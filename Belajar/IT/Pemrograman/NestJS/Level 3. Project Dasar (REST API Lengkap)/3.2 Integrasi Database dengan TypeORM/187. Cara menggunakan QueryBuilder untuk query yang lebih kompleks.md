# 187. Cara menggunakan QueryBuilder untuk query yang lebih kompleks

## Definisi & Target
QueryBuilder memungkinkan pembuatan query SQL yang kompleks secara programatik. Berguna untuk pagination, filter dinamis, join, aggregasi, dan subquery yang tidak bisa dilakukan dengan method repository sederhana.

## Komponen & Nest CLI Command
Tidak ada CLI command.

## Kode Implementasi (Per File)
```typescript
// src/products/products.service.ts
import { Injectable } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Product } from './entities/product.entity';

@Injectable()
export class ProductsService {
  constructor(
    @InjectRepository(Product)
    private productRepository: Repository<Product>,
  ) {}

  async findAllWithQueryBuilder(query: { page?: number; limit?: number; search?: string; minPrice?: number }) {
    const qb = this.productRepository.createQueryBuilder('product');

    qb.where('product.isActive = :isActive', { isActive: true });

    if (query.search) {
      qb.andWhere('product.name ILIKE :search', { search: `%${query.search}%` });
    }

    if (query.minPrice) {
      qb.andWhere('product.price >= :minPrice', { minPrice: query.minPrice });
    }

    const page = query.page || 1;
    const limit = query.limit || 10;

    qb.skip((page - 1) * limit).take(limit);
    qb.orderBy('product.createdAt', 'DESC');

    const [items, total] = await qb.getManyAndCount();

    return {
      data: items,
      meta: { total, page, limit, totalPages: Math.ceil(total / limit) },
    };
  }

  async getProductStats() {
    const stats = await this.productRepository
      .createQueryBuilder('product')
      .select('product.status', 'status')
      .addSelect('COUNT(product.id)', 'count')
      .addSelect('AVG(product.price)', 'averagePrice')
      .groupBy('product.status')
      .getRawMany();
    return { data: stats };
  }
}
```
