# 228. Cara membuat custom configuration file untuk setiap environment

## Definisi & Target
Memisahkan konfigurasi per environment (development, staging, production) menggunakan file terpisah. ConfigModule membaca file berdasarkan nilai `NODE_ENV`.

## Komponen & Nest CLI Command
Tidak ada CLI command.

## Kode Implementasi (Per File)
```typescript
// src/config/configuration.ts
export default () => ({
  port: parseInt(process.env.PORT, 10) || 3000,
  database: {
    host: process.env.DB_HOST || 'localhost',
    port: parseInt(process.env.DB_PORT, 10) || 5432,
    username: process.env.DB_USERNAME || 'postgres',
    password: process.env.DB_PASSWORD || 'postgres',
    database: process.env.DB_DATABASE || 'blog_db',
  },
  jwt: {
    secret: process.env.JWT_SECRET || 'default-secret',
    expiration: process.env.JWT_EXPIRATION || '7d',
  },
  redis: {
    host: process.env.REDIS_HOST || 'localhost',
    port: parseInt(process.env.REDIS_PORT, 10) || 6379,
  },
});
```

```typescript
// src/config/environments/development.ts
export default () => ({
  database: {
    host: 'localhost',
    port: 5432,
    username: 'postgres',
    password: 'postgres',
    database: 'blog_dev',
  },
  jwt: {
    secret: 'dev-secret-key',
    expiration: '30d',
  },
  logger: {
    level: 'debug',
  },
});
```

```typescript
// src/config/environments/production.ts
export default () => ({
  database: {
    host: process.env.DB_HOST,
    port: parseInt(process.env.DB_PORT, 10),
    username: process.env.DB_USERNAME,
    password: process.env.DB_PASSWORD,
    database: process.env.DB_DATABASE,
  },
  jwt: {
    secret: process.env.JWT_SECRET,
    expiration: '7d',
  },
  logger: {
    level: 'info',
  },
});
```

```typescript
// src/app.module.ts
import { Module } from '@nestjs/common';
import { ConfigModule } from '@nestjs/config';
import development from './config/environments/development';
import production from './config/environments/production';

@Module({
  imports: [
    ConfigModule.forRoot({
      isGlobal: true,
      load: [process.env.NODE_ENV === 'production' ? production : development],
    }),
  ],
})
export class AppModule {}
```

## Cara Integrasi & Pengujian
```bash
# Development
NODE_ENV=development npm run start:dev
# Memuat config development.ts (DB: blog_dev, logger level: debug)

# Production
NODE_ENV=production node dist/main
# Memuat config production.ts (DB: dari env, logger level: info)
```
