# 229. Cara mengakses nested configuration menggunakan dot notation

## Definisi & Target
ConfigService.get() mendukung dot notation untuk mengakses nested object configuration. Misalnya `database.host` mengakses `config.database.host`. TypeScript memberikan type inference jika menggunakan interface.

## Komponen & Nest CLI Command
Tidak ada CLI command.

## Kode Implementasi (Per File)
```typescript
// src/config/configuration.ts
export default () => ({
  app: {
    name: 'Blog API',
    version: '1.0.0',
    port: parseInt(process.env.PORT, 10) || 3000,
  },
  database: {
    postgres: {
      host: process.env.DB_HOST || 'localhost',
      port: parseInt(process.env.DB_PORT, 10) || 5432,
      credentials: {
        username: process.env.DB_USERNAME,
        password: process.env.DB_PASSWORD,
      },
    },
    redis: {
      host: process.env.REDIS_HOST || 'localhost',
      port: parseInt(process.env.REDIS_PORT, 10) || 6379,
    },
  },
  jwt: {
    access: {
      secret: process.env.JWT_ACCESS_SECRET,
      expiration: '15m',
    },
    refresh: {
      secret: process.env.JWT_REFRESH_SECRET,
      expiration: '7d',
    },
  },
});
```

```typescript
// src/config/config.interface.ts
export interface AppConfig {
  app: { name: string; version: string; port: number };
  database: {
    postgres: { host: string; port: number; credentials: { username: string; password: string } };
    redis: { host: string; port: number };
  };
  jwt: {
    access: { secret: string; expiration: string };
    refresh: { secret: string; expiration: string };
  };
}
```

```typescript
// src/posts/posts.service.ts
import { Injectable } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class PostsService {
  constructor(
    private configService: ConfigService,
    private prisma: PrismaService,
  ) {
    // Dot notation
    const dbHost = this.configService.get<string>('database.postgres.host');
    const jwtExp = this.configService.get<string>('jwt.access.expiration');
    const appName = this.configService.get<string>('app.name');

    // Type-safe dengan interface
    const dbConfig = this.configService.get('database.postgres');
    console.log(dbConfig.host, dbConfig.port);

    // Default value
    const port = this.configService.get<number>('app.port', 3000);
  }
}
```
