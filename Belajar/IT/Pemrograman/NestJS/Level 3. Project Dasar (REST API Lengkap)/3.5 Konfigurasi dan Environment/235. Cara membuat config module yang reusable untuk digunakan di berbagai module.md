# 235. Cara membuat config module yang reusable untuk digunakan di berbagai module

## Definisi & Target
Membuat SharedConfigModule yang bisa di-import oleh module lain tanpa harus mengkonfigurasi ulang. Module ini mengekspos ConfigService dan konfigurasi khusus yang siap pakai.

## Komponen & Nest CLI Command
```bash
nest g module shared/shared-config --no-spec --flat
```

## Kode Implementasi (Per File)
```typescript
// src/shared/shared-config.module.ts
import { Module, DynamicModule, Provider } from '@nestjs/common';
import { ConfigModule, ConfigService } from '@nestjs/config';
import * as Joi from 'joi';

export interface SharedConfigOptions {
  folder?: string;
  required?: boolean;
}

@Module({})
export class SharedConfigModule {
  static forRoot(options?: SharedConfigOptions): DynamicModule {
    const providers: Provider[] = [
      {
        provide: 'APP_CONFIG_OPTIONS',
        useValue: options || {},
      },
      {
        provide: 'APP_NAME',
        useValue: 'Blog API',
      },
      {
        provide: 'API_PREFIX',
        useFactory: (configService: ConfigService) => {
          return configService.get<string>('API_PREFIX', 'api');
        },
        inject: [ConfigService],
      },
    ];

    return {
      module: SharedConfigModule,
      imports: [
        ConfigModule.forRoot({
          isGlobal: true,
          validationSchema: Joi.object({
            NODE_ENV: Joi.string().valid('development', 'production', 'test').default('development'),
            PORT: Joi.number().default(3000),
            API_PREFIX: Joi.string().default('api'),
          }),
        }),
      ],
      providers,
      exports: [ConfigModule, 'APP_NAME', 'API_PREFIX', 'APP_CONFIG_OPTIONS'],
    };
  }

  static forFeature(config?: Record<string, any>): DynamicModule {
    const providers: Provider[] = Object.entries(config || {}).map(
      ([key, value]) => ({
        provide: `CONFIG_${key.toUpperCase()}`,
        useValue: value,
      }),
    );

    return {
      module: SharedConfigModule,
      providers,
      exports: providers.map((p) => (p as any).provide),
    };
  }
}
```

```typescript
// src/app.module.ts
import { Module } from '@nestjs/common';
import { SharedConfigModule } from './shared/shared-config.module';
import { ProductsModule } from './products/products.module';

@Module({
  imports: [
    SharedConfigModule.forRoot({
      folder: 'config',
      required: true,
    }),
    ProductsModule,
  ],
})
export class AppModule {}
```

```typescript
// src/products/products.module.ts
import { Module, Inject } from '@nestjs/common';
import { SharedConfigModule } from '../shared/shared-config.module';
import { ProductsController } from './products.controller';
import { ProductsService } from './products.service';

@Module({
  imports: [
    SharedConfigModule.forFeature({
      TABLE_NAME: 'products',
      CACHE_TTL: 300,
    }),
  ],
  controllers: [ProductsController],
  providers: [ProductsService],
})
export class ProductsModule {}
```

```typescript
// src/products/products.service.ts
import { Injectable, Inject } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';

@Injectable()
export class ProductsService {
  constructor(
    private configService: ConfigService,
    @Inject('CONFIG_TABLE_NAME') private tableName: string,
    @Inject('CONFIG_CACHE_TTL') private cacheTtl: number,
    @Inject('APP_NAME') private appName: string,
  ) {
    console.log(`Service: ${this.appName}, Table: ${this.tableName}, Cache: ${this.cacheTtl}s`);
  }
}
```

## Cara Integrasi & Pengujian
```bash
# SharedConfigModule bisa dipakai di module manapun
# Cukup import SharedConfigModule.forRoot() sekali di AppModule
# Module lain panggil SharedConfigModule.forFeature() untuk config spesifik
npm run start:dev
# Output: "Service: Blog API, Table: products, Cache: 300s"
```
