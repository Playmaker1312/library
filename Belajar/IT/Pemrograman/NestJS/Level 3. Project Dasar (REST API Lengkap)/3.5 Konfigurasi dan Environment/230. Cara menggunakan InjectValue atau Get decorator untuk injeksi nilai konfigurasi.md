# 230. Cara menggunakan InjectValue atau Get decorator untuk injeksi nilai konfigurasi

## Definisi & Target
Selain ConfigService, NestJS menyediakan decorator `@Inject()` dengan custom token untuk menginjeksi nilai konfigurasi langsung ke constructor. Ini mengurangi boilerplate dan membuat dependency lebih eksplisit.

## Komponen & Nest CLI Command
Tidak ada CLI command.

## Kode Implementasi (Per File)
```typescript
// src/config/database.config.ts
import { registerAs } from '@nestjs/config';

export const DATABASE_CONFIG = 'DATABASE_CONFIG';

export default registerAs(DATABASE_CONFIG, () => ({
  host: process.env.DB_HOST || 'localhost',
  port: parseInt(process.env.DB_PORT || '5432', 10),
  database: process.env.DB_DATABASE || 'blog_db',
}));
```

```typescript
// src/config/config.providers.ts
import { Provider } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';

export const DATABASE_HOST = 'DATABASE_HOST';
export const DATABASE_PORT = 'DATABASE_PORT';
export const JWT_SECRET_TOKEN = 'JWT_SECRET';

export const configProviders: Provider[] = [
  {
    provide: DATABASE_HOST,
    inject: [ConfigService],
    useFactory: (config: ConfigService) => config.get<string>('database.host'),
  },
  {
    provide: DATABASE_PORT,
    inject: [ConfigService],
    useFactory: (config: ConfigService) => config.get<number>('database.port'),
  },
  {
    provide: JWT_SECRET_TOKEN,
    inject: [ConfigService],
    useFactory: (config: ConfigService) => config.get<string>('jwt.secret'),
  },
];
```

```typescript
// src/database/database.module.ts
import { Module, Inject } from '@nestjs/common';
import { ConfigModule } from '@nestjs/config';
import databaseConfig, { DATABASE_CONFIG } from '../config/database.config';
import { configProviders, DATABASE_HOST, DATABASE_PORT } from '../config/config.providers';

class DatabaseConnection {
  constructor(
    @Inject(DATABASE_HOST) private host: string,
    @Inject(DATABASE_PORT) private port: number,
    @Inject(DATABASE_CONFIG) private dbConfig: any,
  ) {
    console.log(`Connecting to ${this.host}:${this.port}`);
    console.log(`Config: ${this.dbConfig.database}`);
  }
}

@Module({
  imports: [ConfigModule.forFeature(databaseConfig)],
  providers: [...configProviders, DatabaseConnection],
})
export class DatabaseModule {}
```
