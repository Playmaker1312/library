# 204. Cara menggunakan Prisma transactions dalam NestJS

## Definisi & Target
Transaction memastikan beberapa operasi database berjalan atomik: jika salah satu gagal, semua perubahan dibatalkan (rollback). Prisma mendukung dua pendekatan: sequential (interactive) dan batch.

## Komponen & Nest CLI Command
Tidak ada CLI command.

## Kode Implementasi (Per File)
```typescript
// src/products/products.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class ProductsService {
  constructor(private prisma: PrismaService) {}

  // Method 1: Interactive transaction
  async transferStock(fromProductId: number, toProductId: number, amount: number) {
    return await this.prisma.$transaction(async (tx) => {
      const fromProduct = await tx.product.findUnique({
        where: { id: fromProductId },
      });

      if (!fromProduct || fromProduct.stock < amount) {
        throw new Error('Insufficient stock');
      }

      await tx.product.update({
        where: { id: fromProductId },
        data: { stock: { decrement: amount } },
      });

      await tx.product.update({
        where: { id: toProductId },
        data: { stock: { increment: amount } },
      });

      return { message: 'Stock transferred successfully' };
    });
  }

  // Method 2: Batch transaction (optimistic)
  async createManyProducts(data: Array<{ name: string; price: number; stock: number }>) {
    return await this.prisma.$transaction(
      data.map((item) =>
        this.prisma.product.create({ data: item }),
      ),
    );
  }

  // Method 3: With isolation level
  async updateWithIsolation(productId: number, data: any) {
    return await this.prisma.$transaction(
      async (tx) => {
        return tx.product.update({
          where: { id: productId },
          data,
        });
      },
      {
        isolationLevel: 'Serializable',
        timeout: 10000,
      },
    );
  }
}
```

```typescript
// src/products/products.controller.ts
import { Controller, Post, Param, Body } from '@nestjs/common';
import { ProductsService } from './products.service';

@Controller('products')
export class ProductsController {
  constructor(private readonly productsService: ProductsService) {}

  @Post('transfer/:from/:to')
  transferStock(
    @Param('from') from: string,
    @Param('to') to: string,
    @Body('amount') amount: number,
  ) {
    return this.productsService.transferStock(+from, +to, amount);
  }
}
```
