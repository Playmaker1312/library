# 203. Cara menggunakan Prisma relations yaitu include dan select

## Definisi & Target
`include` digunakan untuk memuat relasi (eager loading), sementara `select` digunakan untuk memilih field tertentu saja. Prisma membuat kedua operasi ini type-safe.

## Komponen & Nest CLI Command
Tidak ada CLI command.

## Kode Implementasi (Per File)
```prisma
// prisma/schema.prisma
model User {
  id       Int       @id @default(autoincrement())
  email    String    @unique
  name     String
  posts    Post[]
  @@map("users")
}

model Post {
  id       Int     @id @default(autoincrement())
  title    String
  content  String?
  authorId Int
  author   User    @relation(fields: [authorId], references: [id])
  @@map("posts")
}
```

```typescript
// src/posts/posts.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class PostsService {
  constructor(private prisma: PrismaService) {}

  // include — memuat relasi
  async findAllWithAuthor() {
    return await this.prisma.post.findMany({
      include: {
        author: {
          select: { id: true, name: true, email: true },
        },
      },
    });
  }

  // select — memilih field spesifik
  async findAllTitles() {
    return await this.prisma.post.findMany({
      select: {
        id: true,
        title: true,
        author: {
          select: { name: true },
        },
      },
    });
  }

  // Nested create — buat post + user sekaligus
  async createPostWithUser(data: { title: string; authorEmail: string }) {
    return await this.prisma.post.create({
      data: {
        title: data.title,
        author: {
          connect: { email: data.authorEmail },
        },
      },
      include: { author: true },
    });
  }
}
```
