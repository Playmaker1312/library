# 213. Membangun CommentModule dengan nested resource routing

## Definisi & Target
Comment menggunakan nested routing di bawah Post: `POST /posts/:postId/comments` dan `GET /posts/:postId/comments`. Ini menunjukkan konsep resource routing bersarang di NestJS.

## Komponen & Nest CLI Command
```bash
nest g module comments
nest g service comments --no-spec
nest g controller comments --no-spec
```

## Kode Implementasi (Per File)
```typescript
// src/comments/dto/create-comment.dto.ts
import { IsString, IsInt } from 'class-validator';

export class CreateCommentDto {
  @IsString()
  content: string;

  @IsInt()
  authorId: number;
}
```

```typescript
// src/comments/comments.service.ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreateCommentDto } from './dto/create-comment.dto';

@Injectable()
export class CommentsService {
  constructor(private prisma: PrismaService) {}

  async findByPost(postId: number) {
    const comments = await this.prisma.comment.findMany({
      where: { postId },
      include: { author: { select: { id: true, name: true } } },
      orderBy: { createdAt: 'desc' },
    });
    return { data: comments };
  }

  async create(postId: number, dto: CreateCommentDto) {
    const post = await this.prisma.post.findUnique({ where: { id: postId } });
    if (!post) throw new NotFoundException(`Post #${postId} not found`);

    const comment = await this.prisma.comment.create({
      data: {
        content: dto.content,
        postId,
        authorId: dto.authorId,
      },
      include: { author: { select: { id: true, name: true } } },
    });
    return { data: comment, message: 'Comment added' };
  }
}
```

```typescript
// src/comments/comments.controller.ts
import { Controller, Get, Post, Param, Body } from '@nestjs/common';
import { CommentsService } from './comments.service';
import { CreateCommentDto } from './dto/create-comment.dto';

@Controller('posts/:postId/comments')
export class CommentsController {
  constructor(private readonly commentsService: CommentsService) {}

  @Get()
  findByPost(@Param('postId') postId: string) {
    return this.commentsService.findByPost(+postId);
  }

  @Post()
  create(@Param('postId') postId: string, @Body() dto: CreateCommentDto) {
    return this.commentsService.create(+postId, dto);
  }
}
```

## Cara Integrasi & Pengujian
```bash
# Tambah komentar
curl -X POST http://localhost:3000/posts/1/comments \
  -H "Content-Type: application/json" \
  -d '{"content":"Nice post!","authorId":1}'
# Response: 201 Created

# Lihat komentar post
curl http://localhost:3000/posts/1/comments
# Response: { "data": [{ "id": 1, "content": "Nice post!", "author": { ... } }] }
```
