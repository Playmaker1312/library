# 215. Mengimplementasikan soft delete menggunakan field deletedAt pada semua model

## Definisi & Target
Soft delete menandai record sebagai terhapus tanpa menghapus data dari database. Field `deletedAt` diisi timestamp saat delete. Semua query default mengecualikan record yang sudah di-soft-delete.

## Komponen & Nest CLI Command
Tidak ada CLI command.

## Kode Implementasi (Per File)
```typescript
// src/posts/posts.service.ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { UpdatePostDto } from './dto/update-post.dto';

@Injectable()
export class PostsService {
  constructor(private prisma: PrismaService) {}

  // Soft delete — set deletedAt
  async remove(id: number) {
    try {
      await this.prisma.post.update({
        where: { id },
        data: { deletedAt: new Date() },
      });
      return { message: 'Post deleted (soft)' };
    } catch {
      throw new NotFoundException(`Post #${id} not found`);
    }
  }

  // Restore — hapus deletedAt
  async restore(id: number) {
    try {
      await this.prisma.post.update({
        where: { id },
        data: { deletedAt: null },
      });
      return { message: 'Post restored' };
    } catch {
      throw new NotFoundException(`Post #${id} not found`);
    }
  }

  // Hard delete — hapus permanen
  async hardRemove(id: number) {
    try {
      await this.prisma.post.delete({ where: { id } });
      return { message: 'Post permanently deleted' };
    } catch {
      throw new NotFoundException(`Post #${id} not found`);
    }
  }
```

```typescript
// src/posts/posts.controller.ts
import { Controller, Delete, Post, Param } from '@nestjs/common';
import { PostsService } from './posts.service';

@Controller('posts')
export class PostsController {
  constructor(private readonly postsService: PostsService) {}

  @Delete(':id')
  remove(@Param('id') id: string) { return this.postsService.remove(+id); }

  @Post(':id/restore')
  restore(@Param('id') id: string) { return this.postsService.restore(+id); }
}
```
