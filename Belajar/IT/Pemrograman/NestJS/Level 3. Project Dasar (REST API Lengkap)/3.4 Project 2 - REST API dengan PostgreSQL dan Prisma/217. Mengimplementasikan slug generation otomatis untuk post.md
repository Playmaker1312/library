# 217. Mengimplementasikan slug generation otomatis untuk post

## Definisi & Target
Slug adalah versi URL-friendly dari judul post (contoh: "Getting Started with NestJS" → "getting-started-with-nestjs"). Slug dibuat otomatis saat create dan digunakan untuk routing alternatif.

## Komponen & Nest CLI Command
Tidak ada CLI command.

## Kode Implementasi (Per File)
```typescript
// src/posts/helpers/slug.helper.ts
export function generateSlug(title: string): string {
  return title
    .toLowerCase()
    .replace(/[^\w\s-]/g, '')
    .replace(/\s+/g, '-')
    .replace(/-+/g, '-')
    .trim();
}

export function generateUniqueSlug(title: string, existingSlugs: string[]): string {
  let slug = generateSlug(title);
  let counter = 1;
  while (existingSlugs.includes(slug)) {
    slug = `${generateSlug(title)}-${counter}`;
    counter++;
  }
  return slug;
}
```

```typescript
// src/posts/posts.service.ts
import { Injectable, ConflictException, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreatePostDto } from './dto/create-post.dto';
import { UpdatePostDto } from './dto/update-post.dto';
import { generateSlug, generateUniqueSlug } from './helpers/slug.helper';

@Injectable()
export class PostsService {
  constructor(private prisma: PrismaService) {}

  async create(dto: CreatePostDto) {
    const existingSlugs = (await this.prisma.post.findMany({ select: { slug: true } }))
      .map((p) => p.slug);

    const slug = generateUniqueSlug(dto.title, existingSlugs);

    const post = await this.prisma.post.create({
      data: {
        title: dto.title,
        slug,
        content: dto.content,
        status: dto.status || 'DRAFT',
        authorId: dto.authorId,
      },
    });
    return { data: post, message: 'Post created' };
  }

  async findBySlug(slug: string) {
    const post = await this.prisma.post.findFirst({
      where: { slug, deletedAt: null },
      include: {
        author: { select: { id: true, name: true } },
        tags: { include: { tag: true } },
      },
    });
    if (!post) throw new NotFoundException(`Post with slug "${slug}" not found`);
    return { data: post };
  }
}
```

```typescript
// src/posts/posts.controller.ts
import { Controller, Get, Param } from '@nestjs/common';
import { PostsService } from './posts.service';

@Controller('posts')
export class PostsController {
  constructor(private readonly postsService: PostsService) {}

  @Get('slug/:slug')
  findBySlug(@Param('slug') slug: string) {
    return this.postsService.findBySlug(slug);
  }
}
```
