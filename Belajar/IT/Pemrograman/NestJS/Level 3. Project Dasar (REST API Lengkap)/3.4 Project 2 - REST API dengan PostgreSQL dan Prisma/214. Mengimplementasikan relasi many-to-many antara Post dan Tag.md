# 214. Mengimplementasikan relasi many-to-many antara Post dan Tag

## Definisi & Target
Post dan Tag memiliki relasi many-to-many melalui tabel pivot `PostOnTags`. Endpoint memungkinkan menambahkan tag ke post saat create atau update, dan filter post berdasarkan tag.

## Komponen & Nest CLI Command
Tidak ada CLI command. Modifikasi service dan DTO yang sudah ada.

## Kode Implementasi (Per File)
```typescript
// src/posts/dto/create-post.dto.ts
import { IsString, IsInt, IsOptional, IsArray } from 'class-validator';

export class CreatePostDto {
  @IsString()
  title: string;

  @IsString()
  content: string;

  @IsOptional()
  @IsString()
  status?: string;

  @IsInt()
  authorId: number;

  @IsOptional()
  @IsArray()
  @IsInt({ each: true })
  tagIds?: number[];
}
```

```typescript
// src/posts/posts.service.ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreatePostDto } from './dto/create-post.dto';
import { UpdatePostDto } from './dto/update-post.dto';

@Injectable()
export class PostsService {
  constructor(private prisma: PrismaService) {}

  async create(dto: CreatePostDto) {
    const post = await this.prisma.post.create({
      data: {
        title: dto.title,
        slug: dto.title.toLowerCase().replace(/\s+/g, '-'),
        content: dto.content,
        status: dto.status || 'DRAFT',
        authorId: dto.authorId,
        tags: dto.tagIds
          ? { create: dto.tagIds.map((tagId) => ({ tagId })) }
          : undefined,
      },
      include: { tags: { include: { tag: true } } },
    });
    return { data: post, message: 'Post created' };
  }

  async update(id: number, dto: UpdatePostDto) {
    if (dto.tagIds) {
      await this.prisma.postOnTags.deleteMany({ where: { postId: id } });
    }

    const post = await this.prisma.post.update({
      where: { id },
      data: {
        title: dto.title,
        content: dto.content,
        status: dto.status,
        tags: dto.tagIds
          ? { create: dto.tagIds.map((tagId) => ({ tagId })) }
          : undefined,
      },
      include: { tags: { include: { tag: true } } },
    });
    return { data: post, message: 'Post updated' };
  }
}
```

## Cara Integrasi & Pengujian
```bash
# Buat post dengan tag
curl -X POST http://localhost:3000/posts \
  -H "Content-Type: application/json" \
  -d '{"title":"NestJS Guide","content":"...","authorId":1,"tagIds":[1,2]}'
# Response: post dengan tags: [{ tag: { name: "Technology" } }, { tag: { name: "Programming" } }]

# Filter posts by tag
curl "http://localhost:3000/posts?tag=Technology"
```
