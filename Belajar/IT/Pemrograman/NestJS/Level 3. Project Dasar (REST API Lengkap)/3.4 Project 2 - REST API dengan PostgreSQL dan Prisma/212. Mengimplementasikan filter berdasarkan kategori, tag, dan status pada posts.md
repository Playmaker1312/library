# 212. Mengimplementasikan filter berdasarkan kategori, tag, dan status pada posts

## Definisi & Target
Endpoint `GET /posts` mendukung multi-filter: berdasarkan status (DRAFT/PUBLISHED/ARCHIVED), tag, dan pencarian teks pada judul/konten. Filter digabung menggunakan Prisma `AND`/`OR`.

## Komponen & Nest CLI Command
Tidak ada CLI command.

## Kode Implementasi (Per File)
```typescript
// src/posts/posts.service.ts
import { Injectable } from '@nestjs/common';
import { Prisma } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class PostsService {
  constructor(private prisma: PrismaService) {}

  async findAll(query: {
    page: number;
    limit: number;
    status?: string;
    tag?: string;
    search?: string;
  }) {
    const where: Prisma.PostWhereInput = { deletedAt: null };

    if (query.status) {
      where.status = query.status as any;
    }

    if (query.tag) {
      where.tags = {
        some: {
          tag: { name: { equals: query.tag, mode: 'insensitive' } },
        },
      };
    }

    if (query.search) {
      where.OR = [
        { title: { contains: query.search, mode: 'insensitive' } },
        { content: { contains: query.search, mode: 'insensitive' } },
      ];
    }

    const skip = (query.page - 1) * query.limit;

    const [items, total] = await Promise.all([
      this.prisma.post.findMany({
        where,
        skip,
        take: query.limit,
        include: {
          author: { select: { id: true, name: true } },
          tags: { include: { tag: { select: { id: true, name: true } } } },
        },
        orderBy: { createdAt: 'desc' },
      }),
      this.prisma.post.count({ where }),
    ]);

    return {
      data: items,
      meta: { total, page: query.page, limit: query.limit, totalPages: Math.ceil(total / query.limit) },
    };
  }
}
```

```typescript
// src/posts/posts.controller.ts
import { Controller, Get, Query } from '@nestjs/common';
import { PostsService } from './posts.service';

@Controller('posts')
export class PostsController {
  constructor(private readonly postsService: PostsService) {}

  @Get()
  findAll(
    @Query('page') page?: string,
    @Query('limit') limit?: string,
    @Query('status') status?: string,
    @Query('tag') tag?: string,
    @Query('search') search?: string,
  ) {
    return this.postsService.findAll({
      page: page ? parseInt(page) : 1,
      limit: limit ? parseInt(limit) : 10,
      status,
      tag,
      search,
    });
  }
}
```

## Cara Integrasi & Pengujian
```bash
# Filter by status
curl "http://localhost:3000/posts?status=PUBLISHED"

# Filter by tag
curl "http://localhost:3000/posts?tag=Technology"

# Filter by search
curl "http://localhost:3000/posts?search=nestjs"

# Kombinasi
curl "http://localhost:3000/posts?status=PUBLISHED&tag=Programming&page=1&limit=10"
```
