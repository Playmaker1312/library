# 220. Mendokumentasikan semua endpoint menggunakan Swagger

## Definisi & Target
Swagger/OpenAPI menyediakan dokumentasi API interaktif. NestJS menyediakan decorator `@ApiTags`, `@ApiOperation`, `@ApiResponse` untuk mendokumentasikan endpoint secara otomatis.

## Komponen & Nest CLI Command
```bash
npm install @nestjs/swagger
```

File yang dimodifikasi:
- `src/main.ts`
- Controller dan DTO

## Kode Implementasi (Per File)
```typescript
// src/main.ts
import { NestFactory } from '@nestjs/core';
import { SwaggerModule, DocumentBuilder } from '@nestjs/swagger';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  const config = new DocumentBuilder()
    .setTitle('Blog API')
    .setDescription('REST API for blog application with User, Post, Comment, and Tag')
    .setVersion('1.0')
    .addTag('posts', 'Blog post management')
    .addTag('users', 'User management')
    .addTag('comments', 'Post comments')
    .addTag('tags', 'Post tags')
    .build();

  const document = SwaggerModule.createDocument(app, config);
  SwaggerModule.setup('api/docs', app, document);

  await app.listen(3000);
}
bootstrap();
```

```typescript
// src/posts/posts.controller.ts
import {
  Controller, Get, Post, Put, Delete, Param, Body, Query,
} from '@nestjs/common';
import { ApiTags, ApiOperation, ApiResponse, ApiQuery, ApiParam } from '@nestjs/swagger';
import { PostsService } from './posts.service';
import { CreatePostDto } from './dto/create-post.dto';
import { UpdatePostDto } from './dto/update-post.dto';

@ApiTags('posts')
@Controller('posts')
export class PostsController {
  constructor(private readonly postsService: PostsService) {}

  @Get()
  @ApiOperation({ summary: 'Get all posts', description: 'Returns paginated posts' })
  @ApiQuery({ name: 'page', required: false })
  @ApiQuery({ name: 'limit', required: false })
  @ApiQuery({ name: 'status', required: false, enum: ['DRAFT', 'PUBLISHED', 'ARCHIVED'] })
  @ApiResponse({ status: 200, description: 'List of posts' })
  findAll(@Query('page') page?: string, @Query('limit') limit?: string) {
    return this.postsService.findAll({ page: +page || 1, limit: +limit || 10 });
  }

  @Post()
  @ApiOperation({ summary: 'Create a new post' })
  @ApiResponse({ status: 201, description: 'Post created' })
  @ApiResponse({ status: 400, description: 'Validation error' })
  create(@Body() dto: CreatePostDto) {
    return this.postsService.create(dto);
  }

  @Get(':id')
  @ApiOperation({ summary: 'Get post by ID' })
  @ApiParam({ name: 'id', type: Number })
  @ApiResponse({ status: 200, description: 'Post found' })
  @ApiResponse({ status: 404, description: 'Post not found' })
  findOne(@Param('id') id: string) {
    return this.postsService.findOne(+id);
  }

  @Delete(':id')
  @ApiOperation({ summary: 'Delete a post (soft delete)' })
  @ApiResponse({ status: 200, description: 'Post deleted' })
  remove(@Param('id') id: string) {
    return this.postsService.remove(+id);
  }
}
```
