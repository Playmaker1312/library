# 210. Membangun PostModule dengan endpoint create, read, update, dan delete

## Definisi & Target
PostModule menangani CRUD untuk artikel blog. Service menggunakan Prisma untuk operasi database. Fitur slug dan soft delete akan ditambahkan di poin terpisah.

## Komponen & Nest CLI Command
```bash
nest g module posts
nest g service posts --no-spec
nest g controller posts --no-spec
```

## Kode Implementasi (Per File)

// src/posts/dto/create-post.dto.ts
import { IsString, IsInt, IsOptional, IsEnum } from 'class-validator';

export enum PostStatusDto {
  DRAFT = 'DRAFT',
  PUBLISHED = 'PUBLISHED',
  ARCHIVED = 'ARCHIVED',
}

export class CreatePostDto {
  @IsString()
  title: string;

  @IsString()
  content: string;

  @IsOptional()
  @IsEnum(PostStatusDto)
  status?: PostStatusDto;

  @IsInt()
  authorId: number;
}

```typescript
// src/posts/posts.service.ts
import { Injectable, NotFoundException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreatePostDto } from './dto/create-post.dto';
import { UpdatePostDto } from './dto/update-post.dto';

@Injectable()
export class PostsService {
  constructor(private prisma: PrismaService) {}

  async findAll() {
    const posts = await this.prisma.post.findMany({
      where: { deletedAt: null },
      include: { author: { select: { id: true, name: true } } },
      orderBy: { createdAt: 'desc' },
    });
    return { data: posts };
  }

  async findOne(id: number) {
    const post = await this.prisma.post.findFirst({
      where: { id, deletedAt: null },
      include: {
        author: { select: { id: true, name: true } },
        comments: { include: { author: { select: { id: true, name: true } } } },
        tags: { include: { tag: true } },
      },
    });
    if (!post) throw new NotFoundException(`Post #${id} not found`);
    return { data: post };
  }

  async create(dto: CreatePostDto) {
    const post = await this.prisma.post.create({
      data: {
        title: dto.title,
        slug: dto.title.toLowerCase().replace(/\s+/g, '-'),
        content: dto.content,
        status: dto.status || 'DRAFT',
        authorId: dto.authorId,
      },
    });
    return { data: post, message: 'Post created' };
  }

  async update(id: number, dto: UpdatePostDto) {
    try {
      const post = await this.prisma.post.update({ where: { id }, data: dto });
      return { data: post, message: 'Post updated' };
    } catch {
      throw new NotFoundException(`Post #${id} not found`);
    }
  }

  async remove(id: number) {
    try {
      await this.prisma.post.update({ where: { id }, data: { deletedAt: new Date() } });
      return { message: 'Post deleted' };
    } catch {
      throw new NotFoundException(`Post #${id} not found`);
    }
  }
}
```
