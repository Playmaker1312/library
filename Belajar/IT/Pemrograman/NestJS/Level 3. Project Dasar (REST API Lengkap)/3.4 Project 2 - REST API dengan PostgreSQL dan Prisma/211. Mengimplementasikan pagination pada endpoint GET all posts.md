# 211. Mengimplementasikan pagination pada endpoint GET all posts

## Definisi & Target
Endpoint `GET /posts` mendukung pagination menggunakan parameter `page` dan `limit`. Prisma `skip` dan `take` digunakan untuk implementasi pagination yang efisien di level database.

## Komponen & Nest CLI Command
Tidak ada CLI command. Modifikasi controller dan service yang sudah ada.

## Kode Implementasi (Per File)
```typescript
// src/posts/posts.controller.ts
import { Controller, Get, Query } from '@nestjs/common';
import { PostsService } from './posts.service';

@Controller('posts')
export class PostsController {
  constructor(private readonly postsService: PostsService) {}

  @Get()
  findAll(
    @Query('page') page?: string,
    @Query('limit') limit?: string,
    @Query('status') status?: string,
  ) {
    return this.postsService.findAll({
      page: page ? parseInt(page) : 1,
      limit: limit ? parseInt(limit) : 10,
      status,
    });
  }
}
```

```typescript
// src/posts/posts.service.ts
import { Injectable } from '@nestjs/common';
import { Prisma, PostStatus } from '@prisma/client';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class PostsService {
  constructor(private prisma: PrismaService) {}

  async findAll(query: { page: number; limit: number; status?: string }) {
    const where: Prisma.PostWhereInput = { deletedAt: null };

    if (query.status) {
      where.status = query.status as PostStatus;
    }

    const skip = (query.page - 1) * query.limit;

    const [items, total] = await Promise.all([
      this.prisma.post.findMany({
        where,
        skip,
        take: query.limit,
        include: { author: { select: { id: true, name: true } } },
        orderBy: { createdAt: 'desc' },
      }),
      this.prisma.post.count({ where }),
    ]);

    return {
      data: items,
      meta: {
        total,
        page: query.page,
        limit: query.limit,
        totalPages: Math.ceil(total / query.limit),
      },
    };
  }
}
```
