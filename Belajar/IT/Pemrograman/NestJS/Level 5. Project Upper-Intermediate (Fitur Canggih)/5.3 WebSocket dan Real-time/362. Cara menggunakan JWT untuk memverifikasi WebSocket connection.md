# 362. Cara menggunakan JWT untuk memverifikasi WebSocket connection

## Definisi & Target
Mengintegrasikan JWT (JSON Web Token) untuk memverifikasi identitas user saat koneksi WebSocket, menggunakan token yang sama dengan REST API.

## Komponen
File yang dimodifikasi: `src/chat/chat.gateway.ts`

## Kode Implementasi

```typescript
// src/chat/chat.gateway.ts (Gateway with JWT)
import { WebSocketGateway, WebSocketServer, OnGatewayConnection, OnGatewayDisconnect } from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import * as jwt from 'jsonwebtoken';

interface JwtPayload {
  sub: string;
  email: string;
  role: string;
}

@WebSocketGateway({ cors: { origin: '*' } })
export class ChatGateway implements OnGatewayConnection, OnGatewayDisconnect {
  @WebSocketServer()
  server: Server;

  async handleConnection(client: Socket) {
    try {
      const token = this.extractToken(client);
      const payload = jwt.verify(token, process.env.JWT_SECRET) as JwtPayload;
      client.data.user = payload;
      client.join(`user:${payload.sub}`);
      console.log(`User ${payload.email} connected via WebSocket`);
    } catch (error) {
      client.emit('error', { message: 'Invalid or expired token' });
      client.disconnect();
    }
  }

  handleDisconnect(client: Socket) {
    if (client.data.user) {
      console.log(`User ${client.data.user.email} disconnected`);
    }
  }

  private extractToken(client: Socket): string {
    const auth = client.handshake.auth?.token || client.handshake.query?.token;
    if (!auth) throw new Error('No token provided');
    return auth.replace('Bearer ', '');
  }
}
```

```typescript
// src/chat/chat.module.ts (Module)
import { Module } from '@nestjs/common';
import { JwtModule } from '@nestjs/jwt';
import { ChatGateway } from './chat.gateway';

@Module({
  imports: [
    JwtModule.register({
      secret: process.env.JWT_SECRET || 'super-secret',
      signOptions: { expiresIn: '1d' },
    }),
  ],
  providers: [ChatGateway],
})
export class ChatModule {}
```

## Cara Integrasi & Pengujian
```javascript
// Dapatkan JWT dari REST API
const loginRes = await fetch('http://localhost:3000/auth/login', {
  method: 'POST',
  headers: { 'Content-Type': 'application/json' },
  body: JSON.stringify({ email: 'user@test.com', password: '123456' }),
});
const { access_token } = await loginRes.json();

// Gunakan token untuk WebSocket
const socket = io('http://localhost:3000', {
  auth: { token: `Bearer ${access_token}` },
});

socket.on('connect', () => console.log('Authenticated!'));
socket.on('error', (err) => console.error('Auth error:', err));
```
