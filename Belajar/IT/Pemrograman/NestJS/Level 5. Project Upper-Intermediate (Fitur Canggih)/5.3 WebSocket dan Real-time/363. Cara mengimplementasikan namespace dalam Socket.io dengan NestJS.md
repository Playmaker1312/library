# 363. Cara mengimplementasikan namespace dalam Socket.io dengan NestJS

## Definisi & Target
Namespace adalah channel komunikasi terpisah dalam Socket.io yang memungkinkan segmentasi koneksi (misal: `/chat`, `/admin`, `/notifications`) dengan context berbeda.

## Komponen
File yang dimodifikasi: `src/chat/chat.gateway.ts`

## Kode Implementasi

```typescript
// src/chat/chat.gateway.ts (Gateway with namespaces)
import { WebSocketGateway, WebSocketServer, SubscribeMessage } from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';

@WebSocketGateway({ cors: { origin: '*' } })
export class ChatGateway {
  @WebSocketServer()
  server: Server;

  afterInit() {
    const chatNsp = this.server.of('/chat');
    const adminNsp = this.server.of('/admin');
    const notifNsp = this.server.of('/notifications');

    chatNsp.on('connection', (socket) => {
      console.log(`Chat namespace: ${socket.id} connected`);
      socket.emit('welcome', { namespace: 'chat' });
    });

    adminNsp.on('connection', (socket) => {
      console.log(`Admin namespace: ${socket.id} connected`);
      socket.emit('welcome', { namespace: 'admin' });
    });

    notifNsp.on('connection', (socket) => {
      console.log(`Notification namespace: ${socket.id} connected`);
      socket.emit('welcome', { namespace: 'notifications' });
    });
  }
}
```

```typescript
// src/chat/admin.gateway.ts (Separate gateway per namespace)
import { WebSocketGateway, WebSocketServer, SubscribeMessage } from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';

@WebSocketGateway({
  namespace: '/admin',
  cors: { origin: '*' },
})
export class AdminGateway {
  @WebSocketServer()
  server: Server;

  handleConnection(client: Socket) {
    if (client.handshake.query.role !== 'admin') {
      client.disconnect();
      return;
    }
    console.log(`Admin connected: ${client.id}`);
  }

  @SubscribeMessage('admin-broadcast')
  handleAdminBroadcast(client: Socket, payload: { message: string }) {
    this.server.emit('admin-message', {
      from: client.id,
      message: payload.message,
    });
  }
}
```

```typescript
// src/chat/chat.module.ts (Module)
import { Module } from '@nestjs/common';
import { ChatGateway } from './chat.gateway';
import { AdminGateway } from './admin.gateway';

@Module({
  providers: [ChatGateway, AdminGateway],
})
export class ChatModule {}
```

## Cara Integrasi & Pengujian
```javascript
// Default namespace
const chatSocket = io('http://localhost:3000/chat');

// Admin namespace
const adminSocket = io('http://localhost:3000/admin', {
  query: { role: 'admin' }
});

// Notification namespace
const notifSocket = io('http://localhost:3000/notifications');

chatSocket.on('welcome', (data) => console.log('Chat:', data));
adminSocket.on('welcome', (data) => console.log('Admin:', data));
notifSocket.on('welcome', (data) => console.log('Notif:', data));
```
