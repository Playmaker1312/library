# 359. Cara broadcast pesan ke semua client menggunakan server.emit

## Definisi & Target
Mengirim pesan ke semua client yang terhubung menggunakan `server.emit()`, berguna untuk announcement, live scores, atau system-wide notifications.

## Komponen
File yang dimodifikasi: `src/chat/chat.gateway.ts`

## Kode Implementasi

```typescript
// src/chat/chat.gateway.ts (Gateway)
import { WebSocketGateway, WebSocketServer, SubscribeMessage } from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';

@WebSocketGateway({ cors: { origin: '*' } })
export class ChatGateway {
  @WebSocketServer()
  server: Server;

  @SubscribeMessage('broadcast-message')
  handleBroadcast(client: Socket, payload: { message: string }) {
    this.server.emit('broadcast', {
      from: client.id,
      message: payload.message,
      timestamp: new Date().toISOString(),
    });
    return { broadcasted: true, recipients: this.server.engine.clientsCount };
  }

  @SubscribeMessage('system-announcement')
  handleAnnouncement(client: Socket, payload: { title: string; body: string }) {
    if (!this.isAdmin(client)) {
      return { error: 'Unauthorized' };
    }
    this.server.emit('announcement', {
      title: payload.title,
      body: payload.body,
      timestamp: Date.now(),
    });
    return { sent: true };
  }

  private isAdmin(client: Socket): boolean {
    return client.handshake.query.role === 'admin';
  }
}
```

```typescript
// src/chat/chat.service.ts (Service - broadcast from service)
import { Injectable } from '@nestjs/common';
import { Server } from 'socket.io';

@Injectable()
export class ChatService {
  private server: Server;

  setServer(server: Server) {
    this.server = server;
  }

  broadcastSystemUpdate(type: string, data: any) {
    if (this.server) {
      this.server.emit('system-update', { type, data, timestamp: Date.now() });
    }
  }

  broadcastToAll(event: string, data: any) {
    if (this.server) {
      this.server.emit(event, data);
    }
  }
}
```

## Cara Integrasi & Pengujian
```javascript
// Semua client akan menerima:
socket.on('broadcast', (data) => {
  console.log(`${data.from}: ${data.message} at ${data.timestamp}`);
});

socket.on('announcement', (data) => {
  alert(`${data.title}: ${data.body}`);
});

socket.on('system-update', (data) => {
  console.log('System update:', data);
});

// Kirim broadcast
socket.emit('broadcast-message', { message: 'Hello everyone!' });
```
