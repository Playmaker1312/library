# 365. Cara mengintegrasikan WebSocket dengan sistem notifikasi real-time

## Definisi & Target
Mengintegrasikan WebSocket gateway dengan service notifikasi untuk mengirim notifikasi real-time ke client berdasarkan event dari sistem (order baru, message, dll).

## Komponen & Nest CLI Command
```bash
nest g mo notifications
nest g ga notifications/notification
nest g s notifications
```

## Kode Implementasi

```typescript
// src/notifications/notification.gateway.ts (Gateway)
import { WebSocketGateway, WebSocketServer, SubscribeMessage, ConnectedSocket } from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';

@WebSocketGateway({ cors: { origin: '*' } })
export class NotificationGateway {
  @WebSocketServer()
  server: Server;

  handleConnection(client: Socket) {
    const userId = client.handshake.query.userId as string;
    if (userId) {
      client.join(`user:${userId}`);
      console.log(`User ${userId} joined notification room`);
    }
  }

  sendNotification(userId: string, notification: { title: string; message: string; type: string }) {
    this.server.to(`user:${userId}`).emit('notification', {
      ...notification,
      timestamp: Date.now(),
      id: `${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
    });
  }

  sendBulkNotification(userIds: string[], notification: any) {
    for (const userId of userIds) {
      this.sendNotification(userId, notification);
    }
  }
}
```

```typescript
// src/notifications/notification.service.ts (Service)
import { Injectable } from '@nestjs/common';
import { NotificationGateway } from './notification.gateway';

@Injectable()
export class NotificationService {
  constructor(private notificationGateway: NotificationGateway) {}

  async orderConfirmed(userId: string, orderId: string) {
    this.notificationGateway.sendNotification(userId, {
      title: 'Order Confirmed',
      message: `Your order #${orderId} has been confirmed`,
      type: 'order',
    });
  }

  async orderShipped(userId: string, orderId: string, trackingNumber: string) {
    this.notificationGateway.sendNotification(userId, {
      title: 'Order Shipped',
      message: `Your order #${orderId} is on the way. Tracking: ${trackingNumber}`,
      type: 'shipping',
    });
  }

  async newMessage(userId: string, from: string, preview: string) {
    this.notificationGateway.sendNotification(userId, {
      title: `New message from ${from}`,
      message: preview,
      type: 'message',
    });
  }
}
```

```typescript
// src/orders/orders.service.ts (Service - integrate with notification)
import { Injectable } from '@nestjs/common';
import { NotificationService } from '../notifications/notification.service';

@Injectable()
export class OrdersService {
  constructor(private notificationService: NotificationService) {}

  async confirmOrder(orderId: string, userId: string) {
    await this.notificationService.orderConfirmed(userId, orderId);
    return { confirmed: true };
  }
}
```
