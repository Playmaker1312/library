# 395. Cara mengimplementasikan audit log untuk setiap perubahan data

## Definisi & Target
Audit log mencatat setiap perubahan data (create, update, delete) beserta metadata seperti siapa, kapan, IP address, dan data sebelum/sesudah.

## Komponen & Nest CLI Command
```bash
nest g mo audit
nest g s audit
```

## Kode Implementasi

```typescript
// src/audit/audit.service.ts (Service)
import { Injectable, Logger } from '@nestjs/common';

interface AuditEntry {
  id: string;
  action: 'CREATE' | 'UPDATE' | 'DELETE' | 'PUBLISH' | 'ARCHIVE' | 'LOGIN' | 'EXPORT';
  entity: string;
  entityId: string;
  userId: string;
  userName: string;
  tenantId: string;
  changes: {
    before?: Record<string, any>;
    after?: Record<string, any>;
    changedFields?: string[];
  };
  metadata: {
    ipAddress: string;
    userAgent: string;
    requestId: string;
    timestamp: string;
  };
}

@Injectable()
export class AuditService {
  private readonly logger = new Logger(AuditService.name);
  private auditLog: AuditEntry[] = [];

  async log(action: AuditEntry['action'], data: {
    entity: string;
    entityId: string;
    userId: string;
    userName: string;
    tenantId: string;
    before?: any;
    after?: any;
    ipAddress?: string;
    userAgent?: string;
    requestId?: string;
  }): Promise<AuditEntry> {
    const changedFields = this.getChangedFields(data.before, data.after);

    const entry: AuditEntry = {
      id: `audit-${Date.now()}-${Math.random().toString(36).substr(2, 9)}`,
      action,
      entity: data.entity,
      entityId: data.entityId,
      userId: data.userId,
      userName: data.userName,
      tenantId: data.tenantId,
      changes: {
        before: data.before,
        after: data.after,
        changedFields,
      },
      metadata: {
        ipAddress: data.ipAddress || 'unknown',
        userAgent: data.userAgent || 'unknown',
        requestId: data.requestId || 'unknown',
        timestamp: new Date().toISOString(),
      },
    };

    this.auditLog.push(entry);
    this.logger.log(`[AUDIT] ${action} ${data.entity}:${data.entityId} by ${data.userName}`);

    return entry;
  }

  async findByEntity(entity: string, entityId: string, page = 1, limit = 20) {
    const logs = this.auditLog
      .filter((l) => l.entity === entity && l.entityId === entityId)
      .sort((a, b) => new Date(b.metadata.timestamp).getTime() - new Date(a.metadata.timestamp).getTime());

    const start = (page - 1) * limit;
    return {
      data: logs.slice(start, start + limit),
      total: logs.length,
      page,
      limit,
    };
  }

  async findByUser(userId: string, page = 1, limit = 20) {
    const logs = this.auditLog
      .filter((l) => l.userId === userId)
      .sort((a, b) => new Date(b.metadata.timestamp).getTime() - new Date(a.metadata.timestamp).getTime());

    const start = (page - 1) * limit;
    return {
      data: logs.slice(start, start + limit),
      total: logs.length,
      page,
      limit,
    };
  }

  async findByTenant(tenantId: string, page = 1, limit = 20) {
    const logs = this.auditLog
      .filter((l) => l.tenantId === tenantId)
      .sort((a, b) => new Date(b.metadata.timestamp).getTime() - new Date(a.metadata.timestamp).getTime());

    const start = (page - 1) * limit;
    return {
      data: logs.slice(start, start + limit),
      total: logs.length,
      page,
      limit,
    };
  }

  async findByAction(action: string, page = 1, limit = 20) {
    const logs = this.auditLog
      .filter((l) => l.action === action)
      .sort((a, b) => new Date(b.metadata.timestamp).getTime() - new Date(a.metadata.timestamp).getTime());

    const start = (page - 1) * limit;
    return {
      data: logs.slice(start, start + limit),
      total: logs.length,
      page,
      limit,
    };
  }

  async getRecent(limit = 50): Promise<AuditEntry[]> {
    return this.auditLog
      .sort((a, b) => new Date(b.metadata.timestamp).getTime() - new Date(a.metadata.timestamp).getTime())
      .slice(0, limit);
  }

  private getChangedFields(before: any, after: any): string[] {
    if (!before || !after) return [];
    const fields: string[] = [];

    for (const key of Object.keys(after)) {
      if (JSON.stringify(before[key]) !== JSON.stringify(after[key])) {
        fields.push(key);
      }
    }

    return fields;
  }
}
```

```typescript
// src/audit/audit.controller.ts (Controller)
import { Controller, Get, Query, Param, Req } from '@nestjs/common';
import { AuditService } from './audit.service';
import { Request } from 'express';

@Controller('audit')
export class AuditController {
  constructor(private auditService: AuditService) {}

  @Get()
  async getRecent(@Query('limit') limit = 50) {
    return this.auditService.getRecent(+limit);
  }

  @Get('entity/:entity/:entityId')
  async getByEntity(
    @Param('entity') entity: string,
    @Param('entityId') entityId: string,
    @Query('page') page = 1,
    @Query('limit') limit = 20,
  ) {
    return this.auditService.findByEntity(entity, entityId, +page, +limit);
  }

  @Get('user/:userId')
  async getByUser(
    @Param('userId') userId: string,
    @Query('page') page = 1,
    @Query('limit') limit = 20,
  ) {
    return this.auditService.findByUser(userId, +page, +limit);
  }

  @Get('action/:action')
  async getByAction(
    @Param('action') action: string,
    @Query('page') page = 1,
    @Query('limit') limit = 20,
  ) {
    return this.auditService.findByAction(action.toUpperCase(), +page, +limit);
  }
}
```

```typescript
// src/content/content.service.ts (Integrate audit)
import { Injectable } from '@nestjs/common';
import { AuditService } from '../audit/audit.service';

@Injectable()
export class ContentService {
  constructor(private auditService: AuditService) {}

  async updateContent(id: string, data: any, userId: string, userName: string, req?: any) {
    const before = await this.findOne(id);
    const after = { ...before, ...data };

    const updated = await this.update(id, data);

    await this.auditService.log('UPDATE', {
      entity: 'Content',
      entityId: id,
      userId,
      userName,
      tenantId: 'default',
      before,
      after,
      ipAddress: req?.ip,
      userAgent: req?.headers?.['user-agent'],
      requestId: req?.id,
    });

    return updated;
  }
}
```
