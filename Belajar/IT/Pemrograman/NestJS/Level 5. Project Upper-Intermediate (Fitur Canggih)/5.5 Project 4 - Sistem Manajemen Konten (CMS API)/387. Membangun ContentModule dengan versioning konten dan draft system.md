# 387. Membangun ContentModule dengan versioning konten dan draft system

## Definisi & Target
ContentModule menangani CRUD konten dengan versioning (setiap update menyimpan versi sebelumnya) dan draft system (draft -> review -> published).

## Komponen & Nest CLI Command
```bash
nest g mo content
nest g s content
nest g co content
```

## Kode Implementasi

```typescript
// src/content/content.service.ts (Service)
import { Injectable, NotFoundException } from '@nestjs/common';

interface Content {
  id: string;
  title: string;
  slug: string;
  body: string;
  status: string;
  version: number;
  authorId: string;
  categoryId?: string;
  tenantId: string;
  publishedAt?: Date;
  scheduledAt?: Date;
  createdAt: Date;
  updatedAt: Date;
}

interface ContentVersion {
  contentId: string;
  version: number;
  title: string;
  body: string;
  createdBy: string;
  createdAt: Date;
}

@Injectable()
export class ContentService {
  private contents: Content[] = [];
  private versions: ContentVersion[] = [];
  private counter = 0;

  async create(data: Partial<Content>): Promise<Content> {
    const content: Content = {
      id: `content-${++this.counter}`,
      title: data.title,
      slug: this.generateSlug(data.title),
      body: data.body,
      status: 'draft',
      version: 1,
      authorId: data.authorId,
      tenantId: data.tenantId,
      createdAt: new Date(),
      updatedAt: new Date(),
    };

    this.contents.push(content);
    await this.saveVersion(content, data.authorId);
    return content;
  }

  async update(id: string, data: Partial<Content>, userId: string): Promise<Content> {
    const content = this.contents.find((c) => c.id === id);
    if (!content) throw new NotFoundException('Content not found');

    Object.assign(content, data);
    content.version += 1;
    content.updatedAt = new Date();

    await this.saveVersion(content, userId);
    return content;
  }

  async saveAsDraft(id: string, data: Partial<Content>, userId: string): Promise<Content> {
    const content = await this.update(id, { ...data, status: 'draft' }, userId);
    return content;
  }

  async submitForReview(id: string, userId: string): Promise<Content> {
    const content = this.contents.find((c) => c.id === id);
    if (!content) throw new NotFoundException('Content not found');
    content.status = 'review';
    content.updatedAt = new Date();
    return content;
  }

  async publish(id: string, userId: string): Promise<Content> {
    const content = this.contents.find((c) => c.id === id);
    if (!content) throw new NotFoundException('Content not found');
    content.status = 'published';
    content.publishedAt = new Date();
    content.updatedAt = new Date();
    return content;
  }

  async schedule(id: string, scheduledAt: Date, userId: string): Promise<Content> {
    const content = this.contents.find((c) => c.id === id);
    if (!content) throw new NotFoundException('Content not found');
    content.status = 'scheduled';
    content.scheduledAt = scheduledAt;
    content.updatedAt = new Date();
    return content;
  }

  async archive(id: string): Promise<Content> {
    const content = this.contents.find((c) => c.id === id);
    if (!content) throw new NotFoundException('Content not found');
    content.status = 'archived';
    content.updatedAt = new Date();
    return content;
  }

  async getVersions(contentId: string): Promise<ContentVersion[]> {
    return this.versions
      .filter((v) => v.contentId === contentId)
      .sort((a, b) => b.version - a.version);
  }

  async rollback(contentId: string, version: number, userId: string): Promise<Content> {
    const snapshot = this.versions.find((v) => v.contentId === contentId && v.version === version);
    if (!snapshot) throw new NotFoundException('Version not found');

    const content = this.contents.find((c) => c.id === contentId);
    content.title = snapshot.title;
    content.body = snapshot.body;
    content.version += 1;
    content.updatedAt = new Date();

    await this.saveVersion(content, userId);
    return content;
  }

  private async saveVersion(content: Content, userId: string): Promise<void> {
    this.versions.push({
      contentId: content.id,
      version: content.version,
      title: content.title,
      body: content.body,
      createdBy: userId,
      createdAt: new Date(),
    });
  }

  private generateSlug(title: string): string {
    return title.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
  }
}
```
