# 386. Cara membuat PoliciesGuard yang menggunakan CASL untuk pengecekan izin

## Definisi & Target
Membuat guard yang mengecek ability user menggunakan CASL sebelum mengakses endpoint, dengan decorator untuk mendefinisikan permission yang diperlukan.

## Komponen
File: `src/casl/policies.guard.ts`

## Kode Implementasi

```typescript
// src/casl/policies.guard.ts (Guard)
import { Injectable, CanActivate, ExecutionContext, SetMetadata } from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { AbilityFactory, Action, Subjects } from './ability.factory';

export const CHECK_POLICIES_KEY = 'check_policies';
export const CheckPolicies = (action: Action, subject: Subjects) =>
  SetMetadata(CHECK_POLICIES_KEY, { action, subject });

@Injectable()
export class PoliciesGuard implements CanActivate {
  constructor(
    private reflector: Reflector,
    private abilityFactory: AbilityFactory,
  ) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const policy = this.reflector.getAllAndOverride<{ action: Action; subject: Subjects }>(
      CHECK_POLICIES_KEY,
      [context.getHandler(), context.getClass()],
    );

    if (!policy) return true;

    const { user } = context.switchToHttp().getRequest();
    const ability = this.abilityFactory.defineAbilities(user.role, user.id);

    return ability.can(policy.action, policy.subject);
  }
}
```

```typescript
// src/content/content.controller.ts (Controller with policy guard)
import { Controller, Get, Post, Put, Delete, Param, Body, UseGuards, Req } from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { PoliciesGuard, CheckPolicies } from '../casl/policies.guard';
import { Action } from '../casl/ability.factory';
import { ContentService } from './content.service';

@Controller('content')
@UseGuards(AuthGuard('jwt'), PoliciesGuard)
export class ContentController {
  constructor(private contentService: ContentService) {}

  @Get()
  @CheckPolicies(Action.Read, 'Content')
  async findAll() {
    return this.contentService.findAll();
  }

  @Post()
  @CheckPolicies(Action.Create, 'Content')
  async create(@Body() body: any, @Req() req: any) {
    return this.contentService.create({ ...body, authorId: req.user.id });
  }

  @Put(':id')
  @CheckPolicies(Action.Update, 'Content')
  async update(@Param('id') id: string, @Body() body: any) {
    return this.contentService.update(id, body);
  }

  @Delete(':id')
  @CheckPolicies(Action.Delete, 'Content')
  async delete(@Param('id') id: string) {
    return this.contentService.delete(id);
  }

  @Post(':id/publish')
  @CheckPolicies(Action.Publish, 'Content')
  async publish(@Param('id') id: string) {
    return this.contentService.publish(id);
  }
}
```
