# 388. Mengimplementasikan workflow publish yaitu draft, review, scheduled, dan published

## Definisi & Target
Mengimplementasikan workflow status konten: Draft -> Review -> Scheduled/Published -> Archived.

## Komponen
File yang dimodifikasi: `src/content/content.service.ts`

## Kode Implementasi

```typescript
// src/content/content.service.ts (Service with workflow)
import { Injectable, BadRequestException } from '@nestjs/common';

type ContentStatus = 'draft' | 'review' | 'scheduled' | 'published' | 'archived';

interface Content {
  id: string;
  title: string;
  status: ContentStatus;
  scheduledAt?: Date;
  authorId: string;
  reviewerId?: string;
  publishedAt?: Date;
}

const VALID_TRANSITIONS: Record<ContentStatus, ContentStatus[]> = {
  draft: ['review'],
  review: ['draft', 'scheduled', 'published'],
  scheduled: ['published', 'draft'],
  published: ['archived'],
  archived: ['draft'],
};

@Injectable()
export class ContentWorkflowService {
  private contents: Content[] = [];
  private counter = 0;

  async transition(id: string, newStatus: ContentStatus, userId: string): Promise<Content> {
    const content = this.contents.find((c) => c.id === id);
    if (!content) throw new BadRequestException('Content not found');

    const allowed = VALID_TRANSITIONS[content.status];
    if (!allowed.includes(newStatus)) {
      throw new BadRequestException(
        `Cannot transition from ${content.status} to ${newStatus}. Allowed: ${allowed.join(', ')}`,
      );
    }

    content.status = newStatus;

    switch (newStatus) {
      case 'published':
        content.publishedAt = new Date();
        content.reviewerId = userId;
        break;
      case 'review':
        break;
      case 'archived':
        break;
    }

    return content;
  }

  async submitForReview(id: string): Promise<Content> {
    return this.transition(id, 'review', null);
  }

  async approve(id: string, userId: string): Promise<Content> {
    return this.transition(id, 'published', userId);
  }

  async schedule(id: string, scheduledAt: Date): Promise<Content> {
    const content = this.contents.find((c) => c.id === id);
    content.scheduledAt = scheduledAt;
    return this.transition(id, 'scheduled', null);
  }

  async reject(id: string): Promise<Content> {
    return this.transition(id, 'draft', null);
  }

  async archive(id: string): Promise<Content> {
    return this.transition(id, 'archived', null);
  }

  async restore(id: string): Promise<Content> {
    return this.transition(id, 'draft', null);
  }

  async getContentsByStatus(status: ContentStatus): Promise<Content[]> {
    return this.contents.filter((c) => c.status === status);
  }

  async getScheduledContents(): Promise<Content[]> {
    return this.contents.filter(
      (c) => c.status === 'scheduled' && c.scheduledAt <= new Date(),
    );
  }
}
```

```

    +-------+                +--------+
       ^                     |    |
       |      reject        |    | approve
       |                     v    v
       |                 +-----------+     schedule     +-----------+
       +-----------------| Published | <-------------- | Scheduled |
       |                 +-----------+                  +-----------+
       |                      |
       |                   archive
       |                      v
       |                 +----------+

                        +----------+
```
