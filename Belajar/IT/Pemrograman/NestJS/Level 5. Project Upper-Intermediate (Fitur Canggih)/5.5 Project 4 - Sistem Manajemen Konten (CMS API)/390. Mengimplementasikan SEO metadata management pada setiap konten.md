# 390. Mengimplementasikan SEO metadata management pada setiap konten

## Definisi & Target
Menambahkan SEO metadata (meta title, description, keywords, OG image, canonical URL) ke setiap konten untuk optimasi mesin pencari.

## Komponen
File yang dimodifikasi: `src/content/seo.service.ts`

## Kode Implementasi

```typescript
// src/content/seo.service.ts (Service)
import { Injectable } from '@nestjs/common';

interface SeoMetadata {
  metaTitle: string;
  metaDescription: string;
  metaKeywords: string;
  ogImage: string;
  ogTitle: string;
  ogDescription: string;
  canonicalUrl: string;
  noIndex: boolean;
  structuredData?: Record<string, any>;
}

@Injectable()
export class SeoService {
  private seoStore: Map<string, SeoMetadata> = new Map();

  async saveSeo(contentId: string, seo: Partial<SeoMetadata>): Promise<SeoMetadata> {
    const existing = this.seoStore.get(contentId) || this.getDefaultSeo();
    const updated: SeoMetadata = { ...existing, ...seo };
    this.seoStore.set(contentId, updated);
    return updated;
  }

  async getSeo(contentId: string): Promise<SeoMetadata | null> {
    return this.seoStore.get(contentId) || null;
  }

  async generateSeoFromContent(contentId: string, title: string, body: string, imageUrl?: string): Promise<SeoMetadata> {
    const seo: SeoMetadata = {
      metaTitle: this.truncate(title, 60),
      metaDescription: this.truncate(this.stripHtml(body), 160),
      metaKeywords: this.extractKeywords(title, body),
      ogImage: imageUrl || '',
      ogTitle: title,
      ogDescription: this.truncate(this.stripHtml(body), 200),
      canonicalUrl: `${process.env.APP_URL}/content/${contentId}`,
      noIndex: false,
      structuredData: this.generateArticleStructuredData(title, body),
    };

    this.seoStore.set(contentId, seo);
    return seo;
  }

  async getSeoHtmlTags(contentId: string): Promise<string> {
    const seo = await this.getSeo(contentId);
    if (!seo) return '';

    return `
      <title>${seo.metaTitle}</title>
      <meta name="description" content="${seo.metaDescription}" />
      <meta name="keywords" content="${seo.metaKeywords}" />
      <meta property="og:title" content="${seo.ogTitle}" />
      <meta property="og:description" content="${seo.ogDescription}" />
      <meta property="og:image" content="${seo.ogImage}" />
      <link rel="canonical" href="${seo.canonicalUrl}" />
      ${seo.noIndex ? '<meta name="robots" content="noindex" />' : ''}
      <script type="application/ld+json">${JSON.stringify(seo.structuredData)}</script>
    `;
  }

  private getDefaultSeo(): SeoMetadata {
    return {
      metaTitle: '',
      metaDescription: '',
      metaKeywords: '',
      ogImage: '',
      ogTitle: '',
      ogDescription: '',
      canonicalUrl: '',
      noIndex: false,
    };
  }

  private truncate(text: string, maxLength: number): string {
    if (text.length <= maxLength) return text;
    return text.substring(0, maxLength - 3) + '...';
  }

  private stripHtml(html: string): string {
    return html.replace(/<[^>]*>/g, '').replace(/\s+/g, ' ').trim();
  }

  private extractKeywords(title: string, body: string): string {
    const words = (title + ' ' + body).toLowerCase().match(/\b[a-z]{3,}\b/g) || [];
    const unique = [...new Set(words)];
    return unique.slice(0, 10).join(', ');
  }

  private generateArticleStructuredData(title: string, body: string): Record<string, any> {
    return {
      '@context': 'https://schema.org',
      '@type': 'Article',
      headline: title,
      description: this.stripHtml(body).substring(0, 200),
      datePublished: new Date().toISOString(),
      author: { '@type': 'Person', name: 'Author' },
    };
  }
}
```

```typescript
// src/content/content.controller.ts (Controller - SEO endpoints)
import { Controller, Post, Get, Param, Body } from '@nestjs/common';
import { SeoService } from './seo.service';

@Controller('content')
export class ContentSeoController {
  constructor(private seoService: SeoService) {}

  @Post(':id/seo')
  async updateSeo(@Param('id') id: string, @Body() seo: any) {
    return this.seoService.saveSeo(id, seo);
  }

  @Get(':id/seo')
  async getSeo(@Param('id') id: string) {
    return this.seoService.getSeo(id);
  }

  @Post(':id/generate-seo')
  async generateSeo(@Param('id') id: string, @Body() body: { title: string; content: string; image?: string }) {
    return this.seoService.generateSeoFromContent(id, body.title, body.content, body.image);
  }

  @Get(':id/seo-tags')
  async getSeoTags(@Param('id') id: string) {
    return this.seoService.getSeoHtmlTags(id);
  }
}
```
