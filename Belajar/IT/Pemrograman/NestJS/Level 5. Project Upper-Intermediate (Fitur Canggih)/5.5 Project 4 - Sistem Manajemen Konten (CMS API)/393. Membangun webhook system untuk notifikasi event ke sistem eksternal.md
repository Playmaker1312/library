# 393. Membangun webhook system untuk notifikasi event ke sistem eksternal

## Definisi & Target
Webhook system mengirim notifikasi HTTP POST ke URL eksternal saat event tertentu terjadi (content published, media uploaded, user registered).

## Komponen & Nest CLI Command
```bash
nest g mo webhook
nest g s webhook
nest g co webhook
```

## Kode Implementasi

```typescript
// src/webhook/webhook.service.ts (Service)
import { Injectable, Logger } from '@nestjs/common';
import * as crypto from 'crypto';

interface WebhookRegistration {
  id: string;
  name: string;
  url: string;
  events: string[];
  secret: string;
  isActive: boolean;
  retryCount: number;
  timeout: number;
}

@Injectable()
export class WebhookService {
  private readonly logger = new Logger(WebhookService.name);
  private webhooks: WebhookRegistration[] = [];

  async register(data: { name: string; url: string; events: string[] }): Promise<WebhookRegistration> {
    const webhook: WebhookRegistration = {
      id: `wh-${Date.now()}`,
      name: data.name,
      url: data.url,
      events: data.events,
      secret: crypto.randomBytes(32).toString('hex'),
      isActive: true,
      retryCount: 3,
      timeout: 5000,
    };

    this.webhooks.push(webhook);
    return webhook;
  }

  async unregister(id: string): Promise<void> {
    this.webhooks = this.webhooks.filter((w) => w.id !== id);
  }

  async dispatch(event: string, payload: any): Promise<void> {
    const matchingWebhooks = this.webhooks.filter(
      (w) => w.isActive && (w.events.includes(event) || w.events.includes('*')),
    );

    this.logger.log(`Dispatching event "${event}" to ${matchingWebhooks.length} webhooks`);

    for (const webhook of matchingWebhooks) {
      await this.sendWithRetry(webhook, event, payload);
    }
  }

  private async sendWithRetry(webhook: WebhookRegistration, event: string, payload: any, attempt = 1): Promise<void> {
    try {
      const body = JSON.stringify({ event, payload, timestamp: new Date().toISOString() });
      const signature = crypto
        .createHmac('sha256', webhook.secret)
        .update(body)
        .digest('hex');

      const response = await fetch(webhook.url, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          'X-Webhook-Signature': signature,
          'X-Webhook-Event': event,
          'User-Agent': 'NestJS-CMS-Webhook/1.0',
        },
        body,
        signal: AbortSignal.timeout(webhook.timeout),
      });

      if (!response.ok) {
        throw new Error(`HTTP ${response.status}: ${response.statusText}`);
      }

      this.logger.log(`Webhook ${webhook.id} sent successfully to ${webhook.url}`);
    } catch (error) {
      this.logger.error(`Webhook ${webhook.id} attempt ${attempt}/${webhook.retryCount} failed: ${error.message}`);

      if (attempt < webhook.retryCount) {
        const delay = Math.pow(2, attempt) * 1000;
        await new Promise((resolve) => setTimeout(resolve, delay));
        await this.sendWithRetry(webhook, event, payload, attempt + 1);
      } else {
        this.logger.error(`Webhook ${webhook.id} permanently failed after ${attempt} attempts`);
      }
    }
  }

  async getWebhookLogs(webhookId: string): Promise<any[]> {
    return [];
  }
}
```

```typescript
// src/webhook/webhook.controller.ts (Controller)
import { Controller, Post, Get, Param, Body, Delete } from '@nestjs/common';
import { WebhookService } from './webhook.service';

@Controller('webhooks')
export class WebhookController {
  constructor(private webhookService: WebhookService) {}

  @Post()
  async register(@Body() body: { name: string; url: string; events: string[] }) {
    return this.webhookService.register(body);
  }

  @Delete(':id')
  async unregister(@Param('id') id: string) {
    await this.webhookService.unregister(id);
    return { deleted: true };
  }
}
```

```typescript
// src/content/content.service.ts (Integrate webhook dispatch)
import { Injectable } from '@nestjs/common';
import { WebhookService } from '../webhook/webhook.service';

@Injectable()
export class ContentService {
  constructor(private webhookService: WebhookService) {}

  async publishContent(id: string) {
    const content = await this.findOne(id);

    await this.webhookService.dispatch('content.published', {
      id: content.id,
      title: content.title,
      slug: content.slug,
      publishedAt: new Date().toISOString(),
      url: `${process.env.APP_URL}/content/${content.slug}`,
    });

    return content;
  }
}
```
