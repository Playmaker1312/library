# 384. Cara menginstal dan mengkonfigurasi CASL dalam NestJS

## Definisi & Target
Mengintegrasikan CASL dengan NestJS sebagai modul global untuk pengecekan permission.

## Komponen & Nest CLI Command
```bash
npm install @casl/ability @casl/prisma
nest g mo casl
nest g s casl/ability-factory
```

## Kode Implementasi

```typescript
// src/casl/casl.module.ts (Module)
import { Module, Global } from '@nestjs/common';
import { AbilityFactory } from './ability.factory';

@Global()
@Module({
  providers: [AbilityFactory],
  exports: [AbilityFactory],
})
export class CaslModule {}
```

```typescript
// src/casl/ability.factory.ts (Factory)
import { AbilityBuilder, Ability, AbilityClass } from '@casl/ability';
import { Injectable } from '@nestjs/common';

type Action = 'manage' | 'create' | 'read' | 'update' | 'delete' | 'publish';
type Subject = 'Content' | 'Media' | 'User' | 'Role' | 'Setting' | 'all';

@Injectable()
export class AbilityFactory {
  createForUser(user: { id: string; role: string; permissions: Array<{ action: string; subject: string }> }) {
    const { can, cannot, build } = new AbilityBuilder<Ability<[Action, Subject]>>(
      Ability as AbilityClass<Ability<[Action, Subject]>>,
    );

    if (user.role === 'super_admin') {
      can('manage', 'all');
    } else {
      for (const perm of user.permissions) {
        can(perm.action as Action, perm.subject as Subject);
      }
    }

    return build();
  }
}
```

```typescript
// src/casl/policies.guard.ts (Guard)
import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { AbilityFactory } from './ability.factory';

@Injectable()
export class PoliciesGuard implements CanActivate {
  constructor(private abilityFactory: AbilityFactory) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const user = request.user;
    const ability = this.abilityFactory.createForUser(user);

    request.ability = ability;
    return true;
  }
}
```
