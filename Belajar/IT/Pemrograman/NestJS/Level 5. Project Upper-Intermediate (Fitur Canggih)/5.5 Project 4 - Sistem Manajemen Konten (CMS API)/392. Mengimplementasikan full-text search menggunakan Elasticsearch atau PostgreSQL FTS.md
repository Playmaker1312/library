## Kode Implementasi (PostgreSQL FTS)

```typescript
// src/search/search.service.ts (Service - PostgreSQL FTS)
import { Injectable } from '@nestjs/common';

interface SearchResult {
  id: string;
  title: string;
  excerpt: string;
  rank: number;
}

@Injectable()
export class SearchService {
  private documents: Array<{ id: string; title: string; body: string; category: string; tags: string[] }> = [];

  async search(query: string, options?: { category?: string; tags?: string[]; page?: number; limit?: number }): Promise<{ results: SearchResult[]; total: number }> {
    const searchTerms = query.toLowerCase().split(' ').filter(Boolean);

    let results = this.documents
      .map((doc) => ({
        ...doc,
        rank: this.calculateRank(doc, searchTerms),
      }))
      .filter((doc) => doc.rank > 0)
      .sort((a, b) => b.rank - a.rank);

    if (options?.category) {
      results = results.filter((r) => r.category === options.category);
    }

    if (options?.tags?.length) {
      results = results.filter((r) => options.tags.some((t) => r.tags.includes(t)));
    }

    const total = results.length;
    const page = options?.page || 1;
    const limit = options?.limit || 20;
    const start = (page - 1) * limit;

    return {
      results: results.slice(start, start + limit).map((r) => ({
        id: r.id,
        title: this.highlight(r.title, searchTerms),
        excerpt: this.highlight(this.getExcerpt(r.body, searchTerms), searchTerms),
        rank: r.rank,
      })),
      total,
    };
  }

  async indexDocument(doc: { id: string; title: string; body: string; category: string; tags: string[] }): Promise<void> {
    const existing = this.documents.findIndex((d) => d.id === doc.id);
    if (existing > -1) {
      this.documents[existing] = doc;
    } else {
      this.documents.push(doc);
    }
  }

  async removeDocument(id: string): Promise<void> {
    this.documents = this.documents.filter((d) => d.id !== id);
  }

  async reindexAll(docs: Array<{ id: string; title: string; body: string; category: string; tags: string[] }>): Promise<void> {
    this.documents = docs;
  }

  private calculateRank(doc: { title: string; body: string; category: string; tags: string[] }, terms: string[]): number {
    let rank = 0;
    const titleLower = doc.title.toLowerCase();
    const bodyLower = doc.body.toLowerCase();

    for (const term of terms) {
      if (titleLower.includes(term)) rank += 10;
      if (doc.tags.some((t) => t.toLowerCase().includes(term))) rank += 5;
      const bodyCount = (bodyLower.match(new RegExp(term, 'g')) || []).length;
      rank += bodyCount;
    }

    return rank;
  }

  private getExcerpt(body: string, terms: string[]): string {
    const lower = body.toLowerCase();
    let bestIndex = -1;

    for (const term of terms) {
      const idx = lower.indexOf(term);
      if (idx > -1 && (bestIndex === -1 || idx < bestIndex)) {
        bestIndex = idx;
      }
    }

    if (bestIndex === -1) return body.substring(0, 200);
    const start = Math.max(0, bestIndex - 50);
    const end = Math.min(body.length, bestIndex + 150);
    return (start > 0 ? '...' : '') + body.substring(start, end) + (end < body.length ? '...' : '');
  }

  private highlight(text: string, terms: string[]): string {
    let result = text;
    for (const term of terms) {
      const regex = new RegExp(`(${term})`, 'gi');
      result = result.replace(regex, '<mark>$1</mark>');
    }
    return result;
  }
}
```

## Kode Implementasi (Elasticsearch)

```typescript
// src/search/elasticsearch.service.ts (Service - Elasticsearch)
import { Injectable } from '@nestjs/common';
import { ElasticsearchService } from '@nestjs/elasticsearch';

@Injectable()
export class SearchElasticService {
  private index = 'contents';

  constructor(private readonly elasticsearchService: ElasticsearchService) {}

  async createIndex() {
    const exists = await this.elasticsearchService.indices.exists({ index: this.index });
    if (!exists) {
      await this.elasticsearchService.indices.create({
        index: this.index,
        body: {
          settings: { analysis: { analyzer: { indonesian: { type: 'standard' } } } },
          mappings: {
            properties: {
              title: { type: 'text', analyzer: 'indonesian' },
              body: { type: 'text', analyzer: 'indonesian' },
              category: { type: 'keyword' },
              tags: { type: 'keyword' },
              status: { type: 'keyword' },
              publishedAt: { type: 'date' },
            },
          },
        },
      });
    }
  }

  async indexDocument(doc: any) {
    await this.elasticsearchService.index({
      index: this.index,
      id: doc.id,
      body: doc,
      refresh: 'wait_for',
    });
  }

  async search(query: string, page = 1, limit = 20) {
    const { hits } = await this.elasticsearchService.search({
      index: this.index,
      body: {
        query: {
          multi_match: {
            query,
            fields: ['title^3', 'body', 'tags^2'],
            type: 'best_fields',
          },
        },
        highlight: {
          fields: { title: {}, body: {} },
        },
        from: (page - 1) * limit,
        size: limit,
      },
    });

    return {
      results: hits.hits.map((hit: any) => ({
        id: hit._id,
        score: hit._score,
        ...hit._source,
        highlights: hit.highlight,
      })),
      total: hits.total.value,
    };
  }
}
```
