# 383. Mengimplementasikan sistem permission yang granular menggunakan CASL

## Definisi & Target
CASL (Code Access Security Lists) adalah library untuk mendefinisikan permission secara granular berdasarkan action dan subject.

## Komponen & Perintah Instalasi
```bash
npm install @casl/ability @casl/prisma
```

## Konsep CASL
```
Ability = Action + Subject + Conditions

Contoh:
- User bisa update content miliknya sendiri
- Admin bisa delete content apapun
- Viewer hanya bisa read content yang published
```

## Kode Implementasi

```typescript
// src/casl/types.ts (Types)
import { AbilityBuilder, Ability, AbilityClass, ExtractSubjectType, InferSubjects } from '@casl/ability';
import { User } from '@prisma/client';

export type Action = 'manage' | 'create' | 'read' | 'update' | 'delete' | 'publish' | 'archive';

export type Subjects = 'Content' | 'Media' | 'User' | 'Role' | 'Setting' | 'Webhook' | 'ApiKey' | 'all';

export type AppAbility = Ability<[Action, Subjects]>;

export type PermissionObject = {
  action: Action;
  subject: Subjects;
  conditions?: any;
};
```

```typescript
// src/casl/ability.factory.ts (Factory)
import { AbilityBuilder, Ability, AbilityClass, ExtractSubjectType } from '@casl/ability';
import { Injectable } from '@nestjs/common';
import { User } from '@prisma/client';

@Injectable()
export class AbilityFactory {
  createForUser(user: User, permissions: Array<{ action: string; subject: string }>) {
    const { can, cannot, build } = new AbilityBuilder<Ability<[string, string]>>(
      Ability as AbilityClass<Ability<[string, string]>>,
    );

    if (user.role === 'super_admin') {
      can('manage', 'all');
    } else {
      for (const permission of permissions) {
        can(permission.action, permission.subject);
      }

      cannot('delete', 'Content', { authorId: { $ne: user.id } }).because(
        'You can only delete your own content',
      );
    }

    return build({
      detectSubjectType: (item) => item.constructor.name as ExtractSubjectType<string>,
    });
  }
}
```
