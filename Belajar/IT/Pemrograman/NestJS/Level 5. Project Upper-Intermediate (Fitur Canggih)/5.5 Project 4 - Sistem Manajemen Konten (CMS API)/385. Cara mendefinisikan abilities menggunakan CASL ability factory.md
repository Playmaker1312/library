# 385. Cara mendefinisikan abilities menggunakan CASL ability factory

## Definisi & Target
Mendefinisikan aturan permission (abilities) untuk setiap role menggunakan CASL AbilityBuilder.

## Komponen
File yang dimodifikasi: `src/casl/ability.factory.ts`

## Kode Implementasi

```typescript
// src/casl/ability.factory.ts (Factory)
import { Ability, AbilityBuilder, AbilityClass, ExtractSubjectType, InferSubjects } from '@casl/ability';
import { Injectable } from '@nestjs/common';

export enum Action {
  Manage = 'manage',
  Create = 'create',
  Read = 'read',
  Update = 'update',
  Delete = 'delete',
  Publish = 'publish',
  Archive = 'archive',
}

export type Subjects = 'Content' | 'Media' | 'User' | 'Role' | 'Setting' | 'Webhook' | 'ApiKey' | 'all';

export type AppAbility = Ability<[Action, Subjects]>;

@Injectable()
export class AbilityFactory {
  defineAbilities(role: string, userId: string) {
    const { can, cannot, build } = new AbilityBuilder<Ability<[Action, Subjects]>>(
      Ability as AbilityClass<Ability<[Action, Subjects]>>,
    );

    switch (role) {
      case 'super_admin':
        can(Action.Manage, 'all');
        break;

      case 'admin':
        can(Action.Create, ['Content', 'Media', 'Category', 'Tag']);
        can(Action.Read, 'all');
        can(Action.Update, ['Content', 'Media', 'Category', 'Tag']);
        can(Action.Delete, ['Content', 'Media', 'Category', 'Tag']);
        can(Action.Publish, 'Content');
        can(Action.Archive, 'Content');
        can(Action.Read, ['User', 'Role', 'Setting', 'Webhook']);
        can(Action.Update, ['Setting', 'Webhook']);
        break;

      case 'editor':
        can(Action.Create, ['Content', 'Media']);
        can(Action.Read, ['Content', 'Media', 'Category', 'Tag']);
        can(Action.Update, 'Content', { authorId: userId });
        can(Action.Delete, 'Content', { authorId: userId });
        can(Action.Publish, 'Content', { authorId: userId });
        break;

      case 'author':
        can(Action.Create, 'Content');
        can(Action.Read, ['Content', 'Category', 'Tag']);
        can(Action.Update, 'Content', { authorId: userId });
        can(Action.Delete, 'Content', { authorId: userId });
        cannot(Action.Publish, 'Content');
        break;

      case 'viewer':
        can(Action.Read, 'Content', { status: 'published' });
        can(Action.Read, ['Category', 'Tag']);
        break;
    }

    return build({
      detectSubjectType: (item) => item.constructor.name as ExtractSubjectType<Subjects>,
    });
  }
}
```

## Ability Rules
| Role | create | read | update | delete | publish | manage |
|------|--------|------|--------|--------|---------|--------|
| Super Admin | all | all | all | all | all | all |
| Admin | content,media | all | content,media | content,media | content | - |
| Editor | content,media | content | own | own | own | - |
| Author | content | content | own | own | - | - |
| Viewer | - | published | - | - | - | - |
