# 394. Mengimplementasikan API key management untuk akses programatik

## Definisi & Target

```typescript
// src/api-key/api-key.service.ts (Service)
import { Injectable, Logger } from '@nestjs/common';
import * as crypto from 'crypto';

interface ApiKey {
  id: string;
  name: string;
  key: string;
  prefix: string;
  tenantId: string;
  permissions: string[];
  isActive: boolean;
  expiresAt?: Date;
  lastUsedAt?: Date;
  rateLimit: number;
  createdAt: Date;
}

@Injectable()
export class ApiKeyService {
  private readonly logger = new Logger(ApiKeyService.name);
  private apiKeys: ApiKey[] = [];
  private rateLimitMap: Map<string, { count: number; resetAt: Date }> = new Map();

  async generate(data: { name: string; tenantId: string; permissions?: string[]; expiresInDays?: number }): Promise<{ key: string; prefix: string }> {
    const rawKey = crypto.randomBytes(32).toString('hex');
    const prefix = data.name.substring(0, 4).toUpperCase();
    const hashedKey = this.hashKey(rawKey);

    const apiKey: ApiKey = {
      id: `apikey-${Date.now()}`,
      name: data.name,
      key: hashedKey,
      prefix,
      tenantId: data.tenantId,
      permissions: data.permissions || ['read:content'],
      isActive: true,
      rateLimit: 100,
      expiresAt: data.expiresInDays ? new Date(Date.now() + data.expiresInDays * 86400000) : undefined,
      createdAt: new Date(),
    };

    this.apiKeys.push(apiKey);

    return {
      key: `${prefix}_${rawKey}`,
      prefix,
    };
  }

  async validate(key: string): Promise<{ valid: boolean; apiKey?: ApiKey }> {
    const hashedKey = this.hashKey(this.extractRawKey(key));
    const apiKey = this.apiKeys.find((k) => k.key === hashedKey);

    if (!apiKey) return { valid: false };
    if (!apiKey.isActive) return { valid: false };
    if (apiKey.expiresAt && apiKey.expiresAt < new Date()) return { valid: false };

    const rateLimitCheck = this.checkRateLimit(apiKey.id, apiKey.rateLimit);
    if (!rateLimitCheck.allowed) return { valid: false };

    apiKey.lastUsedAt = new Date();
    return { valid: true, apiKey };
  }

  async revoke(id: string): Promise<void> {
    const key = this.apiKeys.find((k) => k.id === id);
    if (key) key.isActive = false;
  }

  async list(tenantId: string): Promise<ApiKey[]> {
    return this.apiKeys
      .filter((k) => k.tenantId === tenantId)
      .map(({ key, ...rest }) => ({ ...rest, key: `${rest.prefix}_****` }));
  }

  async updatePermissions(id: string, permissions: string[]): Promise<void> {
    const key = this.apiKeys.find((k) => k.id === id);
    if (key) key.permissions = permissions;
  }

  private hashKey(key: string): string {
    return crypto.createHash('sha256').update(key).digest('hex');
  }

  private extractRawKey(compositeKey: string): string {
    const parts = compositeKey.split('_');
    return parts.slice(1).join('_');
  }

  private checkRateLimit(apiKeyId: string, limit: number): { allowed: boolean } {
    const now = new Date();
    const record = this.rateLimitMap.get(apiKeyId);

    if (!record || record.resetAt < now) {
      this.rateLimitMap.set(apiKeyId, { count: 1, resetAt: new Date(now.getTime() + 60000) });
      return { allowed: true };
    }

    if (record.count >= limit) {
      return { allowed: false };
    }

    record.count++;
    return { allowed: true };
  }
}
```

```typescript
// src/api-key/api-key.guard.ts (Guard)
import { Injectable, CanActivate, ExecutionContext, UnauthorizedException } from '@nestjs/common';
import { ApiKeyService } from './api-key.service';

@Injectable()
export class ApiKeyGuard implements CanActivate {
  constructor(private apiKeyService: ApiKeyService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const apiKey = request.headers['x-api-key'];

    if (!apiKey) {
      throw new UnauthorizedException('API key is required');
    }

    const { valid, apiKey: keyData } = await this.apiKeyService.validate(apiKey);

    if (!valid) {
      throw new UnauthorizedException('Invalid or expired API key');
    }

    request.apiKey = keyData;
    request.tenantId = keyData.tenantId;
    return true;
  }
}
```

```typescript
// src/api-key/api-key.controller.ts (Controller)
import { Controller, Post, Get, Param, Body, Delete, UseGuards } from '@nestjs/common';
import { ApiKeyService } from './api-key.service';

@Controller('api-keys')
export class ApiKeyController {
  constructor(private apiKeyService: ApiKeyService) {}

  @Post()
  async generate(@Body() body: { name: string; permissions?: string[]; expiresInDays?: number }) {
    return this.apiKeyService.generate({
      ...body,
      tenantId: 'default',
    });
  }

  @Get()
  async list() {
    return this.apiKeyService.list('default');
  }

  @Delete(':id')
  async revoke(@Param('id') id: string) {
    await this.apiKeyService.revoke(id);
    return { revoked: true };
  }
}
```
