# 391. Membangun sistem tag dan kategorisasi yang fleksibel

## Definisi & Target
Membangun sistem kategori (hierarchical) dan tag (flat) untuk mengelompokkan konten dengan fleksibel.

## Komponen & Nest CLI Command
```bash
nest g mo category
nest g s category
nest g co category
nest g mo tag
nest g s tag
nest g co tag
```

// src/category/category.service.ts (Service)
import { Injectable, NotFoundException } from '@nestjs/common';

interface Category {
  id: string;
  name: string;
  slug: string;
  description?: string;
  parentId?: string;
  order: number;
  createdAt: Date;
}

@Injectable()
export class CategoryService {
  private categories: Category[] = [];
  private counter = 0;

  async create(data: { name: string; description?: string; parentId?: string; order?: number }): Promise<Category> {
    if (data.parentId) {
      const parent = this.categories.find((c) => c.id === data.parentId);
      if (!parent) throw new NotFoundException('Parent category not found');
    }

    const category: Category = {
      id: `cat-${++this.counter}`,
      name: data.name,
      slug: this.generateSlug(data.name),
      description: data.description,
      parentId: data.parentId,
      order: data.order || 0,
      createdAt: new Date(),
    };

    this.categories.push(category);
    return category;
  }

  async findAll(): Promise<Category[]> {
    return this.categories.sort((a, b) => a.order - b.order);
  }

  async getTree(): Promise<any[]> {
    const roots = this.categories.filter((c) => !c.parentId);
    return roots.map((root) => this.buildTree(root));
  }

  async getChildren(parentId: string): Promise<Category[]> {
    return this.categories.filter((c) => c.parentId === parentId);
  }

  async getAncestors(categoryId: string): Promise<Category[]> {
    const ancestors: Category[] = [];
    let current = this.categories.find((c) => c.id === categoryId);

    while (current?.parentId) {
      const parent = this.categories.find((c) => c.id === current.parentId);
      if (parent) {
        ancestors.unshift(parent);
        current = parent;
      } else break;
    }

    return ancestors;
  }

  async update(id: string, data: Partial<Category>): Promise<Category> {
    const category = this.categories.find((c) => c.id === id);
    if (!category) throw new NotFoundException('Category not found');

    if (data.name) category.name = data.name;
    if (data.description !== undefined) category.description = data.description;
    if (data.order !== undefined) category.order = data.order;

    return category;
  }

  async delete(id: string): Promise<void> {
    const children = this.categories.filter((c) => c.parentId === id);
    for (const child of children) {
      child.parentId = null;
    }

    const index = this.categories.findIndex((c) => c.id === id);
    if (index > -1) this.categories.splice(index, 1);
  }

  private buildTree(category: Category): any {
    const children = this.categories
      .filter((c) => c.parentId === category.id)
      .map((child) => this.buildTree(child));

    return { ...category, children };
  }

  private generateSlug(name: string): string {
    return name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
  }
}

// src/tag/tag.service.ts (Service)
import { Injectable } from '@nestjs/common';

interface Tag {
  id: string;
  name: string;
  slug: string;
  count: number;
  createdAt: Date;
}

@Injectable()
export class TagService {
  private tags: Tag[] = [];
  private counter = 0;

  async create(name: string): Promise<Tag> {
    const existing = this.tags.find((t) => t.slug === this.generateSlug(name));
    if (existing) return existing;

    const tag: Tag = {
      id: `tag-${++this.counter}`,
      name,
      slug: this.generateSlug(name),
      count: 0,
      createdAt: new Date(),
    };

    this.tags.push(tag);
    return tag;
  }

  async bulkCreate(names: string[]): Promise<Tag[]> {
    return Promise.all(names.map((name) => this.create(name)));
  }

  async findAll(): Promise<Tag[]> {
    return this.tags.sort((a, b) => b.count - a.count);
  }

  async findPopular(limit = 10): Promise<Tag[]> {
    return this.tags.sort((a, b) => b.count - a.count).slice(0, limit);
  }

  async incrementCount(tagId: string): Promise<void> {
    const tag = this.tags.find((t) => t.id === tagId);
    if (tag) tag.count++;
  }

  async decrementCount(tagId: string): Promise<void> {
    const tag = this.tags.find((t) => t.id === tagId);
    if (tag && tag.count > 0) tag.count--;
  }

  async update(id: string, name: string): Promise<Tag> {
    const tag = this.tags.find((t) => t.id === id);
    if (tag) {
      tag.name = name;
      tag.slug = this.generateSlug(name);
    }
    return tag;
  }

  async delete(id: string): Promise<void> {
    const index = this.tags.findIndex((t) => t.id === id);
    if (index > -1) this.tags.splice(index, 1);
  }

  private generateSlug(name: string): string {
    return name.toLowerCase().replace(/[^a-z0-9]+/g, '-').replace(/^-|-$/g, '');
  }
}
