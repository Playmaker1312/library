# 389. Membangun MediaModule dengan upload ke S3 dan image optimization

## Definisi & Target
MediaModule menangani upload file ke S3 (Amazon S3 atau compatible), optimasi gambar (resize, kompresi), dan management media.

## Komponen & Perintah Instalasi
```bash
npm install @aws-sdk/client-s3 @aws-sdk/lib-storage sharp
npm install --save-dev @types/multer
nest g mo media
nest g s media
nest g co media
```

## Kode Implementasi

```typescript
// src/media/media.service.ts (Service)
import { Injectable, Logger } from '@nestjs/common';
import { S3Client, PutObjectCommand, DeleteObjectCommand } from '@aws-sdk/client-s3';
import * as sharp from 'sharp';
import { v4 as uuid } from 'uuid';

@Injectable()
export class MediaService {
  private readonly logger = new Logger(MediaService.name);
  private s3: S3Client;
  private bucket: string;

  constructor() {
    this.s3 = new S3Client({
      region: process.env.AWS_REGION || 'ap-southeast-1',
      credentials: {
        accessKeyId: process.env.AWS_ACCESS_KEY_ID,
        secretAccessKey: process.env.AWS_SECRET_ACCESS_KEY,
      },
    });
    this.bucket = process.env.S3_BUCKET || 'myapp-media';
  }

  async upload(
    file: Express.Multer.File,
    tenantId: string,
    options?: { resize?: { width: number; height?: number } },
  ) {
    const key = `${tenantId}/${uuid()}-${file.originalname}`;
    let buffer = file.buffer;
    let metadata: any = {};

    if (file.mimetype.startsWith('image/') && options?.resize) {
      const resized = await sharp(file.buffer)
        .resize(options.resize.width, options.resize.height, { fit: 'inside' })
        .jpeg({ quality: 80 })
        .toBuffer();

      const thumbnail = await sharp(file.buffer)
        .resize(150, 150, { fit: 'cover' })
        .jpeg({ quality: 60 })
        .toBuffer();

      await this.uploadToS3(`${key}-thumb`, thumbnail, 'image/jpeg');
      metadata.thumbnailUrl = `${process.env.CDN_URL}/${key}-thumb`;
      buffer = resized;
    }

    await this.uploadToS3(key, buffer, file.mimetype);

    return {
      url: `${process.env.CDN_URL}/${key}`,
      thumbnailUrl: metadata.thumbnailUrl,
      filename: file.originalname,
      mimeType: file.mimetype,
      size: buffer.length,
    };
  }

  async delete(key: string): Promise<void> {
    await this.s3.send(new DeleteObjectCommand({ Bucket: this.bucket, Key: key }));
    this.logger.log(`Deleted S3 object: ${key}`);
  }

  private async uploadToS3(key: string, body: Buffer, contentType: string): Promise<void> {
    await this.s3.send(new PutObjectCommand({
      Bucket: this.bucket,
      Key: key,
      Body: body,
      ContentType: contentType,
      ACL: 'public-read',
    }));
  }
}
```

```typescript
// src/media/media.controller.ts (Controller)
import { Controller, Post, Param, Delete, UseInterceptors, UploadedFile, BadRequestException } from '@nestjs/common';
import { FileInterceptor } from '@nestjs/platform-express';
import { MediaService } from './media.service';

@Controller('media')
export class MediaController {
  constructor(private mediaService: MediaService) {}

  @Post('upload')
  @UseInterceptors(FileInterceptor('file'))
  async upload(
    @UploadedFile() file: Express.Multer.File,
  ) {
    if (!file) throw new BadRequestException('No file uploaded');
    return this.mediaService.upload(file, 'default', { resize: { width: 1920 } });
  }

  @Delete(':key')
  async delete(@Param('key') key: string) {
    await this.mediaService.delete(key);
    return { deleted: true };
  }
}
```
