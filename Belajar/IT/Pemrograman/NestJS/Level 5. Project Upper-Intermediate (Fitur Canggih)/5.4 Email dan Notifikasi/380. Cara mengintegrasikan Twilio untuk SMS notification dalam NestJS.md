# 380. Cara mengintegrasikan Twilio untuk SMS notification dalam NestJS

## Definisi & Target
Mengirim SMS notification menggunakan Twilio API untuk notifikasi penting seperti OTP, konfirmasi order, atau reminder.

## Komponen & Perintah Instalasi
```bash
npm install twilio
npm install --save-dev @types/twilio
```

## Kode Implementasi

```typescript
// src/sms/sms.module.ts (Module)
import { Module } from '@nestjs/common';
import { SmsService } from './sms.service';
import { SmsController } from './sms.controller';

@Module({
  controllers: [SmsController],
  providers: [SmsService],
  exports: [SmsService],
})
export class SmsModule {}
```

```typescript
// src/sms/sms.service.ts (Service)
import { Injectable, Logger } from '@nestjs/common';
import * as twilio from 'twilio';

@Injectable()
export class SmsService {
  private readonly logger = new Logger(SmsService.name);
  private client: twilio.Twilio;

  constructor() {
    this.client = twilio(
      process.env.TWILIO_ACCOUNT_SID,
      process.env.TWILIO_AUTH_TOKEN,
    );
  }

  async sendSms(to: string, body: string): Promise<any> {
    try {
      const message = await this.client.messages.create({
        body,
        from: process.env.TWILIO_PHONE_NUMBER,
        to: this.formatPhoneNumber(to),
      });
      this.logger.log(`SMS sent to ${to}: ${message.sid}`);
      return { sid: message.sid, status: message.status };
    } catch (error) {
      this.logger.error(`Failed to send SMS to ${to}: ${error.message}`);
      throw error;
    }
  }

  async sendOtp(phoneNumber: string, otp: string): Promise<any> {
    return this.sendSms(
      phoneNumber,
      `Your OTP code is: ${otp}. Valid for 5 minutes.`,
    );
  }

  async sendOrderConfirmation(phoneNumber: string, orderId: string): Promise<any> {
    return this.sendSms(
      phoneNumber,
      `Order #${orderId} confirmed! We'll notify you when it ships.`,
    );
  }

  async sendShippingUpdate(phoneNumber: string, orderId: string, trackingNumber: string): Promise<any> {
    return this.sendSms(
      phoneNumber,
      `Order #${orderId} has been shipped. Tracking: ${trackingNumber}`,
    );
  }

  async sendReminder(phoneNumber: string, message: string): Promise<any> {
    return this.sendSms(phoneNumber, `Reminder: ${message}`);
  }

  async getMessageStatus(messageSid: string): Promise<string> {
    const message = await this.client.messages(messageSid).fetch();
    return message.status;
  }

  private formatPhoneNumber(phone: string): string {
    if (!phone.startsWith('+')) {
      return `+62${phone.replace(/^0+/, '')}`;
    }
    return phone;
  }
}
```

```typescript
// src/sms/sms.controller.ts (Controller)
import { Controller, Post, Body } from '@nestjs/common';
import { SmsService } from './sms.service';

@Controller('sms')
export class SmsController {
  constructor(private smsService: SmsService) {}

  @Post('send')
  async send(@Body('to') to: string, @Body('message') message: string) {
    return this.smsService.sendSms(to, message);
  }

  @Post('send-otp')
  async sendOtp(@Body('phone') phone: string, @Body('otp') otp: string) {
    return this.smsService.sendOtp(phone, otp);
  }
}
```

