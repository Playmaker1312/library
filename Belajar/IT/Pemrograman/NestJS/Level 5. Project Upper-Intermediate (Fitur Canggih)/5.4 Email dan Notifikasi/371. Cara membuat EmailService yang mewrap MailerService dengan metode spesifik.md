# 371. Cara membuat EmailService yang mewrap MailerService dengan metode spesifik

## Definisi & Target
Membuat `EmailService` sebagai abstraction layer di atas `MailerService` dengan metode spesifik untuk berbagai keperluan (welcome, reset password, order confirmation).

## Komponen & Nest CLI Command
```bash
nest g s mail/email
```

## Kode Implementasi

```typescript
// src/mail/email.service.ts (Service)
import { Injectable } from '@nestjs/common';
import { MailerService } from '@nestjs-modules/mailer';

@Injectable()
export class EmailService {
  constructor(private mailerService: MailerService) {}

  async sendWelcomeEmail(to: string, name: string, verificationToken: string) {
    await this.mailerService.sendMail({
      to,
      subject: 'Welcome to Our Platform!',
      template: 'welcome',
      context: {
        name,
        verificationUrl: `${process.env.APP_URL}/verify?token=${verificationToken}`,
        siteName: 'MyApp',
        year: new Date().getFullYear(),
      },
    });
  }

  async sendResetPasswordEmail(to: string, name: string, resetToken: string) {
    await this.mailerService.sendMail({
      to,
      subject: 'Reset Your Password',
      template: 'reset-password',
      context: {
        name,
        resetCode: resetToken,
        expiryMinutes: 15,
      },
    });
  }

  async sendOrderConfirmation(to: string, customerName: string, order: any) {
    await this.mailerService.sendMail({
      to,
      subject: `Order Confirmed - #${order.id}`,
      template: 'order-confirmation',
      context: {
        customerName,
        orderId: order.id,
        items: order.items,
        total: order.total,
        orderUrl: `${process.env.APP_URL}/orders/${order.id}`,
        siteName: 'MyApp',
        supportEmail: 'support@myapp.com',
        supportPhone: '+62-123-4567',
      },
    });
  }

  async sendInvoice(to: string, invoice: any, pdfBuffer: Buffer) {
    await this.mailerService.sendMail({
      to,
      subject: `Invoice #${invoice.number}`,
      template: 'invoice',
      context: { invoice },
      attachments: [
        {
          filename: `invoice-${invoice.number}.pdf`,
          content: pdfBuffer,
        },
      ],
    });
  }
}
```

```typescript
// src/mail/mail.module.ts (Module)
import { Module } from '@nestjs/common';
import { MailerModule } from '@nestjs-modules/mailer';
import { EmailService } from './email.service';
import { MailController } from './mail.controller';

@Module({
  imports: [
    MailerModule.forRoot({
      transport: {
        host: process.env.SMTP_HOST,
        port: parseInt(process.env.SMTP_PORT, 10),
        auth: { user: process.env.SMTP_USER, pass: process.env.SMTP_PASS },
      },
      defaults: { from: '"MyApp" <noreply@myapp.com>' },
      template: { dir: process.cwd() + '/templates', adapter: new (require('handlebars'))() },
    }),
  ],
  controllers: [MailController],
  providers: [EmailService],
  exports: [EmailService],
})
export class MailModule {}
```

// Di service lain
@Injectable()
export class AuthService {
  constructor(private emailService: EmailService) {}

  async register(email: string, name: string) {
    const token = 'verification-token-123';
    await this.emailService.sendWelcomeEmail(email, name, token);
    return { message: 'Registration successful, check your email' };
  }
}
