# 377. Cara mengimplementasikan retry mechanism untuk email yang gagal terkirim

## Definisi & Target
Menerapkan retry mechanism dengan exponential backoff untuk email yang gagal, serta mencatat kegagalan untuk investigasi.

## Komponen
File yang dimodifikasi: `src/mail/mail.processor.ts`

## Kode Implementasi

```typescript
// src/mail/mail.processor.ts (Processor with retry)
import { Processor, Process, OnQueueFailed, InjectQueue } from '@nestjs/bull';
import { Job, Queue } from 'bull';
import { MailerService } from '@nestjs-modules/mailer';

@Processor('mail')
export class MailProcessor {
  constructor(
    private mailerService: MailerService,
    @InjectQueue('mail') private mailQueue: Queue,
  ) {}

  @Process('send-email')
  async handleSendEmail(job: Job<{ to: string; subject: string; html: string }>) {
    const { to, subject, html } = job.data;

    try {
      await this.mailerService.sendMail({ to, subject, html });
      console.log(`Email sent to ${to} (attempt ${job.attemptsMade + 1})`);
      return { status: 'sent', to };
    } catch (error) {
      console.error(`Failed to send email to ${to}: ${error.message}`);

      if (job.attemptsMade < job.opts.attempts - 1) {
        const nextDelay = Math.pow(2, job.attemptsMade) * 1000;
        console.log(`Retrying in ${nextDelay}ms...`);
      } else {
        await this.handleFinalFailure(job, error);
      }

      throw error;
    }
  }

  private async handleFinalFailure(job: Job, error: Error) {
    console.error(`[FINAL FAILURE] Email to ${job.data.to} failed after ${job.opts.attempts} attempts`);

    await this.mailQueue.add('failed-email-log', {
      to: job.data.to,
      subject: job.data.subject,
      error: error.message,
      attempts: job.attemptsMade,
      failedAt: new Date().toISOString(),
    });
  }

  @Process('failed-email-log')
  async handleFailedEmailLog(job: Job) {
    console.log('Logging failed email:', job.data);
    await this.saveToDatabase(job.data);
  }

  @OnQueueFailed()
  async onFailed(job: Job, error: Error) {
    console.log(`[Event] Job ${job.id} failed: ${error.message}`);
  }

  private async saveToDatabase(data: any) {
    // Simpan ke database untuk investigasi
    console.log('Saved to database:', data);
  }
}
```

```typescript
// src/mail/mail.service.ts (Service - resend failed)
import { Injectable } from '@nestjs/common';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';

@Injectable()
export class MailService {
  constructor(@InjectQueue('mail') private mailQueue: Queue) {}

  async resendFailedEmails() {
    const failed = await this.mailQueue.getFailed();
    for (const job of failed) {
      await job.retry();
    }
    return { resent: failed.length };
  }
}
```
