# 374. Cara mengimplementasikan email notifikasi order kepada pembeli

## Definisi & Target
Mengirim email notifikasi otomatis kepada pembeli saat status order berubah (konfirmasi, pengiriman, selesai).

## Komponen & Nest CLI Command
```bash
nest g mo orders
nest g s orders
nest g co orders
```

## Kode Implementasi

```typescript
// src/orders/orders.service.ts (Service)
import { Injectable } from '@nestjs/common';
import { EmailService } from '../mail/email.service';

interface Order {
  id: string;
  userId: string;
  customerName: string;
  customerEmail: string;
  items: Array<{ name: string; quantity: number; price: number }>;
  total: number;
  status: string;
  trackingNumber?: string;
}

@Injectable()
export class OrdersService {
  private orders: Order[] = [];

  constructor(private emailService: EmailService) {}

  async createOrder(orderData: Partial<Order>): Promise<Order> {
    const order: Order = {
      id: `ORD-${Date.now()}`,
      userId: orderData.userId,
      customerName: orderData.customerName,
      customerEmail: orderData.customerEmail,
      items: orderData.items || [],
      total: this.calculateTotal(orderData.items || []),
      status: 'confirmed',
    };

    this.orders.push(order);

    await this.emailService.sendOrderConfirmation(
      order.customerEmail,
      order.customerName,
      order,
    );

    return order;
  }

  async shipOrder(orderId: string, trackingNumber: string): Promise<Order> {
    const order = this.orders.find((o) => o.id === orderId);
    if (!order) throw new Error('Order not found');

    order.status = 'shipped';
    order.trackingNumber = trackingNumber;

    await this.emailService.sendMail({
      to: order.customerEmail,
      subject: `Order Shipped - #${order.id}`,
      template: 'order-shipped',
      context: {
        customerName: order.customerName,
        orderId: order.id,
        trackingNumber,
        trackingUrl: `${process.env.APP_URL}/track/${trackingNumber}`,
        estimatedDelivery: new Date(Date.now() + 3 * 24 * 60 * 60 * 1000).toLocaleDateString(),
      },
    });

    return order;
  }

  async completeOrder(orderId: string): Promise<Order> {
    const order = this.orders.find((o) => o.id === orderId);
    if (!order) throw new Error('Order not found');

    order.status = 'delivered';

    await this.emailService.sendMail({
      to: order.customerEmail,
      subject: `Order Delivered - #${order.id}`,
      template: 'order-delivered',
      context: {
        customerName: order.customerName,
        orderId: order.id,
        reviewUrl: `${process.env.APP_URL}/orders/${order.id}/review`,
      },
    });

    return order;
  }

  private calculateTotal(items: Array<{ price: number; quantity: number }>): number {
    return items.reduce((sum, item) => sum + item.price * item.quantity, 0);
  }
}
```

```typescript
// src/orders/orders.controller.ts (Controller)
import { Controller, Post, Body, Param, Patch } from '@nestjs/common';
import { OrdersService } from './orders.service';

@Controller('orders')
export class OrdersController {
  constructor(private ordersService: OrdersService) {}

  @Post()
  async create(@Body() body: any) {
    return this.ordersService.createOrder(body);
  }

  @Patch(':id/ship')
  async ship(@Param('id') id: string, @Body('trackingNumber') tn: string) {
    return this.ordersService.shipOrder(id, tn);
  }

  @Patch(':id/complete')
  async complete(@Param('id') id: string) {
    return this.ordersService.completeOrder(id);
  }
}
```
