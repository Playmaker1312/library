# 326. Cara menggunakan Redis untuk menyimpan session dan blacklist token

## Definisi & Target
Menyimpan session user dan daftar token yang di-blacklist (misalnya setelah logout) di Redis dengan TTL, sehingga state management bersifat stateless dan scalable.

## Komponen & Nest CLI Command
```bash
nest g mo auth
nest g s auth
```

## Kode Implementasi

```typescript
// src/auth/session.service.ts (Service)
import { Injectable, Inject, CACHE_MANAGER } from '@nestjs/common';
import { Cache } from 'cache-manager';

@Injectable()
export class SessionService {
  constructor(@Inject(CACHE_MANAGER) private cache: Cache) {}

  async createSession(userId: string, token: string): Promise<void> {
    const sessionKey = `session:${userId}`;
    const existing = await this.cache.get<string[]>(sessionKey);
    const tokens = existing || [];
    tokens.push(token);
    await this.cache.set(sessionKey, tokens, 86400);
  }

  async invalidateSession(userId: string): Promise<void> {
    await this.cache.del(`session:${userId}`);
  }

  async isTokenBlacklisted(token: string): Promise<boolean> {
    const blacklist = await this.cache.get<string[]>('blacklist:tokens');
    return blacklist?.includes(token) || false;
  }

  async blacklistToken(token: string, ttl: number = 86400): Promise<void> {
    const key = 'blacklist:tokens';
    const existing = await this.cache.get<string[]>(key);
    const tokens = existing || [];
    tokens.push(token);
    await this.cache.set(key, tokens, ttl);
  }
}
```

```typescript
// src/auth/auth.guard.ts (Guard)
import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { SessionService } from './session.service';

@Injectable()
export class AuthGuard implements CanActivate {
  constructor(private sessionService: SessionService) {}

  async canActivate(context: ExecutionContext): Promise<boolean> {
    const request = context.switchToHttp().getRequest();
    const token = request.headers.authorization?.replace('Bearer ', '');
    if (!token) return false;

    const isBlacklisted = await this.sessionService.isTokenBlacklisted(token);
    return !isBlacklisted;
  }
}
```
