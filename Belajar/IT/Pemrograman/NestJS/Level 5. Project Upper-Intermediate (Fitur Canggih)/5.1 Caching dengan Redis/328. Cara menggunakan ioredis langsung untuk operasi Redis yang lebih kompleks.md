# 328. Cara menggunakan ioredis langsung untuk operasi Redis yang lebih kompleks

## Definisi & Target
Menggunakan package `ioredis` langsung (tanpa cache-manager) ketika membutuhkan operasi Redis lanjutan seperti Pub/Sub, pipeline, transaction, sorted sets, atau geospatial.

## Komponen & Nest CLI Command
```bash
npm install ioredis
npm install --save-dev @types/ioredis
nest g mo redis
nest g s redis
```

## Kode Implementasi

```typescript
// src/redis/redis.provider.ts (Provider)
import { Provider } from '@nestjs/common';
import Redis from 'ioredis';

export type RedisClient = Redis;

export const REDIS_CLIENT = 'REDIS_CLIENT';

export const redisProvider: Provider = {
  provide: REDIS_CLIENT,
  useFactory: () => {
    return new Redis({
      host: process.env.REDIS_HOST || 'localhost',
      port: parseInt(process.env.REDIS_PORT, 10) || 6379,
      password: process.env.REDIS_PASSWORD || undefined,
      retryStrategy: (times) => Math.min(times * 50, 2000),
    });
  },
};
```

```typescript
// src/redis/redis.service.ts (Service)
import { Injectable, Inject } from '@nestjs/common';
import { REDIS_CLIENT, RedisClient } from './redis.provider';

@Injectable()
export class RedisService {
  constructor(@Inject(REDIS_CLIENT) private readonly redis: RedisClient) {}

  async addToSortedSet(key: string, score: number, member: string): Promise<void> {
    await this.redis.zadd(key, score, member);
  }

  async getTopFromSortedSet(key: string, count: number = 10): Promise<string[]> {
    return this.redis.zrevrange(key, 0, count - 1);
  }

  async publish(channel: string, message: string): Promise<void> {
    await this.redis.publish(channel, message);
  }

  async subscribe(channel: string, callback: (message: string) => void): Promise<void> {
    const subscriber = this.redis.duplicate();
    await subscriber.subscribe(channel);
    subscriber.on('message', (_channel, message) => callback(message));
  }

  async pipelineExample(): Promise<void> {
    const pipeline = this.redis.pipeline();
    pipeline.set('key1', 'value1');
    pipeline.set('key2', 'value2');
    pipeline.incr('counter');
    await pipeline.exec();
  }

  async withTransaction(): Promise<void> {
    const result = await this.redis
      .multi()
      .set('key', 'value')
      .expire('key', 60)
      .get('key')
      .exec();
    return result;
  }
}
```

```typescript
// src/redis/redis.module.ts (Module)
import { Module, Global } from '@nestjs/common';
import { redisProvider } from './redis.provider';
import { RedisService } from './redis.service';

@Global()
@Module({
  providers: [redisProvider, RedisService],
  exports: [RedisService],
})
export class RedisModule {}
```
