# 325. Cara mengimplementasikan cache-aside pattern dalam service

## Definisi & Target
Cache-Aside (Lazy Loading) adalah pattern di mana aplikasi bertanggung jawab mengisi cache. Saat read, cek cache dulu (cache hit return, cache miss ambil dari DB lalu set ke cache). Saat write, update DB lalu hapus cache.

## Komponen
File yang dimodifikasi: `src/products/products.service.ts`

## Kode Implementasi

```typescript
// src/products/products.service.ts (Service)
import { Injectable, Inject, CACHE_MANAGER, Logger } from '@nestjs/common';
import { Cache } from 'cache-manager';

interface Product {
  id: string;
  name: string;
  price: number;
}

@Injectable()
export class ProductsService {
  private readonly logger = new Logger(ProductsService.name);
  private db: Product[] = [
    { id: '1', name: 'Laptop', price: 15000000 },
    { id: '2', name: 'Keyboard', price: 500000 },
  ];

  constructor(@Inject(CACHE_MANAGER) private cache: Cache) {}

  async findById(id: string): Promise<Product> {
    const cacheKey = `product:${id}`;
    let product = await this.cache.get<Product>(cacheKey);

    if (product) {
      this.logger.log(`Cache HIT for ${cacheKey}`);
      return product;
    }

    this.logger.log(`Cache MISS for ${cacheKey}, fetching from DB`);
    product = this.db.find((p) => p.id === id);

    if (product) {
      await this.cache.set(cacheKey, product, 300);
    }

    return product;
  }

  async updateProduct(id: string, data: Partial<Product>): Promise<Product> {
    const index = this.db.findIndex((p) => p.id === id);
    if (index === -1) return null;

    this.db[index] = { ...this.db[index], ...data };

    await this.cache.del(`product:${id}`);
    this.logger.log(`Cache invalidated for product:${id}`);

    return this.db[index];
  }
}
```
