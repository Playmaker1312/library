# 327. Cara mengimplementasikan distributed lock menggunakan Redis dalam NestJS

## Definisi & Target
Distributed lock memungkinkan hanya satu instance aplikasi yang mengeksekusi suatu operasi pada satu waktu, penting untuk race condition di lingkungan multi-instance.

## Komponen & Nest CLI Command
```bash
nest g mo locks
nest g s locks
```

## Kode Implementasi

```typescript
// src/locks/lock.service.ts (Service)
import { Injectable, Inject, CACHE_MANAGER } from '@nestjs/common';
import { Cache } from 'cache-manager';

@Injectable()
export class LockService {
  constructor(@Inject(CACHE_MANAGER) private cache: Cache) {}

  async acquireLock(
    resource: string,
    ttl: number = 10,
    retryDelay: number = 100,
    maxRetries: number = 5,
  ): Promise<boolean> {
    const lockKey = `lock:${resource}`;
    const lockValue = Date.now().toString();

    for (let i = 0; i < maxRetries; i++) {
      const existing = await this.cache.get<string>(lockKey);
      if (!existing) {
        await this.cache.set(lockKey, lockValue, ttl);
        const stored = await this.cache.get<string>(lockKey);
        if (stored === lockValue) return true;
      }
      await this.delay(retryDelay);
    }
    return false;
  }

  async releaseLock(resource: string): Promise<void> {
    await this.cache.del(`lock:${resource}`);
  }

  private delay(ms: number): Promise<void> {
    return new Promise((resolve) => setTimeout(resolve, ms));
  }
}
```

```typescript
// src/locks/lock.controller.ts (Controller)
import { Controller, Post, Body } from '@nestjs/common';
import { LockService } from './lock.service';

@Controller('locks')
export class LockController {
  constructor(private readonly lockService: LockService) {}

  @Post('acquire')
  async acquire(@Body('resource') resource: string) {
    const acquired = await this.lockService.acquireLock(resource);
    return { resource, acquired };
  }

  @Post('release')
  async release(@Body('resource') resource: string) {
    await this.lockService.releaseLock(resource);
    return { resource, released: true };
  }
}
```
