# 330. Cara mengimplementasikan cache warming strategy saat aplikasi startup

## Definisi & Target
Cache warming adalah proses mengisi cache dengan data yang sering diakses segera setelah aplikasi startup, sehingga pengguna pertama tidak mengalami slow response karena cache miss.

## Komponen
File yang dimodifikasi: `src/common/cache/cache-warming.service.ts`

## Kode Implementasi

```typescript
// src/common/cache/cache-warming.service.ts (Service)
import { Injectable, OnApplicationBootstrap, Logger } from '@nestjs/common';
import { CacheService } from './cache.service';

@Injectable()
export class CacheWarmingService implements OnApplicationBootstrap {
  private readonly logger = new Logger(CacheWarmingService.name);

  constructor(private cacheService: CacheService) {}

  async onApplicationBootstrap(): Promise<void> {
    this.logger.log('Starting cache warming...');
    const startTime = Date.now();

    await Promise.all([
      this.warmProducts(),
      this.warmCategories(),
      this.warmConfigurations(),
    ]);

    const duration = Date.now() - startTime;
    this.logger.log(`Cache warming completed in ${duration}ms`);
  }

  private async warmProducts(): Promise<void> {
    const products = await this.fetchProductsFromDb();
    await this.cacheService.set('products:all', products, 300);
    for (const product of products) {
      await this.cacheService.set(`products:${product.id}`, product, 300);
    }
    this.logger.log(`Warmed ${products.length} products`);
  }

  private async warmCategories(): Promise<void> {
    const categories = await this.fetchCategoriesFromDb();
    await this.cacheService.set('categories:all', categories, 600);
    this.logger.log(`Warmed ${categories.length} categories`);
  }

  private async warmConfigurations(): Promise<void> {
    const config = await this.fetchConfigFromDb();
    await this.cacheService.set('app:config', config, 3600);
    this.logger.log('Warmed application configuration');
  }

  private async fetchProductsFromDb(): Promise<any[]> {
    return [{ id: '1', name: 'Laptop' }, { id: '2', name: 'Mouse' }];
  }

  private async fetchCategoriesFromDb(): Promise<any[]> {
    return [{ id: '1', name: 'Electronics' }];
  }

  private async fetchConfigFromDb(): Promise<any> {
    return { siteName: 'My Store', version: '1.0' };
  }
}
```

```typescript
// src/common/cache/cache.module.ts (Module)
import { Module } from '@nestjs/common';
import { CacheService } from './cache.service';
import { CacheWarmingService } from './cache-warming.service';

@Module({
  providers: [CacheService, CacheWarmingService],
  exports: [CacheService],
})
export class CacheServiceModule {}
```
