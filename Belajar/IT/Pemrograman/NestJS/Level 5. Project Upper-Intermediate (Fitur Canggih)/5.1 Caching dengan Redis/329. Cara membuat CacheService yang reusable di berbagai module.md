# 329. Cara membuat CacheService yang reusable di berbagai module

## Definisi & Target
Membuat CacheService terpusat yang membungkus operasi cache-manager dengan pattern typing, logging, dan error handling yang konsisten, sehingga bisa di-inject ke module manapun.

## Komponen & Nest CLI Command
```bash
nest g mo common
nest g s common/cache
```

## Kode Implementasi

```typescript
// src/common/cache/cache.service.ts (Service)
import { Injectable, Inject, CACHE_MANAGER, Logger } from '@nestjs/common';
import { Cache } from 'cache-manager';

@Injectable()
export class CacheService {
  private readonly logger = new Logger(CacheService.name);

  constructor(@Inject(CACHE_MANAGER) private cache: Cache) {}

  async get<T>(key: string): Promise<T | null> {
    try {
      const data = await this.cache.get<T>(key);
      if (data) this.logger.debug(`Cache HIT: ${key}`);
      return data || null;
    } catch (error) {
      this.logger.error(`Cache GET error for ${key}: ${error.message}`);
      return null;
    }
  }

  async set<T>(key: string, value: T, ttl: number = 60): Promise<boolean> {
    try {
      await this.cache.set(key, value, ttl);
      this.logger.debug(`Cache SET: ${key} (TTL: ${ttl}s)`);
      return true;
    } catch (error) {
      this.logger.error(`Cache SET error for ${key}: ${error.message}`);
      return false;
    }
  }

  async del(key: string): Promise<boolean> {
    try {
      await this.cache.del(key);
      this.logger.debug(`Cache DEL: ${key}`);
      return true;
    } catch (error) {
      this.logger.error(`Cache DEL error for ${key}: ${error.message}`);
      return false;
    }
  }

  async getOrSet<T>(key: string, fn: () => Promise<T>, ttl: number = 60): Promise<T> {
    const cached = await this.get<T>(key);
    if (cached) return cached;

    const fresh = await fn();
    await this.set(key, fresh, ttl);
    return fresh;
  }

  async delByPattern(pattern: string): Promise<void> {
    const keys = await this.cache.store.keys(`${pattern}*`);
    const promises = keys.map((key: string) => this.cache.del(key));
    await Promise.all(promises);
    this.logger.log(`Cache DEL by pattern: ${pattern}* (${keys.length} keys)`);
  }
}
```

```typescript
// src/common/cache/cache.module.ts (Module)
import { Module, CacheModule } from '@nestjs/common';
import { CacheService } from './cache.service';

@Module({
  imports: [CacheModule.register()],
  providers: [CacheService],
  exports: [CacheService],
})
export class CacheServiceModule {}
```

## Cara Integrasi & Pengujian
```typescript
// src/products/products.service.ts
@Injectable()
export class ProductsService {
  constructor(private cacheService: CacheService) {}

  async findAll() {
    return this.cacheService.getOrSet('products:all', () => this.fetchFromDb(), 60);
  }
}
```
