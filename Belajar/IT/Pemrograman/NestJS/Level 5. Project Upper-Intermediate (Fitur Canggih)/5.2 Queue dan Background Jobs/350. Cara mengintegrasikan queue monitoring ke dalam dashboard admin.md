# 350. Cara mengintegrasikan queue monitoring ke dalam dashboard admin

## Definisi & Target
Membuat endpoint REST API untuk memonitor status queue, job counts, dan performa worker, yang bisa diintegrasikan ke dashboard admin.

## Komponen & Nest CLI Command
```bash
nest g mo queue-admin
nest g co queue-admin
```

## Kode Implementasi

```typescript
// src/queue-admin/queue-admin.controller.ts (Controller)
import { Controller, Get, Param, Post, UseGuards } from '@nestjs/common';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';

@Controller('admin/queues')
export class QueueAdminController {
  constructor(
    @InjectQueue('email') private emailQueue: Queue,
    @InjectQueue('image') private imageQueue: Queue,
    @InjectQueue('notification') private notificationQueue: Queue,
  ) {}

  @Get()
  async getAllQueues() {
    const queues = await Promise.all(
      [this.emailQueue, this.imageQueue, this.notificationQueue].map(async (q) => ({
        name: q.name,
        ...await this.getQueueStats(q),
      })),
    );
    return queues;
  }

  @Get(':name')
  async getQueue(@Param('name') name: string) {
    const queue = await this.getQueueByName(name);
    if (!queue) return { error: 'Queue not found' };
    return {
      name: queue.name,
      ...await this.getQueueStats(queue),
      jobs: await this.getRecentJobs(queue),
    };
  }

  @Post(':name/retry-all')
  async retryAll(@Param('name') name: string) {
    const queue = await this.getQueueByName(name);
    const failed = await queue.getFailed();
    for (const job of failed) {
      await job.retry();
    }
    return { message: `Retried ${failed.length} jobs` };
  }

  @Post(':name/empty')
  async emptyQueue(@Param('name') name: string) {
    const queue = await this.getQueueByName(name);
    await queue.empty();
    return { message: 'Queue emptied' };
  }

  private async getQueueStats(queue: Queue) {
    const [waiting, active, completed, failed, delayed] = await Promise.all([
      queue.getWaitingCount(),
      queue.getActiveCount(),
      queue.getCompletedCount(),
      queue.getFailedCount(),
      queue.getDelayedCount(),
    ]);
    return { waiting, active, completed, failed, delayed };
  }

  private async getRecentJobs(queue: Queue, limit = 20) {
    const jobs = await Promise.all([
      queue.getJobs(['failed'], 0, limit - 1),
      queue.getJobs(['completed'], 0, limit - 1),
      queue.getJobs(['waiting'], 0, limit - 1),
    ]);
    return jobs.flat().slice(0, limit).map((j) => ({
      id: j.id,
      name: j.name,
      status: 'unknown',
      data: j.data,
      failedReason: j.failedReason,
      timestamp: j.timestamp,
    }));
  }

  private async getQueueByName(name: string): Promise<Queue | null> {
    const map = { email: this.emailQueue, image: this.imageQueue, notification: this.notificationQueue };
    return map[name] || null;
  }
}
```

## Cara Integrasi & Pengujian
```bash
# Lihat semua queue
curl http://localhost:3000/admin/queues
# Response: [{ name: "email", waiting: 5, active: 2, completed: 100, failed: 3, delayed: 1 }, ...]

# Lihat detail queue
curl http://localhost:3000/admin/queues/email

# Retry semua job gagal
curl -X POST http://localhost:3000/admin/queues/email/retry-all

# Kosongkan queue
curl -X POST http://localhost:3000/admin/queues/email/empty
```
