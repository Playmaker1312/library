# 343. Cara mengimplementasikan email sending queue untuk pengiriman email asynchronous

## Definisi & Target
Membuat queue khusus untuk pengiriman email agar proses SMTP tidak memblokir response API. Email akan dikirim di background oleh worker.

## Komponen & Nest CLI Command
```bash
nest g mo mail
nest g pr mail/mail
nest g co mail
```

## Kode Implementasi

```typescript
// src/mail/mail.module.ts (Module)
import { Module } from '@nestjs/common';
import { BullModule } from '@nestjs/bull';
import { MailController } from './mail.controller';
import { MailProcessor } from './mail.processor';

@Module({
  imports: [
    BullModule.registerQueue({ name: 'mail' }),
  ],
  controllers: [MailController],
  providers: [MailProcessor],
})
export class MailModule {}
```

```typescript
// src/mail/mail.controller.ts (Controller)
import { Controller, Post, Body } from '@nestjs/common';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';

@Controller('mail')
export class MailController {
  constructor(@InjectQueue('mail') private mailQueue: Queue) {}

  @Post('send')
  async sendMail(
    @Body('to') to: string,
    @Body('subject') subject: string,
    @Body('html') html: string,
  ) {
    const job = await this.mailQueue.add('send-mail', { to, subject, html }, {
      attempts: 3,
      backoff: { type: 'exponential', delay: 2000 },
    });
    return { message: 'Email queued', jobId: job.id };
  }
}
```

```typescript
// src/mail/mail.processor.ts (Processor)
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';
import * as nodemailer from 'nodemailer';

@Processor('mail')
export class MailProcessor {
  private transporter = nodemailer.createTransport({
    host: process.env.SMTP_HOST,
    port: parseInt(process.env.SMTP_PORT, 10),
    auth: {
      user: process.env.SMTP_USER,
      pass: process.env.SMTP_PASS,
    },
  });

  @Process('send-mail')
  async handleSendMail(job: Job<{ to: string; subject: string; html: string }>) {
    const { to, subject, html } = job.data;
    await this.transporter.sendMail({ from: 'noreply@example.com', to, subject, html });
    return { status: 'sent', to };
  }
}
```
