# 338. Cara mengimplementasikan delayed job menggunakan opsi delay

## Definisi & Target
Menjadwalkan job untuk dieksekusi setelah waktu tertentu, misalnya mengirim email reminder 24 jam setelah registrasi.

## Komponen
File yang dimodifikasi: `src/email/email.controller.ts`

## Kode Implementasi

```typescript
// src/email/email.controller.ts (Controller)
import { Controller, Post, Body } from '@nestjs/common';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';

@Controller('email')
export class EmailController {
  constructor(@InjectQueue('email') private emailQueue: Queue) {}

  @Post('send-delayed')
  async sendDelayed(@Body('email') email: string, @Body('hours') hours: number) {
    const delayMs = hours * 60 * 60 * 1000;
    const job = await this.emailQueue.add('send-delayed', { email }, {
      delay: delayMs,
      attempts: 3,
    });
    return {
      message: `Email will be sent in ${hours} hours`,
      jobId: job.id,
      scheduledAt: new Date(Date.now() + delayMs).toISOString(),
    };
  }

  @Post('reminder')
  async sendReminder(@Body('email') email: string) {
    const job = await this.emailQueue.add('send-reminder', { email, type: 'abandoned-cart' }, {
      delay: 24 * 60 * 60 * 1000,
      attempts: 3,
    });
    return { message: 'Reminder scheduled in 24 hours', jobId: job.id };
  }
}
```

```typescript
// src/email/email.processor.ts (Processor)
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';

@Processor('email')
export class EmailProcessor {
  @Process('send-delayed')
  async handleDelayed(job: Job<{ email: string }>) {
    console.log(`Sending delayed email to ${job.data.email}`);
  }

  @Process('send-reminder')
  async handleReminder(job: Job<{ email: string; type: string }>) {
    console.log(`Sending ${job.data.type} reminder to ${job.data.email}`);
  }
}
```
