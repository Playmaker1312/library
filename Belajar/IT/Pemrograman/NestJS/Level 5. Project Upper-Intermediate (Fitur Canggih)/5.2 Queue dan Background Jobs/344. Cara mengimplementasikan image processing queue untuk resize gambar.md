# 344. Cara mengimplementasikan image processing queue untuk resize gambar

## Definisi & Target
Membuat queue untuk memproses gambar secara asynchronous: resize, kompresi, dan generate thumbnail tanpa memblokir upload API.

## Komponen & Nest CLI Command
```bash
npm install sharp
nest g mo media
nest g pr media/image
nest g co media
```

## Kode Implementasi

```typescript
// src/media/media.module.ts (Module)
import { Module } from '@nestjs/common';
import { BullModule } from '@nestjs/bull';
import { MediaController } from './media.controller';
import { ImageProcessor } from './image.processor';

@Module({
  imports: [BullModule.registerQueue({ name: 'image' })],
  controllers: [MediaController],
  providers: [ImageProcessor],
})
export class MediaModule {}
```

```typescript
// src/media/media.controller.ts (Controller)
import { Controller, Post, Body } from '@nestjs/common';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';

@Controller('media')
export class MediaController {
  constructor(@InjectQueue('image') private imageQueue: Queue) {}

  @Post('process')
  async processImage(@Body() body: { filePath: string }) {
    const job = await this.imageQueue.add('resize', {
      filePath: body.filePath,
      sizes: [
        { width: 150, suffix: 'thumb' },
        { width: 800, suffix: 'medium' },
        { width: 1920, suffix: 'large' },
      ],
    });
    return { message: 'Image processing queued', jobId: job.id };
  }
}
```

```typescript
// src/media/image.processor.ts (Processor)
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';
import * as sharp from 'sharp';
import * as path from 'path';
import * as fs from 'fs';

@Processor('image')
export class ImageProcessor {
  @Process('resize')
  async handleResize(job: Job<{ filePath: string; sizes: Array<{ width: number; suffix: string }> }>) {
    const { filePath, sizes } = job.data;
    const dir = path.dirname(filePath);
    const ext = path.extname(filePath);
    const name = path.basename(filePath, ext);

    const results = [];
    for (const size of sizes) {
      const outputPath = path.join(dir, `${name}_${size.suffix}${ext}`);
      await sharp(filePath)
        .resize(size.width)
        .jpeg({ quality: 80 })
        .toFile(outputPath);
      results.push({ size: size.suffix, path: outputPath });
    }

    return { original: filePath, versions: results };
  }
}
```
