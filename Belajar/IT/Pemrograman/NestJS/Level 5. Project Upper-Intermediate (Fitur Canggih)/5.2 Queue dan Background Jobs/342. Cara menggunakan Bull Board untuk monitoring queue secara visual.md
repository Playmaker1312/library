# 342. Cara menggunakan Bull Board untuk monitoring queue secara visual

## Definisi & Target
Bull Board adalah UI dashboard untuk memonitor queue Bull secara real-time, melihat job status, retry job, dan manage queue dari browser.

## Komponen & Perintah Instalasi
```bash
npm install bull-board
```

## Kode Implementasi

```typescript
// src/main.ts (Main)
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { createBullBoard } from 'bull-board';
import { BullAdapter } from 'bull-board/bullAdapter';
import { ExpressAdapter } from 'bull-board/express';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  const serverAdapter = new ExpressAdapter();
  serverAdapter.setBasePath('/admin/queues');

  const { addQueue, removeQueue, setQueues, replaceQueues } = createBullBoard({
    queues: [],
    serverAdapter,
  });

  await app.listen(3000);
}
bootstrap();
```

```typescript
// src/bull-board/bull-board.module.ts (Module)
import { Module, DynamicModule } from '@nestjs/common';
import { createBullBoard } from 'bull-board';
import { BullAdapter } from 'bull-board/bullAdapter';
import { ExpressAdapter } from 'bull-board/express';

@Module({})
export class BullBoardModule {
  static register(queueNames: string[]): DynamicModule {
    const serverAdapter = new ExpressAdapter();
    serverAdapter.setBasePath('/admin/queues');

    const bullBoard = createBullBoard({
      queues: queueNames.map((name) => new BullAdapter(new Bull(name))),
      serverAdapter,
    });

    return {
      module: BullBoardModule,
      providers: [
        { provide: 'BULL_BOARD', useValue: bullBoard },
        { provide: 'SERVER_ADAPTER', useValue: serverAdapter },
      ],
      exports: ['SERVER_ADAPTER'],
    };
  }
}
```

## Cara Integrasi & Pengujian
```bash
npm install bull-board express

# Akses dashboard di browser
open http://localhost:3000/admin/queues
# Lihat: Job list, status, retry button, data inspection
```

Bull Board menyediakan UI dengan fitur:
- Lihat semua job (waiting, active, completed, failed, delayed)
- Inspect job data dan stack trace error
- Retry job yang gagal
- Hapus job
- Real-time update via WebSocket
