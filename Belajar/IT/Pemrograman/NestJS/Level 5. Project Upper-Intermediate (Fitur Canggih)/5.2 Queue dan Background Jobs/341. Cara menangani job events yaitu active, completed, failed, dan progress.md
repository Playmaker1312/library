# 341. Cara menangani job events yaitu active, completed, failed, dan progress

## Definisi & Target
Mendengarkan lifecycle events dari Bull queue seperti `active`, `completed`, `failed`, dan `progress` untuk logging, notifikasi, atau tindakan lanjutan.

## Komponen
File yang dimodifikasi: `src/email/email.processor.ts`

## Kode Implementasi

```typescript
// src/email/email.listener.ts (Listener)
import { Injectable, Logger } from '@nestjs/common';
import { OnQueueActive, OnQueueCompleted, OnQueueFailed, OnQueueProgress } from '@nestjs/bull';
import { Job } from 'bull';

@Injectable()
export class EmailQueueListener {
  private readonly logger = new Logger(EmailQueueListener.name);

  @OnQueueActive()
  onActive(job: Job) {
    this.logger.log(`[Active] Job ${job.id} - ${job.name} started`);
  }

  @OnQueueCompleted()
  onCompleted(job: Job, result: any) {
    this.logger.log(`[Completed] Job ${job.id} - ${job.name} done: ${JSON.stringify(result)}`);
  }

  @OnQueueFailed()
  onFailed(job: Job, error: Error) {
    this.logger.error(`[Failed] Job ${job.id} - ${job.name}: ${error.message}`);
  }

  @OnQueueProgress()
  onProgress(job: Job, progress: number) {
    this.logger.log(`[Progress] Job ${job.id} - ${progress}% complete`);
  }
}
```

```typescript
// src/email/email.processor.ts (Processor)
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';

@Processor('email')
export class EmailProcessor {
  @Process('send-progress')
  async handleWithProgress(job: Job<{ count: number }>) {
    const total = job.data.count || 10;
    for (let i = 1; i <= total; i++) {
      await new Promise((resolve) => setTimeout(resolve, 500));
      await job.progress(Math.round((i / total) * 100));
    }
    return { processed: total };
  }
}
```

```typescript
// src/email/email.module.ts (Module)
import { Module } from '@nestjs/common';
import { BullModule } from '@nestjs/bull';
import { EmailProcessor } from './email.processor';
import { EmailQueueListener } from './email.listener';

@Module({
  imports: [BullModule.registerQueue({ name: 'email' })],
  providers: [EmailProcessor, EmailQueueListener],
})
export class EmailModule {}
```

## Event Reference
| Decorator | Event | Description |
|-----------|-------|-------------|
| `@OnQueueActive` | `active` | Job mulai diproses |
| `@OnQueueCompleted` | `completed` | Job selesai sukses |
| `@OnQueueFailed` | `failed` | Job gagal |
| `@OnQueueProgress` | `progress` | Progress job diupdate |
| `@OnQueueWaiting` | `waiting` | Job masuk antrian |
| `@OnQueueStalled` | `stalled` | Job dianggap stalled |
