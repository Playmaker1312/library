# 349. Cara menangani dead letter queue untuk job yang gagal

## Definisi & Target
Dead Letter Queue (DLQ) adalah queue khusus yang menampung job yang gagal setelah semua percobaan retry habis, untuk diinspeksi dan diproses ulang secara manual atau otomatis.

## Komponen & Nest CLI Command
```bash
nest g mo dlq
nest g pr dlq/dlq
```

## Kode Implementasi

```typescript
// src/dlq/dlq.module.ts (Module)
import { Module } from '@nestjs/common';
import { BullModule } from '@nestjs/bull';
import { DlqProcessor } from './dlq.processor';
import { EmailModule } from '../email/email.module';

@Module({
  imports: [
    BullModule.registerQueue({ name: 'dead-letter' }),
    EmailModule,
  ],
  providers: [DlqProcessor],
})
export class DlqModule {}
```

```typescript
// src/email/email.processor.ts (Processor with DLQ)
import { Processor, Process, InjectQueue } from '@nestjs/bull';
import { Job, Queue } from 'bull';

@Processor('email')
export class EmailProcessor {
  constructor(@InjectQueue('dead-letter') private dlq: Queue) {}

  @Process('send-mail')
  async handleSendMail(job: Job<{ to: string; subject: string }>) {
    try {
      const success = await this.deliverEmail(job.data.to, job.data.subject);
      if (!success) throw new Error('Delivery failed');
    } catch (error) {
      if (job.attemptsMade >= job.opts.attempts) {
        await this.dlq.add('failed-email', {
          originalJobId: job.id,
          data: job.data,
          error: error.message,
          failedAt: new Date().toISOString(),
        });
      }
      throw error;
    }
  }

  private async deliverEmail(to: string, subject: string): Promise<boolean> {
    return false;
  }
}
```

```typescript
// src/dlq/dlq.processor.ts (Processor for DLQ)
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';

@Processor('dead-letter')
export class DlqProcessor {
  @Process('failed-email')
  async handleFailedEmail(job: Job) {
    console.error('Processing dead letter job:', {
      originalJobId: job.data.originalJobId,
      error: job.data.error,
      failedAt: job.data.failedAt,
    });

    await this.notifyAdmin(job.data);
    return { dlqProcessed: true };
  }

  @Process('retry-dead-letter')
  async retryDeadLetter(job: Job) {
    console.log(`Retrying dead letter job ${job.data.originalJobId}`);
    return { retried: true };
  }

  private async notifyAdmin(data: any) {
    console.log(`[ADMIN ALERT] Job ${data.originalJobId} failed permanently`);
  }
}
```
