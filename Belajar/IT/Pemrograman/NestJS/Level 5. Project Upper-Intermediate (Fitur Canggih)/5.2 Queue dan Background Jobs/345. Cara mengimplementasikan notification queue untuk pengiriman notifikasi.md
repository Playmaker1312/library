# 345. Cara mengimplementasikan notification queue untuk pengiriman notifikasi

## Definisi & Target
Membuat queue untuk mengirim notifikasi ke berbagai channel (in-app, push, email) secara asynchronous dan terstruktur.

## Komponen & Nest CLI Command
```bash
nest g mo notification
nest g pr notification/notification
nest g co notification
```

## Kode Implementasi

```typescript
// src/notification/notification.module.ts (Module)
import { Module } from '@nestjs/common';
import { BullModule } from '@nestjs/bull';
import { NotificationController } from './notification.controller';
import { NotificationProcessor } from './notification.processor';

@Module({
  imports: [BullModule.registerQueue({ name: 'notification' })],
  controllers: [NotificationController],
  providers: [NotificationProcessor],
})
export class NotificationModule {}
```

```typescript
// src/notification/notification.controller.ts (Controller)
import { Controller, Post, Body } from '@nestjs/common';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';

@Controller('notifications')
export class NotificationController {
  constructor(@InjectQueue('notification') private notifQueue: Queue) {}

  @Post('send')
  async sendNotification(@Body() body: { userId: string; title: string; message: string; channels: string[] }) {
    const job = await this.notifQueue.add('send-notification', body, {
      priority: body.channels.includes('sms') ? 1 : 5,
      attempts: 3,
    });
    return { message: 'Notification queued', jobId: job.id };
  }
}
```

```typescript
// src/notification/notification.processor.ts (Processor)
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';

interface NotificationPayload {
  userId: string;
  title: string;
  message: string;
  channels: string[];
}

@Processor('notification')
export class NotificationProcessor {
  @Process('send-notification')
  async handleNotification(job: Job<NotificationPayload>) {
    const { userId, title, message, channels } = job.data;

    for (const channel of channels) {
      switch (channel) {
        case 'in-app':
          await this.saveInAppNotification(userId, title, message);
          break;
        case 'push':
          await this.sendPushNotification(userId, title, message);
          break;
        case 'email':
          await this.sendEmailNotification(userId, title, message);
          break;
        case 'sms':
          await this.sendSmsNotification(userId, message);
          break;
      }
    }
    return { userId, channels };
  }

  private async saveInAppNotification(userId: string, title: string, message: string) {
    console.log(`[In-App] ${userId}: ${title} - ${message}`);
  }

  private async sendPushNotification(userId: string, title: string, message: string) {
    console.log(`[Push] ${userId}: ${title}`);
  }

  private async sendEmailNotification(userId: string, title: string, message: string) {
    console.log(`[Email] ${userId}: ${title}`);
  }

  private async sendSmsNotification(userId: string, message: string) {
    console.log(`[SMS] ${userId}: ${message}`);
  }
}
```
