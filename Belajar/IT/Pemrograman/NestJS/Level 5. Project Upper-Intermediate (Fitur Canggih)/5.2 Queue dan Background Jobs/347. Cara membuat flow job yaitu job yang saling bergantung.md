# 347. Cara membuat flow job yaitu job yang saling bergantung

## Definisi & Target
Flow job memungkinkan job berjalan secara berantai, dimana job berikutnya baru dijalankan setelah job sebelumnya selesai. Berguna untuk pipeline processing.

## Komponen & Nest CLI Command
```bash
nest g mo order
nest g pr order/order
nest g co order
```

## Kode Implementasi

```typescript
// src/order/order.module.ts (Module)
import { Module } from '@nestjs/common';
import { BullModule } from '@nestjs/bull';
import { OrderController } from './order.controller';
import { OrderProcessor } from './order.processor';

@Module({
  imports: [BullModule.registerQueue({ name: 'order' })],
  controllers: [OrderController],
  providers: [OrderProcessor],
})
export class OrderModule {}
```

```typescript
// src/order/order.processor.ts (Processor)
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';

@Processor('order')
export class OrderProcessor {
  @Process('validate')
  async handleValidate(job: Job<{ orderId: string }>) {
    console.log(`[1/3] Validating order ${job.data.orderId}`);
    await new Promise((resolve) => setTimeout(resolve, 1000));
    return { validated: true };
  }

  @Process('charge')
  async handleCharge(job: Job<{ orderId: string }>) {
    console.log(`[2/3] Charging payment for order ${job.data.orderId}`);
    await new Promise((resolve) => setTimeout(resolve, 2000));
    return { charged: true, transactionId: `TXN-${Date.now()}` };
  }

  @Process('notify')
  async handleNotify(job: Job<{ orderId: string; transactionId: string }>) {
    console.log(`[3/3] Sending notification for order ${job.data.orderId}`);
    return { notified: true };
  }
}
```

```typescript
// src/order/order.controller.ts (Controller)
import { Controller, Post, Body } from '@nestjs/common';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';

@Controller('order')
export class OrderController {
  constructor(@InjectQueue('order') private orderQueue: Queue) {}

  @Post('checkout')
  async checkout(@Body('orderId') orderId: string) {
    const validateJob = await this.orderQueue.add('validate', { orderId });

    validateJob.finished().then(async () => {
      const chargeJob = await this.orderQueue.add('charge', { orderId });

      chargeJob.finished().then(async (result) => {
        await this.orderQueue.add('notify', {
          orderId,
          transactionId: result.transactionId,
        });
      });
    });

    return { message: 'Order processing started', orderId };
  }
}
```

```bash
curl -X POST http://localhost:3000/order/checkout -H "Content-Type: application/json" -d '{"orderId":"ORD-001"}'
```
