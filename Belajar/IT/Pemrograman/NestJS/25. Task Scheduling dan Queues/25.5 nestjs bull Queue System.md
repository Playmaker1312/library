# 25. Task Scheduling dan Queues

## 25.5. nestjs/bull: Queue System

**Definisi Singkat:**
`@nestjs/bull` adalah modul Queue untuk NestJS yang berbasis library `Bull` (menggunakan Redis). Bull memungkinkan pengelolaan background job secara andal dengan fitur seperti retry, delay, priority, dan concurrency.

**Analogi:**
Seperti antrean di bank - nasabah (jobs) menunggu di antrean, teller (workers) memproses satu per satu, ada nasabah prioritas (VIP), dan jika teller gagal, nasabah dicoba lagi (retry).

**Poin-Poin Penting:**
- Package: `@nestjs/bull` + `bull` + `ioredis`
- Membutuhkan Redis sebagai backend
- Konsep: Queue, Job, Producer, Consumer, Worker
- Fitur: retry, delay, priority, rate limiting, concurrency
- Bull Board untuk monitoring dashboard
- Mendukung job events dan lifecycle hooks

// bull.module.ts
import { Module } from '@nestjs/common';
import { BullModule } from '@nestjs/bull';
import { ConfigModule, ConfigService } from '@nestjs/config';

@Module({
  imports: [
    // Konfigurasi dasar Bull
    BullModule.forRoot({
      redis: {
        host: 'localhost',
        port: 6379,
        // password: 'your-redis-password',
        // db: 0,
      },
    }),

    // Atau menggunakan async configuration
    BullModule.forRootAsync({
      imports: [ConfigModule],
      useFactory: (configService: ConfigService) => ({
        redis: {
          host: configService.get('REDIS_HOST', 'localhost'),
          port: configService.get('REDIS_PORT', 6379),
          password: configService.get('REDIS_PASSWORD'),
        },
        defaultJobOptions: {
          removeOnComplete: true,
          removeOnFail: false,
          attempts: 3,
          backoff: {
            type: 'exponential',
            delay: 1000,
          },
        },
      }),
      inject: [ConfigService],
    }),

    // Mendaftarkan Queue
    BullModule.registerQueue({
      name: 'email',
      defaultJobOptions: {
        attempts: 5,
        backoff: {
          type: 'fixed',
          delay: 5000,
        },
      },
    }),

    BullModule.registerQueue({
      name: 'image-processing',
      defaultJobOptions: {
        priority: 10,
        removeOnComplete: true,
      },
    }),

    BullModule.registerQueue({
      name: 'notifications',
    }),
  ],
  exports: [BullModule],
})
export class AppBullModule {}

// email.processor.ts - Consumer (Worker)
import { Processor, Process } from '@nestjs/bull';
import { Job } from 'bull';
import { Injectable, Logger } from '@nestjs/common';
import { MailerService } from '@nestjs-modules/mailer';

@Injectable()
@Processor('email')
export class EmailProcessor {
  private readonly logger = new Logger(EmailProcessor.name);

  constructor(private mailerService: MailerService) {}

  // Handler untuk job dengan nama 'send-welcome'
  @Process('send-welcome')
  async handleSendWelcome(job: Job) {
    this.logger.log(`Processing send-welcome job: ${job.id}`);
    this.logger.debug(`Job data: ${JSON.stringify(job.data)}`);

    const { email, name } = job.data;

    try {
      await this.mailerService.sendMail({
        to: email,
        subject: 'Welcome to Our Platform!',
        template: './welcome',
        context: { name },
      });

      this.logger.log(`Welcome email sent to ${email}`);

      // Update progress
      await job.progress(100);

      return { success: true, message: 'Email sent successfully' };
    } catch (error) {
      this.logger.error(`Failed to send email to ${email}: ${error.message}`);
      throw error; // Throw untuk trigger retry
    }
  }

  // Handler untuk job dengan nama 'reset-password'
  @Process('reset-password')
  async handleResetPassword(job: Job) {
    this.logger.log(`Processing reset-password job: ${job.id}`);

    const { email, resetToken } = job.data;

    // Simulasi progress
    await job.progress(50);

    await this.mailerService.sendMail({
      to: email,
      subject: 'Reset Your Password',
      template: './reset-password',
      context: { resetToken },
    });

    await job.progress(100);

    return { success: true };
  }

  // Handler default (untuk job tanpa nama spesifik)
  @Process()
  async handleGenericEmail(job: Job) {
    this.logger.log(`Processing generic email job: ${job.id}`);
    
    const { email, subject, body } = job.data;
    
    await this.mailerService.sendMail({
      to: email,
      subject,
      text: body,
    });

    return { success: true };
  }
}

// email.service.ts - Producer (Menambahkan job ke queue)
import { Injectable, Inject } from '@nestjs/common';
import { Queue } from 'bull';
import { InjectQueue } from '@nestjs/bull';

@Injectable()
export class EmailQueueService {
  constructor(
    @InjectQueue('email') private readonly emailQueue: Queue,
  ) {}

  // Kirim email welcome dengan delay
  async sendWelcomeEmail(email: string, name: string) {
    const job = await this.emailQueue.add(
      'send-welcome',
      { email, name },
      {
        delay: 5000, // Kirim setelah 5 detik
        priority: 1,  // Prioritas tinggi
        attempts: 3,  // Coba 3 kali jika gagal
        backoff: {
          type: 'exponential',
          delay: 1000,
        },
      },
    );

    return { jobId: job.id };
  }

  // Kirim email reset password
  async sendResetPasswordEmail(email: string, resetToken: string) {
    const job = await this.emailQueue.add(
      'reset-password',
      { email, resetToken },
      {
        priority: 1, // Prioritas tinggi
        attempts: 5,
        removeOnComplete: true,
      },
    );

    return { jobId: job.id };
  }

  // Kirim email newsletter massal
  async sendNewsletter(emails: string[], newsletterData: any) {
    const jobs = [];
    
    for (const email of emails) {
      const job = await this.emailQueue.add(
        {
          email,
          subject: newsletterData.subject,
          body: newsletterData.body,
        },
        {
          priority: 10, // Prioritas rendah
          attempts: 2,
          removeOnComplete: true,
        },
      );
      jobs.push(job.id);
    }

    return { jobIds: jobs };
  }

  // Kirim email terjadwal (delay)
  async scheduleEmail(email: string, subject: string, body: string, sendAt: Date) {
    const delay = sendAt.getTime() - Date.now();
    
    if (delay <= 0) {
      throw new Error('Scheduled time must be in the future');
    }

    const job = await this.emailQueue.add(
      { email, subject, body },
      {
        delay,
        jobId: `scheduled_${Date.now()}`,
      },
    );

    return { jobId: job.id, scheduledAt: sendAt };
  }

  // Mendapatkan status job
  async getJobStatus(jobId: string) {
    const job = await this.emailQueue.getJob(jobId);
    
    if (!job) {
      return { error: 'Job not found' };
    }

    const state = await job.getState();
    const progress = job.progress();

    return {
      jobId: job.id,
      state,
      progress,
      data: job.data,
      attemptsMade: job.attemptsMade,
      failedReason: job.failedReason,
      processedOn: job.processedOn,
      finishedOn: job.finishedOn,
    };
  }

  // Membatalkan job
  async cancelJob(jobId: string) {
    const job = await this.emailQueue.getJob(jobId);
    
    if (job) {
      await job.remove();
      return { success: true, message: `Job ${jobId} cancelled` };
    }

    return { error: 'Job not found' };
  }
}

// image-processing.processor.ts - Contoh lain dengan concurrency
import { Processor, Process, OnQueueActive, OnQueueCompleted, OnQueueFailed } from '@nestjs/bull';
import { Job } from 'bull';
import { Logger } from '@nestjs/common';

@Processor({
  name: 'image-processing',
  concurrency: 5, // 5 worker sekaligus
})
export class ImageProcessor {
  private readonly logger = new Logger(ImageProcessor.name);

  @Process('resize')
  async handleResize(job: Job) {
    this.logger.log(`Resizing image: ${job.data.imageId}`);
    
    const { imageId, sizes } = job.data;
    
    // Simulasi progress
    await job.progress(10);
    
    // Logic resize image
    await new Promise(resolve => setTimeout(resolve, 2000));
    
    await job.progress(50);
    
    // Logic upload ke storage
    await new Promise(resolve => setTimeout(resolve, 1000));
    
    await job.progress(100);

    return { success: true, sizes };
  }

  @Process('watermark')
  async handleWatermark(job: Job) {
    this.logger.log(`Adding watermark to image: ${job.data.imageId}`);
    
    await new Promise(resolve => setTimeout(resolve, 1000));
    
    return { success: true };
  }

  // Event listener
  @OnQueueActive()
  onActive(job: Job) {
    this.logger.log(
      `Processing job ${job.id} of type ${job.name} with data ${JSON.stringify(job.data)}`,
    );
  }

  @OnQueueCompleted()
  onCompleted(job: Job, result: any) {
    this.logger.log(
      `Job ${job.id} completed with result ${JSON.stringify(result)}`,
    );
  }

  @OnQueueFailed()
  onFailed(job: Job, err: Error) {
    this.logger.error(
      `Job ${job.id} failed with error: ${err.message}`,
    );
  }
}

// image.service.ts - Producer untuk image processing
import { Injectable } from '@nestjs/common';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';

@Injectable()
export class ImageService {
  constructor(
    @InjectQueue('image-processing') private readonly imageQueue: Queue,
  ) {}

  async addResizeJob(imageId: string, sizes: string[]) {
    const job = await this.imageQueue.add(
      'resize',
      { imageId, sizes },
      {
        priority: sizes.includes('thumbnail') ? 1 : 5, // Thumbnail lebih penting
        attempts: 2,
      },
    );

    return { jobId: job.id };
  }

  async addWatermarkJob(imageId: string, watermarkText: string) {
    const job = await this.imageQueue.add(
      'watermark',
      { imageId, watermarkText },
      {
        priority: 10,
      },
    );

    return { jobId: job.id };
  }
}

**Kasus Penggunaan Nyata:**
1. **Email Sending**: Mengirim email welcome, reset password, newsletter di background agar user tidak menunggu
2. **Image Processing**: Resize, compress, watermark gambar setelah di-upload user
3. **PDF/Report Generation**: Generate laporan PDF yang memakan waktu di background
4. **CSV Import/Export**: Memproses file CSV besar (import data produk, export laporan)
5. **Webhook Delivery**: Mengirim webhook ke third-party dengan retry logic jika gagal
6. **Push Notifications**: Mengirim notifikasi massal ke ribuan device mobile
7. **Database Cleanup**: Membersihkan data lama secara bertahap tanpa membebani DB
8. **Payment Processing**: Memproses pembayaran secara async dengan retry jika gateway error
