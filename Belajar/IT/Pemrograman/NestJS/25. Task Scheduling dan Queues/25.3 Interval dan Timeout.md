# 25. Task Scheduling dan Queues

## 25.3. Interval dan Timeout

**Definisi Singkat:**
`@Interval()` dan `@Timeout()` adalah decorator dari `@nestjs/schedule` untuk menjalankan tugas secara berulang dengan interval tertentu atau satu kali setelah delay tertentu, lebih sederhana daripada cron untuk kasus-kasus sederhana.

**Analogi:**
- `@Interval()` seperti pomodoro timer: berbunyi setiap 25 menit untuk beristirahat
- `@Timeout()` seperti alarm once-off: set alarm untuk 10 menit lagi untuk mengingatkan sesuatu

**Poin-Poin Penting:**
- `@Interval(milliseconds)`: berjalan berulang setiap X ms
- `@Timeout(milliseconds)`: berjalan SATU KALI setelah X ms
- Keduanya dapat diberi nama untuk manajemen via `SchedulerRegistry`
- Berdasarkan JavaScript `setInterval` dan `setTimeout`
- Cocok untuk tugas sederhana tanpa jadwal kompleks

**Contoh Kode:**
```typescript
import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { Interval, Timeout, SchedulerRegistry } from '@nestjs/schedule';

@Injectable()
export class IntervalTimeoutService implements OnModuleInit {
  private readonly logger = new Logger(IntervalTimeoutService.name);
  private counter = 0;

  constructor(private schedulerRegistry: SchedulerRegistry) {}

  onModuleInit() {
    this.logger.log('IntervalTimeoutService initialized');
  }

  // ========== @INTERVAL DECORATOR ==========

  // Berjalan setiap 1 detik (1000ms)
  @Interval(1000)
  handleEverySecond() {
    this.counter++;
    this.logger.debug(`Interval tick #${this.counter}`);
  }

  // Berjalan setiap 5 detik dengan nama
  @Interval('health_check_interval', 5000)
  handleHealthCheck() {
    this.logger.log('Health check running every 5 seconds');
    this.checkDatabaseConnection();
    this.checkRedisConnection();
  }

  // Berjalan setiap 30 detik untuk sync
  @Interval('data_sync_interval', 30000)
  handleDataSync() {
    this.logger.log('Syncing data from external API...');
    this.syncFromExternalAPI();
  }

  // Berjalan setiap 1 menit untuk cleanup
  @Interval('cache_cleanup_interval', 60000)
  handleCacheCleanup() {
    this.logger.log('Cleaning up stale cache entries...');
    this.cleanupStaleCache();
  }

  // ========== @TIMEOUT DECORATOR ==========

  // Berjalan SATU KALI setelah 5 detik aplikasi start
  @Timeout(5000)
  handleStartupTimeout() {
    this.logger.log('Timeout: 5 seconds after startup');
    this.performInitialCacheWarmup();
  }

  // Berjalan sekali dengan nama
  @Timeout('delayed_notification', 10000)
  handleDelayedNotification() {
    this.logger.log('Timeout: Sending delayed notification');
    this.sendWelcomeEmailToNewUsers();
  }

  // Berjalan sekali setelah 30 detik untuk mengecek konfigurasi
  @Timeout('config_validation', 30000)
  handleConfigValidation() {
    this.logger.log('Timeout: Validating configuration');
    this.validateExternalServiceConfigs();
  }

  // ========== DINAMIS INTERVAL & TIMEOUT ==========

  // Membuat interval secara dinamis
  createDynamicInterval(name: string, milliseconds: number) {
    const interval = setInterval(() => {
      this.logger.log(`Dynamic interval '${name}' ticking every ${milliseconds}ms`);
    }, milliseconds);

    this.schedulerRegistry.addInterval(name, interval);
    this.logger.log(`Dynamic interval '${name}' created`);
  }

  // Membuat timeout secara dinamis
  createDynamicTimeout(name: string, milliseconds: number, callback: () => void) {
    const timeout = setTimeout(() => {
      this.logger.log(`Dynamic timeout '${name}' executed after ${milliseconds}ms`);
      callback();
    }, milliseconds);

    this.schedulerRegistry.addTimeout(name, timeout);
    this.logger.log(`Dynamic timeout '${name}' created`);
  }

  // Menghapus interval
  deleteInterval(name: string) {
    const interval = this.schedulerRegistry.getInterval(name);
    clearInterval(interval);
    this.schedulerRegistry.deleteInterval(name);
    this.logger.log(`Interval '${name}' deleted`);
  }

  // Menghapus timeout
  deleteTimeout(name: string) {
    const timeout = this.schedulerRegistry.getTimeout(name);
    clearTimeout(timeout);
    this.schedulerRegistry.deleteTimeout(name);
    this.logger.log(`Timeout '${name}' deleted`);
  }

  // Mendapatkan semua interval
  getAllIntervals() {
    const intervals = this.schedulerRegistry.getIntervals();
    return intervals.map(name => ({ name }));
  }

  // Mendapatkan semua timeout
  getAllTimeouts() {
    const timeouts = this.schedulerRegistry.getTimeouts();
    return timeouts.map(name => ({ name }));
  }

  // ========== CONTOH PENGGUNAAN REALISTIS ==========

  // Real-time metrics setiap 10 detik
  @Interval('metrics_collection', 10000)
  collectMetrics() {
    const cpuUsage = this.getCpuUsage();
    const memoryUsage = this.getMemoryUsage();
    const activeConnections = this.getActiveConnections();

    this.logger.debug('Metrics collected', {
      cpu: cpuUsage,
      memory: memoryUsage,
      connections: activeConnections,
    });

    this.sendMetricsToMonitoringService({
      cpu: cpuUsage,
      memory: memoryUsage,
      connections: activeConnections,
      timestamp: Date.now(),
    });
  }

  // Token refresh setiap 45 menit
  @Interval('token_refresh', 45 * 60 * 1000)
  refreshAuthTokens() {
    this.logger.log('Refreshing authentication tokens...');
    this.refreshExternalServiceTokens();
  }

  // Notifikasi timeout untuk order yang belum dibayar
  createOrderTimeout(orderId: string, minutes: number = 30) {
    const timeoutName = `order_timeout_${orderId}`;
    const milliseconds = minutes * 60 * 1000;

    this.createDynamicTimeout(timeoutName, milliseconds, () => {
      this.logger.log(`Order ${orderId} expired - cancelling...`);
      this.cancelExpiredOrder(orderId);
      this.sendOrderCancelledNotification(orderId);
    });
  }

  // Cancel order timeout jika pembayaran berhasil
  cancelOrderTimeout(orderId: string) {
    const timeoutName = `order_timeout_${orderId}`;
    try {
      this.deleteTimeout(timeoutName);
      this.logger.log(`Order ${orderId} timeout cancelled - payment received`);
    } catch (error) {
      this.logger.warn(`Timeout for order ${orderId} not found or already executed`);
    }
  }

  // ========== PRIVATE HELPER METHODS ==========

  private checkDatabaseConnection() {

  }

  private syncFromExternalAPI() {

  private cleanupStaleCache() {

  }

  private performInitialCacheWarmup() {

  }

  private sendWelcomeEmailToNewUsers() {

  }

  private validateExternalServiceConfigs() {

  }

    return Math.random() * 100;
  }

  }

// Controller untuk Interval & Timeout
import { Controller, Get, Post, Delete, Param, Body } from '@nestjs/common';
import { IntervalTimeoutService } from './interval-timeout.service';

@Controller('scheduler')
export class SchedulerController {
  constructor(private readonly schedulerService: IntervalTimeoutService) {}

  // Interval endpoints
  @Get('intervals')
  getIntervals() {
    return this.schedulerService.getAllIntervals();
  }

  @Post('intervals')
  createInterval(
    @Body('name') name: string,
    @Body('ms') ms: number,
  ) {
    this.schedulerService.createDynamicInterval(name, ms);
    return { message: `Interval ${name} created` };
  }

  @Delete('intervals/:name')
  deleteInterval(@Param('name') name: string) {
    this.schedulerService.deleteInterval(name);
    return { message: `Interval ${name} deleted` };
  }

  // Timeout endpoints
  @Get('timeouts')
  getTimeouts() {
    return this.schedulerService.getAllTimeouts();
  }

  @Post('order-timeout')
  createOrderTimeout(
    @Body('orderId') orderId: string,
    @Body('minutes') minutes: number,
  ) {
    this.schedulerService.createOrderTimeout(orderId, minutes);
    return { message: `Order timeout for ${orderId} created` };
  }

  @Post('cancel-order-timeout/:orderId')
  cancelOrderTimeout(@Param('orderId') orderId: string) {
    this.schedulerService.cancelOrderTimeout(orderId);
    return { message: `Order timeout for ${orderId} cancelled` };
  }
}
```

**Kasus Penggunaan Nyata:**
1. **Order Timeout**: Membuat timeout 30 menit untuk order yang belum dibayar, jika tidak dibayar batalkan otomatis. Jika pembayaran masuk sebelum timeout, cancel timeoutnya.
2. **Real-time Metrics**: Mengumpulkan metrik CPU, memory, koneksi setiap 10 detik untuk monitoring dashboard
3. **Token Refresh**: Me-refresh token autentikasi service eksternal setiap 45 menit sebelum expired
4. **Cache Cleanup**: Membersihkan cache yang sudah expired setiap 1 jam
5. **Delayed Notifications**: Mengirim email selamat datang 5 menit setelah user mendaftar (memberi waktu user untuk verifikasi email dulu)
6. **Connection Pool Health Check**: Mengecek health database connection pool setiap 30 detik
