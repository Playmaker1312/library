# 25. Task Scheduling dan Queues

## 25.4. Dynamic Cron Jobs

**Definisi Singkat:**
Dynamic Cron Jobs adalah cron job yang dibuat, dimodifikasi, atau dihapus secara runtime (saat aplikasi berjalan), tidak hardcoded dengan decorator. Ini memungkinkan penjadwalan yang fleksibel berdasarkan konfigurasi atau input user.

**Analogi:**
Seperti TV yang bisa menyetel acara favorit secara manual - kamu bisa menambah, menghapus, atau mengubah jam tayang kapan saja tanpa harus mematikan TV.

**Poin-Poin Penting:**
- Menggunakan `SchedulerRegistry` untuk manajemen
- Membuat `CronJob` dari package `cron` secara manual
- Dapat dibuat berdasarkan input user/database
- Dapat dimodifikasi (ubah cron expression) saat runtime
- Cocok untuk aplikasi multi-tenant atau user-defined schedules

**Contoh Kode:**
```typescript
import { Injectable, Logger, OnModuleInit } from '@nestjs/common';
import { SchedulerRegistry } from '@nestjs/schedule';
import { CronJob, CronTime } from 'cron';

@Injectable()
export class DynamicCronService implements OnModuleInit {
  private readonly logger = new Logger(DynamicCronService.name);

  constructor(private schedulerRegistry: SchedulerRegistry) {}

  async onModuleInit() {
    // Load cron jobs dari database saat startup
    await this.loadScheduledJobsFromDatabase();
    this.logger.log('Dynamic Cron Service initialized');
  }

  // ========== MEMBUAT CRON JOB SECARA DINAMIS ==========

  async createCronJob(
    name: string,
    cronExpression: string,
    callback: () => void,
    timeZone: string = 'Asia/Jakarta',
  ): Promise<void> {
    // Cek apakah job dengan nama ini sudah ada
    if (this.schedulerRegistry.doesExist('cron', name)) {
      this.logger.warn(`Cron job '${name}' already exists`);
      return;
    }

    // Buat CronJob baru
    const job = new CronJob(
      cronExpression,
      () => {
        this.logger.log(`Executing dynamic cron job: ${name}`);
        callback();
      },
      null, // onComplete callback
      true, // autoStart
      timeZone,
    );

    // Daftarkan ke SchedulerRegistry
    this.schedulerRegistry.addCronJob(name, job);

    // Simpan ke database untuk persistence
    await this.saveJobToDatabase({
      name,
      cronExpression,
      timeZone,
      active: true,
    });

    this.logger.log(`Dynamic cron job '${name}' created with schedule: ${cronExpression}`);
  }

  // ========== MEMUAT CRON JOB DARI DATABASE ==========

  private async loadScheduledJobsFromDatabase(): Promise<void> {
    const jobs = await this.getJobsFromDatabase();

    for (const job of jobs) {
      if (job.active) {
        await this.createCronJob(
          job.name,
          job.cronExpression,
          () => this.executeJob(job.name, job.type, job.config),
          job.timeZone,
        );
      }
    }

    this.logger.log(`Loaded ${jobs.length} cron jobs from database`);
  }

  // ========== MENGUPDATE CRON JOB ==========

  async updateCronJob(
    name: string,
    newCronExpression: string,
    newTimeZone?: string,
  ): Promise<void> {
    try {
      const job = this.schedulerRegistry.getCronJob(name);
      
      // Hentikan job sementara
      job.stop();

      // Buat CronTime baru
      const newTime = new CronTime(newCronExpression, newTimeZone);
      
      // Set waktu baru
      job.setTime(newTime);
      
      // Mulai kembali
      job.start();

      // Update database
      await this.updateJobInDatabase(name, {
        cronExpression: newCronExpression,
        timeZone: newTimeZone,
      });

      this.logger.log(`Cron job '${name}' updated to: ${newCronExpression}`);
    } catch (error) {
      this.logger.error(`Failed to update cron job '${name}': ${error.message}`);
      throw error;
    }
  }

  // ========== MENGHAPUS CRON JOB ==========

  async deleteCronJob(name: string): Promise<void> {
    try {
      const job = this.schedulerRegistry.getCronJob(name);
      job.stop();
      
      this.schedulerRegistry.deleteCronJob(name);
      
      await this.deleteJobFromDatabase(name);
      
      this.logger.log(`Cron job '${name}' deleted`);
    } catch (error) {
      this.logger.error(`Failed to delete cron job '${name}': ${error.message}`);
      throw error;
    }
  }

  // ========== MENGAKTIFKAN/MENONAKTIFKAN ==========

  async toggleCronJob(name: string, active: boolean): Promise<void> {
    const job = this.schedulerRegistry.getCronJob(name);
    
    if (active) {
      job.start();
      this.logger.log(`Cron job '${name}' started`);
    } else {
      job.stop();
      this.logger.log(`Cron job '${name}' stopped`);
    }

    await this.updateJobInDatabase(name, { active });
  }

  // ========== MENJALANKAN SECARA MANUAL ==========

  async triggerCronJob(name: string): Promise<void> {
    const job = this.schedulerRegistry.getCronJob(name);
    
    // Jalankan callbacknya (tanpa menunggu jadwal)
    job.fireOnTick();
    
    this.logger.log(`Cron job '${name}' triggered manually`);
  }

  // ========== MENDAPATKAN STATUS ==========

  getCronJobStatus(name: string): any {
    const job = this.schedulerRegistry.getCronJob(name);
    
    return {
      name,
      running: job.running,
      lastExecution: job.lastDate(),
      nextExecution: job.nextDate(),
    };
  }

  getAllCronJobs(): any[] {
    const jobs = this.schedulerRegistry.getCronJobs();
    const result = [];
    
    jobs.forEach((job, name) => {
      result.push({
        name,
        running: job.running,
        lastExecution: job.lastDate(),
        nextExecution: job.nextDate(),
      });
    });
    
    return result;
  }

  // ========== EXECUTE JOB BERDASARKAN TIPE ==========

  private executeJob(name: string, type: string, config: any): void {
    this.logger.log(`Executing job '${name}' of type '${type}'`);

    switch (type) {
      case 'send_report':
        this.sendReport(config);
        break;
      case 'database_backup':
        this.performDatabaseBackup(config);
        break;
      case 'sync_data':
        this.syncExternalData(config);
        break;
      case 'send_notifications':
        this.sendNotifications(config);
        break;
      default:
        this.logger.warn(`Unknown job type: ${type}`);
    }
  }

  private sendReport(config: any): void {
    this.logger.log(`Sending report to: ${config.recipientEmail}`);
    // Implementasi kirim email
  }

  private performDatabaseBackup(config: any): void {
    this.logger.log(`Backing up database: ${config.databaseName}`);
    // Implementasi backup
  }

  private syncExternalData(config: any): void {
    this.logger.log(`Syncing from: ${config.externalApiUrl}`);
    // Implementasi sync
  }

  private sendNotifications(config: any): void {
    this.logger.log(`Sending notifications to user segment: ${config.segment}`);
    // Implementasi notifikasi
  }

  // ========== DATABASE HELPER METHODS (SIMULASI) ==========

  private async getJobsFromDatabase(): Promise<any[]> {
    return [
      {
        name: 'daily_sales_report',
        cronExpression: '0 0 9 * * *',
        timeZone: 'Asia/Jakarta',
        type: 'send_report',
        config: { recipientEmail: 'admin@company.com' },
        active: true,
      },
      {
        name: 'database_backup',
        cronExpression: '0 0 3 * * *',
        timeZone: 'Asia/Jakarta',
        type: 'database_backup',
        config: { databaseName: 'main_db' },
        active: true,
      },
    ];
  }

  private async saveJobToDatabase(job: any): Promise<void> {
    // Simpan ke database
    this.logger.log(`Job '${job.name}' saved to database`);
  }

  private async updateJobInDatabase(name: string, updates: any): Promise<void> {
    // Update database
    this.logger.log(`Job '${name}' updated in database`);
  }

  private async deleteJobFromDatabase(name: string): Promise<void> {
    // Delete dari database
    this.logger.log(`Job '${name}' deleted from database`);
  }
}

// Controller untuk Dynamic Cron Jobs
import { Controller, Get, Post, Put, Delete, Param, Body } from '@nestjs/common';
import { DynamicCronService } from './dynamic-cron.service';

@Controller('dynamic-cron')
export class DynamicCronController {
  constructor(private readonly dynamicCronService: DynamicCronService) {}

  @Get('jobs')
  getAllJobs() {
    return this.dynamicCronService.getAllCronJobs();
  }

  @Get('jobs/:name')
  getJobStatus(@Param('name') name: string) {
    return this.dynamicCronService.getCronJobStatus(name);
  }

  @Post('jobs')
  async createJob(
    @Body('name') name: string,
    @Body('cronExpression') cronExpression: string,
    @Body('timeZone') timeZone?: string,
    @Body('type') type?: string,
    @Body('config') config?: any,
  ) {
    await this.dynamicCronService.createCronJob(
      name,
      cronExpression,
      () => {
        console.log(`Custom callback for job: ${name}`);
      },
      timeZone,
    );
    return { message: `Job '${name}' created successfully` };
  }

  @Put('jobs/:name')
  async updateJob(
    @Param('name') name: string,
    @Body('cronExpression') cronExpression: string,
    @Body('timeZone') timeZone?: string,
  ) {
    await this.dynamicCronService.updateCronJob(name, cronExpression, timeZone);
    return { message: `Job '${name}' updated successfully` };
  }

  @Delete('jobs/:name')
  async deleteJob(@Param('name') name: string) {
    await this.dynamicCronService.deleteCronJob(name);
    return { message: `Job '${name}' deleted successfully` };
  }

  @Post('jobs/:name/toggle')
  async toggleJob(
    @Param('name') name: string,
    @Body('active') active: boolean,
  ) {
    await this.dynamicCronService.toggleCronJob(name, active);
    return { message: `Job '${name}' ${active ? 'started' : 'stopped'}` };
  }

  @Post('jobs/:name/trigger')
  async triggerJob(@Param('name') name: string) {
    await this.dynamicCronService.triggerCronJob(name);
    return { message: `Job '${name}' triggered manually` };
  }
}
```

**Kasus Penggunaan Nyata:**
1. **Multi-tenant SaaS**: Setiap tenant bisa membuat jadwal backup atau laporan mereka sendiri melalui dashboard
2. **Email Marketing Platform**: User bisa menjadwalkan kapan email campaign dikirim (setiap Senin jam 8 pagi, atau tanggal 1 setiap bulan)
3. **IoT Dashboard**: User bisa menjadwalkan kapan perangkat IoT harus menyala/mati secara otomatis
4. **Reporting System**: Manager bisa membuat jadwal laporan mingguan/bulanan custom sesuai kebutuhan
5. **Social Media Scheduler**: Menjadwalkan postingan Instagram/Twitter pada waktu tertentu yang diatur user
6. **Database Maintenance**: DBA bisa mengubah jadwal maintenance tanpa redeploy aplikasi
