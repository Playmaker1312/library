# 25. Task Scheduling dan Queues

## 25.2. Cron Decorator

**Definisi Singkat:**
`@Cron()` adalah decorator NestJS yang digunakan untuk menandai method agar dijalankan secara otomatis berdasarkan cron expression, dengan opsi konfigurasi seperti nama, timezone, dan disabled.

**Analogi:**
Seperti stiker jadwal yang ditempel di kulkas - menandai kapan tugas harus dilakukan, jam berapa, dan zona waktunya.

**Poin-Poin Penting:**
- Import dari `@nestjs/schedule`
- Parameter: cron expression (string) atau `CronExpression` enum
- Opsi: `name`, `timeZone`, `utcOffset`, `disabled`
- Dapat mengakses `SchedulerRegistry` untuk manajemen runtime
- Cron expression 6 atau 7 field (dengan detik)

**Contoh Kode:**
```typescript
import { Injectable, Logger } from '@nestjs/common';
import { Cron, CronExpression, SchedulerRegistry } from '@nestjs/schedule';
import { CronJob } from 'cron';

@Injectable()
export class CronDecoratorService {
  private readonly logger = new Logger(CronDecoratorService.name);

  constructor(private schedulerRegistry: SchedulerRegistry) {}

  // ========== BASIC CRON DECORATOR ==========

  // Menggunakan string cron expression
  // Setiap detik
  @Cron('* * * * * *')
  handleEverySecond() {
    this.logger.debug('Tick every second');
  }

  // Menggunakan CronExpression enum
  @Cron(CronExpression.EVERY_MINUTE)
  handleEveryMinute() {
    this.logger.log('Tick every minute');
  }

  // Setiap jam
  @Cron(CronExpression.EVERY_HOUR)
  handleEveryHour() {
    this.logger.log('Tick every hour');
  }

  // ========== CRON DENGAN NAMA ==========
  // Memberi nama untuk manajemen melalui SchedulerRegistry

  @Cron('*/5 * * * * *', {
    name: 'health_check',
  })
  handleHealthCheck() {
    this.logger.log('Running health check...');
  }

  // ========== CRON DENGAN TIMEZONE ==========

  @Cron('0 0 9 * * *', {
    name: 'daily_report_jakarta',
    timeZone: 'Asia/Jakarta',
  })
  handleDailyReportJakarta() {
    this.logger.log('Sending daily report - Jakarta Time 9:00 AM');
  }

  @Cron('0 0 9 * * *', {
    name: 'daily_report_singapore',
    timeZone: 'Asia/Singapore',
  })
  handleDailyReportSingapore() {
    this.logger.log('Sending daily report - Singapore Time 9:00 AM');
  }

  // Timezone dengan UTC offset
  @Cron('0 0 12 * * *', {
    name: 'lunch_reminder',
    utcOffset: '+07:00', // UTC+7 sama dengan Jakarta
  })
  handleLunchReminder() {
    this.logger.log('Lunch time reminder (UTC+7 12:00 PM)');
  }

  // ========== CRON YANG BISA DISABLE ==========

  @Cron('*/10 * * * * *', {
    name: 'optional_task',
    disabled: false, // bisa diubah menjadi true untuk menonaktifkan
  })
  handleOptionalTask() {
    this.logger.log('Optional task running');
  }

  // ========== DINAMIS MENGAKTIFKAN/MENONAKTIFKAN CRON ==========

  // Menonaktifkan cron job melalui API/Service
  stopCronJob(jobName: string) {
    const job = this.schedulerRegistry.getCronJob(jobName);
    job.stop();
    this.logger.log(`Cron job '${jobName}' stopped`);
  }

  // Mengaktifkan kembali cron job
  startCronJob(jobName: string) {
    const job = this.schedulerRegistry.getCronJob(jobName);
    job.start();
    this.logger.log(`Cron job '${jobName}' started`);
  }

  // Mendapatkan status cron job
  getCronJobStatus(jobName: string) {
    const job = this.schedulerRegistry.getCronJob(jobName);
    return {
      name: jobName,
      running: job.running,
      lastDate: job.lastDate(),
      nextDate: job.nextDate(),
    };
  }

  // Menjalankan cron job secara manual (trigger)
  triggerCronJob(jobName: string) {
    const job = this.schedulerRegistry.getCronJob(jobName);
    job.fireOnTick();
    this.logger.log(`Cron job '${jobName}' triggered manually`);
  }

  // Mendapatkan semua cron job
  getAllCronJobs() {
    const cronJobs = this.schedulerRegistry.getCronJobs();
    const jobs = [];
    
    cronJobs.forEach((job, name) => {
      jobs.push({
        name,
        running: job.running,
        lastDate: job.lastDate(),
        nextDate: job.nextDate(),
      });
    });
    
    return jobs;
  }

  // ========== CONTOH CRON JOBS REALISTIS ==========

  // Backup database setiap jam 3 pagi
  @Cron('0 0 3 * * *', {
    name: 'database_backup',
    timeZone: 'Asia/Jakarta',
  })
  async performDatabaseBackup() {
    this.logger.log('Starting database backup...');
    
    try {
      // Logic backup database
      await this.backupDatabase();
      this.logger.log('Database backup completed successfully');
    } catch (error) {
      this.logger.error('Database backup failed', error.stack);
      // Kirim alert
      this.sendAlert('Database backup failed');
    }
  }

  // Kirim email notifikasi tagihan setiap hari jam 8 pagi
  @Cron('0 0 8 * * *', {
    name: 'billing_reminder',
    timeZone: 'Asia/Jakarta',
  })
  async sendBillingReminders() {
    this.logger.log('Sending billing reminders...');
    
    const overdueInvoices = await this.getOverdueInvoices();
    
    for (const invoice of overdueInvoices) {
      await this.sendReminderEmail(invoice);
    }
    
    this.logger.log(`Sent ${overdueInvoices.length} billing reminders`);
  }

  // Clear expired sessions setiap jam
  @Cron(CronExpression.EVERY_HOUR, {
    name: 'clear_expired_sessions',
  })
  async clearExpiredSessions() {
    this.logger.log('Clearing expired sessions...');
    
    const deletedCount = await this.deleteExpiredSessionsFromDB();
    
    this.logger.log(`Cleared ${deletedCount} expired sessions`);
  }

  // Private helper methods
  private async backupDatabase() {
    // Implementasi backup
    return new Promise(resolve => setTimeout(resolve, 1000));
  }

  private sendAlert(message: string) {
    this.logger.warn(`ALERT: ${message}`);
  }

  private async getOverdueInvoices() {
    return [];
  }

  private async sendReminderEmail(invoice: any) {
    this.logger.log(`Sending reminder for invoice: ${invoice.id}`);
  }

  private async deleteExpiredSessionsFromDB() {
    return 42;
  }
}

// Controller untuk mengelola Cron Jobs
import { Controller, Get, Post, Param } from '@nestjs/common';
import { CronDecoratorService } from './cron-decorator.service';

@Controller('cron')
export class CronController {
  constructor(private readonly cronService: CronDecoratorService) {}

  @Get('jobs')
  getAllJobs() {
    return this.cronService.getAllCronJobs();
  }

  @Get('job/:name')
  getJobStatus(@Param('name') name: string) {
    return this.cronService.getCronJobStatus(name);
  }

  @Post('job/:name/start')
  startJob(@Param('name') name: string) {
    this.cronService.startCronJob(name);
    return { message: `Job ${name} started` };
  }

  @Post('job/:name/stop')
  stopJob(@Param('name') name: string) {
    this.cronService.stopCronJob(name);
    return { message: `Job ${name} stopped` };
  }

  @Post('job/:name/trigger')
  triggerJob(@Param('name') name: string) {
    this.cronService.triggerCronJob(name);
    return { message: `Job ${name} triggered` };
  }
}
```

**Kasus Penggunaan Nyata:**
1. **Dashboard Admin**: Membuat endpoint untuk mengontrol cron jobs (start/stop/trigger) melalui dashboard tanpa restart aplikasi
2. **Feature Flags**: Menonaktifkan cron job untuk environment tertentu (misal: staging) menggunakan `disabled: process.env.NODE_ENV === 'staging'`
3. **Multi-Tenant Application**: Setiap tenant memiliki jadwal backup sendiri dengan timezone masing-masing
4. **Health Monitoring**: Cron job untuk mengecek kesehatan service eksternal setiap 5 menit, jika gagal kirim alert ke Slack
5. **A/B Testing**: Cron job hanya berjalan untuk grup user tertentu berdasarkan konfigurasi
