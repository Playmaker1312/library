# 28. API Documentation Swagger OpenAPI
## 28.7 ApiQuery ApiParam ApiBody
**Definisi Singkat:** Decorator `@ApiQuery()`, `@ApiParam()`, dan `@ApiBody()` mendokumentasikan parameter request — query string, path parameter, dan request body — secara eksplisit di spesifikasi Swagger.
**Analogi:** Seperti formulir pendaftaran — `@ApiParam` adalah kolom yang sudah tercetak di formulir (fixed), `@ApiQuery` adalah kolom isian tambahan (opsional), dan `@ApiBody` adalah lampiran dokumen.
**Poin-Poin Penting:**
- `@ApiQuery({ name, type, required, description })`
- `@ApiParam({ name, type, description, example })`
- `@ApiBody({ type, description })`
- Bisa dikombinasikan dengan parameter NestJS bawaan
**Contoh Kode:**
```typescript
// products.controller.ts
import { Controller, Get, Post, Param, Query, Body } from '@nestjs/common';
import { ApiTags, ApiOperation, ApiQuery, ApiParam, ApiBody, ApiOkResponse } from '@nestjs/swagger';
import { ProductsService } from './products.service';
import { CreateProductDto } from './dto/create-product.dto';
import { ProductFilterDto } from './dto/product-filter.dto';

@ApiTags('products')
@Controller('products')
export class ProductsController {
  constructor(private readonly productsService: ProductsService) {}

  @Get()
  @ApiOperation({ summary: 'Mencari produk dengan filter' })
  @ApiQuery({
    name: 'search',
    required: false,
    description: 'Kata kunci pencarian',
    example: 'nike',
    type: String,
  })
  @ApiQuery({
    name: 'category',
    required: false,
    description: 'Filter berdasarkan kategori',
    example: 'shoes',
    type: String,
  })
  @ApiQuery({
    name: 'minPrice',
    required: false,
    description: 'Harga minimum',
    example: 100000,
    type: Number,
  })
  @ApiQuery({
    name: 'maxPrice',
    required: false,
    description: 'Harga maksimum',
    example: 5000000,
    type: Number,
  })
  @ApiQuery({
    name: 'sortBy',
    required: false,
    description: 'Urutkan berdasarkan field',
    enum: ['price', 'name', 'createdAt'],
    example: 'price',
  })
  @ApiQuery({
    name: 'order',
    required: false,
    description: 'Arah urutan',
    enum: ['asc', 'desc'],
    example: 'asc',
  })
  @ApiQuery({
    name: 'page',
    required: false,
    description: 'Halaman',
    example: 1,
    type: Number,
  })
  @ApiQuery({
    name: 'limit',
    required: false,
    description: 'Item per halaman',
    example: 10,
    type: Number,
  })
  findAll(
    @Query() filterDto: ProductFilterDto,
    @Query('search') search?: string,
    @Query('page') page?: number,
    @Query('limit') limit?: number,
  ) {
    return this.productsService.findAll({ search, page, limit, ...filterDto });
  }

  @Get(':id')
  @ApiOperation({ summary: 'Mendapatkan produk berdasarkan ID' })
  @ApiParam({
    name: 'id',
    required: true,
    description: 'ID unik produk',
    example: 1,
    type: Number,
  })
  @ApiParam({
    name: 'variant',
    required: false,
    description: 'Varian produk (warna/ukuran)',
    example: 'red',
    type: String,
  })
  findOne(
    @Param('id') id: string,
    @Param('variant') variant?: string,
  ) {
    return this.productsService.findOne(+id, variant);
  }

  @Post()
  @ApiOperation({ summary: 'Membuat produk baru' })
  @ApiBody({
    type: CreateProductDto,
    description: 'Data produk yang akan dibuat',
    required: true,
  })
  create(@Body() createProductDto: CreateProductDto) {
    return this.productsService.create(createProductDto);
  }
}

// dto/product-filter.dto.ts
import { ApiPropertyOptional } from '@nestjs/swagger';
import { IsOptional, IsString, IsNumber, Min } from 'class-validator';
import { Type } from 'class-transformer';

export class ProductFilterDto {
  @ApiPropertyOptional({ description: 'Filter kategori', example: 'shoes' })
  @IsOptional()
  @IsString()
  category?: string;

  @ApiPropertyOptional({ description: 'Harga minimum', example: 100000, type: Number })
  @IsOptional()
  @Type(() => Number)
  @IsNumber()
  @Min(0)
  minPrice?: number;

  @ApiPropertyOptional({ description: 'Harga maksimum', example: 5000000, type: Number })
  @IsOptional()
  @Type(() => Number)
  @IsNumber()
  @Min(0)
  maxPrice?: number;
}

// orders.controller.ts — @ApiBody dengan custom schema
import { Controller, Post, Body } from '@nestjs/common';
import { ApiTags, ApiOperation, ApiBody } from '@nestjs/swagger';

@ApiTags('orders')
@Controller('orders')
export class OrdersController {
  @Post('checkout')
  @ApiOperation({ summary: 'Checkout pesanan' })
  @ApiBody({
    schema: {
      type: 'object',
      required: ['productIds', 'shippingAddress'],
      properties: {
        productIds: {
          type: 'array',
          items: { type: 'number' },
          example: [1, 2, 3],
        },
        shippingAddress: {
          type: 'string',
          example: 'Jl. Merdeka No. 1, Jakarta',
        },
        couponCode: {
          type: 'string',
          example: 'DISKON10',
          nullable: true,
        },
      },
    },
  })
  checkout(@Body() body: any) {
    return { message: 'Checkout berhasil' };
  }
}
```
**Kasus Penggunaan Nyata:** API pencarian produk dengan banyak filter. Dengan `@ApiQuery` yang mendokumentasikan setiap parameter filter, frontend developer bisa melihat parameter apa saja yang tersedia, tipe data, dan contoh nilai — memungkinkan mereka membangun UI filter yang sesuai tanpa perlu membaca dokumentasi terpisah.
