# 28. API Documentation Swagger OpenAPI
## 28.9 Swagger UI Customization
**Definisi Singkat:** Swagger UI dapat dikustomisasi secara ekstensif — mulai dari tema, logo, CSS/JS custom, hingga konfigurasi perilaku UI seperti ekspansi default dan filter.
**Analogi:** Seperti mendekorasi toko — layout dasar sudah dari pabrik, tapi Anda bisa mengganti cat dinding (CSS), menambahkan logo di pintu (custom header), dan mengatur tata letak rak (swaggerOptions) sesuai branding toko Anda.
**Poin-Poin Penting:**
- Opsi `customCss`, `customJs`, `customSiteTitle`
- `swaggerOptions` untuk konfigurasi UI
- `customCssUrl` untuk CSS eksternal
- `customfavIcon` untuk favicon
**Contoh Kode:**
```typescript
// main.ts — full Swagger UI customization
import { NestFactory } from '@nestjs/core';
import { SwaggerModule, DocumentBuilder } from '@nestjs/swagger';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  const config = new DocumentBuilder()
    .setTitle('E-commerce API')
    .setDescription('API platform e-commerce')
    .setVersion('1.0')
    .build();

  const document = SwaggerModule.createDocument(app, config);

  SwaggerModule.setup('api', app, document, {
    // Custom CSS — override styling Swagger UI
    customCss: `
      /* Sembunyikan topbar default */
      .swagger-ui .topbar { display: none; }
      
      /* Custom header */
      .swagger-ui .info .title {
        color: #1a1a2e;
        font-size: 36px;
      }
      
      /* Warna tombol */
      .swagger-ui .btn.execute {
        background-color: #0f3460;
        border-color: #0f3460;
      }
      
      /* Warna response success */
      .swagger-ui .responses-inner .response-col_status {
        color: #16a34a;
      }
      
      /* Custom font */
      .swagger-ui {
        font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
      }
      
      /* Loading animation */
      .swagger-ui .loading-container {
        background: linear-gradient(135deg, #667eea 0%, #764ba2 100%);
      }
    `,

    // Custom CSS dari URL eksternal
    customCssUrl: 'https://cdn.example.com/swagger-custom.css',

    // Custom JavaScript — injected di halaman Swagger UI
    customJs: '/assets/swagger-custom.js',

    // Custom favicon
    customfavIcon: 'https://example.com/favicon.ico',

    // Custom site title
    customSiteTitle: 'MyApp API Docs',

    // Custom JavaScript string (inline)
    customJsStr: `
      console.log('Swagger UI loaded!');
      // Tambahkan custom behavior
      document.addEventListener('DOMContentLoaded', function() {
        // Custom logic
      });
    `,

    // Swagger UI options
    swaggerOptions: {
      // Ekspansi panel: 'list', 'full', 'none'
      docExpansion: 'list',
      
      // Tampilkan filter box
      filter: true,
      
      // Tampilkan durasi request
      showRequestDuration: true,
      
      // Default depth model expansion
      defaultModelsExpandDepth: 3,
      
      // Sembunyikan model yang tidak digunakan
      defaultModelExpandDepth: 2,
      
      // Tampilkan ekstensi vendor (x-*)
      showExtensions: true,
      
      // Tampilkan common extensions
      showCommonExtensions: true,
      
      // Tags sorter
      tagsSorter: 'alpha',
      
      // Operations sorter dalam tag
      operationsSorter: 'alpha',
      
      // Persist authorization di localStorage
      persistAuthorization: true,
      
      // Display request duration
      displayRequestDuration: true,
      
      // Max displayed tags
      maxDisplayedTags: 50,
      
      // Deep linking — URL bisa di-bookmark per endpoint
      deepLinking: {
        enabled: true,
        prefix: 'endpoint',
      },
      
      // Synthetic network response times
      tryItOutEnabled: true,
    },
  });

  await app.listen(3000);
}
bootstrap();

// swagger-custom.js — custom JavaScript
// File: public/swagger-custom.js
(function() {
  // Tambahkan tombol dark mode
  const style = document.createElement('style');
  style.textContent = `
    .dark-mode {
      background-color: #1a1a2e !important;
      color: #e0e0e0 !important;
    }
    .dark-mode .swagger-ui {
      filter: invert(0.9) hue-rotate(180deg);
    }
  `;
  document.head.appendChild(style);
})();
```
```typescript
// main.ts — custom theme switch dengan express static
import { NestFactory } from '@nestjs/core';
import { SwaggerModule, DocumentBuilder } from '@nestjs/swagger';
import { AppModule } from './app.module';
import { NestExpressApplication } from '@nestjs/platform-express';
import { join } from 'path';

async function bootstrap() {
  const app = await NestFactory.create<NestExpressApplication>(AppModule);

  // Serve static files untuk custom CSS/JS
  app.useStaticAssets(join(__dirname, '..', 'public'));

  const config = new DocumentBuilder()
    .setTitle('My API')
    .setVersion('1.0')
    .build();

  const document = SwaggerModule.createDocument(app, config);

  SwaggerModule.setup('api', app, document, {
    customCssUrl: '/swagger-corporate.css',
    customJs: '/swagger-custom.js',
    customSiteTitle: 'Corporate API Docs',
    swaggerOptions: { tryItOutEnabled: true },
  });

  await app.listen(3000);
}
bootstrap();
```
**Kasus Penggunaan Nyata:** Perusahaan yang ingin dokumentasi API sesuai dengan brand guideline — dengan logo perusahaan, warna korporat, dan tata letak yang rapi. Custom Swagger UI memberikan pengalaman profesional bagi pihak ketiga yang mengintegrasikan API.
