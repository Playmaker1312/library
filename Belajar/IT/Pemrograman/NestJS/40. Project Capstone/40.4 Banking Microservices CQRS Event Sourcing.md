# 40. Project Capstone ## 40.4 Banking Microservices: CQRS, Event Sourcing

**Definisi Singkat:** Banking microservices architecture menggunakan CQRS untuk memisahkan transaksi (command) dan read operations (query), Event Sourcing untuk immutable audit trail, dan multiple services untuk account, transaction, fraud detection.

**Analogi:** Banking system seperti bank sungguhan — setiap transaksi dicatat di buku besar immutable (event sourcing), teller (command) proses setor/tarik, dan ATM/saldo (query) baca dari replicated view. Fraud detection adalah service terpisah yang analisis transaksi real-time.

**Poin-Poin Penting:**
- **CQRS** command (transfer, deposit) vs query (balance, history)
- **Event Sourcing** immutable transaction log
- **Microservices** account-service, transaction-service, fraud-service
- **Saga** untuk distributed transaction (transfer antar bank)
- **Projections** read models dari event stream
- **Concurrency** optimistic locking untuk balance

```typescript
export class DepositCommand {
  constructor(
    public readonly accountId: string,
    public readonly amount: number,
    public readonly reference: string,
  ) {}
}

// account-service/src/commands/deposit.handler.ts
@CommandHandler(DepositCommand)
export class DepositHandler implements ICommandHandler<DepositCommand> {
  constructor(
    private readonly eventStore: EventStoreRepository,
    private readonly eventBus: EventBus,
  ) {}

  async execute(command: DepositCommand) {
    const { accountId, amount, reference } = command;

    // Reconstruct aggregate dari events
    const events = await this.eventStore.getEvents(accountId);
    const account = AccountAggregate.reconstruct(events);

    // Business validation
    if (!account || account.status !== 'active') {
      throw new BadRequestException('Account not found or inactive');
    }

    if (amount <= 0) {
      throw new BadRequestException('Deposit amount must be positive');
    }

    // Apply event
    account.apply(new MoneyDepositedEvent(accountId, amount, reference, new Date()));
    account.commit();

    // Save event
    await this.eventStore.save(account.getUncommittedEvents());
  }
}
```

```typescript
// account-service/src/domain/account.aggregate.ts
export class AccountAggregate extends AggregateRoot {
  private id: string;
  private balance: number;
  private status: string;
  private version: number = 0;

  static create(id: string, ownerId: string, initialDeposit: number): AccountAggregate {
    const account = new AccountAggregate();
    account.apply(new AccountCreatedEvent(id, ownerId, initialDeposit));
    return account;
  }

  deposit(amount: number, reference: string): void {
    this.apply(new MoneyDepositedEvent(this.id, amount, reference, new Date()));
  }

  withdraw(amount: number, reference: string): void {
    if (this.balance < amount) {
      throw new Error('Insufficient funds');
    }
    if (this.dailyLimitExceeded(amount)) {
      throw new Error('Daily limit exceeded');
    }
    this.apply(new MoneyWithdrawnEvent(this.id, amount, reference, new Date()));
  }

  transfer(targetAccountId: string, amount: number): void {
    if (this.balance < amount) throw new Error('Insufficient funds');
    this.apply(new MoneyTransferredEvent(this.id, targetAccountId, amount, new Date()));
  }

  freeze(): void {
    if (this.status === 'frozen') throw new Error('Account already frozen');
    this.apply(new AccountFrozenEvent(this.id));
  }

  private onAccountCreatedEvent(event: AccountCreatedEvent) {
    this.id = event.accountId;
    this.balance = event.initialDeposit;
    this.status = 'active';
    this.version = 1;
  }

  private onMoneyDepositedEvent(event: MoneyDepositedEvent) {
    this.balance += event.amount;
    this.version++;
  }

  private onMoneyWithdrawnEvent(event: MoneyWithdrawnEvent) {
    this.balance -= event.amount;
    this.version++;
  }

  static reconstruct(events: any[]): AccountAggregate {
    const account = new AccountAggregate();
    events.forEach((event) => account.apply(event, true));
    return account;
  }
}
```

```typescript
// fraud-service/src/fraud.detector.ts
// Saga untuk transfer antar bank
@Injectable()
export class TransferSaga {
  @Saga()
  transferStarted = (events$: Observable<any>): Observable<ICommand> => {
    return events$.pipe(
      ofType(MoneyTransferredEvent),
      delay(100),
      map((event) => new CheckFraudCommand(event.accountId, event.amount, event.timestamp)),
    );
  };

  @Saga()
  fraudDetected = (events$: Observable<any>): Observable<ICommand> => {
    return events$.pipe(
      ofType(FraudDetectedEvent),
      map((event) => new FreezeAccountCommand(event.accountId, 'Fraud detected')),
    );
  };
}
```
