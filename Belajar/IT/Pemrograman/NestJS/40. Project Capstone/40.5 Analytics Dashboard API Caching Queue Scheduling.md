# 40. Project Capstone ## 40.5 Analytics Dashboard API: Caching, Queue, Scheduling

**Definisi Singkat:** Analytics Dashboard API yang mengagregasi data real-time dengan Redis caching, background job processing dengan message queue (Bull/RabbitMQ), dan scheduled tasks untuk laporan periodik.

**Analogi:** Analytics dashboard seperti command center NASA — data dari berbagai sensor (events) masuk via queue, diproses (aggregation), disimpan sementara di cache (Redis) untuk akses cepat, dan laporan harian otomatis dicetak (scheduled report). Astronot (user) lihat dashboard real-time.

**Poin-Poin Penting:**
- **@nestjs/bull** job queue dengan Redis
- **@nestjs/schedule** cron jobs
- **Redis caching** untuk query berulang
- **Data aggregation** real-time analytics
- **Background processing** heavy computation
- **WebSocket** live updates ke dashboard
- **Rate limiting** untuk API requests

```typescript
// analytics/analytics.service.ts
import { Injectable, Logger } from '@nestjs/common';
import { InjectQueue } from '@nestjs/bull';
import { Queue } from 'bull';
import { Cron, CronExpression } from '@nestjs/schedule';
import { InjectRedis } from '@nestjs/redis';
import Redis from 'ioredis';

@Injectable()
export class AnalyticsService {
  private readonly logger = new Logger(AnalyticsService.name);

  constructor(
    @InjectQueue('analytics') private readonly analyticsQueue: Queue,
    @InjectRedis() private readonly redis: Redis,
  ) {}

  @Cron(CronExpression.EVERY_5_MINUTES)
  async aggregateRealtimeStats() {
    this.logger.log('Running real-time stats aggregation');
    await this.analyticsQueue.add('aggregate:realtime', {
      timestamp: new Date(),
    });
  }

  @Cron('0 0 * * *') // Setiap tengah malam
  async generateDailyReport() {
    this.logger.log('Generating daily report');
    await this.analyticsQueue.add('report:daily', {
      date: new Date().toISOString().split('T')[0],
    });
  }

  @Cron(CronExpression.EVERY_30_MINUTES)
  async cleanupCache() {
    // Hapus cache yang expired
    await this.redis.expire('analytics:dashboard', 1800);
  }

  async trackEvent(event: string, data: Record<string, any>) {
    // Queue event untuk processing async
    await this.analyticsQueue.add('track:event', {
      event,
      data,
      timestamp: new Date(),
    });
  }

  async getDashboardStats(period: '24h' | '7d' | '30d'): Promise<any> {
    const cacheKey = `analytics:dashboard:${period}`;

    // Cek cache
    const cached = await this.redis.get(cacheKey);
    if (cached) {
      return JSON.parse(cached);
    }

    // Hitung dari database (simulasi)
    const stats = await this.computeStats(period);

    // Simpan ke cache (30 menit)
    await this.redis.setex(cacheKey, 1800, JSON.stringify(stats));

    return stats;
  }

  private async computeStats(period: string) {
    return {
      totalUsers: 15000,
      activeUsers: 3200,
      totalRevenue: 45000000,
      conversionRate: 3.2,
      topPages: [
        { path: '/products', views: 15000 },
        { path: '/pricing', views: 8500 },
      ],
      period,
    };
  }
}
```

```typescript
// analytics/analytics.gateway.ts (WebSocket live update)
@WebSocketGateway({ namespace: '/analytics' })
export class AnalyticsGateway implements OnGatewayConnection {
  @WebSocketServer()
  server: Server;

  @SubscribeMessage('subscribe:dashboard')
  handleSubscribe(client: Socket) {
    client.join('dashboard');
  }

  // Dipanggil setelah aggregation selesai
  sendLiveUpdate(data: any) {
    this.server.to('dashboard').emit('analytics:update', data);
  }
}
```
