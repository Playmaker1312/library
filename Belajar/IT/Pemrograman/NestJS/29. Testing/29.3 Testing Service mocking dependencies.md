# 29. Testing
## 29.3 Testing Service mocking dependencies
**Definisi Singkat:** Service di NestJS diuji dengan membuat mock object untuk setiap dependency yang di-inject, menggunakan Jest mock functions untuk mengontrol dan memverifikasi interaksi.
**Analogi:** Seperti pilot yang berlatih di simulator — simulator (mock) menggantikan pesawat asli (real dependency), tapi memberi feedback yang realistis sehingga pilot bisa berlatih prosedur tanpa resiko.
**Poin-Poin Penting:**
- Gunakan `jest.fn()` untuk membuat mock method
- Mock repository, service, atau client eksternal
- Gunakan `mockResolvedValue()` dan `mockRejectedValue()`
- Verifikasi interaksi dengan `toHaveBeenCalled()`
**Contoh Kode:**
```typescript
// orders.service.ts — service yang akan di-test
import { Injectable, NotFoundException } from '@nestjs/common';
import { InjectRepository } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { Order } from './entities/order.entity';
import { InventoryService } from '../inventory/inventory.service';
import { NotificationService } from '../notification/notification.service';

@Injectable()
export class OrdersService {
  constructor(
    @InjectRepository(Order)
    private ordersRepository: Repository<Order>,
    private inventoryService: InventoryService,
    private notificationService: NotificationService,
  ) {}

  async create(orderData: any): Promise<Order> {
    // Cek stok
    const isAvailable = await this.inventoryService.checkAvailability(orderData.productId, orderData.quantity);
    if (!isAvailable) {
      throw new Error('Stok tidak mencukupi');
    }

    // Kurangi stok
    await this.inventoryService.reduceStock(orderData.productId, orderData.quantity);

    // Simpan order
    const order = this.ordersRepository.create(orderData);
    const savedOrder = await this.ordersRepository.save(order);

    // Kirim notifikasi
    await this.notificationService.sendOrderConfirmation(savedOrder);

    return savedOrder;
  }

  async findOne(id: number): Promise<Order> {
    const order = await this.ordersRepository.findOneBy({ id });
    if (!order) {
      throw new NotFoundException(`Order #${id} tidak ditemukan`);
    }
    return order;
  }
}
```
```typescript
// orders.service.spec.ts — comprehensive test with mocks
import { Test, TestingModule } from '@nestjs/testing';
import { OrdersService } from './orders.service';
import { getRepositoryToken } from '@nestjs/typeorm';
import { Order } from './entities/order.entity';
import { InventoryService } from '../inventory/inventory.service';
import { NotificationService } from '../notification/notification.service';
import { NotFoundException } from '@nestjs/common';

describe('OrdersService', () => {
  let service: OrdersService;
  let mockOrdersRepository: any;
  let mockInventoryService: any;
  let mockNotificationService: any;

  beforeEach(async () => {
    // Buat mock untuk semua dependency
    mockOrdersRepository = {
      create: jest.fn(),
      save: jest.fn(),
      findOneBy: jest.fn(),
      find: jest.fn(),
    };

    mockInventoryService = {
      checkAvailability: jest.fn(),
      reduceStock: jest.fn(),
    };

    mockNotificationService = {
      sendOrderConfirmation: jest.fn(),
    };

    const module: TestingModule = await Test.createTestingModule({
      providers: [
        OrdersService,
        { provide: getRepositoryToken(Order), useValue: mockOrdersRepository },
        { provide: InventoryService, useValue: mockInventoryService },
        { provide: NotificationService, useValue: mockNotificationService },
      ],
    }).compile();

    service = module.get<OrdersService>(OrdersService);
  });

  it('should be defined', () => {
    expect(service).toBeDefined();
  });

  describe('create', () => {
    const orderData = { productId: 1, quantity: 2, userId: 1 };

    it('should create order successfully', async () => {
      // Mock semua method yang dipanggil
      mockInventoryService.checkAvailability.mockResolvedValue(true);
      mockInventoryService.reduceStock.mockResolvedValue(undefined);
      const createdOrder = { id: 1, ...orderData };
      mockOrdersRepository.create.mockReturnValue(createdOrder);
      mockOrdersRepository.save.mockResolvedValue(createdOrder);
      mockNotificationService.sendOrderConfirmation.mockResolvedValue(undefined);

      const result = await service.create(orderData);

      // Verifikasi hasil
      expect(result).toEqual(createdOrder);

      // Verifikasi urutan pemanggilan
      expect(mockInventoryService.checkAvailability).toHaveBeenCalledWith(1, 2);
      expect(mockInventoryService.reduceStock).toHaveBeenCalledWith(1, 2);
      expect(mockOrdersRepository.create).toHaveBeenCalledWith(orderData);
      expect(mockOrdersRepository.save).toHaveBeenCalledWith(createdOrder);
      expect(mockNotificationService.sendOrderConfirmation).toHaveBeenCalledWith(createdOrder);
    });

    it('should throw error when stock is insufficient', async () => {
      // Mock stok tidak tersedia
      mockInventoryService.checkAvailability.mockResolvedValue(false);

      // Harus throw error
      await expect(service.create(orderData)).rejects.toThrow('Stok tidak mencukupi');

      // Pastikan tidak lanjut ke proses berikutnya
      expect(mockInventoryService.reduceStock).not.toHaveBeenCalled();
      expect(mockOrdersRepository.save).not.toHaveBeenCalled();
      expect(mockNotificationService.sendOrderConfirmation).not.toHaveBeenCalled();
    });

    it('should handle repository save failure', async () => {
      mockInventoryService.checkAvailability.mockResolvedValue(true);
      mockInventoryService.reduceStock.mockResolvedValue(undefined);
      mockOrdersRepository.create.mockReturnValue(orderData);
      mockOrdersRepository.save.mockRejectedValue(new Error('Database error'));

      await expect(service.create(orderData)).rejects.toThrow('Database error');
    });
  });

  describe('findOne', () => {
    it('should return an order if found', async () => {
      const expectedOrder = { id: 1, productId: 1 };
      mockOrdersRepository.findOneBy.mockResolvedValue(expectedOrder);

      const result = await service.findOne(1);

      expect(result).toEqual(expectedOrder);
      expect(mockOrdersRepository.findOneBy).toHaveBeenCalledWith({ id: 1 });
    });

    it('should throw NotFoundException when order not found', async () => {
      mockOrdersRepository.findOneBy.mockResolvedValue(null);

      await expect(service.findOne(999)).rejects.toThrow(NotFoundException);
    });
  });
});
```
**Kasus Penggunaan Nyata:** Service `OrdersService` memiliki tiga dependency — `Repository<Order>`, `InventoryService`, `NotificationService`. Dalam unit test, ketiganya di-mock sehingga test hanya fokus pada logika bisnis di `OrdersService`, bukan pada implementasi dependency. Jika terjadi error, developer langsung tahu masalahnya di service mana.
