# 29. Testing
## 29.8 Supertest untuk HTTP E2E Test
**Definisi Singkat:** Supertest (`supertest` package) adalah library testing HTTP yang memungkinkan simulasi request HTTP ke aplikasi NestJS yang sedang berjalan, memverifikasi status code, response body, header, dan cookie.
**Analogi:** Seperti pelanggan misterius (mystery shopper) — Anda mengirim "pembeli palsu" ke restoran, dan dia melaporkan apakah pelayanan, kualitas makanan, dan tagihan sesuai standar — tanpa perlu menjadi bagian dari dapur.
**Poin-Poin Penting:**
- `supertest` (atau `@types/supertest`)
- `request(app.getHttpServer())` untuk test HTTP
- Chainable API: `.get()`, `.post()`, `.send()`, `.expect()`
- Test status code, body, header, auth
**Contoh Kode:**
```typescript
// app.e2e-spec.ts — full E2E test dengan Supertest
import { Test, TestingModule } from '@nestjs/testing';
import { INestApplication, ValidationPipe } from '@nestjs/common';
import * as request from 'supertest';
import { AppModule } from '../src/app.module';

describe('E2E Test — HTTP dengan Supertest', () => {
  let app: INestApplication;

  beforeAll(async () => {
    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [AppModule],
    }).compile();

    app = moduleFixture.createNestApplication();

    // Konfigurasi aplikasi SEPERTI di main.ts
    app.setGlobalPrefix('api');
    app.useGlobalPipes(
      new ValidationPipe({
        whitelist: true,
        forbidNonWhitelisted: true,
        transform: true,
      }),
    );

    await app.init();
  });

  afterAll(async () => {
    await app.close();
  });

  describe('GET /api/users', () => {
    it('should return 200 and array of users', async () => {
      const response = await request(app.getHttpServer())
        .get('/api/users')
        .expect('Content-Type', /json/)
        .expect(200);

      expect(Array.isArray(response.body)).toBe(true);
    });

    it('should return 401 without auth token', async () => {
      // Jika endpoint menggunakan auth guard
      await request(app.getHttpServer())
        .get('/api/users')
        .expect(401);
    });
  });

  describe('GET /api/users/:id', () => {
    it('should return 200 for existing user', async () => {
      const response = await request(app.getHttpServer())
        .get('/api/users/1')
        .expect(200);

      expect(response.body).toHaveProperty('id', 1);
    });

    it('should return 404 for non-existing user', async () => {
      await request(app.getHttpServer())
        .get('/api/users/999')
        .expect(404);
    });
  });

  describe('POST /api/users', () => {
    it('should create user and return 201', async () => {
      const newUser = {
        name: 'John Doe',
        email: 'john@example.com',
        password: 'secret123',
      };

      const response = await request(app.getHttpServer())
        .post('/api/users')
        .send(newUser)
        .expect(201);

      expect(response.body).toHaveProperty('id');
      expect(response.body.name).toBe('John Doe');
    });

    it('should return 400 for invalid data', async () => {
      const invalidUser = {
        name: '',  // Nama kosong — validasi gagal
        email: 'not-an-email',
      };

      const response = await request(app.getHttpServer())
        .post('/api/users')
        .send(invalidUser)
        .expect(400);

      expect(response.body).toHaveProperty('message');
      expect(Array.isArray(response.body.message)).toBe(true);
    });
  });
});
```

```typescript
// auth.e2e-spec.ts — test dengan JWT authentication
import * as request from 'supertest';
import { INestApplication } from '@nestjs/common';

describe('Auth E2E — dengan JWT', () => {
  let app: INestApplication;
  let accessToken: string;

  beforeAll(async () => {
    // ... setup app
  });

  describe('POST /api/auth/login', () => {
    it('should return JWT token for valid credentials', async () => {
      const response = await request(app.getHttpServer())
        .post('/api/auth/login')
        .send({ email: 'admin@test.com', password: 'password123' })
        .expect(200);

      expect(response.body).toHaveProperty('accessToken');
      accessToken = response.body.accessToken;
    });

    it('should return 401 for invalid credentials', async () => {
      await request(app.getHttpServer())
        .post('/api/auth/login')
        .send({ email: 'wrong@test.com', password: 'wrong' })
        .expect(401);
    });
  });

  describe('GET /api/protected — with auth', () => {
    it('should return 200 with valid token', async () => {
      await request(app.getHttpServer())
        .get('/api/protected')
        .set('Authorization', `Bearer ${accessToken}`)
        .expect(200);
    });

    it('should return 401 without token', async () => {
      await request(app.getHttpServer())
        .get('/api/protected')
        .expect(401);
    });

    it('should return 401 with invalid token', async () => {
      await request(app.getHttpServer())
        .get('/api/protected')
        .set('Authorization', 'Bearer invalid-token')
        .expect(401);
    });
  });
});
```

```typescript
// Helper untuk database — bersihkan data antar test
export class TestHelper {
  static async clearDatabase(app: INestApplication) {
    const entities = [];
    // Dapatkan semua entity dari connection
    const connection = app.get(await import('@nestjs/typeorm').then(m => m.getDataSourceToken('default')));
    for (const entity of entities) {
      const repository = connection.getRepository(entity);
      await repository.delete({});
    }
  }

  static async createAuthToken(app: INestApplication, role = 'user'): Promise<string> {
    const response = await request(app.getHttpServer())
      .post('/api/auth/login')
      .send({ email: `${role}@test.com`, password: 'password123' });

    return response.body.accessToken;
  }
}
```

**Kasus Penggunaan Nyata:** CI/CD pipeline menjalankan E2E test dengan Supertest setiap kali ada pull request. Test mencakup: registrasi user, login dengan berbagai skenario, CRUD produk dengan autentikasi, dan verifikasi error handling — memastikan tidak ada regression sebelum kode di-merge ke branch utama.
