# 29. Testing
## 29.7 Integration Testing E2E
**Definisi Singkat:** Integration/E2E testing di NestJS menguji aplikasi secara utuh dari request HTTP hingga response, memverifikasi interaksi semua komponen (controller, service, guard, pipe, interceptor) bekerja bersama.
**Analogi:** Seperti uji coba restoran secara keseluruhan — bukan hanya menguji mesin kopi (unit test), tapi memesan kopi dari menu (HTTP request), melihat pelayan memproses, barista membuat kopi, dan kopi tersaji dengan benar (response).
**Poin-Poin Penting:**
- Gunakan `Test.createTestingModule()` dengan modul asli
- `app.init()` untuk menjalankan instance NestJS
- Setup global pipes, guards, interceptors seperti production
- Test seluruh flow aplikasi
**Contoh Kode:**
```typescript
// app.e2e-spec.ts — E2E test dengan modul asli
import { Test, TestingModule } from '@nestjs/testing';
import { INestApplication, ValidationPipe } from '@nestjs/common';
import { AppModule } from '../src/app.module';

describe('AppModule — Integration Test (E2E)', () => {
  let app: INestApplication;

  beforeAll(async () => {
    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [AppModule],  // Import AppModule ASLI (bukan mock)
    }).compile();

    app = moduleFixture.createNestApplication();

    // Setup global pipe SEPERTI di production
    app.useGlobalPipes(
      new ValidationPipe({
        whitelist: true,
        forbidNonWhitelisted: true,
        transform: true,
      }),
    );

    // Setup global filter
    app.useGlobalFilters(new (await import('../src/common/filters/http-exception.filter')).HttpExceptionFilter());

    await app.init();
  });

  afterAll(async () => {
    await app.close();
  });

  it('should be defined', () => {
    expect(app).toBeDefined();
  });
});
```
```typescript
// users.integration.spec.ts — Integration test spesifik modul
import { Test, TestingModule } from '@nestjs/testing';
import { INestApplication } from '@nestjs/common';
import { getRepositoryToken } from '@nestjs/typeorm';
import { UsersModule } from '../src/users/users.module';
import { User } from '../src/users/entities/user.entity';

describe('UsersModule — Integration Test', () => {
  let app: INestApplication;
  let mockUserRepository: any;

  beforeAll(async () => {
    // Mock repository dengan data in-memory
    mockUserRepository = {
      find: jest.fn().mockResolvedValue([
        { id: 1, name: 'John', email: 'john@test.com' },
        { id: 2, name: 'Jane', email: 'jane@test.com' },
      ]),
      findOneBy: jest.fn().mockImplementation(({ id }) => {
        const users = {
          1: { id: 1, name: 'John', email: 'john@test.com' },
          2: { id: 2, name: 'Jane', email: 'jane@test.com' },
        };
        return Promise.resolve(users[id] || null);
      }),
      save: jest.fn().mockImplementation((user) => Promise.resolve({ id: 3, ...user })),
    };

    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [UsersModule],
    })
      .overrideProvider(getRepositoryToken(User))
      .useValue(mockUserRepository)
      .compile();

    app = moduleFixture.createNestApplication();
    await app.init();
  });

  afterAll(async () => {
    await app.close();
  });

  describe('UsersService — integration', () => {
    it('should return all users', async () => {
      const usersService = app.get(await import('../src/users/users.service').then(m => m.UsersService));
      const users = await usersService.findAll();
      expect(users).toHaveLength(2);
      expect(users[0]).toHaveProperty('name', 'John');
    });

    it('should find user by id', async () => {
      const usersService = app.get(await import('../src/users/users.service').then(m => m.UsersService));
      const user = await usersService.findOne(1);
      expect(user).toBeDefined();
      expect(user.name).toBe('John');
    });

    it('should return null for non-existent user', async () => {
      const usersService = app.get(await import('../src/users/users.service').then(m => m.UsersService));
      const user = await usersService.findOne(999);
      expect(user).toBeNull();
    });
  });
});

// Integration test untuk database — dengan test database
import { Test, TestingModule } from '@nestjs/testing';
import { INestApplication } from '@nestjs/common';
import { TypeOrmModule, getRepositoryToken } from '@nestjs/typeorm';
import { Repository } from 'typeorm';
import { User } from '../src/users/entities/user.entity';
import { UsersModule } from '../src/users/users.module';
import { UsersService } from '../src/users/users.service';

describe('Database Integration Test', () => {
  let app: INestApplication;
  let userRepository: Repository<User>;
  let usersService: UsersService;

  beforeAll(async () => {
    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [
        // Gunakan database test (SQLite in-memory)
        TypeOrmModule.forRoot({
          type: 'sqlite',
          database: ':memory:',
          entities: [User],
          synchronize: true,  // Auto-create schema
        }),
        UsersModule,
      ],
    }).compile();

    app = moduleFixture.createNestApplication();
    await app.init();

    userRepository = app.get<Repository<User>>(getRepositoryToken(User));
    usersService = app.get<UsersService>(UsersService);
  });

  afterAll(async () => {
    await app.close();
  });

  beforeEach(async () => {
    // Bersihkan database sebelum setiap test
    await userRepository.clear();
  });

  it('should create and retrieve user', async () => {
    const created = await usersService.create({ name: 'Test', email: 'test@test.com' });
    expect(created.id).toBeDefined();

    const found = await usersService.findOne(created.id);
    expect(found.name).toBe('Test');
  });
});
```
**Kasus Penggunaan Nyata:** Sebelum rilis, tim QA menjalankan E2E test yang mensimulasikan flow lengkap: registrasi user baru → login → melihat produk → membuat pesanan → logout. Test ini memverifikasi semua komponen (validation pipe, auth guard, controller, service, repository) bekerja sama dengan benar.
