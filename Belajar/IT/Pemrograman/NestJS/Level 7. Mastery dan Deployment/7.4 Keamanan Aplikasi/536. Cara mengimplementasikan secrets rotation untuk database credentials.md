# 536. Cara mengimplementasikan secrets rotation untuk database credentials

## Definisi & Target
Rotasi secrets secara periodik (password database, API keys) untuk meminimalkan risiko jika bocor. Target: credential baru tanpa downtime aplikasi.

## Komponen & Nest CLI Command
```
npm install @nestjs/config
```

## Kode Implementasi (Per File)

```
// src/config/secrets-rotation.service.ts
import { Injectable, OnModuleInit } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';

@Injectable()
export class SecretsRotationService implements OnModuleInit {
  constructor(private config: ConfigService) {}

  onModuleInit() {
    this.loadSecrets();
  }

  private async loadSecrets() {
    if (this.config.get('VAULT_PROVIDER') === 'aws') {
      await this.loadFromAwsSecretsManager();
    } else if (this.config.get('VAULT_PROVIDER') === 'hashicorp') {
      await this.loadFromVault();
    }
  }

  private async loadFromAwsSecretsManager() {
    const AWS = await import('@aws-sdk/client-secrets-manager');
    const client = new AWS.SecretsManagerClient({
      region: this.config.get('AWS_REGION'),
    });

    const response = await client.send(
      new AWS.GetSecretValueCommand({
        SecretId: this.config.get('DB_SECRET_ARN'),
      }),
    );

    const secret = JSON.parse(response.SecretString);
    process.env.DATABASE_URL = secret.databaseUrl;
  }

  private async loadFromVault() {
    const response = await fetch(
      `${this.config.get('VAULT_ADDR')}/v1/${this.config.get('VAULT_SECRET_PATH')}`,
      {
        headers: {
          'X-Vault-Token': this.config.get('VAULT_TOKEN'),
        },
      },
    );

    const data = await response.json();
    process.env.DATABASE_URL = data.data.data.databaseUrl;
  }

  async rotateDatabasePassword() {
    const newPassword = crypto.randomBytes(24).toString('hex');
    await this.updateDatabasePassword(newPassword);
    await this.updateSecretInVault(newPassword);
  }
```
