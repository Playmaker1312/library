# 495. Mengimplementasikan tenant-aware rate limiting

## Definisi & Target
Rate limiting yang dihitung per tenant — tenant A punya quota terpisah dari tenant B. Target: throttle guard yang menggunakan `tenantId` sebagai key.

## Komponen & Nest CLI Command
```
npm install @nestjs/throttler
nest g module rate-limit
```

## Kode Implementasi (Per File)

```
// src/rate-limit/tenant-throttler.guard.ts
import { Injectable, ExecutionContext } from '@nestjs/common';
import { ThrottlerGuard, ThrottlerException } from '@nestjs/throttler';
import { TenantContextService } from '../../tenancy/tenant-context.service';

@Injectable()
export class TenantThrottlerGuard extends ThrottlerGuard {
  constructor(private readonly context: TenantContextService) {
    super();
  }

  protected async getTracker(req: Record<string, any>): Promise<string> {
    const tenantId = this.context.getTenantId() || req.ip;
    return `tenant:${tenantId}`;
  }

  protected async getLimitAndTtl(
    context: ExecutionContext,
    options: { limit: number; ttl: number },
  ): Promise<{ limit: number; ttl: number }> {
    const req = context.switchToHttp().getRequest();
    const plan = req.tenant?.plan ?? 'free';

    const limits: Record<string, { limit: number; ttl: number }> = {
      free: { limit: 100, ttl: 60000 },
      pro: { limit: 1000, ttl: 60000 },
      enterprise: { limit: 10000, ttl: 60000 },
    };

    return limits[plan] ?? limits.free;
  }
}
```

```
// src/rate-limit/rate-limit.module.ts
import { Module } from '@nestjs/common';
import { ThrottlerModule } from '@nestjs/throttler';
import { TenantThrottlerGuard } from './tenant-throttler.guard';
import { TenancyModule } from '../../tenancy/tenancy.module';

@Module({
  imports: [
    ThrottlerModule.forRoot([{ limit: 100, ttl: 60000 }]),
    TenancyModule,
  ],
  providers: [
    { provide: 'APP_GUARD', useClass: TenantThrottlerGuard },
  ],
})
export class RateLimitModule {}
```

## Cara Integrasi & Pengujian
Import `RateLimitModule`. Uji: tenant Free → kirim >100 request/menit → 429. Upgrade ke Pro → quota lebih besar.
