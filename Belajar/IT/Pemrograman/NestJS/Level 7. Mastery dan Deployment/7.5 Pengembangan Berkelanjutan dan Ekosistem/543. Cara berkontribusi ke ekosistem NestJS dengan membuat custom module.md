# 543. Cara berkontribusi ke ekosistem NestJS dengan membuat custom module

## Definisi & Target
Buat custom module NestJS yang bisa dipublikasikan ke npm. Target: module reusable dengan konfigurasi async (forRoot/forRootAsync) seperti module-module resmi.

## Komponen & Nest CLI Command
```
nest g library my-cache
npm run build:my-cache
```

## Kode Implementasi (Per File)

```
// src/index.ts — entry point library
export { MyCacheModule } from './my-cache.module';
export { MyCacheService } from './my-cache.service';
export { CACHE_OPTIONS, CacheOptions } from './cache-options.interface';
```

```
// src/cache-options.interface.ts
export interface CacheOptions {
  host: string;
  port: number;
  ttl?: number;
  maxItems?: number;
}

export const CACHE_OPTIONS = 'CACHE_OPTIONS';
```

```
// src/my-cache.module.ts
import { DynamicModule, Module, Provider } from '@nestjs/common';
import { MyCacheService } from './my-cache.service';
import { CACHE_OPTIONS, CacheOptions } from './cache-options.interface';

@Module({})
export class MyCacheModule {
  static forRoot(options: CacheOptions): DynamicModule {
    return {
      module: MyCacheModule,
      providers: [
        { provide: CACHE_OPTIONS, useValue: options },
        MyCacheService,
      ],
      exports: [MyCacheService],
      global: options.global ?? false,
    };
  }

  static forRootAsync(options: {
    imports?: any[];
    useFactory: (...args: any[]) => Promise<CacheOptions> | CacheOptions;
    inject?: any[];
  }): DynamicModule {
    return {
      module: MyCacheModule,
      imports: options.imports ?? [],
      providers: [
        {
          provide: CACHE_OPTIONS,
          useFactory: options.useFactory,
          inject: options.inject ?? [],
        },
        MyCacheService,
      ],
      exports: [MyCacheService],
    };
  }
}
```

```
// src/my-cache.service.ts
import { Inject, Injectable } from '@nestjs/common';
import { CACHE_OPTIONS, CacheOptions } from './cache-options.interface';

@Injectable()
export class MyCacheService {
  private store = new Map<string, { value: any; expiresAt: number }>();

  constructor(@Inject(CACHE_OPTIONS) private options: CacheOptions) {}

  set(key: string, value: any, ttl?: number) {
    const expiresAt = Date.now() + (ttl ?? this.options.ttl ?? 60000);
    this.store.set(key, { value, expiresAt });
  }

  get<T>(key: string): T | undefined {
    const entry = this.store.get(key);
    if (!entry || entry.expiresAt < Date.now()) {
      this.store.delete(key);
      return undefined;
    }
    return entry.value as T;
  }
}
```

## Cara Integrasi & Pengujian
Build library: `nest build my-cache`. Test lokal: `npm link`, lalu `npm link @myorg/my-cache` di project lain. Publikasikan: `npm publish`.
