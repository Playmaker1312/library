# 547. Cara mengintegrasikan NestJS dengan tRPC untuk type-safe API

## Definisi & Target
tRPC memberikan type safety antara backend dan frontend tanpa REST/GraphQL boilerplate. Target: endpoint tRPC di dalam NestJS yang memberikan tipe otomatis ke klien.

## Komponen & Nest CLI Command
```
npm install @trpc/server @trpc/client zod
nest g module trpc
```

## Kode Implementasi (Per File)

```
// src/trpc/trpc.context.ts
import { inferAsyncReturnType } from '@trpc/server';
import { CreateFastifyContextOptions } from '@trpc/server/adapters/fastify';
import { RequestContextService } from '../cls/context.service';

export async function createContext(
  opts: CreateFastifyContextOptions,
  contextService: RequestContextService,
) {
  return {
    req: opts.req,
    res: opts.res,
    tenantId: contextService.tenantId,
    userId: contextService.userId,
  };
}

export type TrpcContext = inferAsyncReturnType<typeof createContext>;
```

```
// src/trpc/trpc.router.ts
import { initTRPC, TRPCError } from '@trpc/server';
import { z } from 'zod';
import { TrpcContext } from './trpc.context';
import { UsersService } from '../../modules/users/users.service';

const t = initTRPC.context<TrpcContext>().create();

const requireAuth = t.middleware(({ ctx, next }) => {
  if (!ctx.userId) throw new TRPCError({ code: 'UNAUTHORIZED' });
  return next({ ctx: { ...ctx, user: { id: ctx.userId } } });
});

const requireTenant = t.middleware(({ ctx, next }) => {
  if (!ctx.tenantId) throw new TRPCError({ code: 'BAD_REQUEST', message: 'Tenant required' });
  return next({ ctx: { ...ctx, tenant: { id: ctx.tenantId } } });
});

export function createAppRouter(usersService: UsersService) {
  return t.router({
    user: t.router({
      list: t.procedure.use(requireAuth).use(requireTenant).query(() => {
        return usersService.findAll();
      }),
      byId: t.procedure.use(requireAuth).input(z.string()).query(({ input }) => {
        return usersService.findById(input);
      }),
      create: t.procedure.use(requireAuth).input(z.object({
        email: z.string().email(),
        name: z.string().min(2),
      })).mutation(({ input }) => {
        return usersService.create(input);
      }),
    }),
    health: t.procedure.query(() => ({ status: 'ok' })),
  });
}

export type AppRouter = ReturnType<typeof createAppRouter>;
```

## Cara Integrasi & Pengujian
Buat adapter tRPC di NestJS (Fastify atau Express). Client: `import { createTRPCClient } from '@trpc/client'`. Semua request/response ter-validasi oleh type TypeScript.
