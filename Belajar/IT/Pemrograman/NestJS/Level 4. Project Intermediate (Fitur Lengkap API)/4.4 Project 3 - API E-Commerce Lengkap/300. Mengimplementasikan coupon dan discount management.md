# 300. Mengimplementasikan coupon dan discount management

## Definisi & Target
Coupon management: admin membuat kode diskon (persentase/nominal), user bisa apply ke order. Validasi expiry, min order, usage limit.

## Komponen & Nest CLI Command
```bash
nest g module coupons && nest g service coupons && nest g controller coupons
```

## Kode Implementasi (Per File)
// src/coupons/coupons.service.ts
```typescript
import { Injectable, NotFoundException, BadRequestException } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { CreateCouponDto } from './dto/create-coupon.dto';

@Injectable()
export class CouponsService {
  constructor(private readonly prisma: PrismaService) {}

  async create(dto: CreateCouponDto) {
    return this.prisma.coupon.create({ data: dto });
  }

  async validateAndApply(code: string, orderTotal: number) {
    const coupon = await this.prisma.coupon.findUnique({ where: { code } });
    if (!coupon) throw new NotFoundException('Coupon not found');
    if (!coupon.isActive) throw new BadRequestException('Coupon is inactive');
    if (coupon.expiresAt < new Date()) throw new BadRequestException('Coupon has expired');
    if (coupon.usedCount >= coupon.usageLimit) throw new BadRequestException('Coupon usage limit reached');
    if (orderTotal < coupon.minOrder) {
      throw new BadRequestException(`Minimum order Rp ${coupon.minOrder.toLocaleString()} required`);
    }

    let discount = 0;
    if (coupon.type === 'percentage') {
      discount = (orderTotal * coupon.discount) / 100;
    } else {
      discount = coupon.discount;
    }

    return {
      code: coupon.code,
      discount,
      totalAfterDiscount: orderTotal - discount,
    };
  }

  async applyCoupon(code: string, orderId: string) {
    const order = await this.prisma.order.findUnique({ where: { id: orderId } });
    if (!order) throw new NotFoundException('Order not found');

    const result = await this.validateAndApply(code, order.total);

    await this.prisma.coupon.update({
      where: { code },
      data: { usedCount: { increment: 1 } },
    });

    await this.prisma.order.update({
      where: { id: orderId },
      data: { total: result.totalAfterDiscount },
    });

    return result;
  }

  async findAll() {
    return this.prisma.coupon.findMany();
  }

  async remove(id: string) {
    const coupon = await this.prisma.coupon.findUnique({ where: { id } });
    if (!coupon) throw new NotFoundException('Coupon not found');
    await this.prisma.coupon.delete({ where: { id } });
  }
}
```

// src/coupons/dto/create-coupon.dto.ts
```typescript
import { IsString, IsNumber, Min, IsEnum, IsDateString, IsOptional } from 'class-validator';

export enum CouponType {
  PERCENTAGE = 'percentage',
  NOMINAL = 'nominal',
}

export class CreateCouponDto {
  @IsString()
  code: string;

  @IsNumber()
  @Min(0)
  discount: number;

  @IsEnum(CouponType)
  type: CouponType;

  @IsOptional()
  @IsNumber()
  @Min(0)
  minOrder?: number;

  @IsOptional()
  @IsNumber()
  @Min(1)
  usageLimit?: number;

  @IsDateString()
  expiresAt: string;
}
```
