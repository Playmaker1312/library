# 266. Cara mengintegrasikan Cloudinary untuk image management dalam NestJS

## Definisi & Target
Cloudinary adalah SaaS untuk image/video management dengan fitur upload, transformasi, optimasi, dan CDN.

## Komponen & Nest CLI Command
```bash
npm install cloudinary
```

## Kode Implementasi (Per File)
// src/media/cloudinary.service.ts
```typescript
import { Injectable } from '@nestjs/common';
import { v2 as cloudinary, UploadApiResponse, UploadApiErrorResponse } from 'cloudinary';
import { ConfigService } from '@nestjs/config';

@Injectable()
export class CloudinaryService {
  constructor(private readonly configService: ConfigService) {
    cloudinary.config({
      cloud_name: this.configService.get('CLOUDINARY_CLOUD_NAME'),
      api_key: this.configService.get('CLOUDINARY_API_KEY'),
      api_secret: this.configService.get('CLOUDINARY_API_SECRET'),
    });
  }

  async uploadImage(file: Express.Multer.File): Promise<UploadApiResponse> {
    return new Promise((resolve, reject) => {
      const upload = cloudinary.uploader.upload_stream(
        { folder: 'products', resource_type: 'image' },
        (error: UploadApiErrorResponse, result: UploadApiResponse) => {
          if (error) reject(error);
          else resolve(result);
        },
      );
      upload.end(file.buffer);
    });
  }

  async deleteImage(publicId: string): Promise<void> {
    await cloudinary.uploader.destroy(publicId);
  }

  getOptimizedUrl(publicId: string, width = 500, height = 500): string {
    return cloudinary.url(publicId, { width, height, crop: 'fill', quality: 'auto', fetch_format: 'auto' });
  }
}
```

## Cara Integrasi & Pengujian
```bash
POST /upload/cloudinary -F "file=@image.jpg"
```
Tambahkan `.env`: `CLOUDINARY_CLOUD_NAME`, `CLOUDINARY_API_KEY`, `CLOUDINARY_API_SECRET`. File dikirim via memoryStorage.
