# 26. Email dan Notifications

## 26.1. nestjs-modules/mailer: Setup

**Definisi Singkat:**
`@nestjs-modules/mailer` adalah modul email resmi untuk NestJS yang menyediakan wrapper untuk `nodemailer` dengan dukungan template engine (Handlebars, Pug, EJS) dan konfigurasi yang mudah.

**Analogi:**
Seperti kantor pos yang lengkap - ada perangko (SMTP config), amplop (email templates), dan petugas pengirim (transport) yang mengantarkan surat (email) ke tujuan.

**Poin-Poin Penting:**
- Package: `@nestjs-modules/mailer` + `nodemailer`
- Berbasis Nodemailer
- Dukungan template: Handlebars, Pug, EJS
- Async configuration via `useFactory`
- Dukungan multiple transports
- Dapat di-inject sebagai `MailerService`

        adapter: new HandlebarsAdapter(),

// mail.module.ts - Async Configuration dengan ConfigService
import { Module } from '@nestjs/common';
import { MailerModule } from '@nestjs-modules/mailer';
import { HandlebarsAdapter } from '@nestjs-modules/mailer/dist/adapters/handlebars.adapter';
import { ConfigModule, ConfigService } from '@nestjs/config';
import { join } from 'path';

@Module({
  imports: [
    MailerModule.forRootAsync({
      imports: [ConfigModule],
      useFactory: (configService: ConfigService) => ({
        transport: {
          host: configService.get('MAIL_HOST', 'smtp.gmail.com'),
          port: configService.get('MAIL_PORT', 587),
          secure: configService.get('MAIL_SECURE', false),
          auth: {
            user: configService.get('MAIL_USER'),
            pass: configService.get('MAIL_PASSWORD'),
          },
          
          // Opsi tambahan
          ignoreTLS: false,
          requireTLS: true,
          connectionTimeout: 10000,
          greetingTimeout: 10000,
          socketTimeout: 10000,
          
          // Pooling untuk performance
          pool: true,
          maxConnections: 5,
          maxMessages: 100,
          rateDelta: 1000,
          rateLimit: 10, // 10 email per detik
        },
        
        defaults: {
          from: configService.get('MAIL_FROM', '"My App" <no-reply@myapp.com>'),
          replyTo: configService.get('MAIL_REPLY_TO', 'support@myapp.com'),
        },
        
        template: {
          dir: join(__dirname, 'templates'),
          adapter: new HandlebarsAdapter({
            // Custom helpers untuk Handlebars
            helpers: {
              formatDate: (date: Date) => {
                return new Date(date).toLocaleDateString('id-ID');
              },
              formatCurrency: (amount: number) => {
                return new Intl.NumberFormat('id-ID', {
                  style: 'currency',
                  currency: 'IDR',
                }).format(amount);
              },
            },
          }),
          options: {
            strict: true,
          },
        },
        
        // Preview email di browser untuk development
        preview: process.env.NODE_ENV === 'development',
      }),
      inject: [ConfigService],
    }),
  ],
})
export class AppMailModule {}

// mail.service.ts - Menggunakan MailerService
import { Injectable, Logger } from '@nestjs/common';
import { MailerService } from '@nestjs-modules/mailer';

@Injectable()
export class MailService {
  private readonly logger = new Logger(MailService.name);

  constructor(private readonly mailerService: MailerService) {}

  // Kirim email plain text
  async sendPlainTextEmail(to: string, subject: string, text: string) {
    try {
      await this.mailerService.sendMail({
        to,
        subject,
        text,
      });
      this.logger.log(`Plain text email sent to ${to}`);
    } catch (error) {
      this.logger.error(`Failed to send email to ${to}: ${error.message}`);
      throw error;
    }
  }

  // Kirim email HTML
  async sendHtmlEmail(to: string, subject: string, html: string) {
    await this.mailerService.sendMail({
      to,
      subject,
      html,
    });
    this.logger.log(`HTML email sent to ${to}`);
  }

  // Kirim email dengan template Handlebars
  async sendWelcomeEmail(user: { email: string; name: string }) {
    await this.mailerService.sendMail({
      to: user.email,
      subject: 'Welcome to My App! 🎉',
      template: './welcome', // File: templates/welcome.hbs
      context: {
        name: user.name,
        email: user.email,
        loginUrl: `${process.env.APP_URL}/login`,
        dashboardUrl: `${process.env.APP_URL}/dashboard`,
        supportEmail: 'support@myapp.com',
        year: new Date().getFullYear(),
      },
    });
    this.logger.log(`Welcome email sent to ${user.email}`);
  }

  // Kirim email reset password
  async sendResetPasswordEmail(
    email: string,
    name: string,
    resetToken: string,
  ) {
    const resetUrl = `${process.env.APP_URL}/reset-password?token=${resetToken}`;
    
    await this.mailerService.sendMail({
      to: email,
      subject: 'Reset Your Password',
      template: './reset-password',
      context: {
        name,
        resetUrl,
        expiryHours: 24,
        supportEmail: 'support@myapp.com',
      },
    });
    this.logger.log(`Reset password email sent to ${email}`);
  }

  // Kirim email dengan attachment
  async sendEmailWithAttachment(
    to: string,
    subject: string,
    template: string,
    context: any,
    attachments: Array<{
      filename: string;
      path: string;
      contentType?: string;
    }>,
  ) {
    await this.mailerService.sendMail({
      to,
      subject,
      template,
      context,
      attachments,
    });
    this.logger.log(`Email with attachment sent to ${to}`);
  }

  // Kirim email ke banyak penerima (BCC)
  async sendBulkEmail(
    recipients: string[],
    subject: string,
    template: string,
    context: any,
  ) {
    await this.mailerService.sendMail({
      to: 'undisclosed-recipients:;',
      bcc: recipients.join(', '),
      subject,
      template,
      context,
    });
    this.logger.log(`Bulk email sent to ${recipients.length} recipients`);
  }
}

// mail.controller.ts - Contoh penggunaan di API
import { Controller, Post, Body } from '@nestjs/common';
import { MailService } from './mail.service';

@Controller('mail')
export class MailController {
  constructor(private readonly mailService: MailService) {}

  @Post('test')
  async sendTestEmail(@Body('to') to: string) {
    await this.mailService.sendPlainTextEmail(
      to,
      'Test Email',
      'This is a test email from NestJS Mailer Module.',
    );
    return { message: 'Test email sent' };
  }

  @Post('welcome')
  async sendWelcome(@Body() body: { email: string; name: string }) {
    await this.mailService.sendWelcomeEmail(body);
    return { message: 'Welcome email sent' };
  }

  @Post('reset-password')
  async sendResetPassword(
    @Body() body: { email: string; name: string; resetToken: string },
  ) {
    await this.mailService.sendResetPasswordEmail(
      body.email,
      body.name,
      body.resetToken,
    );
    return { message: 'Reset password email sent' };
  }
}

**Kasus Penggunaan Nyata:**
1. **User Onboarding**: Kirim email selamat datang setelah user mendaftar, dengan tombol login dan informasi akun
2. **Password Reset**: Kirim email dengan link reset password yang berlaku selama 24 jam
3. **Order Confirmation**: Kirim email konfirmasi setelah pembayaran sukses, dengan detail order dan total harga
4. **Email Verification**: Kirim kode verifikasi email saat registrasi
5. **Newsletter**: Kirim newsletter bulanan ke subscriber dengan template yang menarik
6. **Notification System**: Kirim notifikasi penting (perubahan password, login dari device baru, dll)
