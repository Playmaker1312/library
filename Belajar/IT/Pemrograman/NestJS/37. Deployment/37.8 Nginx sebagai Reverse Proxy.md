# 37. Deployment ## 37.8 Nginx sebagai Reverse Proxy

**Definisi Singkat:** Nginx sebagai reverse proxy diletakkan di depan aplikasi NestJS untuk menangani SSL termination, load balancing, caching, rate limiting, dan serving static files. Nginx menerima request dari internet dan meneruskannya ke aplikasi Node.js.

**Analogi:** Nginx seperti resepsionis gedung perkantoran — semua tamu (request) masuk melalui resepsionis, bukan langsung ke kantor (aplikasi). Resepsionis memeriksa identitas (SSL), mengarahkan ke lantai yang tepat (load balancing), dan menolak tamu mencurigakan (rate limiting).

**Poin-Poin Penting:**
- **Reverse proxy** forward request ke Node.js
- **SSL termination** handle HTTPS certificates
- **Load balancing** distribusi ke multiple instances
- **Caching** static files dan proxy cache
- **Rate limiting** batasi request per IP
- **WebSocket** proxy support
- **Gzip compression** lebih efisien daripada Node.js

**Contoh Kode:**
```nginx
# /etc/nginx/sites-available/nestjs-app
upstream nestjs_cluster {
    # Load balancing — round-robin (default)
    server 127.0.0.1:3001;
    server 127.0.0.1:3002;
    server 127.0.0.1:3003;

    # Opsi: least_connections — kirim ke server dengan koneksi paling sedikit
    # least_conn;

    # Opsi: ip_hash — sticky session
    # ip_hash;

    keepalive 64;
}

server {
    listen 80;
    server_name api.example.com;
    return 301 https://$server_name$request_uri; # Redirect HTTP -> HTTPS
}

server {
    listen 443 ssl http2;
    server_name api.example.com;

    # SSL Configuration
    ssl_certificate /etc/letsencrypt/live/api.example.com/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/api.example.com/privkey.pem;
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_ciphers HIGH:!aNULL:!MD5;

    # Security headers
    add_header Strict-Transport-Security "max-age=31536000; includeSubDomains" always;
    add_header X-Frame-Options "SAMEORIGIN" always;
    add_header X-Content-Type-Options "nosniff" always;
    add_header X-XSS-Protection "1; mode=block" always;

    # Logs
    access_log /var/log/nginx/nestjs-access.log;
    error_log /var/log/nginx/nestjs-error.log;

    # Max upload size
    client_max_body_size 10M;

    # Gzip compression
    gzip on;
    gzip_types application/json application/javascript text/css text/plain;
    gzip_min_length 1000;
    gzip_comp_level 6;
    gzip_vary on;
    gzip_proxied any;

    # Proxy cache config
    proxy_cache_path /var/cache/nginx levels=1:2 keys_zone=api_cache:10m max_size=1g inactive=60m;

    location / {
        proxy_pass http://nestjs_cluster;
        proxy_http_version 1.1;
        proxy_set_header Upgrade $http_upgrade;
        proxy_set_header Connection 'upgrade';
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto $scheme;
        proxy_cache_bypass $http_upgrade;

        # Timeouts
        proxy_connect_timeout 60s;
        proxy_send_timeout 60s;
        proxy_read_timeout 60s;

        # Buffering
        proxy_buffering on;
        proxy_buffer_size 4k;
        proxy_buffers 8 4k;
        proxy_busy_buffers_size 8k;
    }

    # Static files — langsung dari Nginx (tanpa ke Node.js)
    location /static/ {
        alias /var/www/nestjs-app/public/;
        expires 30d;
        add_header Cache-Control "public, immutable";
    }

    # Health check — jangan di-cache
    location /health {
        proxy_pass http://nestjs_cluster;
        proxy_cache off;
        proxy_no_cache 1;
        proxy_cache_bypass 1;
    }
}
```

```typescript
// main.ts — penting: trust proxy untuk IP asli
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);

  // Trust proxy — Nginx ada di depan
  app.set('trust proxy', 1);

  // Rate limiting berdasarkan IP asli (dari X-Forwarded-For)
  // app.use(rateLimit({ windowMs: 60000, max: 100 }));

  await app.listen(process.env.PORT || 3000);
}
bootstrap();
```

```nginx
# /etc/nginx/sites-available/nestjs-ws (WebSocket support)
location /ws/ {
    proxy_pass http://nestjs_cluster;
    proxy_http_version 1.1;
    proxy_set_header Upgrade $http_upgrade;
    proxy_set_header Connection "upgrade";
    proxy_set_header Host $host;
    proxy_read_timeout 86400s; # WebSocket long-lived connection
}
```

**Kasus Penggunaan Nyata:** Aplikasi NestJS production dengan 3 PM2 instances (port 3001-3003) di belakang Nginx. Nginx handle SSL, gzip compression, serve static files langsung, dan load balance dengan least_connections. Jika satu instance down, Nginx otomatis skip.
