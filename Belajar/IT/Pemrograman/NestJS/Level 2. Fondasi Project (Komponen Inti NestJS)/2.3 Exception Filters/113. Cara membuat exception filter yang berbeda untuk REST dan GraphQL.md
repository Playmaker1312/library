# 113. Cara membuat exception filter yang berbeda untuk REST dan GraphQL

## Definisi & Target
`ArgumentsHost.getType()` mengembalikan `'http'` untuk REST dan `'graphql'` untuk GraphQL. Exception filter bisa menangani kedua konteks secara berbeda: REST mengembalikan JSON, GraphQL bisa mengembalikan format Apollo/GraphQL error yang sesuai.

## Komponen & Nest CLI Command
Tidak ada CLI command.
File: `src/common/filters/universal-exception.filter.ts`

## Kode Implementasi (Per File)

```typescript
// src/common/filters/universal-exception.filter.ts
import { ExceptionFilter, Catch, ArgumentsHost, HttpException } from '@nestjs/common';
import { Request, Response } from 'express';

@Catch(HttpException)
export class UniversalExceptionFilter implements ExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const type = host.getType();

    if (type === 'http') {
      this.handleRest(exception, host);
    } else if (type === 'graphql') {
      this.handleGraphQL(exception);
    }
  }

  private handleRest(exception: HttpException, host: ArgumentsHost) {
    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();
    const request = ctx.getRequest<Request>();
    const status = exception.getStatus();

    response.status(status).json({
      statusCode: status,
      message: exception.message,
      timestamp: new Date().toISOString(),
      path: request.url,
    });
  }

  private handleGraphQL(exception: HttpException) {
    const status = exception.getStatus();
    const message = exception.message;

    const gqlError = {
      message,
      extensions: {
        code: `HTTP_${status}`,
        statusCode: status,
        timestamp: new Date().toISOString(),
      },
    };

    throw gqlError;
  }
}
```

```typescript
// src/common/filters/context-specific.filter.ts
import { ExceptionFilter, Catch, ArgumentsHost, HttpException } from '@nestjs/common';
import { GqlArgumentsHost, GqlExceptionFilter } from '@nestjs/graphql';

@Catch(HttpException)
export class ContextSpecificFilter implements GqlExceptionFilter {
  catch(exception: HttpException, host: ArgumentsHost) {
    const gqlHost = GqlArgumentsHost.create(host);
    const context = gqlHost.getContext();
    const info = gqlHost.getInfo();

    const status = exception.getStatus();

    if (host.getType() === 'graphql') {
      return {
        message: exception.message,
        extensions: {
          code: 'INTERNAL_ERROR',
          statusCode: status,
          fieldName: info?.fieldName,
        },
      };
    }

    const ctx = host.switchToHttp();
    const response = ctx.getResponse<Response>();
    response.status(status).json({ statusCode: status, message: exception.message });
  }
}
```

## Cara Integrasi & Pengujian

Filter ini mendeteksi otomatis apakah request REST atau GraphQL dan memberikan error response yang sesuai.

**Test dengan curl:**
```bash
curl http://localhost:3000/users/999
# {"statusCode":404,"message":"User #999 not found","timestamp":"...","path":"/users/999"}
```
