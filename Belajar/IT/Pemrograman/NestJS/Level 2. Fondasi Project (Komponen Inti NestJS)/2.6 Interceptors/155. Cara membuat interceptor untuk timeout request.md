# 155. Cara membuat interceptor untuk timeout request

## Definisi & Target
Timeout interceptor membatasi waktu eksekusi handler. Jika handler tidak selesai dalam batas waktu yang ditentukan, interceptor akan membatalkan request dan mengembalikan error `Request Timeout`. Mencegah endpoint yang lambat menggantung resource server.

## Komponen & Nest CLI Command
```
nest g interceptor common/timeout --no-spec
```
File: `src/common/interceptors/timeout.interceptor.ts`

## Kode Implementasi (Per File)

```typescript
// src/common/interceptors/timeout.interceptor.ts
import { Injectable, NestInterceptor, ExecutionContext, CallHandler, RequestTimeoutException } from '@nestjs/common';
import { Observable, throwError, TimeoutError, catchError, timeout } from 'rxjs';

@Injectable()
export class TimeoutInterceptor implements NestInterceptor {
  constructor(private readonly timeoutMs: number = 5000) {}

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    return next.handle().pipe(
      timeout(this.timeoutMs),
      catchError(err => {
        if (err instanceof TimeoutError) {
          return throwError(() => new RequestTimeoutException(
            `Request exceeded ${this.timeoutMs}ms timeout`,
          ));
        }
        return throwError(() => err);
      }),
    );
  }
}
```

```typescript
// src/common/interceptors/timeout-with-logging.interceptor.ts
import { Injectable, NestInterceptor, ExecutionContext, CallHandler, Logger, RequestTimeoutException } from '@nestjs/common';
import { Observable, throwError, TimeoutError, catchError, timeout, tap } from 'rxjs';

@Injectable()
export class TimeoutWithLoggingInterceptor implements NestInterceptor {
  private readonly logger = new Logger('Timeout');
  private readonly defaultTimeout = 10000;

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();

    return next.handle().pipe(
      timeout(this.defaultTimeout),
      tap({
        error: err => {
          if (err instanceof TimeoutError) {
            this.logger.error(`Timeout: ${request.method} ${request.url}`);
          }
        },
      }),
      catchError(err => {
        if (err instanceof TimeoutError) {
          return throwError(() => new RequestTimeoutException(
            `Server timeout - ${request.method} ${request.url}`,
          ));
        }
        return throwError(() => err);
      }),
    );
  }
}
```

```typescript
// src/users/users.controller.ts
import { Controller, Get, UseInterceptors } from '@nestjs/common';
import { TimeoutInterceptor } from '../common/interceptors/timeout.interceptor';

@Controller('users')
@UseInterceptors(new TimeoutInterceptor(3000))
export class UsersController {
  @Get()
  findAll() {
    return [{ id: 1, name: 'Alice' }];
  }

  @Get('slow')
  async getSlow() {
    await new Promise(resolve => setTimeout(resolve, 5000));
    return { message: 'This is slow' };
  }
}
```

## Cara Integrasi & Pengujian

Endpoint yang memakan waktu lebih lama dari timeout akan mendapat error 408.

**Test dengan curl:**
```bash
curl http://localhost:3000/users/slow
# {"message":"Request exceeded 3000ms timeout","statusCode":408}
```
