# 154. Cara membuat interceptor untuk caching response

## Definisi & Target
Interceptor caching menyimpan response handler di memory sehingga request berikutnya dengan parameter yang sama bisa dilayani langsung dari cache—tanpa mengeksekusi handler lagi. Cocok untuk endpoint yang datanya jarang berubah.

## Komponen & Nest CLI Command
```
nest g interceptor common/cache --no-spec
```
File: `src/common/interceptors/cache.interceptor.ts`

## Kode Implementasi (Per File)

```typescript
// src/common/interceptors/cache.interceptor.ts
import { Injectable, NestInterceptor, ExecutionContext, CallHandler } from '@nestjs/common';
import { Observable, of, tap } from 'rxjs';
import { Request } from 'express';

@Injectable()
export class CacheInterceptor implements NestInterceptor {
  private cache = new Map<string, { data: any; expiry: number }>();
  private readonly defaultTtl = 30000;

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest<Request>();

    if (request.method !== 'GET') {
      return next.handle();
    }

    const key = this.buildCacheKey(request);
    const cached = this.cache.get(key);

    if (cached && Date.now() < cached.expiry) {
      return of(cached.data);
    }

    return next.handle().pipe(
      tap(data => {
        this.cache.set(key, {
          data,
          expiry: Date.now() + this.defaultTtl,
        });
      }),
    );
  }

  private buildCacheKey(request: Request): string {
    return `${request.url}-${JSON.stringify(request.query)}`;
  }

  clearCache() {
    this.cache.clear();
  }
}
```

```typescript
// src/common/interceptors/cache-with-ttl.interceptor.ts
import { Injectable, NestInterceptor, ExecutionContext, CallHandler } from '@nestjs/common';
import { Observable, of, tap } from 'rxjs';

@Injectable()
export class CacheWithTtlInterceptor implements NestInterceptor {
  private cache = new Map<string, any>();
  private timestamps = new Map<string, number>();

  constructor(private readonly ttlMs: number = 30000) {}

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const key = `${request.method}-${request.url}`;

    if (request.method !== 'GET') {
      return next.handle();
    }

    const cached = this.cache.get(key);
    const timestamp = this.timestamps.get(key);

    if (cached && timestamp && Date.now() - timestamp < this.ttlMs) {
      return of(cached);
    }

    return next.handle().pipe(
      tap(data => {
        this.cache.set(key, data);
        this.timestamps.set(key, Date.now());
      }),
    );
  }
}
```

## Cara Integrasi & Pengujian

Request GET yang sama dalam waktu TTL akan mengembalikan data dari cache.

**Test dengan curl:**
```bash
curl http://localhost:3000/users
# Response pertama: dari handler (lambat)
# Response kedua (dalam 30 detik): dari cache (cepat)
```
