# 31. GraphQL
## 31.3 Resolvers Query Mutation Subscription
**Definisi Singkat:** Resolver di GraphQL adalah komponen yang menangani operasi — Query (membaca data), Mutation (mengubah data), dan Subscription (real-time events), analog dengan Controller di REST.
**Analogi:** Query seperti petugas informasi (membaca data), Mutation seperti teller bank (mengubah data), Subscription seperti langganan koran (mendapat update otomatis).
**Poin-Poin Penting:**
- `@Query()` untuk operasi baca
- `@Mutation()` untuk operasi tulis
- `@Subscription()` untuk real-time
- `@Resolver()` decorator class
**Contoh Kode:**
```typescript
// users.resolver.ts — Query & Mutation
import { Resolver, Query, Mutation, Args, Int, ID } from '@nestjs/graphql';
import { UsersService } from './users.service';
import { User } from './entities/user.entity';
import { CreateUserInput } from './dto/create-user.input';
import { UpdateUserInput } from './dto/update-user.input';

@Resolver(() => User)
export class UsersResolver {
  constructor(private readonly usersService: UsersService) {}

  // QUERY: Mendapatkan semua users
  @Query(() => [User], {
    name: 'users',                    // Nama query di schema
    description: 'Mendapatkan daftar semua pengguna',
    nullable: 'items',                // Item dalam array bisa null
  })
  async findAll(): Promise<User[]> {
    return this.usersService.findAll();
  }

  // QUERY: Mendapatkan user by ID
  @Query(() => User, {
    name: 'user',
    description: 'Mendapatkan pengguna berdasarkan ID',
    nullable: true,                   // Bisa return null jika tidak ditemukan
  })
  async findOne(
    @Args('id', { type: () => Int, description: 'ID pengguna' }) id: number,
  ): Promise<User | null> {
    return this.usersService.findOne(id);
  }

  // MUTATION: Membuat user baru
  @Mutation(() => User, {
    name: 'createUser',
    description: 'Membuat pengguna baru',
  })
  async createUser(
    @Args('input', { type: () => CreateUserInput }) input: CreateUserInput,
  ): Promise<User> {
    return this.usersService.create(input);
  }

  // MUTATION: Update user
  @Mutation(() => User, {
    name: 'updateUser',
    description: 'Memperbarui data pengguna',
  })
  async updateUser(
    @Args('id', { type: () => Int }) id: number,
    @Args('input', { type: () => UpdateUserInput }) input: UpdateUserInput,
  ): Promise<User> {
    return this.usersService.update(id, input);
  }

  // MUTATION: Delete user
  @Mutation(() => Boolean, {
    name: 'deleteUser',
    description: 'Menghapus pengguna',
  })
  async deleteUser(
    @Args('id', { type: () => Int }) id: number,
  ): Promise<boolean> {
    await this.usersService.remove(id);
    return true;
  }
}
```
```typescript
// posts.resolver.ts — Subscription example
import { Resolver, Query, Mutation, Subscription, Args, Int } from '@nestjs/graphql';
import { PubSub } from 'graphql-subscriptions';
import { Post } from './entities/post.entity';

const pubSub = new PubSub();

@Resolver(() => Post)
export class PostsResolver {
  @Query(() => [Post])
  async posts() {
    return [];  // Fetch from database
  }

  // MUTATION: Create post + publish subscription event
  @Mutation(() => Post)
  async createPost(
    @Args('title') title: string,
    @Args('content') content: string,
  ): Promise<Post> {
    const newPost = { id: Date.now(), title, content } as Post;

    // Publish event ke subscribers
    pubSub.publish('postCreated', { postCreated: newPost });

    return newPost;
  }

  // SUBSCRIPTION: Listen for new posts (real-time)
  @Subscription(() => Post, {
    name: 'postCreated',
    description: 'Mendapatkan notifikasi saat post baru dibuat',
    // Filter — hanya subscribe post tertentu
    filter: (payload, variables) => {
      // payload: { postCreated: Post }
      // variables: { authorId? }
      return true; // Terima semua post
    },
  })
  postCreated() {
    // Return asyncIterator untuk event 'postCreated'
    return pubSub.asyncIterator('postCreated');
  }
}
```
