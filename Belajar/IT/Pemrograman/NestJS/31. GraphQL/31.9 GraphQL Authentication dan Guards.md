# 31. GraphQL
## 31.9 GraphQL Authentication dan Guards
**Definisi Singkat:** Autentikasi dan otorisasi di GraphQL NestJS menggunakan Guards yang sama seperti REST — `@UseGuards(AuthGuard)` — dengan `ExecutionContext` yang menangani konteks GraphQL.
**Analogi:** Seperti kartu akses gedung — setiap orang (request) harus tap kartu (auth guard) untuk masuk lift, dan kartu menentukan lantai mana yang bisa diakses (roles guard).
**Poin-Poin Penting:**
- `AuthGuard('jwt')` dari `@nestjs/passport`
- `@UseGuards()` di resolver atau method
- Context GraphQL untuk akses user
- Custom guard untuk GraphQL
**Contoh Kode:**
```typescript
// app.module.ts — setup JWT auth + GraphQL context
import { Module } from '@nestjs/common';
import { GraphQLModule } from '@nestjs/graphql';
import { ApolloDriver, ApolloDriverConfig } from '@nestjs/apollo';
import { AuthModule } from './auth/auth.module';

@Module({
  imports: [
    GraphQLModule.forRoot<ApolloDriverConfig>({
      driver: ApolloDriver,
      autoSchemaFile: 'schema.gql',
      // Context dari Express request — akses user dari JWT
      context: ({ req, connection }) => {
        // Untuk HTTP (Query/Mutation)
        if (req) {
          return { req, user: req.user };
        }
        // Untuk WebSocket (Subscription)
        if (connection) {
          return { req: connection.context, user: connection.context.user };
        }
        return {};
      },
      // Untuk subscriptions dengan auth
      subscriptions: {
        'graphql-ws': {
          onConnect: (context: any) => {
            const { connectionParams } = context;
            // Validasi token dari connection params
            const token = connectionParams?.Authorization?.replace('Bearer ', '');
            if (!token) {
              throw new Error('Unauthorized');
            }
            // Verify token dan set user
            // const user = verifyToken(token);
            // context.user = user;
          },
        },
      },
    }),
    AuthModule,
  ],
})
export class AppModule {}
```

```typescript
import { JwtModule } from '@nestjs/jwt';
import { PassportModule } from '@nestjs/passport';
import { JwtStrategy } from './jwt.strategy';

@Module({
  imports: [
    PassportModule.register({ defaultStrategy: 'jwt' }),
    JwtModule.register({
      secret: process.env.JWT_SECRET || 'secret-key',
      signOptions: { expiresIn: '1h' },
    }),
  ],
  providers: [JwtStrategy],
  exports: [PassportModule, JwtModule],
})
export class AuthModule {}
```

```typescript
// auth/jwt.strategy.ts
import { Injectable, UnauthorizedException } from '@nestjs/common';
import { PassportStrategy } from '@nestjs/passport';
import { ExtractJwt, Strategy } from 'passport-jwt';

@Injectable()
export class JwtStrategy extends PassportStrategy(Strategy) {
  constructor() {
    super({
      jwtFromRequest: ExtractJwt.fromAuthHeaderAsBearerToken(),
      ignoreExpiration: false,
      secretOrKey: process.env.JWT_SECRET || 'secret-key',
    });
  }

  async validate(payload: any) {
    return { id: payload.sub, email: payload.email, roles: payload.roles };
  }
}
```

```typescript
// users.resolver.ts — auth guards di GraphQL resolver
import { Resolver, Query, Mutation, Args, Context } from '@nestjs/graphql';
import { UseGuards } from '@nestjs/common';
import { AuthGuard } from '@nestjs/passport';
import { User } from './entities/user.entity';
import { RolesGuard } from '../auth/roles.guard';
import { Roles } from '../auth/roles.decorator';

@Resolver(() => User)
export class UsersResolver {
  // PUBLIC — tanpa autentikasi
  @Query(() => [User])
  async publicUsers() {
    return this.usersService.findAll();
  }

  // PROTECTED — perlu JWT token
  @Query(() => User)
  @UseGuards(AuthGuard('jwt'))  // Guard dari Passport
  async me(@Context() context: any) {
    // context.user berasal dari JWT strategy
    const userId = context.user.id;
    return this.usersService.findOne(userId);
  }

  // ADMIN ONLY — perlu JWT + role admin
  @Mutation(() => User)
  @UseGuards(AuthGuard('jwt'), RolesGuard)  // Multiple guards
  @Roles('admin')  // Custom decorator
  async deleteUser(@Args('id') id: number) {
    return this.usersService.remove(id);
  }

  // Dapatkan user dari context
  @Query(() => User)
  @UseGuards(AuthGuard('jwt'))
  async getProfile(@Context() context: any): Promise<User> {
    // context.user = { id, email, roles }
    return this.usersService.findOne(context.user.id);
  }
}
```

```typescript
// auth/roles.guard.ts — custom guard untuk GraphQL
import { Injectable, CanActivate, ExecutionContext } from '@nestjs/common';
import { Reflector } from '@nestjs/core';
import { GqlExecutionContext } from '@nestjs/graphql';
import { ROLES_KEY } from './roles.decorator';

@Injectable()
export class RolesGuard implements CanActivate {
  constructor(private reflector: Reflector) {}

  canActivate(context: ExecutionContext): boolean {
    // Konversi ExecutionContext ke GqlExecutionContext
    const ctx = GqlExecutionContext.create(context);
    const requiredRoles = this.reflector.getAllAndOverride<string[]>(ROLES_KEY, [
      ctx.getHandler(),
      ctx.getClass(),
    ]);

    if (!requiredRoles) {
      return true;
    }

    // Akses user dari GraphQL context
    const { user } = ctx.getContext();
    return requiredRoles.some((role) => user.roles?.includes(role));
  }
}
```

```typescript
// auth/roles.decorator.ts
import { SetMetadata } from '@nestjs/common';

export const ROLES_KEY = 'roles';
export const Roles = (...roles: string[]) => SetMetadata(ROLES_KEY, roles);
```

**Kasus Penggunaan Nyata:** Panel admin GraphQL. `Query me()` hanya bisa diakses user terautentikasi (AuthGuard). `Mutation deleteUser()` hanya untuk admin (RolesGuard). User biasa mendapat error "Forbidden resource" jika mencoba akses endpoint admin. Semua terintegrasi dengan JWT yang sama dengan REST endpoint.
