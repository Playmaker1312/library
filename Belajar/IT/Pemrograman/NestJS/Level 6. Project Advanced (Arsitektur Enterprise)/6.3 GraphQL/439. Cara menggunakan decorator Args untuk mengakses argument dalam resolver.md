# 439. Cara menggunakan decorator Args untuk mengakses argument dalam resolver

## Definisi & Target
`@Args()` decorator digunakan untuk mengekstrak arguments dari GraphQL query atau mutation. NestJS GraphQL mendukung berbagai cara mengakses args: single arg, arg object, atau parsed input type.

## Komponen & Nest CLI Command
```
Decorator variants:
  @Args('id') id: string                    // Single argument
  @Args() filter: PostsFilterInput          // All arguments as object
  @Args('input') input: CreatePostInput     // Named input argument
  @Args('id', { type: () => ID }) id: string // Dengan type casting
  @Args('ids', { type: () => [String] }) ids: string[] // Array argument
```

## Kode Implementasi (Per File)
```typescript
// src/posts/posts.resolver.ts (args examples)
import { Resolver, Query, Mutation, Args, ID, Int } from '@nestjs/graphql';
import { Post } from './models/post.model';
import { PostsService } from './posts.service';
import { CreatePostInput } from './dto/create-post.input';
import { PostsFilterInput } from './dto/posts-filter.input';

@Resolver(() => Post)
export class PostsResolver {
  constructor(private readonly postsService: PostsService) {}

  @Query(() => Post, { nullable: true })
  findPost(@Args('id', { type: () => ID }) id: string) {
    return this.postsService.findOne(id);
  }

  @Query(() => [Post])
  findPostsByAuthor(
    @Args('authorId', { type: () => ID }) authorId: string,
    @Args('published', { type: () => Boolean, nullable: true }) published?: boolean,
  ) {
    return this.postsService.findByAuthor(authorId, published);
  }

  @Query(() => [Post])
  searchPosts(@Args() filter: PostsFilterInput) {
    return this.postsService.search(filter);
  }

  @Mutation(() => Post)
  createPost(@Args('input') input: CreatePostInput) {
    return this.postsService.create(input);
  }

  @Query(() => [Post])
  paginatedPosts(
    @Args('skip', { type: () => Int, defaultValue: 0 }) skip: number,
    @Args('take', { type: () => Int, defaultValue: 10 }) take: number,
  ) {
    return this.postsService.findAll({ skip, take });
  }
}
```

## Cara Integrasi & Pengujian
Test berbagai bentuk args di Playground:
```graphql
query { findPost(id: "1") { title } }
query { searchPosts(search: "graphql", published: true) { id } }
query { paginatedPosts(skip: 0, take: 5) { title } }
```
