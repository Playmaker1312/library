# 444. Cara menggunakan decorator Context untuk mengakses request context

## Definisi & Target
`@Context()` decorator memberikan akses ke object context GraphQL yang berisi request, response, dan data kustom lainnya. Targetnya adalah mengakses request headers, user data, atau shared services di resolver.

## Komponen & Nest CLI Command
```
@Context() ctx                      // Seluruh context
@Context('req') req                 // Specific property
@Context('req.headers') headers     // Nested property
```

## Kode Implementasi (Per File)
```typescript
// src/common/decorators/gql-context.decorator.ts
import { createParamDecorator, ExecutionContext } from '@nestjs/common';
import { GqlExecutionContext } from '@nestjs/graphql';

export const GqlContext = createParamDecorator(
  (property: string | undefined, context: ExecutionContext) => {
    const ctx = GqlExecutionContext.create(context);
    const gqlContext = ctx.getContext();
    return property ? gqlContext[property] : gqlContext;
  },
);
```

```typescript
// src/posts/posts.resolver.ts (context usage)
import { Resolver, Query, Mutation, Context } from '@nestjs/graphql';
import { Post } from './models/post.model';

@Resolver(() => Post)
export class PostsResolver {
  @Query(() => [Post])
  posts(@Context() context: any) {
    const userAgent = context.req.headers['user-agent'];
    console.log(`Request from: ${userAgent}`);
    return this.postsService.findAll();
  }

  @Mutation(() => Post)
  createPost(
    @Context('req') req: any,
    @Args('input') input: CreatePostInput,
  ) {
    const userId = req.user?.id || 'anonymous';
    return this.postsService.create({ ...input, authorId: userId });
  }

  @Query(() => String)
  myIp(@Context('req.headers') headers: any) {
    return headers['x-forwarded-for'] || headers['x-real-ip'] || 'unknown';
  }
}
```

```typescript
// src/app.module.ts (context setup)
GraphQLModule.forRoot<ApolloDriverConfig>({
  driver: ApolloDriver,
  autoSchemaFile: 'src/schema.gql',
  context: ({ req, res, connection }) => {
    const token = req?.headers?.authorization || '';
    const user = token ? decodeToken(token) : null;
    return { req, res, user, customData: 'shared' };
  },
}),
```

## Cara Integrasi & Pengujian
Akses headers, user, atau data kustom via `@Context()`. Data di context bisa di-set di konfigurasi GraphQLModule.
