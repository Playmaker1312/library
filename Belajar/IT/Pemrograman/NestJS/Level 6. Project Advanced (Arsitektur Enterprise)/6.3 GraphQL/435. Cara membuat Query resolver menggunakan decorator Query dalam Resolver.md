# 435. Cara membuat Query resolver menggunakan decorator Query dalam Resolver

## Definisi & Target
`@Query()` decorator mendefinisikan query endpoint di GraphQL. Resolver class berisi method yang merespon query dari client. Targetnya adalah membuat endpoint read-only untuk mengambil data.

## Komponen & Nest CLI Command
```
Generate:
  nest g resolver authors

Decorators:
  @Resolver(() => Author)    // Resolver untuk Author type
  @Query(() => [Author])     // Query yang return array Author
  @Query(() => Author)       // Query yang return single Author
  @Args()                    // Mengakses arguments
```

## Kode Implementasi (Per File)
```typescript
// src/authors/authors.resolver.ts
import { Resolver, Query, Args, ID, Int } from '@nestjs/graphql';
import { Author } from './models/author.model';
import { AuthorsService } from './authors.service';

@Resolver(() => Author)
export class AuthorsResolver {
  constructor(private readonly authorsService: AuthorsService) {}

  @Query(() => [Author], { name: 'authors' })
  findAll() {
    return this.authorsService.findAll();
  }

  @Query(() => Author, { name: 'author', nullable: true })
  findOne(@Args('id', { type: () => ID }) id: string) {
    return this.authorsService.findOne(id);
  }

  @Query(() => Int, { name: 'authorCount' })
  count() {
    return this.authorsService.count();
  }
}
```

```typescript
// src/authors/authors.service.ts
import { Injectable } from '@nestjs/common';

@Injectable()
export class AuthorsService {
  private authors: Author[] = [
    { id: '1', name: 'J.K. Rowling', age: 58, genres: ['Fantasy'], isActive: true, createdAt: new Date() },
    { id: '2', name: 'George R.R. Martin', age: 75, genres: ['Fantasy', 'Sci-Fi'], isActive: true, createdAt: new Date() },
  ];

  findAll() {
    return this.authors;
  }

  findOne(id: string) {
    return this.authors.find((a) => a.id === id);
  }

  count() {
    return this.authors.length;
  }
}
```
