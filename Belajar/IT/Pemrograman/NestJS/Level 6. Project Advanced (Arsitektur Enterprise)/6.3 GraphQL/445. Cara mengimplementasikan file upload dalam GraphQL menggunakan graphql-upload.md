# 445. Cara mengimplementasikan file upload dalam GraphQL menggunakan graphql-upload

## Definisi & Target
File upload di GraphQL menggunakan multipart request spec. `graphql-upload` menyediakan `Upload` scalar type untuk menangani file upload. Targetnya adalah upload file via mutation GraphQL.

## Komponen & Nest CLI Command
Instalasi: `npm install graphql-upload`

```
GraphQL scalar: Upload
  Representasi: Promise<{ filename, mimetype, encoding, createReadStream }>
```

## Kode Implementasi (Per File)
```typescript
// src/common/scalars/upload.scalar.ts
import { Scalar, CustomScalar } from '@nestjs/graphql';
import { Kind, ValueNode } from 'graphql';

@Scalar('Upload')
export class UploadScalar implements CustomScalar<any, any> {
  description = 'File upload scalar type';

  parseValue(value: any) {
    return value;
  }

  serialize(value: any) {
    return value;
  }

  parseLiteral(ast: ValueNode) {
    if (ast.kind === Kind.STRING) return ast.value;
    return null;
  }
}
```

```typescript
// src/files/dto/upload-file.input.ts
import { InputType, Field } from '@nestjs/graphql';
import * as GraphQLUpload from 'graphql-upload/GraphQLUpload.js';

@InputType()
export class UploadFileInput {
  @Field(() => GraphQLUpload)
  file: any;

  @Field({ nullable: true })
  folder?: string;
}
```

```typescript
// src/files/files.resolver.ts
import { Resolver, Mutation, Args } from '@nestjs/graphql';
import { FileUpload } from 'graphql-upload';
import { createWriteStream } from 'fs';
import { join } from 'path';
import { v4 as uuid } from 'uuid';

const GraphQLUpload = require('graphql-upload/GraphQLUpload.js');

@Resolver()
export class FilesResolver {
  @Mutation(() => String)
  async uploadFile(
    @Args({ name: 'file', type: () => GraphQLUpload })
    { createReadStream, filename }: FileUpload,
    @Args('folder', { nullable: true }) folder?: string,
  ) {
    const uniqueName = `${uuid()}-${filename}`;
    const path = join(process.cwd(), 'uploads', folder || '', uniqueName);
    await new Promise((resolve, reject) => {
      createReadStream()
        .pipe(createWriteStream(path))
        .on('finish', resolve)
        .on('error', reject);
    });
    return uniqueName;
  }

  @Mutation(() => Boolean)
  async uploadMultipleFiles(
    @Args({ name: 'files', type: () => [GraphQLUpload] })
    files: FileUpload[],
  ) {
    for (const file of files) {
      const { createReadStream, filename } = file;
      const path = join(process.cwd(), 'uploads', filename);
      await new Promise((resolve, reject) => {
        createReadStream()
          .pipe(createWriteStream(path))
          .on('finish', resolve)
          .on('error', reject);
      });
    }
    return true;
  }
}
```

## Cara Integrasi & Pengujian
Test dengan Altair atau Postman (multipart):
```graphql
mutation($file: Upload!) {
  uploadFile(file: $file)
}
```
