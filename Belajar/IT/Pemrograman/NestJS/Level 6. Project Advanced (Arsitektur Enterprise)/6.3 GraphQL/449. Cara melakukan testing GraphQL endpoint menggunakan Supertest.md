# 449. Cara melakukan testing GraphQL endpoint menggunakan Supertest

## Definisi & Target
Testing GraphQL endpoint mirip dengan testing REST, tapi body request menggunakan format GraphQL query. Supertest mengirim POST request ke `/graphql` dengan query string.

## Komponen & Nest CLI Command
Instalasi: `npm install --save-dev supertest @types/supertest`

```
Format request:
  POST /graphql
  Body: { query: "...", variables: { ... }, operationName: "..." }
```

## Kode Implementasi (Per File)
```typescript
// test/graphql.e2e-spec.ts
import { Test, TestingModule } from '@nestjs/testing';
import { INestApplication } from '@nestjs/common';
import * as request from 'supertest';
import { AppModule } from '../src/app.module';

describe('GraphQL (e2e)', () => {
  let app: INestApplication;

  beforeAll(async () => {
    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [AppModule],
    }).compile();

    app = moduleFixture.createNestApplication();
    await app.init();
  });

  afterAll(async () => {
    await app.close();
  });

  describe('Queries', () => {
    it('should return users list', async () => {
      const res = await request(app.getHttpServer())
        .post('/graphql')
        .send({ query: '{ users { id name } }' })
        .expect(200);

      expect(res.body.data.users).toBeDefined();
      expect(Array.isArray(res.body.data.users)).toBe(true);
    });

    it('should return single user', async () => {
      const query = `
        query GetUser($userId: ID!) {
          user(id: $userId) { name email }
        }
      `;
      const res = await request(app.getHttpServer())
        .post('/graphql')
        .send({ query, variables: { userId: '1' } })
        .expect(200);

      expect(res.body.data.user.name).toBeDefined();
    });
  });

  describe('Mutations', () => {
    it('should create a user', async () => {
      const mutation = `
        mutation CreateUser($input: CreateUserInput!) {
          createUser(input: $input) { id name email }
        }
      `;
      const res = await request(app.getHttpServer())
        .post('/graphql')
        .send({
          query: mutation,
          variables: { input: { name: 'Test', email: 'test@test.com' } },
        })
        .expect(200);

      expect(res.body.data.createUser.id).toBeDefined();
      expect(res.body.data.createUser.name).toEqual('Test');
    });
  });

  describe('Errors', () => {
    it('should return validation error', async () => {
      const res = await request(app.getHttpServer())
        .post('/graphql')
        .send({ query: '{ unknownField }' })
        .expect(400);

      expect(res.body.errors).toBeDefined();
    });
  });
});
```

## Cara Integrasi & Pengujian
Jalankan dengan `npm run test:e2e`. Pastikan AppModule terimport dengan benar. Gunakan variable untuk test parameterized query.
