# 442. Cara menginstal dan menggunakan NestJS DataLoader package

## Definisi & Target
`nestjs-dataloader` atau `@nestjs/graphql-dataloader` menyediakan integrasi DataLoader yang lebih seamless dengan NestJS DI. Alternatif lain adalah membuat factory sendiri seperti poin 441.

## Komponen & Nest CLI Command
Instalasi: `npm install --save @nestjs/graphql-dataloader dataloader`

```
Package: @nestjs/graphql-dataloader
  Menyediakan decorator @Loader() untuk inject DataLoader
```

## Kode Implementasi (Per File)
```typescript
// src/common/loaders/author.loader.ts
import { Injectable, Scope } from '@nestjs/common';
import * as DataLoader from 'dataloader';
import { AuthorsService } from '../../authors/authors.service';
import { Author } from '../../authors/models/author.model';

@Injectable({ scope: Scope.REQUEST })
export class AuthorLoader {
  constructor(private readonly authorsService: AuthorsService) {}

  getLoader() {
    return new DataLoader<string, Author | null>(async (ids: readonly string[]) => {
      const authors = await this.authorsService.findByIds(ids as string[]);
      const authorMap = new Map(authors.map((a) => [a.id, a]));
      return ids.map((id) => authorMap.get(id) || null);
    });
  }
}
```

```typescript
// src/posts/posts.resolver.ts (dengan @Loader decorator)
import { Resolver, ResolveField, Parent } from '@nestjs/graphql';
import { Loader } from '@nestjs/graphql-dataloader';
import { Post } from './models/post.model';
import { Author } from '../authors/models/author.model';
import { AuthorLoader } from '../common/loaders/author.loader';
import * as DataLoader from 'dataloader';

@Resolver(() => Post)
export class PostsResolver {
  @ResolveField(() => Author, { nullable: true })
  async author(
    @Parent() post: Post,
    @Loader(AuthorLoader) authorLoader: DataLoader<string, Author>,
  ) {
    if (!post.authorId) return null;
    return authorLoader.load(post.authorId);
  }
}
```

```typescript
// src/common/loaders/loaders.module.ts
import { Module } from '@nestjs/common';
import { GraphQLModule } from '@nestjs/graphql';
import { ApolloDriver, ApolloDriverConfig } from '@nestjs/apollo';
import { AuthorLoader } from './author.loader';

@Module({
  imports: [
    GraphQLModule.forRoot<ApolloDriverConfig>({
      driver: ApolloDriver,
      autoSchemaFile: 'src/schema.gql',
    }),
  ],
  providers: [AuthorLoader],
  exports: [AuthorLoader],
})
export class LoadersModule {}
```

## Cara Integrasi & Pengujian
Pastikan loader di-scope sebagai `REQUEST` agar instance per request. Inject via `@Loader(ClassName)` di resolver field.
