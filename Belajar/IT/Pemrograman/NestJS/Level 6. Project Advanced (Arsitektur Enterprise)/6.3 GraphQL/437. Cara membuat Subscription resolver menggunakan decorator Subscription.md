# 437. Cara membuat Subscription resolver menggunakan decorator Subscription

## Definisi & Target
`@Subscription()` decorator memungkinkan client menerima data real-time via WebSocket. Subscription menggunakan PubSub mechanism dari GraphQL untuk mengirim event ke subscriber.

## Komponen & Nest CLI Command
```
Instalasi tambahan:
  npm install graphql-subscriptions (jika perlu PubSub manual)

Decorators:
  @Subscription(() => Post)          // Subscribe ke event
  @Subscription(() => Post, { filter }) // Dengan filter
```

## Kode Implementasi (Per File)
```typescript
// src/posts/posts.resolver.ts (tambahan subscription)
import { Resolver, Subscription, Mutation, Args } from '@nestjs/graphql';
import { Inject, UseGuards } from '@nestjs/common';
import { PubSub } from 'graphql-subscriptions';
import { Post } from './models/post.model';

const PUB_SUB = new PubSub();

@Resolver(() => Post)
export class PostsResolver {
  @Mutation(() => Post)
  createPost(@Args('input') input: CreatePostInput) {
    const post = this.postsService.create(input);
    PUB_SUB.publish('postCreated', { postCreated: post });
    return post;
  }

  @Subscription(() => Post, {
    name: 'postCreated',
    filter: (payload, variables) => {
      return variables.authorId ? payload.postCreated.authorId === variables.authorId : true;
    },
    resolve: (payload) => payload.postCreated,
  })
  subscribeToPostCreated(
    @Args('authorId', { type: () => String, nullable: true }) authorId?: string,
  ) {
    return PUB_SUB.asyncIterator('postCreated');
  }
}
```

```typescript
// src/common/pubsub.module.ts
import { Module, Global } from '@nestjs/common';
import { PubSub } from 'graphql-subscriptions';

@Global()
@Module({
  providers: [
    {
      provide: 'PUB_SUB',
      useValue: new PubSub(),
    },
  ],
  exports: ['PUB_SUB'],
})
export class PubSubModule {}
```
