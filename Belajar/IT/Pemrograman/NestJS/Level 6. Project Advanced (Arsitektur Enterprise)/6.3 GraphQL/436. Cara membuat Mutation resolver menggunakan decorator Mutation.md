# 436. Cara membuat Mutation resolver menggunakan decorator Mutation

## Definisi & Target
`@Mutation()` decorator mendefinisikan mutation endpoint untuk create, update, atau delete data. Mutation bisa mengembalikan data yang dimodifikasi. Targetnya adalah membuat operasi write di GraphQL.

## Komponen & Nest CLI Command
```
Generate:
  nest g resolver posts
  nest g service posts

Decorators:
  @Mutation(() => Post)           // Create operation
  @Mutation(() => Post)           // Update operation
  @Mutation(() => Boolean)       // Delete operation
  @Args('input') input: CreatePostInput  // Input type
```

## Kode Implementasi (Per File)
```typescript
// src/posts/posts.resolver.ts
import { Resolver, Mutation, Args, ID } from '@nestjs/graphql';
import { Post } from './models/post.model';
import { PostsService } from './posts.service';
import { CreatePostInput } from './dto/create-post.input';
import { UpdatePostInput } from './dto/update-post.input';

@Resolver(() => Post)
export class PostsResolver {
  constructor(private readonly postsService: PostsService) {}

  @Mutation(() => Post, { name: 'createPost' })
  create(@Args('input') input: CreatePostInput) {
    return this.postsService.create(input);
  }

  @Mutation(() => Post, { name: 'updatePost' })
  update(
    @Args('id', { type: () => ID }) id: string,
    @Args('input') input: UpdatePostInput,
  ) {
    return this.postsService.update(id, input);
  }

  @Mutation(() => Boolean, { name: 'deletePost' })
  remove(@Args('id', { type: () => ID }) id: string) {
    return this.postsService.remove(id);
  }
}
```

```typescript
// src/posts/posts.service.ts
import { Injectable, NotFoundException } from '@nestjs/common';

@Injectable()
export class PostsService {
  private posts: Post[] = [];

  create(input: CreatePostInput) {
    const post: Post = { id: String(this.posts.length + 1), ...input, viewCount: 0, published: false };
    this.posts.push(post);
    return post;
  }

  update(id: string, input: UpdatePostInput) {
    const index = this.posts.findIndex((p) => p.id === id);
    if (index === -1) throw new NotFoundException();
    this.posts[index] = { ...this.posts[index], ...input };
    return this.posts[index];
  }

  remove(id: string) {
    const index = this.posts.findIndex((p) => p.id === id);
    if (index === -1) return false;
    this.posts.splice(index, 1);
    return true;
  }
}
```
