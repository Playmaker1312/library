# 405. Cara membuat end-to-end test menggunakan Supertest bersama NestJS

## Definisi & Target
E2E test menguji aplikasi dari HTTP request hingga response, melibatkan semua komponen termasuk database. Supertest digunakan untuk mensimulasikan HTTP request ke aplikasi NestJS yang berjalan.

## Komponen & Nest CLI Command
Instalasi: `npm install --save-dev supertest @types/supertest`

File e2e diletakkan di folder `test/`.

## Kode Implementasi (Per File)
```typescript
// test/app.e2e-spec.ts
import { Test, TestingModule } from '@nestjs/testing';
import { INestApplication } from '@nestjs/common';
import * as request from 'supertest';
import { AppModule } from '../src/app.module';

describe('App E2E', () => {
  let app: INestApplication;

  beforeAll(async () => {
    const moduleFixture: TestingModule = await Test.createTestingModule({
      imports: [AppModule],
    }).compile();

    app = moduleFixture.createNestApplication();
    await app.init();
  });

  afterAll(async () => {
    await app.close();
  });

  describe('GET /cats', () => {
    it('should return 200 and array', async () => {
      const res = await request(app.getHttpServer())
        .get('/cats')
        .expect(200);

      expect(Array.isArray(res.body)).toBe(true);
    });
  });

  describe('POST /cats', () => {
    it('should create a new cat', async () => {
      const res = await request(app.getHttpServer())
        .post('/cats')
        .send({ name: 'Kitty', breed: 'Siamese' })
        .expect(201);

      expect(res.body.name).toEqual('Kitty');
      expect(res.body.id).toBeDefined();
    });

    it('should return 400 for invalid data', async () => {
      await request(app.getHttpServer())
        .post('/cats')
        .send({})
        .expect(400);
    });
  });
});
```

## Cara Integrasi & Pengujian
Jalankan dengan `npm run test:e2e` atau `npx jest --config ./test/jest-e2e.json`. Gunakan database test terpisah.
