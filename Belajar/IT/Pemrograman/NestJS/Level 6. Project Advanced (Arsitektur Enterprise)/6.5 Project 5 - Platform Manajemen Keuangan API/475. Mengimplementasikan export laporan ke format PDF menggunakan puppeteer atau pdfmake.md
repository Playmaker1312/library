# 475. Mengimplementasikan export laporan ke format PDF (puppeteer/pdfmake)

## Definisi & Target
Export laporan ke PDF memungkinkan user mendownload laporan keuangan untuk dicetak atau disimpan. Dua pendekatan: puppeteer (HTML to PDF, lebih presisi) atau pdfmake (generate PDF langsung).

## Komponen & Nest CLI Command
Instalasi puppeteer: `npm install puppeteer`
Instalasi pdfmake: `npm install pdfmake`

## Kode Implementasi (Per File)
```typescript
// src/report/export/pdf-export.service.ts (pdfmake)
import { Injectable } from '@nestjs/common';
import * as PdfPrinter from 'pdfmake';
import { TDocumentDefinitions } from 'pdfmake/interfaces';

@Injectable()
export class PdfExportService {
  private printer = new PdfPrinter({
    Roboto: {
      normal: 'node_modules/pdfmake/src/fonts/Roboto-Regular.ttf',
      bold: 'node_modules/pdfmake/src/fonts/Roboto-Medium.ttf',
    },
  });

  generateCashFlowReport(data: { income: number; expense: number; netFlow: number; categories: any }) {
    const docDefinition: TDocumentDefinitions = {
      content: [
        { text: 'Cash Flow Report', style: 'header' },
        { text: `Generated: ${new Date().toLocaleDateString()}`, margin: [0, 0, 0, 20] },
        { text: `Total Income: Rp ${data.income.toLocaleString()}`, margin: [0, 5] },
        { text: `Total Expense: Rp ${data.expense.toLocaleString()}`, margin: [0, 5] },
        { text: `Net Flow: Rp ${data.netFlow.toLocaleString()}`, margin: [0, 5, 0, 20], bold: true },
        { text: 'Spending by Category', style: 'subheader' },
        {
          table: {
            headerRows: 1,
            widths: ['*', 'auto'],
            body: [
              ['Category', 'Amount'],
              ...Object.entries(data.categories).map(([name, amount]) => [
                name,
                `Rp ${(amount as number).toLocaleString()}`,
              ]),
            ],
          },
        },
      ],
      styles: {
        header: { fontSize: 18, bold: true, margin: [0, 0, 0, 10] },
        subheader: { fontSize: 14, bold: true, margin: [0, 10, 0, 5] },
      },
      defaultStyle: { font: 'Roboto' },
    };

    const pdfDoc = this.printer.createPdfKitDocument(docDefinition);
    return pdfDoc;
  }
}
```

```typescript
// src/report/export/puppeteer-export.service.ts
import { Injectable } from '@nestjs/common';
import * as puppeteer from 'puppeteer';

@Injectable()
export class PuppeteerExportService {
  async generatePdfFromHtml(html: string): Promise<Buffer> {
    const browser = await puppeteer.launch({
      headless: true,
      args: ['--no-sandbox', '--disable-setuid-sandbox'],
    });
    const page = await browser.newPage();
    await page.setContent(html, { waitUntil: 'networkidle0' });
    const pdf = await page.pdf({
      format: 'A4',
      margin: { top: '20mm', bottom: '20mm', left: '15mm', right: '15mm' },
    });
    await browser.close();
    return pdf;
  }

  async generateReportHtml(data: any): Promise<string> {
    return `
      <html>
        <head><style>body { font-family: Arial; } h1 { color: #333; } table { width: 100%; border-collapse: collapse; } th, td { border: 1px solid #ddd; padding: 8px; } </style></head>
        <body>
          <h1>Financial Report</h1>
          <p>Income: Rp ${data.income.toLocaleString()}</p>
          <p>Expense: Rp ${data.expense.toLocaleString()}</p>
          <table>
            <tr><th>Category</th><th>Amount</th></tr>
            ${Object.entries(data.categories).map(([name, amount]) => `<tr><td>${name}</td><td>Rp ${(amount as number).toLocaleString()}</td></tr>`).join('')}
          </table>
        </body>
      </html>
    `;
  }
}
```
