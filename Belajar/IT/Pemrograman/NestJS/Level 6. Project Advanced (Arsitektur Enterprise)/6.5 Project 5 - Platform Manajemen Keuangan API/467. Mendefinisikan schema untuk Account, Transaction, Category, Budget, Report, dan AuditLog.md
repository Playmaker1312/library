# 467. Mendefinisikan schema Account, Transaction, Category, Budget, Report, AuditLog

## Definisi & Target
Schema database adalah fondasi platform keuangan. Targetnya adalah mendefinisikan model Prisma yang mendukung double-entry bookkeeping, audit trail, dan relasi antar entitas.

## Komponen & Nest CLI Command
Generate Prisma: `npx prisma init` lalu `npx prisma migrate dev`

## Kode Implementasi (Per File)
```prisma
// prisma/schema.prisma
model User {
  id        String    @id @default(uuid())
  email     String    @unique
  password  String
  name      String
  is2faEnabled Boolean @default(false)
  twoFactorSecret String?
  accounts  Account[]
  budgets   Budget[]
  createdAt DateTime  @default(now())
  updatedAt DateTime  @updatedAt
}

model Account {
  id        String        @id @default(uuid())
  userId    String
  user      User          @relation(fields: [userId], references: [id])
  name      String
  type      AccountType   @default(CHECKING)
  balance   Decimal       @default(0) @db.Decimal(15, 2)
  currency  String        @default("IDR")
  isActive  Boolean       @default(true)
  transactions Transaction[]
  createdAt DateTime      @default(now())
  updatedAt DateTime      @updatedAt
}

enum AccountType { CHECKING SAVINGS CREDIT_CARD INVESTMENT }

model Category {
  id        String    @id @default(uuid())
  userId    String
  name      String
  type      CategoryType
  icon      String?
  transactions Transaction[]
  budgets   Budget[]
}

enum CategoryType { INCOME EXPENSE }

model Transaction {
  id          String       @id @default(uuid())
  accountId   String
  account     Account      @relation(fields: [accountId], references: [id])
  categoryId  String?
  category    Category?    @relation(fields: [categoryId], references: [id])
  amount      Decimal      @db.Decimal(15, 2)
  type        TransactionType
  description String?
  referenceId String?      @unique
  isReconciled Boolean     @default(false)
  entries     JournalEntry[]
  createdAt   DateTime     @default(now())
}

enum TransactionType { CREDIT DEBIT }

model JournalEntry {
  id            String    @id @default(uuid())
  transactionId String
  transaction   Transaction @relation(fields: [transactionId], references: [id])
  accountId     String
  amount        Decimal   @db.Decimal(15, 2)
  side          JournalSide
  createdAt     DateTime  @default(now())
}

enum JournalSide { DEBIT CREDIT }

model Budget {
  id         String     @id @default(uuid())
  userId     String
  user       User       @relation(fields: [userId], references: [id])
  categoryId String
  category   Category   @relation(fields: [categoryId], references: [id])
  amount     Decimal    @db.Decimal(15, 2)
  period     BudgetPeriod
  startDate  DateTime
  endDate    DateTime
  alertAt    Decimal    @default(80) @db.Decimal(5, 2)
}

enum BudgetPeriod { WEEKLY MONTHLY YEARLY }

model AuditLog {
  id        String   @id @default(uuid())
  userId    String?
  action    String
  entity    String
  entityId  String?
  oldValue  Json?
  newValue  Json?
  ipAddress String?
  userAgent String?
  createdAt DateTime @default(now())
}
```

## Cara Integrasi & Pengujian
Jalankan `npx prisma migrate dev` untuk generate migrasi. Gunakan Prisma Studio untuk melihat data: `npx prisma studio`.
