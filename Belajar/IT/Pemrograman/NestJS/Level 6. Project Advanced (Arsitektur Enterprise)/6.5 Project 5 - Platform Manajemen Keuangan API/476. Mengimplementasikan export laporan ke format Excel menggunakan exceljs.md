# 476. Mengimplementasikan export laporan ke format Excel menggunakan exceljs

## Definisi & Target
Export ke Excel memungkinkan user menganalisa data keuangan di spreadsheet. ExcelJS adalah library untuk create, read, dan manipulate Excel files (.xlsx).

## Komponen & Nest CLI Command
Instalasi: `npm install exceljs`

## Kode Implementasi (Per File)
```typescript
// src/report/export/excel-export.service.ts
import { Injectable } from '@nestjs/common';
import * as ExcelJS from 'exceljs';

@Injectable()
export class ExcelExportService {
  async generateTransactionExport(transactions: any[]) {
    const workbook = new ExcelJS.Workbook();
    workbook.creator = 'Finance App';
    workbook.created = new Date();

    const sheet = workbook.addWorksheet('Transactions');
    sheet.columns = [
      { header: 'Date', key: 'date', width: 15 },
      { header: 'Description', key: 'description', width: 30 },
      { header: 'Category', key: 'category', width: 15 },
      { header: 'Type', key: 'type', width: 10 },
      { header: 'Amount', key: 'amount', width: 15 },
      { header: 'Reference', key: 'reference', width: 20 },
    ];

    sheet.getRow(1).font = { bold: true };
    sheet.getRow(1).fill = {
      type: 'pattern',
      pattern: 'solid',
      fgColor: { argb: 'FF4472C4' },
    };
    sheet.getRow(1).font = { bold: true, color: { argb: 'FFFFFFFF' } };

    for (const tx of transactions) {
      sheet.addRow({
        date: tx.createdAt?.toISOString().slice(0, 10),
        description: tx.description || '',
        category: tx.category?.name || '-',
        type: tx.type,
        amount: Number(tx.amount),
        reference: tx.referenceId || '',
      });
    }

    const totalRow = sheet.addRow({
      description: 'TOTAL',
      amount: transactions.reduce((s, t) => s + Number(t.amount), 0),
    });
    totalRow.font = { bold: true };

    sheet.getColumn('amount').numFmt = '#,##0';
    return workbook;
  }

  async generateMonthlyReport(monthlyData: any[]) {
    const workbook = new ExcelJS.Workbook();
    const sheet = workbook.addWorksheet('Monthly Report');
    sheet.columns = [
      { header: 'Month', key: 'month', width: 10 },
      { header: 'Income', key: 'income', width: 15 },
      { header: 'Expense', key: 'expense', width: 15 },
      { header: 'Net Flow', key: 'netFlow', width: 15 },
    ];

    for (const data of monthlyData) {
      sheet.addRow({
        month: data.month,
        income: data.totalIncome,
        expense: data.totalExpense,
        netFlow: data.netFlow,
      });
    }

    const chart = sheet.addImage(
      await workbook.xlsx.writeBuffer().then(() => {}),
    );

    return workbook;
  }

  async writeToBuffer(workbook: ExcelJS.Workbook): Promise<Buffer> {
    const buffer = await workbook.xlsx.writeBuffer();
    return buffer as Buffer;
  }
}
```

```typescript
// src/report/export/export.controller.ts
import { Controller, Get, Res, UseGuards } from '@nestjs/common';
import { Response } from 'express';
import { ExcelExportService } from './excel-export.service';
import { ReportService } from '../report.service';
import { JwtAuthGuard } from '../../auth/guards/jwt-auth.guard';

@Controller('export')
@UseGuards(JwtAuthGuard)
export class ExportController {
  constructor(
    private readonly excelService: ExcelExportService,
    private readonly reportService: ReportService,
  ) {}

  @Get('excel/transactions')
  async exportTransactions(@Res() res: Response) {
    const data = await this.reportService.getAllTransactions();
    const workbook = await this.excelService.generateTransactionExport(data);
    const buffer = await this.excelService.writeToBuffer(workbook);
    res.setHeader('Content-Type', 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet');
    res.setHeader('Content-Disposition', 'attachment; filename=transactions.xlsx');
    res.send(buffer);
  }
}
```
