# 477. Membangun AuditLogModule yang mencatat setiap perubahan data otomatis

## Definisi & Target
Audit log mencatat siapa, apa, kapan, dan data sebelum/sesudah setiap perubahan. Targetnya adalah memenuhi requirement compliance dan mendeteksi aktivitas mencurigakan.

## Komponen & Nest CLI Command
```
Generate:
  nest g module audit-log
  nest g service audit-log
```

## Kode Implementasi (Per File)
```typescript
// src/audit-log/audit-log.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';
import { Request } from 'express';

@Injectable()
export class AuditLogService {
  constructor(private readonly prisma: PrismaService) {}

  async log(data: {
    userId?: string;
    action: 'CREATE' | 'UPDATE' | 'DELETE' | 'LOGIN' | 'EXPORT';
    entity: string;
    entityId?: string;
    oldValue?: any;
    newValue?: any;
    request?: Request;
  }) {
    return this.prisma.auditLog.create({
      data: {
        userId: data.userId,
        action: data.action,
        entity: data.entity,
        entityId: data.entityId,
        oldValue: data.oldValue || undefined,
        newValue: data.newValue || undefined,
        ipAddress: data.request?.ip || data.request?.socket?.remoteAddress,
        userAgent: data.request?.headers?.['user-agent'],
      },
    });
  }

  async findByUser(userId: string, page: number = 1, limit: number = 50) {
    return this.prisma.auditLog.findMany({
      where: { userId },
      orderBy: { createdAt: 'desc' },
      skip: (page - 1) * limit,
      take: limit,
    });
  }

  async findByEntity(entity: string, entityId: string) {
    return this.prisma.auditLog.findMany({
      where: { entity, entityId },
      orderBy: { createdAt: 'desc' },
    });
  }

  async getSuspiciousActivity(thresholdMinutes: number = 5) {
    const recent = new Date(Date.now() - thresholdMinutes * 60 * 1000);
    const logs = await this.prisma.auditLog.findMany({
      where: { createdAt: { gte: recent } },
      orderBy: { createdAt: 'desc' },
    });

    const ipCount: Record<string, number> = {};
    for (const log of logs) {
      const ip = log.ipAddress || 'unknown';
      ipCount[ip] = (ipCount[ip] || 0) + 1;
    }

    return Object.entries(ipCount)
      .filter(([_, count]) => count > 10)
      .map(([ip, count]) => ({ ip, requestCount: count, suspicious: true }));
  }
}
```

```typescript
// src/common/interceptors/audit-log.interceptor.ts
import { Injectable, NestInterceptor, ExecutionContext, CallHandler } from '@nestjs/common';
import { Observable } from 'rxjs';
import { tap } from 'rxjs/operators';
import { AuditLogService } from '../../audit-log/audit-log.service';
import { Reflector } from '@nestjs/core';

@Injectable()
export class AuditLogInterceptor implements NestInterceptor {
  constructor(
    private readonly auditLog: AuditLogService,
    private readonly reflector: Reflector,
  ) {}

  intercept(context: ExecutionContext, next: CallHandler): Observable<any> {
    const request = context.switchToHttp().getRequest();
    const auditAction = this.reflector.get<string>('auditAction', context.getHandler());

    if (!auditAction) return next.handle();

    return next.handle().pipe(
      tap((response) => {
        this.auditLog.log({
          userId: request.user?.id,
          action: auditAction as any,
          entity: context.getClass().name,
          newValue: response,
          request,
        });
      }),
    );
  }
}
```
