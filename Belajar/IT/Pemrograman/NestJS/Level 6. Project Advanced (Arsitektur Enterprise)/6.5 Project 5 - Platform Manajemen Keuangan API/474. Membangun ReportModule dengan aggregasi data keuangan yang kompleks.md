# 474. Membangun ReportModule dengan aggregasi data keuangan kompleks

## Definisi & Target
Report module menyediakan laporan keuangan agregat: cash flow, spending by category, balance history, dll. Targetnya adalah memberikan insight finansial dari data transaksi.

## Komponen & Nest CLI Command
```
Generate:
  nest g module report
  nest g service report
  nest g controller report
```

## Kode Implementasi (Per File)
```typescript
// src/report/report.service.ts
import { Injectable } from '@nestjs/common';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class ReportService {
  constructor(private readonly prisma: PrismaService) {}

  async cashFlow(userId: string, startDate: Date, endDate: Date) {
    const transactions = await this.prisma.transaction.findMany({
      where: {
        account: { userId },
        createdAt: { gte: startDate, lte: endDate },
      },
      include: { category: true },
    });

    const income = transactions.filter((t) => t.type === 'CREDIT');
    const expense = transactions.filter((t) => t.type === 'DEBIT');

    return {
      totalIncome: income.reduce((sum, t) => sum + Number(t.amount), 0),
      totalExpense: expense.reduce((sum, t) => sum + Number(t.amount), 0),
      netFlow: income.reduce((sum, t) => sum + Number(t.amount), 0) -
        expense.reduce((sum, t) => sum + Number(t.amount), 0),
      incomeByCategory: this.groupByCategory(income),
      expenseByCategory: this.groupByCategory(expense),
    };
  }

  async spendingTrend(userId: string, months: number = 6) {
    const monthlyData = [];
    for (let i = months - 1; i >= 0; i--) {
      const date = new Date();
      date.setMonth(date.getMonth() - i);
      const startOfMonth = new Date(date.getFullYear(), date.getMonth(), 1);
      const endOfMonth = new Date(date.getFullYear(), date.getMonth() + 1, 0);
      const flow = await this.cashFlow(userId, startOfMonth, endOfMonth);
      monthlyData.push({
        month: startOfMonth.toISOString().slice(0, 7),
        ...flow,
      });
    }
    return monthlyData;
  }

  async balanceHistory(userId: string) {
    const accounts = await this.prisma.account.findMany({
      where: { userId },
      select: { id: true, name: true, balance: true, createdAt: true },
    });
    return accounts;
  }

  async transactionSummary(userId: string, date: Date) {
    const startOfDay = new Date(date.getFullYear(), date.getMonth(), date.getDate());
    const endOfDay = new Date(startOfDay.getTime() + 86400000);

    return this.prisma.transaction.count({
      where: {
        account: { userId },
        createdAt: { gte: startOfDay, lte: endOfDay },
      },
    });
  }

  private groupByCategory(transactions: any[]) {
    const grouped: Record<string, number> = {};
    for (const t of transactions) {
      const cat = t.category?.name || 'Uncategorized';
      grouped[cat] = (grouped[cat] || 0) + Number(t.amount);
    }
    return grouped;
  }
}
```

```typescript
// src/report/report.controller.ts
import { Controller, Get, Query, UseGuards } from '@nestjs/common';
import { ReportService } from './report.service';
import { JwtAuthGuard } from '../auth/guards/jwt-auth.guard';
import { CurrentUser } from '../common/decorators/current-user.decorator';

@Controller('reports')
@UseGuards(JwtAuthGuard)
export class ReportController {
  constructor(private readonly reportService: ReportService) {}

  @Get('cash-flow')
  cashFlow(@CurrentUser() user: any, @Query('start') start: string, @Query('end') end: string) {
    return this.reportService.cashFlow(user.id, new Date(start), new Date(end));
  }

  @Get('spending-trend')
  spendingTrend(@CurrentUser() user: any, @Query('months') months?: string) {
    return this.reportService.spendingTrend(user.id, months ? parseInt(months) : 6);
  }

  @Get('balance')
  balance(@CurrentUser() user: any) {
    return this.reportService.balanceHistory(user.id);
  }
}
```
