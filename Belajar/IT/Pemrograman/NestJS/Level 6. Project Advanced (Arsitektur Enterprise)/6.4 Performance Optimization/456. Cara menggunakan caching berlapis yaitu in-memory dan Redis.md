# 456. Cara menggunakan caching berlapis yaitu in-memory dan Redis

## Definisi & Target
Caching berlapis (layered caching) menggunakan in-memory cache (cepat) sebagai L1 dan Redis (distributed) sebagai L2. Targetnya adalah mengurangi beban database dengan data yang di-cache di multiple layer.

## Komponen & Nest CLI Command
```
Layered cache flow:
  1. Cek L1 (in-memory) -> return jika ada
  2. Cek L2 (Redis) -> return + simpan di L1
  3. Query database -> simpan di L1 + L2
```

## Kode Implementasi (Per File)
```typescript
// src/cache/layered-cache.service.ts
import { Injectable, Inject } from '@nestjs/common';
import { CACHE_MANAGER } from '@nestjs/cache-manager';
import { Cache } from 'cache-manager';

@Injectable()
export class LayeredCacheService {
  constructor(@Inject(CACHE_MANAGER) private readonly cacheManager: Cache) {}

  async get<T>(key: string, fallback: () => Promise<T>, ttl: number = 60): Promise<T> {
    const cached = await this.cacheManager.get<T>(key);
    if (cached) return cached;

    const data = await fallback();
    await this.cacheManager.set(key, data, ttl * 1000);
    return data;
  }

  async invalidate(pattern: string) {
    await this.cacheManager.del(pattern);
  }

  async invalidateMany(keys: string[]) {
    await Promise.all(keys.map((k) => this.cacheManager.del(k)));
  }
}
```

```typescript
// src/cache/cache.module.ts (multi-layer config)
import { Module } from '@nestjs/common';
import { CacheModule } from '@nestjs/cache-manager';
import { redisStore } from 'cache-manager-redis-yet';
import { LayeredCacheService } from './layered-cache.service';

@Module({
  imports: [
    CacheModule.registerAsync({
      useFactory: async () => ({
        store: await redisStore({
          socket: { host: 'localhost', port: 6379 },
          ttl: 60 * 1000,
        }),
        isGlobal: true,
        max: 100,
      }),
    }),
  ],
  providers: [LayeredCacheService],
  exports: [LayeredCacheService],
})
export class AppCacheModule {}
```

```typescript
// src/posts/posts.service.ts (using cache)
import { Injectable } from '@nestjs/common';
import { LayeredCacheService } from '../cache/layered-cache.service';
import { PrismaService } from '../prisma/prisma.service';

@Injectable()
export class PostsService {
  constructor(
    private readonly prisma: PrismaService,
    private readonly cache: LayeredCacheService,
  ) {}

  async getPost(id: string) {
    return this.cache.get(
      `post:${id}`,
      () => this.prisma.post.findUnique({ where: { id } }),
      300,
    );
  }

  async updatePost(id: string, data: any) {
    const post = await this.prisma.post.update({ where: { id }, data });
    await this.cache.invalidate(`post:${id}`);
    return post;
  }
}
```
