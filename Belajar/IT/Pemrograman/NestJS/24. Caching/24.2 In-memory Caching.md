# 24. Caching

## 24.2. In-memory Caching

**Definisi Singkat:**
In-memory Caching adalah mekanisme penyimpanan cache di dalam memori (RAM) aplikasi, menggunakan struktur data seperti Map atau objek untuk akses super cepat tanpa I/O disk atau network call.

**Analogi:**
Seperti meja kerja - menyimpan barang yang sedang digunakan agar bisa dijangkau dalam sekejap, berbeda dengan lemari arsip (database) yang perlu dibuka dan dicari.

**Poin-Poin Penting:**
- Default store dari `cache-manager` adalah in-memory
- Sangat cepat (nanoseconds access)
- Data hilang ketika aplikasi restart
- Tidak cocok untuk multi-instance/cluster
- Cocok untuk development dan simple use cases
- Batasan memori sesuai dengan RAM yang dialokasikan

**Contoh Kode:**
```typescript
// In-memory cache dasar dengan Map
import { Injectable } from '@nestjs/common';

@Injectable()
export class SimpleInMemoryCache {
  private cache = new Map<string, { value: any; expiry: number }>();

  set(key: string, value: any, ttlMs: number = 60000): void {
    this.cache.set(key, {
      value,
      expiry: Date.now() + ttlMs,
    });
  }

  get<T = any>(key: string): T | undefined {
    const item = this.cache.get(key);
    
    if (!item) return undefined;
    
    // Cek apakah sudah expired
    if (Date.now() > item.expiry) {
      this.cache.delete(key);
      return undefined;
    }
    
    return item.value as T;
  }

  delete(key: string): void {
    this.cache.delete(key);
  }

  clear(): void {
    this.cache.clear();
  }

  // Cleanup otomatis untuk item expired
  private cleanupInterval: NodeJS.Timeout;

  constructor() {
    // Jalankan cleanup setiap 1 menit
    this.cleanupInterval = setInterval(() => this.cleanup(), 60000);
  }

  private cleanup(): void {
    const now = Date.now();
    for (const [key, item] of this.cache.entries()) {
      if (now > item.expiry) {
        this.cache.delete(key);
      }
    }
  }
}

// Menggunakan NestJS CacheManager dengan in-memory store
import { Module } from '@nestjs/common';
import { CacheModule } from '@nestjs/cache-manager';
import { caching } from 'cache-manager';

@Module({
  imports: [
    CacheModule.register({
      // Store default adalah 'memory'
      store: 'memory',
      ttl: 10, // 10 detik
      max: 100, // maksimal 100 items
      
      // Opsi tambahan untuk memory store
      refreshThreshold: 3, // refresh cache jika sisa TTL < 3 detik
      allowStale: true, // izinkan return stale data saat refresh
    }),
  ],
})
export class InMemoryCacheModule {}

// Product Service dengan in-memory cache
import { Injectable, Inject } from '@nestjs/common';
import { Cache } from 'cache-manager';
import { CACHE_MANAGER } from '@nestjs/cache-manager';

@Injectable()
export class ProductService {
  constructor(@Inject(CACHE_MANAGER) private cacheManager: Cache) {}

  private products = [
    { id: 1, name: 'Laptop Gaming', price: 25000000, stock: 50 },
    { id: 2, name: 'Mouse Wireless', price: 350000, stock: 200 },
    { id: 3, name: 'Keyboard Mechanical', price: 850000, stock: 100 },
  ];

  async getProduct(id: number): Promise<any> {
    const cacheKey = `product:${id}`;
    
    // Coba dari cache terlebih dahulu
    const cached = await this.cacheManager.get<any>(cacheKey);
    
    if (cached) {
      console.log('Returning from in-memory cache');
      return cached;
    }

    console.log('Querying database...');
    // Simulasi query database yang lambat
    await this.delay(100);
    
    const product = this.products.find(p => p.id === id);
    
    if (product) {
      // Simpan ke cache selama 5 menit
      await this.cacheManager.set(cacheKey, product, 300000);
    }

    return product;
  }

  async getTopProducts(): Promise<any[]> {
    const cacheKey = 'products:top';
    
    const cached = await this.cacheManager.get<any[]>(cacheKey);
    if (cached) return cached;

    await this.delay(200);
    
    const topProducts = this.products
      .sort((a, b) => b.stock - a.stock)
      .slice(0, 5);

    await this.cacheManager.set(cacheKey, topProducts, 60000); // 1 menit
    
    return topProducts;
  }

  async updateProduct(id: number, data: any): Promise<any> {
    const index = this.products.findIndex(p => p.id === id);
    if (index === -1) return null;

    this.products[index] = { ...this.products[index], ...data };

    // Invalidate cache setelah update
    const cacheKey = `product:${id}`;
    await this.cacheManager.del(cacheKey);
    
    // Juga hapus cache list products
    await this.cacheManager.del('products:top');

    return this.products[index];
  }

  private delay(ms: number): Promise<void> {
    return new Promise(resolve => setTimeout(resolve, ms));
  }
}

// In-memory cache dengan LRU (Least Recently Used)
import { Injectable } from '@nestjs/common';
import { LRUCache } from 'lru-cache';

@Injectable()
export class LruCacheService {
  private cache: LRUCache<string, any>;

  constructor() {
    this.cache = new LRUCache({
      max: 500, // maksimal 500 items
      ttl: 300000, // 5 menit
      allowStale: true,
      updateAgeOnGet: true, // perbarui TTL ketika diakses
      
      // Berdasarkan ukuran memori (opsional)
      maxSize: 50 * 1024 * 1024, // 50MB
      sizeCalculation: (value) => {
        return JSON.stringify(value).length;
      },
    });
  }

  get<T = any>(key: string): T | undefined {
    return this.cache.get(key) as T;
  }

  set(key: string, value: any, ttl?: number): void {
    this.cache.set(key, value, ttl ? { ttl } : undefined);
  }

  delete(key: string): void {
    this.cache.delete(key);
  }

  has(key: string): boolean {
    return this.cache.has(key);
  }

  clear(): void {
    this.cache.clear();
  }

  getStats(): any {
    return {
      size: this.cache.size,
      itemCount: this.cache.keys.length,
    };
  }
}
```

**Kasus Penggunaan Nyata:**
- Men-cache session data yang aktif untuk 1000 pengguna online
- Menyimpan konfigurasi aplikasi yang di-load saat startup
- Caching hasil perhitungan yang sering dipakai (seperti shipping cost calculator)
- Development dan testing environment (tidak perlu setup Redis/Memcached)
- Aplikasi single-instance yang tidak membutuhkan shared cache antar instance
