# 24. Caching

## 24.1. nestjs/cache-manager

**Definisi Singkat:**
`@nestjs/cache-manager` adalah modul caching resmi NestJS yang menyediakan abstraksi untuk berbagai cache store (in-memory, Redis, dll) dengan API yang konsisten dan mudah digunakan.

**Analogi:**
Seperti lemari pendingin (kulkas) di dapur - menyimpan makanan yang sering digunakan agar tidak perlu pergi ke supermarket (database) setiap kali membutuhkan.

**Poin-Poin Penting:**
- Package: `@nestjs/cache-manager` + `cache-manager`
- Berdasarkan library `cache-manager`
- Mendukung berbagai store: memory, Redis, MongoDB, dll
- Decorator: `@Cacheable()`, `@CacheEvict()`, `@CachePut()`
- Dapat menggunakan `CacheInterceptor` untuk caching otomatis
- Support async configuration dan multiple stores

**Contoh Kode:**
```typescript
// Install: npm install @nestjs/cache-manager cache-manager

// cache.module.ts - Basic Setup
import { Module } from '@nestjs/common';
import { CacheModule } from '@nestjs/cache-manager';

@Module({
  imports: [
    CacheModule.register({
      ttl: 5, // default TTL: 5 detik
      max: 100, // maksimum item di cache
      isGlobal: true, // membuat module global
    }),
  ],
})
export class AppCacheModule {}

// cache.module.ts - Async Configuration
import { Module } from '@nestjs/common';
import { CacheModule } from '@nestjs/cache-manager';
import { ConfigModule, ConfigService } from '@nestjs/config';

@Module({
  imports: [
    CacheModule.registerAsync({
      isGlobal: true,
      imports: [ConfigModule],
      useFactory: (configService: ConfigService) => ({
        ttl: configService.get('CACHE_TTL', 300), // 5 menit default
        max: configService.get('CACHE_MAX_ITEMS', 1000),
      }),
      inject: [ConfigService],
    }),
  ],
})
export class AppCacheModule {}

// Menggunakan CacheService di Controller
import { Controller, Get, Inject } from '@nestjs/common';
import { Cache } from 'cache-manager';
import { CACHE_MANAGER } from '@nestjs/cache-manager';

@Controller('users')
export class UserController {
  constructor(@Inject(CACHE_MANAGER) private cacheManager: Cache) {}

  @Get()
  async getUsers() {
    const cacheKey = 'users:all';
    
    // Coba dapatkan dari cache
    const cachedUsers = await this.cacheManager.get(cacheKey);
    if (cachedUsers) {
      return { source: 'cache', data: cachedUsers };
    }

    // Jika tidak ada di cache, ambil dari database
    const users = await this.fetchFromDatabase();
    
    // Simpan ke cache dengan TTL 60 detik
    await this.cacheManager.set(cacheKey, users, 60000);
    
    return { source: 'database', data: users };
  }

  @Get('clear-cache')
  async clearCache() {
    // Hapus semua cache
    await this.cacheManager.reset();
    return { message: 'Cache cleared' };
  }

  private async fetchFromDatabase() {
    return [
      { id: 1, name: 'John Doe' },
      { id: 2, name: 'Jane Doe' },
    ];
  }
}

// Menggunakan Cache di Service dengan decorator
import { Injectable } from '@nestjs/common';
import { Cacheable, CacheEvict, CachePut } from '@nestjs/cache-manager';

@Injectable()
export class ProductService {
  private products = [
    { id: 1, name: 'Laptop', price: 15000000 },
    { id: 2, name: 'Mouse', price: 250000 },
  ];

  // Cache hasil method ini dengan key otomatis
  @Cacheable({ ttl: 300 })
  async findAll() {
    console.log('Querying database for products...');
    return this.products;
  }

  // Cache dengan custom key
  @Cacheable({ key: 'product_{id}', ttl: 600 })
  async findOne(id: number) {
    console.log(`Querying database for product #${id}...`);
    return this.products.find(p => p.id === id);
  }

  // Update cache setelah operasi
  @CachePut({ key: 'product_{id}' })
  async update(id: number, data: any) {
    const index = this.products.findIndex(p => p.id === id);
    if (index !== -1) {
      this.products[index] = { ...this.products[index], ...data };
    }
    return this.products[index];
  }

  // Hapus cache setelah operasi
  @CacheEvict({ key: 'products' })
  async create(data: any) {
    const newProduct = { id: Date.now(), ...data };
    this.products.push(newProduct);
    return newProduct;
  }

  // Hapus semua cache dengan pattern
  @CacheEvict({ pattern: 'product_*' })
  async clearAllProductsCache() {
    return true;
  }
}

// Module dengan CacheInterceptor untuk auto-cache semua GET endpoint
import { Module, CacheInterceptor } from '@nestjs/common';
import { APP_INTERCEPTOR } from '@nestjs/core';
import { CacheModule } from '@nestjs/cache-manager';
import { ProductController } from './product.controller';
import { ProductService } from './product.service';

@Module({
  imports: [
    CacheModule.register({
      ttl: 120, // 2 menit
      isGlobal: true,
    }),
  ],
  controllers: [ProductController],
  providers: [
    ProductService,
    {
      provide: APP_INTERCEPTOR,
      useClass: CacheInterceptor,
    },
  ],
})
export class ProductModule {}
```

**Kasus Penggunaan Nyata:**
Menyimpan data master seperti daftar provinsi, kategori produk, atau konfigurasi aplikasi yang jarang berubah. Men-cache response API GET yang sering diakses dan memakan resource besar (seperti laporan dengan complex query) untuk mengurangi beban database dan meningkatkan response time dari 2 detik menjadi <100ms.
