# 24. Caching

## 24.7. Custom Cache Decorator

**Definisi Singkat:**
Custom Cache Decorator adalah decorator buatan sendiri yang memperluas atau memodifikasi perilaku caching default NestJS, memungkinkan logika caching yang lebih kompleks dan spesifik sesuai kebutuhan aplikasi.

**Analogi:**
Seperti membuat tombol pintas di remote TV yang menjalankan beberapa perintah sekaligus (contoh: tombol "Movie Mode" yang mematikan lampu, menyalakan TV, dan mengatur volume).

**Poin-Poin Penting:**
- Menggunakan NestJS Custom Decorator pattern
- Menggabungkan dengan Interceptors
- Dapat menambahkan custom logic: conditional caching, dynamic TTL, cache fallback
- Menggunakan `Reflector` untuk membaca metadata
- Dapat dikombinasikan dengan Guards/Pipes

**Contoh Kode:**
```typescript
// cache.decorator.ts - Custom Cache Decorator
import { SetMetadata, applyDecorators, UseInterceptors } from '@nestjs/common';
import { CacheInterceptor, CacheKey, CacheTTL } from '@nestjs/cache-manager';

// Custom metadata key
export const CACHE_CONDITIONAL = 'cache_conditional';
export const CACHE_USER_SPECIFIC = 'cache_user_specific';
export const CACHE_DYNAMIC_TTL = 'cache_dynamic_ttl';

// Decorator 1: Conditional Cache (hanya cache jika kondisi terpenuhi)
export function ConditionalCache(condition: (context: any) => boolean) {
  return SetMetadata(CACHE_CONDITIONAL, condition);
}

// Decorator 2: User-Specific Cache (cache per user)
export function UserSpecificCache() {
  return SetMetadata(CACHE_USER_SPECIFIC, true);
}

// Decorator 3: Dynamic TTL (TTL berdasarkan data)
export function DynamicTTL(calculateTTL: (data: any) => number) {
  return SetMetadata(CACHE_DYNAMIC_TTL, calculateTTL);
}

// Decorator Kombinasi: Smart Cache
export function SmartCache(options: {
  key?: string;
  ttl?: number;
  userSpecific?: boolean;
  condition?: (context: any) => boolean;
}) {
  const decorators: MethodDecorator[] = [];

  if (options.key) {
    decorators.push(CacheKey(options.key));
  }

  if (options.ttl) {
    decorators.push(CacheTTL(options.ttl));
  }

  if (options.userSpecific) {
    decorators.push(UserSpecificCache());
  }

  if (options.condition) {
    decorators.push(ConditionalCache(options.condition));
  }

  return applyDecorators(...decorators);
}

// custom-cache.interceptor.ts
import {
  Injectable,
  ExecutionContext,
  Inject,
  Logger,
} from '@nestjs/common';
import { CacheInterceptor, CACHE_KEY_METADATA, CACHE_TTL_METADATA } from '@nestjs/cache-manager';
import { Cache } from 'cache-manager';
import { CACHE_MANAGER } from '@nestjs/cache-manager';
import { Reflector } from '@nestjs/core';
import {
  CACHE_CONDITIONAL,
  CACHE_USER_SPECIFIC,
  CACHE_DYNAMIC_TTL,
} from './cache.decorator';

@Injectable()
export class CustomCacheInterceptor extends CacheInterceptor {
  @Inject(CACHE_MANAGER) cacheManager: Cache;
  reflector: Reflector;
  private logger = new Logger('CustomCacheInterceptor');

  // Override: Generate cache key dengan tambahan user ID
  trackBy(context: ExecutionContext): string | undefined {
    const request = context.switchToHttp().getRequest();
    const handler = context.getHandler();

    // Cek apakah user-specific cache
    const isUserSpecific = this.reflector.get(CACHE_USER_SPECIFIC, handler);
    
    if (isUserSpecific && request.user) {
      const baseKey = this.reflector.get(CACHE_KEY_METADATA, handler);
      const userId = request.user.id;
      
      if (baseKey) {
        return `user:${userId}:${baseKey}`;
      }
      
      const url = request.url;
      return `user:${userId}:${url}`;
    }

    return super.trackBy(context);
  }

  // Override: Cek apakah request harus di-cache
  protected isRequestCacheable(context: ExecutionContext): boolean {
    const handler = context.getHandler();
    const request = context.switchToHttp().getRequest();

    // Cek conditional cache
    const conditionalCache = this.reflector.get(
      CACHE_CONDITIONAL,
      handler,
    );

    if (conditionalCache) {
      const shouldCache = conditionalCache({ request, context });
      if (!shouldCache) {
        this.logger.debug('Skipping cache due to conditional check');
        return false;
      }
    }

    // Cek header Cache-Control
    const cacheControl = request.headers['cache-control'];
    if (cacheControl === 'no-cache' || cacheControl === 'max-age=0') {
      this.logger.debug('Skipping cache due to Cache-Control: no-cache');
      return false;
    }

    return super.isRequestCacheable(context);
  }

  // Override: Dapatkan TTL secara dinamis
  protected cachedTTL(context: ExecutionContext): number {
    const handler = context.getHandler();
    
    // Cek dynamic TTL
    const dynamicTTL = this.reflector.get(CACHE_DYNAMIC_TTL, handler);
    if (dynamicTTL) {
      // Note: dynamicTTL bisa dihitung berdasarkan response data
      // Untuk contoh ini, kita gunakan default dulu
      this.logger.debug('Dynamic TTL calculator available');
    }

    // Cek custom TTL dari environment
    const defaultTTL = this.defaultTTL;
    const request = context.switchToHttp().getRequest();
    
    // TTL lebih panjang untuk static data
    if (request.url.includes('/categories') || request.url.includes('/config')) {
      return 3600; // 1 jam
    }
    
    // TTL lebih pendek untuk search
    if (request.url.includes('/search')) {
      return 30; // 30 detik
    }

    return defaultTTL;
  }
}

// Decorator untuk cache dengan fallback ke source lain
import { SetMetadata } from '@nestjs/common';

export const CACHE_FALLBACK = 'cache_fallback';

export function CacheWithFallback(options: {
  key: string;
  ttl: number;
  fallbackService?: string;
  fallbackMethod?: string;
}) {
  return SetMetadata(CACHE_FALLBACK, options);
}

import { Request } from 'express';
import {
  ConditionalCache,
  UserSpecificCache,
  DynamicTTL,
  SmartCache,
  CacheWithFallback,
} from './cache.decorator';
import { CustomCacheInterceptor } from './custom-cache.interceptor';
import { UseInterceptors } from '@nestjs/common';
import { CacheKey, CacheTTL } from '@nestjs/cache-manager';

@Controller('products')
@UseInterceptors(CustomCacheInterceptor)
export class ProductController {
  private products = [];

  // Conditional Cache: Hanya cache jika bukan admin
  @Get()
  @ConditionalCache(({ request }) => request.user?.role !== 'admin')
  @CacheKey('products:all')
  @CacheTTL(120)
  async findAll() {
    return this.products;
  }

  // User-Specific Cache: Setiap user punya cache sendiri
  @Get('favorites')
  @UserSpecificCache()
  @CacheKey('products:favorites')
  @CacheTTL(300)
  async getFavorites(@Req() req: Request) {
    const userId = req.user?.id;
    return this.fetchUserFavorites(userId);
  }

  // Smart Cache: Kombinasi beberapa opsi
  @Get('recommendations')
  @SmartCache({
    key: 'products:recommendations',
    ttl: 600,
    userSpecific: true,
    condition: ({ request }) => !request.headers['x-no-cache'],
  })
  async getRecommendations(@Req() req: Request) {
    return this.fetchRecommendations(req.user?.id);
  }

  // Dynamic TTL berdasarkan kategori
  @Get('category/:categoryId')
  @CacheKey('products:category_{categoryId}')
  async getByCategory(@Param('categoryId') categoryId: string) {
    const products = await this.fetchByCategory(categoryId);

// Multi-Level Cache Decorator
import { SetMetadata } from '@nestjs/common';

export const MULTI_LEVEL_CACHE = 'multi_level_cache';

export function MultiLevelCache(options: {
  key: string;
  l1TTL: number;  // In-memory TTL
  l2TTL: number;  // Redis TTL
}) {
  return SetMetadata(MULTI_LEVEL_CACHE, options);
}
```

**Kasus Penggunaan Nyata:**
1. **User-Specific Dashboard**: Setiap user melihat data yang berbeda, menggunakan `@UserSpecificCache()` untuk memisahkan cache per user dengan pattern `user:{id}:dashboard`
2. **Admin vs Regular User**: Admin selalu melihat data real-time, regular user dapat cached data - menggunakan `@ConditionalCache()`
3. **Multi-Region Caching**: L1 cache di memory (lokal), L2 cache di Redis (shared) - menghemat network call antar region
4. **A/B Testing Cache**: Fitur yang berbeda untuk user yang berbeda, cache key menyertakan variant ID
5. **Graceful Degradation**: Jika database down, kembalikan cached data meskipun sudah expired - cache with fallback
