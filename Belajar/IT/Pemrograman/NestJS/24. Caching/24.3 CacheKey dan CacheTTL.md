# 24. Caching

## 24.3. CacheKey dan CacheTTL

**Definisi Singkat:**
`CacheKey` adalah identifier unik untuk setiap item di cache, sedangkan `CacheTTL` (Time To Live) adalah durasi berapa lama item tersebut tetap valid di cache sebelum dianggap expired.

**Analogi:**
CacheKey seperti nomor locker di gym (untuk menemukan barangmu), CacheTTL seperti batas waktu penyewaan locker (setelah itu barang bisa diambil/dihapus).

**Poin-Poin Penting:**
- CacheKey harus unik untuk setiap data yang berbeda
- TTL diukur dalam miliseconds
- Decorator: `@CacheKey()`, `@CacheTTL()`
- Dapat di-override per method/endpoint
- Pattern umum: `resource:{id}:{variation}`
- TTL yang terlalu pendek = kurang efektif, terlalu lama = data stale

**Contoh Kode:**
```typescript
// CacheKey manual di Controller
import { Controller, Get, Param, Query, Inject } from '@nestjs/common';
import { Cache } from 'cache-manager';
import { CACHE_MANAGER } from '@nestjs/cache-manager';

@Controller('products')
export class ProductController {
  constructor(@Inject(CACHE_MANAGER) private cacheManager: Cache) {}

  // Pattern: products:{id}
  @Get(':id')
  async getProduct(@Param('id') id: string) {
    const cacheKey = `products:${id}`;
    
    let product = await this.cacheManager.get<any>(cacheKey);
    
    if (!product) {
      product = await this.fetchFromDB(id);
      // TTL 5 menit = 300000ms
      await this.cacheManager.set(cacheKey, product, 300000);
    }
    
    return product;
  }

  // Pattern: products:list:{page}:{limit}:{category}
  @Get()
  async getProducts(
    @Query('page') page: number = 1,
    @Query('limit') limit: number = 10,
    @Query('category') category?: string,
  ) {
    let cacheKey = `products:list:${page}:${limit}`;
    if (category) {
      cacheKey += `:category:${category}`;
    }

    let products = await this.cacheManager.get<any[]>(cacheKey);
    
    if (!products) {
      products = await this.fetchListFromDB(page, limit, category);
      // TTL lebih singkat untuk list (sering berubah)
      await this.cacheManager.set(cacheKey, products, 60000); // 1 menit
    }
    
    return products;
  }

  // Cache dengan TTL berbeda berdasarkan kondisi
  @Get('featured')
  async getFeaturedProducts() {
    const cacheKey = 'products:featured';
    const isPeakHour = this.isPeakHour();
    
    let products = await this.cacheManager.get<any[]>(cacheKey);
    
    if (!products) {
      products = await this.fetchFeaturedFromDB();
      
      // TTL lebih panjang saat peak hour untuk mengurangi DB load
      const ttl = isPeakHour ? 300000 : 120000; // 5 menit vs 2 menit
      await this.cacheManager.set(cacheKey, products, ttl);
    }
    
    return products;
  }

  private isPeakHour(): boolean {
    const hour = new Date().getHours();
    return hour >= 10 && hour <= 20; // 10:00 - 20:00
  }

  private async fetchFromDB(id: string) {
    return { id, name: 'Product ' + id, price: 100000 };
  }

  private async fetchListFromDB(page: number, limit: number, category?: string) {
    return [];
  }

  private async fetchFeaturedFromDB() {
    return [];
  }
}

// Menggunakan decorator @CacheKey dan @CacheTTL
import { Controller, Get, Param, Query } from '@nestjs/common';
import { CacheKey, CacheTTL } from '@nestjs/cache-manager';

@Controller('articles')
export class ArticleController {
  // CacheTTL 5 menit
  @Get(':id')
  @CacheKey('article_{id}')
  @CacheTTL(300000)
  async getArticle(@Param('id') id: string) {
    return this.fetchArticle(id);
  }

  // TTL lebih singkat untuk list
  @Get()
  @CacheKey('articles_list')
  @CacheTTL(60000)
  async getArticles(@Query('page') page: number = 1) {
    return this.fetchArticles(page);
  }

  // TTL dinamis berdasarkan konfigurasi
  @Get('popular')
  @CacheKey('articles_popular')
  async getPopularArticles() {
    return this.fetchPopularArticles();
  }

  private async fetchArticle(id: string) {
    return { id, title: 'Article ' + id };
  }

  private async fetchArticles(page: number) {
    return [];
  }

  private async fetchPopularArticles() {
    return [];
  }
}

// Advanced: Cache Key Generator
import { Injectable, ExecutionContext } from '@nestjs/common';
import { CacheKey, CacheInterceptor, CACHE_KEY_METADATA } from '@nestjs/cache-manager';
import { Reflector } from '@nestjs/core';

@Injectable()
export class CustomCacheInterceptor extends CacheInterceptor {
  protected reflector: Reflector;

  trackBy(context: ExecutionContext): string | undefined {
    const cacheKey = this.reflector.get(
      CACHE_KEY_METADATA,
      context.getHandler(),
    );

    if (cacheKey) {
      const request = context.switchToHttp().getRequest();
      const params = request.params;
      const query = request.query;
      const userId = request.user?.id;

      let finalKey = cacheKey;
      
      // Replace placeholder {param}
      for (const [key, value] of Object.entries(params)) {
        finalKey = finalKey.replace(`{${key}}`, value);
      }

      // Tambahkan query params jika ada
      if (Object.keys(query).length > 0) {
        finalKey += ':' + this.serializeQuery(query);
      }

      // Tambahkan userId untuk data user-specific
      if (userId) {
        finalKey = `user:${userId}:` + finalKey;
      }

      return finalKey;
    }

    return super.trackBy(context);
  }

  private serializeQuery(query: any): string {
    const sortedKeys = Object.keys(query).sort();
    const pairs = sortedKeys.map(k => `${k}:${query[k]}`);
    return pairs.join(':');
  }
}

// Contoh penggunaan cache key yang baik vs buruk
const cacheKeys = {
  // BURUK: terlalu generik
  bad1: 'data',
  bad2: 'products',
  bad3: 'user_' + 123, // tanpa namespace

  // BAIK: memiliki namespace, detail, unik
  good1: 'users:profile:123',
  good2: 'products:detail:456:lang:en',
  good3: 'categories:list:active:page:1:limit:20',
  good4: 'user:123:notifications:unread:count',
  good5: 'orders:statistics:2024:01:user:456',
};

// Dynamic TTL berdasarkan ukuran data
function calculateTTL(dataSize: number): number {
  // Data kecil = TTL lebih panjang
  if (dataSize < 1024) return 3600000; // 1 jam
  
  // Data sedang
  if (dataSize < 10240) return 1800000; // 30 menit
  
  // Data besar
  return 600000; // 10 menit
}
```

**Kasus Penggunaan Nyata:**
- `users:profile:{userId}`: cache profil pengguna dengan TTL 1 jam
- `products:detail:{productId}:{currency}`: cache harga produk dalam berbagai mata uang
- `orders:report:{year}:{month}:{branchId}`: cache laporan bulanan per cabang dengan TTL 24 jam
- `search:products:{keyword}:{page}:{filters}`: cache hasil pencarian dengan TTL pendek (5 menit)
- `config:{appId}:{environment}`: cache konfigurasi aplikasi dengan TTL panjang (6 jam) + refresh background
