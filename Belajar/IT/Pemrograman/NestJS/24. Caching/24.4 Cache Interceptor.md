# 24. Caching

## 24.4. Cache Interceptor

**Definisi Singkat:**
`CacheInterceptor` adalah interceptor bawaan NestJS yang secara otomatis melakukan caching untuk HTTP GET endpoint, menyimpan response berdasarkan URL dan mengembalikan cached response tanpa perlu menulis kode caching manual.

**Analogi:**
Seperti pelayan restoran yang hafal pesanan pelanggan tetap - ketika pelanggan yang sama datang lagi, pelayan langsung tahu pesanannya tanpa perlu mencatat ulang.

**Poin-Poin Penting:**
- Auto-cache semua GET endpoint secara default
- Menggunakan URL sebagai cache key
- Dapat di-override dengan `@CacheKey()` dan `@CacheTTL()`
- Dapat didaftarkan sebagai global interceptor
- Dapat di-extend untuk kustomisasi perilaku caching
- Hanya bekerja dengan HTTP GET requests secara default

**Contoh Kode:**
```typescript
// Mendaftarkan CacheInterceptor secara Global
import { Module } from '@nestjs/common';
import { APP_INTERCEPTOR } from '@nestjs/core';
import { CacheModule, CacheInterceptor } from '@nestjs/cache-manager';
import { ProductController } from './product.controller';
import { ProductService } from './product.service';

@Module({
  imports: [
    CacheModule.register({
      ttl: 120, // Default TTL 2 menit
      max: 100,
      isGlobal: true,
    }),
  ],
  controllers: [ProductController],
  providers: [
    ProductService,
    {
      provide: APP_INTERCEPTOR,
      useClass: CacheInterceptor,
    },
  ],
})
export class AppModule {}

// Controller dengan Cache Interceptor
import { Controller, Get, Post, Put, Delete, Param, Body } from '@nestjs/common';
import { CacheKey, CacheTTL } from '@nestjs/cache-manager';
import { ProductService } from './product.service';

@Controller('products')
export class ProductController {
  constructor(private readonly productService: ProductService) {}

  // Auto-cached dengan CacheInterceptor
  // Cache key: /products (dari URL)
  // Default TTL: 120 detik (dari module config)
  @Get()
  async findAll() {
    console.log('Executing findAll...');
    return this.productService.findAll();
  }

  // Custom cache key dan TTL
  @Get('featured')
  @CacheKey('products:featured')
  @CacheTTL(300) // 300 detik = 5 menit
  async getFeatured() {
    console.log('Executing getFeatured...');
    return this.productService.getFeatured();
  }

  // Cache dengan parameter
  @Get(':id')
  @CacheKey('product_{id}')
  @CacheTTL(600) // 10 menit
  async findOne(@Param('id') id: string) {
    console.log(`Executing findOne for id: ${id}...`);
    return this.productService.findOne(+id);
  }

  // POST/PUT/DELETE TIDAK di-cache secara otomatis
  // (CacheInterceptor hanya untuk GET)
  @Post()
  async create(@Body() data: any) {
    return this.productService.create(data);
  }

  @Put(':id')
  async update(@Param('id') id: string, @Body() data: any) {
    return this.productService.update(+id, data);
  }
}

// Custom Cache Interceptor
import {
  CacheInterceptor,
  CACHE_KEY_METADATA,
  CACHE_TTL_METADATA,
} from '@nestjs/cache-manager';
import {
  Injectable,
  ExecutionContext,
  Inject,
  CacheManager,
  CACHE_MANAGER,
} from '@nestjs/common';
import { Reflector } from '@nestjs/core';

@Injectable()
export class CustomCacheInterceptor extends CacheInterceptor {
  @Inject(CACHE_MANAGER) cacheManager: CacheManager;
  reflector: Reflector;

  // Override method untuk menentukan cache key
  trackBy(context: ExecutionContext): string | undefined {
    const httpAdapter = this.httpAdapterHost.httpAdapter;
    const request = context.switchToHttp().getRequest();
    const response = context.switchToHttp().getResponse();

    // Cek custom cache key dari metadata
    const cacheKey = this.reflector.get(
      CACHE_KEY_METADATA,
      context.getHandler(),
    );

    if (cacheKey) {
      // Ganti placeholder dengan parameter
      let key = cacheKey;
      const params = request.params;
      for (const [param, value] of Object.entries(params)) {
        key = key.replace(`{${param}}`, value);
      }
      return key;
    }

    // Default: gunakan URL sebagai cache key
    // Tambahkan juga user ID untuk user-specific cache
    const url = httpAdapter.getRequestUrl(request);
    const userId = request.user?.id;
    
    return userId ? `user:${userId}:${url}` : url;
  }

  // Override untuk menentukan TTL
  protected cachedTTL(context: ExecutionContext): number {
    const cacheTTL = this.reflector.get(
      CACHE_TTL_METADATA,
      context.getHandler(),
    );

    if (cacheTTL !== undefined) {
      return cacheTTL;
    }

    // Custom TTL berdasarkan endpoint
    const request = context.switchToHttp().getRequest();
    const url = request.url;

    // TTL lebih panjang untuk static data
    if (url.includes('/categories') || url.includes('/config')) {
      return 3600; // 1 jam
    }

    // TTL lebih pendek untuk data yang sering berubah
    if (url.includes('/search') || url.includes('/realtime')) {
      return 30; // 30 detik
    }

    // Default TTL dari module config
    return this.defaultTTL;
  }

  // Override untuk menentukan apakah response harus di-cache
  protected isRequestCacheable(context: ExecutionContext): boolean {
    const request = context.switchToHttp().getRequest();
    const method = request.method;

    // Hanya GET yang di-cache
    if (method !== 'GET') {
      return false;
    }

    // Jangan cache jika ada header 'no-cache'
    if (request.headers['cache-control'] === 'no-cache') {
      return false;
    }

    // Jangan cache untuk admin user (mereka butuh data terbaru)
    if (request.user?.role === 'admin') {
      return false;
    }

    return true;
  }
}

// Menggunakan Custom Cache Interceptor
import { Module } from '@nestjs/common';
import { APP_INTERCEPTOR } from '@nestjs/core';
import { CacheModule } from '@nestjs/cache-manager';
import { CustomCacheInterceptor } from './custom-cache.interceptor';

@Module({
  imports: [
    CacheModule.register({
      ttl: 120,
      isGlobal: true,
    }),
  ],
  providers: [
    {
      provide: APP_INTERCEPTOR,
      useClass: CustomCacheInterceptor,
    },
  ],
})
export class AppModule {}

// Contoh lengkap dengan Service dan Cache Invalidation
import { Injectable } from '@nestjs/common';
import { CacheKey, CacheTTL, CacheEvict } from '@nestjs/cache-manager';

@Injectable()
export class ProductService {
  private products = [
    { id: 1, name: 'Laptop', price: 15000000 },
    { id: 2, name: 'Mouse', price: 250000 },
  ];

  @CacheKey('products:all')
  @CacheTTL(120)
  async findAll() {
    console.log('Querying database for all products...');
    return this.products;
  }

  @CacheKey('products:{id}')
  @CacheTTL(300)
  async findOne(id: number) {
    console.log(`Querying database for product #${id}...`);
    return this.products.find(p => p.id === id);
  }

  // Hapus cache setelah create
  @CacheEvict({ key: 'products:all' })
  async create(data: any) {
    const newProduct = { id: this.products.length + 1, ...data };
    this.products.push(newProduct);
    return newProduct;
  }

  // Hapus beberapa cache setelah update
  @CacheEvict({ keys: ['products:all', 'products:{id}'] })
  async update(id: number, data: any) {
    const index = this.products.findIndex(p => p.id === id);
    if (index !== -1) {
      this.products[index] = { ...this.products[index], ...data };
    }
    return this.products[index];
  }

  // Hapus semua cache dengan pattern
  @CacheEvict({ pattern: 'products:*' })
  async clearAllCache() {
    return true;
  }
}
```

**Kasus Penggunaan Nyata:**
- Membuat blog CMS: semua artikel GET di-cache otomatis, ketika admin mengupdate artikel, cache di-invalidate secara otomatis
- API e-commerce: endpoint produk di-cache selama 5 menit, endpoint pencarian di-cache selama 30 detik
- Microservices: CacheInterceptor di API Gateway untuk mengurangi request ke service downstream
- User-specific cache: menambahkan user ID ke cache key agar setiap pengguna mendapatkan cache-nya sendiri
