# 32. Microservices
## 32.7 ClientProxy dan ClientModule
**Definisi Singkat:** `ClientProxy` adalah class abstrak untuk mengirim pesan/event ke microservice, sedangkan `ClientsModule` adalah module yang mendaftarkan dan mengkonfigurasi client-client tersebut dengan dependency injection.
**Analogi:** `ClientProxy` seperti telepon — alat untuk berkomunikasi. `ClientsModule` seperti buku telepon — mendaftarkan nomor (konfigurasi) siapa yang bisa dihubungi dan melalui jalur apa.
**Poin-Poin Penting:**
- `ClientsModule.register()` untuk registrasi client
- `@Inject('SERVICE_NAME') ClientProxy` untuk injeksi
- `client.send(pattern, data)` — request-response
- `client.emit(event, data)` — fire-and-forget
**Contoh Kode:**
```typescript
// clients.module.ts — registrasi multiple clients
import { Module } from '@nestjs/common';
import { ClientsModule, Transport } from '@nestjs/microservices';

@Module({
  imports: [
    ClientsModule.register([
      // TCP Client
      {
        name: 'USER_SERVICE',          // Token untuk @Inject()
        transport: Transport.TCP,
        options: {
          host: 'localhost',
          port: 3001,
        },
      },
      // Redis Client
      {
        name: 'NOTIFICATION_SERVICE',
        transport: Transport.REDIS,
        options: {
          host: 'localhost',
          port: 6379,
        },
      },
      // RabbitMQ Client
      {
        name: 'ORDER_SERVICE',
        transport: Transport.RMQ,
        options: {
          urls: ['amqp://guest:guest@localhost:5672'],
          queue: 'orders_queue',
          queueOptions: { durable: true },
          persistent: true,
          noAck: false,
        },
      },
      // Kafka Client
      {
        name: 'ANALYTICS_SERVICE',
        transport: Transport.KAFKA,
        options: {
          client: {
            brokers: ['localhost:9092'],
            clientId: 'api-gateway',
          },
          consumer: {
            groupId: 'analytics-consumer',
          },
        },
      },
    ]),
  ],
  exports: [ClientsModule],  // Export supaya module lain bisa inject
})
export class ServiceClientsModule {}
```
```typescript
// gateway.service.ts — menggunakan multiple clients
import { Injectable, Inject } from '@nestjs/common';
import { ClientProxy } from '@nestjs/microservices';
import { lastValueFrom } from 'rxjs';

@Injectable()
export class GatewayService {
  constructor(
    @Inject('USER_SERVICE') private userClient: ClientProxy,
    @Inject('NOTIFICATION_SERVICE') private notificationClient: ClientProxy,
    @Inject('ORDER_SERVICE') private orderClient: ClientProxy,
    @Inject('ANALYTICS_SERVICE') private analyticsClient: ClientProxy,
  ) {}

  async getUserProfile(userId: number) {
    // Request-response ke USER_SERVICE
    return lastValueFrom(
      this.userClient.send({ cmd: 'get_user' }, { id: userId })
    );
  }

  async createOrder(orderData: any) {
    // Request-response ke ORDER_SERVICE
    const order = await lastValueFrom(
      this.orderClient.send({ cmd: 'create_order' }, orderData)
    );

    // Fire-forget events (tidak perlu response)
    this.notificationClient.emit('order_confirmed', {
      orderId: order.id,
      userId: order.userId,
    });

    this.analyticsClient.emit('order_created', {
      orderId: order.id,
      amount: order.total,
    });

    return order;
  }

  async getDashboard(userId: number) {
    // Parallel calls — kirim ke multiple services sekaligus
    const [user, orders, notifications] = await Promise.all([
      lastValueFrom(this.userClient.send({ cmd: 'get_user' }, { id: userId })),
      lastValueFrom(this.orderClient.send({ cmd: 'get_user_orders' }, { userId })),
      lastValueFrom(this.notificationClient.send({ cmd: 'get_notifications' }, { userId })),
    ]);

    return { user, orders, notifications };
  }
}
```
```typescript
// Dynamic client — membuat ClientProxy secara manual
import { Injectable } from '@nestjs/common';
import { ClientProxy, ClientProxyFactory, Transport } from '@nestjs/microservices';

@Injectable()
export class DynamicClientService {
  private clients: Map<string, ClientProxy> = new Map();

  getClient(name: string, options: any): ClientProxy {
    if (!this.clients.has(name)) {
      this.clients.set(
        name,
        ClientProxyFactory.create({
          transport: Transport.TCP,
          options: options,
        }),
      );
    }
    return this.clients.get(name);
  }

  async sendToService(service: string, pattern: any, data: any) {
    const client = this.getClient(service, { host: service, port: 3001 });
    return client.send(pattern, data).toPromise();
  }
}
```

```typescript
// app.module.ts — registrasi dengan async options
import { Module } from '@nestjs/common';
import { ClientsModule, Transport } from '@nestjs/microservices';
import { ConfigService } from '@nestjs/config';

@Module({
  imports: [
    ClientsModule.registerAsync([
      {
        name: 'ORDER_SERVICE',
        useFactory: (config: ConfigService) => ({
          transport: Transport.RMQ,
          options: {
            urls: [config.get<string>('RABBITMQ_URL')],
            queue: config.get<string>('ORDER_QUEUE'),
          },
        }),
        inject: [ConfigService],
      },
    ]),
  ],
})
export class AppModule {}
```
**Kasus Penggunaan Nyata:** API Gateway service yang menerima request HTTP dari mobile app. Gateway menggunakan `ClientProxy` untuk berkomunikasi dengan 5 microservices backend — User Service, Product Service, Order Service, Payment Service, Notification Service. Gateway meng-agregasi data dari multiple services untuk satu response.
