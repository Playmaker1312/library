# 32. Microservices
## 32.10 Service Discovery pengenalan
**Definisi Singkat:** Service Discovery adalah mekanisme untuk microservices menemukan lokasi (host:port) service lain secara dinamis, tanpa hardcode URL, biasanya menggunakan registry seperti Consul, etcd, atau Kubernetes DNS.
**Analogi:** Seperti buku telepon — dulu Anda harus hafal nomor telepon teman (hardcode URL). Sekarang Anda buka kontak (service registry), cari nama teman (service name), dan dapat nomornya (IP:port) secara otomatis.
**Poin-Poin Penting:**
- Client-side discovery vs Server-side discovery
- Service registry: Consul, etcd, ZooKeeper
- Kubernetes: DNS-based discovery (service-name.namespace)
- NestJS: `@nestjs/microservices` tidak built-in — perlu library tambahan
**Contoh Kode:**
```typescript
// Tanpa Service Discovery — hardcode URL (masalah!)
// ❌ MASALAH: URL hardcode, harus update manual jika service pindah
const client = ClientProxyFactory.create({
  transport: Transport.TCP,
  options: {
    host: '192.168.1.100',  // Hardcode — fragile!
    port: 3001,
  },
});

// ✅ SOLUSI: Gunakan environment variables atau service registry
@Injectable()
export class DynamicClientFactory {
  constructor(private configService: ConfigService) {}

  createClient(serviceName: string): ClientProxy {
    // Baca dari environment variable: USER_SERVICE_HOST, USER_SERVICE_PORT
    const host = this.configService.get(`${serviceName}_HOST`);
    const port = this.configService.get(`${serviceName}_PORT`);

    if (!host || !port) {
      throw new Error(`Service ${serviceName} tidak ditemukan di konfigurasi`);
    }

    return ClientProxyFactory.create({
      transport: Transport.TCP,
      options: { host, port: parseInt(port, 10) },
    });
  }
}
```

```typescript
// consul.service.ts — Service Discovery dengan Consul
import { Injectable, OnModuleInit } from '@nestjs/common';
import * as Consul from 'consul';

@Injectable()
export class ConsulService implements OnModuleInit {
  private consul: Consul.Consul;
  private serviceCache: Map<string, { host: string; port: number }> = new Map();

  constructor() {
    this.consul = new Consul({
      host: process.env.CONSUL_HOST || 'localhost',
      port: parseInt(process.env.CONSUL_PORT || '8500', 10),
    });
  }

  async onModuleInit() {
    // Watch service changes
    this.watchServices();
  }

  // Register service ke Consul
  async registerService(name: string, port: number) {
    const id = `${name}-${port}-${Date.now()}`;
    await this.consul.agent.service.register({
      id,
      name,
      address: require('os').hostname(),
      port,
      check: {
        http: `http://localhost:${port}/health`,
        interval: '10s',
        timeout: '5s',
        deregistercriticalserviceafter: '30s',
      },
    });
    console.log(`[Consul] Service ${name} registered with id ${id}`);
  }

  // Dapatkan service address dari registry
  async getServiceAddress(name: string): Promise<{ host: string; port: number } | null> {
    // Cek cache dulu
    if (this.serviceCache.has(name)) {
      return this.serviceCache.get(name);
    }

    try {
      const services = await this.consul.catalog.service.nodes(name);
      if (services.length === 0) return null;

      // Pilih service (bisa load balance round-robin)
      const service = services[0];
      const address = {
        host: service.ServiceAddress || service.Address,
        port: service.ServicePort,
      };

      this.serviceCache.set(name, address);
      return address;
    } catch (error) {
      console.error(`[Consul] Gagal mendapatkan service ${name}:`, error.message);
      return null;
    }
  }

  // Watch perubahan service
  private watchServices() {
    this.consul.watch({
      method: this.consul.health.service,
      options: { passing: true },
    });
  }
}
```

```typescript
// kubernetes-discovery.service.ts — Kubernetes DNS-based discovery
import { Injectable } from '@nestjs/common';
import { ClientProxyFactory, Transport } from '@nestjs/microservices';
import * as dns from 'dns';

@Injectable()
export class KubernetesDiscoveryService {
  // Di Kubernetes, service discovery melalui DNS:
  // Format: <service-name>.<namespace>.svc.cluster.local
  // Contoh: user-service.default.svc.cluster.local

  async resolveService(serviceName: string, namespace = 'default'): Promise<string> {
    return new Promise((resolve, reject) => {
      const hostname = `${serviceName}.${namespace}.svc.cluster.local`;

      dns.resolveSrv(`_http._tcp.${hostname}`, (err, addresses) => {
        if (err) {
          // Fallback ke hostname langsung
          resolve(hostname);
          return;
        }
        if (addresses.length > 0) {
          resolve(addresses[0].name);
        } else {
          resolve(hostname);
        }
      });
    });
  }

  async createClientForService(serviceName: string, port = 3001): Promise<any> {
    const host = await this.resolveService(serviceName);

    return ClientProxyFactory.create({
      transport: Transport.TCP,
      options: { host, port },
    });
  }
}
```

```typescript
// service-registry.module.ts — abstraction untuk service registry
import { Module, Global } from '@nestjs/common';
import { ConfigService } from '@nestjs/config';
import { ClientProxyFactory, Transport, ClientProxy } from '@nestjs/microservices';

@Global()
@Module({
  providers: [
    {
      provide: 'SERVICE_REGISTRY',
      useFactory: (config: ConfigService) => {
        // Factory pattern — buat client sesuai environment
        const registryType = config.get('SERVICE_REGISTRY_TYPE', 'env');

        if (registryType === 'consul') {
          // Gunakan Consul
          return new ConsulService();
        } else if (registryType === 'kubernetes') {
          // Gunakan K8s DNS
          return new KubernetesDiscoveryService();
        } else {
          // Default: environment variables
          return {
            getClient: (name: string): ClientProxy => {
              const host = config.get(`${name.toUpperCase()}_HOST`, 'localhost');
              const port = config.get(`${name.toUpperCase()}_PORT`, '3001');
              return ClientProxyFactory.create({
                transport: Transport.TCP,
                options: { host, port: parseInt(port, 10) },
              });
            },
          };
        }
      },
      inject: [ConfigService],
    },
  ],
  exports: ['SERVICE_REGISTRY'],
})
export class ServiceRegistryModule {}
```
```typescript
// Penggunaan service registry di service
import { Injectable, Inject } from '@nestjs/common';
import { ClientProxy } from '@nestjs/microservices';

@Injectable()
export class OrderService {
  constructor(
    @Inject('SERVICE_REGISTRY') private registry: any,
  ) {}

  async getProductData(productId: number) {
    // Dapatkan client untuk product-service secara dinamis
    const productClient = this.registry.getClient('product_service');

    return productClient.send({ cmd: 'get_product' }, { id: productId }).toPromise();
  }
}
```
**Kasus Penggunaan Nyata:** Di Kubernetes, saat deploy `order-service` dengan 3 replica, masing-masing instance punya IP berbeda. Service discovery DNS (`order-service.default.svc.cluster.local`) secara otomatis me-load balance request ke salah satu replica. Jika ada replica yang restart dengan IP baru, DNS otomatis update tanpa perubahan kode.
