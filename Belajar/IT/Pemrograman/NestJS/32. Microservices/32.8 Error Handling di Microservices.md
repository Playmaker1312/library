# 32. Microservices
## 32.8 Error Handling di Microservices
**Definisi Singkat:** Error handling di microservices NestJS meliputi mekanisme retry, timeout, circuit breaker, dead letter queue, dan exception filters — memastikan kegagalan satu service tidak merusak seluruh sistem.
**Analogi:** Seperti sistem pengiriman paket — jika kurir tidak bisa mengantar (error), paket akan dicoba lagi besok (retry), jika terus gagal dikirim ke gudang penampungan (DLQ), dan pengirim diberi tahu (notification).
**Poin-Poin Penting:**
- `RpcException` untuk error messages
- Retry mechanism di client options
- Timeout handling
- RabbitMQ: dead letter exchange, ACK/NACK
- Circuit breaker pattern
**Contoh Kode:**
```typescript
// orders.controller.ts — RpcException dan error handling
import { Controller } from '@nestjs/common';
import { MessagePattern, Payload, RpcException, Ctx, RmqContext } from '@nestjs/microservices';

@Controller()
export class OrdersController {
  private readonly orders = new Map<number, any>();

  @MessagePattern({ cmd: 'get_order' })
  async getOrder(@Payload() data: { id: number }) {
    const order = this.orders.get(data.id);

    if (!order) {
      // 1️⃣ RpcException — error yang dikirim ke client
      throw new RpcException({
        statusCode: 404,
        message: `Order #${data.id} tidak ditemukan`,
        code: 'ORDER_NOT_FOUND',
        timestamp: new Date().toISOString(),
      });
    }

    return order;
  }

  @MessagePattern({ cmd: 'create_order' })
  async createOrder(@Payload() data: any) {
    try {
      // Validasi data
      if (!data.items || data.items.length === 0) {
        throw new RpcException({
          statusCode: 400,
          message: 'Order harus memiliki minimal 1 item',
          code: 'INVALID_ORDER',
        });
      }

      // Proses dengan kemungkinan error
      const order = { id: Date.now(), ...data, status: 'created' };
      this.orders.set(order.id, order);
      return order;

    } catch (error) {
      // 2️⃣ Wrap unexpected error
      if (error instanceof RpcException) throw error;
      throw new RpcException({
        statusCode: 500,
        message: 'Gagal membuat order',
        code: 'INTERNAL_ERROR',
      });
    }
  }

  // 3️⃣ RabbitMQ — manual ACK/NACK dengan error handling
  @MessagePattern('process_order')
  async processOrder(
    @Payload() data: any,
    @Ctx() context: RmqContext,
  ) {
    const channel = context.getChannelRef();
    const originalMsg = context.getMessage();

    try {
      // Proses order
      await this.validatePayment(data);
      await this.reserveInventory(data);

      // ✅ ACK — hapus dari queue
      channel.ack(originalMsg);
      return { success: true };

    } catch (error) {
      // ❌ NACK — kirim ke dead letter queue
      channel.nack(originalMsg, false, false);  // requeue = false → DLQ
      throw new RpcException({
        statusCode: 500,
        message: error.message,
        code: 'PROCESSING_FAILED',
      });
    }
  }
}
```

```typescript
// app.module.ts — client options dengan retry & timeout
import { Module } from '@nestjs/common';
import { ClientsModule, Transport } from '@nestjs/microservices';

@Module({
  imports: [
    ClientsModule.register([
      {
        name: 'ORDER_SERVICE',
        transport: Transport.RMQ,
        options: {
          urls: ['amqp://localhost:5672'],
          queue: 'orders_queue',
          queueOptions: {
            durable: true,
            // Dead Letter Exchange untuk pesan gagal
            deadLetterExchange: 'dlx',
            deadLetterRoutingKey: 'orders.dlx',
            // Message TTL (expired)
            messageTtl: 30000,  // 30 detik
            // Max queue length
            maxLength: 10000,
          },
          // Retry options
          reconnectAttempts: 10,
          reconnectDelay: 3000,  // 3 detik antar retry
          // Prefetch count
          prefetchCount: 5,
        },
      },
      {
        name: 'USER_SERVICE',
        transport: Transport.TCP,
        options: {
          host: 'localhost',
          port: 3001,
          // Retry untuk TCP
          retryAttempts: 3,
          retryDelay: 1000,
          // Timeout
          timeout: 5000,  // 5 detik timeout
        },
      },
    ]),
  ],
})
export class AppModule {}
```

```typescript
// client-error-handler.service.ts — error handling di client side
import { Injectable, Inject } from '@nestjs/common';
import { ClientProxy, RpcException } from '@nestjs/microservices';
import { lastValueFrom, timeout, catchError, retry, throwError } from 'rxjs';
import { Logger } from '@nestjs/common';

@Injectable()
export class ResilientClient {
  private readonly logger = new Logger(ResilientClient.name);
  private circuitOpen = false;
  private failureCount = 0;
  private readonly threshold = 5;

  constructor(@Inject('ORDER_SERVICE') private client: ClientProxy) {}

  async sendWithRetry(pattern: any, data: any, maxRetries = 3) {
    // Circuit breaker
    if (this.circuitOpen) {
      this.logger.warn('Circuit breaker OPEN — request ditolak');
      throw new Error('Service tidak tersedia (circuit breaker)');
    }

    try {
      const result = await lastValueFrom(
        this.client.send(pattern, data).pipe(
          retry(maxRetries),          // Retry otomatis
          timeout(10000),             // Timeout 10 detik
          catchError((error) => {
            this.logger.error(`Error: ${error.message}`);
            if (error.name === 'TimeoutError') {
              return throwError(() => new Error('Service timeout'));
            }
            // Handle RpcException
            if (error?.code === 'ORDER_NOT_FOUND') {
              return throwError(() => new Error(error.message));
            }
            return throwError(() => error);
          }),
        ),
      );

      // Success — reset circuit breaker
      this.failureCount = 0;
      return result;

    } catch (error) {
      this.failureCount++;

      // Open circuit jika terlalu banyak gagal
      if (this.failureCount >= this.threshold) {
        this.circuitOpen = true;
        this.logger.warn('Circuit breaker OPEN!');

        // Reset circuit setelah 30 detik
        setTimeout(() => {
          this.circuitOpen = false;
          this.failureCount = 0;
          this.logger.log('Circuit breaker CLOSED (reset)');
        }, 30000);
      }

      throw error;
    }
  }
}
```

```typescript
// exception-filter.ts — global RpcException filter
import { Catch, ArgumentsHost } from '@nestjs/common';
import { BaseRpcExceptionFilter } from '@nestjs/microservices';
import { Observable, throwError } from 'rxjs';

@Catch()
export class AllExceptionsFilter extends BaseRpcExceptionFilter {
  catch(exception: any, host: ArgumentsHost): Observable<any> {
    // Log semua error
    console.error('[RPC ERROR]', {
      message: exception.message,
      stack: exception.stack,
      timestamp: new Date().toISOString(),
    });

    return throwError(() => ({
      status: 'error',
      message: exception.message || 'Internal server error',
      timestamp: new Date().toISOString(),
    }));
  }
}
```
**Kasus Penggunaan Nyata:** Sistem pembayaran microservices. Jika `payment-service` sedang down, `order-service` tidak boleh langsung gagal. Dengan retry (3x), timeout (10 detik), dan dead letter queue, pesan pembayaran akan di-coba ulang. Jika tetap gagal, masuk DLQ untuk diproses manual oleh tim operasional.
