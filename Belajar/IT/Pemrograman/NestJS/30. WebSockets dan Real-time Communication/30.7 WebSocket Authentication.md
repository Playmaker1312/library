# 30. WebSockets dan Real-time Communication
## 30.7 WebSocket Authentication
**Definisi Singkat:** WebSocket authentication memvalidasi identitas client saat koneksi dibuat, biasanya menggunakan JWT token yang dikirim melalui handshake auth, lalu diverifikasi di middleware atau lifecycle hook.
**Analogi:** Seperti check-in di hotel — Anda harus menunjukkan kartu identitas (token JWT) sebelum mendapatkan kunci kamar (koneksi WebSocket). Resepsionis (middleware) memeriksa validitas identitas sebelum mengizinkan masuk.
**Poin-Poin Penting:**
- `socket.handshake.auth.token` untuk kirim token
- Middleware di `afterInit` via `server.use()`
- `socket.data` untuk menyimpan data user
- Guard juga bisa digunakan di gateway
**Contoh Kode:**
```typescript
// auth.gateway.ts — autentikasi WebSocket
import {
  WebSocketGateway,
  WebSocketServer,
  OnGatewayInit,
  OnGatewayConnection,
  OnGatewayDisconnect,
  SubscribeMessage,
  ConnectedSocket,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { Logger, UnauthorizedException } from '@nestjs/common';
import { JwtService } from '@nestjs/jwt';

@WebSocketGateway({
  cors: { origin: '*' },
  namespace: '/auth',
})
export class AuthGateway implements OnGatewayInit, OnGatewayConnection, OnGatewayDisconnect {
  private readonly logger = new Logger(AuthGateway.name);

  constructor(private jwtService: JwtService) {}

  @WebSocketServer()
  server: Server;

  // Middleware autentikasi — dijalankan SEBELUM koneksi diterima
  afterInit(server: Server) {
    server.use(async (socket, next) => {
      try {
        // Ekstrak token dari handshake
        const token = socket.handshake.auth.token
          || socket.handshake.headers.authorization?.replace('Bearer ', '')
          || socket.handshake.query.token as string;

        if (!token) {
          return next(new Error('Unauthorized: Token tidak ditemukan'));
        }

        // Verifikasi JWT
        const payload = await this.jwtService.verifyAsync(token, {
          secret: process.env.JWT_SECRET,
        });

        // Simpan data user di socket untuk digunakan handler
        socket.data.user = {
          id: payload.sub,
          email: payload.email,
          roles: payload.roles,
        };

        this.logger.log(`User terautentikasi: ${payload.email}`);
        next();
      } catch (error) {
        this.logger.warn(`Autentikasi gagal: ${error.message}`);
        next(new Error('Unauthorized: Token tidak valid'));
      }
    });
  }

  // Client yang terautentikasi akan sampai di sini
  handleConnection(client: Socket) {
    const user = client.data.user;
    this.logger.log(`User ${user.email} terhubung`);

    // Join room personal untuk direct messages
    client.join(`user:${user.id}`);

    // Kirim konfirmasi
    client.emit('authenticated', {
      user: { id: user.id, email: user.email },
      message: 'Autentikasi berhasil',
    });
  }

  handleDisconnect(client: Socket) {
    if (client.data.user) {
      this.logger.log(`User ${client.data.user.email} terputus`);
    }
  }

  // Method yang hanya bisa diakses user terautentikasi
  @SubscribeMessage('getProfile')
  handleGetProfile(@ConnectedSocket() client: Socket) {
    // Akses data user dari middleware
    return { user: client.data.user };
  }

  @SubscribeMessage('adminAction')
  handleAdminAction(@ConnectedSocket() client: Socket) {
    // Cek role — hanya admin
    if (!client.data.user.roles.includes('admin')) {
      return { error: 'Forbidden: Hanya untuk admin' };
    }
    return { success: true, message: 'Aksi admin berhasil' };
  }
}
```

```typescript
// ws-auth.guard.ts — WebSocket Guard (untuk gateway)
import { CanActivate, ExecutionContext, Injectable } from '@nestjs/common';
import { WsException } from '@nestjs/websockets';
import { Observable } from 'rxjs';

@Injectable()
export class WsAuthGuard implements CanActivate {
  canActivate(context: ExecutionContext): boolean | Promise<boolean> | Observable<boolean> {
    const client = context.switchToWs().getClient();
    const user = client.data.user;

    if (!user) {
      throw new WsException('Unauthorized');
    }

    return true;
  }
}
```

```typescript
// gateway.with.guard.ts — Guard di WebSocket Gateway (NestJS 10+)
import {
  WebSocketGateway,
  SubscribeMessage,
  MessageBody,
  ConnectedSocket,
} from '@nestjs/websockets';
import { UseGuards } from '@nestjs/common';
import { WsAuthGuard } from './ws-auth.guard';
import { Socket } from 'socket.io';

@WebSocketGateway({ namespace: '/protected', cors: { origin: '*' } })
export class ProtectedGateway {
  @UseGuards(WsAuthGuard)  // Guard di level method
  @SubscribeMessage('sensitiveData')
  handleSensitiveData(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: any,
  ) {
    // Hanya user terautentikasi yang bisa akses
    return { data: 'sensitive', user: client.data.user };
  }
}
```
