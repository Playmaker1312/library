# 30. WebSockets dan Real-time Communication
## 30.5 Broadcasting Messages
**Definisi Singkat:** Broadcasting adalah mekanisme mengirim pesan ke satu atau banyak client WebSocket, bisa ke semua client (`server.emit()`), semua kecuali pengirim (`client.broadcast.emit()`), atau spesifik room (`server.to(room).emit()`).
**Analogi:** Seperti sistem pengumuman di bandara — ada pengumuman ke seluruh terminal (broadcast global), ke terminal tertentu (room), atau pesan pribadi ke petugas tertentu (direct message).
**Poin-Poin Penting:**
- `server.emit()` — broadcast ke semua client
- `client.broadcast.emit()` — semua client kecuali pengirim
- `server.to(room).emit()` — client dalam room tertentu
- `client.emit()` — hanya ke client spesifik
**Contoh Kode:**
```typescript
// broadcast.gateway.ts — berbagai cara broadcast
import {
  WebSocketGateway,
  WebSocketServer,
  SubscribeMessage,
  ConnectedSocket,
  MessageBody,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';

@WebSocketGateway({
  cors: { origin: '*' },
  namespace: '/broadcast',
})
export class BroadcastGateway {
  @WebSocketServer()
  server: Server;

  // 1️⃣ Global broadcast — ke SEMUA client yang terhubung
  // Semua client di namespace ini akan menerima pesan
  sendGlobalAnnouncement(title: string, message: string) {
    this.server.emit('announcement', {
      title,
      message,
      timestamp: new Date().toISOString(),
    });
  }

  // 2️⃣ Broadcast exclude sender — client.broadcast
  // Client yang mengirim tidak akan menerima pesan
  @SubscribeMessage('broadcast')
  handleBroadcast(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { message: string; username: string },
  ) {
    // Kirim ke semua client KECUALI pengirim
    client.broadcast.emit('broadcastMessage', {
      username: payload.username,
      message: payload.message,
      timestamp: new Date().toISOString(),
    });

    // Konfirmasi ke pengirim bahwa pesan sudah terkirim
    client.emit('broadcastSent', { status: 'ok' });
  }

  // 3️⃣ Room broadcast — hanya client dalam room tertentu
  @SubscribeMessage('sendToRoom')
  handleSendToRoom(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { room: string; message: string; username: string },
  ) {
    // Kirim ke semua client di room (termasuk pengirim jika di room)
    this.server.to(payload.room).emit('roomMessage', {
      username: payload.username,
      message: payload.message,
      room: payload.room,
      timestamp: new Date().toISOString(),
    });
  }

  // 4️⃣ Kirim ke room, kecuali pengirim
  @SubscribeMessage('sendToRoomExcludeSender')
  handleSendToRoomExclude(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { room: string; message: string },
  ) {
    // broadcast.to() — kirim ke semua client di room KECUALI pengirim
    client.to(payload.room).emit('roomMessage', {
      message: payload.message,
      sender: client.id,
    });
  }

  // 5️⃣ Direct message — ke client spesifik
  @SubscribeMessage('directMessage')
  handleDirectMessage(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { to: string; message: string },
  ) {
    // Kirim hanya ke client dengan ID tertentu
    this.server.to(payload.to).emit('directMessage', {
      from: client.id,
      message: payload.message,
      timestamp: new Date().toISOString(),
    });

    // Notifikasi ke pengirim
    client.emit('directMessageSent', {
      to: payload.to,
      status: 'delivered',
    });
  }

  // 6️⃣ Volatile event — pesan yang boleh hilang (tidak guarantee sampai)
  sendVolatileUpdate(data: any) {
    // Volatile: jika client sedang terputus, pesan tidak akan di-cache
    this.server.volatile.emit('liveUpdate', data);
  }

  // 7️⃣ Compressed event — untuk data besar
  sendLargeData(data: any) {
    // Gunakan compression untuk data besar
    this.server.emit('largeData', data, { compress: true });
  }

  // 8️⃣ Broadcast dengan callback — konfirmasi diterima
  @SubscribeMessage('broadcastWithAck')
  handleBroadcastWithAck(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { message: string },
  ) {
    // Broadcast dan minta acknowledgment
    this.server.emit('requireAck', { message: payload.message });
    return { status: 'broadcasting', message: payload.message };
  }
}
```
