# 30. WebSockets dan Real-time Communication
## 30.6 Rooms dan Namespaces
**Definisi Singkat:** Rooms adalah sub-kanal dalam satu koneksi WebSocket yang memungkinkan pengelompokan client untuk broadcast terarah; Namespaces adalah kanal terpisah yang memisahkan concern aplikasi dalam satu server.
**Analogi:** Rooms seperti ruang rapat dalam satu gedung — Anda hanya mendengar percakapan di ruangan Anda. Namespaces seperti gedung terpisah — `chat` dan `analytics` adalah gedung berbeda dengan sistem keamanan sendiri.
**Poin-Poin Penting:**
- `client.join(room)` — masuk room
- `client.leave(room)` — keluar room
- `server.to(room).emit()` — broadcast ke room
- `@WebSocketGateway({ namespace: '/chat' })` — namespace
**Contoh Kode:**
```typescript
// room.gateway.ts — manajemen rooms
import {
  WebSocketGateway,
  WebSocketServer,
  SubscribeMessage,
  ConnectedSocket,
  MessageBody,
} from '@nestjs/websockets';
import { Server, Socket } from 'socket.io';
import { Logger } from '@nestjs/common';

@WebSocketGateway({
  cors: { origin: '*' },
  namespace: '/chat',
})
export class RoomGateway {
  private readonly logger = new Logger(RoomGateway.name);

  @WebSocketServer()
  server: Server;

  // JOIN room
  @SubscribeMessage('joinRoom')
  handleJoinRoom(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { roomId: string; password?: string },
  ) {
    // Cek apakah room terproteksi password
    // (implementasi sederhana)
    client.join(payload.roomId);
    this.logger.log(`Client ${client.id} join room ${payload.roomId}`);

    // Kirim notifikasi ke room
    this.server.to(payload.roomId).emit('userJoined', {
      clientId: client.id,
      roomId: payload.roomId,
      onlineInRoom: this.getRoomSize(payload.roomId),
    });

    return {
      success: true,
      room: payload.roomId,
      onlineInRoom: this.getRoomSize(payload.roomId),
    };
  }

  // LEAVE room
  @SubscribeMessage('leaveRoom')
  handleLeaveRoom(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { roomId: string },
  ) {
    client.leave(payload.roomId);
    this.logger.log(`Client ${client.id} leave room ${payload.roomId}`);

    this.server.to(payload.roomId).emit('userLeft', {
      clientId: client.id,
      roomId: payload.roomId,
      onlineInRoom: this.getRoomSize(payload.roomId),
    });
  }

  // GET available rooms
  @SubscribeMessage('getRooms')
  handleGetRooms(@ConnectedSocket() client: Socket) {
    const rooms = Array.from(this.server.sockets.adapter.rooms.keys())
      .filter((room) => !room.startsWith(client.id))  // Filter default room
      .map((room) => ({
        id: room,
        clients: this.getRoomSize(room),
      }));

    return { rooms };
  }

  // LIST all rooms + client count
  @SubscribeMessage('getRoomInfo')
  handleGetRoomInfo(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { roomId: string },
  ) {
    const room = this.server.sockets.adapter.rooms.get(payload.roomId);
    if (!room) {
      return { exists: false };
    }

    return {
      exists: true,
      roomId: payload.roomId,
      clientCount: room.size,
      clients: Array.from(room),
    };
  }

  // JOIN multiple rooms sekaligus
  @SubscribeMessage('joinMultipleRooms')
  handleJoinMultipleRooms(
    @ConnectedSocket() client: Socket,
    @MessageBody() payload: { roomIds: string[] },
  ) {
    payload.roomIds.forEach((roomId) => client.join(roomId));
    return { joined: payload.roomIds };
  }

  // LEAVE all rooms kecuali default room (client.id)
  @SubscribeMessage('leaveAllRooms')
  handleLeaveAllRooms(@ConnectedSocket() client: Socket) {
    const rooms = Array.from(client.rooms);
    rooms.forEach((room) => {
      if (room !== client.id) {
        client.leave(room);
      }
    });
    return { left: rooms.filter((r) => r !== client.id) };
  }

  private getRoomSize(roomId: string): number {
    const room = this.server.sockets.adapter.rooms.get(roomId);
    return room ? room.size : 0;
  }
}
```

```typescript
// namespace-setup.ts — multiple namespaces
import { WebSocketGateway, WebSocketServer } from '@nestjs/websockets';
import { Server } from 'socket.io';

// Namespace 'chat' — untuk messaging
@WebSocketGateway({ namespace: '/chat', cors: { origin: '*' } })
export class ChatNamespaceGateway {
  @WebSocketServer()
  server: Server;
}

// Namespace 'notifications' — untuk notifikasi
@WebSocketGateway({ namespace: '/notifications', cors: { origin: '*' } })
export class NotificationNamespaceGateway {
  @WebSocketServer()
  server: Server;
}

// Namespace 'analytics' — untuk data dashboard
@WebSocketGateway({ namespace: '/analytics', cors: { origin: '*' } })
export class AnalyticsNamespaceGateway {
  @WebSocketServer()
  server: Server;
}
```
