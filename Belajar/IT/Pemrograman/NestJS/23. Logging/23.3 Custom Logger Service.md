# 23. Logging

## 23.3. Custom Logger Service

**Definisi Singkat:**
Custom Logger Service adalah implementasi logger sendiri yang mengimplementasikan `LoggerService` interface, memungkinkan kustomisasi penuh format output, transport, dan perilaku logging.

**Analogi:**
Mengganti buku catatan standar dengan sistem pencatatan otomatis yang bisa mengirim pesan ke Slack, menyimpan ke database, dan memformat pesan sesuai kebutuhan perusahaan.

**Poin-Poin Penting:**
- Implementasikan `LoggerService` interface dari `@nestjs/common`
- Dapat di-inject sebagai provider
- Dapat dijadikan logger global di `main.ts`
- Metode yang harus diimplementasikan: `log()`, `error()`, `warn()`, `debug()`, `verbose()`
- Dapat meng-extend `ConsoleLogger` untuk mempertahankan perilaku default

**Contoh Kode:**
```typescript
// my-logger.service.ts
import { Injectable, LoggerService, LogLevel, ConsoleLogger } from '@nestjs/common';

@Injectable()
export class MyLogger extends ConsoleLogger {
  constructor() {
    super();
  }

  log(message: string, context?: string) {
    // Tambahkan kustomisasi sebelum menulis log
    const timestamp = new Date().toISOString();
    const formattedMessage = `[${timestamp}] [LOG] ${message}`;
    super.log(formattedMessage, context);
  }

  error(message: string, trace?: string, context?: string) {
    const timestamp = new Date().toISOString();
    const formattedMessage = `[${timestamp}] [ERROR] ${message}`;
    super.error(formattedMessage, trace, context);
    
    // Bisa tambahkan: kirim alert ke Slack/email
    this.sendAlertToSlack(formattedMessage, trace);
  }

  warn(message: string, context?: string) {
    const timestamp = new Date().toISOString();
    const formattedMessage = `[${timestamp}] [WARN] ${message}`;
    super.warn(formattedMessage, context);
  }

  debug(message: string, context?: string) {
    if (process.env.NODE_ENV !== 'production') {
      const timestamp = new Date().toISOString();
      const formattedMessage = `[${timestamp}] [DEBUG] ${message}`;
      super.debug(formattedMessage, context);
    }
  }

  verbose(message: string, context?: string) {
    if (process.env.NODE_ENV === 'development') {
      const timestamp = new Date().toISOString();
      const formattedMessage = `[${timestamp}] [VERBOSE] ${message}`;
      super.verbose(formattedMessage, context);
    }
  }

  private sendAlertToSlack(message: string, trace?: string) {
    // Implementasi pengiriman alert ke Slack
    console.log('Sending alert to Slack:', message);
  }
}

// Implementasi lengkap LoggerService interface
import { Injectable, LoggerService } from '@nestjs/common';

@Injectable()
export class CustomLogger implements LoggerService {
  private context?: string;

  setContext(context: string) {
    this.context = context;
  }

  log(message: any, ...optionalParams: any[]) {
    const context = optionalParams[optionalParams.length - 1] || this.context;
    console.log(`[${new Date().toISOString()}] [LOG] [${context}] ${message}`);
  }

  error(message: any, ...optionalParams: any[]) {
    const trace = optionalParams.length > 1 ? optionalParams[0] : undefined;
    const context = optionalParams[optionalParams.length - 1] || this.context;
    console.error(`[${new Date().toISOString()}] [ERROR] [${context}] ${message}`);
    if (trace) {
      console.error(trace);
    }
  }

  warn(message: any, ...optionalParams: any[]) {
    const context = optionalParams[optionalParams.length - 1] || this.context;
    console.warn(`[${new Date().toISOString()}] [WARN] [${context}] ${message}`);
  }

  debug?(message: any, ...optionalParams: any[]) {
    const context = optionalParams[optionalParams.length - 1] || this.context;
    console.debug(`[${new Date().toISOString()}] [DEBUG] [${context}] ${message}`);
  }

  verbose?(message: any, ...optionalParams: any[]) {
    const context = optionalParams[optionalParams.length - 1] || this.context;
    console.log(`[${new Date().toISOString()}] [VERBOSE] [${context}] ${message}`);
  }

  fatal?(message: any, ...optionalParams: any[]) {
    const context = optionalParams[optionalParams.length - 1] || this.context;
    console.error(`[${new Date().toISOString()}] [FATAL] [${context}] ${message}`);
    process.exit(1);
  }
}

// Menggunakan Custom Logger di main.ts
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { MyLogger } from './my-logger.service';

async function bootstrap() {
  const app = await NestFactory.create(AppModule, {
    logger: new MyLogger(),  // Gunakan custom logger global
  });

  // Atau menggunakan DI untuk logger
  // const logger = app.get<MyLogger>(MyLogger);
  // app.useLogger(logger);

  await app.listen(3000);
}
bootstrap();

// Module configuration
import { Module } from '@nestjs/common';
import { MyLogger } from './my-logger.service';

@Module({
  providers: [MyLogger],
  exports: [MyLogger],
})
export class LoggerModule {}

// Menggunakan di Service
import { Injectable } from '@nestjs/common';
import { MyLogger } from './my-logger.service';

@Injectable()
export class UserService {
  constructor(private logger: MyLogger) {
    this.logger.setContext(UserService.name);
  }

  findAll() {
    this.logger.log('Fetching all users');
  }
}
```

**Kasus Penggunaan Nyata:**
Membuat logger yang secara otomatis mengirim error kritis ke Slack channel tim engineering, menyimpan semua log ke file dengan format JSON terstruktur, dan menambahkan `requestId` pada setiap log untuk tracing request di microservices architecture.
