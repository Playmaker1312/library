# 23. Logging

## 23.5. Request Logging Middleware

**Definisi Singkat:**
Request Logging Middleware adalah middleware NestJS yang mencatat setiap permintaan HTTP yang masuk, mencakup informasi seperti method, endpoint, status code, response time, dan metadata request lainnya.

**Analogi:**
Seperti petugas keamanan di gedung yang mencatat setiap orang yang masuk (siapa, kapan, tujuan, berapa lama di dalam).

**Poin-Poin Penting:**
- Mengimplementasikan `NestMiddleware` interface
- Dapat diterapkan secara global atau untuk route tertentu
- Biasa mencatat: method, URL, status code, response time, IP address
- Dapat diintegrasikan dengan Correlation ID untuk tracing
- Dapat mengecualikan route tertentu (seperti health check)

**Contoh Kode:**
```typescript
// request-logging.middleware.ts
import { Injectable, NestMiddleware, Logger } from '@nestjs/common';
import { Request, Response, NextFunction } from 'express';

@Injectable()
export class RequestLoggingMiddleware implements NestMiddleware {
  private logger = new Logger('HTTP');

  use(req: Request, res: Response, next: NextFunction) {
    const { method, originalUrl, ip } = req;
    const userAgent = req.get('user-agent') || '';
    const startTime = Date.now();

    // Response listener untuk mencatat setelah response dikirim
    res.on('finish', () => {
      const { statusCode } = res;
      const contentLength = res.get('content-length');
      const responseTime = Date.now() - startTime;

      const logMessage = `${method} ${originalUrl} ${statusCode} ${contentLength || 0}bytes ${responseTime}ms - ${userAgent} ${ip}`;

      // Tentukan log level berdasarkan status code
      if (statusCode >= 500) {
        this.logger.error(logMessage);
      } else if (statusCode >= 400) {
        this.logger.warn(logMessage);
      } else {
        this.logger.log(logMessage);
      }
    });

    next();
  }
}

// request-logging.module.ts - dengan correlation ID
import { Injectable, NestMiddleware, Logger } from '@nestjs/common';
import { Request, Response, NextFunction } from 'express';
import { randomUUID } from 'crypto';

// Extend Express Request type untuk correlation ID
declare global {
  namespace Express {
    interface Request {
      correlationId: string;
    }
  }
}

@Injectable()
export class CorrelationIdMiddleware implements NestMiddleware {
  use(req: Request, res: Response, next: NextFunction) {
    // Ambil correlation ID dari header, atau buat baru
    const correlationId = req.headers['x-correlation-id'] as string || randomUUID();
    
    req.correlationId = correlationId;
    
    // Set di response header juga
    res.setHeader('X-Correlation-ID', correlationId);
    
    next();
  }
}

@Injectable()
export class AdvancedRequestLoggingMiddleware implements NestMiddleware {
  private logger = new Logger('HTTP');

  use(req: Request, res: Response, next: NextFunction) {
    const { method, originalUrl, ip, correlationId } = req;
    const userAgent = req.get('user-agent') || '';
    const startTime = Date.now();

    // Log request masuk
    this.logger.verbose(
      `[${correlationId}] Incoming request: ${method} ${originalUrl}`,
    );

    // Log body (opsional, hati-hati dengan sensitif data)
    if (Object.keys(req.body || {}).length > 0) {
      const safeBody = this.sanitizeSensitiveData({ ...req.body });
      this.logger.debug(
        `[${correlationId}] Request body: ${JSON.stringify(safeBody)}`,
      );
    }

    res.on('finish', () => {
      const { statusCode } = res;
      const responseTime = Date.now() - startTime;

      const logData = {
        correlationId,
        method,
        url: originalUrl,
        statusCode,
        responseTime: `${responseTime}ms`,
        ip,
        userAgent: userAgent.substring(0, 100),
      };

      const logMessage = `[${correlationId}] ${method} ${originalUrl} ${statusCode} - ${responseTime}ms`;

      if (statusCode >= 500) {
        this.logger.error(logMessage, JSON.stringify(logData));
      } else if (statusCode >= 400) {
        this.logger.warn(logMessage);
      } else {
        this.logger.log(logMessage);
      }
    });

    next();
  }

  private sanitizeSensitiveData(body: any): any {
    const sensitiveFields = ['password', 'token', 'secret', 'creditCard'];
    
    for (const field of sensitiveFields) {
      if (body[field]) {
        body[field] = '***MASKED***';
      }
    }
    
    return body;
  }
}

// Menerapkan middleware di module
import { Module, NestModule, MiddlewareConsumer, RequestMethod } from '@nestjs/common';
import { RequestLoggingMiddleware, CorrelationIdMiddleware, AdvancedRequestLoggingMiddleware } from './request-logging.middleware';
import { AppController } from './app.controller';
import { AppService } from './app.service';

@Module({
  controllers: [AppController],
  providers: [AppService],
})
export class AppModule implements NestModule {
  configure(consumer: MiddlewareConsumer) {
    consumer
      // Apply correlation ID middleware dulu
      .apply(CorrelationIdMiddleware)
      .forRoutes('*')
      // Apply logging middleware
      .apply(AdvancedRequestLoggingMiddleware)
      .exclude(
        { path: 'health', method: RequestMethod.GET },
        { path: 'metrics', method: RequestMethod.GET },
      )
      .forRoutes('*');
  }
}

// Atau apply secara global di main.ts
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { RequestLoggingMiddleware } from './request-logging.middleware';

async function bootstrap() {
  const app = await NestFactory.create(AppModule);
  
  // Apply middleware secara global
  app.use(new RequestLoggingMiddleware().use);
  
  await app.listen(3000);
}
bootstrap();
```

**Kasus Penggunaan Nyata:**
Mencatat setiap request API untuk keperluan audit dan debugging, menambahkan correlation ID untuk melacak request antar microservices, mengukur response time setiap endpoint untuk performance monitoring, dan mendeteksi pola traffic yang mencurigakan (seperti brute force attack).
