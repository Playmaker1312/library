# 23. Logging

## 23.4. Integrasi Winston & Pino

**Definisi Singkat:**
Winston dan Pino adalah library logging populer untuk Node.js yang menawarkan fitur lebih lengkap seperti multiple transport, formatting, dan structured logging. NestJS dapat diintegrasikan dengan keduanya.

**Analogi:**
Winston seperti perpustakaan alat tulis lengkap dengan berbagai pena dan kertas, sementara Pino seperti pena bolpoin super cepat yang dirancang untuk kinerja tinggi.

**Poin-Poin Penting:**
- Winston: Paling populer, fleksibel, banyak plugin
- Pino: Sangat cepat (ekstrem), JSON out-of-the-box, low overhead
- Package `nest-winston` untuk integrasi Winston dengan NestJS
- Package `nestjs-pino` untuk integrasi Pino dengan NestJS
- Mendukung transport: Console, File, HTTP, Database, Cloud services

**Contoh Kode:**
```typescript
// ========== WINSTON INTEGRATION ==========
// Install: npm install nest-winston winston

// winston-logger.module.ts
import { Module } from '@nestjs/common';
import { WinstonModule, utilities } from 'nest-winston';
import * as winston from 'winston';

@Module({
  imports: [
    WinstonModule.forRoot({
      // Format log
      format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.ms(),
        utilities.format.nestLike('MyApp', {
          colors: true,
          prettyPrint: true,
        }),
      ),
      
      // Transport (tempat output log)
      transports: [
        // Console transport
        new winston.transports.Console({
          level: process.env.NODE_ENV === 'production' ? 'info' : 'debug',
        }),
        
        // File transport untuk error
        new winston.transports.File({
          filename: 'logs/error.log',
          level: 'error',
          format: winston.format.combine(
            winston.format.timestamp(),
            winston.format.json(),
          ),
        }),
        
        // File transport untuk semua log
        new winston.transports.File({
          filename: 'logs/combined.log',
          format: winston.format.combine(
            winston.format.timestamp(),
            winston.format.json(),
          ),
        }),
      ],
    }),
  ],
  exports: [WinstonModule],
})
export class WinstonLoggerModule {}

// Menggunakan Winston di Service
import { Injectable } from '@nestjs/common';
import { InjectLogger } from '@nestjs/common';
import { WINSTON_MODULE_PROVIDER } from 'nest-winston';
import { Logger } from 'winston';

@Injectable()
export class ProductService {
  constructor(
    @InjectLogger() private readonly logger: Logger,
  ) {}

  async createProduct(data: any) {
    this.logger.info('Creating product', { 
      data: { name: data.name, price: data.price },
      correlationId: 'req-123',
    });

    try {
      // Logic create product
      this.logger.info('Product created successfully', { productId: 1 });
    } catch (error) {
      this.logger.error('Failed to create product', {
        error: error.message,
        stack: error.stack,
      });
    }
  }
}

// ========== PINO INTEGRATION ==========
// Install: npm install nestjs-pino pino pino-pretty

// pino-logger.module.ts
import { Module } from '@nestjs/common';
import { LoggerModule } from 'nestjs-pino';

@Module({
  imports: [
    LoggerModule.forRoot({
      pinoHttp: {
        level: process.env.NODE_ENV === 'production' ? 'info' : 'debug',
        transport: process.env.NODE_ENV === 'production'
          ? undefined  // Production: JSON tanpa pretty print
          : {
              target: 'pino-pretty',
              options: {
                colorize: true,
                levelFirst: true,
                translateTime: 'yyyy-dd-mm, h:MM:ss TT',
              },
            },
        customProps: (req, res) => ({
          context: 'HTTP',
          requestId: req.headers['x-request-id'],
        }),
        customLogLevel: (req, res, err) => {
          if (res.statusCode >= 400 && res.statusCode < 500) {
            return 'warn';
          }
          if (res.statusCode >= 500 || err) {
            return 'error';
          }
          return 'info';
        },
      },
    }),
  ],
  exports: [LoggerModule],
})
export class PinoLoggerModule {}

// Menggunakan Pino Logger
import { Injectable } from '@nestjs/common';
import { Logger } from 'nestjs-pino';

@Injectable()
export class OrderService {
  constructor(private readonly logger: Logger) {}

  async processOrder(orderId: string) {
    this.logger.info({ orderId }, 'Processing order');
    
    this.logger.debug({ orderId, step: 1 }, 'Validating order');
    
    try {
      // Logic
      this.logger.info({ orderId, duration: 150 }, 'Order processed successfully');
    } catch (error) {
      this.logger.error({ orderId, err: error }, 'Order processing failed');
    }
  }
}

// main.ts - Menggunakan Winston sebagai logger global
import { NestFactory } from '@nestjs/core';
import { AppModule } from './app.module';
import { WinstonModule } from 'nest-winston';
import * as winston from 'winston';

async function bootstrap() {
  const app = await NestFactory.create(AppModule, {
    logger: WinstonModule.createLogger({
      format: winston.format.combine(
        winston.format.timestamp(),
        winston.format.json(),
      ),
      transports: [
        new winston.transports.Console(),
      ],
    }),
  });

  await app.listen(3000);
}
bootstrap();
```

**Kasus Penggunaan Nyata:**
Menggunakan Winston untuk aplikasi yang membutuhkan fleksibilitas tinggi dengan banyak transport (kirim log ke Elasticsearch, CloudWatch, dan file secara bersamaan). Menggunakan Pino untuk microservices yang membutuhkan kinerja logging ekstrem dengan overhead minimal, khususnya pada environment dengan traffic tinggi.
