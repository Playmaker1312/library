# 89. Type Predicates - pola lanjutan

**Definisi Singkat:** Pola lanjutan type predicates mencakup teknik validasi bertingkat, discriminated union dengan field dinamis, filter array dengan type guard, dan type narrowing kompleks di berbagai skenario nyata.

**Analogi:** Seperti pemeriksaan keamanan bandara bertingkat — pertama periksa ID (type guard level 1), lalu periksa boarding pass (level 2), lalu cocokkan dengan daftar penumpang (level 3) — setiap level mempersempit identitas penumpang.

**Poin-Poin Penting:**
- Array.filter dengan type predicate bisa mempersempit tipe array
- Type guard bisa dikomposisikan (bersarang)
- `this is Type` di class method untuk narrowing instance type
- Type guard untuk tipe bersarang (nested objects, recursive types)

**Contoh Kode:**
```typescript
// BENAR: pola lanjutan type predicates

// 1. Array filter dengan type predicate
interface SuccessResult<T> {
  ok: true;
  data: T;
}
interface ErrorResult {
  ok: false;
  error: string;
}

type Result<T> = SuccessResult<T> | ErrorResult;

function isSuccess<T>(result: Result<T>): result is SuccessResult<T> {
  return result.ok === true;
}

function isError<T>(result: Result<T>): result is ErrorResult {
  return !result.ok;
}

const results: Result<number>[] = [
  { ok: true, data: 42 },
  { ok: false, error: "gagal" },
  { ok: true, data: 100 },
];

// Filter menghasilkan tipe yang tepat
const successes: SuccessResult<number>[] = results.filter(isSuccess);
const errors: ErrorResult[] = results.filter(isError);

successes.forEach((s) => console.log(s.data.toFixed(2))); // OK: number
errors.forEach((e) => console.log(e.error.toUpperCase())); // OK: string

// 2. Nested type guard untuk objek kompleks
interface Address {
  street: string;
  city: string;
  zipCode: string;
}

interface UserWithAddress {
  id: number;
  name: string;
  address: Address;
}

interface UserWithoutAddress {
  id: number;
  name: string;
  address?: undefined;
}

type MyUser = UserWithAddress | UserWithoutAddress;

function hasAddress(user: MyUser): user is UserWithAddress {
  return user.address !== undefined && typeof user.address.street === "string";
}

function getCity(user: MyUser): string {
  if (hasAddress(user)) {
    // Di sini user adalah UserWithAddress
    return user.address.city;
  }
  return "Alamat tidak tersedia";
}

// 3. Type guard dengan discriminated union dinamis
type Shape =
  | { kind: "circle"; radius: number }
  | { kind: "rectangle"; width: number; height: number }
  | { kind: "triangle"; base: number; height: number };

const shapeGuards = {
  circle: (s: Shape): s is Extract<Shape, { kind: "circle" }> =>
    s.kind === "circle",
  rectangle: (s: Shape): s is Extract<Shape, { kind: "rectangle" }> =>
    s.kind === "rectangle",
  triangle: (s: Shape): s is Extract<Shape, { kind: "triangle" }> =>
    s.kind === "triangle",
};

function calculateArea(shape: Shape): number {
  if (shapeGuards.circle(shape)) return Math.PI * shape.radius ** 2;
  if (shapeGuards.rectangle(shape)) return shape.width * shape.height;
  if (shapeGuards.triangle(shape)) return (shape.base * shape.height) / 2;
  throw new Error("Unknown shape");
}

// 4. Type guard untuk tipe recursive
type TreeNode<T> =
  | { type: "leaf"; value: T }
  | { type: "branch"; children: TreeNode<T>[] };

function isLeaf<T>(node: TreeNode<T>): node is TreeNode<T> & { type: "leaf" } {
  return node.type === "leaf";
}

function isBranch<T>(node: TreeNode<T>): node is TreeNode<T> & { type: "branch" } {
  return node.type === "branch";
}

function getLeafValues<T>(node: TreeNode<T>): T[] {
  if (isLeaf(node)) return [node.value];
  return node.children.flatMap(getLeafValues);
}

// ERROR: type predicate harus boolean function
```
