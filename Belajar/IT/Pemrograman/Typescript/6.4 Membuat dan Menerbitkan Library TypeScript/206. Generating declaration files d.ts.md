# 206. Generating declaration files d.ts

**Definisi Singkat:** Cara menghasilkan file `.d.ts` (declaration files) dari kode TypeScript, yang berisi type definitions untuk digunakan oleh pengguna library.

**Analogi:** Seperti menu restoran - declaration file adalah menu yang menunjukkan apa yang tersedia (types, functions, classes) dan bagaimana memesannya (signature), tanpa harus melihat ke dapur (implementasi).

**Poin-Poin Penting:**
- Aktifkan `"declaration": true` di tsconfig.json
- `*.d.ts` hanya berisi type definitions, bukan implementasi
- `declarationMap: true` untuk source maps dari .d.ts ke .ts
- `stripInternal: true` untuk menghapus `/** @internal */`
- Gunakan `export` untuk types yang ingin diekspos

**Contoh Kode:**
```typescript
// ========== KONFIGURASI ==========

// BENAR: tsconfig.json untuk generate declaration files
{
  "compilerOptions": {
    "target": "ES2020",
    "module": "NodeNext",
    "strict": true,
    
    // Yang penting untuk declaration files:
    "declaration": true,           // Generate .d.ts files
    "declarationMap": true,        // Generate .d.ts.map untuk source maps
    "declarationDir": "./dist/types",  // Opsional: tempat output .d.ts
    "stripInternal": true,         // Hapus yang ditandai @internal
    "emitDeclarationOnly": false,   // Hanya generate .d.ts, tanpa .js
    
    // Untuk debugging dan source maps:
    "sourceMap": true,
    "inlineSources": true,
    
    "outDir": "./dist",
    "rootDir": "./src"
  },
  "include": ["src/**/*"],
  "exclude": ["node_modules", "dist", "**/*.test.ts"]
}

// Atau jika hanya ingin generate declaration tanpa JS:
// "emitDeclarationOnly": true

// Jalankan dengan: tsc -p tsconfig.json

// ========== CONTOH KODE SUMBER ==========

// File: src/math.ts

/**
 * Adds two numbers together
 * @param a - First number
 * @param b - Second number
 * @returns Sum of a and b
 */
export function add(a: number, b: number): number {
  return a + b;
}

/**
 * Subtracts second number from first
 * @internal - This is for internal use only
 */
export function subtract(a: number, b: number): number {
  return a - b;
}

// Generic function
export function identity<T>(value: T): T {
  return value;
}

// Interface
export interface Calculator {
  add(a: number, b: number): number;
  subtract(a: number, b: number): number;
}

// Class
export class SimpleCalculator implements Calculator {
  add(a: number, b: number): number {
    return a + b;
  }
  
  subtract(a: number, b: number): number {
    return a - b;
  }
}

// Type alias
export type Operation = 'add' | 'subtract' | 'multiply' | 'divide';

// ========== OUTPUT DECLARATION FILE (.d.ts) ==========
// File: dist/math.d.ts (hasil generate otomatis)

/**
 * Adds two numbers together
 * @param a - First number
 * @param b - Second number
 * @returns Sum of a and b
 */
export declare function add(a: number, b: number): number;

// Perhatikan: subtract tidak ada karena @internal dan stripInternal: true

export declare function identity<T>(value: T): T;

export interface Calculator {
  add(a: number, b: number): number;
  subtract(a: number, b: number): number;
}

export declare class SimpleCalculator implements Calculator {
  add(a: number, b: number): number;
  subtract(a: number, b: number): number;
}

export type Operation = 'add' | 'subtract' | 'multiply' | 'divide';

// Untuk module JavaScript yang tidak punya types

// File: types/some-js-library.d.ts
declare module 'some-js-library' {
  export interface Config {
    url: string;
    retries?: number;
  }
  
  export function initialize(config: Config): void;
  
  export function getData(id: string): Promise<unknown>;
  
  export const version: string;
  
  export class Client {
    constructor(config: Config);
    connect(): Promise<void>;
    disconnect(): void;
  }
}

// Module augmentation untuk menambah types ke module yang sudah ada
// File: types/express.d.ts
import 'express';

declare global {
  namespace Express {
    interface Request {
      // Tambahkan properti custom ke Request
      userId: string;
      userRole: 'admin' | 'user';
    }
    
    interface Response {
      success<T>(data: T): void;
      error(message: string, statusCode?: number): void;
    }
  }
}

export {};

// ========== KONFIGURASI DENGAN TSUP ==========
// tsup adalah bundler modern yang otomatis generate .d.ts

// package.json scripts:
// "build": "tsup src/index.ts --format cjs,esm --dts"

// Atau dengan konfigurasi tsup.config.ts
import { defineConfig } from 'tsup';

export default defineConfig({
  entry: ['src/index.ts', 'src/utils.ts'],  // Multiple entry points
  format: ['cjs', 'esm'],                    // Dual package
  dts: true,                                  // Generate .d.ts
  sourcemap: true,                            // Source maps
  clean: true,                                // Bersihkan dist sebelum build
  minify: true,                               // Minify output
  splitting: false,                           // Tidak split chunks
  outDir: 'dist',
  external: ['react', 'react-dom'],          // Peer dependencies
  
  // Untuk declaration files yang lebih baik:
  dts: {
    entry: ['src/index.ts'],  // Entry untuk .d.ts
    // resolve: true          // Resolve types dari dependencies
  }
});

// ========== TIPS PENTING ==========

// 1. Ekspor types yang dibutuhkan pengguna
// ✅ BENAR
export interface User { /* ... */ }
export type Status = 'active' | 'inactive';

// ❌ SALAH - types tidak diekspor
interface InternalUser { /* ... */ }  // Hanya internal
type InternalStatus = 'draft' | 'published';  // Hanya internal

// 2. Gunakan `export type` untuk type-only exports
// (TypeScript 3.8+, lebih jelas dan membantu bundler)
export type { User, Status };
export { add, subtract };  // Value exports

// 3. Dokumentasi dengan JSDoc akan muncul di .d.ts
/**
 * Creates a new user
 * @param data - User data
 * @returns Created user with ID
 */
export function createUser(data: Omit<User, 'id'>): User {
  // ...
}

// 4. Hindari `any` di public API
// ❌ SALAH
export function process(data: any): any { /* ... */ }

// ✅ BENAR
export function process<T>(data: T): Processed<T> { /* ... */ }

// 5. Test declaration files dengan `tsd` atau `attw`
// Package: `tsd` untuk test type definitions
// File: test-d/index.test-d.ts
import { expectType } from 'tsd';
import { add, identity, Client, type User } from '../dist';

// Test return types
expectType<number>(add(1, 2));
expectType<string>(identity('hello'));
expectType<number>(identity(42));

// Test class
const client = new Client({ apiKey: 'test' });
expectType<Promise<unknown>>(client.get('/path'));
```

**Kasus Penggunaan Nyata:**
- Memberikan type safety untuk pengguna library JavaScript
- Auto-completion dan IntelliSense di IDE pengguna
- Dokumentasi otomatis dari JSDoc comments
- Memastikan API library digunakan dengan benar
- Refactoring library tidak merusak pengguna (jika types berubah dengan benar)
