# 207. Dual package CJS dan ESM dengan TypeScript

**Definisi Singkat:** Membuat package TypeScript yang mendukung kedua module system: CommonJS (require) dan ES Modules (import/export), sehingga kompatibel dengan berbagai environment Node.js dan bundler.

**Analogi:** Seperti toko yang menerima pembayaran dengan tunai dan kartu - Anda menyediakan beberapa opsi pembayaran (module system) agar semua pelanggan (pengguna library) bisa bertransaksi dengan nyaman.

**Poin-Poin Penting:**
- `type: "module"` di package.json untuk ESM default
- Gunakan ekstensi `.cjs` untuk CommonJS, `.mjs` untuk ESM
- `exports` field di package.json untuk mendefinisikan entry points
- Bundler seperti `tsup`, `rollup`, atau `esbuild` memudahkan dual package
- Perhatikan perbedaan `__dirname`, `require`, `import.meta.url`

**Contoh Kode:**
```typescript
// ========== KONFIGURASI DASAR ==========

// BENAR: package.json untuk dual package
{
  "name": "my-dual-package",
  "version": "1.0.0",
  
  // Tidak set "type": "module" jika menggunakan ekstensi file untuk membedakan
  // Atau set "type": "module" dan gunakan .cjs untuk CommonJS
  
  // Field exports (PENTING untuk dual package)
  "exports": {
    // Entry point utama
    ".": {
      "import": "./dist/index.mjs",      // Untuk ESM (import)
      "require": "./dist/index.cjs",      // Untuk CJS (require)
      "types": "./dist/index.d.ts"        // Type definitions
    },
    
    // Sub-path exports
    "./utils": {
      "import": "./dist/utils/index.mjs",
      "require": "./dist/utils/index.cjs",
      "types": "./dist/utils/index.d.ts"
    },
    
    // Untuk kompatibilitas tools lama
    "./package.json": "./package.json"
  },
  
  // Legacy fields (untuk Node.js < 12.7.0 dan tools lama)
  "main": "./dist/index.cjs",
  "module": "./dist/index.mjs",
  "types": "./dist/index.d.ts",
  "typings": "./dist/index.d.ts",
  
  "files": ["dist", "src"],
  
  "scripts": {
    "build": "tsup",  // Gunakan tsup untuk build dual format
    "build:rollup": "rollup -c",  // Atau dengan rollup
    "test": "vitest"
  },
  
  "devDependencies": {
    "tsup": "^8.0.0",
    "typescript": "^5.0.0",
    "rollup": "^4.0.0",
    "@rollup/plugin-typescript": "^11.0.0"
  }
}

// ========== DENGAN TSUP (PALING MUDAH) ==========

// File: tsup.config.ts
import { defineConfig } from 'tsup';

export default defineConfig({
  // Entry points
  entry: [
    'src/index.ts',
    'src/utils/index.ts'
  ],
  
  // Format output
  format: ['cjs', 'esm'],  // Generate kedua format
  
  // Extension file (opsional, default sudah benar)
  // cjs: .cjs, esm: .mjs ketika package.json tidak ada "type": "module"
  
  // Generate declaration files
  dts: true,
  
  // Source maps
  sourcemap: true,
  
  // Clean output directory sebelum build
  clean: true,
  
  // Output directory
  outDir: 'dist',
  
  // Splitting (hanya untuk ESM)
  splitting: false,
  
  // External dependencies (tidak dibundle)
  external: ['react', 'vue'],
  
  // Shim untuk __dirname, __filename di ESM
  shims: true,
  
  // Untuk masing-masing format bisa dikonfigurasi
  // cjsInterop: true,  // Tambahkan interop untuk CJS
});

// Hasil build di dist/:
// dist/
// ├── index.cjs        # CommonJS
// ├── index.mjs        # ESM
// ├── index.d.ts       # Types
// ├── utils/
// │   ├── index.cjs
// │   ├── index.mjs
// │   └── index.d.ts
// └── ...

// ========== DENGAN ROLLUP (KONFIGURASI MANUAL) ==========

// File: rollup.config.js
import typescript from '@rollup/plugin-typescript';
import { nodeResolve } from '@rollup/plugin-node-resolve';
import commonjs from '@rollup/plugin-commonjs';
import dts from 'rollup-plugin-dts';

const external = ['react', 'react-dom'];  // Peer deps

export default [
  // ESM Build
  {
    input: 'src/index.ts',
    output: {
      file: 'dist/index.mjs',
      format: 'esm',
      sourcemap: true
    },
    plugins: [
      nodeResolve({ extensions: ['.ts', '.tsx'] }),
      commonjs(),
      typescript({ tsconfig: './tsconfig.build.json' })
    ],
    external
  },
  
  // CommonJS Build
  {
    input: 'src/index.ts',
    output: {
      file: 'dist/index.cjs',
      format: 'cjs',
      sourcemap: true,
      exports: 'named'
    },
    plugins: [
      nodeResolve({ extensions: ['.ts', '.tsx'] }),
      commonjs(),
      typescript({ tsconfig: './tsconfig.build.json' })
    ],
    external
  },
  
  // Declaration Files
]
// ... konfigurasi rollup-plugin-dts

// ========== KODE SUMBER YANG KOMPATIBEL ==========

// File: src/index.ts

// ✅ BENAR: Kode yang bekerja di CJS dan ESM

// Gunakan import/export ESM di source
// Bundler akan mengkonversi ke CJS jika perlu
export { add, subtract } from './math';
export { createClient } from './client';
export type { User, Config } from './types';

// ⚠️ PERHATIKAN: Perbedaan CJS vs ESM

// ❌ Jangan gunakan __dirname/__filename secara langsung
// Ini hanya ada di CJS
const path1 = path.join(__dirname, 'file.txt');  // ❌ Tidak bekerja di ESM

// ✅ Gunakan cara yang kompatibel
import { fileURLToPath } from 'url';
import { dirname, join } from 'path';

// Dapatkan __dirname di ESM
const __filename = fileURLToPath(import.meta.url);
const __dirname = dirname(__filename);
const path2 = join(__dirname, 'file.txt');  // ✅ Bekerja di ESM

// Atau gunakan path relatif dari import.meta.url
const dataPath = new URL('./data.json', import.meta.url);

// ❌ Jangan campur require dan import di source
const lodash = require('lodash');  // ❌ Hanya CJS
import lodash from 'lodash';      // ✅ ESM (bundler akan handle)

// ✅ Gunakan import di source, biarkan bundler mengkonversi
import _ from 'lodash';

// ========== TEST KEDUA FORMAT ==========

// Test CJS
// File: test-cjs.js
const { add, createClient } = require('my-dual-package');
console.log(add(1, 2));  // 3
const client = createClient({ apiKey: 'test' });

// Test ESM
// File: test-esm.mjs
import { add, createClient } from 'my-dual-package';
console.log(add(1, 2));  // 3
const client = createClient({ apiKey: 'test' });

// Test TypeScript
// File: test-ts.ts
import { add, createClient, type Config } from 'my-dual-package';

const result: number = add(1, 2);
const config: Config = { apiKey: 'test' };

// ========== KONFIGURASI CONDITIONAL EXPORTS ==========

// Package.json dengan exports yang lebih kompleks
{
  "exports": {
    ".": {
      // TypeScript 4.7+ mendukung "types" condition
      "types": "./dist/index.d.ts",
      
      // Bundler (Webpack, Vite, Rollup, Parcel)
      "import": "./dist/index.mjs",
      "require": "./dist/index.cjs",
      
      // Node.js specific
      "node": {
        "import": "./dist/index-node.mjs",
        "require": "./dist/index-node.cjs"
      },
      
      // Default fallback
      "default": "./dist/index.mjs"
    },
    
    // Sub-modules dengan pattern
    "./features/*": {
      "import": "./dist/features/*.mjs",
      "require": "./dist/features/*.cjs",
      "types": "./dist/features/*.d.ts"
    }
  },
  
  // Untuk TypeScript < 4.7 yang tidak support "types" condition
  "typesVersions": {
    "*": {
      "*": ["dist/*"]
    }
  }
}

// ========== MASALAH UMUM DAN SOLUSI ==========

// Masalah 1: Default export di CJS vs ESM
// Di CJS: module.exports = ...
// Di ESM: export default ...

// Solusi: Gunakan named exports atau pastikan interop benar
// Di tsup config: cjsInterop: true

// Masalah 2: __dirname di ESM
// Solusi: Gunakan import.meta.url dan fileURLToPath

import { fileURLToPath } from 'url';
import { dirname } from 'path';

const getDirname = (metaUrl: string) => dirname(fileURLToPath(metaUrl));
const __dirname = getDirname(import.meta.url);

// Masalah 3: JSON import di ESM
// Di CJS: const data = require('./data.json')
// Di ESM: Butuh assert atau import assertions

// Solusi 1: Import dengan assert (Node.js 16+)
import data from './data.json' assert { type: 'json' };

// Solusi 2: Baca dengan fs
import { readFileSync } from 'fs';
const data = JSON.parse(readFileSync('./data.json', 'utf-8'));

// Masalah 4: Dual package hazard
// Saat CJS dan ESM mengimport package yang sama tapi mendapatkan instance berbeda

// Solusi: Hindari stateful singleton di module level
// atau pastikan state disimpan di tempat yang sama

// ❌ SALAH - stateful di module
let counter = 0;
export function increment() { return ++counter; }

// ✅ LEBIH BAIK - buat class untuk state
export class Counter {
  private count = 0;
  increment() { return ++this.count; }
}

// ========== TYPE DEFINITIONS UNTUK DUAL PACKAGE ==========

// Type definitions tidak peduli CJS atau ESM
// Yang penting ekspor types dengan benar

// File: src/types.ts
export interface User {
  id: string;
  name: string;
}

export type Role = 'admin' | 'user';

// File: src/index.ts
// Re-export types
export type { User, Role } from './types';
export * from './math';
export * from './client';

// Atau dengan export type (TypeScript 3.8+)
export type { User, Role };
export { add, createClient };

// Test penggunaan TypeScript
// import { add, createClient, type User, type Role } from 'my-package';
```

**Kasus Penggunaan Nyata:**
- Library yang digunakan di project CJS (require) dan ESM (import)
- Kompatibilitas dengan Node.js lama dan baru
- Kompatibilitas dengan berbagai bundler (Webpack, Vite, Rollup, Parcel)
- Mendukung TypeScript dan JavaScript pengguna
- Mudah migrasi dari CJS ke ESM tanpa breaking change
