# 205. Menulis source code yang type-safe

**Definisi Singkat:** Praktik terbaik menulis kode library TypeScript yang type-safe, termasuk penggunaan generic, type narrowing, utility types, dan menghindari `any`.

**Analogi:** Seperti membangun jembatan yang kuat - setiap bagian (tipe) harus pas dengan sempurna, tidak ada celah (any) yang bisa menyebabkan kegagalan, dan desainnya fleksibel untuk berbagai beban (generic).

**Poin-Poin Penting:**
- Hindari `any` sebisa mungkin, gunakan `unknown` jika perlu
- Gunakan generic untuk fleksibilitas tanpa kehilangan type safety
- Type narrowing untuk menangani union types
- Gunakan branded types untuk nilai dengan makna khusus
- Overload function untuk signature yang berbeda
- Defensive programming dengan runtime validation

**Contoh Kode:**
```typescript
// ❌ KESALAHAN UMUM: Penggunaan any
function badFetch(url: any): Promise<any> {
  return fetch(url).then(r => r.json() as any);
}

// Penggunaan - tidak ada type safety sama sekali
const data = badFetch('/api/users');
data.then(d => d.nonexistentMethod());  // Tidak ada error compile-time!

// ✅ BENAR: Gunakan generic dan tipe yang jelas
interface ApiResponse<T> {
  data: T;
  success: boolean;
  timestamp: string;
}

async function fetchApi<T>(url: string): Promise<ApiResponse<T>> {
  const response = await fetch(url);
  const json = await response.json() as ApiResponse<T>;
  return json;
}

// Atau lebih aman: validasi runtime dengan schema
import { z } from 'zod';

const UserSchema = z.object({
  id: z.string(),
  name: z.string(),
  email: z.string().email()
});

type User = z.infer<typeof UserSchema>;

async function fetchUser(id: string): Promise<User> {
  const response = await fetch(`/api/users/${id}`);
  const json = await response.json();
  
  // Validasi runtime!
  const result = UserSchema.safeParse(json);
  if (!result.success) {
    throw new Error(`Invalid user data: ${result.error.message}`);
  }
  
  return result.data;  // Tipe: User (100% type-safe)
}

// BENAR: Generic dengan constraints
interface HasId {
  id: string | number;
}

interface HasTimestamps {
  createdAt: Date;
  updatedAt: Date;
}

// Generic function yang hanya menerima tipe dengan id
function findById<T extends HasId>(items: T[], id: T['id']): T | undefined {
  return items.find(item => item.id === id);
}

// Generic dengan multiple constraints
function isRecent<T extends HasTimestamps>(item: T, days: number = 7): boolean {
  const weekAgo = new Date();
  weekAgo.setDate(weekAgo.getDate() - days);
  return item.createdAt > weekAgo;
}

// Penggunaan
interface User {
  id: string;
  name: string;
  createdAt: Date;
  updatedAt: Date;
}

const users: User[] = [
  { id: '1', name: 'John', createdAt: new Date(), updatedAt: new Date() }
];

const user = findById(users, '1');  // OK, User memiliki id
const recent = isRecent(users[0]);  // OK, User memiliki timestamps

// findById([{ name: 'Test' }], '1');  // Error: tidak ada id

// BENAR: Function overloading untuk signature berbeda
// Mendukung beberapa cara pemanggilan dengan type safety

interface Config {
  url: string;
  timeout?: number;
}

// Overload signatures
function createClient(url: string): Client;
function createClient(config: Config): Client;
function createClient(urlOrConfig: string | Config, options?: object): Client;

// Implementation signature (hanya satu implementasi)
function createClient(
  urlOrConfig: string | Config,
  options?: object
): Client {
  if (typeof urlOrConfig === 'string') {
    return new Client({ url: urlOrConfig, ...options });
  }
  return new Client({ ...urlOrConfig, ...options });
}

// Penggunaan - TypeScript tahu overload mana yang cocok
const client1 = createClient('https://api.example.com');  // OK
const client2 = createClient({ url: 'https://api.example.com', timeout: 5000 });  // OK
// createClient(123);  // Error: tidak cocok dengan overload mana pun

// BENAR: Branded Types untuk nilai dengan makna khusus
// Masalah: string userId dan string productId tidak bisa dibedakan
type UserId = string & { __brand: 'UserId' };
type ProductId = string & { __brand: 'ProductId' };

// Helper untuk membuat branded type
function createUserId(id: string): UserId {
  return id as UserId;
}

function createProductId(id: string): ProductId {
  return id as ProductId;
}

// Fungsi yang hanya menerima UserId
function fetchUser(id: UserId): Promise<User> {
  return db.users.findById(id);
}

const userId = createUserId('user_123');
const productId = createProductId('prod_456');

fetchUser(userId);  // OK
// fetchUser(productId);  // Error! ProductId tidak sama dengan UserId
// fetchUser('string biasa');  // Error!

// BENAR: Type narrowing untuk menangani union types
type Result<T, E = Error> = 
  | { success: true; data: T }
  | { success: false; error: E };

function handleResult<T>(result: Result<T>): T {
  if (result.success) {
    // Type narrowing: result.data bertipe T
    return result.data;
  }
  // Type narrowing: result.error bertipe Error
  throw result.error;
}

// Type guard untuk narrowing
function isString(value: unknown): value is string {
  return typeof value === 'string';
}

function isNumber(value: unknown): value is number {
  return typeof value === 'number';
}

function processValue(value: unknown): void {
  if (isString(value)) {
    console.log(value.toUpperCase());  // value bertipe string
  } else if (isNumber(value)) {
    console.log(value.toFixed(2));  // value bertipe number
  }
}

// BENAR: Utility types untuk transformasi tipe
interface Original {
  id: string;
  name: string;
  createdAt: Date;
  updatedAt: Date;
}

// Partial - semua properti menjadi optional
type PartialOriginal = Partial<Original>;
// { id?: string; name?: string; ... }

// Required - semua properti menjadi required
type RequiredOriginal = Required<{ id?: string; name?: string }>;
// { id: string; name: string }

// Pick - pilih properti tertentu
type PickOriginal = Pick<Original, 'id' | 'name'>;
// { id: string; name: string }

// Omit - hilangkan properti tertentu
type OmitOriginal = Omit<Original, 'createdAt' | 'updatedAt'>;
// { id: string; name: string }

// ReturnType - dapatkan return type dari function
type FetchUserReturn = ReturnType<typeof fetchUser>;
// Promise<User>

// Parameters - dapatkan parameter type dari function
type FetchUserParams = Parameters<typeof fetchUser>;
// [UserId]

// Awaited - dapatkan tipe di dalam Promise
type UnwrappedUser = Awaited<Promise<User>>;
// User

// BENAR: Const assertions untuk literal types
// Masalah: inferensi menjadi string/number biasa, bukan literal
const badConfig = {
  endpoint: '/api/users',
  method: 'GET'
};
// badConfig.method bertipe string, bukan 'GET'

const goodConfig = {
  endpoint: '/api/users',
  method: 'GET'
} as const;
// goodConfig.method bertipe 'GET' (literal type)

// Bisa juga digunakan pada array
const roles = ['admin', 'user', 'moderator'] as const;
type Role = typeof roles[number];  // 'admin' | 'user' | 'moderator'

// BENAR: Menghindari any dengan unknown
// unknown lebih aman karena harus di-narrowing sebelum digunakan
function processUnknown(value: unknown): void {
  // value.toUpperCase();  // Error: unknown tidak bisa digunakan langsung
  
  if (typeof value === 'string') {
    value.toUpperCase();  // OK: sudah di-narrowing
  }
  
  if (typeof value === 'number') {
    value.toFixed(2);  // OK
  }
}

// BENAR: Runtime validation untuk data dari luar
// Semua data dari API, user input, localStorage harus divalidasi
import { z } from 'zod';

const ConfigSchema = z.object({
  apiKey: z.string().min(10),
  baseUrl: z.string().url(),
  timeout: z.number().int().positive().optional().default(30000),
  debug: z.boolean().optional().default(false)
});

type Config = z.infer<typeof ConfigSchema>;

function loadConfig(): Config {
  // Data dari environment variable (tidak type-safe!)
  const rawConfig = {
    apiKey: process.env.API_KEY,
    baseUrl: process.env.BASE_URL,
    timeout: process.env.TIMEOUT ? parseInt(process.env.TIMEOUT) : undefined,
    debug: process.env.DEBUG === 'true'
  };
  
  // Validasi dan transformasi menjadi type-safe
  const result = ConfigSchema.safeParse(rawConfig);
  
  if (!result.success) {
    throw new Error(`Invalid config: ${result.error.message}`);
  }
  
  return result.data;  // 100% type-safe dan tervalidasi
}
```

**Kasus Penggunaan Nyata:**
- Membuat library API yang type-safe untuk pengguna
- Menghindari bug runtime karena ketidakcocokan tipe
- Generic membuat library fleksibel untuk berbagai tipe data
- Branded types mencegah kesalahan pencampuran ID
- Function overload memberikan API yang intuitif namun type-safe
