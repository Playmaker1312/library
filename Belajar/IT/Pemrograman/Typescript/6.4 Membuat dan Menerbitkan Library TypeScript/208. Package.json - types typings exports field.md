# 208. Package.json - types typings exports field

**Definisi Singkat:** Memahami dan mengkonfigurasi field penting di package.json untuk TypeScript library: `types`, `typings`, dan terutama `exports` untuk entry points modern.

**Analogi:** Seperti peta dan tanda di gedung perkantoran - `exports` adalah peta lengkap yang menunjukkan semua ruangan (entry points), `types` adalah tanda yang menunjukkan di mana arsip (type definitions) berada.

**Poin-Poin Penting:**
- `types` / `typings` - Menunjuk ke file declaration utama (.d.ts)
- `exports` - Field modern untuk mendefinisikan semua entry points
- `exports` mendukung conditional exports (import, require, types, node, default)
- `typesVersions` - Untuk TypeScript < 4.7 yang tidak support `exports` types
- Urutan conditions di `exports` penting (lebih spesifik di atas)

**Contoh Kode:**
```typescript
// ========== FIELDS LAMA (LEGACY) ==========

// types dan typings (keduanya sama, types lebih modern)
{
  "name": "my-library",
  "version": "1.0.0",
  
  // Menunjuk ke main file
  "main": "./dist/index.js",
  
  // Menunjuk ke type definitions (sama dengan typings)
  "types": "./dist/index.d.ts",
  "typings": "./dist/index.d.ts",  // Legacy, sama dengan types
  
  // Untuk ESM
  "module": "./dist/index.esm.js",
  "jsnext:main": "./dist/index.esm.js"  // Legacy
}

// Kekurangan fields legacy:
// - Tidak mendukung sub-paths dengan baik
// - Tidak mendukung conditional exports
// - Tidak bisa membedakan import vs require

// ========== EXPORTS FIELD (MODERN) ==========

// ✅ BENAR: Package.json dengan exports field
{
  "name": "@org/my-library",
  "version": "1.0.0",
  "type": "module",
  
  // ========== EXPORTS FIELD ==========
  "exports": {
    // Entry point utama: "."
    ".": {
      // TypeScript types (untuk TypeScript 4.7+)
      "types": "./dist/index.d.ts",
      
      // Untuk ESM import
      "import": "./dist/index.mjs",
      
      // Untuk CJS require
      "require": "./dist/index.cjs",
      
      // Untuk Node.js spesifik
      "node": {
        "import": "./dist/index-node.mjs",
        "require": "./dist/index-node.cjs"
      },
      
      // Untuk browser (jika berbeda dengan node)
      "browser": "./dist/index-browser.mjs",
      
      // Development condition (untuk debug mode)
      "development": "./dist/index-dev.mjs",
      
      // Default fallback (harus paling bawah)
      "default": "./dist/index.mjs"
    },
    
    // Sub-path: "./utils"
    "./utils": {
      "types": "./dist/utils.d.ts",
      "import": "./dist/utils.mjs",
      "require": "./dist/utils.cjs"
    },
    
    // Sub-path dengan pattern
    "./components/*": {
      "types": "./dist/components/*.d.ts",
      "import": "./dist/components/*.mjs",
      "require": "./dist/components/*.cjs"
    },
    
    // Internal hanya untuk package sendiri
    "./internal/*": {
      "internal": "./src/internal/*.ts"
    },
    
    // Package.json (untuk tools yang membutuhkan)
    "./package.json": "./package.json"
  },
  
  // ========== TYPES FIELD (BACKUP) ==========
  // Untuk tools yang belum mendukung exports dengan types
  "types": "./dist/index.d.ts",
  
  // ========== MAIN/MODULE (BACKUP) ==========
  // Untuk Node.js < 12.7.0 yang tidak mendukung exports
  "main": "./dist/index.cjs",
  "module": "./dist/index.mjs",
  
  // ========== TYPES VERSIONS ==========
  // Untuk TypeScript < 4.7 yang tidak support "types" di exports
  "typesVersions": {
    ">=4.7": {
      "*": ["dist/*"]
    },
    ">=3.1": {
      "*": ["dist/types-legacy/*"]
    },
    "*": {
      "*": ["dist/*"]
    }
  },
  
  // ========== FILES YANG DIPUBLISH ==========
  "files": [
    "dist",
    "src"  // Opsional: untuk debugging
  ],
  
  "scripts": {
    "build": "tsup src/index.ts src/utils.ts --format cjs,esm --dts"
  }
}

// ========== PENGGUNAAN DI KONSUMER ==========

// Setelah exports dikonfigurasi dengan benar:

// ✅ Bisa import entry point utama
import { mainFunction } from '@org/my-library';
const { mainFunction } = require('@org/my-library');

// ✅ Bisa import sub-path
import { helper } from '@org/my-library/utils';
const { helper } = require('@org/my-library/utils');

// ✅ Bisa import dengan pattern
import { Button } from '@org/my-library/components/Button';

// ❌ Tidak bisa import yang tidak diexport
import { secret } from '@org/my-library/internal/secret';
// Error: Module not found (karena hanya "internal" condition)

// ========== CONTOH KONFIGURASI BERBAGAI SCENARIO ==========

// SCENARIO 1: Library Sederhana (Single Entry Point)
{
  "name": "simple-lib",
  "version": "1.0.0",
  "main": "./dist/index.cjs",
  "module": "./dist/index.mjs",
  "types": "./dist/index.d.ts",
  "exports": {
    ".": {
      "types": "./dist/index.d.ts",
      "import": "./dist/index.mjs",
      "require": "./dist/index.cjs"
    }
  }
}

// SCENARIO 2: Library dengan Sub-modules
{
  "name": "feature-lib",
  "version": "1.0.0",
  "exports": {
    ".": {
      "types": "./dist/index.d.ts",
      "import": "./dist/index.mjs",
      "require": "./dist/index.cjs"
    },
    "./react": {
      "types": "./dist/react.d.ts",
      "import": "./dist/react.mjs",
      "require": "./dist/react.cjs",
      "peerDependencies": { "react": ">=17.0.0" }
    },
    "./vue": {
      "types": "./dist/vue.d.ts",
      "import": "./dist/vue.mjs",
      "require": "./dist/vue.cjs"
    }
  }
}

// Penggunaan:
// import { core } from 'feature-lib';
// import { useFeature } from 'feature-lib/react';
// import { featureComponent } from 'feature-lib/vue';

// SCENARIO 3: Library dengan Browser dan Node.js berbeda
{
  "name": "universal-lib",
  "version": "1.0.0",
  "exports": {
    ".": {
      "types": "./dist/index.d.ts",
      "node": {
        "import": "./dist/node.mjs",
        "require": "./dist/node.cjs"
      },
      "browser": {
        "import": "./dist/browser.mjs",
        "require": "./dist/browser.cjs"
      },
      "default": "./dist/browser.mjs"
    }
  }
}

// ========== URAIAN KONDISI (CONDITIONS) ==========

// Urutan kondisi dari PALING SPESIFIK ke PALING UMUM
{
  "exports": {
    ".": {
      // Paling spesifik di atas
      "types": "./dist/index.d.ts",      // TypeScript saja
      "node": {                           // Node.js saja
        "import": "./dist/node-esm.mjs",  // Node.js + ESM import
        "require": "./dist/node-cjs.cjs"  // Node.js + CJS require
      },
      "browser": "./dist/browser.mjs",    // Browser saja
      "default": "./dist/fallback.mjs"    // Fallback terakhir
    }
  }
}

// Kondisi yang tersedia:
// - "types"        → TypeScript (4.7+)
// - "node"         → Node.js runtime
// - "browser"      → Browser environment
// - "import"       → ESM import
// - "require"      → CJS require
// - "development"  → Development mode (Node.js 20.12+)
// - "production"   → Production mode
// - "default"      → Fallback

// ========== TYPE VERSIONS (UNTUK TYPESCRIPT LAMA) ==========

// Sebelum TypeScript 4.7, "types" di exports tidak didukung
// Gunakan "typesVersions" sebagai fallback
{
  "typesVersions": {
    // Untuk TypeScript 4.7+ (sudah support exports)
    ">=4.7": {
      "*": ["dist/*"]
    },
    
    // Untuk TypeScript 3.1 - 4.6
    ">=3.1 <4.7": {
      "*": ["dist/legacy/*"]
    },
    
    // Fallback untuk versi lain
    "*": {
      "*": ["dist/*"]
    }
  }
}

// ========== PRAKTIK TERBAIK ==========

// ✅ 1. Selalu sertakan "types" di exports (jika ada types)
"exports": {
  ".": {
    "types": "./dist/index.d.ts",  // ✅ Penting untuk TypeScript
    "import": "./dist/index.mjs",
    "require": "./dist/index.cjs"
  }
}

// ✅ 2. Sertakan juga field "types" legacy (untuk tools lama)
{
  "types": "./dist/index.d.ts",
  "exports": {
    /* ... */
  }
}

// ✅ 3. Urutkan conditions dari spesifik ke umum
"exports": {
  ".": {
    "types": "./types.d.ts",    // Paling spesifik
    "node": { /* ... */ },
    "import": "./esm.mjs",
    "require": "./cjs.cjs",
    "default": "./fallback.mjs" // Paling umum di bawah
  }
}

// ✅ 4. Jangan lupa sertakan "./package.json"
"exports": {
  "./package.json": "./package.json"
}

// ✅ 5. Gunakan "files" untuk membatasi apa yang dipublish
{
  "files": [
    "dist",
    "!dist/**/*.test.js",  // Kecualikan test files
    "src"  // Opsional untuk source maps
  ]
}

// ❌ 6. Jangan gunakan "main" saja tanpa "exports" (modern)
// - Kurang fleksibel
// - Tidak mendukung conditional exports

// ========== TESTING EXPORTS ==========

// Untuk test apakah exports bekerja dengan benar:

// 1. Test ESM import
// File: test-esm.mjs
import { fn } from 'my-library';
import { helper } from 'my-library/utils';

// 2. Test CJS require
// File: test-cjs.js
const { fn } = require('my-library');
const { helper } = require('my-library/utils');

// 3. Test TypeScript
// File: test-ts.ts
import { fn, type SomeType } from 'my-library';
import { helper } from 'my-library/utils';

const value: SomeType = fn();

// 4. Test dengan publint
// npm install -D publint
// npx publint
// → Memeriksa apakah package.json dikonfigurasi dengan benar
```

**Kasus Penggunaan Nyata:**
- Memberikan entry points yang jelas untuk pengguna library
- Mendukung berbagai module system (ESM, CJS)
- Mendukung berbagai environment (Node.js, browser)
- Type safety untuk semua import paths
- Mengontrol apa yang bisa di-import oleh pengguna (encapsulation)
- Kemudahan refactoring tanpa breaking change pengguna
