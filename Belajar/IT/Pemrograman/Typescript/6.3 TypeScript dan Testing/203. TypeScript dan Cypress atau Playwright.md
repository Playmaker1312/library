# 203. TypeScript dan Cypress atau Playwright

**Definisi Singkat:** Menggunakan TypeScript dengan end-to-end testing frameworks seperti Cypress dan Playwright untuk menulis test E2E yang type-safe dan maintainable.

**Analogi:** Seperti robot penguji produk otomatis - Cypress/Playwright adalah robotnya, dan TypeScript adalah instruksi yang jelas sehingga robot tahu persis apa yang harus diperiksa dan bagaimana melakukannya.

**Poin-Poin Penting:**
- Cypress dan Playwright memiliki dukungan TypeScript native
- Custom commands bisa di-extend dengan tipe
- Page Object Model (POM) dengan TypeScript untuk maintainability
- Fixtures dan selectors yang type-safe
- Auto-completion untuk semua API methods

**Contoh Kode:**

// File: pages/LoginPage.ts
import { Page, Locator, expect } from '@playwright/test';

interface UserCredentials {
  email: string;
  password: string;
}

export class LoginPage {
  readonly page: Page;
  readonly emailInput: Locator;
  readonly passwordInput: Locator;
  readonly submitButton: Locator;
  readonly errorMessage: Locator;

  constructor(page: Page) {
    this.page = page;
    this.emailInput = page.getByLabel('Email');
    this.passwordInput = page.getByLabel('Password');
    this.submitButton = page.getByRole('button', { name: 'Sign In' });
    this.errorMessage = page.getByTestId('error-message');
  }

  async goto(): Promise<void> {
    await this.page.goto('/login');
  }

  async fillForm(credentials: UserCredentials): Promise<void> {
    await this.emailInput.fill(credentials.email);
    await this.passwordInput.fill(credentials.password);
  }

  async submit(): Promise<void> {
    await this.submitButton.click();
  }

  async login(credentials: UserCredentials): Promise<void> {
    await this.fillForm(credentials);
    await this.submit();
  }

  async expectErrorMessage(message: string): Promise<void> {
    await expect(this.errorMessage).toBeVisible();
    await expect(this.errorMessage).toContainText(message);
  }

  async expectPageLoaded(): Promise<void> {
    await expect(this.emailInput).toBeVisible();
    await expect(this.passwordInput).toBeVisible();
  }
}

// File: pages/DashboardPage.ts
import { Page, Locator, expect } from '@playwright/test';

export class DashboardPage {
  readonly page: Page;
  readonly welcomeMessage: Locator;
  readonly navigationMenu: Locator;
  readonly logoutButton: Locator;

  constructor(page: Page) {
    this.page = page;
    this.welcomeMessage = page.getByTestId('welcome-message');
    this.navigationMenu = page.getByTestId('navigation');
    this.logoutButton = page.getByRole('button', { name: 'Logout' });
  }

  async expectPageLoaded(): Promise<void> {
    await expect(this.welcomeMessage).toBeVisible();
    await expect(this.navigationMenu).toBeVisible();
    await this.page.waitForURL('**/dashboard');
  }

  async expectWelcomeMessage(username: string): Promise<void> {
    await expect(this.welcomeMessage).toContainText(`Welcome, ${username}`);
  }

  async logout(): Promise<void> {
    await this.logoutButton.click();
  }
}

// BENAR: Test dengan Playwright + TypeScript
// File: tests/auth.spec.ts
import { test, expect } from '@playwright/test';
import { LoginPage } from '../pages/LoginPage';
import { DashboardPage } from '../pages/DashboardPage';
import { testUsers } from '../fixtures/test-data';

test.describe('Authentication', () => {
  let loginPage: LoginPage;
  let dashboardPage: DashboardPage;

  test.beforeEach(async ({ page }) => {
    loginPage = new LoginPage(page);
    dashboardPage = new DashboardPage(page);
  });

  test('should login successfully with valid credentials', async () => {
    await loginPage.goto();
    await loginPage.expectPageLoaded();
    
    // Type-safe: testUsers.admin memiliki tipe TestUser
    await loginPage.login(testUsers.admin);
    
    await dashboardPage.expectPageLoaded();
    await dashboardPage.expectWelcomeMessage(testUsers.admin.name);
  });

  test('should show error with invalid credentials', async () => {
    await loginPage.goto();
    
    await loginPage.login({
      email: 'invalid@example.com',
      password: 'wrongpassword'
    });
    
    await loginPage.expectErrorMessage('Invalid email or password');
  });

  test('should logout successfully', async ({ page }) => {
    // Login first
    await loginPage.goto();
    await loginPage.login(testUsers.admin);
    await dashboardPage.expectPageLoaded();
    
    // Logout
    await dashboardPage.logout();
    
    // Should redirect to login
    await loginPage.expectPageLoaded();
    await expect(page).toHaveURL('/login');
  });
});

// BENAR: Cypress Custom Commands dengan tipe
// File: cypress/support/commands.ts
declare global {
  namespace Cypress {
    interface Chainable {
      login(email: string, password: string): Chainable<void>;
      loginByRole(role: 'admin' | 'user'): Chainable<void>;
      getByTestId(id: string): Chainable<JQuery<HTMLElement>>;
    }
  }
}

Cypress.Commands.add('login', (email: string, password: string) => {
  cy.visit('/login');
  cy.get('input[name="email"]').type(email);
  cy.get('input[name="password"]').type(password);
  cy.get('button[type="submit"]').click();
  cy.url().should('include', '/dashboard');
});

Cypress.Commands.add('loginByRole', (role: 'admin' | 'user') => {
  const credentials = {
    admin: { email: 'admin@test.com', password: 'admin123' },
    user: { email: 'user@test.com', password: 'user123' }
  };
  cy.login(credentials[role].email, credentials[role].password);
});

Cypress.Commands.add('getByTestId', (id: string) => {
  return cy.get(`[data-testid="${id}"]`);
});

export {};

// BENAR: Cypress Test dengan TypeScript
// File: cypress/e2e/login.cy.ts
interface User {
  email: string;
  password: string;
}

const testUsers: Record<'admin' | 'user', User> = {
  admin: { email: 'admin@test.com', password: 'admin123' },
  user: { email: 'user@test.com', password: 'user123' }
};

describe('Login Flow', () => {
  beforeEach(() => {
    cy.visit('/login');
  });

  it('should login as admin', () => {
    // Type-safe custom command
    cy.login(testUsers.admin.email, testUsers.admin.password);
    cy.url().should('include', '/dashboard');
    cy.getByTestId('welcome-message').should('contain', 'Admin');
  });

  it('should login as user using custom command', () => {
    cy.loginByRole('user');  // Type-safe: hanya 'admin' | 'user'
    // cy.loginByRole('invalid');  // Error!
    cy.url().should('include', '/dashboard');
  });

  it('should show error for invalid credentials', () => {
    cy.get('input[name="email"]').type('invalid@test.com');
    cy.get('input[name="password"]').type('wrong');
    cy.get('button[type="submit"]').click();
    
    cy.getByTestId('error')
      .should('be.visible')
      .and('contain', 'Invalid credentials');
  });
});

// BENAR: Page Object Model di Cypress
// File: cypress/pages/LoginPage.ts
class LoginPage {
  elements = {
    emailInput: () => cy.get('input[name="email"]'),
    passwordInput: () => cy.get('input[name="password"]'),
    submitButton: () => cy.get('button[type="submit"]'),
    errorMessage: () => cy.getByTestId('error')
  };

  visit(): Cypress.Chainable<Cypress.AUTWindow> {
    return cy.visit('/login');
  }

  fillEmail(email: string): Cypress.Chainable<JQuery<HTMLElement>> {
    return this.elements.emailInput().type(email);
  }

  fillPassword(password: string): Cypress.Chainable<JQuery<HTMLElement>> {
    return this.elements.passwordInput().type(password);
  }

  submit(): Cypress.Chainable<JQuery<HTMLElement>> {
    return this.elements.submitButton().click();
  }

  login(email: string, password: string): void {
    this.fillEmail(email);
    this.fillPassword(password);
    this.submit();
  }
}

export default LoginPage;
