# 201. Type-safe Test Fixtures dan Factories

**Definisi Singkat:** Membuat test data (fixtures) dan factories yang type-safe untuk menghasilkan data test yang konsisten dan sesuai dengan tipe aplikasi.

**Analogi:** Seperti pabrik mainan - factory adalah mesin yang menghasilkan mainan (test data) dengan spesifikasi yang konsisten, dan fixtures adalah mainan siap pakai untuk testing.

**Poin-Poin Penting:**
- Gunakan `faker.js` untuk data dummy realistis
- Factory pattern untuk membuat objek dengan default values
- Builder pattern untuk konfigurasi bertahap
- `Partial<T>` untuk override field tertentu
- `as const` untuk fixtures yang tidak berubah

**Contoh Kode:**
```typescript
// Install: npm install -D @faker-js/faker
import { faker } from '@faker-js/faker';
import { describe, it, expect } from 'vitest';

// Tipe aplikasi
interface User {
  id: string;
  name: string;
  email: string;
  password: string;
  role: 'admin' | 'user' | 'moderator';
  isActive: boolean;
  createdAt: Date;
  profile?: UserProfile;
}

interface UserProfile {
  avatar: string;
  bio: string;
  location: string;
  birthDate?: Date;
}

interface Post {
  id: string;
  title: string;
  content: string;
  authorId: string;
  tags: string[];
  isPublished: boolean;
  publishedAt?: Date;
  createdAt: Date;
  updatedAt: Date;
}

// BENAR: Fixtures sederhana (data statis)
export const userFixtures: User[] = [
  {
    id: 'user-1',
    name: 'Alice Johnson',
    email: 'alice@example.com',
    password: 'hashed-password-1',
    role: 'admin',
    isActive: true,
    createdAt: new Date('2024-01-01'),
    profile: {
      avatar: 'https://example.com/avatar1.png',
      bio: 'I am an admin user',
      location: 'New York'
    }
  },
  {
    id: 'user-2',
    name: 'Bob Smith',
    email: 'bob@example.com',
    password: 'hashed-password-2',
    role: 'user',
    isActive: true,
    createdAt: new Date('2024-01-15')
    // Tidak ada profile
  }
] as const;  // as const agar tipe menjadi literal (readonly)

// Fixture tunggal
export const adminUser: User = userFixtures[0];
export const regularUser: User = userFixtures[1];

// BENAR: Factory Functions (dinamis, dengan faker)
function createUser(overrides: Partial<User> = {}): User {
  return {
    id: faker.string.uuid(),
    name: faker.person.fullName(),
    email: faker.internet.email(),
    password: faker.internet.password(),
    role: faker.helpers.arrayElement(['admin', 'user', 'moderator']),
    isActive: faker.datatype.boolean(),
    createdAt: faker.date.past(),
    profile: faker.datatype.boolean() ? {
      avatar: faker.image.avatar(),
      bio: faker.lorem.paragraph(),
      location: faker.location.city(),
      birthDate: faker.date.past({ years: 50 })
    } : undefined,
    ...overrides  // Override dengan nilai custom
  };
}

function createPost(overrides: Partial<Post> = {}): Post {
  const now = new Date();
  const isPublished = faker.datatype.boolean();
  
  return {
    id: faker.string.uuid(),
    title: faker.lorem.sentence({ min: 5, max: 15 }),
    content: faker.lorem.paragraphs({ min: 3, max: 10 }),
    authorId: faker.string.uuid(),
    tags: faker.lorem.words({ min: 1, max: 5 }),
    isPublished,
    publishedAt: isPublished ? faker.date.past({ years: 1 }) : undefined,
    createdAt: faker.date.past({ years: 1 }),
    updatedAt: now,
    ...overrides
  };
}

// BENAR: Factory dengan helper methods
const userFactory = {
  // User dengan role spesifik
  admin(overrides: Partial<User> = {}): User {
    return createUser({ ...overrides, role: 'admin' });
  },
  
  user(overrides: Partial<User> = {}): User {
    return createUser({ ...overrides, role: 'user' });
  },
  
  moderator(overrides: Partial<User> = {}): User {
    return createUser({ ...overrides, role: 'moderator' });
  },
  
  // User dengan profile
  withProfile(overrides: Partial<User> = {}): User {
    return createUser({
      ...overrides,
      profile: {
        avatar: faker.image.avatar(),
        bio: faker.lorem.paragraph(),
        location: faker.location.city(),
        ...overrides.profile
      }
    });
  },
  
  // User yang inactive
  inactive(overrides: Partial<User> = {}): User {
    return createUser({ ...overrides, isActive: false });
  },
  
  // Multiple users
  many(count: number, overrides: Partial<User> = {}): User[] {
    return Array.from({ length: count }, () => createUser(overrides));
  }
};

const postFactory = {
  published(overrides: Partial<Post> = {}): Post {
    const now = new Date();
    return createPost({
      ...overrides,
      isPublished: true,
      publishedAt: overrides.publishedAt || now
    });
  },
  
  draft(overrides: Partial<Post> = {}): Post {
    return createPost({
      ...overrides,
      isPublished: false,
      publishedAt: undefined
    });
  },
  
  byAuthor(authorId: string, overrides: Partial<Post> = {}): Post {
    return createPost({ ...overrides, authorId });
  },
  
  withTags(tags: string[], overrides: Partial<Post> = {}): Post {
    return createPost({ ...overrides, tags });
  }
};

// BENAR: Builder Pattern untuk kontrol lebih detail
class UserBuilder {
  private user: User;

  constructor() {
    this.user = createUser();
  }

  withId(id: string): this {
    this.user.id = id;
    return this;
  }

  withName(name: string): this {
    this.user.name = name;
    return this;
  }

  withEmail(email: string): this {
    this.user.email = email;
    return this;
  }

  asAdmin(): this {
    this.user.role = 'admin';
    return this;
  }

  asUser(): this {
    this.user.role = 'user';
    return this;
  }

  inactive(): this {
    this.user.isActive = false;
    return this;
  }

  withProfile(profile: Partial<UserProfile> = {}): this {
    this.user.profile = {
      avatar: faker.image.avatar(),
      bio: faker.lorem.paragraph(),
      location: faker.location.city(),
      ...profile
    };
    return this;
  }

  build(): User {
    return { ...this.user };
  }
}

// Penggunaan di test
describe('User factories', () => {
  describe('fixtures', () => {
    it('should have admin user fixture', () => {
      expect(adminUser.role).toBe('admin');
      expect(adminUser.isActive).toBe(true);
      expect(adminUser.profile).toBeDefined();
    });

    it('should have regular user fixture', () => {
      expect(regularUser.role).toBe('user');
      expect(regularUser.profile).toBeUndefined();
    });
  });

  describe('factory functions', () => {
    it('should create user with defaults', () => {
      const user = createUser();
      
      // Semua field harus ada
      expect(user.id).toBeDefined();
      expect(user.name).toBeDefined();
      expect(user.email).toContain('@');
      expect(['admin', 'user', 'moderator']).toContain(user.role);
    });

    it('should create user with overrides', () => {
      const customUser = createUser({
        name: 'Custom Name',
        role: 'admin',
        isActive: false
      });
      
      expect(customUser.name).toBe('Custom Name');
      expect(customUser.role).toBe('admin');
      expect(customUser.isActive).toBe(false);
      // Field lain masih menggunakan default dari faker
      expect(customUser.id).toBeDefined();
    });
  });

  describe('factory helpers', () => {
    it('should create admin user', () => {
      const admin = userFactory.admin();
      expect(admin.role).toBe('admin');
    });

    it('should create multiple users', () => {
      const users = userFactory.many(5, { role: 'user' });
      
      expect(users).toHaveLength(5);
      users.forEach(user => {
        expect(user.role).toBe('user');
      });
    });

    it('should create published post', () => {
      const published = postFactory.published();
      
      expect(published.isPublished).toBe(true);
      expect(published.publishedAt).toBeDefined();
    });
  });

  describe('builder pattern', () => {
    it('should build user with custom properties', () => {
      const user = new UserBuilder()
        .withId('custom-id')
        .withName('John Builder')
        .withEmail('builder@example.com')
        .asAdmin()
        .withProfile({ bio: 'Built with builder pattern' })
        .build();
      
      expect(user.id).toBe('custom-id');
      expect(user.name).toBe('John Builder');
      expect(user.email).toBe('builder@example.com');
      expect(user.role).toBe('admin');
      expect(user.profile?.bio).toBe('Built with builder pattern');
    });
  });
});

// ERROR: Fixtures tanpa tipe
const badFixtures: any[] = [
  { name: 'Test' }  // Tidak ada field penting seperti id, email
];

// ✅ Lebih baik: gunakan tipe eksplisit
const goodFixtures: User[] = [
  createUser({ name: 'Test' })  // Factory memastikan semua field ada
];
```

**Kasus Penggunaan Nyata:**
- Membuat data test yang realistis untuk unit/integration test
- Memastikan data test selalu memiliki semua field yang diperlukan
- Mudah membuat variasi data (admin, user, active, inactive, dll)
- Seed data untuk database development/testing
- Test yang konsisten karena data tidak hardcoded secara manual
