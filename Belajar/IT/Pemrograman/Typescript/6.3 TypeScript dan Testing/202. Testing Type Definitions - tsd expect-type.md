# 202. Testing Type Definitions - tsd dan expect-type

**Definisi Singkat:** Cara menguji type definitions TypeScript untuk memastikan tipe yang di-expect sesuai dengan yang sebenarnya, menggunakan tools seperti `tsd` dan `expect-type`.

**Analogi:** Seperti quality control untuk blueprint - Anda memastikan bahwa spesifikasi (type definitions) yang tertulis benar-benar sesuai dengan apa yang diinginkan.

**Poin-Poin Penting:**
- `tsd` - Test runner untuk TypeScript types
- `expect-type` - Assertion library untuk type tests
- Type tests mengevaluasi di compile-time, bukan runtime
- `expectType<T>(value)` - Memastikan value bertipe T
- `assertType<T>(value)` - Mirip tapi dengan assertion
- `expectNotType<T>(value)` - Memastikan value BUKAN bertipe T

**Contoh Kode:**
```typescript
// Install: npm install -D tsd expect-type typescript

// Struktur project:
// src/
//   index.ts
//   types.ts
// test-d/
//   index.test-d.ts  <-- File test untuk types

// ========== FILE: src/types.ts ==========
// Type definitions yang akan diuji

export interface User {
  id: string;
  name: string;
  email: string;
  role: 'admin' | 'user' | 'moderator';
  profile?: UserProfile;
}

export interface UserProfile {
  avatar: string;
  bio: string;
  location?: string;
}

export type UserRole = User['role'];

// Utility types
export type PartialExcept<T, K extends keyof T> = Partial<Omit<T, K>> & Pick<T, K>;
export type RequiredOnly<T, K extends keyof T> = Required<Pick<T, K>> & Partial<Omit<T, K>>;

// Fungsi dengan tipe
export function createUser(data: Omit<User, 'id' | 'createdAt'>): User {
  return {
    id: crypto.randomUUID(),
    name: data.name,
    email: data.email,
    role: data.role || 'user',
    profile: data.profile
  } as User;
}

export function updateUser(user: User, updates: Partial<User>): User {
  return { ...user, ...updates };
}

export function hasProfile(user: User): user is User & { profile: UserProfile } {
  return user.profile !== undefined;
}

// Generic function
export function pick<T, K extends keyof T>(obj: T, keys: K[]): Pick<T, K> {
  const result = {} as Pick<T, K>;
  keys.forEach(key => {
    if (key in obj) {
      result[key] = obj[key];
    }
  });
  return result;
}

// ========== FILE: test-d/index.test-d.ts ==========
// Test untuk type definitions
// Jalankan dengan: npx tsd

import { expectType, expectError, expectAssignable, expectNotAssignable } from 'tsd';
import { expectTypeOf } from 'expect-type';
import { 
  User, 
  UserProfile, 
  UserRole,
  PartialExcept,
  createUser, 
  updateUser, 
  hasProfile,
  pick
} from '../src/index';

// BENAR: Menguji tipe interface
describe('User interface', () => {
  it('should have correct property types', () => {
    const user: User = {
      id: '1',
      name: 'John',
      email: 'john@example.com',
      role: 'admin',
      profile: { avatar: 'url', bio: 'test' }
    };

    // expectType<T>(value) - memastikan value bertipe T
    expectType<string>(user.id);
    expectType<string>(user.name);
    expectType<string>(user.email);
    expectType<UserRole>(user.role);
    expectType<UserProfile | undefined>(user.profile);
  });

  it('should have UserRole as union type', () => {
    // UserRole harus 'admin' | 'user' | 'moderator'
    expectType<'admin' | 'user' | 'moderator'>('admin' as UserRole);
    
    // String biasa tidak boleh di-assign ke UserRole
    expectError((): UserRole => 'invalid-role');
  });
});

// BENAR: Menguji utility types
describe('PartialExcept utility', () => {
  it('should make all properties partial except specified', () => {
    // PartialExcept<User, 'id' | 'email'>
    // id dan email harus required, lainnya optional
    
    type TestType = PartialExcept<User, 'id' | 'email'>;
    
    // Valid: id dan email required, name tidak perlu
    const valid: TestType = {
      id: '1',
      email: 'test@example.com'
    };
    
    expectType<TestType>(valid);
    
    // Invalid: id harus ada
    expectError((): TestType => ({
      email: 'test@example.com'  // id hilang
    }));
    
    // Invalid: email harus ada
    expectError((): TestType => ({
      id: '1'  // email hilang
    }));
  });
});

// BENAR: Menguji return type fungsi
describe('createUser function', () => {
  it('should return User type', () => {
    const result = createUser({
      name: 'John',
      email: 'john@example.com',
      role: 'user'
    });
    
    expectType<User>(result);
    expectType<string>(result.id);  // id ada di return
  });

  it('should require certain parameters', () => {
    // name dan email required
    expectError(createUser({
      name: 'John'  // email hilang
    }));
    
    // role optional (default 'user')
    createUser({
      name: 'John',
      email: 'john@example.com'  // role tidak perlu
    });
  });
});

// BENAR: Menguji dengan expect-type (alternatif library)
describe('expect-type library', () => {
  it('should test types with expectTypeOf', () => {
    const user: User = {
      id: '1',
      name: 'John',
      email: 'john@example.com',
      role: 'admin'
    };

    // Test tipe user
    expectTypeOf(user).toEqualTypeOf<User>();
    expectTypeOf(user.id).toBeString();
    expectTypeOf(user.role).toEqualTypeOf<'admin' | 'user' | 'moderator'>();
    expectTypeOf(user.profile).toEqualTypeOf<UserProfile | undefined>();
    
    // Test tipe function
    expectTypeOf(createUser).toBeFunction();
    expectTypeOf(createUser).parameter(0).toMatchTypeOf<{ name: string; email: string }>();
    expectTypeOf(createUser).returns.toEqualTypeOf<User>();
    
    // Test generic function
    const picked = pick(user, ['id', 'name']);
    expectTypeOf(picked).toEqualTypeOf<Pick<User, 'id' | 'name'>>();
    expectTypeOf(picked.id).toBeString();
    expectTypeOf(picked.name).toBeString();
    // expectTypeOf(picked.email).toBeString();  // Error: email tidak di-pick
  });
});

// BENAR: Menguji type narrowing
describe('type narrowing', () => {
  it('should narrow type with hasProfile type guard', () => {
    const user: User = {
      id: '1',
      name: 'John',
      email: 'john@example.com',
      role: 'user'
    };

    if (hasProfile(user)) {
      // Di dalam blok ini, user.profile tidak boleh undefined
      expectTypeOf(user.profile).toEqualTypeOf<UserProfile>();
      expectTypeOf(user.profile.avatar).toBeString();
      expectTypeOf(user.profile.bio).toBeString();
    } else {
      // Di luar, masih bisa undefined
      expectTypeOf(user.profile).toEqualTypeOf<UserProfile | undefined>();
    }
  });
});

// BENAR: Menguji assignability
describe('assignability tests', () => {
  it('should test what can be assigned', () => {
    // User bisa di-assign ke object dengan properti subset
    type BasicUser = { id: string; name: string };
    
    const user: User = {
      id: '1',
      name: 'John',
      email: 'john@example.com',
      role: 'user'
    };
    
    // User bisa di-assign ke BasicUser (karena memiliki id dan name)
    expectAssignable<BasicUser>(user);
    
    // Tapi BasicUser TIDAK bisa di-assign ke User (kurang properti)
    const basic: BasicUser = { id: '2', name: 'Jane' };
    expectNotAssignable<User>(basic);
  });
});

// BENAR: Menguji generic types
describe('generic pick function', () => {
  const user: User = {
    id: '1',
    name: 'John',
    email: 'john@example.com',
    role: 'admin'
  };

  it('should return correct Pick type', () => {
    const picked = pick(user, ['id', 'name']);
    
    // picked harus bertipe Pick<User, 'id' | 'name'>
    expectType<{ id: string; name: string }>(picked);
    
    // Akses properti yang di-pick
    expectType<string>(picked.id);
    expectType<string>(picked.name);
    
    // Error mengakses properti yang tidak di-pick
    expectError(picked.email);
  });

  it('should only allow keys of T', () => {
    // Hanya key dari User yang diizinkan
    pick(user, ['id', 'name']);  // OK
    
    // Error: 'invalid' bukan key dari User
    expectError(pick(user, ['id', 'invalid']));
  });
});

// ERROR: Test yang seharusnya gagal

    // expectType<User>({ id: '1' });  // ❌ Kurang properti

});
```

**Kasus Penggunaan Nyata:**
- Menguji utility types buatan sendiri
- Memastikan fungsi mengembalikan tipe yang benar
- Memastikan generic types bekerja seperti yang diharapkan
- Membuat type definitions untuk library (pastikan API types benar)
- Regression test untuk type definitions saat refactoring
