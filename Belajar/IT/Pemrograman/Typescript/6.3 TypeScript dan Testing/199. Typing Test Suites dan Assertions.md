# 199. Typing Test Suites dan Assertions

**Definisi Singkat:** Cara menulis test suites dan assertions yang type-safe di TypeScript, memastikan nilai yang diuji memiliki tipe yang benar dan assertion menggunakan matcher yang sesuai.

**Analogi:** Seperti pemeriksa kualitas yang memiliki daftar spesifikasi jelas - ia tahu persis apa yang harus diperiksa (tipe data) dan bagaimana memeriksanya (assertion/matcher).

**Poin-Poin Penting:**
- Gunakan tipe eksplisit untuk test data jika perlu
- Type narrowing untuk menguji nilai dengan tipe union
- `expect(value)` mengembalikan matcher dengan tipe yang sesuai
- Custom assertions bisa di-extend dengan tipe
- Gunakan `as const` untuk literal types dalam test data

**Contoh Kode:**
```typescript
import { describe, it, expect } from 'vitest';  // atau jest

// Tipe untuk test
interface User {
  id: string;
  name: string;
  email: string;
  role: 'admin' | 'user';
  profile?: {
    avatar?: string;
    bio?: string;
  };
}

type ApiResponse<T> = 
  | { success: true; data: T }
  | { success: false; error: string };

// BENAR: Test data dengan tipe eksplisit
const testUsers: User[] = [
  {
    id: '1',
    name: 'Alice Admin',
    email: 'alice@example.com',
    role: 'admin',
    profile: {
      avatar: 'alice.png',
      bio: 'I am an admin'
    }
  },
  {
    id: '2',
    name: 'Bob User',
    email: 'bob@example.com',
    role: 'user'
    // profile tidak ada
  }
];

// Fungsi yang akan diuji
function findUserById(users: User[], id: string): User | undefined {
  return users.find(u => u.id === id);
}

function isAdmin(user: User): boolean {
  return user.role === 'admin';
}

function processApiResponse(response: ApiResponse<User>): User {
  if (response.success) {
    return response.data;  // Type narrowing: data bertipe User
  }
  throw new Error(response.error);
}

// BENAR: Test suites dengan type safety
describe('user functions', () => {
  describe('findUserById', () => {
    it('should find user by id', () => {
      const user = findUserById(testUsers, '1');
      
      // Type narrowing setelah toBeDefined
      expect(user).toBeDefined();
      
      if (user) {
        // Di dalam blok if, user bertipe User (bukan undefined)
        expect(user.name).toBe('Alice Admin');
        expect(user.email).toBe('alice@example.com');
      }
    });

    it('should return undefined for non-existent id', () => {
      const user = findUserById(testUsers, '999');
      expect(user).toBeUndefined();
    });
  });

  describe('isAdmin', () => {
    it('should return true for admin users', () => {
      const admin = testUsers[0];  // Alice is admin
      expect(isAdmin(admin)).toBe(true);
    });

    it('should return false for regular users', () => {
      const user = testUsers[1];  // Bob is regular user
      expect(isAdmin(user)).toBe(false);
    });
  });

  describe('processApiResponse', () => {
    it('should return data on success response', () => {
      const successResponse: ApiResponse<User> = {
        success: true,
        data: testUsers[0]
      };
      
      const result = processApiResponse(successResponse);
      expect(result).toEqual(testUsers[0]);
    });

    it('should throw error on failed response', () => {
      const errorResponse: ApiResponse<User> = {
        success: false,
        error: 'User not found'
      };
      
      expect(() => processApiResponse(errorResponse)).toThrow('User not found');
    });
  });
});

// BENAR: Menguji optional properties dengan narrowing
describe('user profile', () => {
  it('should have profile for admin user', () => {
    const admin = testUsers[0];
    
    // Cek bahwa profile ada
    expect(admin.profile).toBeDefined();
    
    // Type narrowing
    if (admin.profile) {
      expect(admin.profile.avatar).toBe('alice.png');
      expect(admin.profile.bio).toContain('admin');
    }
  });

  it('should not have profile for regular user', () => {
    const user = testUsers[1];
    expect(user.profile).toBeUndefined();
  });
});

// BENAR: Custom type guard untuk testing
function isSuccessResponse<T>(response: ApiResponse<T>): response is { success: true; data: T } {
  return response.success === true;
}

describe('type guard test', () => {
  it('should narrow type with custom guard', () => {
    const response: ApiResponse<User> = {
      success: true,
      data: testUsers[0]
    };

    expect(isSuccessResponse(response)).toBe(true);

    if (isSuccessResponse(response)) {
      // response.data bertipe User
      expect(response.data.name).toBe('Alice Admin');
    }
  });
});

// BENAR: Menguji dengan literal types
type Status = 'pending' | 'active' | 'inactive';

function getStatusMessage(status: Status): string {
  switch (status) {
    case 'pending':
      return 'Your account is pending approval';
    case 'active':
      return 'Your account is active';
    case 'inactive':
      return 'Your account is inactive';
  }
}

describe('getStatusMessage', () => {
  it('should return correct messages', () => {
    // as const memastikan tipe literal, bukan string
    const testCases = [
      { status: 'pending' as const, expected: 'pending approval' },
      { status: 'active' as const, expected: 'active' },
      { status: 'inactive' as const, expected: 'inactive' }
    ];

    testCases.forEach(({ status, expected }) => {
      expect(getStatusMessage(status)).toContain(expected);
    });
  });
});

// ERROR: Menguji tanpa type safety
describe('bad practices', () => {
  it('should not use any', () => {
    const user: any = testUsers[0];
    
    // ❌ Tidak ada error meskipun properti tidak ada
    expect(user.nonexistent).toBeUndefined();  // Runtime error!
    
    // ❌ Bisa memanggil fungsi dengan argumen salah
    isAdmin(user as any);  // Tidak ada pengecekan tipe
  });
});
```

**Kasus Penggunaan Nyata:**
- Memastikan test data selalu sesuai dengan tipe yang diharapkan
- Type narrowing membantu menguji union types dengan aman
- Refactoring tipe akan memunculkan error di test jika tidak sesuai
- Custom type guards membuat test lebih ekspresif dan type-safe
- Literal types dengan `as const` memastikan test case tidak bisa menggunakan nilai sembarang
