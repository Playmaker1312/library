# 195. Typing WebSocket connections

**Definisi Singkat:** Cara men-tipe WebSocket connections dan pesan yang dikirimkan, memastikan komunikasi real-time antara client dan server memiliki type safety.

**Analogi:** Seperti walkie-talkie dengan saluran khusus - setiap tipe pesan memiliki salurannya sendiri, dan kedua pihak tahu persis format pesan yang akan diterima.

**Poin-Poin Penting:**
- Definisikan tipe untuk semua message yang mungkin
- Tagged union untuk membedakan tipe pesan
- `ws` library untuk WebSocket di Node.js
- `socket.io` memiliki dukungan TypeScript yang baik
- Type guards untuk memproses pesan masuk

**Contoh Kode:**
```typescript
// Install untuk ws: npm install ws
// npm install -D @types/ws

// Atau untuk Socket.IO: npm install socket.io

// BENAR: Definisikan tipe pesan dengan tagged union
interface ChatMessage {
  type: 'chat';
  payload: {
    roomId: string;
    senderId: string;
    content: string;
    timestamp: number;
  };
}

interface UserJoinedMessage {
  type: 'user_joined';
  payload: {
    userId: string;
    username: string;
    roomId: string;
  };
}

interface UserLeftMessage {
  type: 'user_left';
  payload: {
    userId: string;
    roomId: string;
  };
}

interface TypingMessage {
  type: 'typing';
  payload: {
    userId: string;
    roomId: string;
    isTyping: boolean;
  };
}

// Union type untuk semua pesan yang mungkin
type WebSocketMessage = 
  | ChatMessage 
  | UserJoinedMessage 
  | UserLeftMessage 
  | TypingMessage;

// Tipe untuk pesan dari client ke server
type ClientToServerMessage = WebSocketMessage;

// Tipe untuk pesan dari server ke client
type ServerToClientMessage = WebSocketMessage | {
  type: 'error';
  payload: { message: string };
};

// BENAR: Server WebSocket dengan ws library
import { WebSocketServer, WebSocket } from 'ws';
import http from 'http';

const server = http.createServer();
const wss = new WebSocketServer({ server });

// Type guard untuk memvalidasi pesan
function isWebSocketMessage(data: unknown): data is WebSocketMessage {
  if (typeof data !== 'object' || data === null) return false;
  
  const msg = data as Record<string, unknown>;
  if (typeof msg.type !== 'string') return false;
  if (typeof msg.payload !== 'object' || msg.payload === null) return false;
  
  // Validasi berdasarkan type
  switch (msg.type) {
    case 'chat': {
      const payload = msg.payload as Record<string, unknown>;
      return typeof payload.roomId === 'string' &&
             typeof payload.senderId === 'string' &&
             typeof payload.content === 'string';
    }
    case 'user_joined':
    case 'user_left':
    case 'typing':
      return true;  // Simplified validation
    default:
      return false;
  }
}

// Fungsi helper untuk mengirim pesan dengan tipe aman
function sendMessage(ws: WebSocket, message: ServerToClientMessage) {
  ws.send(JSON.stringify(message));
}

// Fungsi untuk broadcast ke semua client
function broadcast(message: ServerToClientMessage, exclude?: WebSocket) {
  wss.clients.forEach((client) => {
    if (client !== exclude && client.readyState === WebSocket.OPEN) {
      sendMessage(client, message);
    }
  });
}

wss.on('connection', (ws) => {
  console.log('Client connected');

  // Kirim pesan selamat datang
  sendMessage(ws, {
    type: 'user_joined',
    payload: {
      userId: 'system',
      username: 'System',
      roomId: 'general'
    }
  });

  ws.on('message', (data) => {
    try {
      const parsed = JSON.parse(data.toString());
      
      // Type narrowing dengan type guard
      if (!isWebSocketMessage(parsed)) {
        sendMessage(ws, {
          type: 'error',
          payload: { message: 'Invalid message format' }
        });
        return;
      }

      // Type narrowing berdasarkan discriminant (type)
      switch (parsed.type) {
        case 'chat':
          // parsed.payload bertipe ChatMessage['payload']
          console.log(`[${parsed.payload.roomId}] ${parsed.payload.senderId}: ${parsed.payload.content}`);
          
          // Broadcast ke semua client
          broadcast({
            type: 'chat',
            payload: parsed.payload
          }, ws);
          break;

        case 'user_joined':
          console.log(`User ${parsed.payload.username} joined ${parsed.payload.roomId}`);
          broadcast(parsed, ws);
          break;

        case 'typing':
          broadcast(parsed, ws);
          break;
      }
    } catch (error) {
      sendMessage(ws, {
        type: 'error',
        payload: { message: 'Failed to parse message' }
      });
    }
  });

  ws.on('close', () => {
    console.log('Client disconnected');
  });
});

// BENAR: Dengan Socket.IO (lebih simple untuk TypeScript)
import { Server, Socket } from 'socket.io';
import { createServer } from 'http';

const httpServer = createServer();

interface ServerToClientEvents {
  chat: (message: ChatMessage['payload']) => void;
  userJoined: (data: UserJoinedMessage['payload']) => void;
  userLeft: (data: UserLeftMessage['payload']) => void;
  typing: (data: TypingMessage['payload']) => void;
}

interface ClientToServerEvents {
  chat: (message: ChatMessage['payload'], callback: (ack: { received: boolean }) => void) => void;
  joinRoom: (roomId: string) => void;
  typing: (data: { roomId: string; isTyping: boolean }) => void;
}

interface InterServerEvents {
  ping: () => void;
}

interface SocketData {
  userId: string;
  username: string;
}

const io = new Server<
  ClientToServerEvents,
  ServerToClientEvents,
  InterServerEvents,
  SocketData
>(httpServer, {
  cors: { origin: "*" }
});

io.on('connection', (socket) => {
  // socket.data bertipe SocketData
  socket.data.userId = 'user_' + socket.id;
  socket.data.username = 'Guest';

  // Client events - autocomplete tersedia!
  socket.on('joinRoom', (roomId) => {
    socket.join(roomId);
    
    // Emit ke room
    io.to(roomId).emit('userJoined', {
      userId: socket.data.userId,
      username: socket.data.username,
      roomId
    });
  });

  socket.on('chat', (message, callback) => {
    io.to(message.roomId).emit('chat', message);
    callback({ received: true });
  });

  socket.on('typing', (data) => {
    socket.to(data.roomId).emit('typing', {
      ...data,
      userId: socket.data.userId
    });
  });
});
```
