# Proyek aplikasi PWA untuk catatan offline dengan Service Worker

## Analogi: Perpustakaan Digital

Aplikasi catatan offline seperti buku catatan pribadi yang selalu bisa dibawa dan ditulis kapan pun, bahkan saat perpustakaan tutup. Pengguna bisa menulis catatan baru, membaca catatan lama, dan mengeditnya tanpa perlu koneksi internet. Begitu koneksi tersedia, catatan otomatis tersinkronisasi dengan server perpustakaan pusat. Service Worker seperti asisten pribadi yang memastikan buku catatan selalu tersedia meskipun jaringan bermasalah. Aplikasi ini bisa diinstal di layar utama perangkat seperti aplikasi native.

## Penjelasan Detail

Proyek ini membangun aplikasi PWA catatan offline lengkap dengan kemampuan installable, kerja offline, dan sinkronisasi. Fitur utama: tambah/edit/hapus catatan (menggunakan IndexedDB untuk penyimpanan lokal), Service Worker untuk caching dan offline access, Web App Manifest untuk instalasi, dan Background Sync untuk sinkronisasi saat online. Aplikasi ini menggunakan arsitektur App Shell: shell HTML dan CSS di-cache saat install, konten catatan diambil dari IndexedDB. Pengguna dapat menambahkan catatan saat offline dan catatan akan tersinkronisasi secara otomatis ketika koneksi kembali. Proyek ini mendemonstrasikan integrasi berbagai teknologi PWA dalam satu aplikasi fungsional.

```html
  <script>
    const DB_NAME = 'CatatanKuDB';
    const DB_VERSION = 1;
    let db;

    function bukaDB() {
      return new Promise((resolve, reject) => {
        const req = indexedDB.open(DB_NAME, DB_VERSION);
        req.onupgradeneeded = (event) => {
          const database = event.target.result;
          if (!database.objectStoreNames.contains('catatan')) {
            database.createObjectStore('catatan', { keyPath: 'id', autoIncrement: true });
          }
        };
        req.onsuccess = () => { db = req.result; resolve(db); };
        req.onerror = () => reject(req.error);
      });
    }

    async function tambahCatatan(catatan) {
      const tx = db.transaction('catatan', 'readwrite');
      const store = tx.objectStore('catatan');
      return new Promise((resolve, reject) => {
        const req = store.add(catatan);
        req.onsuccess = () => resolve(req.result);
        req.onerror = () => reject(req.error);
      });
    }

    async function getAllCatatan() {
      const tx = db.transaction('catatan', 'readonly');
      const store = tx.objectStore('catatan');
      return new Promise((resolve) => {
        const result = [];
        store.openCursor().onsuccess = (event) => {
          const cursor = event.target.result;
          if (cursor) {
            result.push(cursor.value);
            cursor.continue();
          } else {
            resolve(result);
          }
        };
      });
    }

    async function hapusCatatan(id) {
      const tx = db.transaction('catatan', 'readwrite');
      const store = tx.objectStore('catatan');
      return new Promise((resolve) => {
        store.delete(id).onsuccess = () => resolve();
      });
    }

    async function renderCatatan() {
      const container = document.getElementById('daftarCatatan');
      const catatanList = await getAllCatatan();

      if (catatanList.length === 0) {
        container.innerHTML = '<p class="empty">Belum ada catatan. Buat catatan pertama Anda!</p>';
        return;
      }

      container.innerHTML = catatanList.reverse().map(c => `
        <div class="catatan-item">
          <h3>${escapeHtml(c.judul)}</h3>
          <p>${escapeHtml(c.isi.substring(0, 100))}${c.isi.length > 100 ? '...' : ''}</p>
          <div class="meta">
            ${new Date(c.tanggal).toLocaleString('id-ID')}
            <span class="status-badge ${c.synced ? 'synced' : 'pending'}">
              ${c.synced ? 'Tersimpan' : 'Menunggu sinkron'}
            </span>
          </div>
          <div class="aksi">
            <button class="btn btn-danger" onclick="hapusCatatanHandler(${c.id})">Hapus</button>
          </div>
        </div>
      `).join('');
    }

    function escapeHtml(str) {
      const div = document.createElement('div');
      div.textContent = str;
      return div.innerHTML;
    }

    window.hapusCatatanHandler = async (id) => {
      await hapusCatatan(id);
      renderCatatan();
    };

    document.getElementById('simpanBtn').addEventListener('click', async () => {
      const judul = document.getElementById('judulInput').value.trim();
      const isi = document.getElementById('isiInput').value.trim();
      if (!judul || !isi) return;

      await tambahCatatan({
        judul,
        isi,
        tanggal: new Date().toISOString(),
        synced: navigator.onLine
      });

      document.getElementById('judulInput').value = '';
      document.getElementById('isiInput').value = '';
      renderCatatan();
    });

    document.getElementById('statusKoneksi').textContent =
      navigator.onLine ? 'Online' : 'Offline';
    window.addEventListener('online', () => {
      document.getElementById('statusKoneksi').textContent = 'Online';
    });
    window.addEventListener('offline', () => {
      document.getElementById('statusKoneksi').textContent = 'Offline';
    });

    bukaDB().then(() => renderCatatan());

    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.register('sw-catatan.js');
    }
  </script>
</body>
</html>
```

```javascript
// sw-catatan.js
const CACHE = 'catatanku-v1';
const URLS = ['/', '/index.html', '/manifest.json'];

self.addEventListener('install', event => {
  event.waitUntil(
    caches.open(CACHE).then(cache => cache.addAll(URLS))
  );
});

self.addEventListener('fetch', event => {
  event.respondWith(
    caches.match(event.request).then(res => res || fetch(event.request))
  );
});
```

## Poin Penting
- Aplikasi catatan berfungsi penuh secara offline dengan penyimpanan IndexedDB.
- Service Worker menyediakan akses offline ke shell aplikasi dan konten yang sudah di-cache.
- Web App Manifest memungkinkan instalasi aplikasi di layar utama perangkat seperti aplikasi native.
