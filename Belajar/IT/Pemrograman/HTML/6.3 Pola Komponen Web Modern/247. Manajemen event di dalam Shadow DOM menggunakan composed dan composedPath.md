# Manajemen Event di dalam Shadow DOM Menggunakan composed dan composedPath

## Analogi: Perpustakaan Digital

Bayangkan sebuah kotak kaca di perpustakaan yang berisi buku langka. Jika seseorang mengetuk kotak kaca dari dalam (misal, alarm keamanan berbunyi), suara tersebut harus bisa didengar oleh petugas di luar kotak. Namun, beberapa suara internal (seperti detak mekanisme pengunci) seharusnya tidak terdengar ke luar. Dalam Shadow DOM, event yang terjadi di dalam shadow tree perlu "menembus" ke luar agar bisa ditangani oleh elemen host atau komponen induk. Properti `composed` pada event menentukan apakah event dapat menembus batas Shadow DOM. `composedPath()` mengembalikan array node yang dilalui event saat merambat, termasuk node di shadow tree.

## Penjelasan Detail

Event yang terjadi di dalam Shadow DOM memiliki perilaku perambatan yang berbeda. Secara default, event custom bersifat `composed: false` — tidak menembus batas Shadow DOM. Event browser umum seperti `click`, `keydown`, `focus` bersifat `composed: true` — bisa didengar dari luar. Properti `composed` pada event menentukan apakah event dapat melewati shadow boundary. Metode `composedPath()` mengembalikan array node yang dilalui event dalam urutan perambatan, termasuk node di shadow tree dan light DOM. Event retargeting terjadi ketika event melewati batas shadow: `target` event akan diubah menjadi shadow host untuk menjaga enkapsulasi. `composedPath()` mengungkapkan target asli.

```html
    var log = document.getElementById('log');
    var hasOutsideListener = false;

    function addLog(pesan) {
      var entry = document.createElement('div');
      entry.className = 'log-entry';
      var waktu = new Date().toLocaleTimeString();
      entry.textContent = '[' + waktu + '] ' + pesan;
      log.insertBefore(entry, log.firstChild);
      if (log.children.length > 15) {
        log.removeChild(log.lastChild);
      }
    }

    // Template untuk Shadow Box
    var template = document.createElement('template');
    template.innerHTML =
      '<style>' +
        '.box { border: 2px solid #1565c0; border-radius: 8px; padding: 20px; background: #e3f2fd; }' +
        '.box h3 { margin: 0 0 10px 0; color: #1565c0; }' +
        '.internal-btn { background: #1565c0; color: white; border: none; padding: 10px 20px; border-radius: 4px; cursor: pointer; margin: 5px; }' +
        '.internal-btn:hover { background: #1976d2; }' +
        '.internal-div { background: #bbdefb; padding: 15px; border-radius: 4px; margin: 10px 0; }' +
        '.slot-content { margin-top: 10px; }' +
      '</style>' +
      '<div class="box">' +

          '<p>🔘 Ini DIV di dalam Shadow DOM.</p>' +

        '</div>' +
        '<div class="slot-content">' +
          '<slot name="konten"></slot>' +
        '</div>' +
      '</div>';

    class ShadowBox extends HTMLElement {
      constructor() {
        super();
        this._shadowRoot = this.attachShadow({ mode: 'open' });
        this._shadowRoot.appendChild(template.content.cloneNode(true));
      }

      connectedCallback() {
        var shadow = this._shadowRoot;
        var btn = shadow.getElementById('btn-dalam');
        var internalDiv = shadow.getElementById('internal-div');

        btn.addEventListener('click', function (event) {
          addLog('🖱️ Tombol DI KLIK (di dalam Shadow DOM)');
          addLog('   event.target: ' + (event.target.tagName || event.target));
          addLog('   event.composed: ' + event.composed);
          addLog('   composedPath length: ' + event.composedPath().length);
          addLog('   composedPath[0]: ' + (event.composedPath()[0].tagName || event.composedPath()[0]));
          addLog('   composedPath[last]: ' + (event.composedPath()[event.composedPath().length - 1].tagName || 'document'));
        });

        // Event listener untuk event custom di dalam shadow
        shadow.addEventListener('custom-internal', function (event) {
          addLog('📨 Event "custom-internal" diterima di Shadow DOM (composed: ' + event.composed + ')');
        });
      }

      // Mengekspos method untuk mengirim event
      kirimEventCustom(composed) {
        var shadow = this._shadowRoot;
        var event = new CustomEvent('custom-internal', {
          bubbles: true,
          composed: composed,
          detail: { pesan: 'Halo dari Shadow DOM!' }
        });
        shadow.getElementById('internal-div').dispatchEvent(event);
        addLog('📤 Event custom-internal dikirim (composed: ' + composed + ')');
      }
    }

    customElements.define('shadow-box', ShadowBox);

    // Listener dari luar shadow root
    function tambahListenerLuar() {
      var box = document.getElementById('demo-box');

      if (hasOutsideListener) {
        addLog('⚠️ Listener luar sudah ditambahkan');
        return;
      }

      box.addEventListener('click', function (event) {
        addLog('🔗 [LISTENER LUAR] click event diterima!');
        addLog('   event.target (retargeted): ' + (event.target.tagName || event.target));
        addLog('   event.composed: ' + event.composed);
        addLog('   composedPath includes shadow root: ' + (event.composedPath().some(function (n) { return n instanceof ShadowRoot; })));
      });

      box.addEventListener('custom-internal', function (event) {
        addLog('🔗 [LISTENER LUAR] Event "custom-internal" diterima! (composed: ' + event.composed + ')');
        addLog('   detail: ' + JSON.stringify(event.detail));
      });

      hasOutsideListener = true;
      addLog('✅ Listener luar ditambahkan ke shadow-box');
    }

    function kirimEventCustom() {
      var box = document.getElementById('demo-box');
      if (box.kirimEventCustom) {
        box.kirimEventCustom(false); // composed: false
      }
    }

    function kirimEventCustomComposed() {
      var box = document.getElementById('demo-box');
      if (box.kirimEventCustom) {
        box.kirimEventCustom(true); // composed: true
      }
    }

    function bersihkanLog() {
      log.innerHTML = '<div class="log-entry">🗑️ Log dibersihkan pada ' + new Date().toLocaleTimeString() + '</div>';
    }
  </script>
</body>
</html>
```

## Poin Penting
- Event browser default (click, keydown, focus) memiliki `composed: true` — otomatis menembus shadow boundary.
- Event custom harus secara eksplisit menyetel `composed: true` jika ingin didengar dari luar Shadow DOM.
- Gunakan `composedPath()` untuk debugging — ini menunjukkan path lengkap event termasuk node di dalam shadow tree.
