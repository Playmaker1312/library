# Pengujian Web Components Menggunakan Tools seperti Web Component Tester

## Analogi: Perpustakaan Digital

Sebelum rak buku baru dipasang di perpustakaan, rak tersebut harus diuji: apakah bisa menahan beban, apakah labelnya terbaca, apakah pintunya terbuka dengan benar. Pengujian Web Components seperti quality control untuk setiap komponen sebelum digunakan di aplikasi nyata. Web Component Tester, Open WC, dan Playwright adalah alat-alat yang memungkinkan kita menguji komponen secara otomatis: memeriksa apakah elemen terbentuk dengan benar, apakah atribut berfungsi, apakah event dipicu, dan apakah Shadow DOM berisi struktur yang diharapkan.

## Penjelasan Detail

Pengujian Web Components dapat dilakukan di berbagai level. Unit test menguji logika komponen secara terisolasi menggunakan framework seperti Mocha atau Jasmine dengan Web Component Tester (WCT) — alat dari Polymer yang menyediakan browser environment. Integration test menguji komponen dalam konteks halaman menggunakan Playwright atau Cypress — alat ini dapat menembus Shadow DOM untuk memeriksa elemen internal. Open WC (open-wc.org) menyediakan rekomendasi toolchain modern menggunakan @web/test-runner dengan Mocha/Chai. Pengujian mencakup: render awal, perubahan atribut, event handling, aksesibilitas, dan interaksi pengguna. Penting untuk menguji di berbagai browser karena implementasi Web Components bisa berbeda.

```html
  <div class="card">

import { html, fixture, expect } from '@open-wc/testing';
import '../src/kalkulator-buku.js';

describe('kalkulator-buku', () => {
  it('merender dengan benar', async () => {
    const el = await fixture(html`&lt;kalkulator-buku harga-satuan="5000" jumlah="3"&gt;&lt;/kalkulator-buku&gt;`);
    const shadow = el.shadowRoot;
    expect(shadow.querySelector('.total')).to.exist;
    expect(shadow.querySelector('.total').textContent).to.include('Rp 15.000');
  });

  it('merespon perubahan atribut jumlah', async () => {
    const el = await fixture(html`&lt;kalkulator-buku harga-satuan="5000" jumlah="2"&gt;&lt;/kalkulator-buku&gt;`);
    el.setAttribute('jumlah', '5');
    await el.updateComplete;
    expect(el.shadowRoot.querySelector('.total').textContent).to.include('Rp 25.000');
  });

  it('mengirim event saat tombol diklik', async () => {
    const el = await fixture(html`&lt;kalkulator-buku&gt;&lt;/kalkulator-buku&gt;`);
    const handler = sinon.spy();
    el.addEventListener('hitung', handler);
    el.shadowRoot.querySelector('button').click();
    expect(handler.calledOnce).to.be.true;
  });
});

  <!-- Definisi komponen yang diuji -->
  <script>
    class KalkulatorBuku extends HTMLElement {
      constructor() {
        super();
        this._shadow = this.attachShadow({ mode: 'open' });
        this._shadow.innerHTML =
          '<style>' +
            ':host { display: block; padding: 15px; border: 2px solid #4caf50; border-radius: 8px; background: #f1f8e9; }' +
            '.row { display: flex; justify-content: space-between; padding: 5px 0; }' +
            '.total { font-size: 1.3em; font-weight: bold; color: #2e7d32; margin-top: 10px; padding-top: 10px; border-top: 2px solid #c8e6c9; }' +
            'button { background: #4caf50; color: white; border: none; padding: 8px 16px; border-radius: 4px; cursor: pointer; margin-top: 10px; }' +
            'button:hover { background: #66bb6a; }' +
          '</style>' +
          '<div class="row">Harga satuan: <span id="harga">Rp 5.000</span></div>' +
          '<div class="row">Jumlah: <span id="jumlah">3</span> buku</div>' +
          '<div class="total">Total: <span id="total">Rp 15.000</span></div>' +
          '<button id="btn-hitung">🔢 Hitung Ulang</button>';

        var btn = this._shadow.getElementById('btn-hitung');
        btn.addEventListener('click', function () {
          this.hitung();
          this.dispatchEvent(new CustomEvent('hitung', {
            detail: { total: this._total },
            bubbles: true,
            composed: true
          }));
        }.bind(this));
      }

      get updateComplete() {
        return Promise.resolve();
      }

      static get observedAttributes() {
        return ['harga-satuan', 'jumlah'];
      }

      attributeChangedCallback() {
        this.hitung();
      }

      connectedCallback() {
        this.hitung();
      }

      hitung() {
        var harga = parseInt(this.getAttribute('harga-satuan')) || 0;
        var jumlah = parseInt(this.getAttribute('jumlah')) || 0;
        this._total = harga * jumlah;

        var hargaEl = this._shadow.getElementById('harga');
        var jumlahEl = this._shadow.getElementById('jumlah');
        var totalEl = this._shadow.getElementById('total');

        if (hargaEl) hargaEl.textContent = 'Rp ' + harga.toLocaleString('id-ID');
        if (jumlahEl) jumlahEl.textContent = jumlah;
        if (totalEl) totalEl.textContent = 'Rp ' + this._total.toLocaleString('id-ID');
      }
    }

    customElements.define('kalkulator-buku', KalkulatorBuku);
  </script>

  <script>
    // Simulasi test runner
    var testResults = [];

    function test(name, fn) {
      try {
        fn();
        testResults.push({ name: name, status: 'pass', message: '' });
      } catch (e) {
        testResults.push({ name: name, status: 'fail', message: e.message });
      }
    }

    function assert(condition, message) {
      if (!condition) throw new Error(message || 'Assertion failed');
    }

    function assertEquals(actual, expected, message) {
      if (actual !== expected) {
        throw new Error(message || 'Expected ' + expected + ' but got ' + actual);
      }
    }

    function jalankanTest() {
      testResults = [];

      // Test 1: Render
      test('Render: komponen terbentuk dengan benar', function () {
        var el = document.getElementById('kalkulator-demo');
        assert(el.shadowRoot !== null, 'Shadow root harus ada');
        assert(el.shadowRoot.querySelector('.total') !== null, 'Elemen .total harus ada');
        assert(el.shadowRoot.querySelector('#btn-hitung') !== null, 'Tombol hitung harus ada');
      });

      // Test 2: Total awal
      test('Kalkulasi: total awal benar (harga 5000 x jumlah 3 = 15000)', function () {
        var el = document.getElementById('kalkulator-demo');
        el.setAttribute('harga-satuan', '5000');
        el.setAttribute('jumlah', '3');
        el.hitung();
        var totalText = el.shadowRoot.querySelector('#total').textContent;
        assert(totalText.includes('15.000') || totalText.includes('15000'), 'Total harus Rp 15.000, tetapi: ' + totalText);
      });

      // Test 3: Perubahan atribut
      test('Observasi: perubahan atribut jumlah memperbarui total', function () {
        var el = document.getElementById('kalkulator-demo');
        el.setAttribute('harga-satuan', '10000');
        el.setAttribute('jumlah', '2');
        el.hitung();
        var totalText = el.shadowRoot.querySelector('#total').textContent;
        assert(totalText.includes('20.000') || totalText.includes('20000'), 'Total harus Rp 20.000');
      });

      // Test 4: Event
      test('Event: tombol hitung mengirim event "hitung"', function () {
        var el = document.getElementById('kalkulator-demo');
        var eventFired = false;
        var handler = function () { eventFired = true; };
        el.addEventListener('hitung', handler);
        el.shadowRoot.querySelector('#btn-hitung').click();
        el.removeEventListener('hitung', handler);
        assert(eventFired, 'Event "hitung" harus dikirim saat tombol diklik');
      });

      // Test 5: Aksesibilitas
      test('Aksesibilitas: tombol memiliki teks yang bermakna', function () {
        var el = document.getElementById('kalkulator-demo');
        var btn = el.shadowRoot.querySelector('#btn-hitung');
        assert(btn.textContent.trim().length > 0, 'Tombol harus memiliki teks');
      });

      // Test 6: Atribut dinamis (tanpa nilai)
      test('Edge case: harga-satuan dan jumlah tidak diset', function () {
        var tempEl = document.createElement('kalkulator-buku');
        document.body.appendChild(tempEl);
        tempEl.hitung();
        var totalText = tempEl.shadowRoot.querySelector('#total').textContent;
        document.body.removeChild(tempEl);
        assert(totalText.includes('0'), 'Jika tidak ada atribut, total harus 0');
      });

      tampilkanHasilTest();
    }

    function tampilkanHasilTest() {
      var output = document.getElementById('test-output');
      var passed = testResults.filter(function (t) { return t.status === 'pass'; }).length;
      var failed = testResults.filter(function (t) { return t.status === 'fail'; }).length;
      var total = testResults.length;

      var html =
        '<div style="margin-bottom:15px;padding:15px;background:' + (failed === 0 ? '#e8f5e9' : '#ffebee') + ';border-radius:8px;">' +
          '<strong>' + (failed === 0 ? '✅ Semua lulus!' : '❌ Ada yang gagal!') + '</strong> ' +
          passed + '/' + total + ' lulus' +
        '</div>';

      testResults.forEach(function (t) {
        var icon = t.status === 'pass' ? '✅' : '❌';
        var color = t.status === 'pass' ? '#4caf50' : '#ef5350';
        html +=
          '<div class="test-card">' +
            '<div class="test-header" style="color:' + color + ';">' +
              icon + ' ' + t.name +
              '<span>' + t.status.toUpperCase() + '</span>' +
            '</div>' +
            (t.message ? '<div class="test-body" style="color:#ef5350;">' + t.message + '</div>' : '') +
          '</div>';
      });

      output.innerHTML = html;
    }

    function resetTest() {
      document.getElementById('test-output').innerHTML = '<p style="color:#888;">Klik "Jalankan Test Suite" untuk memulai pengujian.</p>';
    }

    function ubahJumlah(id) {
      var el = document.getElementById(id);
      var jml = prompt('Masukkan jumlah buku:', el.getAttribute('jumlah') || '3');
      if (jml) {
        el.setAttribute('jumlah', jml);
      }
    }

    function cekTotal(id) {
      var el = document.getElementById(id);
      var totalEl = el.shadowRoot.querySelector('#total');
      alert('💰 Total saat ini: ' + totalEl.textContent);
    }
  </script>
</body>
</html>
```

## Poin Penting
- Gunakan `@open-wc/testing` untuk integrasi Mocha/Chai/Sinon yang siap pakai dengan fixture HTML untuk Web Components.
- Untuk pengujian cross-browser, gunakan Playwright atau @web/test-runner dengan berbagai browser (Chrome, Firefox, Safari).
- Shadow DOM bisa diakses dengan `element.shadowRoot.querySelector()` di test — pastikan menggunakan mode `open` untuk testability.
