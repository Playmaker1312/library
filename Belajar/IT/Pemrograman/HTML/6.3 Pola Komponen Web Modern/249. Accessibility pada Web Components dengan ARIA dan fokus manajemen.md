# Accessibility pada Web Components dengan ARIA dan Fokus Manajemen

## Analogi: Perpustakaan Digital

Perpustakaan digital harus bisa diakses oleh semua pengunjung, termasuk mereka yang menggunakan pembaca layar atau hanya keyboard. Rak buku yang dirancang dengan baik memiliki label Braille, kontras warna yang jelas, dan bisa dinavigasi dengan tombol tab. Web Components yang aksesibel perlu memberikan informasi yang sama: `role` dan `aria-*` seperti label Braille untuk pembaca layar, `tabindex` untuk navigasi keyboard, dan manajemen fokus untuk memastikan pengguna keyboard tahu di mana mereka berada. Tanpa ini, komponen kustom menjadi tidak berguna bagi pengguna dengan disabilitas.

## Penjelasan Detail

Accessibility (a11y) pada Web Components memerlukan perhatian khusus karena Shadow DOM menyembunyikan struktur internal dari pembaca layar. Prinsip utamanya: gunakan semantic HTML di dalam Shadow DOM, tambahkan `role` dan `aria-*` attribute yang sesuai, kelola fokus secara manual saat diperlukan (seperti modal atau autocomplete), dan uji dengan pembaca layar nyata. ARIA (Accessible Rich Internet Applications) menyediakan atribut seperti `aria-label`, `aria-labelledby`, `aria-expanded`, `aria-live` untuk memberikan konteks ke assistive technology. Manajemen fokus penting untuk komponen interaktif — gunakan `tabindex="0"` untuk membuat elemen focusable, dan panggil `.focus()` secara programatis saat diperlukan.

```html
  <div class="card">

  <script>
    var triggerElement = null;

    // Modal Dialog Component
    class ModalDialog extends HTMLElement {
      constructor() {
        super();
        this._shadow = this.attachShadow({ mode: 'open' });
        this._previousFocus = null;

        this._shadow.innerHTML =
          '<style>' +
            ':host { display: none; position: fixed; top: 0; left: 0; width: 100%; height: 100%; z-index: 1000; }' +
            ':host([open]) { display: block; }' +
            '.overlay { position: fixed; top: 0; left: 0; width: 100%; height: 100%; background: rgba(0,0,0,0.5); }' +
            '.modal { position: fixed; top: 50%; left: 50%; transform: translate(-50%, -50%); background: white; padding: 30px; border-radius: 12px; max-width: 500px; width: 90%; max-height: 80vh; overflow-y: auto; box-shadow: 0 10px 40px rgba(0,0,0,0.3); }' +
            '.close-btn { position: absolute; top: 10px; right: 15px; background: none; border: none; font-size: 1.5em; cursor: pointer; padding: 5px 10px; border-radius: 4px; }' +
            '.close-btn:focus-visible { outline: 3px solid #1565c0; outline-offset: 2px; }' +
            ':host([open]) .modal { animation: slideIn 0.2s ease; }' +
            '@keyframes slideIn { from { opacity: 0; transform: translate(-50%, -45%) scale(0.95); } to { opacity: 1; transform: translate(-50%, -50%) scale(1); } }' +
          '</style>' +
          '<div class="overlay" part="overlay"></div>' +
          '<div class="modal" role="dialog" aria-modal="true" part="modal">' +
            '<button class="close-btn" part="close-btn" aria-label="Tutup dialog">✕</button>' +
            '<slot name="konten"></slot>' +
          '</div>';

        this._shadow.querySelector('.close-btn').addEventListener('click', this.close.bind(this));
        this._shadow.querySelector('.overlay').addEventListener('click', this.close.bind(this));

        this._handleKeyDown = this._handleKeyDown.bind(this);
        this._focusTrap = this._focusTrap.bind(this);
      }

      connectedCallback() {
        document.addEventListener('keydown', this._handleKeyDown);
      }

      disconnectedCallback() {
        document.removeEventListener('keydown', this._handleKeyDown);
      }

      _handleKeyDown(e) {
        if (e.key === 'Escape' && this.hasAttribute('open')) {
          this.close();
        }
        if (e.key === 'Tab' && this.hasAttribute('open')) {
          this._focusTrap(e);
        }
      }

      _focusTrap(e) {
        var focusable = this._shadow.querySelectorAll('button, [tabindex]:not([tabindex="-1"])');
        if (focusable.length === 0) return;
        var first = focusable[0];
        var last = focusable[focusable.length - 1];

        if (e.shiftKey && document.activeElement === first) {
          e.preventDefault();
          last.focus();
        } else if (!e.shiftKey && document.activeElement === last) {
          e.preventDefault();
          first.focus();
        }
      }

      open() {
        this.setAttribute('open', '');
        this._previousFocus = document.activeElement;
        setTimeout(function () {
          var closeBtn = this._shadow.querySelector('.close-btn');
          if (closeBtn) closeBtn.focus();
        }.bind(this), 50);
      }

      close() {
        this.removeAttribute('open');
        if (this._previousFocus) {
          this._previousFocus.focus();
          this._previousFocus = null;
        }
      }
    }

    customElements.define('modal-dialog', ModalDialog);

    function bukaModal() {
      var modal = document.getElementById('modal-buku');
      triggerElement = document.activeElement;
      modal.open();
    }

    function tutupModal() {
      var modal = document.getElementById('modal-buku');
      modal.close();
    }

    // Accordion Component
    class AccordionBuku extends HTMLElement {
      constructor() {
        super();
        this._shadow = this.attachShadow({ mode: 'open' });
        this._shadow.innerHTML =
          '<style>' +
            '.accordion-item { border: 1px solid #e0e0e0; border-radius: 8px; margin: 8px 0; overflow: hidden; }' +
            '.accordion-header { display: flex; align-items: center; padding: 12px 16px; background: #f5f5f5; cursor: pointer; border: none; width: 100%; text-align: left; font-size: 1em; font-weight: bold; color: #1565c0; }' +
            '.accordion-header:hover { background: #e3f2fd; }' +
            '.accordion-header:focus-visible { outline: 3px solid #1565c0; outline-offset: -3px; }' +
            '.accordion-header .icon { margin-right: 10px; transition: transform 0.2s; }' +
            '.accordion-header[aria-expanded="true"] .icon { transform: rotate(90deg); }' +
            '.accordion-panel { padding: 0 16px; max-height: 0; overflow: hidden; transition: max-height 0.3s ease, padding 0.3s ease; }' +
            '.accordion-panel.open { max-height: 300px; padding: 16px; }' +
          '</style>' +
          '<div part="accordion"><slot name="item"></slot></div>';
      }

      connectedCallback() {
        this.addEventListener('slotchange', this._initItems.bind(this));
        this._initItems();
      }

      _initItems() {
        var items = this.querySelectorAll('[slot="item"]');
        items.forEach(function (item, index) {
          var header = item.querySelector('[slot="header"]');
          var panel = item.querySelector('[slot="panel"]');

          if (!header || !panel) return;

          var headerId = 'accordion-header-' + index;
          var panelId = 'accordion-panel-' + index;

          var btn = document.createElement('button');
          btn.className = 'accordion-header';
          btn.setAttribute('aria-expanded', 'false');
          btn.setAttribute('aria-controls', panelId);
          btn.setAttribute('id', headerId);
          btn.innerHTML = '<span class="icon">▶</span> ' + header.textContent;

          var panelDiv = document.createElement('div');
          panelDiv.className = 'accordion-panel';
          panelDiv.setAttribute('role', 'region');
          panelDiv.setAttribute('aria-labelledby', headerId);
          panelDiv.setAttribute('id', panelId);
          panelDiv.innerHTML = panel.innerHTML;

          item.innerHTML = '';
          item.appendChild(btn);
          item.appendChild(panelDiv);

          btn.addEventListener('click', function () {
            var isOpen = btn.getAttribute('aria-expanded') === 'true';
            btn.setAttribute('aria-expanded', String(!isOpen));
            panelDiv.classList.toggle('open');
          });
        });
      }
    }

    customElements.define('accordion-buku', AccordionBuku);
  </script>
</body>
</html>
```

## Poin Penting
- Selalu gunakan semantic HTML di dalam Shadow DOM — jangan andalkan ARIA saja, karena ARIA tidak mengubah perilaku atau semantik elemen.
- Kelola fokus secara manual untuk komponen seperti modal, autocomplete, dan sidebar — pastikan fokus tidak "kabur" ke luar komponen.
- Gunakan `aria-live` region untuk memberi tahu pembaca layar tentang perubahan konten dinamis tanpa memindahkan fokus.
