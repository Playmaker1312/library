# Background Sync API untuk sinkronisasi data saat offline

## Analogi: Perpustakaan Digital

Background Sync seperti petugas perpustakaan yang mencatat permintaan anggota saat sedang ada perbaikan jaringan. Ketika listrik padam atau jaringan terputus, anggota tetap bisa menitipkan slip permintaan buku. Petugas menyimpan semua slip itu dalam kotak khusus. Begitu jaringan kembali normal, petugas segera memproses semua slip yang tertunda. Pengunjung tidak perlu mengingat untuk mengirim ulang permintaan karena petugas menangani semuanya secara otomatis.

## Penjelasan Detail

Background Sync API memungkinkan aplikasi web mendefer tugas hingga koneksi internet tersedia kembali. API ini sangat berguna untuk aplikasi yang perlu mengirim data ke server, seperti formulir offline, chat messages, atau pembaruan data. Prosesnya dimulai dengan mendaftarkan sync event di Service Worker dari halaman utama menggunakan `registration.sync.register('nama-sync')`. Saat koneksi kembali, browser akan membangkitkan event `sync` di Service Worker yang dapat ditangani untuk melakukan sinkronisasi. Sync events bersifat "best-effort" tetapi browser akan terus mencoba hingga berhasil. Ada juga `Periodic Background Sync` untuk sinkronisasi berkala meskipun aplikasi tidak aktif. API ini memerlukan Service Worker yang aktif dan koneksi HTTPS. Tidak semua browser mendukung fitur ini, jadi penting untuk melakukan feature detection.

## Contoh Kode Implementasi

```html
<!DOCTYPE html>
<html lang="id">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>Background Sync Demo</title>
</head>
<body>
  <h1>Catatan Offline</h1>
  <textarea id="catatanInput" rows="3" cols="40" placeholder="Tulis catatan..."></textarea>
  <br>
  <button onclick="simpanCatatan()">Simpan Catatan</button>
  <ul id="daftarCatatan"></ul>

  <script>
    async function simpanCatatan() {
      const input = document.getElementById('catatanInput');
      const teks = input.value.trim();
      if (!teks) return;

      const catatan = {
        id: Date.now(),
        teks: teks,
        waktu: new Date().toISOString()
      };

      // Simpan ke IndexedDB dulu
      const db = await bukaDatabase();
      await simpanKeDB(db, catatan);
      tampilkanCatatan();

      // Daftarkan background sync
      if ('serviceWorker' in navigator && 'SyncManager' in window) {
        const reg = await navigator.serviceWorker.ready;
        await reg.sync.register('sync-catatan');
        console.log('Background sync terdaftar');
      } else {
        // Fallback: kirim langsung
        kirimKeServer(catatan);
      }

      input.value = '';
    }

    async function bukaDatabase() {
      return new Promise((resolve, reject) => {
        const req = indexedDB.open('CatatanDB', 1);
        req.onupgradeneeded = () => {
          req.result.createObjectStore('catatan', { keyPath: 'id' });
        };
        req.onsuccess = () => resolve(req.result);
        req.onerror = () => reject(req.error);
      });
    }

    async function simpanKeDB(db, catatan) {
      const tx = db.transaction('catatan', 'readwrite');
      return new Promise((resolve, reject) => {
        tx.objectStore('catatan').add(catatan);
        tx.oncomplete = () => resolve();
        tx.onerror = () => reject(tx.error);
      });
    }

    async function kirimKeServer(catatan) {
      console.log('Mengirim catatan:', catatan);
      // Simulasi fetch ke server
    }

    async function tampilkanCatatan() {
      const db = await bukaDatabase();
      const tx = db.transaction('catatan', 'readonly');
      const semua = await new Promise(resolve => {
        const result = [];
        tx.objectStore('catatan').openCursor().onsuccess = e => {
          const cursor = e.target.result;
          if (cursor) {
            result.push(cursor.value);
            cursor.continue();
          } else {
            resolve(result);
          }
        };
      });
      const ul = document.getElementById('daftarCatatan');
      ul.innerHTML = semua.map(c => `<li>${c.teks} <small>${new Date(c.waktu).toLocaleString()}</small></li>`).join('');
    }

    if ('serviceWorker' in navigator) {
      navigator.serviceWorker.register('sw-sync.js');
    }
  </script>
</body>
</html>
```

```javascript
// sw-sync.js
self.addEventListener('install', () => self.skipWaiting());
self.addEventListener('activate', event => event.waitUntil(clients.claim()));

self.addEventListener('sync', event => {
  if (event.tag === 'sync-catatan') {
    event.waitUntil(sinkronisasiCatatan());
  }
});

async function sinkronisasiCatatan() {
  const db = await bukaDatabase();
  const tx = db.transaction('catatan', 'readonly');
  const catatanList = await new Promise(resolve => {
    const result = [];
    tx.objectStore('catatan').openCursor().onsuccess = e => {
      const cursor = e.target.result;
      if (cursor) {
        result.push(cursor.value);
        cursor.continue();
      } else {
        resolve(result);
      }
    };
  });

  for (const catatan of catatanList) {
    try {
      const res = await fetch('/api/catatan', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify(catatan)
      });
      if (res.ok) {
        const tx2 = db.transaction('catatan', 'readwrite');
        tx2.objectStore('catatan').delete(catatan.id);
        await new Promise(r => { tx2.oncomplete = r; });
      }
    } catch (err) {
      console.error('Gagal sinkron:', catatan.id, err);
      throw err; // Browser akan coba lagi
    }
  }
}

function bukaDatabase() {
  return new Promise((resolve, reject) => {
    const req = indexedDB.open('CatatanDB', 1);
    req.onupgradeneeded = () => {
      req.result.createObjectStore('catatan', { keyPath: 'id' });
    };
    req.onsuccess = () => resolve(req.result);
    req.onerror = () => reject(req.error);
  });
}
```

## Poin Penting
- Background Sync memungkinkan aplikasi menunda pekerjaan hingga koneksi tersedia.
- Sync event terdaftar di Service Worker dengan tag unik untuk identifikasi.
- Data yang belum tersinkron biasanya disimpan sementara di IndexedDB sebelum dikirim.
