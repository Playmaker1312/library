# 290. Mengimplementasikan Debounce Function untuk Menunda Request saat Mengetik

## Definisi & Target
Membuat fungsi debounce JavaScript yang menunda eksekusi fungsi sampai user selesai mengetik selama periode tertentu.

## Analogi Alur
Debounce seperti pintu otomatis: pintu tidak akan terbuka setiap kali ada gerakan kecil, tapi hanya jika ada gerakan konsisten selama beberapa detik. Ini mencegah pintu terbuka-tutup terus.

## Poin-Poin Penting
- Fungsi debounce menerima callback dan delay
- Return function baru yang memiliki timer internal
- ClearTimeout setiap kali fungsi dipanggil
- SetTimeout baru dengan delay yang ditentukan
- this dan arguments harus di-pass ke callback
- Bisa menggunakan leading option (eksekusi pertama langsung)
- Implementasi: function debounce(fn, delay) { let timer; return function(...args) { clearTimeout(timer); timer = setTimeout(() => fn.apply(this, args), delay); }; }

let keyPressCount = 0, apiCallCount = 0, savedCount = 0;
const delay = 1000;

function debounce(fn, delay) {
let timer = null;
return function(...args) {
const context = this;
clearTimeout(timer);
timer = setTimeout(() => fn.apply(context, args), delay);
};
}

function simulateAPICall(query) {
apiCallCount++;
document.getElementById('apiCount').textContent = apiCallCount;
const log = document.getElementById('logArea');
const entry = document.createElement('div');
entry.className = 'log-entry api';
entry.textContent = `🔍 API Request: "${query}" (${new Date().toLocaleTimeString()})`;
log.appendChild(entry);
log.scrollTop = log.scrollHeight;
const saved = keyPressCount - apiCallCount;
document.getElementById('savedCount').textContent = saved;
}

const debouncedAPI = debounce(simulateAPICall, delay);

document.getElementById('demoInput').addEventListener('input', function() {
keyPressCount++;
document.getElementById('keyCount').textContent = keyPressCount;
const query = this.value.trim();
const log = document.getElementById('logArea');
const entry = document.createElement('div');
entry.className = 'log-entry typing';
entry.textContent = `⌨️ Mengetik... "${query}"`;
log.appendChild(entry);
log.scrollTop = log.scrollHeight;
if (query.length >= 2) debouncedAPI(query);
else {
const saved = keyPressCount - apiCallCount;
document.getElementById('savedCount').textContent = saved;
}
});
