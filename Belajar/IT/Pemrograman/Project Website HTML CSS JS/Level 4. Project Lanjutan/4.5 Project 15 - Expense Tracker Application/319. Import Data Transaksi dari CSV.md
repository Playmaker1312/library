# 319. Import Data Transaksi dari CSV

## Definisi & Target
Menambahkan fitur import data transaksi dari file CSV yang bisa diupload oleh user.

## Analogi Alur
Import CSV seperti memasukkan data dari spreadsheet ke buku kas: Anda tinggal mengupload file, dan semua transaksi langsung tercatat tanpa harus mengetik manual.

## Poin-Poin Penting
- Input file type="file" accept=".csv"
- Parse CSV dengan FileReader
- Validasi format: header harus sesuai
- Kolom: tanggal, tipe, kategori, catatan, jumlah
- Konversi tipe data (string ke number, dll)
- Merge dengan data existing (append)
- Tampilkan jumlah data yang berhasil diimport

<script>
let existingTransactions = [
{ id: Date.now(), date: '2025-01-15', type: 'income', category: 'Gaji', note: 'Gaji Bulanan', amount: 5000000 }
];

const dropZone = document.getElementById('dropZone');
['dragover', 'dragenter'].forEach(evt => {
dropZone.addEventListener(evt, (e) => { e.preventDefault(); dropZone.classList.add('dragover'); });
});
['dragleave', 'drop'].forEach(evt => {
dropZone.addEventListener(evt, (e) => { e.preventDefault(); dropZone.classList.remove('dragover'); });
});
dropZone.addEventListener('drop', (e) => {
e.preventDefault();
const file = e.dataTransfer.files[0];
if (file) handleFile(file);
});

function handleFile(file) {
const result = document.getElementById('result');
if (!file || !file.name.endsWith('.csv')) {
result.className = 'result error';
result.textContent = '❌ File harus berformat CSV';
result.style.display = 'block';
return;
}

const reader = new FileReader();
reader.onload = function(e) {
try {
const text = e.target.result;
const lines = text.split('\n').filter(line => line.trim());
if (lines.length < 2) throw new Error('File kosong');

const header = lines[0].toLowerCase();
if (!header.includes('tanggal') || !header.includes('jumlah')) {
throw new Error('Format header tidak sesuai');
}

let imported = 0;
for (let i = 1; i < lines.length; i++) {
const cols = lines[i].split(',').map(c => c.trim().replace(/^"|"$/g, ''));
if (cols.length < 5) continue;
const [date, typeStr, category, note, amountStr] = cols;
const amount = parseInt(amountStr.replace(/[^0-9]/g, ''));
if (!date || isNaN(amount) || amount <= 0) continue;
const type = typeStr.toLowerCase().includes('pemasukan') ? 'income' : 'expense';
existingTransactions.push({
id: Date.now() + imported,
date,
type,
category,
note,
amount
});
imported++;
}

if (imported === 0) throw new Error('Tidak ada data valid');
result.className = 'result success';
result.textContent = `✅ Berhasil mengimport ${imported} transaksi! Total: ${existingTransactions.length} transaksi`;
result.style.display = 'block';
} catch (err) {
result.className = 'result error';
result.textContent = '❌ Gagal import: ' + err.message;
result.style.display = 'block';
}
};
reader.readAsText(file);
}
</script>
