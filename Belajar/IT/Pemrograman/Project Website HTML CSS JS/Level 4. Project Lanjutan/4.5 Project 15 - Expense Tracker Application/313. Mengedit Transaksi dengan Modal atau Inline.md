# 313. Mengedit Transaksi dengan Modal atau Inline

## Definisi & Target
Menambahkan fitur edit transaksi yang sudah ada, baik melalui modal popup atau inline editing.

## Analogi Alur
Edit transaksi seperti mengoreksi buku kas: Anda bisa mengubah jumlah yang salah, mengganti kategori, atau memperbaiki tanggal transaksi.

## Poin-Poin Penting
- Tombol edit (✏️) di setiap baris transaksi
- Modal form yang terisi data lama
- Simpan perubahan dengan update array
- Validasi input seperti form tambah
- Batalkan edit tanpa perubahan
- Update tampilan setelah edit

## Kode Implementasi (Gabungan) - HTML, CSS, JavaScript
```html
<!DOCTYPE html>
<html lang="id">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Edit Transaksi</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; background: #f0f2f5; padding: 30px; }
.container { max-width: 550px; margin: 0 auto; }
h1 { text-align: center; color: #333; font-size: 20px; margin-bottom: 16px; }
.transaction-list { background: white; border-radius: 14px; box-shadow: 0 2px 15px rgba(0,0,0,0.06); }
.transaction-item { display: flex; justify-content: space-between; align-items: center; padding: 12px 16px; border-bottom: 1px solid #f0f0f0; }
.transaction-item:last-child { border-bottom: none; }
.t-info { flex: 1; }
.t-note { font-size: 14px; color: #333; }
.t-meta { font-size: 11px; color: #999; }
.t-amount { font-size: 14px; font-weight: 600; margin: 0 12px; white-space: nowrap; }
.t-amount.income { color: #2ecc71; }
.t-amount.expense { color: #e74c3c; }
.btn-edit { background: none; border: none; color: #ccc; font-size: 16px; cursor: pointer; padding: 4px 8px; border-radius: 6px; }
.btn-edit:hover { color: #667eea; background: #f0f2ff; }

.modal-overlay { display: none; position: fixed; inset: 0; background: rgba(0,0,0,0.4); justify-content: center; align-items: center; z-index: 1000; }
.modal-overlay.show { display: flex; }
.modal { background: white; border-radius: 18px; padding: 28px; width: 90%; max-width: 420px; box-shadow: 0 20px 60px rgba(0,0,0,0.3); animation: modalIn 0.3s; }
@keyframes modalIn { from { opacity: 0; transform: scale(0.9); } to { opacity: 1; transform: scale(1); } }
.modal h2 { text-align: center; color: #333; margin-bottom: 18px; font-size: 18px; }
.form-group { margin-bottom: 14px; }
.form-group label { display: block; font-size: 13px; color: #555; margin-bottom: 4px; }
.form-group input, .form-group select { width: 100%; padding: 10px 12px; border: 2px solid #ddd; border-radius: 10px; font-size: 14px; outline: none; }
.form-group input:focus, .form-group select:focus { border-color: #667eea; }
.modal-actions { display: flex; gap: 10px; margin-top: 18px; }
.btn-primary { flex: 1; padding: 12px; background: #667eea; border: none; border-radius: 10px; color: white; font-size: 15px; font-weight: 600; cursor: pointer; }
.btn-secondary { flex: 1; padding: 12px; background: #f0f2f5; border: none; border-radius: 10px; color: #555; font-size: 15px; cursor: pointer; }
</style>
</head>
<body>
<div class="container">
<h1>✏️ Edit Transaksi</h1>
<div class="transaction-list" id="transactionList"></div>
</div>

<div class="modal-overlay" id="modalEdit">
<div class="modal">
<h2>✏️ Edit Transaksi</h2>
<select id="editType" style="width:100%;padding:10px;border:2px solid #ddd;border-radius:10px;margin-bottom:14px;font-size:14px;outline:none">
<option value="income">💰 Pemasukan</option><option value="expense">💸 Pengeluaran</option>
</select>
<div class="form-group">
<label>Catatan</label><input type="text" id="editNote" placeholder="Catatan">
</div>
<div class="form-group">
<label>Jumlah (Rp)</label><input type="number" id="editAmount" placeholder="50000" min="1">
</div>
<div class="form-group">
<label>Kategori</label>
<select id="editCategory" style="width:100%;padding:10px;border:2px solid #ddd;border-radius:10px;font-size:14px;outline:none">
<option value="Makanan">Makanan</option><option value="Transport">Transport</option>
<option value="Hiburan">Hiburan</option><option value="Belanja">Belanja</option>
<option value="Tagihan">Tagihan</option><option value="Gaji">Gaji</option><option value="Lainnya">Lainnya</option>
</select>
</div>
<div class="modal-actions">
<button class="btn-secondary" onclick="closeModal()">Batal</button>
<button class="btn-primary" onclick="saveEdit()">Simpan</button>
</div>
</div>
</div>

<script>
let transactions = [
{ id: 1, type: 'income', note: 'Gaji Bulanan', category: 'Gaji', amount: 5000000, date: '2025-01-15' },
{ id: 2, type: 'expense', note: 'Nasi Goreng', category: 'Makanan', amount: 25000, date: '2025-01-16' },
{ id: 3, type: 'expense', note: 'Bensin', category: 'Transport', amount: 100000, date: '2025-01-17' },
{ id: 4, type: 'expense', note: 'Netflix', category: 'Hiburan', amount: 65000, date: '2025-01-19' }
];

let editingId = null;

function formatRp(num) { return 'Rp' + num.toLocaleString('id-ID'); }

function render() {
const list = document.getElementById('transactionList');
list.innerHTML = transactions.map(t => `
<div class="transaction-item">
  <div class="t-info">
    <div class="t-note">${t.note}</div>
    <div class="t-meta">${t.category} • Rp${t.amount.toLocaleString()}</div>
  </div>
  <div class="t-amount ${t.type}">${formatRp(t.amount)}</div>
  <button class="btn-edit" onclick="openEdit(${t.id})">✏️</button>
</div>
`).join('');
}

function openEdit(id) {
const t = transactions.find(tr => tr.id === id);
if (!t) return;
editingId = id;
document.getElementById('editType').value = t.type;
document.getElementById('editNote').value = t.note;
document.getElementById('editAmount').value = t.amount;
document.getElementById('editCategory').value = t.category;
document.getElementById('modalEdit').classList.add('show');
}

function closeModal() {
document.getElementById('modalEdit').classList.remove('show');
editingId = null;
}

function saveEdit() {
const t = transactions.find(tr => tr.id === editingId);
if (!t) return;
t.type = document.getElementById('editType').value;
t.note = document.getElementById('editNote').value.trim();
t.amount = parseInt(document.getElementById('editAmount').value);
t.category = document.getElementById('editCategory').value;
if (!t.note || !t.amount || t.amount <= 0) return;
closeModal();
render();
}

render();
</script>
</body>
</html>
```

## Cara Integrasi
Simpan sebagai HTML. Klik ✏️ untuk membuka modal edit, ubah data, klik Simpan.
