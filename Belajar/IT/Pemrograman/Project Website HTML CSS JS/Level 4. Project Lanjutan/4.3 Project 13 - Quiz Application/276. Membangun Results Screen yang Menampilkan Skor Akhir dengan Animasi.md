# 276. Membangun Results Screen yang Menampilkan Skor Akhir dengan Animasi

## Definisi & Target
Membuat layar hasil akhir quiz dengan animasi yang menampilkan skor, pesan sesuai performa, dan tombol untuk mengulang quiz.

## Analogi Alur
Results screen seperti papan pengumuman kelulusan: nama-nama yang lulus diumumkan dengan meriah, ada tepuk tangan (animasi), dan ada ajakan untuk mengikuti lagi.

## Poin-Poin Penting
- Tampilkan skor akhir dengan animasi counter
- Pesan berbeda berdasarkan performa (sempurna, bagus, sedang, kurang)
- Ikon atau emoji yang sesuai
- Animasi confetti atau bintang untuk skor tinggi
- Tombol "Coba Lagi" untuk restart quiz
- Statistik: benar, salah, tidak dijawab
- Waktu total penyelesaian

## Kode Implementasi (Gabungan) - HTML, CSS, JavaScript
```html
<!DOCTYPE html>
<html lang="id">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Results Screen</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; background: linear-gradient(135deg, #667eea 0%, #764ba2 100%); min-height: 100vh; display: flex; justify-content: center; align-items: center; padding: 20px; }
.result-card { background: white; border-radius: 24px; padding: 40px; width: 100%; max-width: 480px; box-shadow: 0 25px 80px rgba(0,0,0,0.2); text-align: center; animation: scaleIn 0.6s ease; }
@keyframes scaleIn { from { opacity: 0; transform: scale(0.8); } to { opacity: 1; transform: scale(1); } }
.result-icon { font-size: 64px; margin-bottom: 10px; animation: bounce 1s ease infinite; }
@keyframes bounce { 0%,100% { transform: translateY(0); } 50% { transform: translateY(-10px); } }
.result-title { font-size: 28px; font-weight: 700; margin-bottom: 5px; }
.result-subtitle { color: #888; font-size: 14px; margin-bottom: 20px; }
.score-circle { width: 140px; height: 140px; border-radius: 50%; margin: 0 auto 20px; background: conic-gradient(#667eea var(--pct, 0%), #e0e0e0 var(--pct, 0%)); display: flex; align-items: center; justify-content: center; }
.score-circle-inner { width: 120px; height: 120px; border-radius: 50%; background: white; display: flex; flex-direction: column; align-items: center; justify-content: center; }
.score-number { font-size: 42px; font-weight: 700; color: #667eea; line-height: 1; }
.score-total { font-size: 16px; color: #888; }
.stats { display: grid; grid-template-columns: repeat(3, 1fr); gap: 10px; margin-bottom: 20px; }
.stat-box { background: #f5f7ff; border-radius: 12px; padding: 12px; }
.stat-label { color: #888; font-size: 11px; }
.stat-value { font-size: 22px; font-weight: 600; margin-top: 4px; }
.stat-value.green { color: #4caf50; }
.stat-value.red { color: #e94560; }
.stat-value.gray { color: #888; }
.restart-btn { padding: 16px 40px; background: linear-gradient(135deg, #667eea, #764ba2); color: white; border: none; border-radius: 12px; font-size: 18px; font-weight: 600; cursor: pointer; transition: all 0.3s; }
.restart-btn:hover { transform: translateY(-2px); box-shadow: 0 10px 30px rgba(102,126,234,0.4); }
.confetti { position: fixed; font-size: 24px; animation: confettiFall 3s linear forwards; pointer-events: none; z-index: 10; }
@keyframes confettiFall { 0% { transform: translateY(-100vh) rotate(0deg); opacity: 1; } 100% { transform: translateY(100vh) rotate(720deg); opacity: 0; } }
</style>
</head>
<body>
<div class="result-card" id="resultCard">
<div class="result-icon" id="resultIcon">🎉</div>
<div class="result-title" id="resultTitle">Quiz Selesai!</div>
<div class="result-subtitle" id="resultSubtitle">Lihat hasil kamu di bawah ini</div>
<div class="score-circle" id="scoreCircle" style="--pct:0%">
<div class="score-circle-inner"><div class="score-number" id="scoreNumber">0</div><div class="score-total">/ 10</div></div>
</div>
<div class="stats">
<div class="stat-box"><div class="stat-label">✅ Benar</div><div class="stat-value green" id="correctCount">0</div></div>
<div class="stat-box"><div class="stat-label">❌ Salah</div><div class="stat-value red" id="wrongCount">0</div></div>
<div class="stat-box"><div class="stat-label">⏭️ Lewat</div><div class="stat-value gray" id="skippedCount">0</div></div>
</div>
<button class="restart-btn" onclick="restartQuiz()">🔄 Coba Lagi</button>
</div>
<script>
function triggerConfetti() {
const emojis = ['🎉', '⭐', '🎊', '✨', '🏆', '💫'];
for (let i = 0; i < 30; i++) {
setTimeout(() => {
const el = document.createElement('div');
el.className = 'confetti';
el.textContent = emojis[Math.floor(Math.random() * emojis.length)];
el.style.left = Math.random() * 100 + 'vw';
el.style.animationDuration = (2 + Math.random() * 2) + 's';
el.style.fontSize = (18 + Math.random() * 20) + 'px';
document.body.appendChild(el);
setTimeout(() => el.remove(), 4000);
}, i * 100);
}
}

function showResults(correct, wrong, skipped) {
const total = correct + wrong + skipped;
const pct = total > 0 ? (correct / total) * 100 : 0;

let icon, title, subtitle;
if (pct >= 90) { icon = '🏆'; title = 'Luar Biasa!'; subtitle = 'Kamu benar-benar ahli!'; }
else if (pct >= 70) { icon = '🎉'; title = 'Bagus Sekali!'; subtitle = 'Pertahankan prestasimu!'; }
else if (pct >= 50) { icon = '👍'; title = 'Cukup Baik'; subtitle = 'Masih perlu belajar lagi.'; }
else { icon = '💪'; title = 'Semangat!'; subtitle = 'Jangan menyerah, coba lagi!'; }

document.getElementById('resultIcon').textContent = icon;
document.getElementById('resultTitle').textContent = title;
document.getElementById('resultSubtitle').textContent = subtitle;
document.getElementById('correctCount').textContent = correct;
document.getElementById('wrongCount').textContent = wrong;
document.getElementById('skippedCount').textContent = skipped;

const circle = document.getElementById('scoreCircle');
const number = document.getElementById('scoreNumber');
let count = 0;
const interval = setInterval(() => {
count++;
number.textContent = count;
circle.style.setProperty('--pct', `${(count / total) * 100}%`);
if (count >= correct) clearInterval(interval);
}, 50);

if (pct >= 70) triggerConfetti();
}

function restartQuiz() {
alert('Quiz akan dimulai ulang!');
location.reload();
}

showResults(8, 1, 1);
</script>
</body>
</html>
```

## Cara Integrasi
Simpan sebagai HTML. Lihat results screen dengan animasi score counter, circle progress, confetti untuk skor tinggi, dan statistik lengkap.
