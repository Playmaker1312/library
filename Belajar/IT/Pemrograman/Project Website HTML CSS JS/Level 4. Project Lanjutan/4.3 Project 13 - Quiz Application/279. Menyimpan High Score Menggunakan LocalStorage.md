# 279. Menyimpan High Score Menggunakan LocalStorage

## Definisi & Target
Menyimpan skor tertinggi (high score) yang pernah dicapai user ke LocalStorage agar dapat dibandingkan dengan skor baru.

## Analogi Alur
High score seperti rekor di papan arcade: setiap kali Anda bermain, skor Anda dicatat. Jika skor baru melebihi rekor sebelumnya, papan rekor diperbarui. Rekor tetap ada meskipun mesin dimatikan.

## Poin-Poin Penting
- Simpan high score dengan key tertentu di LocalStorage
- Load high score saat halaman dimuat
- Bandingkan skor baru dengan high score yang tersimpan
- Jika skor baru lebih tinggi, update high score
- Tampilkan high score di results screen
- Animasi saat high score terpecahkan (new record!)
- Tampilkan di welcome screen sebagai motivasi

## Kode Implementasi (Gabungan) - HTML, CSS, JavaScript
```html
<!DOCTYPE html>
<html lang="id">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>High Score Quiz</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; background: linear-gradient(135deg, #1e3c72, #2a5298); min-height: 100vh; display: flex; justify-content: center; align-items: center; padding: 20px; }
.container { background: rgba(255,255,255,0.1); backdrop-filter: blur(15px); border-radius: 24px; padding: 30px; width: 100%; max-width: 480px; border: 1px solid rgba(255,255,255,0.15); color: white; text-align: center; }
h1 { font-size: 24px; margin-bottom: 5px; }
.subtitle { opacity: 0.6; font-size: 13px; margin-bottom: 25px; }
.high-score-badge { background: rgba(255,215,0,0.15); border: 1px solid rgba(255,215,0,0.3); border-radius: 12px; padding: 12px 20px; display: inline-flex; align-items: center; gap: 10px; margin-bottom: 25px; font-size: 14px; }
.high-score-badge .score { color: #ffd700; font-weight: 600; font-size: 18px; }
.quiz-area { min-height: 200px; display: flex; flex-direction: column; align-items: center; justify-content: center; }
.question-text { font-size: 18px; margin-bottom: 20px; }
.options { display: grid; gap: 10px; width: 100%; }
.option { padding: 14px 18px; border: 1px solid rgba(255,255,255,0.15); border-radius: 12px; cursor: pointer; transition: all 0.3s; }
.option:hover { border-color: rgba(255,255,255,0.4); background: rgba(255,255,255,0.05); }
.option.correct { background: rgba(76,175,80,0.3); border-color: #4caf50; }
.option.wrong { background: rgba(233,69,96,0.3); border-color: #e94560; }
.next-btn { padding: 14px 40px; margin-top: 20px; border: none; border-radius: 12px; background: rgba(255,255,255,0.15); color: white; font-size: 16px; cursor: pointer; display: none; }
.next-btn.show { display: inline-block; }
.result-area { display: none; }
.result-area.show { display: block; }
.result-score { font-size: 56px; font-weight: 700; margin: 15px 0; }
.new-record { color: #ffd700; font-size: 18px; animation: pulse 1s infinite; }
@keyframes pulse { 0%,100% { opacity: 1; } 50% { opacity: 0.5; } }
.restart-btn { padding: 14px 40px; margin-top: 20px; border: none; border-radius: 12px; background: rgba(255,255,255,0.15); color: white; font-size: 16px; cursor: pointer; }
</style>
</head>
<body>
<div class="container">
<h1>🏆 High Score</h1>
<p class="subtitle">Kalahkan rekor tertinggi!</p>
<div class="high-score-badge">🏅 High Score: <span class="score" id="highScoreDisplay">0</span></div>
<div class="quiz-area" id="quizArea">
<div class="question-text" id="questionText">Memulai quiz...</div>
<div class="options" id="optionsContainer"></div>
<button class="next-btn" id="nextBtn" onclick="nextQuestion()">Selanjutnya →</button>
</div>
<div class="result-area" id="resultArea">
<h2>🎉 Quiz Selesai!</h2>
<div class="result-score" id="finalScore">0/5</div>
<div id="recordMessage"></div>
<button class="restart-btn" onclick="restartQuiz()">🔄 Main Lagi</button>
</div>
</div>
<script>
const STORAGE_KEY = 'quiz_highscore';
let questions = [
{ q: 'Apa ibu kota Indonesia?', o: ['Jakarta', 'Bandung', 'Surabaya', 'Yogyakarta'], c: 0 },
{ q: 'Siapa presiden pertama RI?', o: ['Soeharto', 'Soekarno', 'Habibie', 'Megawati'], c: 1 },
{ q: 'CSS singkatan dari?', o: ['Cascading Style Sheets', 'Creative System', 'Computer Sheets', 'Colorful Style'], c: 0 },
{ q: 'Planet terdekat matahari?', o: ['Venus', 'Bumi', 'Merkurius', 'Mars'], c: 2 },
{ q: 'HTML singkatan dari?', o: ['Hyper Text Markup Language', 'High Tech Language', 'Home Tool Language', 'Hyperlink Language'], c: 0 }
];
let current = 0, score = 0, answered = false;

function loadHighScore() {
try { return parseInt(localStorage.getItem(STORAGE_KEY)) || 0; } catch { return 0; }
}

function saveHighScore(val) {
try { localStorage.setItem(STORAGE_KEY, val.toString()); } catch {}
}

function updateHighScoreDisplay() {
document.getElementById('highScoreDisplay').textContent = loadHighScore();
}

function render() {
const q = questions[current];
answered = false;
document.getElementById('questionText').textContent = q.q;
document.getElementById('nextBtn').classList.remove('show');
const container = document.getElementById('optionsContainer');
container.innerHTML = q.o.map((opt, i) => `<div class="option" onclick="selectAnswer(${i})">${opt}</div>`).join('');
}

function selectAnswer(idx) {
if (answered) return;
answered = true;
const q = questions[current];
document.querySelectorAll('.option').forEach((el, i) => {
if (i === q.c) el.classList.add('correct');
if (i === idx && idx !== q.c) el.classList.add('wrong');
});
if (idx === q.c) score++;
document.getElementById('nextBtn').classList.add('show');
}

function nextQuestion() {
current++;
if (current < questions.length) render();
else {
document.getElementById('quizArea').style.display = 'none';
const ra = document.getElementById('resultArea');
ra.classList.add('show');
document.getElementById('finalScore').textContent = `${score}/${questions.length}`;
const hs = loadHighScore();
const rm = document.getElementById('recordMessage');
if (score > hs) {
saveHighScore(score);
updateHighScoreDisplay();
rm.innerHTML = '<div class="new-record">🏆 NEW RECORD! 🏆</div>';
} else {
rm.innerHTML = `<p style="opacity:0.6">High score: ${hs}</p>`;
}
}
}

function restartQuiz() {
current = 0; score = 0;
document.getElementById('resultArea').classList.remove('show');
document.getElementById('quizArea').style.display = 'flex';
render();
}

updateHighScoreDisplay();
render();
</script>
</body>
</html>
```

## Cara Integrasi
Simpan sebagai HTML. Mainkan quiz dan lihat high score tersimpan. Jika skor melebihi rekor, tampilkan "NEW RECORD!".
