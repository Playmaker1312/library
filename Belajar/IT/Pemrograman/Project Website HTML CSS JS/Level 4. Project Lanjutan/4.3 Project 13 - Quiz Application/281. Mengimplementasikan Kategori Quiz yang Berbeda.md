# 281. Mengimplementasikan Kategori Quiz yang Berbeda

## Definisi & Target
Menambahkan kategori quiz yang berbeda (misal: Pengetahuan Umum, Teknologi, Sejarah) yang bisa dipilih user sebelum memulai quiz.

## Analogi Alur
Kategori quiz seperti memilih mata pelajaran di sekolah: Anda bisa memilih ingin ujian Matematika, IPA, atau Sejarah. Setiap kategori memiliki kumpulan soal yang berbeda.

## Poin-Poin Penting
- Array of categories dengan id dan nama
- Setiap pertanyaan memiliki properti category
- Filter pertanyaan berdasarkan kategori yang dipilih
- Tampilan pemilihan kategori yang menarik
- Simpan kategori yang dipilih di state
- Tampilkan kategori aktif di header quiz

```html
<script>
const allQuestions = {
general: [
{ q: 'Apa ibu kota Indonesia?', o: ['Jakarta', 'Bandung', 'Surabaya', 'Yogyakarta'], c: 0 },
{ q: 'Berapa jumlah provinsi di Indonesia?', o: ['34', '33', '35', '32'], c: 0 },
{ q: 'Apa bahasa resmi Indonesia?', o: ['Bahasa Indonesia', 'Bahasa Jawa', 'Bahasa Melayu', 'Bahasa Inggris'], c: 0 },
{ q: 'Apa mata uang Indonesia?', o: ['Rupiah', 'Ringgit', 'Dollar', 'Baht'], c: 0 },
{ q: 'Apa lambang negara Indonesia?', o: ['Garuda Pancasila', 'Bhinneka Tunggal Ika', 'Pancasila', 'UUD 1945'], c: 0 }
],
tech: [
{ q: 'Apa kepanjangan HTML?', o: ['Hyper Text Markup Language', 'High Tech Modern Language', 'Home Tool Markup Language', 'Hyperlink Text Management Language'], c: 0 },
{ q: 'CSS digunakan untuk?', o: ['Styling', 'Struktur', 'Database', 'Network'], c: 0 },
{ q: 'JavaScript adalah bahasa?', o: ['Pemrograman', 'Markup', 'Styling', 'Query'], c: 0 },
{ q: 'Apa itu API?', o: ['Application Programming Interface', 'Automated Program Interface', 'Advanced Programming Internet', 'Application Process Integration'], c: 0 },
{ q: 'LocalStorage menyimpan data di?', o: ['Browser', 'Server', 'Database', 'File'], c: 0 }
]
};
let currentCat = null, questions = [], current = 0, score = 0, answered = false;

function selectCategory(cat) {
currentCat = cat;
questions = allQuestions[cat];
current = 0; score = 0;
document.getElementById('catGrid').style.display = 'none';
document.getElementById('backBtn').style.display = 'block';
document.getElementById('quizHeader').classList.add('show');
document.getElementById('questionText').classList.add('show');
document.getElementById('optionsContainer').classList.add('show');
document.getElementById('catLabel').textContent = `📂 ${cat.charAt(0).toUpperCase() + cat.slice(1)}`;
render();
}

function showCategories() {
currentCat = null;
document.getElementById('catGrid').style.display = 'grid';
document.getElementById('backBtn').style.display = 'none';
document.getElementById('quizHeader').classList.remove('show');
document.getElementById('questionText').classList.remove('show');
document.getElementById('optionsContainer').classList.remove('show');
document.getElementById('nextBtn').classList.remove('show');
}

function render() {
const q = questions[current];
answered = false;
document.getElementById('progressLabel').textContent = `Soal ${current + 1}/${questions.length}`;
document.getElementById('questionText').textContent = q.q;
document.getElementById('nextBtn').classList.remove('show');
document.getElementById('optionsContainer').innerHTML = q.o.map((opt, i) => `<div class="option" onclick="selectAnswer(${i})">${opt}</div>`).join('');
}

function selectAnswer(idx) {
if (answered) return;
answered = true;
const q = questions[current];
document.querySelectorAll('.option').forEach((el, i) => {
if (i === q.c) el.classList.add('correct');
if (i === idx && idx !== q.c) el.classList.add('wrong');
});
if (idx === q.c) score++;
document.getElementById('nextBtn').classList.add('show');
}

function nextQuestion() {
current++;
if (current < questions.length) render();
else {
document.querySelector('.container').innerHTML = `<h1 style="text-align:center;color:#e94560">🎉 Selesai!</h1><p style="text-align:center;color:#fff;font-size:24px;margin:20px">Skor: ${score}/${questions.length}</p><button onclick="location.reload()" style="display:block;margin:20px auto;padding:14px 40px;background:#667eea;border:none;border-radius:12px;color:white;font-size:16px;cursor:pointer">🔄 Pilih Kategori Lagi</button>`;
}
}
</script>
</body>
</html>
```
