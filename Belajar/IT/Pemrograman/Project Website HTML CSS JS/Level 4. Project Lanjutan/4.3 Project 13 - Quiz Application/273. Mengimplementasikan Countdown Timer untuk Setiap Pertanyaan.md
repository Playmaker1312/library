# 273. Mengimplementasikan Countdown Timer untuk Setiap Pertanyaan

## Definisi & Target
Menambahkan countdown timer untuk setiap pertanyaan. Jika waktu habis, jawaban dianggap salah dan otomatis pindah ke pertanyaan berikutnya.

## Analogi Alur
Countdown timer seperti bom waktu di permainan: Anda harus menjawab sebelum waktu habis. Semakin cepat menjawab, semakin baik. Jika waktu habis, Anda kehilangan kesempatan.

## Poin-Poin Penting
- Timer countdown mundur dari 15-30 detik
- Tampilkan timer secara visual (angka + progress bar)
- Warna timer berubah ketika waktu tinggal sedikit (merah)
- Auto-submit saat waktu habis
- Timer berhenti setelah user menjawab
- Reset timer untuk setiap pertanyaan baru
- Gunakan setInterval dengan interval 1000ms

## Kode Implementasi (Gabungan) - HTML, CSS, JavaScript
```html
<!DOCTYPE html>
<html lang="id">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Quiz Timer</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; background: linear-gradient(135deg, #1a1a2e, #16213e); min-height: 100vh; display: flex; justify-content: center; align-items: center; padding: 20px; }
.container { background: #0f3460; border-radius: 20px; padding: 30px; width: 100%; max-width: 500px; }

.timer-section { display: flex; align-items: center; gap: 15px; margin-bottom: 20px; }
.timer-ring { width: 56px; height: 56px; border-radius: 50%; background: conic-gradient(#667eea var(--pct, 100%), #1a3a6e var(--pct, 100%)); display: flex; align-items: center; justify-content: center; flex-shrink: 0; }
.timer-ring-inner { width: 44px; height: 44px; border-radius: 50%; background: #0f3460; display: flex; align-items: center; justify-content: center; color: #fff; font-size: 18px; font-weight: 600; font-family: monospace; }
.timer-bar { flex: 1; height: 8px; background: #1a3a6e; border-radius: 4px; overflow: hidden; }
.timer-fill { height: 100%; background: #667eea; border-radius: 4px; transition: width 1s linear; }
.timer-bar.warning .timer-fill { background: #f39c12; }
.timer-bar.danger .timer-fill { background: #e94560; }
.question-text { color: #eee; font-size: 18px; margin-bottom: 20px; }
.options { display: grid; gap: 10px; }
.option { padding: 14px 18px; background: #1a3a6e; border: 2px solid #2a4a8e; border-radius: 12px; cursor: pointer; color: #ccc; font-size: 15px; transition: all 0.3s; }
.option:hover { border-color: #667eea; }
.option.correct { background: rgba(76,175,80,0.2); border-color: #4caf50; color: #4caf50; }
.option.wrong { background: rgba(233,69,96,0.2); border-color: #e94560; color: #e94560; }
.option.timeout { opacity: 0.5; border-color: #555; }
.next-btn { width: 100%; padding: 14px; margin-top: 15px; background: #667eea; border: none; border-radius: 12px; color: white; font-size: 16px; cursor: pointer; font-weight: 600; display: none; }
.next-btn.show { display: block; }
</style>
</head>
<body>
<div class="container">
<h1>⏱️ Quiz Timer</h1>
<div class="timer-section">
<div class="timer-ring" id="timerRing"><div class="timer-ring-inner" id="timerText">15</div></div>
<div class="timer-bar" id="timerBar"><div class="timer-fill" id="timerFill" style="width:100%"></div></div>
</div>
<div class="question-text" id="questionText">Memuat pertanyaan...</div>
<div class="options" id="optionsContainer"></div>
<button class="next-btn" id="nextBtn" onclick="nextQuestion()">Soal Berikutnya →</button>
</div>
<script>
const questions = [
{ q: 'Apa ibu kota Indonesia?', o: ['Jakarta', 'Bandung', 'Surabaya', 'Yogyakarta'], c: 0 },
{ q: 'Siapa presiden pertama RI?', o: ['Soeharto', 'Soekarno', 'Habibie', 'Megawati'], c: 1 },
{ q: 'CSS adalah singkatan dari?', o: ['Cascading Style Sheets', 'Creative Style System', 'Computer Style Sheets', 'Colorful Style Sheets'], c: 0 }
];
let current = 0, score = 0, timeLeft = 15, answered = false, timerInterval = null;
const TOTAL_TIME = 15;

function render() {
const q = questions[current];
answered = false;
clearInterval(timerInterval);
timeLeft = TOTAL_TIME;
updateTimer();
document.getElementById('questionText').textContent = q.q;
document.getElementById('nextBtn').classList.remove('show');
const container = document.getElementById('optionsContainer');
container.innerHTML = q.o.map((opt, i) => `<div class="option" onclick="selectAnswer(${i})">${opt}</div>`).join('');
timerInterval = setInterval(tick, 1000);
}

function tick() {
timeLeft--;
updateTimer();
if (timeLeft <= 0) {
clearInterval(timerInterval);
timeoutAnswer();
}
}

function updateTimer() {
const pct = (timeLeft / TOTAL_TIME) * 100;
document.getElementById('timerText').textContent = timeLeft;
document.getElementById('timerRing').style.setProperty('--pct', `${pct}%`);
document.getElementById('timerFill').style.width = `${pct}%`;
const bar = document.getElementById('timerBar');
bar.classList.toggle('warning', timeLeft <= 5 && timeLeft > 2);
bar.classList.toggle('danger', timeLeft <= 2);
document.getElementById('timerText').style.color = timeLeft <= 2 ? '#e94560' : timeLeft <= 5 ? '#f39c12' : '#fff';
}

function selectAnswer(idx) {
if (answered) return;
answered = true;
clearInterval(timerInterval);
const q = questions[current];
const options = document.querySelectorAll('.option');
options.forEach((el, i) => {
if (i === q.c) el.classList.add('correct');
if (i === idx && idx !== q.c) el.classList.add('wrong');
});
if (idx === q.c) score++;
document.getElementById('nextBtn').classList.add('show');
}

function timeoutAnswer() {
if (answered) return;
answered = true;
const q = questions[current];
document.querySelectorAll('.option').forEach((el, i) => {
if (i === q.c) el.classList.add('correct');
else el.classList.add('timeout');
});
document.getElementById('nextBtn').classList.add('show');
}

function nextQuestion() {
current++;
if (current < questions.length) render();
else document.querySelector('.container').innerHTML = `<h1 style="text-align:center;color:#e94560">🎉 Selesai!</h1><p style="text-align:center;font-size:24px;color:#fff;margin-top:20px">Skor: ${score}/${questions.length}</p>`;
}
render();
</script>
</body>
</html>
```

## Cara Integrasi
Simpan sebagai HTML. Jawab sebelum timer habis. Timer ring dan bar menunjukkan sisa waktu. Warna berubah saat waktu menipis.
