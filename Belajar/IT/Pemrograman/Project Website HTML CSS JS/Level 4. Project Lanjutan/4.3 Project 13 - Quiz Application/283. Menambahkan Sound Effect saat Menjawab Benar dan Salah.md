# 283. Menambahkan Sound Effect saat Menjawab Benar dan Salah

## Definisi & Target
Menambahkan efek suara (sound effect) saat user menjawab pertanyaan: suara berbeda untuk jawaban benar dan salah.

## Analogi Alur
Sound effect seperti bel di sekolah: bel berbeda untuk masuk kelas (benar) dan pulang sekolah (salah). Suara memberikan umpan balik auditori yang memperkuat feedback visual.

## Poin-Poin Penting
- Gunakan Audio API JavaScript dengan file suara pendek
- Buat suara sintetis menggunakan Web Audio API (tanpa file)
- Atau gunakan base64 encoded audio untuk kemandirian
- Suara benar: nada tinggi ceria (ding)
- Suara salah: nada rendah (buzz)
- Suara timeout: suara peringatan
- Jangan memutar suara jika user tidak berinteraksi (autoplay policy)

## Kode Implementasi (Gabungan) - HTML, CSS, JavaScript
```html
<!DOCTYPE html>
<html lang="id">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Quiz Sound Effects</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; background: #1a1a2e; min-height: 100vh; display: flex; justify-content: center; align-items: center; padding: 20px; }
.container { background: #16213e; border-radius: 20px; padding: 30px; width: 100%; max-width: 500px; }
h1 { text-align: center; color: #e94560; margin-bottom: 20px; font-size: 22px; }
.question-text { color: #eee; font-size: 18px; margin-bottom: 20px; }
.options { display: grid; gap: 10px; }
.option { padding: 14px 18px; background: #0f3460; border: 2px solid #1a3a6e; border-radius: 12px; cursor: pointer; color: #ccc; font-size: 15px; transition: all 0.3s; }
.option:hover { border-color: #667eea; }
.option.correct { background: rgba(76,175,80,0.2); border-color: #4caf50; color: #4caf50; }
.option.wrong { background: rgba(233,69,96,0.2); border-color: #e94560; color: #e94560; }
.next-btn { width: 100%; padding: 14px; margin-top: 15px; background: #e94560; color: white; border: none; border-radius: 12px; font-size: 16px; cursor: pointer; display: none; font-weight: 600; }
.next-btn.show { display: block; }
.sound-toggle { position: fixed; top: 20px; right: 20px; background: #0f3460; border: 1px solid #1a3a6e; border-radius: 50%; width: 44px; height: 44px; display: flex; align-items: center; justify-content: center; cursor: pointer; font-size: 20px; color: #fff; }
.score-text { text-align: center; color: #888; font-size: 14px; margin-bottom: 15px; }
</style>
</head>
<body>
<div class="sound-toggle" id="soundToggle" onclick="toggleSound()">🔊</div>
<div class="container">
<h1>🔊 Quiz with Sound</h1>
<div class="score-text" id="scoreText">Skor: 0</div>
<div class="question-text" id="questionText">Apa ibu kota Indonesia?</div>
<div class="options" id="optionsContainer">
<div class="option" onclick="selectAnswer(0)">Jakarta</div>
<div class="option" onclick="selectAnswer(1)">Bandung</div>
<div class="option" onclick="selectAnswer(2)">Surabaya</div>
<div class="option" onclick="selectAnswer(3)">Yogyakarta</div>
</div>
<button class="next-btn" id="nextBtn" onclick="nextQuestion()">Selanjutnya →</button>
</div>
<script>
let soundEnabled = true;
const AudioCtx = window.AudioContext || window.webkitAudioContext;
let audioCtx = null;

function getAudioCtx() { if (!audioCtx) audioCtx = new AudioCtx(); return audioCtx; }

function playCorrectSound() {
if (!soundEnabled) return;
try {
const ctx = getAudioCtx();
const osc = ctx.createOscillator();
const gain = ctx.createGain();
osc.connect(gain);
gain.connect(ctx.destination);
osc.frequency.setValueAtTime(523.25, ctx.currentTime);
osc.frequency.setValueAtTime(659.25, ctx.currentTime + 0.1);
osc.frequency.setValueAtTime(783.99, ctx.currentTime + 0.2);
gain.gain.setValueAtTime(0.3, ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.4);
osc.start(ctx.currentTime);
osc.stop(ctx.currentTime + 0.4);
} catch {}
}

function playWrongSound() {
if (!soundEnabled) return;
try {
const ctx = getAudioCtx();
const osc = ctx.createOscillator();
const gain = ctx.createGain();
osc.connect(gain);
gain.connect(ctx.destination);
osc.type = 'sawtooth';
osc.frequency.setValueAtTime(200, ctx.currentTime);
osc.frequency.setValueAtTime(150, ctx.currentTime + 0.2);
gain.gain.setValueAtTime(0.3, ctx.currentTime);
gain.gain.exponentialRampToValueAtTime(0.01, ctx.currentTime + 0.3);
osc.start(ctx.currentTime);
osc.stop(ctx.currentTime + 0.3);
} catch {}
}

function toggleSound() {
soundEnabled = !soundEnabled;
document.getElementById('soundToggle').textContent = soundEnabled ? '🔊' : '🔇';
}

const questions = [
{ q: 'Apa ibu kota Indonesia?', o: ['Jakarta', 'Bandung', 'Surabaya', 'Yogyakarta'], c: 0 },
{ q: 'Siapa presiden pertama RI?', o: ['Soeharto', 'Soekarno', 'Habibie', 'Megawati'], c: 1 },
{ q: 'CSS singkatan dari?', o: ['Cascading Style Sheets', 'Creative System', 'Computer Sheets', 'Colorful Style'], c: 0 }
];
let current = 0, score = 0, answered = false;

function render() {
const q = questions[current];
answered = false;
document.getElementById('scoreText').textContent = `Skor: ${score}`;
document.getElementById('questionText').textContent = q.q;
document.getElementById('nextBtn').classList.remove('show');
document.getElementById('optionsContainer').innerHTML = q.o.map((opt, i) => `<div class="option" onclick="selectAnswer(${i})">${opt}</div>`).join('');
}

function selectAnswer(idx) {
if (answered) return;
answered = true;
const q = questions[current];
document.querySelectorAll('.option').forEach((el, i) => {
if (i === q.c) { el.classList.add('correct'); }
if (i === idx && idx !== q.c) { el.classList.add('wrong'); }
});
if (idx === q.c) { score++; playCorrectSound(); }
else { playWrongSound(); }
document.getElementById('scoreText').textContent = `Skor: ${score}`;
document.getElementById('nextBtn').classList.add('show');
}

function nextQuestion() {
current++;
if (current < questions.length) render();
else { document.querySelector('.container').innerHTML = `<h1 style="text-align:center;color:#e94560">🎉 Selesai!</h1><p style="text-align:center;color:#fff;font-size:24px;margin:20px">Skor: ${score}/${questions.length}</p>`; }
}

render();
</script>
</body>
</html>
```

## Cara Integrasi
Simpan sebagai HTML. Jawab pertanyaan - suara "ding" untuk benar dan "buzz" untuk salah. Gunakan tombol speaker untuk toggle suara on/off.
