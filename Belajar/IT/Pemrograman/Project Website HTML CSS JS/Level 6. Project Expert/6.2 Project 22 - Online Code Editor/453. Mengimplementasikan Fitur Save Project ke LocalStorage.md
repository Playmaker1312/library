# 453. Mengimplementasikan Fitur Save Project ke LocalStorage

## Definisi & Target
Mengimplementasikan fitur save project ke LocalStorage. User bisa menyimpan project code editor dengan nama tertentu dan memuatnya kembali nanti. Mendukung multiple saved projects.

## Analogi Alur
Seperti Save As di text editor: Anda memberi nama file, menyimpannya, dan bisa membukanya lagi nanti. Semua project tersimpan di LocalStorage browser.

## Poin-Poin Penting
- Modal form untuk memberi nama project
- Tombol Save menyimpan HTML, CSS, JS ke LocalStorage
- Sidebar atau dropdown untuk memuat project tersimpan
- Auto-save dengan interval tertentu
- Hapus project yang tidak diperlukan

<script>
const htmlEditor = CodeMirror.fromTextArea(document.getElementById('htmlEditor'), { mode: 'xml', theme: 'dracula', lineNumbers: true });
const cssEditor = CodeMirror.fromTextArea(document.getElementById('cssEditor'), { mode: 'css', theme: 'dracula', lineNumbers: true });
const jsEditor = CodeMirror.fromTextArea(document.getElementById('jsEditor'), { mode: 'javascript', theme: 'dracula', lineNumbers: true });
const preview = document.getElementById('preview');
const projectNameEl = document.getElementById('projectName');

let currentProject = 'Untitled';

function updatePreview() {
  const doc = `<!DOCTYPE html><html><head><style>${cssEditor.getValue()}</style></head><body>${htmlEditor.getValue()}<script>${jsEditor.getValue()}<\/script></body></html>`;
  preview.srcdoc = doc;
}

[htmlEditor, cssEditor, jsEditor].forEach(e => e.on('change', updatePreview));

function getProjects() {
  return JSON.parse(localStorage.getItem('codeEditorProjects')) || {};
}

function saveProjects(data) {
  localStorage.setItem('codeEditorProjects', JSON.stringify(data));
}

function saveProject() {
  const name = document.getElementById('saveName').value.trim();
  if (!name) return;
  const projects = getProjects();
  projects[name] = {
    html: htmlEditor.getValue(),
    css: cssEditor.getValue(),
    js: jsEditor.getValue(),
    savedAt: new Date().toISOString()
  };
  saveProjects(projects);
  currentProject = name;
  projectNameEl.textContent = name;
  closeModals();
}

function showSaveModal() {
  document.getElementById('saveName').value = currentProject;
  document.getElementById('saveModal').classList.add('show');
}

function showLoadModal() {
  const list = document.getElementById('projectList');
  const projects = getProjects();
  const entries = Object.entries(projects);
  if (entries.length === 0) {
    list.innerHTML = '<div style="color:#888;padding:10px">No saved projects</div>';
  } else {
    list.innerHTML = entries.map(([name, data]) =>
      `<div class="project-item" onclick="loadProject('${name}')">
        <span class="name">${name}</span>
        <span class="date">${new Date(data.savedAt).toLocaleDateString()}</span>
      </div>`
    ).join('');
  }
  document.getElementById('loadModal').classList.add('show');
}

function loadProject(name) {
  const projects = getProjects();
  const data = projects[name];
  if (!data) return;
  htmlEditor.setValue(data.html);
  cssEditor.setValue(data.css);
  jsEditor.setValue(data.js);
  currentProject = name;
  projectNameEl.textContent = name;
  updatePreview();
  closeModals();
}

function closeModals() {
  document.querySelectorAll('.modal').forEach(m => m.classList.remove('show'));
}

updatePreview();
</script>

## Cara Integrasi
Simpan file HTML, buka di browser. Klik Save untuk menyimpan project dengan nama tertentu. Klik Load untuk membuka daftar project tersimpan. Project disimpan di LocalStorage dengan key "codeEditorProjects".
