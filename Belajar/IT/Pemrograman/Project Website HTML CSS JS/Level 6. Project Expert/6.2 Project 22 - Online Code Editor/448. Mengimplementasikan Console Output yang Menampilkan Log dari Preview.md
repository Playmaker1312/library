# 448. Mengimplementasikan Console Output yang Menampilkan Log dari Preview

## Definisi & Target
Membangun console output yang menangkap log dari kode JavaScript yang dijalankan di preview iframe. Console menampilkan console.log, console.warn, console.error, dan error lainnya.

## Analogi Alur
Seperti console di browser DevTools: setiap pesan log, warning, atau error dari kode yang dijalankan akan muncul di panel console di bawah editor.

## Poin-Poin Penting
- Override console.log/warn/error di dalam iframe
- Tangkap pesan dan kirim ke parent window via postMessage
- Tampilkan pesan dengan styling berbeda (log=putih, warn=kuning, error=merah)
- Clear console setiap kali preview diupdate
- Auto-scroll ke pesan terbaru

```html
<html lang="id">
<head>

<title>Console Output</title>

  grid-template-columns: 1fr 1fr;

</style>
</head>
<body>
<div class="editor-section">
  <div class="editor-area">
    <div class="label">JavaScript Code</div>
    <textarea id="jsCode">console.log('Hello World!');
console.log('User clicked button');
console.warn('This is a warning message');
console.error('This is an error message');

// Test objects
console.log({name: 'John', age: 30});

// Test arrays
console.log([1, 2, 3, 4, 5]);

// Throw error
try {
  throw new Error('Something went wrong');
} catch(e) {
  console.error(e.message);
}
</textarea>
  </div>
  <div class="preview-section">
    <div class="label">Preview</div>
    <iframe class="preview-frame" id="preview"></iframe>
  </div>
</div>
<div class="console-panel">
  <div class="header">
    <span>⚡ Console Output</span>
    <button class="clear-btn" onclick="clearConsole()">Clear</button>
  </div>

    <div class="empty">Console output will appear here...</div>
  </div>
</div>
<script>
const jsCode = document.getElementById('jsCode');
const preview = document.getElementById('preview');
const consoleOutput = document.getElementById('consoleOutput');

function injectConsoleCapture(iframeWin) {
  const captureCode = `
    (function() {
      var logs = [];
      function send(type, args) {
        var msg = Array.from(args).map(function(a) {
          try { return typeof a === 'object' ? JSON.stringify(a) : String(a); }
          catch(e) { return String(a); }
        }).join(' ');
        window.parent.postMessage({type: 'console', method: type, message: msg}, '*');
      }
      console.log = function() { send('log', arguments); };
      console.warn = function() { send('warn', arguments); };
      console.error = function() { send('error', arguments); };
      window.onerror = function(msg, url, line) {
        window.parent.postMessage({type: 'console', method: 'error', message: msg + ' (line ' + line + ')'}, '*');
      };
    })();
  `;
  const script = iframeWin.document.createElement('script');
  script.textContent = captureCode;
  iframeWin.document.head.appendChild(script);
}

function updatePreview() {
  const js = jsCode.value;
  const doc = `<!DOCTYPE html><html><head><meta charset="UTF-8"></head><body>
<script>${js}<\/script>
</body></html>`;
  preview.srcdoc = doc;
  setTimeout(() => {
    try {
      injectConsoleCapture(preview.contentWindow);
    } catch(e) {}
  }, 100);
}

window.addEventListener('message', function(e) {
  if (e.data && e.data.type === 'console') {
    addConsoleEntry(e.data.method, e.data.message);
  }
});

function addConsoleEntry(method, message) {
  const empty = consoleOutput.querySelector('.empty');
  if (empty) empty.remove();
  const entry = document.createElement('div');
  entry.className = 'entry ' + method;
  entry.innerHTML = '<span class="prefix">[' + method.toUpperCase() + ']</span>' + escapeHtml(message);
  consoleOutput.appendChild(entry);
  consoleOutput.scrollTop = consoleOutput.scrollHeight;
}

function escapeHtml(text) {
  var d = document.createElement('div');
  d.textContent = text;
  return d.innerHTML;
}

function clearConsole() {
  consoleOutput.innerHTML = '<div class="empty">Console output will appear here...</div>';
}

jsCode.addEventListener('input', updatePreview);
updatePreview();
</script>
</body>
</html>
```

## Cara Integrasi
Simpan file HTML, buka di browser. Kode JavaScript di editor akan dijalankan di iframe. Console.log/warn/error ditangkap dan ditampilkan di panel console bawah. Warna berbeda untuk setiap level log. Klik Clear untuk membersihkan console.
