# 449. Mengimplementasikan Fitur Auto-Complete Dasar

## Definisi & Target
Mengimplementasikan fitur auto-complete dasar untuk kode HTML, CSS, dan JavaScript menggunakan CodeMirror addon show-hint. Fitur ini menampilkan saran saat mengetik kata kunci atau tag.

## Analogi Alur
Seperti predictive text di smartphone: saat Anda mulai mengetik, muncul saran kata yang relevan. Anda bisa memilih saran untuk mempercepat penulisan kode.

## Poin-Poin Penting
- Integrasi CodeMirror show-hint addon
- HTML hint untuk tag dan attribute
- CSS hint untuk properti dan value
- JavaScript hint untuk keyword dan function
- Trigger hint otomatis saat mengetik karakter tertentu

## Kode Implementasi (Gabungan) - HTML, CSS, JavaScript

```html
<!DOCTYPE html>
<html lang="id">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Auto Complete Editor</title>
<link rel="stylesheet" href="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/codemirror.min.css">
<link rel="stylesheet" href="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/theme/dracula.min.css">
<link rel="stylesheet" href="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/addon/hint/show-hint.min.css">
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; height: 100vh; overflow: hidden; background: #1a1a2e; }

.top-panel { display: grid; grid-template-columns: 1fr 1fr 1fr; gap: 4px; padding: 4px; background: #0f3460; }
.editor-panel { display: flex; flex-direction: column; background: #16213e; border-radius: 6px; overflow: hidden; }
.editor-panel .header { background: #0f3460; padding: 8px 12px; font-size: 12px; color: #e94560; }
.editor-panel .CodeMirror { flex: 1; height: auto !important; font-size: 13px; }
.bottom-panel { background: #0f3460; padding: 4px; }
.preview-frame { width: 100%; height: 100%; background: white; border-radius: 6px; border: none; }
</style>
</head>
<body>
<div class="app">
  <div class="top-panel">
    <div class="editor-panel">
      <div class="header">HTML <span style="color:#888;font-size:10px">(type < to trigger hints)</span></div>
      <textarea id="htmlEditor"><div>
  <h1>Auto Complete Demo</h1>
  <p>Type a tag like <a to see hints</p>
</div></textarea>
    </div>
    <div class="editor-panel">
      <div class="header">CSS <span style="color:#888;font-size:10px">(type property to trigger hints)</span></div>
      <textarea id="cssEditor">body {
  font-family: Arial;
  margin: 0;
  padding: 20px;
}
h1 {
  color: #333;
}</textarea>
    </div>
    <div class="editor-panel">
      <div class="header">JavaScript <span style="color:#888;font-size:10px">(type keyword for hints)</span></div>
      <textarea id="jsEditor">function hello() {
  console.log('Hello!');
}

document.querySelector('h1').style.color = '#e94560';</textarea>
    </div>
  </div>
  <div class="bottom-panel">
    <iframe class="preview-frame" id="preview"></iframe>
  </div>
</div>
<script src="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/codemirror.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/mode/xml/xml.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/mode/css/css.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/mode/javascript/javascript.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/addon/hint/show-hint.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/addon/hint/xml-hint.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/addon/hint/html-hint.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/addon/hint/css-hint.min.js"></script>
<script src="https://cdnjs.cloudflare.com/ajax/libs/codemirror/5.65.16/addon/hint/javascript-hint.min.js"></script>
<script>
const htmlEditor = CodeMirror.fromTextArea(document.getElementById('htmlEditor'), {
  mode: 'xml',
  theme: 'dracula',
  lineNumbers: true,
  autoCloseTags: true,
  extraKeys: {'Ctrl-Space': 'autocomplete'}
});
htmlEditor.on('inputRead', function(cm, change) {
  if (change.text.length === 1) {
    var ch = change.text[0];
    if (ch === '<' || ch === ' ' || ch === '=' || ch === '/') {
      CodeMirror.commands.autocomplete(cm, null, {completeSingle: false});
    }
  }
});

const cssEditor = CodeMirror.fromTextArea(document.getElementById('cssEditor'), {
  mode: 'css',
  theme: 'dracula',
  lineNumbers: true,
  extraKeys: {'Ctrl-Space': 'autocomplete'}
});
cssEditor.on('inputRead', function(cm, change) {
  if (change.text.length === 1) {
    var ch = change.text[0];
    if (ch === ':' || ch === ';' || ch === ' ' || ch === '-') {
      CodeMirror.commands.autocomplete(cm, null, {completeSingle: false});
    }
  }
});

const jsEditor = CodeMirror.fromTextArea(document.getElementById('jsEditor'), {
  mode: 'javascript',
  theme: 'dracula',
  lineNumbers: true,
  extraKeys: {'Ctrl-Space': 'autocomplete'}
});
jsEditor.on('inputRead', function(cm, change) {
  if (change.text.length === 1 && /[a-zA-Z0-9._$]/.test(change.text[0])) {
    CodeMirror.commands.autocomplete(cm, null, {completeSingle: false});
  }
});

const editors = [htmlEditor, cssEditor, jsEditor];
const preview = document.getElementById('preview');

function updatePreview() {
  const html = htmlEditor.getValue();
  const css = cssEditor.getValue();
  const js = jsEditor.getValue();
  const doc = `<!DOCTYPE html><html><head><style>${css}</style></head><body>${html}<script>${js}<\/script></body></html>`;
  preview.srcdoc = doc;
}

editors.forEach(e => e.on('change', updatePreview));
updatePreview();
</script>
</body>
</html>
```

## Cara Integrasi
Simpan file HTML, buka di browser. Mulai mengetik tag HTML (misal <di), properti CSS (misal colo), atau JS keyword (misal doc) untuk melihat saran auto-complete. Tekan Enter untuk memilih saran.
