# 459. Mengimplementasikan Filter Berdasarkan Kategori, Harga, Rating, dan Ketersediaan

## Definisi & Target
Mengimplementasikan sistem filter lengkap untuk produk berdasarkan kategori, rentang harga, rating minimal, dan ketersediaan stok. Filter bekerja secara real-time tanpa reload halaman.

## Analogi Alur
Seperti sommelier yang membantu memilih wine: Anda menyebutkan kriteria (jenis, harga, kualitas), dan sommelier akan menyaring semua pilihan yang sesuai.

## Poin-Poin Penting
- Filter kategori menggunakan multiple checkbox
- Filter harga menggunakan range slider dual-handle
- Filter rating menggunakan bintang interaktif
- Filter ketersediaan stok (in stock only)
- Kombinasi semua filter bekerja bersamaan (AND logic)

<script>

  { id: 6, name: 'Running Shoes', price: 249, rating: 4.6, category: 'fashion', emoji: '👟', inStock: true },
  { id: 7, name: 'Bluetooth Speaker', price: 199, rating: 4.3, category: 'electronics', emoji: '🔊', inStock: true },
  { id: 8, name: 'Throw Pillow', price: 49, rating: 3.8, category: 'home', emoji: '🛋️', inStock: true },
  { id: 9, name: 'Sunglasses', price: 129, rating: 4.1, category: 'fashion', emoji: '🕶️', inStock: true },

let minRating = 0;
let selectedCategory = null;

function filter() {
  const cats = Array.from(document.querySelectorAll('.f-cat:checked')).map(cb => cb.value);
  const minP = parseInt(document.getElementById('priceMin').value);
  const maxP = parseInt(document.getElementById('priceMax').value);
  const stockOnly = document.getElementById('fStock').checked;
  return products.filter(p => {
    if (!cats.includes(p.category)) return false;
    if (p.price < minP || p.price > maxP) return false;
    if (p.rating < minRating) return false;
    if (stockOnly && !p.inStock) return false;
    return true;
  });
}

function render() {
  const result = filter();
  document.getElementById('resultCount').textContent = `Showing ${result.length} of ${products.length} products`;
  const grid = document.getElementById('productGrid');
  grid.innerHTML = result.map(p => `
    <div class="product-card">
      <div class="emoji">${p.emoji}</div>
      <h4>${p.name}</h4>
      <div class="rating">${'★'.repeat(Math.floor(p.rating))} ${p.rating}</div>
      <div class="price">Rp${p.price}k</div>
      <div class="stock ${p.inStock ? 'in' : 'out'}">${p.inStock ? '✓ In Stock' : '✗ Out of Stock'}</div>
    </div>
  `).join('');

    ${cats.length < 3 ? '<span class="filter-tag">' + cats.join(', ') + '</span>' : ''}
    ${minRating > 0 ? '<span class="filter-tag">★ ' + minRating + '+</span>' : ''}
  `.trim() || '<span style="color:#ccc">No active filters</span>';
}

let cats = [];

function setRating(r) {
  minRating = r;
  document.querySelectorAll('.star-filter label').forEach(l => l.classList.remove('active'));
  document.querySelectorAll('.star-filter label')[r === 0 ? 0 : r === 4 ? 1 : 2].classList.add('active');
  render();
}

document.querySelectorAll('.f-cat, #fStock').forEach(el => el.addEventListener('change', render));

});
document.getElementById('priceMax').addEventListener('input', function() {
