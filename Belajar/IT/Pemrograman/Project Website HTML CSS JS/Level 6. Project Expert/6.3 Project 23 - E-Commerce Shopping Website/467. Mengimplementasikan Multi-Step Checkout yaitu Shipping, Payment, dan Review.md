# 467. Mengimplementasikan Multi-Step Checkout yaitu Shipping, Payment, dan Review

## Definisi & Target
Mengimplementasikan multi-step checkout dengan tiga langkah: Shipping (alamat pengiriman), Payment (metode pembayaran), dan Review (konfirmasi pesanan). Setiap langkah memiliki form sendiri dengan navigasi next/back.

## Analogi Alur
Seperti proses pendaftaran bertahap: pertama isi data diri, lalu pilih pembayaran, terakhir review sebelum submit. Setiap langkah harus diisi sebelum lanjut ke langkah berikutnya.

## Poin-Poin Penting
- Tiga step: Shipping → Payment → Review
- Progress indicator (step bar) di bagian atas
- Navigation buttons: Back dan Next
- Validasi per step sebelum lanjut
- Review step menampilkan ringkasan semua data

<script>
let currentStep = 1;
const totalSteps = 3;

function updateUI() {
  for (let i = 1; i <= totalSteps; i++) {
    document.getElementById('step' + i).className = 'step' + (i === currentStep ? ' active' : (i < currentStep ? ' completed' : ''));
    document.getElementById('div' + (i < totalSteps ? i : ''))?.classList.toggle('completed', i < currentStep);
  }
  document.querySelectorAll('.step-content').forEach((el, i) => el.classList.toggle('active', i + 1 === currentStep));
  document.getElementById('backBtn').style.visibility = currentStep === 1 ? 'hidden' : 'visible';
  const nextBtn = document.getElementById('nextBtn');
  if (currentStep === totalSteps) {
    nextBtn.textContent = 'Place Order ✅';
    nextBtn.className = 'btn-next btn-place';
  } else {
    nextBtn.textContent = 'Next →';
    nextBtn.className = 'btn-next';
  }
}

function nextStep() {
  if (currentStep === 1) {
    const name = document.getElementById('sName').value.trim();
    const phone = document.getElementById('sPhone').value.trim();
    const address = document.getElementById('sAddress').value.trim();
    if (!name || !phone || !address) { alert('Please fill required fields'); return; }
    document.getElementById('reviewName').textContent = 'Name: ' + name;
    document.getElementById('reviewPhone').textContent = 'Phone: ' + phone;
    document.getElementById('reviewAddress').textContent = 'Address: ' + address + ', ' + document.getElementById('sCity').value;
  }
  if (currentStep === 2) {
    const payment = document.querySelector('input[name="payment"]:checked');
    document.getElementById('reviewPayment').textContent = 'Method: ' + payment.nextElementSibling.textContent.trim();
  }
  if (currentStep < totalSteps) {
    currentStep++;
    updateUI();
  } else {
    alert('Order placed successfully! 🎉');
  }
}

function prevStep() {
  if (currentStep > 1) { currentStep--; updateUI(); }
}

document.querySelectorAll('.payment-option').forEach(el => el.addEventListener('click', function() {
  document.querySelectorAll('.payment-option').forEach(o => o.classList.remove('selected'));
  this.classList.add('selected');
  this.querySelector('input').checked = true;
}));

updateUI();
</script>
