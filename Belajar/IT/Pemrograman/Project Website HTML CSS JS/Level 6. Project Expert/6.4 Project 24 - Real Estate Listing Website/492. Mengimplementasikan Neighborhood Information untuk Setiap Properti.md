# 492. Mengimplementasikan Neighborhood Information untuk Setiap Properti

## Definisi & Target
Membangun panel informasi lingkungan (neighborhood) sekitar properti, termasuk fasilitas umum seperti sekolah, rumah sakit, pusat perbelanjaan, transportasi, dan taman.

## Analogi Alur
Seperti brosur properti yang mencantumkan "Fasilitas Sekitar": dalam jarak 1 km ada sekolah, 500 m ada rumah sakit, 2 km ada mal — membantu pembeli menilai kelayakan lokasi.

## Poin-Poin Penting
- Data neighborhood dalam format JSON per properti
- Ikon dan label untuk tiap kategori fasilitas
- Jarak dalam meter/kilometer
- Tampilan card grid yang rapi
- Filter fasilitas berdasarkan kategori

const properties = [
  { id: 1, title: 'Rumah Modern 2 Lantai - Menteng', location: 'Jakarta Pusat' },

function init() {
  let sel = document.getElementById('propertySelect');
  sel.innerHTML = properties.map(p => `<option value="${p.id}">${p.title} - ${p.location}</option>`).join('');
  sel.addEventListener('change', () => render(parseInt(sel.value)));
  render(properties[0].id);
}
function render(id) {
  let hood = neighborhoods[id];
  if (!hood) return;
  let cats = [...new Set(hood.facilities.map(f => f.category))];
  document.getElementById('chipContainer').innerHTML = `<button class="chip active" data-cat="all" onclick="filterFacilities('all',this)">Semua (${hood.facilities.length})</button>` +
    cats.map(c => {
      let count = hood.facilities.filter(f => f.category === c).length;
      return `<button class="chip" data-cat="${c}" onclick="filterFacilities('${c}',this)">${c} (${count})</button>`;
    }).join('');
  showFacilities(hood.facilities, 'all');
  document.getElementById('summaryArea').innerHTML = cats.map(c => {
    let count = hood.facilities.filter(f => f.category === c).length;
    return `<div class="summary-item"><div class="count">${count}</div><div class="label">${c}</div></div>`;
  }).join('');
  if (hood.walkScore) {
    let ws = document.getElementById('walkScore');
    ws.style.display = 'flex';
    document.getElementById('wsScore').textContent = hood.walkScore.score;
    document.getElementById('wsLabel').textContent = hood.walkScore.label;
    document.getElementById('wsDesc').textContent = hood.walkScore.desc;
  }
}
function showFacilities(facilities, cat) {
  let filtered = cat === 'all' ? facilities : facilities.filter(f => f.category === cat);
  let grid = document.getElementById('neighborGrid');
  if (filtered.length === 0) {
    grid.innerHTML = '<p style="color:#888;text-align:center;padding:40px">Tidak ada fasilitas untuk kategori ini.</p>';
    return;
  }
  grid.innerHTML = filtered.map(f => {
    let d = f.distance >= 1000 ? (f.distance/1000).toFixed(1) + ' km' : f.distance + ' m';
    return `<div class="neighbor-card">
      <div class="icon">${f.icon}</div>
      <div class="name">${f.name}</div>
      <div class="category">${f.category}</div>
      <div class="distance">Jarak: <strong>${d}</strong></div>
    </div>`;
  }).join('');
}
function filterFacilities(cat, btn) {
  document.querySelectorAll('.chip').forEach(c => c.classList.remove('active'));
  btn.classList.add('active');
  let hood = neighborhoods[parseInt(document.getElementById('propertySelect').value)];
  showFacilities(hood.facilities, cat);
}
