# 495. Responsive & Performance Polish untuk Seluruh Website

## Definisi & Target
Melakukan penyempurnaan akhir (polish) untuk seluruh website real estate, mencakup responsivitas di semua perangkat, optimasi performa rendering, animasi halus, lazy loading, dan a11y (accessibility) dasar.

## Analogi Alur
Seperti rumah yang sudah dibangun kemudian dibersihkan dan dirapikan: mengecat ulang, merapikan kabel, memasang lampu yang tepat — membuat pengunjung merasa nyaman.

## Poin-Poin Penting
- Layout responsif (mobile, tablet, desktop)
- Lazy loading untuk gambar properti
- Animasi transisi halus (CSS transitions)
- Navigasi mobile-friendly dengan hamburger menu
- Optimasi performa dengan debounce dan throttling
- Accessibility (focus, aria-label, alt text)

let displayCount = 6;
function debounce(fn, ms) {
  let timer;
  return function(...args) {
    clearTimeout(timer);
    timer = setTimeout(() => fn.apply(this, args), ms);
  };
}
function createCard(p) {
  let fmtPrice = p.price >= 1000000000 ? 'Rp' + (p.price / 1000000000).toFixed(1) + ' M' : 'Rp' + (p.price / 1000000).toFixed(0) + ' Jt';
  let badgeHtml = p.badge ? `<span class="card-badge">${p.badge}</span>` : '';
  let bedHtml = p.bed > 0 ? `<span>${p.bed} KT</span>` : '';
  let bathHtml = p.bath > 0 ? `<span>${p.bath} KM</span>` : '';
  return `<div class="property-card" role="article" aria-label="${p.title}">
    <div class="card-img">
      ${badgeHtml}
      <span style="font-size:64px">${p.img}</span>
    </div>
    <div class="card-body">
      <h3>${p.title}</h3>
      <div class="price">${fmtPrice}</div>
      <div class="location">${p.location}</div>
      <div class="card-details">
        <span>${p.type}</span>
        ${bedHtml}
        ${bathHtml}
      </div>
    </div>
  </div>`;
}
function renderProperties(props) {

function applyFilters() {
  let type = document.getElementById('filterType').value;
  let bed = document.getElementById('filterBed').value;
  let loc = document.getElementById('filterLoc').value.toLowerCase();
  let sort = document.getElementById('filterSort').value;
  let filtered = allProperties.filter(p => {
    if (type && p.type !== type) return false;
    if (bed) {
      if (bed === '5+') { if (p.bed < 5) return false; }
      else if (p.bed !== parseInt(bed)) return false;
    }
    if (loc && !p.location.toLowerCase().includes(loc)) return false;
    return true;
  });
  if (sort === 'price-asc') filtered.sort((a, b) => a.price - b.price);
  else if (sort === 'price-desc') filtered.sort((a, b) => b.price - a.price);
  displayCount = 6;
  renderProperties(filtered);
}
function loadMore() {
  displayCount += 6;
  applyFilters();
