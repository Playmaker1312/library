# 491. Membangun Fitur Saved Searches & Favorite Listings

## Definisi & Target
Membangun fitur untuk menyimpan pencarian favorit dan properti yang disukai (favorites) berbasis LocalStorage. Pengguna dapat menyimpan kriteria pencarian dan menandai properti favorit untuk akses cepat.

## Analogi Alur
Seperti agen properti yang mencatat preferensi klien di buku catatan: Anda menyebutkan kriteria, agen mencatat, dan saat properti cocok muncul, agen segera memberitahu Anda.

## Poin-Poin Penting
- Tombol favorite pada setiap card properti dengan ikon toggle
- Halaman khusus untuk menampilkan properti favorit
- Simpan kriteria pencarian (filter) sebagai saved search
- Hapus dan kelola saved searches
- Data persist dengan LocalStorage

<script>
let properties = [
  { id: 1, title: 'Rumah Modern 2 Lantai', price: 2500000000, location: 'Jakarta Selatan', type: 'Rumah', bedrooms: 4, bathrooms: 3, img: '🏡' },

  { id: 4, title: 'Ruko 3 Lantai Pusat Kota', price: 5000000000, location: 'Bandung', type: 'Ruko', bedrooms: 2, bathrooms: 2, img: '🏪' },
  { id: 5, title: 'Villa Pegunungan', price: 1800000000, location: 'Puncak', type: 'Rumah', bedrooms: 5, bathrooms: 3, img: '🏔️' },
  { id: 6, title: 'Apartemen 2BR Modern', price: 1200000000, location: 'Surabaya', type: 'Apartemen', bedrooms: 2, bathrooms: 1, img: '🏢' },
  { id: 7, title: 'Rumah Minimalis', price: 950000000, location: 'Depok', type: 'Rumah', bedrooms: 3, bathrooms: 2, img: '🏡' },
  { id: 8, title: 'Tanah Kavling 200m', price: 800000000, location: 'Bogor', type: 'Tanah', bedrooms: 0, bathrooms: 0, img: '🌳' }
];
let favorites = JSON.parse(localStorage.getItem('re24_favorites') || '[]');
let savedSearches = JSON.parse(localStorage.getItem('re24_savedSearches') || '[]');

function toggleFavorite(id) {
  let idx = favorites.indexOf(id);
  if (idx > -1) favorites.splice(idx, 1);
  else favorites.push(id);
  localStorage.setItem('re24_favorites', JSON.stringify(favorites));
  renderFavorites();
  renderSearchResults(document.getElementById('searchResults').dataset.query || '');
}

function renderFavorites() {
  let grid = document.getElementById('favGrid');
  let empty = document.getElementById('favEmpty');
  let favItems = properties.filter(p => favorites.includes(p.id));
  if (favItems.length === 0) {
    grid.innerHTML = '';
    empty.style.display = 'block';
  } else {
    empty.style.display = 'none';
    grid.innerHTML = favItems.map(p => createCard(p)).join('');
  }
  document.querySelectorAll('.tab')[0].textContent = `Favorites (${favItems.length})`;
}

function createCard(p) {
  let isFav = favorites.includes(p.id);
  return `<div class="property-card">
    <button class="fav-btn ${isFav ? 'active' : ''}" onclick="toggleFavorite(${p.id})">${isFav ? '&#10084;' : '&#9825;'}</button>
    <div style="height:180px;display:flex;align-items:center;justify-content:center;font-size:64px;background:#eef2f7;color:#aaa">${p.img}</div>
    <div class="property-info">
      <h3>${p.title}</h3>
      <div class="price">Rp${(p.price / 1000000).toFixed(0)} Juta</div>
      <div class="location">${p.location}</div>
      <div class="details">
        <span>${p.type}</span>
        <span>${p.bedrooms} KT</span>
        <span>${p.bathrooms} KM</span>
      </div>
    </div>
  </div>`;
}

function saveCurrentSearch() {
  let criteria = {
    type: document.getElementById('filterType').value,
    bedrooms: document.getElementById('filterBedrooms').value,
    minPrice: document.getElementById('filterMinPrice').value,
    maxPrice: document.getElementById('filterMaxPrice').value,
    location: document.getElementById('filterLocation').value,
    date: new Date().toLocaleDateString()
  };
  savedSearches.unshift(criteria);
  if (savedSearches.length > 20) savedSearches.pop();
  localStorage.setItem('re24_savedSearches', JSON.stringify(savedSearches));
  renderSearches();
}

function deleteSearch(idx) {
  savedSearches.splice(idx, 1);
  localStorage.setItem('re24_savedSearches', JSON.stringify(savedSearches));
  renderSearches();
}

function renderSearches() {
  let list = document.getElementById('searchList');
  let empty = document.getElementById('searchEmpty');
  if (savedSearches.length === 0) {
    list.innerHTML = '';
    empty.style.display = 'block';
  } else {
    empty.style.display = 'none';
    list.innerHTML = savedSearches.map((s, i) => {
      let parts = [];
      if (s.type) parts.push(`<span>Type: <strong>${s.type}</strong></span>`);
      if (s.bedrooms) parts.push(`<span>Kamar: <strong>${s.bedrooms}</strong></span>`);
      if (s.minPrice || s.maxPrice) parts.push(`<span>Harga: <strong>${s.minPrice || '0'} - ${s.maxPrice || '∞'}</strong></span>`);
      if (s.location) parts.push(`<span>Lokasi: <strong>${s.location}</strong></span>`);
      return `<div class="search-item">
        <div class="criteria">${parts.join('')} <span style="color:#aaa;font-size:12px">${s.date || ''}</span></div>
        <div class="search-actions">
          <button class="btn btn-primary" onclick="applySearch(${i})" style="padding:6px 14px;font-size:12px">Apply</button>
          <button class="btn btn-danger" onclick="deleteSearch(${i})" style="padding:6px 14px;font-size:12px">Delete</button>
        </div>
      </div>`;
    }).join('');
  }
  document.querySelectorAll('.tab')[1].textContent = `Saved Searches (${savedSearches.length})`;
}

function applySearch(idx) {
  let s = savedSearches[idx];
  if (s.type) document.getElementById('filterType').value = s.type;
  if (s.bedrooms) document.getElementById('filterBedrooms').value = s.bedrooms;
  if (s.minPrice) document.getElementById('filterMinPrice').value = s.minPrice;
  if (s.maxPrice) document.getElementById('filterMaxPrice').value = s.maxPrice;
  if (s.location) document.getElementById('filterLocation').value = s.location;
  document.querySelectorAll('.tab')[2].click();
  doSearch();
}

function doSearch() {
  let type = document.getElementById('filterType').value;
  let bed = document.getElementById('filterBedrooms').value;
  let minP = document.getElementById('filterMinPrice').value;
  let maxP = document.getElementById('filterMaxPrice').value;
  let loc = document.getElementById('filterLocation').value.toLowerCase();
  let results = properties.filter(p => {
    if (type && p.type !== type) return false;
    if (bed && p.bedrooms !== parseInt(bed)) return false;
    if (minP && p.price < parseInt(minP)) return false;
    if (maxP && p.price > parseInt(maxP)) return false;
    if (loc && !p.location.toLowerCase().includes(loc)) return false;
    return true;
  });
  let container = document.getElementById('searchResults');
  if (results.length === 0) {
    container.innerHTML = '<div class="empty"><div class="icon">&#128270;</div><p>Tidak ada properti yang cocok.</p></div>';
  } else {
    container.innerHTML = results.map(p => createCard(p)).join('');
  }
  container.dataset.query = 'searched';
}
document.querySelectorAll('.tab').forEach(tab => {
  tab.addEventListener('click', function() {
    document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
    document.querySelectorAll('.section').forEach(s => s.classList.remove('active'));
    this.classList.add('active');
    document.getElementById(this.dataset.tab).classList.add('active');
  });
});
renderFavorites();
renderSearches();
</script>

## Cara Integrasi
Simpan file, buka di browser. Gunakan tab "New Search" untuk mencari properti dan klik tombol favorite (hati) pada card. Saved search dapat disimpan dengan tombol "Save Search" setelah melakukan pencarian. Semua data tersimpan di LocalStorage browser.
