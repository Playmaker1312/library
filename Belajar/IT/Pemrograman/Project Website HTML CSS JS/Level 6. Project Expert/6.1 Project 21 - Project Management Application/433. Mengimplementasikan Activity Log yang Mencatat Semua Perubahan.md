# 433. Mengimplementasikan Activity Log yang Mencatat Semua Perubahan

## Definisi & Target
Membangun activity log yang mencatat setiap perubahan yang terjadi dalam aplikasi. Setiap aksi seperti create, update, delete project/task akan tercatat dengan timestamp dan user yang melakukan.

## Analogi Alur
Seperti black box di pesawat: setiap aksi yang terjadi direkam secara kronologis dengan detail siapa, apa, kapan, sehingga bisa di-audit kapan saja.

## Poin-Poin Penting
- Setiap log berisi: id, action, entityType, entityId, details, userId, timestamp
- Log ditampilkan secara kronologis terbalik (terbaru di atas)
- Filter berdasarkan entity type (project, task, team)
- Search log berdasarkan kata kunci
- Auto-prune log jika sudah terlalu banyak

<script>
let logs = JSON.parse(localStorage.getItem('pmLogs')) || [];
let currentFilter = 'all';

function addLog(action, entityType, entityId, details, userId) {
  logs.unshift({
    id: 'log' + Date.now(),
    action,
    entityType,
    entityId,
    details,
    userId: userId || 'system',
    timestamp: new Date().toISOString()
  });
  if (logs.length > 200) logs = logs.slice(0, 200);
  localStorage.setItem('pmLogs', JSON.stringify(logs));
  render();
}

function setFilter(filter) {
  currentFilter = filter;
  document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
  document.querySelector(`[data-filter="${filter}"]`).classList.add('active');
  render();
}

function render() {
  const search = document.getElementById('searchInput').value.toLowerCase();
  const list = document.getElementById('logList');
  let filtered = logs;
  if (currentFilter !== 'all') filtered = filtered.filter(l => l.entityType === currentFilter);
  if (search) filtered = filtered.filter(l =>
    l.action.toLowerCase().includes(search) ||
    l.entityType.toLowerCase().includes(search) ||
    (l.details || '').toLowerCase().includes(search)
  );
  if (filtered.length === 0) {
    list.innerHTML = '<div class="empty">No activity found</div>';
    return;
  }
  list.innerHTML = filtered.map(log => {
    const icons = { create: '➕', update: '✏️', delete: '🗑️' };
    const time = new Date(log.timestamp).toLocaleString();
    return `<div class="log-entry">
      <div class="log-icon ${log.action}">${icons[log.action] || '📌'}</div>
      <div class="log-content">
        <div class="action"><strong>${log.action.charAt(0).toUpperCase() + log.action.slice(1)}</strong> ${log.entityType}: ${log.entityId}</div>
        <div class="details">${log.details || ''}</div>
        <div class="time">${time} · by ${log.userId}</div>
      </div>
    </div>`;
  }).join('');
}

addLog('create', 'project', 'p102', 'Created new project "Mobile App v2"', 'alice');
addLog('update', 'task', 't202', 'Updated priority from "medium" to "high"', 'charlie');

addLog('update', 'project', 'p101', 'Changed deadline from 2025-07-01 to 2025-07-15', 'alice');

</script>

## Cara Integrasi
Simpan file HTML, buka di browser. Log awal sudah disediakan sebagai contoh. Panggil `addLog(action, entityType, entityId, details, userId)` dari modul lain untuk mencatat perubahan. Filter dan search untuk navigasi log.
