# 432. Membangun Team Management dengan Member Roles dan Permissions

## Definisi & Target
Membangun sistem manajemen tim dengan role-based access control. Setiap member memiliki role (Admin, Manager, Developer, Viewer) dengan permission berbeda untuk mengakses fitur aplikasi.

## Analogi Alur
Seperti struktur organisasi perusahaan: CEO (Admin) memiliki akses penuh, Manager bisa mengelola project dan task, Developer hanya mengerjakan task, Viewer hanya melihat.

## Poin-Poin Penting
- Setiap member memiliki id, name, email, role, avatar
- Role: admin, manager, developer, viewer
- Permissions: create, read, update, delete untuk setiap entity
- UI menampilkan daftar member dengan filter role
- Form invite member baru

<script>
const rolePermissions = {
  admin: { project: 'crud', task: 'crud', team: 'crud', settings: 'crud' },
  manager: { project: 'crud', task: 'crud', team: 'r', settings: 'r' },
  developer: { project: 'r', task: 'crud', team: 'r', settings: 'r' },
  viewer: { project: 'r', task: 'r', team: 'r', settings: 'r' }
};

let members = JSON.parse(localStorage.getItem('pmMembers')) || [
  { id: 'm1', name: 'Alice Johnson', email: 'alice@example.com', role: 'admin', avatar: 'AJ' },
  { id: 'm2', name: 'Bob Smith', email: 'bob@example.com', role: 'manager', avatar: 'BS' },
  { id: 'm3', name: 'Charlie Brown', email: 'charlie@example.com', role: 'developer', avatar: 'CB' },
  { id: 'm4', name: 'Diana Prince', email: 'diana@example.com', role: 'viewer', avatar: 'DP' }
];

function save() { localStorage.setItem('pmMembers', JSON.stringify(members)); }

function render(roleFilter) {
  const list = document.getElementById('memberList');
  const filtered = roleFilter && roleFilter !== 'all' ? members.filter(m => m.role === roleFilter) : members;
  document.getElementById('memberCount').textContent = filtered.length + ' members';
  list.innerHTML = filtered.map(m => {
    const perms = rolePermissions[m.role] || {};
    return `<div class="member-card">
      <div class="avatar">${m.avatar || m.name.charAt(0).toUpperCase()}</div>
      <div class="info"><h3>${m.name}</h3><p>${m.email}</p></div>
      <span class="role-badge ${m.role}">${m.role}</span>
      <div class="permissions">
        <span>P: <span class="${perms.project?.includes('c') ? 'yes' : 'no'}">${perms.project?.includes('c') ? '✓' : '✗'}</span></span>
        <span>T: <span class="${perms.task?.includes('c') ? 'yes' : 'no'}">${perms.task?.includes('c') ? '✓' : '✗'}</span></span>
        <span>S: <span class="${perms.settings?.includes('c') ? 'yes' : 'no'}">${perms.settings?.includes('c') ? '✓' : '✗'}</span></span>
      </div>
    </div>`;
  }).join('');
}

function filterMembers(role) {
  document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
  document.querySelector(`[data-filter="${role}"]`).classList.add('active');
  render(role);
}

function showModal() { document.getElementById('modal').classList.add('show'); }
document.getElementById('modal').onclick = e => { if (e.target === e.currentTarget) e.target.classList.remove('show'); };

function addMember() {
  const name = document.getElementById('memberName').value.trim();
  const email = document.getElementById('memberEmail').value.trim();
  const role = document.getElementById('memberRole').value;
  if (!name || !email) return;
  const parts = name.split(' ');
  const avatar = (parts[0].charAt(0) + (parts[1]?.charAt(0) || '')).toUpperCase();
  members.push({ id: 'm' + Date.now(), name, email, role, avatar });
  save();
  render();
  document.getElementById('modal').classList.remove('show');
  document.getElementById('memberName').value = '';
  document.getElementById('memberEmail').value = '';
}

save();
render('all');
</script>
