# 436. Membangun Reporting Section dengan Chart Progress Project

## Definisi & Target
Membangun halaman reporting dengan chart visual untuk menampilkan progress project. Menggunakan Canvas API untuk membuat chart batang, donut, dan line chart yang menunjukkan statistik project.

## Analogi Alur
Seperti dashboard direktur perusahaan: chart-chart menampilkan kesehatan project dalam bentuk visual yang mudah dipahami, seperti status tasks, progress per project, dan trend penyelesaian.

## Poin-Poin Penting
- Donut chart untuk status tasks (todo, in_progress, done)
- Bar chart untuk progress per project
- Statistik card untuk metrik utama
- Semua chart dibuat dengan Canvas API murni
- Responsive terhadap ukuran container

## Kode Implementasi (Gabungan) - HTML, CSS, JavaScript

```html
<!DOCTYPE html>
<html lang="id">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Reporting Section</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body {
  font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif;
  background: #f0f2f5;

  margin-bottom: 25px;

      <h3>Project Progress</h3>
      <canvas id="barChart" width="400" height="300"></canvas>
    </div>
    <div class="chart-card full">
      <h3>Task Completion Trend</h3>
      <canvas id="lineChart" width="800" height="250"></canvas>
    </div>
  </div>
</div>
<script>
const projects = [
  { name: 'Website', tasks: [5, 8, 3] },
  { name: 'Mobile App', tasks: [2, 4, 9] },
  { name: 'Backend', tasks: [7, 3, 0] },
  { name: 'Design', tasks: [1, 2, 6] }
];
const total = { todo: 15, inProgress: 17, done: 18 };

document.getElementById('statsRow').innerHTML = `
  <div class="stat-card"><div class="val">${total.todo + total.inProgress + total.done}</div><div class="lbl">Total Tasks</div></div>
  <div class="stat-card"><div class="val">${total.done}</div><div class="lbl">Completed</div></div>
  <div class="stat-card"><div class="val">${total.inProgress}</div><div class="lbl">In Progress</div></div>
  <div class="stat-card"><div class="val">${total.todo}</div><div class="lbl">Todo</div></div>
`;

function drawDonut() {
  const c = document.getElementById('donutChart');
  const ctx = c.getContext('2d');
  const cx = 200, cy = 140, r = 90, total = 15 + 17 + 18;
  const data = [
    { val: 15, color: '#e94560', label: 'Todo' },
    { val: 17, color: '#f57c00', label: 'In Progress' },
    { val: 18, color: '#388e3c', label: 'Done' }
  ];
  let start = -Math.PI / 2;
  ctx.clearRect(0, 0, 400, 300);
  data.forEach(d => {
    const angle = (d.val / total) * Math.PI * 2;
    ctx.beginPath();
    ctx.arc(cx, cy, r, start, start + angle);
    ctx.arc(cx, cy, r * 0.6, start + angle, start, true);
    ctx.closePath();
    ctx.fillStyle = d.color;
    ctx.fill();
    start += angle;
  });
  ctx.fillStyle = '#333';
  ctx.font = 'bold 18px Arial';
  ctx.textAlign = 'center';
  ctx.fillText(total, cx, cy + 6);
  let ly = 10;
  data.forEach(d => {
    ctx.fillStyle = d.color;
    ctx.fillRect(10, ly, 12, 12);
    ctx.fillStyle = '#333';
    ctx.font = '12px Arial';
    ctx.textAlign = 'left';
    ctx.fillText(d.label + ' (' + d.val + ')', 26, ly + 10);
    ly += 20;
  });
}

function drawBars() {
  const c = document.getElementById('barChart');
  const ctx = c.getContext('2d');
  const w = 400, h = 250;
  ctx.clearRect(0, 0, w, h);
  const max = Math.max(...projects.map(p => p.tasks.reduce((a,b) => a+b, 0)));
  const barW = 50, gap = 30, startX = 40;
  projects.forEach((p, i) => {
    const x = startX + i * (barW + gap);
    const totalTasks = p.tasks.reduce((a,b) => a+b, 0);
    const bh = (totalTasks / max) * 180;
    const y = h - 40 - bh;
    const colors = ['#e94560', '#f57c00', '#388e3c'];
    let accum = 0;
    p.tasks.forEach((t, j) => {
      const th = (t / max) * 180;
      ctx.fillStyle = colors[j];
      ctx.fillRect(x, h - 40 - accum - th, barW, th);
      accum += th;
    });
    ctx.fillStyle = '#333';
    ctx.font = '12px Arial';
    ctx.textAlign = 'center';
    ctx.fillText(p.name, x + barW / 2, h - 10);
  });
  ctx.fillStyle = '#888';
  ctx.fillRect(10, 15, 10, 10);
  ctx.fillStyle = '#333';
  ctx.font = '10px Arial';
  ctx.textAlign = 'left';
  ctx.fillText('Todo', 24, 24);
  ctx.fillStyle = '#f57c00';
  ctx.fillRect(10, 30, 10, 10);
  ctx.fillStyle = '#333';
  ctx.fillText('In Progress', 24, 39);
  ctx.fillStyle = '#388e3c';
  ctx.fillRect(10, 45, 10, 10);
  ctx.fillText('Done', 24, 54);
}

function drawLine() {
  const c = document.getElementById('lineChart');
  const ctx = c.getContext('2d');
  const w = 800, h = 220;
  ctx.clearRect(0, 0, w, h);
  const data = [5, 8, 12, 15, 18, 22, 25, 28, 30, 33];
  const weeks = ['W1','W2','W3','W4','W5','W6','W7','W8','W9','W10'];
  const max = Math.max(...data);
  const padL = 40, padR = 20, padT = 20, padB = 30;
  const chartW = w - padL - padR, chartH = h - padT - padB;
  ctx.strokeStyle = '#e0e0e0';
  ctx.lineWidth = 1;
  for (let i = 0; i <= 5; i++) {
    const y = padT + (chartH / 5) * i;
    ctx.beginPath();
    ctx.moveTo(padL, y);
    ctx.lineTo(w - padR, y);
    ctx.stroke();
    ctx.fillStyle = '#aaa';
    ctx.font = '10px Arial';
    ctx.textAlign = 'right';
    ctx.fillText(Math.round(max - (max / 5) * i), padL - 5, y + 3);
  }
  ctx.beginPath();
  data.forEach((d, i) => {
    const x = padL + (chartW / (data.length - 1)) * i;
    const y = padT + chartH - (d / max) * chartH;
    if (i === 0) ctx.moveTo(x, y);
    else ctx.lineTo(x, y);
  });
  ctx.strokeStyle = '#667eea';
  ctx.lineWidth = 3;
  ctx.stroke();
  data.forEach((d, i) => {
    const x = padL + (chartW / (data.length - 1)) * i;
    const y = padT + chartH - (d / max) * chartH;
    ctx.beginPath();
    ctx.arc(x, y, 5, 0, Math.PI * 2);
    ctx.fillStyle = '#667eea';
    ctx.fill();
    ctx.strokeStyle = 'white';
    ctx.lineWidth = 2;
    ctx.stroke();
  });
  data.forEach((d, i) => {
    const x = padL + (chartW / (data.length - 1)) * i;
    ctx.fillStyle = '#333';
    ctx.font = '10px Arial';
    ctx.textAlign = 'center';
    ctx.fillText(weeks[i], x, h - 5);
  });
}

drawDonut();
drawBars();
drawLine();
</script>
</body>
</html>
```

## Cara Integrasi
Simpan file HTML dan buka di browser. Tiga chart digambar menggunakan Canvas API: donut chart untuk distribusi status task, bar chart untuk progress per project, dan line chart untuk trend penyelesaian. Data bisa diganti dengan data real dari LocalStorage.
