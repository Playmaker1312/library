# 434. Membangun Notification System untuk Update Project

## Definisi & Target
Membangun sistem notifikasi yang memberi tahu user tentang update project. Notifikasi muncul secara real-time saat ada perubahan, dengan indikator badge dan dropdown panel.

## Analogi Alur
Seperti bell notifikasi di smartphone: ada icon lonceng yang menunjukkan jumlah notifikasi belum dibaca. Klik untuk melihat daftar notifikasi, bisa ditandai sudah dibaca atau dihapus.

## Poin-Poin Penting
- Notifikasi berisi: id, title, message, type, isRead, timestamp, link
- Icon bell dengan badge counter untuk notifikasi unread
- Dropdown panel menampilkan daftar notifikasi
- Mark as read untuk notifikasi individual atau semua
- Auto-generate notifikasi saat ada activity log tertentu

let notifications = JSON.parse(localStorage.getItem('pmNotifications')) || [];

function addNotification(title, message, type, link) {
  notifications.unshift({
    id: 'n' + Date.now(),
    title,
    message,
    type: type || 'info',
    isRead: false,
    timestamp: new Date().toISOString(),
    link: link || '#'
  });
  if (notifications.length > 100) notifications = notifications.slice(0, 100);
  localStorage.setItem('pmNotifications', JSON.stringify(notifications));
  render();
}

function render() {
  const list = document.getElementById('notifList');
  const unread = notifications.filter(n => !n.isRead).length;
  document.getElementById('notifBadge').textContent = unread > 99 ? '99+' : unread;
  if (notifications.length === 0) {
    list.innerHTML = '<div class="empty">No notifications yet</div>';
    return;
  }
  list.innerHTML = notifications.map((n, i) => {
    const icons = { info: '📌', update: '🔄', warning: '⚠️', success: '✅' };
    const time = new Date(n.timestamp).toLocaleString();
    return `<div class="notif-item ${n.isRead ? '' : 'unread'}" onclick="markRead('${n.id}')">
      <div class="notif-icon">${icons[n.type] || '📌'}</div>
      <div class="notif-content">
        <div class="title">${n.title}</div>
        <div class="message">${n.message}</div>
        <div class="time">${time}</div>
      </div>
      <div class="notif-actions">
        <button onclick="event.stopPropagation();removeNotif('${n.id}')">🗑️</button>
      </div>
    </div>`;
  }).join('');
}

function markRead(id) {
  const n = notifications.find(x => x.id === id);
  if (n) { n.isRead = true; localStorage.setItem('pmNotifications', JSON.stringify(notifications)); render(); }
}

function markAllRead() {
  notifications.forEach(n => n.isRead = true);
  localStorage.setItem('pmNotifications', JSON.stringify(notifications));
  render();
}

function removeNotif(id) {
  notifications = notifications.filter(n => n.id !== id);
  localStorage.setItem('pmNotifications', JSON.stringify(notifications));
  render();
}

addNotification('Task Completed', 'Task "Frontend Development" has been marked as done by Alice', 'success');

## Cara Integrasi
Simpan file HTML, buka di browser. Panggil `addNotification(title, message, type, link)` dari modul lain. Badge bell menampilkan jumlah unread. Klik notifikasi untuk mark as read. Tombol "Mark All Read" untuk membersihkan semua.
