# 429. Mengimplementasikan Multi-Project Management dengan Navigasi Antar Project

## Definisi & Target
Membangun fitur multi-project management dengan navigasi untuk berpindah antar project. Setiap project memiliki halaman detail sendiri dan sidebar navigasi untuk memilih project.

## Analogi Alur
Seperti aplikasi Slack yang memiliki banyak workspace: user bisa membuat, memilih, dan berpindah antar project melalui sidebar navigasi yang rapi.

## Poin-Poin Penting
- Sidebar menampilkan daftar semua project
- Fitur create project baru dengan modal form
- Setiap project memiliki halaman dengan tabs (Tasks, Timeline, Team, Files)
- Navigasi yang smooth tanpa reload halaman
- Active project disimpan di state aplikasi

<script>
let data = JSON.parse(localStorage.getItem('pmApp')) || { projects: {}, tasks: {}, members: {}, comments: {}, activityLogs: [], notifications: [], attachments: [] };
let activeProjectId = null;

function saveData() { localStorage.setItem('pmApp', JSON.stringify(data)); }

function renderNav() {
  const nav = document.getElementById('projectNav');
  nav.innerHTML = Object.values(data.projects).map(p =>
    `<li class="${p.id === activeProjectId ? 'active' : ''}" onclick="selectProject('${p.id}')">
      <span class="status-dot ${p.status}"></span>${p.name}
    </li>`
  ).join('');
}

function selectProject(id) {
  activeProjectId = id;
  renderNav();
  renderProject();
}

function renderProject() {
  const p = data.projects[activeProjectId];
  if (!p) { document.getElementById('mainContent').innerHTML = '<h1>Select a project</h1>'; return; }
  document.getElementById('mainContent').innerHTML = `
    <h1>${p.name}</h1>
    <p style="color:#888;margin-bottom:20px">${p.description || ''} · Priority: ${p.priority} · Deadline: ${p.deadline || '-'}</p>
    <div class="tabs">
      <div class="tab active" onclick="showTab(this,'tasks')">Tasks</div>
      <div class="tab" onclick="showTab(this,'timeline')">Timeline</div>
      <div class="tab" onclick="showTab(this,'team')">Team</div>
    </div>
    <div id="tabTasks" class="tab-content active"><p>Task list will appear here</p></div>
    <div id="tabTimeline" class="tab-content"><p>Gantt chart will appear here</p></div>
    <div id="tabTeam" class="tab-content"><p>Team members will appear here</p></div>
  `;
}

function showTab(el, name) {
  document.querySelectorAll('.tab').forEach(t => t.classList.remove('active'));
  document.querySelectorAll('.tab-content').forEach(t => t.classList.remove('active'));
  el.classList.add('active');
  document.getElementById('tab' + name.charAt(0).toUpperCase() + name.slice(1)).classList.add('active');
}

document.getElementById('addProjectBtn').onclick = () => document.getElementById('projectModal').classList.add('show');
document.getElementById('saveProjectBtn').onclick = () => {
  const id = 'p' + Date.now();
  data.projects[id] = { id, name: document.getElementById('projectName').value || 'Untitled', description: document.getElementById('projectDesc').value, status: 'active', priority: document.getElementById('projectPriority').value, deadline: document.getElementById('projectDeadline').value, createdAt: new Date().toISOString() };
  saveData();
  document.getElementById('projectModal').classList.remove('show');
  document.getElementById('projectName').value = '';
  document.getElementById('projectDesc').value = '';
  selectProject(id);
};

if (Object.keys(data.projects).length > 0) {
  activeProjectId = Object.keys(data.projects)[0];
}
renderNav();
renderProject();
</script>

## Cara Integrasi
Simpan sebagai file HTML, buka di browser. Sidebar menampilkan daftar project, klik untuk navigasi. Tombol "New Project" membuka modal untuk membuat project baru. Tab system menyediakan navigasi antar modul dalam project.
