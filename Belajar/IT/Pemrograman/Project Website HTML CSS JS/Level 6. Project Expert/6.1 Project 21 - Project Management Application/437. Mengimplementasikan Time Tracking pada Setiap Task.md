# 437. Mengimplementasikan Time Tracking pada Setiap Task

## Definisi & Target
Membangun fitur time tracking untuk mencatat waktu yang dihabiskan pada setiap task. Fitur ini mencakup start/stop timer, manual time entry, dan total waktu per task.

## Analogi Alur
Seperti aplikasi stopwatch di smartphone: Anda tekan start saat mulai mengerjakan task, pause saat istirahat, dan stop saat selesai. Total waktu tercatat otomatis.

## Poin-Poin Penting
- Timer dengan tombol Start, Pause, Resume, Stop
- Display waktu dalam format HH:MM:SS
- Manual time entry untuk menambahkan waktu secara manual
- Total waktu tersimpan di data task
- Log waktu per entry (start-end duration)

```html
<script>
let state = { currentTaskId: null, seconds: 0, isRunning: false, isPaused: false, interval: null, startTime: null };
let timeLogs = JSON.parse(localStorage.getItem('pmTimeLogs')) || {};

const tasks = [
  { id: 't1', name: 'Frontend Development' },
  { id: 't2', name: 'Backend API Design' },
  { id: 't3', name: 'Database Schema' },
  { id: 't4', name: 'UI/UX Design' }
];

function init() {
  const sel = document.getElementById('taskSelect');
  sel.innerHTML = tasks.map(t => `<option value="${t.id}">${t.name}</option>`).join('');
  sel.onchange = () => { if (!state.isRunning) loadTaskLogs(); };
  loadTaskLogs();
}

function loadTaskLogs() {
  const taskId = document.getElementById('taskSelect').value;
  state.currentTaskId = taskId;
  if (!timeLogs[taskId]) timeLogs[taskId] = { entries: [], totalSeconds: 0 };
  renderLogs();
}

function formatTime(secs) {
  const h = String(Math.floor(secs / 3600)).padStart(2, '0');
  const m = String(Math.floor((secs % 3600) / 60)).padStart(2, '0');
  const s = String(secs % 60).padStart(2, '0');
  return h + ':' + m + ':' + s;
}

function updateDisplay() {
  document.getElementById('timerDisplay').textContent = formatTime(state.seconds);
}

function startTimer() {
  if (state.isRunning) return;
  state.isRunning = true;
  state.isPaused = false;
  state.startTime = Date.now() - state.seconds * 1000;
  state.interval = setInterval(() => {
    state.seconds = Math.floor((Date.now() - state.startTime) / 1000);
    updateDisplay();
  }, 100);
  document.getElementById('startBtn').disabled = true;
  document.getElementById('pauseBtn').disabled = false;
  document.getElementById('stopBtn').disabled = false;
}

function pauseTimer() {
  if (!state.isRunning || state.isPaused) return;
  state.isPaused = true;
  clearInterval(state.interval);
  state.seconds = Math.floor((Date.now() - state.startTime) / 1000);
  updateDisplay();
  document.getElementById('pauseBtn').textContent = '▶ Resume';
  document.getElementById('pauseBtn').onclick = resumeTimer;
}

function resumeTimer() {
  state.isPaused = false;
  state.startTime = Date.now() - state.seconds * 1000;
  state.interval = setInterval(() => {
    state.seconds = Math.floor((Date.now() - state.startTime) / 1000);
    updateDisplay();
  }, 100);
  document.getElementById('pauseBtn').textContent = '⏸ Pause';
  document.getElementById('pauseBtn').onclick = pauseTimer;
}

function stopTimer() {
  clearInterval(state.interval);
  state.isRunning = false;
  state.isPaused = false;
  const taskId = state.currentTaskId;
  const now = new Date().toISOString();
  const entry = { start: state.startTime ? new Date(state.startTime).toISOString() : now, end: now, duration: state.seconds };
  if (!timeLogs[taskId]) timeLogs[taskId] = { entries: [], totalSeconds: 0 };
  timeLogs[taskId].entries.push(entry);
  timeLogs[taskId].totalSeconds += state.seconds;
  localStorage.setItem('pmTimeLogs', JSON.stringify(timeLogs));
  document.getElementById('startBtn').disabled = false;
  document.getElementById('pauseBtn').disabled = true;
  document.getElementById('stopBtn').disabled = true;
  document.getElementById('pauseBtn').textContent = '⏸ Pause';
  document.getElementById('pauseBtn').onclick = pauseTimer;
  state.seconds = 0;
  updateDisplay();
  loadTaskLogs();
}

function renderLogs() {
  const taskId = state.currentTaskId;
  const log = timeLogs[taskId] || { entries: [], totalSeconds: 0 };
  const entriesDiv = document.getElementById('logEntries');
  if (log.entries.length === 0) {
    entriesDiv.innerHTML = '<div style="color:#666;font-size:13px">No time entries yet</div>';
  } else {
    entriesDiv.innerHTML = [...log.entries].reverse().slice(0, 10).map(e =>
      `<div class="log-entry">
        <span>${new Date(e.start).toLocaleTimeString()} - ${new Date(e.end).toLocaleTimeString()}</span>
        <span>${formatTime(e.duration)}</span>
      </div>`
    ).join('');
  }
  document.getElementById('totalTime').textContent = formatTime(log.totalSeconds);
}

init();
</script>
</body>
</html>
```

## Cara Integrasi
Simpan file HTML, buka di browser. Pilih task dari dropdown, tekan Start untuk memulai timer. Pause dan Resume untuk jeda. Stop untuk menyimpan entry. Total waktu tersimpan per task di LocalStorage.
