# 439. Mengimplementasikan Comment Thread pada Setiap Task

## Definisi & Target
Membangun sistem comment thread pada setiap task yang memungkinkan anggota tim berdiskusi. Comments memiliki nested replies, timestamp, dan author info.

## Analogi Alur
Seperti thread diskusi di forum: setiap task memiliki thread komentar sendiri. Anggota tim bisa menambahkan komentar, membalas komentar yang sudah ada, dan melihat riwayat diskusi secara kronologis.

## Poin-Poin Penting
- Setiap comment memiliki id, taskId, author, content, parentId, timestamp
- Comments ditampilkan secara kronologis
- Replies ditampilkan sebagai nested di bawah parent comment
- Form input untuk menambahkan comment baru
- Delete comment dengan konfirmasi

<script>
const TASK_ID = 't1';
let comments = JSON.parse(localStorage.getItem('pmComments')) || {};
if (!comments[TASK_ID]) comments[TASK_ID] = [];

function save() { localStorage.setItem('pmComments', JSON.stringify(comments)); }

function getComments() { return comments[TASK_ID] || []; }

function addComment() {
  const input = document.getElementById('commentInput');
  const text = input.value.trim();
  if (!text) return;
  comments[TASK_ID].push({
    id: 'c' + Date.now(),
    taskId: TASK_ID,
    parentId: null,
    author: 'Current User',
    content: text,
    timestamp: new Date().toISOString()
  });
  save();
  input.value = '';
  render();
}

function addReply(parentId) {
  const input = document.getElementById('replyInput_' + parentId);
  if (!input) return;
  const text = input.value.trim();
  if (!text) return;
  comments[TASK_ID].push({
    id: 'c' + Date.now(),
    taskId: TASK_ID,
    parentId: parentId,
    author: 'Current User',
    content: text,
    timestamp: new Date().toISOString()
  });
  save();
  render();
}

function deleteComment(id) {
  if (!confirm('Delete this comment?')) return;
  comments[TASK_ID] = comments[TASK_ID].filter(c => c.id !== id && c.parentId !== id);
  save();
  render();
}

function showReplyInput(parentId) {
  document.querySelectorAll('.reply-input-area').forEach(el => el.remove());
  const parent = document.getElementById('comment_' + parentId);
  if (!parent) return;
  const div = document.createElement('div');
  div.className = 'reply-input-area';
  div.id = 'replyArea_' + parentId;
  div.innerHTML = `<input type="text" id="replyInput_${parentId}" placeholder="Write a reply..." onkeypress="if(event.key==='Enter') addReply('${parentId}')"><button onclick="addReply('${parentId}')">Reply</button>`;
  parent.after(div);
  document.getElementById('replyInput_' + parentId).focus();
}

function render() {
  const thread = document.getElementById('commentThread');
  const allComments = getComments();
  if (allComments.length === 0) {
    thread.innerHTML = '<div class="empty">No comments yet. Start the discussion!</div>';
    return;
  }
  const parents = allComments.filter(c => !c.parentId);
  thread.innerHTML = parents.map(p => {
    const replies = allComments.filter(c => c.parentId === p.id);
    const time = new Date(p.timestamp).toLocaleString();
    const avatar = p.author.split(' ').map(w => w.charAt(0)).join('').toUpperCase().slice(0, 2);
    return `<div class="comment" id="comment_${p.id}">
      <div class="avatar">${avatar}</div>
      <div class="content">
        <div><span class="author">${p.author}</span><span class="time">${time}</span></div>
        <div class="text">${p.content}</div>
        <div class="actions">
          <button onclick="showReplyInput('${p.id}')">Reply</button>
          <button class="delete-btn" onclick="deleteComment('${p.id}')">Delete</button>
        </div>
      </div>
    </div>
    ${replies.map(r => {
      const rt = new Date(r.timestamp).toLocaleString();
      const ra = r.author.split(' ').map(w => w.charAt(0)).join('').toUpperCase().slice(0, 2);
      return `<div class="comment reply">
        <div class="avatar" style="background:linear-gradient(135deg,#f093fb,#f5576c)">${ra}</div>
        <div class="content">
          <div><span class="author">${r.author}</span><span class="time">${rt}</span></div>
          <div class="text">${r.content}</div>
          <div class="actions">
            <button class="delete-btn" onclick="deleteComment('${r.id}')">Delete</button>
          </div>
        </div>
      </div>`;
    }).join('')}`;
  }).join('');
}

if (comments[TASK_ID].length === 0) {
  comments[TASK_ID] = [
    { id: 'c1', taskId: TASK_ID, parentId: null, author: 'Alice', content: 'I have completed the initial layout. Ready for review.', timestamp: '2025-06-10T09:00:00Z' },

  ];
  save();
}

</script>

## Cara Integrasi
Simpan file HTML, buka di browser. Tulis comment di input field dan tekan Send. Klik Reply pada comment untuk membalas. Delete untuk menghapus comment (beserta replies-nya). Data tersimpan per taskId di LocalStorage.
