# 440. Menyimpan Seluruh Data ke LocalStorage dengan Struktur yang Terorganisir

## Definisi & Target
Menggabungkan seluruh modul aplikasi project management dengan penyimpanan data terpusat ke LocalStorage. Semua data (projects, tasks, members, comments, logs, notifications, attachments, timeLogs) tersimpan dalam satu sistem terorganisir.

## Analogi Alur
Seperti lemari arsip kantor dengan banyak laci: setiap laci (key) berisi folder (entity) yang tertata rapi. Semua arsip bisa diakses dari satu sistem terpusat.

## Poin-Poin Penting
- Data disimpan dengan key prefix "pm" untuk menghindari konflik
- Struktur data: pmApp untuk data utama, pmMembers, pmTimeLogs, pmComments, pmNotifications, pmAttachments, pmLogs
- Setiap modul terpisah namun terintegrasi melalui shared data store
- Fungsi export/import data untuk backup
- Auto-save setelah setiap perubahan

const STORE_KEYS = ['pmApp', 'pmMembers', 'pmTimeLogs', 'pmComments', 'pmNotifications', 'pmAttachments', 'pmLogs'];

function getData() {
  const data = {};
  STORE_KEYS.forEach(key => {
    try {
      const val = localStorage.getItem(key);
      data[key] = val ? JSON.parse(val) : null;
    } catch(e) {
      data[key] = { error: e.message };
    }
  });
  return data;
}

function refreshStats() {
  const data = getData();
  const stats = document.getElementById('dataStats');
  const keyLabels = {
    pmApp: 'Main App Data',
    pmMembers: 'Team Members',
    pmTimeLogs: 'Time Logs',
    pmComments: 'Comments',
    pmNotifications: 'Notifications',
    pmAttachments: 'Attachments',
    pmLogs: 'Activity Logs'
  };
  stats.innerHTML = STORE_KEYS.map(key => {
    const val = data[key];
    let count = 0;
    if (val) {
      if (Array.isArray(val)) count = val.length;
      else if (typeof val === 'object') count = Object.keys(val).length;
    }
    return `<div class="data-card">
      <h3>${keyLabels[key] || key}</h3>
      <div class="key">${key}</div>
      <div class="count">${count}</div>
    </div>`;
  }).join('');

  document.getElementById('dataPreview').textContent =
    JSON.stringify(data, null, 2);
}

function exportData() {
  const data = getData();
  const blob = new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' });
  const url = URL.createObjectURL(blob);
  const a = document.createElement('a');
  a.href = url;
  a.download = 'pm-backup-' + new Date().toISOString().slice(0, 10) + '.json';
  a.click();
  URL.revokeObjectURL(url);
}

function importData(event) {
  const file = event.target.files[0];
  if (!file) return;
  const reader = new FileReader();
  reader.onload = function(e) {
    try {
      const data = JSON.parse(e.target.result);
      STORE_KEYS.forEach(key => {
        if (data[key] !== undefined) {
          localStorage.setItem(key, JSON.stringify(data[key]));
        }
      });
      refreshStats();
      alert('Data imported successfully!');
    } catch(err) {
      alert('Invalid file format: ' + err.message);
    }
  };
  reader.readAsText(file);
  event.target.value = '';
}

function clearAll() {
  if (!confirm('Are you sure you want to clear ALL data? This cannot be undone!')) return;
  if (!confirm('Really? All projects, tasks, comments will be deleted!')) return;
  STORE_KEYS.forEach(key => localStorage.removeItem(key));
  refreshStats();
  alert('All data cleared.');
}

if (!localStorage.getItem('pmApp')) {
  const sample = {
    projects: {
      p1: { id: 'p1', name: 'Sample Project', description: 'A sample project', status: 'active', priority: 'medium', deadline: '2025-07-01', createdAt: new Date().toISOString() }

    tasks: {

    },
    members: {},
    comments: {},
    activityLogs: [],
    notifications: [],
    attachments: []
  };
  localStorage.setItem('pmApp', JSON.stringify(sample));
}

refreshStats();
</script>

## Cara Integrasi
Simpan file HTML, buka di browser. Halaman ini menampilkan semua data yang tersimpan di LocalStorage. Gunakan Export untuk backup data, Import untuk restore dari file backup. Setiap modul aplikasi menyimpan data dengan key yang sudah ditentukan.
