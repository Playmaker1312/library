# 69. Memahami Fetch API untuk Mengambil Data

## Definisi & Target
Fetch API adalah antarmuka JavaScript modern untuk membuat HTTP request dan mengambil data dari server atau API. Menggantikan XMLHttpRequest (XHR) yang lebih tua.

## Analogi Alur
Fetch API seperti Anda memesan makanan via telepon. Anda menelepon restoran (request ke server), menunggu sebentar, lalu mendapat makanan (response). Jika restoran tutup, Anda dapat pesan error.

## Poin-Poin Penting
- `fetch(url)` mengembalikan Promise
- Response perlu di-parsing: `.json()` untuk JSON, `.text()` untuk teks
- Default method adalah GET
- Untuk POST, PUT, DELETE: tambahkan options object
- Fetch Promise hanya reject untuk network error, bukan HTTP error
- Cek `response.ok` atau `response.status` untuk HTTP error handling
- Gunakan `async/await` untuk kode yang lebih rapi

```javascript
var result = document.getElementById('fetchResult');
var loading = document.getElementById('loading');
var statusEl = document.getElementById('fetchStatus');
var statusDot = document.getElementById('statusDot');
var API = 'https://jsonplaceholder.typicode.com';

function setStatus(text, isError) {
  statusEl.textContent = text;
  statusDot.className = 'status-dot' + (isError ? ' error' : ' success');
}

function showLoading(show) {
  loading.style.display = show ? 'block' : 'none';
}

function handleError(err) {
  result.textContent = 'Error: ' + err.message;
  setStatus('Error: ' + err.message, true);
  showLoading(false);
}

document.getElementById('btnGet').addEventListener('click', function() {
  showLoading(true);
  setStatus('Mengambil data...', false);
  fetch(API + '/todos/1')
    .then(function(response) {
      if (!response.ok) {
        throw new Error('HTTP ' + response.status);
      }
      return response.json();
    })
    .then(function(data) {
      result.textContent = JSON.stringify(data, null, 2);
      setStatus('Sukses: Data diterima', false);
      showLoading(false);
    })
    .catch(handleError);
});

document.getElementById('btnGetUsers').addEventListener('click', function() {
  showLoading(true);
  setStatus('Mengambil data users...', false);
  fetch(API + '/users')
    .then(function(response) { return response.json(); })
    .then(function(data) {
      var output = 'Menampilkan ' + data.length + ' users:\n';
      for (var i = 0; i < data.length; i++) {
        output += (i + 1) + '. ' + data[i].name + ' (' + data[i].email + ')\n';
      }
      result.textContent = output;
      setStatus('Sukses: ' + data.length + ' users dimuat', false);
      showLoading(false);
    })
    .catch(handleError);
});

document.getElementById('btnPost').addEventListener('click', function() {
  showLoading(true);
  fetch(API + '/posts', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({
      title: 'Data Baru',
      body: 'Ini konten post baru dari Fetch API',
      userId: 1
    })
  })
  .then(function(response) { return response.json(); })
  .then(function(data) {
    result.textContent = JSON.stringify(data, null, 2);
    setStatus('Sukses: Data terkirim (ID: ' + data.id + ')', false);
    showLoading(false);
  })
  .catch(handleError);
});

document.getElementById('btnError').addEventListener('click', function() {
  showLoading(true);
  fetch('https://jsonplaceholder.typicode.com/invalid-endpoint')
    .then(function(response) {
      if (!response.ok) {
        throw new Error('HTTP Error: ' + response.status);
      }
      return response.json();
    })
    .then(function(data) {
      result.textContent = JSON.stringify(data, null, 2);
      showLoading(false);
    })
    .catch(handleError);
});
```

## Cara Integrasi
1. Panggil `fetch(url)` dengan URL API endpoint
2. Chain `.then(response => response.json())` untuk parse JSON
3. Chain `.then(data => { ... })` untuk menggunakan data
4. Tambahkan `.catch(error => { ... })` untuk error handling
5. Untuk POST: tambahkan options object dengan method, headers, body
