# 68. Cara Menghapus Elemen DOM

## Definisi & Target
Dua metode utama untuk menghapus elemen DOM: `element.remove()` (modern, langsung) dan `parent.removeChild(element)` (tradisional, dari parent).

## Analogi Alur
remove() seperti mengeluarkan buku dari rak dan langsung membuangnya. removeChild() seperti minta petugas perpustakaan (parent) untuk mengeluarkan buku tertentu dari rak.

## Poin-Poin Penting
- `element.remove()`: menghapus elemen langsung (ES6+, lebih simpel)
- `parent.removeChild(child)`: menghapus child dari parent (kompatibel semua browser)
- `element.innerHTML = ''`: menghapus semua child (tapi tidak aman untuk event listener)
- Hapus event listener sebelum menghapus untuk mencegah memory leak
- `replaceChild(new, old)`: mengganti elemen lama dengan baru
- Setelah dihapus, elemen masih ada di memori jika masih ada referensi

```javascript
var container = document.getElementById('cardContainer');
var log = document.getElementById('removeLog');
var cardId = 5;

function logMsg(msg) {
  log.textContent = msg;
}

function addDeleteListeners() {
  var deleteBtns = document.querySelectorAll('.btn-delete');
  for (var i = 0; i < deleteBtns.length; i++) {
    deleteBtns[i].addEventListener('click', function() {
      var card = this.parentElement;
      var id = card.getAttribute('data-id');
      card.remove();
      logMsg('Card ' + id + ' dihapus dengan remove()');
    });
  }
}

document.getElementById('btnRemoveFirst').addEventListener('click', function() {
  var firstCard = container.querySelector('.demo-card');
  if (firstCard) {
    var id = firstCard.getAttribute('data-id');
    container.removeChild(firstCard);
    logMsg('Card ' + id + ' dihapus dengan removeChild()');
  } else {
    logMsg('Tidak ada card yang bisa dihapus');
  }
});

document.getElementById('btnRemoveLast').addEventListener('click', function() {
  var cards = container.querySelectorAll('.demo-card');
  if (cards.length > 0) {
    var lastCard = cards[cards.length - 1];
    var id = lastCard.getAttribute('data-id');
    lastCard.remove();
    logMsg('Card ' + id + ' dihapus dengan remove()');
  } else {
    logMsg('Tidak ada card yang bisa dihapus');
  }
});

document.getElementById('btnReplaceCard').addEventListener('click', function() {
  var card2 = container.querySelector('[data-id="2"]');
  if (card2) {
    var newCard = document.createElement('div');
    newCard.className = 'demo-card';
    newCard.setAttribute('data-id', 'new');
    newCard.innerHTML = '<span>Card Baru (pengganti)</span><button class="btn-delete">X</button>';
    container.replaceChild(newCard, card2);
    addDeleteListeners();
    logMsg('Card 2 diganti dengan card baru');
  } else {
    logMsg('Card 2 sudah tidak ada');
  }
});

document.getElementById('btnResetCards').addEventListener('click', function() {
  container.innerHTML = '';
  var cards = [
    { id: 1, text: 'Card 1' },
    { id: 2, text: 'Card 2' },
    { id: 3, text: 'Card 3' },
    { id: 4, text: 'Card 4' }
  ];
  for (var i = 0; i < cards.length; i++) {
    var div = document.createElement('div');
    div.className = 'demo-card';
    div.setAttribute('data-id', cards[i].id);
    div.innerHTML = '<span>' + cards[i].text + '</span><button class="btn-delete">X</button>';
    container.appendChild(div);
  }
  addDeleteListeners();
  logMsg('Card di-reset ke awal');
});

addDeleteListeners();
logMsg('Klik X pada card atau tombol kontrol');
```

## Cara Integrasi
1. Gunakan `element.remove()` untuk menghapus elemen (cara modern dan simpel)
2. Gunakan `parent.removeChild(child)` jika perlu referensi parent
3. Gunakan `innerHTML = ''` untuk mengosongkan semua child
4. Setelah menghapus, pastikan tidak ada referensi ke elemen untuk garbage collection
5. Event listener pada elemen yang dihapus tetap di memori jika tidak di-remove
