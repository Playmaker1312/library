# 75. Memahami Cara Menangani Error dalam Fetch API

## Definisi & Target
Error handling dalam Fetch API menggunakan `try/catch` (dengan async/await) atau `.catch()` (dengan Promise chain) untuk menangani network error, HTTP error, dan error parsing.

## Analogi Alur
Error handling seperti asuransi perjalanan. Saat perjalanan (fetch) gagal karena jalan rusak (network error) atau tempat tujuan tutup (HTTP 404), asuransi (catch block) akan menangani situasi: memberi tahu Anda dan memberikan solusi alternatif.

## Poin-Poin Penting
- Fetch hanya reject untuk network error (koneksi putus, DNS gagal)
- HTTP 404, 500, dll tidak otomatis reject - perlu cek `response.ok`
- `response.ok` = true jika status 200-299
- `response.status` memberikan kode HTTP
- `try/catch` untuk async/await, `.catch()` untuk Promise chain
- `finally` block untuk kode yang selalu dijalankan (loading selesai)
- Parsing error: `response.json()` juga bisa gagal (invalid JSON)
- Berikan feedback ke user saat error terjadi

```javascript
var output = document.getElementById('errorOutput');
var httpStatus = document.getElementById('httpStatus');
var resultContainer = document.querySelector('.error-result');
var log = document.getElementById('errorLog');

var API = 'https://jsonplaceholder.typicode.com';

function setStatus(text, isError) {
  log.textContent = 'Status: ' + text;
  resultContainer.className = 'error-result' + (isError ? ' status-error' : ' status-success');
}

async function safeFetch(url, label) {
  try {
    setStatus('Mengambil ' + label + '...', false);
    httpStatus.textContent = 'Loading...';

    var response = await fetch(url);

    httpStatus.textContent = response.status + ' ' + response.statusText;

    if (!response.ok) {
      throw new Error('HTTP Error ' + response.status + ': ' + response.statusText + ' untuk ' + label);
    }

    var data = await response.json();
    var text = typeof data === 'object' ? JSON.stringify(data, null, 2).substring(0, 500) : String(data);
    output.textContent = 'Sukses! Data dari ' + label + ':\n\n' + text;
    setStatus('Sukses: ' + label + ' - Status ' + response.status, false);

  } catch (err) {
    var errorMsg = '';
    if (err.name === 'TypeError') {
      errorMsg = 'Network Error: Tidak bisa terhubung ke server.\n' +
        'Pesan: ' + err.message + '\n\n' +
        'Penyebab: Koneksi internet terputus, server down, atau CORS error.';
    } else if (err.message.startsWith('HTTP Error')) {
      errorMsg = 'HTTP Error tertangkap:\n' + err.message + '\n\n' +
        'Solusi: Periksa endpoint API atau coba lagi nanti.';
    } else if (err instanceof SyntaxError) {
      errorMsg = 'Parse Error: Response bukan JSON valid.\n' +
        'Pesan: ' + err.message + '\n\n' +
        'Solusi: Periksa Content-Type response dari server.';
    } else {
      errorMsg = 'Error: ' + err.message;
    }
    output.textContent = errorMsg;
    setStatus('Error: ' + err.message, true);
  }
}

document.getElementById('btnSuccess').addEventListener('click', function() {
  safeFetch(API + '/todos/1', 'Todo 1');
});

document.getElementById('btn404').addEventListener('click', function() {
  safeFetch(API + '/nonexistent', 'endpoint tidak ada');

});

document.getElementById('btnAll').addEventListener('click', async function() {
  output.textContent = 'Menjalankan semua test secara sequential...\n\n';
  httpStatus.textContent = 'Multi-test...';

  var tests = [
    { url: API + '/todos/1', label: 'Sukses' },
    { url: API + '/nonexistent', label: '404' },
    { url: 'https://httpstat.us/500', label: '500' },
    { url: 'https://www.google.com', label: 'Invalid JSON' }
  ];

  for (var i = 0; i < tests.length; i++) {
    output.textContent += 'Test ' + (i + 1) + ': ' + tests[i].label + '\n';
    output.textContent += '  Mencoba ' + tests[i].url + '...\n';
    try {
      var r = await fetch(tests[i].url);
      output.textContent += '  Status: ' + r.status + ' ' + r.statusText + '\n';
      if (!r.ok) {
        output.textContent += '  -> Error (HTTP ' + r.status + ')\n';
      } else {
        var txt = await r.text();
        output.textContent += '  -> Sukses (' + txt.length + ' bytes)\n';
      }
    } catch (err) {
      output.textContent += '  -> Network Error: ' + err.message + '\n';
    }
    output.textContent += '\n';
  }

  setStatus('Semua test selesai', false);
});
```

## Cara Integrasi
1. Bungkus fetch dalam `try/catch` untuk menangkap semua error
2. Cek `response.ok` setelah fetch untuk mendeteksi HTTP error
3. Beri pesan error yang informatif ke user
4. Gunakan `finally` untuk menyembunyikan loading indicator
5. Bedakan jenis error (network, HTTP, parsing) untuk pesan yang tepat
