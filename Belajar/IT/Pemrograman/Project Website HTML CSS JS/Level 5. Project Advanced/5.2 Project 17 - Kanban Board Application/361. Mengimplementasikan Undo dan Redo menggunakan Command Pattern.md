# 361. Mengimplementasikan Undo dan Redo menggunakan Command Pattern

## Definisi & Target
Mengimplementasikan fitur Undo dan Redo untuk operasi pada board menggunakan Command Pattern. Targetnya adalah pengguna bisa membatalkan atau mengulangi aksi.

## Analogi Alur
Seperti mesin waktu yang bisa memutar kembali atau memajukan perubahan. Setiap aksi disimpan sebagai snapshot yang bisa dikembalikan.

## Poin-Poin Penting
- Setiap aksi menyimpan state sebelum dan sesudah
- Stack untuk undo dan redo
- Tombol Undo (Ctrl+Z) dan Redo (Ctrl+Shift+Z)
- Batas maksimal stack untuk menghindari memory leak

## Kode Implementasi (Gabungan) - HTML, CSS, JavaScript
```html
<!DOCTYPE html>
<html lang="id">
<head>
<meta charset="UTF-8">
<meta name="viewport" content="width=device-width, initial-scale=1.0">
<title>Undo Redo</title>
<style>
* { margin: 0; padding: 0; box-sizing: border-box; }
body { font-family: 'Segoe UI', Tahoma, Geneva, Verdana, sans-serif; background: #1a1a2e; height: 100vh; }
.header { padding: 16px 24px; background: #16213e; color: #fff; display: flex; align-items: center; gap: 12px; }
.header button { padding: 6px 14px; background: #2a3a6a; color: #fff; border: none; border-radius: 6px; cursor: pointer; font-size: 13px; }
.header button:disabled { opacity: 0.4; cursor: not-allowed; }
.board { display: flex; gap: 16px; padding: 20px; height: calc(100vh - 60px); overflow-x: auto; }
.column { min-width: 280px; background: #1e2a4a; border-radius: 10px; display: flex; flex-direction: column; max-height: 100%; }
.col-header { padding: 12px 14px; color: #fff; font-weight: 600; border-bottom: 1px solid rgba(255,255,255,0.1); }
.col-body { flex: 1; padding: 8px; }
.card { background: #2a3a6a; color: #e0e0e0; padding: 10px 12px; border-radius: 6px; font-size: 13px; margin-bottom: 6px; display: flex; justify-content: space-between; }
.card .del { background: none; border: none; color: #e53935; cursor: pointer; }
.add-card-btn { padding: 8px; color: rgba(255,255,255,0.4); cursor: pointer; font-size: 13px; text-align: center; }
</style>
</head>
<body>
<div class="header"><span>📋 Undo/Redo</span><button id="undoBtn" disabled>↩ Undo</button><button id="redoBtn" disabled>↪ Redo</button></div>
<div class="board" id="board"></div>
<script>
const MAX_HISTORY = 20;
let data = [{ id: 'c1', title: 'To Do', cards: [{ id: 'k1', title: 'Card A' }] }, { id: 'c2', title: 'Done', cards: [] }];
let undoStack = [JSON.stringify(data)];
let redoStack = [];
function saveState() { undoStack.push(JSON.stringify(data)); if (undoStack.length > MAX_HISTORY) undoStack.shift(); redoStack = []; updateButtons(); }
function updateButtons() { document.getElementById('undoBtn').disabled = undoStack.length <= 1; document.getElementById('redoBtn').disabled = redoStack.length === 0; }
function undo() { if (undoStack.length <= 1) return; redoStack.push(undoStack.pop()); data = JSON.parse(undoStack[undoStack.length - 1]); render(); updateButtons(); }
function redo() { if (redoStack.length === 0) return; undoStack.push(redoStack.pop()); data = JSON.parse(undoStack[undoStack.length - 1]); render(); updateButtons(); }
document.getElementById('undoBtn').addEventListener('click', undo);
document.getElementById('redoBtn').addEventListener('click', redo);
document.addEventListener('keydown', e => { if (e.ctrlKey && e.key === 'z' && !e.shiftKey) { e.preventDefault(); undo(); } if (e.ctrlKey && e.key === 'z' && e.shiftKey) { e.preventDefault(); redo(); } if (e.ctrlKey && e.key === 'y') { e.preventDefault(); redo(); } });
function render() {
const board = document.getElementById('board');
board.innerHTML = '';
data.forEach(col => {
const colDiv = document.createElement('div');
colDiv.className = 'column';
let cardsHtml = col.cards.map(c => `<div class="card"><span>${c.title}</span><button class="del" data-col="${col.id}" data-card="${c.id}">×</button></div>`).join('');
colDiv.innerHTML = `<div class="col-header">${col.title}</div><div class="col-body">${cardsHtml}<div class="add-card-btn" data-col="${col.id}">+ Tambah Card</div></div>`;
colDiv.querySelectorAll('.del').forEach(btn => btn.addEventListener('click', function() {
const col = data.find(c => c.id === this.dataset.col);
if (!col) return; col.cards = col.cards.filter(c => c.id !== this.dataset.card); saveState(); render();
}));
colDiv.querySelector('.add-card-btn').addEventListener('click', function() {
const title = prompt('Judul:');
if (title && title.trim()) { const col = data.find(c => c.id === this.dataset.col); if (col) { col.cards.push({ id: 'k' + Date.now(), title }); saveState(); render(); } }
});
board.appendChild(colDiv);
});
}
render(); updateButtons();
</script>
</body>
</html>
```

## Cara Integrasi
Setiap aksi memanggil saveState() yang menyimpan snapshot JSON data ke undoStack. Undo mengembalikan state sebelumnya dari stack, redo mengambil dari redoStack.
