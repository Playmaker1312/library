# 515. Cara Mengimplementasikan Subscription Billing Menggunakan Cashier Stripe

## Definisi & Target
Subscription billing melibatkan pembuatan paket berlangganan, penanganan siklus tagihan, pembaruan kartu kredit, dan manajemen status subscription. Targetnya adalah mengelola siklus hidup langganan pengguna secara end-to-end.

## Komponen yang Terlibat
- Laravel Cashier + Stripe
- Subscription plans (Price ID)
- Invoice management
- Payment method update
- Webhook events

## Perintah Artisan CLI
```bash
php artisan make:controller SubscriptionController
php artisan make:mail SubscriptionRenewalMail
```

## Kode Implementasi (Per File)

**Subscription Manager (app/Services/SubscriptionService.php)**
```php
namespace App\Services;

use App\Models\User;
use Illuminate\Support\Facades\Log;

class SubscriptionService
{
    public function getSubscriptionStatus(User $user): array
    {
        if (!$user->subscribed()) {
            return ['status' => 'none'];
        }

        $subscription = $user->subscription('default');

        return [
            'status' => $subscription->stripe_status,
            'plan' => $subscription->stripe_price,
            'ends_at' => $subscription->ends_at,
            'on_trial' => $subscription->onTrial(),
            'on_grace_period' => $subscription->onGracePeriod(),
            'cancelled' => $subscription->cancelled(),
        ];
    }

    public function createSubscription(User $user, string $priceId, string $paymentMethodId): void
    {
        $user->newSubscription('default', $priceId)
            ->create($paymentMethodId);
    }

    public function updatePaymentMethod(User $user, string $paymentMethodId): void
    {
        $user->updateDefaultPaymentMethod($paymentMethodId);
    }

    public function getInvoices(User $user, int $limit = 12): array
    {
        return $user->invoices()
            ->take($limit)
            ->map(fn ($invoice) => [
                'id' => $invoice->id,
                'amount' => $invoice->total(),
                'currency' => $invoice->currency,
                'status' => $invoice->status,
                'date' => $invoice->date()->toDateString(),
                'pdf' => $invoice->invoice_pdf,
            ])
            ->toArray();
    }

    public function handleFailedPayment(User $user): void
    {
        Log::warning('Payment failed for user', ['user_id' => $user->id]);
    }
}
```

**Subscription API Controller (app/Http/Controllers/Api/SubscriptionController.php)**
```php
namespace App\Http\Controllers\Api;

use App\Http\Controllers\Controller;
use App\Services\SubscriptionService;
use Illuminate\Http\Request;

class SubscriptionController extends Controller
{
    public function __construct(
        private SubscriptionService $subscriptionService
    ) {}

    public function status(Request $request)
    {
        return response()->json(
            $this->subscriptionService->getSubscriptionStatus($request->user())
        );
    }

    public function subscribe(Request $request)
    {
        $validated = $request->validate([
            'price_id' => 'required|string',
            'payment_method_id' => 'required|string',
        ]);

        $this->subscriptionService->createSubscription(
            $request->user(),
            $validated['price_id'],
            $validated['payment_method_id']
        );

        return response()->json(['message' => 'Subscription created']);
    }

    public function cancel(Request $request)
    {
        $request->user()->subscription('default')->cancel();

        return response()->json(['message' => 'Subscription cancelled']);
    }

    public function resume(Request $request)
    {
        $request->user()->subscription('default')->resume();

        return response()->json(['message' => 'Subscription resumed']);
    }

    public function invoices(Request $request)
    {
        return response()->json(
            $this->subscriptionService->getInvoices($request->user())
        );
    }

    public function updatePaymentMethod(Request $request)
    {
        $validated = $request->validate([
            'payment_method_id' => 'required|string',
        ]);

        $this->subscriptionService->updatePaymentMethod(
            $request->user(),
            $validated['payment_method_id']
        );

        return response()->json(['message' => 'Payment method updated']);
    }
}
```

**Middleware untuk Subscription Check (app/Http/Middleware/EnsureSubscribed.php)**
```php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class EnsureSubscribed
{
    public function handle(Request $request, Closure $next): Response
    {
        if (!$request->user()?->subscribed()) {
            return redirect()->route('plans')
                ->with('error', 'Silakan berlangganan untuk mengakses fitur ini.');
        }

        return $next($request);
    }
}
```

**Webhook Handling di CashierController**
```php
namespace App\Http\Controllers;

use Laravel\Cashier\Http\Controllers\WebhookController;
use Illuminate\Support\Facades\Log;

class CashierWebhookController extends WebhookController
{
    public function handleInvoicePaymentSucceeded($payload): void
    {
        Log::info('Payment succeeded', ['invoice' => $payload['data']['object']['id']]);
    }

    public function handleInvoicePaymentFailed($payload): void
    {
        $customerId = $payload['data']['object']['customer'];
        $user = User::where('stripe_id', $customerId)->first();

        if ($user) {
            Log::warning('Payment failed for user', ['user_id' => $user->id]);
        }
    }
}
```
