# 412. Mengimplementasikan Online Status Indicator Menggunakan Presence Channel

## Definisi & Target
Membangun online status indicator yang menampilkan status user (online, offline, idle) secara real-time menggunakan presence channel. Status akan terupdate otomatis ketika user terhubung/terputus dari WebSocket.

## Komponen yang Terlibat
- Presence channel 'online'
- User model (is_online, last_active_at)
- Online status service
- Status indicator component
- Idle detection (frontend)
- Heartbeat mechanism

### Event - UserStatusChanged
```php
<?php
namespace App\Events;

use App\Models\User;
use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Broadcasting\PresenceChannel;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class UserStatusChanged implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public User $user,
        public string $status
    ) {}

    public function broadcastOn(): PresenceChannel
    {
        return new PresenceChannel('online');
    }

    public function broadcastAs(): string
    {
        return 'status.changed';
    }

    public function broadcastWith(): array
    {
        return [
            'user_id' => $this->user->id,
            'name' => $this->user->name,
            'status' => $this->status,
            'timestamp' => now()->toISO8601String(),
        ];
    }
}
```

### Service - OnlineStatusService
```php
<?php
namespace App\Services;

use App\Events\UserStatusChanged;
use App\Models\User;
use Illuminate\Support\Facades\Cache;

class OnlineStatusService
{
    private const ONLINE_TIMEOUT = 300;

    public function setOnline(User $user): void
    {
        $user->update([
            'is_online' => true,
            'last_active_at' => now(),
            'online_status' => 'online',
        ]);

        Cache::put(
            "user_online_{$user->id}",
            true,
            now()->addSeconds(self::ONLINE_TIMEOUT)
        );

        UserStatusChanged::dispatch($user, 'online');
    }

    public function setOffline(User $user): void
    {
        $user->update([
            'is_online' => false,
            'online_status' => 'offline',
        ]);

        Cache::forget("user_online_{$user->id}");

        UserStatusChanged::dispatch($user, 'offline');
    }

    public function setIdle(User $user): void
    {
        $user->update([
            'online_status' => 'idle',
        ]);

        UserStatusChanged::dispatch($user, 'idle');
    }

    public function heartbeat(User $user): void
    {
        $user->update(['last_active_at' => now()]);

        Cache::put(
            "user_online_{$user->id}",
            true,
            now()->addSeconds(self::ONLINE_TIMEOUT)
        );

        if (!$user->is_online) {
            $this->setOnline($user);
        }

        if ($user->online_status === 'idle') {
            $this->setOnline($user);
        }
    }

    public function isOnline(User $user): bool
    {
        return Cache::get("user_online_{$user->id}", false);
    }

    public function getOnlineUsers(): array
    {
        return User::where('is_online', true)
            ->where('last_active_at', '>', now()->subMinutes(10))
            ->get(['id', 'name', 'avatar_url', 'online_status', 'last_active_at'])
            ->toArray();
    }

    public function getOnlineCount(): int
    {
        return User::where('is_online', true)
            ->where('last_active_at', '>', now()->subMinutes(10))
            ->count();
    }
}
```

### Middleware - TrackUserActivity
```php
<?php
namespace App\Http\Middleware;

use App\Services\OnlineStatusService;
use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class TrackUserActivity
{
    public function __construct(
        private OnlineStatusService $onlineStatus
    ) {}

    public function handle(Request $request, Closure $next): Response
    {
        if ($request->user()) {
            $this->onlineStatus->heartbeat($request->user());
        }

        return $next($request);
    }
}
```

### Controller - OnlineStatusController
```php
<?php
namespace App\Http\Controllers;

use App\Services\OnlineStatusService;
use Illuminate\Http\Request;

class OnlineStatusController extends Controller
{
    public function __construct(
        private OnlineStatusService $onlineStatus
    ) {}

    public function heartbeat(Request $request)
    {
        if ($request->user()) {
            $this->onlineStatus->heartbeat($request->user());
        }

        return response()->json(['status' => 'ok']);
    }

    public function onlineUsers()
    {
        $users = $this->onlineStatus->getOnlineUsers();
        $count = $this->onlineStatus->getOnlineCount();

        return response()->json([
            'data' => $users,
            'count' => $count,
        ]);
    }

    public function checkStatus(Request $request, User $user)
    {
        $isOnline = $this->onlineStatus->isOnline($user);

        return response()->json([
            'user_id' => $user->id,
            'online' => $isOnline,
            'status' => $isOnline ? 'online' : 'offline',
        ]);
    }
}
```

### Frontend - Heartbeat Service
```javascript
// resources/js/services/heartbeat-service.js
import axios from 'axios';

class HeartbeatService {
    constructor() {
        this.interval = null;
        this.frequency = 60000;
        this.idleTimeout = null;
        this.isIdle = false;
    }

    start() {
        this.sendHeartbeat();

        this.interval = setInterval(() => {
            if (!this.isIdle) {
                this.sendHeartbeat();
            }
        }, this.frequency);

        this.setupIdleDetection();
    }

    stop() {
        if (this.interval) {
            clearInterval(this.interval);
            this.interval = null;
        }
        if (this.idleTimeout) {
            clearTimeout(this.idleTimeout);
        }
        document.removeEventListener('mousemove', this.resetIdle);
        document.removeEventListener('keydown', this.resetIdle);
        document.removeEventListener('scroll', this.resetIdle);
    }

    setupIdleDetection() {
        const resetIdle = () => {
            this.isIdle = false;
            clearTimeout(this.idleTimeout);

            if (this.lastStatus === 'idle') {
                this.sendHeartbeat();
            }

            this.idleTimeout = setTimeout(() => {
                this.isIdle = true;
                this.sendIdleStatus();
            }, 300000);
        };

        this.resetIdle = resetIdle;

        document.addEventListener('mousemove', resetIdle);
        document.addEventListener('keydown', resetIdle);
        document.addEventListener('scroll', resetIdle);
        document.addEventListener('click', resetIdle);

        this.idleTimeout = setTimeout(() => {
            this.isIdle = true;
            this.sendIdleStatus();
        }, 300000);
    }

    async sendHeartbeat() {
        try {
            await axios.post('/status/heartbeat');
            this.lastStatus = 'online';
        } catch (e) {
            console.error('Heartbeat failed', e);
        }
    }

    async sendIdleStatus() {
        try {
            await axios.post('/status/heartbeat', { status: 'idle' });
            this.lastStatus = 'idle';
        } catch (e) {
            console.error('Idle status failed', e);
        }
    }
}

export const heartbeatService = new HeartbeatService();
```

### Online Status List Component
```vue
<template>
    <div class="bg-white rounded-lg shadow p-4">
        <div class="flex items-center justify-between mb-4">
            <h3 class="font-semibold flex items-center gap-2">
                <span>Online</span>
                <span class="bg-green-100 text-green-800 text-xs px-2 py-0.5 rounded-full">
                    {{ onlineUsers.length }}
                </span>
            </h3>
            <button @click="showAll = !showAll" class="text-xs text-blue-600">
                {{ showAll ? 'Online saja' : 'Semua' }}
            </button>
        </div>

        <div class="space-y-1">
            <div v-for="user in filteredUsers" :key="user.id">
                <OnlineStatusBadge :user="user" :status="getStatus(user.id)" @select="openChat" />
            </div>
        </div>

        <div v-if="filteredUsers.length === 0" class="text-center py-6 text-gray-400 text-sm">
            <p>Tidak ada user online</p>
        </div>
    </div>
</template>

<script setup>
import { ref, computed, onMounted, onUnmounted } from 'vue';
import echo from '../echo';
import OnlineStatusBadge from './OnlineStatusBadge.vue';

const onlineUsers = ref([]);
const allUsers = ref([]);
const showAll = ref(false);

const filteredUsers = computed(() => {
    return showAll.value ? allUsers.value : onlineUsers.value;
});

function getStatus(userId) {
    const user = onlineUsers.value.find(u => u.id === userId);
    return user?.status || 'offline';
}

let channel = null;

onMounted(() => {
    channel = echo.join('online');

    channel.here((users) => {
        onlineUsers.value = users;
    });

    channel.joining((user) => {
        if (!onlineUsers.value.find(u => u.id === user.id)) {
            onlineUsers.value.push(user);
        }
    });

    channel.leaving((user) => {
        onlineUsers.value = onlineUsers.value.filter(u => u.id !== user.id);
    });

    channel.listen('.status.changed', (data) => {
        const user = onlineUsers.value.find(u => u.id === data.user_id);
        if (user) {
            user.status = data.status;
        }
    });

    loadAllUsers();
});

onUnmounted(() => {
    if (channel) echo.leave('online');
});

async function loadAllUsers() {
    try {
        const { data } = await axios.get('/api/users');
        allUsers.value = data.data;
    } catch (e) {
        console.error(e);
    }
}

function openChat(user) {
    // navigate to chat
}
</script>
```
