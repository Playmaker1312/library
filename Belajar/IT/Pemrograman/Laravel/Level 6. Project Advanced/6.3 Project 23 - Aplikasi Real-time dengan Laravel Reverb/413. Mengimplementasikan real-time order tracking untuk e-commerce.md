# 413. Mengimplementasikan Real-time Order Tracking untuk E-commerce

## Definisi & Target
Membangun fitur real-time order tracking untuk e-commerce dimana user bisa melihat status pesanan mereka berubah secara live. Setiap perubahan status (pending, confirmed, processing, shipped, delivered) akan di-broadcast ke user secara real-time.

## Komponen yang Terlibat
- Order model dengan status workflow
- OrderStatusUpdated event
- Private channel per order
- Tracking timeline component
- Map integration for shipping
- Push notification
- Estimated delivery time

## Perintah Artisan CLI
```bash
# Membuat event
php artisan make:event OrderStatusUpdated

# Membuat model untuk tracking
php artisan make:model OrderTracking -m

# Membuat controller
php artisan make:controller OrderTrackingController

# Membuat notification
php artisan make:notification OrderShipped
```

### Event - OrderStatusUpdated
```php
<?php
namespace App\Events;

use App\Models\Order;
use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Broadcasting\PrivateChannel;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class OrderStatusUpdated implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public Order $order,
        public string $oldStatus,
        public string $newStatus
    ) {}

    public function broadcastOn(): PrivateChannel
    {
        return new PrivateChannel('order.' . $this->order->id);
    }

    public function broadcastAs(): string
    {
        return 'order.status.updated';
    }

    public function broadcastWith(): array
    {
        return [
            'order_id' => $this->order->id,
            'order_number' => $this->order->order_number,
            'old_status' => $this->oldStatus,
            'new_status' => $this->newStatus,
            'status_label' => $this->order->status_label,
            'tracking' => $this->getLatestTracking(),
            'timestamp' => now()->toISO8601String(),
            'estimated_delivery' => $this->order->estimated_delivery?->toISO8601String(),
        ];
    }

    private function getLatestTracking(): ?array
    {
        $tracking = $this->order->trackings()->latest()->first();

        return $tracking ? [
            'status' => $tracking->status,
            'location' => $tracking->location,
            'description' => $tracking->description,
            'timestamp' => $tracking->timestamp->toISO8601String(),
        ] : null;
    }
}
```

### Controller - OrderTrackingController
```php
<?php
namespace App\Http\Controllers;

use App\Events\OrderStatusUpdated;
use App\Models\Order;
use App\Models\OrderTracking;
use Illuminate\Http\Request;

class OrderTrackingController extends Controller
{
    public function show(Order $order)
    {
        if ($order->user_id !== auth()->id() && !auth()->user()->is_admin) {
            abort(403);
        }

        $order->load('trackings', 'items.product');

        return view('orders.tracking', compact('order'));
    }

    public function updateStatus(Request $request, Order $order)
    {
        $validated = $request->validate([
            'status' => 'required|string|in:pending,confirmed,processing,shipped,delivered,cancelled',
            'location' => 'nullable|string|max:255',
            'description' => 'nullable|string|max:500',
            'metadata' => 'nullable|array',
        ]);

        $oldStatus = $order->status;
        $order->update(['status' => $validated['status']]);

        $tracking = OrderTracking::create([
            'order_id' => $order->id,
            'status' => $validated['status'],
            'location' => $validated['location'] ?? null,
            'description' => $validated['description'] ?? null,
            'metadata' => $validated['metadata'] ?? null,
        ]);

        broadcast(new OrderStatusUpdated($order, $oldStatus, $validated['status']));

        if ($validated['status'] === 'shipped') {
            $order->user->notify(new \App\Notifications\OrderShipped($order));
        }

        return response()->json([
            'message' => 'Status updated.',
            'tracking' => $tracking,
        ]);
    }

    public function getTracking(Order $order)
    {
        if ($order->user_id !== auth()->id() && !auth()->user()->is_admin) {
            abort(403);
        }

        $trackings = $order->trackings()->latest()->get();

        return response()->json([
            'data' => $trackings,
            'current_status' => $order->status,
            'estimated_delivery' => $order->estimated_delivery,
        ]);
    }

    public function timeline(Order $order)
    {
        if ($order->user_id !== auth()->id() && !auth()->user()->is_admin) {
            abort(403);
        }

        $timeline = collect(config('orders.statuses'))->map(function ($label, $status) use ($order) {
            $tracking = $order->trackings()->where('status', $status)->first();

            return [
                'status' => $status,
                'label' => $label,
                'completed' => $tracking !== null || $order->status === $status,
                'active' => $order->status === $status,
                'timestamp' => $tracking?->timestamp,
                'location' => $tracking?->location,
                'description' => $tracking?->description,
            ];
        });

        return response()->json(['data' => $timeline]);
    }
}
```

### Vue Component - Tracking Timeline
```vue
<template>
    <div class="max-w-2xl mx-auto">
        <div class="bg-white rounded-xl shadow-sm p-6 mb-6">
            <div class="flex items-center justify-between mb-4">
                <div>
                    <h2 class="text-lg font-semibold">Pesanan #{{ order.order_number }}</h2>
                    <p class="text-sm text-gray-500">
                        Estimasi sampai: {{ estimatedDelivery }}
                    </p>
                </div>
                <span :class="['px-3 py-1 rounded-full text-sm font-medium', statusClass]">
                    {{ order.status_label }}
                </span>
            </div>

            <div class="relative">
                <div v-for="(step, index) in timeline" :key="step.status"
                     class="flex items-start gap-4 pb-8 relative">
                    <div class="flex flex-col items-center">
                        <div :class="['w-10 h-10 rounded-full flex items-center justify-center text-sm font-bold',
                            step.active ? 'bg-blue-600 text-white ring-4 ring-blue-100' :
                            step.completed ? 'bg-green-500 text-white' : 'bg-gray-200 text-gray-400']">
                            <span v-if="step.completed">&#10003;</span>
                            <span v-else>{{ index + 1 }}</span>
                        </div>
                        <div v-if="index < timeline.length - 1"
                             :class="['w-0.5 h-full absolute top-10 left-5',
                             step.completed ? 'bg-green-500' : 'bg-gray-200']">
                        </div>
                    </div>
                    <div class="flex-1 pt-1.5">
                        <h4 :class="['font-medium', step.active ? 'text-blue-600' : step.completed ? 'text-green-600' : 'text-gray-400']">
                            {{ step.label }}
                        </h4>
                        <p v-if="step.description" class="text-sm text-gray-500 mt-1">{{ step.description }}</p>
                        <p v-if="step.location" class="text-sm text-gray-400 mt-0.5">📍 {{ step.location }}</p>
                        <p v-if="step.timestamp" class="text-xs text-gray-400 mt-0.5">
                            {{ formatDate(step.timestamp) }}
                        </p>
                    </div>
                </div>
            </div>
        </div>

        <!-- Live Map (when shipped) -->
        <div v-if="currentStatus === 'shipped'" class="bg-white rounded-xl shadow-sm p-6">
            <h3 class="font-semibold mb-4">Lacak Pengiriman</h3>
            <div id="tracking-map" class="h-64 bg-gray-100 rounded-lg">
                <!-- Map integration here -->
            </div>
        </div>
    </div>
</template>

<script setup>
import { ref, computed, onMounted, onUnmounted } from 'vue';
import echo from '../echo';
import axios from 'axios';

const props = defineProps({ orderId: Number });

const order = ref({});
const timeline = ref([]);
const currentStatus = ref('');

const estimatedDelivery = computed(() => {
    return order.value.estimated_delivery
        ? new Date(order.value.estimated_delivery).toLocaleDateString('id-ID', {
            weekday: 'long', day: 'numeric', month: 'long'
          })
        : 'Belum tersedia';
});

const statusClass = computed(() => {
    const colors = {
        pending: 'bg-yellow-100 text-yellow-800',
        confirmed: 'bg-blue-100 text-blue-800',
        processing: 'bg-indigo-100 text-indigo-800',
        shipped: 'bg-purple-100 text-purple-800',
        delivered: 'bg-green-100 text-green-800',
        cancelled: 'bg-red-100 text-red-800',
    };
    return colors[order.value.status] || 'bg-gray-100';
});

onMounted(async () => {
    await loadData();

    echo.private(`order.${props.orderId}`)
        .listen('.order.status.updated', (data) => {
            order.value.status = data.new_status;
            currentStatus.value = data.new_status;
            loadTimeline();
        });
});

async function loadData() {
    try {
        const [orderRes, timelineRes] = await Promise.all([
            axios.get(`/api/orders/${props.orderId}`),
            axios.get(`/orders/${props.orderId}/tracking/timeline`),
        ]);
        order.value = orderRes.data.data;
        currentStatus.value = order.value.status;
        timeline.value = timelineRes.data.data;
    } catch (e) {
        console.error(e);
    }
}

async function loadTimeline() {
    try {
        const { data } = await axios.get(`/orders/${props.orderId}/tracking/timeline`);
        timeline.value = data.data;
    } catch (e) {
        console.error(e);
    }
}

function formatDate(date) {
    return new Date(date).toLocaleDateString('id-ID', {
        day: 'numeric', month: 'long', year: 'numeric',
        hour: '2-digit', minute: '2-digit',
    });
}
</script>
```

### Artisan Command - Simulate Order Tracking
```php
<?php
namespace App\Console\Commands;

use App\Events\OrderStatusUpdated;
use App\Models\Order;
use App\Models\OrderTracking;
use Illuminate\Console\Command;

class SimulateOrderTracking extends Command
{
    protected $signature = 'order:simulate-tracking {order}';
    protected $description = 'Simulate order status changes';

    public function handle(): int
    {
        $order = Order::findOrFail($this->argument('order'));

        $statuses = ['confirmed', 'processing', 'shipped', 'delivered'];
        $locations = ['Gudang A', 'Sorting Center', 'Kota Tujuan', 'Alamat Pengiriman'];

        foreach ($statuses as $i => $status) {
            $this->info("Updating to: {$status}");
            sleep(3);

            $oldStatus = $order->status;
            $order->update(['status' => $status]);

            OrderTracking::create([
                'order_id' => $order->id,
                'status' => $status,
                'location' => $locations[$i] ?? null,
                'description' => "Status berubah menjadi {$status}",
            ]);

            OrderStatusUpdated::dispatch($order, $oldStatus, $status);
        }

        $this->info('Tracking simulation completed.');

        return Command::SUCCESS;
    }
}
```
