# 411. Mengimplementasikan Real-time Collaborative Editing Sederhana

## Definisi & Target
Membangun fitur collaborative editing sederhana dimana multiple user bisa mengedit dokumen yang sama secara real-time. Menggunakan Laravel Reverb untuk broadcast perubahan dan Operational Transformation dasar untuk konsistensi.

## Komponen yang Terlibat
- Document model
- Cursors position tracking
- Edit operations broadcast
- Conflict resolution (OT basic)
- Presence channel untuk co-editors
- Auto-save mechanism

## Perintah Artisan CLI
```bash
# Membuat model
php artisan make:model Document -m
php artisan make:model DocumentEdit -m

# Membuat event
php artisan make:event DocumentUpdated
php artisan make:event CursorMoved

# Membuat controller
php artisan make:controller CollaborativeController
```

### Event - DocumentUpdated
```php
<?php
namespace App\Events;

use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Broadcasting\PresenceChannel;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class DocumentUpdated implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public int $documentId,
        public array $operation,
        public int $userId,
        public int $version
    ) {}

    public function broadcastOn(): PresenceChannel
    {
        return new PresenceChannel('document.' . $this->documentId);
    }

    public function broadcastAs(): string
    {
        return 'document.updated';
    }

    public function broadcastWith(): array
    {
        return [
            'operation' => $this->operation,
            'user_id' => $this->userId,
            'version' => $this->version,
            'timestamp' => now()->toISO8601String(),
        ];
    }
}
```

### Helper - OperationalTransform
```php
<?php
namespace App\Helpers;

class OperationalTransform
{
    public static function transform(array $operation, array $against): array
    {
        $action = $operation['action'] ?? 'insert';
        $position = $operation['position'];
        $text = $operation['text'] ?? '';
        $length = $operation['length'] ?? strlen($text);

        $againstAction = $against['action'] ?? '';
        $againstPosition = $against['position'];
        $againstLength = $against['length'] ?? strlen($against['text'] ?? '');

        if ($againstAction === 'insert' && $againstPosition <= $position) {
            $operation['position'] += $againstLength;
        }

        if ($againstAction === 'delete') {
            $delStart = $againstPosition;
            $delEnd = $againstPosition + $againstLength;

            if ($position >= $delEnd) {
                $operation['position'] -= $againstLength;
            } elseif ($position > $delStart && $position < $delEnd) {
                $operation['position'] = $delStart;
            }
        }

        return $operation;
    }

    public static function apply(string $content, array $operation): string
    {
        $action = $operation['action'];
        $position = $operation['position'];
        $text = $operation['text'] ?? '';
        $length = $operation['length'] ?? strlen($text);

        return match ($action) {
            'insert' => substr_replace($content, $text, $position, 0),
            'delete' => substr_replace($content, '', $position, $length),
            'replace' => substr_replace($content, $text, $position, $length),
            default => $content,
        };
    }
}
```

### Channel Auth - routes/channels.php
```php
<?php
use App\Models\Document;
use Illuminate\Support\Facades\Broadcast;

Broadcast::channel('document.{documentId}', function ($user, int $documentId) {
    $document = Document::find($documentId);

    if (!$document) {
        return false;
    }

    $isCollaborator = $document->collaborators()->where('user_id', $user->id)->exists();
    $isOwner = $document->user_id === $user->id;

    if (!$isOwner && !$isCollaborator) {
        return false;
    }

    return [
        'id' => $user->id,
        'name' => $user->name,
        'avatar' => $user->avatar_url,
        'color' => $document->collaborators()
            ->where('user_id', $user->id)
            ->value('color') ?? '#3B82F6',
    ];
});
```

### Controller - CollaborativeController
```php
<?php
namespace App\Http\Controllers;

use App\Events\CursorMoved;
use App\Events\DocumentUpdated;
use App\Helpers\OperationalTransform;
use App\Models\Document;
use Illuminate\Http\Request;

class CollaborativeController extends Controller
{
    public function index()
    {
        $documents = Document::where('user_id', auth()->id())
            ->orWhereHas('collaborators', fn($q) => $q->where('user_id', auth()->id()))
            ->with('user')
            ->latest()
            ->get();

        return view('documents.index', compact('documents'));
    }

    public function show(Document $document)
    {
        if ($document->user_id !== auth()->id() &&
            !$document->collaborators()->where('user_id', auth()->id())->exists()) {
            abort(403);
        }

        $document->load('user', 'collaborators.user');

        return view('documents.edit', compact('document'));
    }

    public function update(Request $request, Document $document)
    {
        $validated = $request->validate([
            'operation' => 'required|array',
            'operation.action' => 'required|in:insert,delete,replace',
            'operation.position' => 'required|integer|min:0',
            'operation.text' => 'nullable|string',
            'operation.length' => 'nullable|integer',
            'version' => 'required|integer',
        ]);

        $operation = $validated['operation'];
        $newVersion = $validated['version'];

        $recentEdits = $document->edits()
            ->where('version', '>=', $document->version)
            ->orderBy('version')
            ->get();

        foreach ($recentEdits as $edit) {
            $operation = OperationalTransform::transform(
                $operation,
                ['action' => $edit->action, 'position' => $edit->position, 'text' => $edit->text, 'length' => $edit->length]
            );
        }

        $document->content = OperationalTransform::apply($document->content, $operation);
        $document->version = $newVersion;
        $document->save();

        $document->edits()->create([
            'user_id' => auth()->id(),
            'action' => $operation['action'],
            'position' => $operation['position'],
            'text' => $operation['text'] ?? null,
            'length' => $operation['length'] ?? null,
            'version' => $newVersion,
        ]);

        broadcast(new DocumentUpdated(
            $document->id,
            $operation,
            auth()->id(),
            $newVersion
        ))->toOthers();

        return response()->json(['version' => $newVersion]);
    }

    public function cursor(Request $request, Document $document)
    {
        $validated = $request->validate([
            'position' => 'required|integer|min:0',
            'selection_start' => 'nullable|integer',
            'selection_end' => 'nullable|integer',
        ]);

        broadcast(new CursorMoved(
            $document->id,
            auth()->id(),
            auth()->user()->name,
            $validated['position'],
            $validated['selection_start'] ?? null,
            $validated['selection_end'] ?? null
        ))->toOthers();

        return response()->json(['status' => 'ok']);
    }

    public function collaborators(Request $request, Document $document)
    {
        $request->validate(['email' => 'required|email|exists:users,email']);

        $user = \App\Models\User::where('email', $request->email)->first();

        if ($document->user_id === $user->id) {
            return response()->json(['message' => 'Pemilik dokumen.'], 422);
        }

        $document->collaborators()->firstOrCreate(
            ['user_id' => $user->id],
            ['color' => sprintf('#%06X', mt_rand(0, 0xFFFFFF))]
        );

        return response()->json(['message' => 'Collaborator added.']);
    }
}
```

### Frontend - Collaborative Editor (Vanilla JS)
```javascript
// resources/js/collaborative-editor.js
import echo from './echo';

class CollaborativeEditor {
    constructor(documentId, userId, textarea) {
        this.documentId = documentId;
        this.userId = userId;
        this.textarea = textarea;
        this.version = 1;
        this.remoteCursors = new Map();
        this.isRemoteUpdate = false;

        this.setupEcho();
        this.setupTextarea();
    }

    setupEcho() {
        this.channel = echo.join(`document.${this.documentId}`);

        this.channel.here((users) => {
            this.updateCollaborators(users);
        });

        this.channel.joining((user) => {
            this.showToast(`${user.name} bergabung`);
        });

        this.channel.leaving((user) => {
            this.removeCursor(user.id);
            this.showToast(`${user.name} keluar`);
        });

        this.channel.listen('.document.updated', (data) => {
            if (data.user_id !== this.userId) {
                this.applyRemoteOperation(data.operation);
                this.version = data.version;
            }
        });

        this.channel.listen('.cursor.moved', (data) => {
            if (data.user_id !== this.userId) {
                this.updateRemoteCursor(data);
            }
        });
    }

    setupTextarea() {
        this.textarea.addEventListener('input', (e) => {
            if (this.isRemoteUpdate) return;

            const operation = {
                action: 'insert',
                position: e.target.selectionStart,
                text: this.getNewText(e),
            };

            this.sendOperation(operation);
        });

        this.textarea.addEventListener('keyup', (e) => {
            if (e.key === 'Backspace' || e.key === 'Delete') {
                this.sendOperation({
                    action: 'delete',
                    position: e.target.selectionStart,
                    length: 1,
                });
            }
        });

        this.textarea.addEventListener('click', () => this.broadcastCursor());
        this.textarea.addEventListener('keyup', () => this.broadcastCursor());
    }

    sendOperation(operation) {
        this.version++;

        axios.put(`/documents/${this.documentId}`, {
            operation,
            version: this.version,
        });
    }

    applyRemoteOperation(operation) {
        this.isRemoteUpdate = true;
        const content = this.textarea.value;
        let newContent;

        switch (operation.action) {
            case 'insert':
                newContent = content.substring(0, operation.position) +
                             operation.text +
                             content.substring(operation.position);
                break;
            case 'delete':
                newContent = content.substring(0, operation.position) +
                             content.substring(operation.position + operation.length);
                break;
        }

        this.textarea.value = newContent;
        this.isRemoteUpdate = false;
    }

    broadcastCursor() {
        axios.post(`/documents/${this.documentId}/cursor`, {
            position: this.textarea.selectionStart,
        });
    }

    updateRemoteCursor(data) {
        let cursor = this.remoteCursors.get(data.user_id);
        if (!cursor) {
            cursor = document.createElement('div');
            cursor.className = 'remote-cursor';
            cursor.innerHTML = `<span class="cursor-label">${data.user_name}</span>`;
            this.textarea.parentElement.appendChild(cursor);
            this.remoteCursors.set(data.user_id, cursor);
        }

        const charWidth = 8;
        const lineHeight = 20;
        const text = this.textarea.value;
        const lines = text.substring(0, data.position).split('\n');
        const line = lines.length - 1;
        const col = lines[lines.length - 1].length;

        cursor.style.left = `${col * charWidth}px`;
        cursor.style.top = `${line * lineHeight}px`;
    }

    removeCursor(userId) {
        const cursor = this.remoteCursors.get(userId);
        if (cursor) {
            cursor.remove();
            this.remoteCursors.delete(userId);
        }
    }

    updateCollaborators(users) {
        console.log('Active collaborators:', users);
    }

    showToast(message) {
        const toast = document.createElement('div');
        toast.className = 'fixed bottom-4 right-4 bg-gray-800 text-white px-4 py-2 rounded-lg shadow-lg text-sm';
        toast.textContent = message;
        document.body.appendChild(toast);
        setTimeout(() => toast.remove(), 3000);
    }
}

export default CollaborativeEditor;
```
