# 414. Mengimplementasikan Live Auction Bidding Menggunakan Reverb

## Definisi & Target
Membangun sistem live auction bidding dimana peserta bisa melihat bid terbaru secara real-time, melakukan bid, dan melihat countdown timer. Menggunakan presence channel untuk tracking peserta auction dan broadcast untuk update bid.

## Komponen yang Terlibat
- Auction model
- Bid model
- Presence channel untuk peserta
- Private channel untuk bid updates
- Countdown timer
- Auto bid increment
- Bid validation (higher bid required)
- Winner determination

### Event - NewBidPlaced
```php
<?php
namespace App\Events;

use App\Models\Auction;
use App\Models\Bid;
use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Broadcasting\PresenceChannel;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class NewBidPlaced implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public Auction $auction,
        public Bid $bid
    ) {}

    public function broadcastOn(): PresenceChannel
    {
        return new PresenceChannel('auction.' . $this->auction->id);
    }

    public function broadcastAs(): string
    {
        return 'bid.placed';
    }

    public function broadcastWith(): array
    {
        return [
            'auction_id' => $this->auction->id,
            'bid_id' => $this->bid->id,
            'amount' => (float) $this->bid->amount,
            'formatted_amount' => 'Rp ' . number_format($this->bid->amount, 0, ',', '.'),
            'user_id' => $this->bid->user_id,
            'user_name' => $this->bid->user->name,
            'bid_count' => $this->auction->bid_count,
            'current_price' => (float) $this->auction->current_price,
            'time_left' => $this->auction->ends_at->diffInSeconds(now()),
            'timestamp' => now()->toISO8601String(),
        ];
    }
}
```

### Model - Auction
```php
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\Relations\BelongsTo;

class Auction extends Model
{
    protected $fillable = [
        'title', 'description', 'starting_price', 'current_price',
        'reserve_price', 'bid_increment', 'seller_id', 'winner_id',
        'status', 'starts_at', 'ends_at', 'bid_count', 'bidder_count', 'metadata',
    ];

    protected $casts = [
        'starting_price' => 'decimal:2',
        'current_price' => 'decimal:2',
        'reserve_price' => 'decimal:2',
        'bid_increment' => 'decimal:2',
        'starts_at' => 'datetime',
        'ends_at' => 'datetime',
        'metadata' => 'array',
    ];

    public function bids(): HasMany
    {
        return $this->hasMany(Bid::class);
    }

    public function seller(): BelongsTo
    {
        return $this->belongsTo(User::class, 'seller_id');
    }

    public function winner(): BelongsTo
    {
        return $this->belongsTo(User::class, 'winner_id');
    }

    public function highestBid(): ?Bid
    {
        return $this->bids()->latestOfMany();
    }

    public function isActive(): bool
    {
        return $this->status === 'active'
            && $this->starts_at <= now()
            && $this->ends_at > now();
    }

    public function timeLeft(): int
    {
        return max(0, now()->diffInSeconds($this->ends_at, false));
    }

    public function placeBid(User $user, float $amount, bool $isAutoBid = false, ?float $maxAutoBid = null): Bid
    {
        $bid = $this->bids()->create([
            'user_id' => $user->id,
            'amount' => $amount,
            'is_auto_bid' => $isAutoBid,
            'max_auto_bid' => $maxAutoBid,
        ]);

        $this->increment('bid_count');
        $this->update(['current_price' => $amount]);

        $bidderIds = $this->bids()->distinct()->pluck('user_id');
        $this->update(['bidder_count' => $bidderIds->count()]);

        return $bid;
    }
}
```

### Controller - AuctionController
```php
<?php
namespace App\Http\Controllers;

use App\Events\AuctionStatusChanged;
use App\Events\NewBidPlaced;
use App\Models\Auction;
use App\Models\Bid;
use Illuminate\Http\Request;

class AuctionController extends Controller
{
    public function index()
    {
        $auctions = Auction::where('status', 'active')
            ->where('ends_at', '>', now())
            ->withCount('bids')
            ->latest()
            ->paginate(12);

        return view('auctions.index', compact('auctions'));
    }

    public function show(Auction $auction)
    {
        $auction->load(['bids.user', 'seller']);

        return view('auctions.show', compact('auction'));
    }

    public function bid(Request $request, Auction $auction)
    {
        $validated = $request->validate([
            'amount' => 'required|numeric|min:0',
            'is_auto_bid' => 'boolean',
            'max_auto_bid' => 'nullable|numeric|min:0',
        ]);

        if (!$auction->isActive()) {
            return response()->json(['message' => 'Lelang tidak aktif.'], 422);
        }

        $minBid = $auction->current_price + $auction->bid_increment;

        if ($validated['amount'] < $minBid) {
            return response()->json([
                'message' => "Bid minimal: Rp " . number_format($minBid, 0, ',', '.'),
                'minimum_bid' => $minBid,
            ], 422);
        }

        $user = $request->user();

        if ($auction->highestBid()?->user_id === $user->id) {
            return response()->json(['message' => 'Anda sudah menjadi bidder tertinggi.'], 422);
        }

        $bid = $auction->placeBid(
            $user,
            $validated['amount'],
            $validated['is_auto_bid'] ?? false,
            $validated['max_auto_bid'] ?? null
        );

        $bid->load('user');

        broadcast(new NewBidPlaced($auction, $bid));

        return response()->json([
            'message' => 'Bid berhasil!',
            'bid' => $bid,
            'current_price' => $auction->fresh()->current_price,
        ], 201);
    }

    public function participants(Auction $auction)
    {
        $bidders = $auction->bids()
            ->with('user')
            ->select('user_id')
            ->distinct()
            ->get()
            ->pluck('user');

        return response()->json(['data' => $bidders]);
    }
}
```

### Vue Component - Live Auction Room
```vue
<template>
    <div class="max-w-4xl mx-auto">
        <div class="bg-white rounded-xl shadow-sm p-6 mb-4">
            <div class="flex justify-between items-start">
                <div>
                    <h1 class="text-2xl font-bold">{{ auction.title }}</h1>
                    <p class="text-gray-500 mt-1">{{ auction.description }}</p>
                </div>
                <div class="text-right">
                    <div class="text-3xl font-bold text-green-600">
                        Rp {{ formatPrice(currentPrice) }}
                    </div>
                    <div class="text-sm text-gray-500">Harga Saat Ini</div>
                </div>
            </div>

            <div class="grid grid-cols-3 gap-4 mt-6">
                <div class="text-center p-3 bg-gray-50 rounded-lg">
                    <div class="text-2xl font-bold" :class="timeClass">{{ formattedTime }}</div>
                    <div class="text-xs text-gray-500">Sisa Waktu</div>
                </div>
                <div class="text-center p-3 bg-gray-50 rounded-lg">
                    <div class="text-2xl font-bold">{{ bidCount }}</div>
                    <div class="text-xs text-gray-500">Total Bid</div>
                </div>
                <div class="text-center p-3 bg-gray-50 rounded-lg">
                    <div class="text-2xl font-bold">{{ participantCount }}</div>
                    <div class="text-xs text-gray-500">Peserta</div>
                </div>
            </div>
        </div>

        <div class="grid md:grid-cols-3 gap-4">
            <div class="md:col-span-2">
                <div class="bg-white rounded-xl shadow-sm p-6">
                    <h3 class="font-semibold mb-4">Riwayat Bid</h3>
                    <div class="space-y-2 max-h-96 overflow-y-auto">
                        <div v-for="bid in bids" :key="bid.id"
                             :class="['flex justify-between items-center p-3 rounded-lg',
                             bid.user_id === currentUserId ? 'bg-blue-50 border border-blue-200' : 'bg-gray-50']">
                            <div>
                                <div class="font-medium text-sm">{{ bid.user.name }}</div>
                                <div class="text-xs text-gray-400">{{ formatDate(bid.created_at) }}</div>
                            </div>
                            <div class="font-bold text-green-600">
                                Rp {{ formatPrice(bid.amount) }}
                            </div>
                        </div>
                    </div>
                </div>
            </div>

            <div>
                <div class="bg-white rounded-xl shadow-sm p-6">
                    <h3 class="font-semibold mb-4">Pasang Bid</h3>

                    <div class="text-sm text-gray-500 mb-2">
                        Bid minimal: Rp {{ formatPrice(minBid) }}
                    </div>

                    <div class="flex gap-2 mb-3">
                        <button v-for="amount in quickAmounts" :key="amount"
                                @click="bidAmount = amount"
                                :class="['px-3 py-1 text-sm rounded border',
                                bidAmount === amount ? 'bg-blue-600 text-white border-blue-600' : 'hover:bg-gray-50']">
                            +{{ formatPrice(amount) }}
                        </button>
                    </div>

                    <input v-model="bidAmount" type="number"
                           class="w-full border rounded-lg px-4 py-2 mb-3"
                           :min="minBid" :placeholder="'Min: ' + minBid">

                    <button @click="placeBid"
                            :disabled="!canBid || loading"
                            class="w-full bg-blue-600 text-white py-3 rounded-lg font-semibold hover:bg-blue-700 disabled:opacity-50 disabled:cursor-not-allowed">
                        {{ loading ? 'Memproses...' : 'Pasang Bid' }}
                    </button>

                    <p v-if="error" class="text-red-500 text-sm mt-2">{{ error }}</p>
                </div>

                <div class="bg-white rounded-xl shadow-sm p-6 mt-4">
                    <h3 class="font-semibold mb-3">Peserta ({{ participantCount }})</h3>
                    <div v-for="p in participants" :key="p.id"
                         class="flex items-center gap-2 mb-2">
                        <span class="w-2 h-2 bg-green-500 rounded-full"></span>
                        <span class="text-sm">{{ p.name }}</span>
                    </div>
                </div>
            </div>
        </div>
    </div>
</template>

<script setup>
import { ref, computed, onMounted, onUnmounted } from 'vue';
import axios from 'axios';
import echo from '../echo';

const props = defineProps({ auctionId: Number, currentUserId: Number });

const auction = ref({});
const bids = ref([]);
const participants = ref([]);
const bidAmount = ref(0);
const loading = ref(false);
const error = ref('');
const timeLeft = ref(0);
const timerInterval = ref(null);

const currentPrice = computed(() => auction.value.current_price || 0);
const bidCount = computed(() => auction.value.bid_count || 0);
const participantCount = computed(() => participants.value.length);

const minBid = computed(() => {
    return (currentPrice.value || auction.value.starting_price || 0) + (auction.value.bid_increment || 1000);
});

const canBid = computed(() => {
    return auction.value.status === 'active' && bidAmount.value >= minBid.value;
});

const formattedTime = computed(() => {
    const t = timeLeft.value;
    if (t <= 0) return 'Selesai';
    const h = Math.floor(t / 3600);
    const m = Math.floor((t % 3600) / 60);
    const s = t % 60;
    return `${h.toString().padStart(2, '0')}:${m.toString().padStart(2, '0')}:${s.toString().padStart(2, '0')}`;
});

const timeClass = computed(() => {
    if (timeLeft.value <= 0) return 'text-red-600';
    if (timeLeft.value < 300) return 'text-orange-600 animate-pulse';
    return 'text-gray-800';
});

const quickAmounts = computed(() => {
    const inc = auction.value.bid_increment || 1000;
    return [inc, inc * 2, inc * 5, inc * 10];
});

let channel = null;

onMounted(async () => {
    await loadAuction();
    startTimer();

    channel = echo.join(`auction.${props.auctionId}`);

    channel.here((users) => { participants.value = users; });
    channel.joining((user) => {
        if (!participants.value.find(p => p.id === user.id)) {
            participants.value.push(user);
        }
    });
    channel.leaving((user) => {
        participants.value = participants.value.filter(p => p.id !== user.id);
    });

    channel.listen('.bid.placed', (data) => {
        auction.value.current_price = data.current_price;
        auction.value.bid_count = data.bid_count;
        bids.value.unshift({
            id: data.bid_id, user_id: data.user_id,
            user: { name: data.user_name },
            amount: data.amount, created_at: data.timestamp,
        });
        timeLeft.value = data.time_left;
    });

    channel.listen('.auction.status.changed', (data) => {
        auction.value.status = data.status;
        if (data.status === 'ended' || data.status === 'sold') {
            alert(`Lelang selesai! Pemenang: ${data.winner_name}`);
        }
    });
});

onUnmounted(() => {
    clearInterval(timerInterval.value);
    if (channel) echo.leave(`auction.${props.auctionId}`);
});

async function loadAuction() {
    try {
        const { data } = await axios.get(`/api/auctions/${props.auctionId}`);
        auction.value = data.data;
        bids.value = data.data.bids.reverse();
        timeLeft.value = Math.max(0, Math.floor((new Date(data.data.ends_at) - new Date()) / 1000));
    } catch (e) { console.error(e); }
}

function startTimer() {
    timerInterval.value = setInterval(() => {
        if (timeLeft.value > 0) {
            timeLeft.value--;
        }
    }, 1000);
}

async function placeBid() {
    if (!canBid.value) return;
    loading.value = true;
    error.value = '';

    try {
        const { data } = await axios.post(`/api/auctions/${props.auctionId}/bid`, {
            amount: bidAmount.value,
        });

        bidAmount.value = 0;
    } catch (e) {
        error.value = e.response?.data?.message || 'Gagal memasang bid.';
    } finally {
        loading.value = false;
    }
}

function formatPrice(value) {
    return Number(value).toLocaleString('id-ID');
}

function formatDate(date) {
    return new Date(date).toLocaleTimeString('id-ID', { hour: '2-digit', minute: '2-digit', second: '2-digit' });
}
</script>
```
