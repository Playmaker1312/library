# 406. Mengimplementasikan Presence Channel untuk Mengetahui Siapa yang Sedang Online

## Definisi & Target
Mengimplementasikan presence channel yang memungkinkan tracking user yang sedang online dalam suatu channel. Presence channel adalah private channel dengan kemampuan untuk melihat siapa saja yang sedang subscribe. Targetnya adalah fitur "user online" dan "who is typing" secara real-time.

## Komponen yang Terlibat
- PresenceChannel class
- Echo.join() method
- here(), joining(), leaving() callbacks
- User online status model
- Presence authorization
- Online/offline tracking

## Perintah Artisan CLI
```bash
# Membuat event untuk presence
php artisan make:event UserOnline
php artisan make:event UserOffline

# Membuat model untuk tracking
php artisan make:model PresenceLog -m
```

## Kode Implementasi (Per File)

### Channel Authorization - routes/channels.php
```php
<?php
use App\Models\ChatRoom;
use App\Models\User;
use Illuminate\Support\Facades\Broadcast;

// Presence channel untuk room chat
Broadcast::channel('chat.room.{roomId}', function (User $user, int $roomId) {
    $room = ChatRoom::find($roomId);

    if (!$room || !$room->members()->where('user_id', $user->id)->exists()) {
        return false;
    }

    return [
        'id' => $user->id,
        'name' => $user->name,
        'avatar' => $user->avatar_url,
        'role' => $user->role,
        'last_active' => $user->last_active_at?->diffForHumans(),
    ];
});

// Presence channel untuk online status global
Broadcast::channel('online', function (User $user) {
    return [
        'id' => $user->id,
        'name' => $user->name,
        'avatar' => $user->avatar_url,
        'status' => 'online',
    ];
});

// Presence channel untuk team workspace
Broadcast::channel('team.{teamId}', function (User $user, int $teamId) {
    if (!$user->teams()->where('team_id', $teamId)->exists()) {
        return false;
    }

    return [
        'id' => $user->id,
        'name' => $user->name,
        'avatar' => $user->avatar_url,
        'role' => $user->role,
    ];
});
```

### Event - PresenceChannelEvent
```php
<?php
namespace App\Events;

use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Broadcasting\PresenceChannel;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class UserTyping implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public int $roomId,
        public int $userId,
        public string $userName,
        public bool $typing
    ) {}

    public function broadcastOn(): PresenceChannel
    {
        return new PresenceChannel('chat.room.' . $this->roomId);
    }

    public function broadcastAs(): string
    {
        return 'user.typing';
    }

    public function broadcastWith(): array
    {
        return [
            'user_id' => $this->userId,
            'user_name' => $this->userName,
            'typing' => $this->typing,
        ];
    }
}
```

### Model - PresenceLog
```php
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class PresenceLog extends Model
{
    protected $fillable = [
        'user_id', 'channel', 'action', 'metadata', 'ip_address'
    ];

    protected $casts = [
        'metadata' => 'array',
    ];

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public static function logPresence(int $userId, string $channel, string $action, array $metadata = []): void
    {
        self::create([
            'user_id' => $userId,
            'channel' => $channel,
            'action' => $action,
            'metadata' => $metadata,
            'ip_address' => request()->ip(),
        ]);

        User::where('id', $userId)->update([
            'last_active_at' => now(),
            'is_online' => $action === 'join',
        ]);
    }
}
```

### Controller - PresenceController
```php
<?php
namespace App\Http\Controllers;

use App\Events\UserTyping;
use App\Models\PresenceLog;
use Illuminate\Http\Request;

class PresenceController extends Controller
{
    public function typing(Request $request)
    {
        $validated = $request->validate([
            'room_id' => 'required|integer',
            'typing' => 'required|boolean',
        ]);

        $user = $request->user();

        UserTyping::dispatch(
            $validated['room_id'],
            $user->id,
            $user->name,
            $validated['typing']
        );

        return response()->json(['status' => 'ok']);
    }

    public function onlineUsers()
    {
        $onlineUsers = User::where('is_online', true)
            ->where('last_active_at', '>', now()->subMinutes(5))
            ->get(['id', 'name', 'avatar_url', 'last_active_at']);

        return response()->json([
            'data' => $onlineUsers,
            'count' => $onlineUsers->count(),
        ]);
    }
}
```

### Frontend - Presence Channel Service
```javascript
// resources/js/services/presence-service.js
import echo from '../echo';

export class PresenceService {
    constructor() {
        this.channels = new Map();
        this.onlineUsers = new Map();
    }

    joinOnlineStatus(callbacks = {}) {
        const channel = echo.join('online');

        channel.here((users) => {
            users.forEach(user => this.onlineUsers.set(user.id, user));
            if (callbacks.onHere) callbacks.onUsers(Array.from(this.onlineUsers.values()));
        });

        channel.joining((user) => {
            this.onlineUsers.set(user.id, user);
            if (callbacks.onJoining) callbacks.onUserJoined(user);
        });

        channel.leaving((user) => {
            this.onlineUsers.delete(user.id);
            if (callbacks.onLeaving) callbacks.onUserLeft(user);
        });

        channel.listen('.user.typing', (data) => {
            if (callbacks.onTyping) callbacks.onTyping(data);
        });

        this.channels.set('online', channel);

        return () => {
            echo.leave('online');
            this.channels.delete('online');
        };
    }

    joinChatRoom(roomId, callbacks = {}) {
        const channel = echo.join(`chat.room.${roomId}`);

        channel.here((users) => {
            if (callbacks.onHere) callbacks.onUsers(users);
        });

        channel.joining((user) => {
            if (callbacks.onJoining) callbacks.onUserJoined(user);
        });

        channel.leaving((user) => {
            if (callbacks.onLeaving) callbacks.onUserLeft(user);
        });

        channel.listen('.user.typing', (data) => {
            if (callbacks.onTyping) callbacks.onTyping(data);
        });

        this.channels.set(`chat.room.${roomId}`, channel);

        return () => {
            echo.leave(`chat.room.${roomId}`);
            this.channels.delete(`chat.room.${roomId}`);
        };
    }

    getOnlineCount() {
        return this.onlineUsers.size;
    }

    isUserOnline(userId) {
        return this.onlineUsers.has(userId);
    }
}

export const presenceService = new PresenceService();
```

### Vue Component - Online Users List
```vue
<template>
    <div class="bg-white rounded-lg shadow p-4">
        <h3 class="font-semibold mb-4 flex items-center gap-2">
            <span class="w-2 h-2 bg-green-500 rounded-full inline-block"></span>
            Online ({{ onlineUsers.length }})
        </h3>

        <ul class="space-y-2">
            <li v-for="user in onlineUsers" :key="user.id"
                class="flex items-center gap-3 p-2 rounded-lg hover:bg-gray-50">
                <div class="relative">
                    <img :src="user.avatar || '/default-avatar.png'"
                         class="w-8 h-8 rounded-full">
                    <span class="absolute bottom-0 right-0 w-3 h-3 bg-green-500 border-2 border-white rounded-full"></span>
                </div>
                <div>
                    <div class="text-sm font-medium">{{ user.name }}</div>
                    <div class="text-xs text-green-600">Online</div>
                </div>
            </li>
        </ul>

        <div v-if="onlineUsers.length === 0" class="text-center text-gray-400 py-4 text-sm">
            Tidak ada user online
        </div>
    </div>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue';
import { presenceService } from '../services/presence-service';

const onlineUsers = ref([]);
let unsubscribe = null;

onMounted(() => {
    unsubscribe = presenceService.joinOnlineStatus({
        onUsers: (users) => { onlineUsers.value = users; },
        onUserJoined: (user) => {
            onlineUsers.value = [...onlineUsers.value, user];
        },
        onUserLeft: (user) => {
            onlineUsers.value = onlineUsers.value.filter(u => u.id !== user.id);
        },
    });
});

onUnmounted(() => {
    if (unsubscribe) unsubscribe();
});
</script>
```

### Vue Component - Typing Indicator
```vue
<template>
    <div class="text-sm text-gray-500 italic">
        <span v-if="typingUsers.length === 1">{{ typingUsers[0].name }} sedang mengetik...</span>
        <span v-else-if="typingUsers.length === 2">
            {{ typingUsers[0].name }} dan {{ typingUsers[1].name }} sedang mengetik...
        </span>
        <span v-else-if="typingUsers.length > 2">
            {{ typingUsers[0].name }} dan {{ typingUsers.length - 1 }} lainnya sedang mengetik...
        </span>
    </div>
</template>

<script setup>
import { ref, onMounted, onUnmounted } from 'vue';

const props = defineProps({ roomId: Number, currentUserId: Number });

const typingUsers = ref([]);
let typingTimeout = null;

onMounted(() => {
    window.Echo.join(`chat.room.${props.roomId}`)
        .listen('.user.typing', (data) => {
            if (data.user_id === props.currentUserId) return;

            if (data.typing) {
                if (!typingUsers.value.find(u => u.id === data.user_id)) {
                    typingUsers.value.push({ id: data.user_id, name: data.user_name });
                }
            } else {
                typingUsers.value = typingUsers.value.filter(u => u.id !== data.user_id);
            }

            clearTimeout(typingTimeout);
            typingTimeout = setTimeout(() => {
                typingUsers.value = [];
            }, 3000);
        });
});

onUnmounted(() => {
    clearTimeout(typingTimeout);
    window.Echo.leave(`chat.room.${props.roomId}`);
});

function emitTyping(isTyping) {
    axios.post('/presence/typing', {
        room_id: props.roomId,
        typing: isTyping,
    });
}

defineExpose({ emitTyping });
</script>
```
