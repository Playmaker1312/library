# 410. Mengimplementasikan Live Dashboard yang Update Data Tanpa Reload Menggunakan Livewire

## Definisi & Target
Membangun live dashboard yang menampilkan data real-time seperti statistik penjualan, user aktif, dan metrik lainnya menggunakan Laravel Livewire dengan Reverb broadcasting. Data akan update otomatis tanpa perlu refresh halaman.

## Komponen yang Terlibat
- Laravel Livewire
- Reverb broadcasting
- Livewire polling vs broadcast
- Dashboard widgets
- Real-time charts (Chart.js)
- Metrics calculation

## Perintah Artisan CLI
```bash
# Install Livewire
composer require livewire/livewire

# Membuat Livewire components
php artisan make:livewire Dashboard/StatsOverview
php artisan make:livewire Dashboard/RevenueChart
php artisan make:livewire Dashboard/RecentOrders
php artisan make:livewire Dashboard/OnlineUsers
php artisan make:livewire Dashboard/SalesMetrics

# Membuat event untuk broadcast
php artisan make:event DashboardUpdated
```

## Kode Implementasi (Per File)

### Event - DashboardUpdated
```php
<?php
namespace App\Events;

use Illuminate\Broadcasting\Channel;
use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class DashboardUpdated implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public array $metrics,
        public string $section = 'all'
    ) {}

    public function broadcastOn(): Channel
    {
        return new Channel('dashboard');
    }

    public function broadcastAs(): string
    {
        return 'dashboard.updated';
    }

    public function broadcastWith(): array
    {
        return [
            'section' => $this->section,
            'metrics' => $this->metrics,
            'timestamp' => now()->toISO8601String(),
        ];
    }
}
```

### Livewire Component - StatsOverview
```php
<?php
namespace App\Http\Livewire\Dashboard;

use App\Events\DashboardUpdated;
use Livewire\Component;
use Illuminate\Support\Facades\DB;

class StatsOverview extends Component
{
    public int $totalOrders = 0;
    public int $totalRevenue = 0;
    public int $totalUsers = 0;
    public int $pendingOrders = 0;
    public float $revenueGrowth = 0;
    public string $period = 'today';

    protected $listeners = ['echo:dashboard,DashboardUpdated' => 'refreshFromBroadcast'];

    protected function getListeners(): array
    {
        return [
            "echo:dashboard,DashboardUpdated" => 'refreshFromBroadcast',
        ];
    }

    public function mount(): void
    {
        $this->calculateStats();
    }

    public function refreshFromBroadcast(array $payload): void
    {
        if ($payload['section'] === 'stats' || $payload['section'] === 'all') {
            $this->totalOrders = $payload['metrics']['total_orders'] ?? $this->totalOrders;
            $this->totalRevenue = $payload['metrics']['total_revenue'] ?? $this->totalRevenue;
            $this->totalUsers = $payload['metrics']['total_users'] ?? $this->totalUsers;
            $this->pendingOrders = $payload['metrics']['pending_orders'] ?? $this->pendingOrders;
        }
    }

    public function calculateStats(): void
    {
        $this->totalOrders = DB::table('orders')
            ->when($this->period === 'today', fn($q) => $q->whereDate('created_at', today()))
            ->when($this->period === 'week', fn($q) => $q->whereBetween('created_at', [now()->startOfWeek(), now()]))
            ->when($this->period === 'month', fn($q) => $q->whereMonth('created_at', now()->month))
            ->count();

        $this->totalRevenue = DB::table('orders')
            ->when($this->period === 'today', fn($q) => $q->whereDate('created_at', today()))
            ->when($this->period === 'week', fn($q) => $q->whereBetween('created_at', [now()->startOfWeek(), now()]))
            ->when($this->period === 'month', fn($q) => $q->whereMonth('created_at', now()->month))
            ->sum('total');

        $this->totalUsers = DB::table('users')
            ->when($this->period === 'today', fn($q) => $q->whereDate('created_at', today()))
            ->count();

        $this->pendingOrders = DB::table('orders')
            ->where('status', 'pending')
            ->count();
    }

    public function updatedPeriod(): void
    {
        $this->calculateStats();
    }

    public function render()
    {
        return view('livewire.dashboard.stats-overview');
    }
}
```

### Livewire Component - RevenueChart
```php
<?php
namespace App\Http\Livewire\Dashboard;

use Livewire\Component;
use Illuminate\Support\Facades\DB;

class RevenueChart extends Component
{
    public array $chartData = [];
    public string $range = '7days';

    protected $listeners = ['echo:dashboard,DashboardUpdated' => 'refreshChart'];

    public function mount(): void
    {
        $this->loadChartData();
    }

    public function refreshChart(array $payload): void
    {
        if ($payload['section'] === 'revenue' || $payload['section'] === 'all') {
            $this->chartData = $payload['metrics']['chart_data'] ?? [];
        }
    }

    public function loadChartData(): void
    {
        $days = match ($this->range) {
            '7days' => 7,
            '30days' => 30,
            '90days' => 90,
            default => 7,
        };

        $data = DB::table('orders')
            ->select(
                DB::raw('DATE(created_at) as date'),
                DB::raw('SUM(total) as revenue'),
                DB::raw('COUNT(*) as orders')
            )
            ->where('created_at', '>=', now()->subDays($days))
            ->groupBy('date')
            ->orderBy('date')
            ->get();

        $this->chartData = [
            'labels' => $data->pluck('date')->toArray(),
            'revenue' => $data->pluck('revenue')->toArray(),
            'orders' => $data->pluck('orders')->toArray(),
        ];
    }

    public function render()
    {
        return view('livewire.dashboard.revenue-chart');
    }
}
```

### Livewire Component - RecentOrders
```php
<?php
namespace App\Http\Livewire\Dashboard;

use App\Models\Order;
use Livewire\Component;

class RecentOrders extends Component
{
    public array $orders = [];

    protected $listeners = ['echo:dashboard,DashboardUpdated' => 'refreshOrders'];

    public function mount(): void
    {
        $this->loadOrders();
    }

    public function refreshOrders(array $payload): void
    {
        if ($payload['section'] === 'orders' || $payload['section'] === 'all') {
            $newOrders = $payload['metrics']['orders'] ?? [];
            array_unshift($this->orders, ...$newOrders);
            $this->orders = array_slice($this->orders, 0, 10);
        }
    }

    public function loadOrders(): void
    {
        $this->orders = Order::with('user')
            ->latest()
            ->take(10)
            ->get()
            ->toArray();
    }

    public function render()
    {
        return view('livewire.dashboard.recent-orders');
    }
}
```

### Dashboard Controller - Broadcasting Updates
```php
<?php
namespace App\Http\Controllers;

use App\Events\DashboardUpdated;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;

class DashboardController extends Controller
{
    public function index()
    {
        return view('dashboard.live');
    }

    public function broadcastUpdate(Request $request)
    {
        $metrics = [
            'total_orders' => DB::table('orders')->count(),
            'total_revenue' => DB::table('orders')->sum('total'),
            'total_users' => DB::table('users')->count(),
            'pending_orders' => DB::table('orders')->where('status', 'pending')->count(),
        ];

        DashboardUpdated::dispatch($metrics, 'stats');

        return response()->json(['message' => 'Dashboard updated.']);
    }
}
```

### Artisan Command - Periodic Broadcast
```php
<?php
namespace App\Console\Commands;

use App\Events\DashboardUpdated;
use Illuminate\Console\Command;

class BroadcastDashboard extends Command
{
    protected $signature = 'dashboard:broadcast';
    protected $description = 'Broadcast dashboard updates';

    public function handle(): int
    {
        $metrics = [
            'total_orders' => \App\Models\Order::count(),
            'total_revenue' => \App\Models\Order::sum('total'),
            'total_users' => \App\Models\User::count(),
            'pending_orders' => \App\Models\Order::where('status', 'pending')->count(),
        ];

        DashboardUpdated::dispatch($metrics, 'stats');

        $this->info('Dashboard broadcasted.');

        return Command::SUCCESS;
    }
}
```

## Cara Integrasi & Pengujian
1. Install Livewire: composer require livewire/livewire
2. Buat Livewire components untuk setiap widget dashboard
3. Setup Echo listener di Livewire component
4. Broadcast event ketika ada data baru (order, user, dll)
5. Buka dashboard di browser dan verifikasi update real-time
6. Test dengan membuat order baru → widget terupdate
7. Test polling vs broadcast (broadcast lebih efisien)
8. Test chart real-time update
9. Test schedule untuk periodic broadcast
10. Monitor performa WebSocket connections
