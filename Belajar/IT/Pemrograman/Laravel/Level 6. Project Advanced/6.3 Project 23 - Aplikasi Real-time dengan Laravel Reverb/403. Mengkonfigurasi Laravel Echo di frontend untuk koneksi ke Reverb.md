# 403. Mengkonfigurasi Laravel Echo di Frontend untuk Koneksi ke Reverb

## Definisi & Target
Mengkonfigurasi Laravel Echo di frontend (JavaScript) untuk terhubung ke WebSocket server Reverb. Echo adalah JavaScript library yang menyediakan API sederhana untuk subscribe ke channel, listen event, dan handle real-time data.

## Komponen yang Terlibat
- Laravel Echo (npm package)
- Pusher JS (protocol implementation)
- Vite configuration
- Environment variables
- Channel subscription
- Event listener
- Connection state management

## Perintah Artisan CLI
```bash
# Install dependencies
npm install --save-dev laravel-echo pusher-js

# Build assets
npm run build
```

## Kode Implementasi (Per File)

### Echo Configuration - resources/js/echo.js
```javascript
import Echo from 'laravel-echo';
import Pusher from 'pusher-js';

window.Pusher = Pusher;

const echo = new Echo({
    broadcaster: 'reverb',
    key: import.meta.env.VITE_REVERB_APP_KEY,
    wsHost: import.meta.env.VITE_REVERB_HOST || window.location.hostname,
    wsPort: import.meta.env.VITE_REVERB_PORT ?? 8080,
    wssPort: import.meta.env.VITE_REVERB_PORT ?? 443,
    forceTLS: (import.meta.env.VITE_REVERB_SCHEME ?? 'https') === 'https',
    enabledTransports: ['ws', 'wss'],
    disableStats: true,
    authEndpoint: '/broadcasting/auth',
    auth: {
        headers: {
            'X-CSRF-TOKEN': (() => {
                const meta = document.querySelector('meta[name="csrf-token"]');
                return meta?.getAttribute('content') ?? '';
            })(),
        },
    },
});

echo.connector.pusher.connection.bind('state_change', (states) => {
    console.log(`WebSocket: ${states.previous} -> ${states.current}`);
});

export default echo;
```

### Echo Service - Helper Functions
```javascript
// resources/js/services/echo-service.js
import echo from '../echo';

export class EchoService {
    constructor() {
        this.channels = new Map();
        this.listeners = new Map();
    }

    subscribePublic(channelName, eventName, callback) {
        const channel = echo.channel(channelName);
        channel.listen(eventName, callback);

        const key = `${channelName}:${eventName}`;
        if (!this.channels.has(key)) {
            this.channels.set(key, channel);
            this.listeners.set(key, callback);
        }

        return () => this.unsubscribe(channelName, eventName);
    }

    subscribePrivate(channelName, eventName, callback) {
        const channel = echo.private(channelName);
        channel.listen(eventName, callback);

        const key = `private-${channelName}:${eventName}`;
        this.channels.set(key, channel);
        this.listeners.set(key, callback);

        return () => this.unsubscribe(channelName, eventName, true);
    }

    subscribePresence(channelName, events = {}) {
        const channel = echo.join(channelName);

        if (events.joined) {
            channel.here(events.joined);
        }
        if (events.joining) {
            channel.joining(events.joining);
        }
        if (events.leaving) {
            channel.leaving(events.leaving);
        }
        if (events.listening) {
            Object.entries(events.listening).forEach(([event, callback]) => {
                channel.listen(event, callback);
            });
        }

        return channel;
    }

    unsubscribe(channelName, eventName, isPrivate = false) {
        const prefix = isPrivate ? 'private-' : '';
        const key = `${prefix}${channelName}:${eventName}`;

        if (this.channels.has(key)) {
            const channel = this.channels.get(key);
            channel.stopListening(eventName);
            this.channels.delete(key);
            this.listeners.delete(key);
        }
    }

    leaveChannel(channelName) {
        echo.leave(channelName);
    }

    leavePrivateChannel(channelName) {
        echo.leave(`private-${channelName}`);
    }

    leavePresenceChannel(channelName) {
        echo.leave(`presence-${channelName}`);
    }

    disconnect() {
        echo.disconnect();
    }

    connect() {
        echo.connect();
    }

    isConnected() {
        return echo.connector.pusher.connection.state === 'connected';
    }

    getConnectionState() {
        return echo.connector.pusher.connection.state;
    }
}

export const echoService = new EchoService();
```

### React Hook - useEcho
```javascript
// resources/js/hooks/useEcho.js
import { useEffect, useRef, useCallback } from 'react';
import { echoService } from '../services/echo-service';

export function useEcho(channelName, eventName, callback, options = {}) {
    const callbackRef = useRef(callback);
    callbackRef.current = callback;

    useEffect(() => {
        const isPrivate = options.isPrivate ?? false;
        const unsubscribe = isPrivate
            ? echoService.subscribePrivate(channelName, eventName, (data) => {
                callbackRef.current(data);
            })
            : echoService.subscribePublic(channelName, eventName, (data) => {
                callbackRef.current(data);
            });

        return () => {
            if (unsubscribe) unsubscribe();
        };
    }, [channelName, eventName, options.isPrivate]);

    const isConnected = useCallback(() => echoService.isConnected(), []);

    return { isConnected };
}
```

### Vue Composables - useEcho
```javascript
// resources/js/composables/useEcho.js
import { ref, onMounted, onUnmounted } from 'vue';
import { echoService } from '../services/echo-service';

export function useEcho(channelName, eventName, options = {}) {
    const data = ref(null);
    const isConnected = ref(false);
    let unsubscribe = null;

    onMounted(() => {
        isConnected.value = echoService.isConnected();

        const isPrivate = options.isPrivate ?? false;
        unsubscribe = isPrivate
            ? echoService.subscribePrivate(channelName, eventName, (payload) => {
                data.value = payload;
                if (options.onEvent) options.onEvent(payload);
            })
            : echoService.subscribePublic(channelName, eventName, (payload) => {
                data.value = payload;
                if (options.onEvent) options.onEvent(payload);
            });
    });

    onUnmounted(() => {
        if (unsubscribe) unsubscribe();
    });

    return { data, isConnected };
}
```

## Cara Integrasi & Pengujian
1. Install frontend dependencies: npm install --save-dev laravel-echo pusher-js
2. Setup Echo di resources/js/bootstrap.js atau file terpisah
3. Pastikan VITE_REVERB_* environment variables ter-set
4. Build assets: npm run build
5. Buka halaman dan cek console log "Terhubung ke Reverb"
6. Test subscribe ke public channel dari browser console:
   ```javascript
   window.Echo.channel('test').listen('TestEvent', (e) => console.log(e));
   ```
7. Broadcast event dari backend: event(new App\Events\TestEvent('data'))
8. Verifikasi event diterima di browser
9. Test connection state change (disconnect/reconnect)
10. Test private channel authentication
