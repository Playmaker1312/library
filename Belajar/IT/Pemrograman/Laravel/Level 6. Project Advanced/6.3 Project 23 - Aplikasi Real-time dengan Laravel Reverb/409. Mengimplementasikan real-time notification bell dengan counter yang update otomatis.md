# 409. Mengimplementasikan Real-time Notification Bell dengan Counter yang Update Otomatis

## Definisi & Target
Membangun komponen notification bell dengan counter yang update otomatis secara real-time menggunakan Laravel Reverb. Counter akan bertambah ketika notifikasi baru masuk dan berkurang ketika user membaca notifikasi. Targetnya adalah UX notifikasi yang responsif.

## Komponen yang Terlibat
- Broadcast notification listener
- Live counter update
- Dropdown notification list
- Mark as read functionality
- Browser notification API
- Sound notification
- Vue/Livewire component

## Perintah Artisan CLI
```bash
# Membuat notification controller
```

### Controller - NotificationController
```php
<?php
namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Notifications\DatabaseNotification;

class NotificationController extends Controller
{
    public function index(Request $request)
    {
        $notifications = $request->user()
            ->notifications()
            ->latest()
            ->paginate(10);

        return response()->json([
            'data' => $notifications->items(),
            'meta' => [
                'total' => $notifications->total(),
                'has_more' => $notifications->hasMorePages(),
                'unread_count' => $request->user()->unreadNotifications()->count(),
            ],
        ]);
    }

    public function unreadCount(Request $request)
    {
        $count = $request->user()->unreadNotifications()->count();

        return response()->json(['count' => $count]);
    }

    public function markAsRead(Request $request, $id)
    {
        $notification = $request->user()
            ->notifications()
            ->where('id', $id)
            ->first();

        if ($notification) {
            $notification->markAsRead();
        }

        $unreadCount = $request->user()->unreadNotifications()->count();

        return response()->json([
            'message' => 'Marked as read.',
            'unread_count' => $unreadCount,
        ]);
    }

    public function markAllAsRead(Request $request)
    {
        $count = $request->user()->unreadNotifications()->count();
        $request->user()->unreadNotifications()->update(['read_at' => now()]);

        return response()->json([
            'message' => "{$count} notifikasi ditandai dibaca.",
            'unread_count' => 0,
        ]);
    }
}
```

### Broadcast Notification Handler
```javascript
// resources/js/services/notification-service.js
import echo from '../echo';
import axios from 'axios';

class NotificationService {
    constructor() {
        this.listeners = new Map();
        this.unreadCount = 0;
        this.callbacks = {
            onCountChange: null,
            onNewNotification: null,
        };
    }

    async init(userId, callbacks = {}) {
        this.callbacks = callbacks;

        try {
            const response = await axios.get('/notifications/unread-count');
            this.unreadCount = response.data.count;
            this.emitCountChange();
        } catch (e) {
            console.error('Failed to load unread count', e);
        }

        echo.private(`user.${userId}`)
            .notification((notification) => {
                this.unreadCount++;
                this.emitCountChange();

                if (this.callbacks.onNewNotification) {
                    this.callbacks.onNewNotification(notification);
                }

                this.showBrowserNotification(notification);
                this.playNotificationSound();
            });
    }

    decrementCount(amount = 1) {
        this.unreadCount = Math.max(0, this.unreadCount - amount);
        this.emitCountChange();
    }

    resetCount() {
        this.unreadCount = 0;
        this.emitCountChange();
    }

    getCount() {
        return this.unreadCount;
    }

    emitCountChange() {
        if (this.callbacks.onCountChange) {
            this.callbacks.onCountChange(this.unreadCount);
        }

        const event = new CustomEvent('notification-count-change', {
            detail: { count: this.unreadCount },
        });
        window.dispatchEvent(event);
    }

    showBrowserNotification(notification) {
        if (!('Notification' in window)) return;

        if (Notification.permission === 'granted') {
            new Notification(notification.title, {
                body: notification.body,
                icon: '/favicon.ico',
                tag: 'notification',
            });
        } else if (Notification.permission !== 'denied') {
            Notification.requestPermission();
        }
    }

    playNotificationSound() {
        const audio = new Audio('/sounds/notification.mp3');
        audio.volume = 0.3;
        audio.play().catch(() => {});
    }
}

export const notificationService = new NotificationService();
```

```vue
<script setup>
import { ref, onMounted, onUnmounted, computed } from 'vue';
import axios from 'axios';
import { notificationService } from '../services/notification-service';

const props = defineProps({ userId: { type: Number, required: true } });

const showDropdown = ref(false);
const notifications = ref([]);
const unreadCount = ref(0);
const loading = ref(false);
const hasMore = ref(true);
const page = ref(1);

const displayCount = computed(() => {
    return unreadCount.value > 99 ? '99+' : unreadCount.value;
});

onMounted(() => {
    notificationService.init(props.userId, {
        onCountChange: (count) => {
            unreadCount.value = count;
        },
        onNewNotification: (notification) => {
            notifications.value.unshift(notification);
        },
    });

    window.addEventListener('notification-count-change', (e) => {
        unreadCount.value = e.detail.count;
    });
});

async function toggleDropdown() {
    showDropdown.value = !showDropdown.value;
    if (showDropdown.value && notifications.value.length === 0) {
        await loadNotifications();
    }
}

async function loadNotifications() {
    loading.value = true;
    try {
        const response = await axios.get('/notifications/list', {
            params: { page: page.value },
        });
        notifications.value = [...notifications.value, ...response.data.data];
        hasMore.value = response.data.meta.has_more;
    } catch (e) {
        console.error(e);
    } finally {
        loading.value = false;
    }
}

async function loadMore() {
    page.value++;
    await loadNotifications();
}

async function markAsRead(notif) {
    if (!notif.read_at) {
        try {
            await axios.post(`/notifications/${notif.id}/mark-read`);
            notif.read_at = new Date();
            notificationService.decrementCount();
        } catch (e) {
            console.error(e);
        }
    }
}

async function markAllAsRead() {
    try {
        await axios.post('/notifications/mark-all-read');
        notifications.value.forEach(n => n.read_at = new Date());
        notificationService.resetCount();
    } catch (e) {
        console.error(e);
    }
}

function getIcon(notif) {
    const icons = {
        'check-circle': '✅',
        'message': '💬',
        'warning': '⚠️',
        'info': 'ℹ️',
        'error': '❌',
        'bell': '🔔',
    };
    return icons[notif.data?.icon] || '🔔';
}

function formatTime(date) {
    const d = new Date(date);
    const now = new Date();
    const diff = now - d;

    if (diff < 60000) return 'Baru saja';
    if (diff < 3600000) return `${Math.floor(diff / 60000)}m`;
    if (diff < 86400000) return `${Math.floor(diff / 3600000)}j`;
    return d.toLocaleDateString('id-ID', { day: 'numeric', month: 'short' });
}

function onScroll(event) {
    const el = event.target;
    if (el.scrollHeight - el.scrollTop - el.clientHeight < 100 && hasMore.value && !loading.value) {
        loadMore();
    }
}
</script>
```

### Livewire Component Alternative
```php
<?php
namespace App\Http\Livewire;

use Livewire\Component;

class NotificationBell extends Component
{
    public int $unreadCount = 0;
    public bool $showDropdown = false;
    public array $notifications = [];

    protected $listeners = ['notificationReceived' => 'refreshCount'];

    public function mount(): void
    {
        $this->refreshCount();
    }

    public function refreshCount(): void
    {
        $user = auth()->user();
        if ($user) {
            $this->unreadCount = $user->unreadNotifications()->count();
            $this->notifications = $user->notifications()
                ->latest()
                ->take(10)
                ->get()
                ->toArray();
        }
    }

    public function toggle(): void
    {
        $this->showDropdown = !$this->showDropdown;
        if ($this->showDropdown) {
            $this->refreshCount();
        }
    }

    public function markAsRead(string $notificationId): void
    {
        $notification = auth()->user()->notifications()->find($notificationId);
        if ($notification) {
            $notification->markAsRead();
            $this->refreshCount();
        }
    }

    public function markAllAsRead(): void
    {
        auth()->user()->unreadNotifications()->update(['read_at' => now()]);
        $this->refreshCount();
    }

    public function render()
    {
        return view('livewire.notification-bell');
    }
}
```
