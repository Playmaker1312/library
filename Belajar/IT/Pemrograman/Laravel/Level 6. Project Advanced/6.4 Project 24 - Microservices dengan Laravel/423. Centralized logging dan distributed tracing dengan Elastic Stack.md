# 423. Centralized Logging dan Distributed Tracing dengan Elastic Stack

## Definisi & Target
Mengimplementasikan centralized logging dan distributed tracing untuk memonitor dan mendebug interaksi antar microservices. Menggunakan Elastic Stack (Elasticsearch, Logstash, Kibana) untuk log aggregation dan OpenTelemetry untuk distributed tracing.

## Komponen yang Terlibat
- Elastic Stack (ELK)
- Filebeat / Logstash untuk log shipping
- OpenTelemetry SDK
- Distributed tracing dengan Trace IDs
- Structured logging (JSON)
- Correlation IDs per request
- Log enrichment (service name, trace ID)
- Kibana dashboards
- Alerting & monitoring

### Correlation ID Middleware
```php
<?php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Str;
use Illuminate\Support\Facades\Log;

class RequestTracing
{
    public function handle(Request $request, Closure $next): mixed
    {
        $traceId = $request->header('X-Trace-Id', (string) Str::uuid());
        $spanId = (string) Str::random(16);

        $request->merge([
            'trace_id' => $traceId,
            'span_id' => $spanId,
        ]);

        Log::withContext([
            'trace_id' => $traceId,
            'span_id' => $spanId,
            'service' => config('app.name'),
            'environment' => app()->environment(),
        ]);

        $startTime = microtime(true);

        $response = $next($request);

        $duration = (microtime(true) - $startTime) * 1000;

        $response->headers->set('X-Trace-Id', $traceId);
        $response->headers->set('X-Span-Id', $spanId);

        Log::info('Request processed', [
            'method' => $request->method(),
            'path' => $request->path(),
            'status' => $response->status(),
            'duration_ms' => round($duration, 2),
            'ip' => $request->ip(),
        ]);

        return $response;
    }
}
```

### Structured Log Helper
```php
<?php
namespace App\Services;

use Illuminate\Support\Facades\Log;

class StructuredLogger
{
    public static function info(string $message, array $context = []): void
    {
        Log::info($message, self::enrichContext($context));
    }

    public static function warning(string $message, array $context = []): void
    {
        Log::warning($message, self::enrichContext($context));
    }

    public static function error(string $message, array $context = []): void
    {
        Log::error($message, self::enrichContext($context));
    }

    public static function critical(string $message, array $context = []): void
    {
        Log::critical($message, self::enrichContext($context));
    }

    public static function debug(string $message, array $context = []): void
    {
        Log::debug($message, self::enrichContext($context));
    }

    public static function serviceCall(string $targetService, string $method, string $path, int $status, float $duration): void
    {
        self::info('Service call', [
            'type' => 'service_call',
            'target_service' => $targetService,
            'method' => $method,
            'path' => $path,
            'status_code' => $status,
            'duration_ms' => round($duration, 2),
        ]);
    }

    public static function databaseQuery(string $connection, string $query, float $duration): void
    {
        self::debug('Database query', [
            'type' => 'db_query',
            'connection' => $connection,
            'query' => $query,
            'duration_ms' => round($duration, 2),
        ]);
    }

    public static function queueJob(string $job, string $queue, string $status, ?string $error = null): void
    {
        self::info('Queue job', [
            'type' => 'queue_job',
            'job' => $job,
            'queue' => $queue,
            'status' => $status,
            'error' => $error,
        ]);
    }

    private static function enrichContext(array $context): array
    {
        return array_merge($context, [
            'service' => config('app.name'),
            'host' => gethostname(),
            'memory_mb' => round(memory_get_usage(true) / 1024 / 1024, 2),
            'php_version' => PHP_VERSION,
        ]);
    }
}
```

### HTTP Client Logging for Service Calls
```php
<?php
namespace App\Providers;

use Illuminate\Support\ServiceProvider;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;

class HttpClientLoggingServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        Http::globalRequestMiddleware(function ($request) {
            $startTime = microtime(true);

            $request->withHeaders([
                'X-Trace-Id' => request()->header('X-Trace-Id', (string) str()->uuid()),
                'X-Span-Id' => (string) str()->random(16),
                'X-Service-Name' => config('app.name'),
            ]);

            return function ($response) use ($request, $startTime) {
                $duration = (microtime(true) - $startTime) * 1000;

                Log::info('Outbound HTTP request', [
                    'type' => 'service_call',
                    'method' => $request->method(),
                    'url' => $request->url(),
                    'status' => $response->status(),
                    'duration_ms' => round($duration, 2),
                    'trace_id' => $request->header('X-Trace-Id'),
                    'target_service' => self::extractServiceName($request->url()),
                ]);
            };
        });
    }

    private static function extractServiceName(string $url): string
    {
        $services = [
            'user-service' => 'user',
            'order-service' => 'order',
            'product-service' => 'product',
            'payment-service' => 'payment',
            'notification-service' => 'notification',
        ];

        foreach ($services as $host => $name) {
            if (str_contains($url, $host)) {
                return $name;
            }
        }

        return 'unknown';
    }
}
```

### Distributed Tracing with Trace Context Propagation
```php
<?php
namespace App\Services;

use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Str;

class DistributedTracer
{
    private static array $spans = [];

    public static function startSpan(string $name, string $type = 'internal'): string
    {
        $spanId = (string) Str::random(16);

        self::$spans[$spanId] = [
            'name' => $name,
            'type' => $type,
            'trace_id' => request()->header('X-Trace-Id', (string) Str::uuid()),
            'parent_span_id' => request()->header('X-Span-Id'),
            'start_time' => microtime(true),
            'tags' => [],
        ];

        return $spanId;
    }

    public static function endSpan(string $spanId, array $tags = []): void
    {
        if (!isset(self::$spans[$spanId])) {
            return;
        }

        $span = self::$spans[$spanId];
        $duration = (microtime(true) - $span['start_time']) * 1000;

        $spanData = array_merge($span, [
            'duration_ms' => round($duration, 2),
            'end_time' => microtime(true),
            'tags' => $tags,
        ]);

        unset(self::$spans[$spanId]);

        Cache::store('redis')->lpush(
            'spans:' . $span['trace_id'],
            json_encode($spanData)
        );
    }

    public static function addTag(string $spanId, string $key, string $value): void
    {
        if (isset(self::$spans[$spanId])) {
            self::$spans[$spanId]['tags'][$key] = $value;
        }
    }

    public static function getTrace(string $traceId): array
    {
        $spans = Cache::store('redis')->lrange("spans:{$traceId}", 0, -1);

        return array_map(fn ($s) => json_decode($s, true), $spans);
    }

    public static function getActiveSpanId(): ?string
    {
        return array_key_last(self::$spans) ?: null;
    }

    public static function makeHttpCall(string $service, string $method, string $path, array $data = []): mixed
    {
        $spanId = self::startSpan("call:{$service}:{$path}", 'service_call');

        $traceId = request()->header('X-Trace-Id', (string) Str::uuid());

        try {
            $response = Http::withHeaders([
                'X-Trace-Id' => $traceId,
                'X-Span-Id' => $spanId,
                'X-Service-Name' => config('app.name'),
            ])->$method(
                rtrim(env(strtoupper($service) . '_SERVICE_URL'), '/') . '/' . ltrim($path, '/'),
                $data
            );

            self::endSpan($spanId, [
                'status_code' => $response->status(),
                'target_service' => $service,
                'method' => $method,
                'path' => $path,
            ]);

            return $response;
        } catch (\Exception $e) {
            self::endSpan($spanId, [
                'error' => $e->getMessage(),
                'target_service' => $service,
            ]);
            throw $e;
        }
    }
}
```

### Trace Viewer Endpoint
```php
<?php
namespace App\Http\Controllers;

use App\Services\DistributedTracer;
use Illuminate\Http\Request;

class TraceController extends Controller
{
    public function show(string $traceId): mixed
    {
        $spans = DistributedTracer::getTrace($traceId);

        $trace = [
            'trace_id' => $traceId,
            'duration_ms' => $this->calculateDuration($spans),
            'span_count' => count($spans),
            'services' => $this->extractServices($spans),
            'waterfall' => $this->buildWaterfall($spans),
        ];

        return response()->json($trace);
    }

    public function recent(Request $request): mixed
    {
        $limit = $request->input('limit', 20);

        // Get recent trace IDs from spans stored in Redis
        $traceIds = [];

        return response()->json([
            'data' => array_map(
                fn ($id) => $this->show($id),
                array_slice($traceIds, 0, $limit)
            ),
        ]);
    }

    private function calculateDuration(array $spans): float
    {
        if (empty($spans)) return 0;

        $minStart = min(array_column($spans, 'start_time'));
        $maxEnd = max(array_column($spans, 'end_time'));

        return round(($maxEnd - $minStart) * 1000, 2);
    }

    private function extractServices(array $spans): array
    {
        return array_values(array_unique(
            array_merge(
                [config('app.name')],
                array_column($spans, 'tags.service_name') ?? []
            )
        ));
    }

    private function buildWaterfall(array $spans): array
    {
        usort($spans, fn ($a, $b) => $a['start_time'] <=> $b['start_time']);

        return array_map(fn ($span) => [
            'name' => $span['name'],
            'type' => $span['type'],
            'start_offset_ms' => round(($span['start_time'] - $spans[0]['start_time']) * 1000, 2),
            'duration_ms' => $span['duration_ms'],
            'parent_span_id' => $span['parent_span_id'],
        ], $spans);
    }
}
```

### Log Enricher Service Provider
```php
<?php
namespace App\Providers;

use Illuminate\Support\ServiceProvider;
use Illuminate\Support\Facades\Log;

class LogEnricherServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        Log::afterLogging(function ($channel, $level, $message, $context) {
            // Enrich log context with system metrics
            $context['memory_peak_mb'] = round(memory_get_peak_usage(true) / 1024 / 1024, 2);
            $context['load_average'] = sys_getloadavg()[0] ?? 0;
            $context['php_memory_limit'] = ini_get('memory_limit');

            // Add queue size
            try {
                $context['queue_size'] = app('queue')->connection()->size();
            } catch (\Exception $e) {
                // Queue not available
            }

            return compact('channel', 'level', 'message', 'context');
        });
    }
}
```
