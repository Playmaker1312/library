# 419. Inter-Service Communication dengan HTTP, Queue, dan Event Broadcasting

## Definisi & Target
Mengimplementasikan pola komunikasi antar microservices: synchronous via HTTP/REST, asynchronous via Queue (RabbitMQ/Redis), dan event-driven via Laravel Broadcasting. Targetnya adalah memilih pola komunikasi yang tepat untuk setiap interaksi antar service.

## Komponen yang Terlibat
- HTTP Client (Guzzle)
- Laravel Queue (RabbitMQ, Redis, SQS)
- Laravel Events + Listeners
- Event Broadcasting
- Message serialization
- Idempotency handling
- Retry & dead letter queues
- Service mesh (optional)

## Kode Implementasi (Per File)

### Synchronous HTTP Communication
```php
<?php
namespace App\Services;

use Illuminate\Support\Facades\Http;
use Illuminate\Http\Client\Response;
use Illuminate\Http\Client\RequestException;

class ServiceMeshClient
{
    private array $serviceUrls;

    public function __construct()
    {
        $this->serviceUrls = [
            'user' => env('USER_SERVICE_URL', 'http://user-service:8001'),
            'product' => env('PRODUCT_SERVICE_URL', 'http://product-service:8002'),
            'order' => env('ORDER_SERVICE_URL', 'http://order-service:8003'),
            'payment' => env('PAYMENT_SERVICE_URL', 'http://payment-service:8004'),
            'notification' => env('NOTIFICATION_SERVICE_URL', 'http://notification-service:8005'),
        ];
    }

    public function call(string $service, string $method, string $path, array $data = []): Response
    {
        $url = rtrim($this->serviceUrls[$service], '/') . '/' . ltrim($path, '/');

        $response = Http::withHeaders([
            'X-Service-Name' => config('app.name'),
            'X-Request-Id' => (string) str()->uuid(),
            'X-Trace-Id' => $this->getTraceId(),
        ])->timeout(10)->retry(3, 100)->$method($url, $data);

        if ($response->failed()) {
            throw new ServiceCommunicationException(
                "Failed to communicate with {$service}: {$response->status()}"
            );
        }

        return $response;
    }

    public function getUser(int $id): array
    {
        return $this->call('user', 'get', "/api/users/{$id}")->json();
    }

    public function getProduct(int $id): array
    {
        return $this->call('product', 'get', "/api/products/{$id}")->json();
    }

    public function createOrder(array $data): array
    {
        return $this->call('order', 'post', '/api/orders', $data)->json();
    }

    private function getTraceId(): string
    {
        return request()->header('X-Trace-Id', (string) str()->uuid());
    }
}

class ServiceCommunicationException extends \RuntimeException {}
```

### Asynchronous Communication via Queue
```php
<?php
namespace App\Services;

use Illuminate\Support\Facades\Queue;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;

class AsyncServiceBus
{
    public function dispatch(string $service, string $action, array $payload): void
    {
        $message = [
            'service' => $service,
            'action' => $action,
            'payload' => $payload,
            'metadata' => [
                'source_service' => config('app.name'),
                'request_id' => request()->header('X-Request-Id', (string) str()->uuid()),
                'trace_id' => request()->header('X-Trace-Id', (string) str()->uuid()),
                'timestamp' => now()->toISOString(),
            ],
        ];

        Queue::connection('rabbitmq')
            ->pushOn("service:{$service}", new ServiceJob($message));
    }

    public function broadcast(string $event, array $payload): void
    {
        $message = [
            'event' => $event,
            'payload' => $payload,
            'metadata' => [
                'source_service' => config('app.name'),
                'event_id' => (string) str()->uuid(),
                'timestamp' => now()->toISOString(),
            ],
        ];

        Queue::connection('rabbitmq')
            ->pushOn('events', new ServiceJob($message));
    }
}
```

### Service Job Handler
```php
<?php
namespace App\Jobs;

use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Bus\Queueable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;

class ServiceJob implements ShouldQueue
{
    use InteractsWithQueue, Queueable, SerializesModels;

    public int $timeout = 30;
    public int $tries = 3;

    public function __construct(
        private array $message
    ) {}

    public function handle(): void
    {
        $service = $this->message['service'];
        $action = $this->message['action'];
        $payload = $this->message['payload'];
        $metadata = $this->message['metadata'];

        $serviceUrls = [
            'user' => env('USER_SERVICE_URL'),
            'order' => env('ORDER_SERVICE_URL'),
        ];

        $url = rtrim($serviceUrls[$service], '/') . "/api/internal/{$action}";

        try {
            $response = Http::withHeaders([
                'X-Service-Name' => config('app.name'),
                'X-Request-Id' => $metadata['request_id'],
                'X-Trace-Id' => $metadata['trace_id'],
                'X-Idempotency-Key' => $metadata['event_id'] ?? $metadata['request_id'],
            ])->timeout(25)->post($url, $payload);

            if ($response->failed()) {
                throw new \Exception("Service {$service} responded with {$response->status()}");
            }

        } catch (\Exception $e) {
            Log::error("ServiceJob failed", [
                'service' => $service,
                'action' => $action,
                'error' => $e->getMessage(),
            ]);

            if ($this->attempts() >= $this->tries) {
                $this->sendToDeadLetter($e->getMessage());
            }

            throw $e;
        }
    }

    private function sendToDeadLetter(string $error): void
    {
        Log::critical("Message sent to dead letter queue", [
            'message' => $this->message,
            'error' => $error,
        ]);
    }
}
```

### Event-Driven Communication via Broadcasting
```php
<?php
namespace App\Events;

use Illuminate\Broadcasting\Channel;
use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class OrderPlaced implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public array $order,
        public array $metadata = []
    ) {
        $this->metadata = [
            'source_service' => config('app.name'),
            'event_id' => (string) str()->uuid(),
            'timestamp' => now()->toISOString(),
        ];
    }

    public function broadcastOn(): array
    {
        return [
            new Channel('orders'),
            new Channel('service:notification'),
            new Channel('service:analytics'),
        ];
    }

    public function broadcastAs(): string
    {
        return 'order.placed';
    }

    public function broadcastWith(): array
    {
        return [
            'order_id' => $this->order['id'],
            'customer_id' => $this->order['customer_id'],
            'total' => $this->order['total'],
            'items' => $this->order['items'],
            'status' => 'pending',
            'metadata' => $this->metadata,
        ];
    }
}
```

### Listener di Service Lain
```php
<?php
namespace App\Listeners;

use App\Events\OrderPlaced;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;

class HandleOrderPlaced
{
    public function handle(OrderPlaced $event): void
    {
        // Notification Service: send confirmation
        $this->sendOrderConfirmation($event->order);

        // Inventory Service: reduce stock
        $this->reduceInventory($event->order['items']);

        // Analytics Service: track event
        $this->trackAnalytics($event);
    }

    private function sendOrderConfirmation(array $order): void
    {
        Http::post(env('NOTIFICATION_SERVICE_URL') . '/api/internal/send-email', [
            'user_id' => $order['customer_id'],
            'template' => 'order-confirmation',
            'data' => $order,
        ]);
    }

    private function reduceInventory(array $items): void
    {
        foreach ($items as $item) {
            Http::post(env('PRODUCT_SERVICE_URL') . '/api/internal/reduce-stock', [
                'product_id' => $item['product_id'],
                'quantity' => $item['quantity'],
            ]);
        }
    }

    private function trackAnalytics(OrderPlaced $event): void
    {
        Log::info('Order analytics event', $event->broadcastWith());
    }
}
```

### Idempotency Middleware
```php
<?php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Cache;

class Idempotency
{
    public function handle(Request $request, Closure $next): mixed
    {
        $key = $request->header('X-Idempotency-Key');

        if (!$key) {
            return $next($request);
        }

        $cacheKey = 'idempotency:' . $key;

        if ($cached = Cache::get($cacheKey)) {
            return response()->json($cached['data'], $cached['status']);
        }

        $response = $next($request);

        if ($response->getStatusCode() < 500) {
            Cache::put($cacheKey, [
                'data' => $response->getData(),
                'status' => $response->getStatusCode(),
            ], now()->addHours(24));
        }

        return $response;
    }
}
```
