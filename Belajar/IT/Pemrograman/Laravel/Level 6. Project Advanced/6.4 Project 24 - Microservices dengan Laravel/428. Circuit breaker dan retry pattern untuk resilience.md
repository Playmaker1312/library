# 428. Circuit Breaker dan Retry Pattern untuk Resilience

## Definisi & Target
Mengimplementasikan circuit breaker pattern dan retry mechanism untuk meningkatkan resilience microservices. Meliputi circuit breaker states (closed, open, half-open), retry with backoff, bulkhead pattern, timeout handling, dan graceful degradation.

## Komponen yang Terlibat
- Circuit Breaker pattern
- Retry with exponential backoff
- Bulkhead pattern
- Timeout handling
- Fallback mechanism
- Graceful degradation
- Failure caching
- Health monitoring

### Circuit Breaker Implementation
```php
<?php
namespace App\Resilience;

use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Log;

class CircuitBreaker
{
    private string $serviceName;
    private int $failureThreshold;
    private int $successThreshold;
    private int $timeout;

    public function __construct(
        string $serviceName,
        int $failureThreshold = 5,
        int $successThreshold = 3,
        int $timeout = 30
    ) {
        $this->serviceName = $serviceName;
        $this->failureThreshold = $failureThreshold;
        $this->successThreshold = $successThreshold;
        $this->timeout = $timeout;
    }

    public function call(callable $operation, ?callable $fallback = null): mixed
    {
        $state = $this->getState();

        if ($state === 'open') {
            if (!$this->isTimeoutExpired()) {
                Log::warning("Circuit breaker open for {$this->serviceName}, using fallback");
                return $this->executeFallback($fallback);
            }
            $this->setState('half-open');
        }

        try {
            $result = $operation();
            $this->onSuccess();
            return $result;
        } catch (\Exception $e) {
            $this->onFailure($e);
            return $this->executeFallback($fallback);
        }
    }

    public function isAvailable(): bool
    {
        $state = $this->getState();
        if ($state === 'open') {
            return $this->isTimeoutExpired();
        }
        return true;
    }

    private function getState(): string
    {
        return Cache::get("circuit:{$this->serviceName}:state", 'closed');
    }

    private function setState(string $state): void
    {
        Cache::put("circuit:{$this->serviceName}:state", $state, $this->timeout);
        Log::info("Circuit breaker {$this->serviceName} -> {$state}");
    }

    private function isTimeoutExpired(): bool
    {
        $openedAt = Cache::get("circuit:{$this->serviceName}:opened_at");
        if (!$openedAt) return true;
        return now()->diffInSeconds($openedAt) >= $this->timeout;
    }

    private function onSuccess(): void
    {
        $state = $this->getState();

        if ($state === 'half-open') {
            $successCount = Cache::increment("circuit:{$this->serviceName}:success_count", 1);
            if ($successCount >= $this->successThreshold) {
                $this->reset();
            }
        } else {
            $this->reset();
        }
    }

    private function onFailure(\Exception $e): void
    {
        $failureCount = Cache::increment("circuit:{$this->serviceName}:failure_count", 1);

        Log::warning("Circuit breaker failure for {$this->serviceName}", [
            'failure_count' => $failureCount,
            'error' => $e->getMessage(),
        ]);

        if ($failureCount >= $this->failureThreshold) {
            $this->open();
        }
    }

    private function open(): void
    {
        $this->setState('open');
        Cache::put("circuit:{$this->serviceName}:opened_at", now(), $this->timeout * 2);
    }

    private function reset(): void
    {
        $this->setState('closed');
        Cache::forget("circuit:{$this->serviceName}:failure_count");
        Cache::forget("circuit:{$this->serviceName}:success_count");
        Cache::forget("circuit:{$this->serviceName}:opened_at");
    }

    private function executeFallback(?callable $fallback): mixed
    {
        if ($fallback) {
            return $fallback();
        }

        throw new ServiceUnavailableException(
            "Service {$this->serviceName} is unavailable"
        );
    }
}

class ServiceUnavailableException extends \RuntimeException {}
```

### Retry Handler with Exponential Backoff
```php
<?php
namespace App\Resilience;

use Illuminate\Support\Facades\Log;

class RetryHandler
{
    private int $maxRetries;
    private int $baseDelay;
    private float $multiplier;
    private int $maxDelay;
    private array $retryableStatuses;

    public function __construct(
        int $maxRetries = 3,
        int $baseDelay = 100,
        float $multiplier = 2.0,
        int $maxDelay = 10000,
        array $retryableStatuses = [408, 429, 500, 502, 503, 504]
    ) {
        $this->maxRetries = $maxRetries;
        $this->baseDelay = $baseDelay;
        $this->multiplier = $multiplier;
        $this->maxDelay = $maxDelay;
        $this->retryableStatuses = $retryableStatuses;
    }

    public function execute(callable $operation, string $operationName = ''): mixed
    {
        $lastException = null;

        for ($attempt = 1; $attempt <= $this->maxRetries; $attempt++) {
            try {
                $result = $operation();

                if ($result instanceof \Illuminate\Http\Client\Response) {
                    $status = $result->status();
                    if (in_array($status, $this->retryableStatuses)) {
                        throw new RetryableException("HTTP {$status}", $status);
                    }
                }

                return $result;

            } catch (RetryableException $e) {
                $lastException = $e;

                if ($attempt < $this->maxRetries) {
                    $delay = $this->calculateDelay($attempt);
                    Log::warning("Retry {$attempt}/{$this->maxRetries} for {$operationName}", [
                        'delay_ms' => $delay,
                        'error' => $e->getMessage(),
                    ]);
                    usleep($delay * 1000);
                }
            } catch (\Exception $e) {
                if ($this->isRetryable($e)) {
                    $lastException = $e;
                    if ($attempt < $this->maxRetries) {
                        $delay = $this->calculateDelay($attempt);
                        usleep($delay * 1000);
                    }
                } else {
                    throw $e;
                }
            }
        }

        throw $lastException ?? new \RuntimeException("Operation {$operationName} failed after {$this->maxRetries} retries");
    }

    private function calculateDelay(int $attempt): int
    {
        $delay = $this->baseDelay * pow($this->multiplier, $attempt - 1);
        $jitter = mt_rand(0, (int) ($delay * 0.1));
        return min((int) $delay + $jitter, $this->maxDelay);
    }

    private function isRetryable(\Exception $e): bool
    {
        return $e instanceof \Illuminate\Http\Client\ConnectionException
            || $e instanceof \Illuminate\Http\Client\RequestException
            || $e instanceof \GuzzleHttp\Exception\ConnectException;
    }
}

class RetryableException extends \RuntimeException {}
```

### Bulkhead Pattern
```php
<?php
namespace App\Resilience;

use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Log;

class Bulkhead
{
    private string $name;
    private int $maxConcurrent;
    private int $maxQueueSize;
    private array $queue = [];

    public function __construct(
        string $name,
        int $maxConcurrent = 10,
        int $maxQueueSize = 50
    ) {
        $this->name = $name;
        $this->maxConcurrent = $maxConcurrent;
        $this->maxQueueSize = $maxQueueSize;
    }

    public function execute(callable $operation): mixed
    {
        $currentCount = $this->getCurrentCount();

        if ($currentCount >= $this->maxConcurrent) {
            if (count($this->queue) >= $this->maxQueueSize) {
                throw new BulkheadFullException(
                    "Bulkhead {$this->name} is full. {$currentCount} concurrent, {$this->queue} queued."
                );
            }

            return $this->enqueue($operation);
        }

        return $this->run($operation);
    }

    private function run(callable $operation): mixed
    {
        $this->incrementCount();

        try {
            $result = $operation();
            return $result;
        } finally {
            $this->decrementCount();
            $this->processQueue();
        }
    }

    private function enqueue(callable $operation): mixed
    {
        $this->queue[] = $operation;

        Log::info("Request queued in bulkhead {$this->name}", [
            'queue_size' => count($this->queue),
            'concurrent' => $this->getCurrentCount(),
        ]);

        // In production, this would use a proper queue system
        // For simplicity, we wait until a slot is available
        while ($this->getCurrentCount() >= $this->maxConcurrent) {
            usleep(10000); // 10ms
        }

        return $this->run(array_shift($this->queue));
    }

    private function processQueue(): void
    {
        if (!empty($this->queue) && $this->getCurrentCount() < $this->maxConcurrent) {
            $operation = array_shift($this->queue);
            $this->run($operation);
        }
    }

    private function getCurrentCount(): int
    {
        return Cache::get("bulkhead:{$this->name}:count", 0);
    }

    private function incrementCount(): void
    {
        Cache::increment("bulkhead:{$this->name}:count", 1);
    }

    private function decrementCount(): void
    {
        Cache::decrement("bulkhead:{$this->name}:count", 1);
    }
}

class BulkheadFullException extends \RuntimeException {}
```

### Resilience Service Proxy
```php
<?php
namespace App\Services;

use App\Resilience\CircuitBreaker;
use App\Resilience\RetryHandler;
use App\Resilience\Bulkhead;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Http;

class ResilienceServiceProxy
{
    private array $circuitBreakers = [];
    private RetryHandler $retryHandler;
    private array $bulkheads = [];
    private array $timeouts;

    public function __construct()
    {
        $this->retryHandler = new RetryHandler();
        $this->timeouts = [
            'user-service' => 5,
            'product-service' => 5,
            'order-service' => 10,
            'payment-service' => 15,
            'notification-service' => 5,
        ];
    }

    public function call(
        string $service,
        string $method,
        string $path,
        array $data = [],
        ?callable $fallback = null
    ): mixed {
        $circuitBreaker = $this->getCircuitBreaker($service);
        $bulkhead = $this->getBulkhead($service);

        return $bulkhead->execute(function () use ($circuitBreaker, $service, $method, $path, $data, $fallback) {
            return $circuitBreaker->call(
                function () use ($service, $method, $path, $data) {
                    return $this->retryHandler->execute(
                        function () use ($service, $method, $path, $data) {
                            return $this->makeRequest($service, $method, $path, $data);
                        },
                        "{$service}:{$method}:{$path}"
                    );
                },
                $fallback
            );
        });
    }

    public function getCachedOrCall(string $service, string $path, int $ttlSeconds = 60): mixed
    {
        $cacheKey = "resilience:{$service}:" . md5($path);

        return Cache::remember($cacheKey, $ttlSeconds, function () use ($service, $path) {
            return $this->call($service, 'get', $path);
        });
    }

    private function makeRequest(string $service, string $method, string $path, array $data): mixed
    {
        $urls = [
            'user-service' => env('USER_SERVICE_URL'),
            'product-service' => env('PRODUCT_SERVICE_URL'),
            'order-service' => env('ORDER_SERVICE_URL'),
            'payment-service' => env('PAYMENT_SERVICE_URL'),
            'notification-service' => env('NOTIFICATION_SERVICE_URL'),
        ];

        $baseUrl = $urls[$service] ?? throw new \InvalidArgumentException("Unknown service: {$service}");
        $url = rtrim($baseUrl, '/') . '/' . ltrim($path, '/');
        $timeout = $this->timeouts[$service] ?? 10;

        $response = Http::timeout($timeout)
            ->withHeaders(['X-Service-Name' => config('app.name')])
            ->$method($url, $data);

        if ($response->failed()) {
            throw new \RuntimeException(
                "Service {$service} responded with {$response->status()}: {$response->body()}"
            );
        }

        return $response->json();
    }

    private function getCircuitBreaker(string $service): CircuitBreaker
    {
        if (!isset($this->circuitBreakers[$service])) {
            $this->circuitBreakers[$service] = new CircuitBreaker(
                $service,
                failureThreshold: 5,
                successThreshold: 3,
                timeout: 30
            );
        }
        return $this->circuitBreakers[$service];
    }

    private function getBulkhead(string $service): Bulkhead
    {
        if (!isset($this->bulkheads[$service])) {
            $configs = [
                'user-service' => [20, 100],
                'product-service' => [30, 150],
                'order-service' => [15, 50],
                'payment-service' => [10, 30],
                'notification-service' => [25, 100],
            ];

            [$maxConcurrent, $maxQueue] = $configs[$service] ?? [10, 50];
            $this->bulkheads[$service] = new Bulkhead($service, $maxConcurrent, $maxQueue);
        }
        return $this->bulkheads[$service];
    }
}
```

### Graceful Degradation Example
```php
<?php
namespace App\Services;

use App\Resilience\CircuitBreaker;
use Illuminate\Support\Facades\Cache;

class CatalogServiceWithDegradation
{
    private ResilienceServiceProxy $proxy;

    public function __construct(ResilienceServiceProxy $proxy)
    {
        $this->proxy = $proxy;
    }

    public function getProduct(int $productId): array
    {
        $fallback = function () use ($productId) {
            return $this->getStaleProductData($productId);
        };

        return $this->proxy->call(
            'product-service',
            'get',
            "/api/products/{$productId}",
            [],
            $fallback
        );
    }

    public function getRelatedProducts(int $productId): array
    {
        try {
            return $this->proxy->getCachedOrCall(
                'product-service',
                "/api/products/{$productId}/related",
                300
            );
        } catch (\Exception $e) {
            // Graceful degradation: return empty recommendation
            return [
                'data' => [],
                'source' => 'fallback',
                'message' => 'Recommendations temporarily unavailable',
            ];
        }
    }

    public function searchProducts(string $query): array
    {
        $cacheKey = 'product_search:' . md5($query);

        $results = Cache::remember($cacheKey, 60, function () use ($query) {
            try {
                return $this->proxy->call(
                    'product-service',
                    'get',
                    '/api/products',
                    ['q' => $query, 'limit' => 20]
                );
            } catch (\Exception $e) {
                return null;
            }
        });

        if ($results === null) {
            return [
                'data' => [],
                'source' => 'cache',
                'message' => 'Showing cached results. Some data may be outdated.',
            ];
        }

        return $results;
    }

    private function getStaleProductData(int $productId): array
    {
        $cached = Cache::get("product:{$productId}");

        if ($cached) {
            return [
                ...$cached,
                '_meta' => [
                    'source' => 'stale_cache',
                    'cached_at' => $cached['_cached_at'] ?? null,
                    'warning' => 'Showing cached product data',
                ],
            ];
        }

        // Minimal response when everything fails
        return [
            'id' => $productId,
            'name' => 'Product Temporarily Unavailable',
            'price' => 0,
            'stock' => 0,
            'is_active' => false,
            '_meta' => [
                'source' => 'minimal',
                'warning' => 'Product service unavailable',
            ],
        ];
    }
}
```

### Resilience Middleware
```php
<?php
namespace App\Http\Middleware;

use App\Resilience\CircuitBreaker;
use Closure;
use Illuminate\Http\Request;

class ResilienceMiddleware
{
    public function handle(Request $request, Closure $next): mixed
    {
        $cb = new CircuitBreaker(
            serviceName: config('app.name'),
            failureThreshold: 10,
            successThreshold: 3,
            timeout: 15
        );

        return $cb->call(
            fn () => $next($request),
            fn () => response()->json([
                'error' => 'Service temporarily unavailable',
                'retry_after' => 30,
            ], 503)->withHeaders(['Retry-After' => 30])
        );
    }
}
```

### Timeout Handler
```php
<?php
namespace App\Resilience;

use Illuminate\Support\Facades\Log;

class TimeoutHandler
{
    public static function execute(callable $operation, int $timeoutMs, string $name = ''): mixed
    {
        $start = microtime(true);

        $result = $operation();

        $elapsed = (microtime(true) - $start) * 1000;

        if ($elapsed > $timeoutMs) {
            Log::warning("Operation {$name} exceeded timeout", [
                'timeout_ms' => $timeoutMs,
                'actual_ms' => round($elapsed, 2),
            ]);
        }

        return $result;
    }

    public static function executeWithDeadline(callable $operation, int $deadlineMs, string $name = ''): mixed
    {
        $start = microtime(true);

        $result = $operation();

        $elapsed = (microtime(true) - $start) * 1000;

        if ($elapsed > $deadlineMs) {
            throw new TimeoutException(
                "Operation {$name} exceeded deadline of {$deadlineMs}ms (took {$elapsed}ms)"
            );
        }

        return $result;
    }
}

class TimeoutException extends \RuntimeException {}
```
