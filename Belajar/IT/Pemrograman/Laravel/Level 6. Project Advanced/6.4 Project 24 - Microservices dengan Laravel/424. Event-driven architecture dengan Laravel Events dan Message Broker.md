# 424. Event-Driven Architecture dengan Laravel Events dan Message Broker

## Definisi & Target
Mengimplementasikan event-driven architecture menggunakan Laravel Events dan message broker (RabbitMQ / Apache Kafka). Service berkomunikasi secara asynchronous melalui events, meningkatkan decoupling dan resilience.

## Komponen yang Terlibat
- Laravel Events
- Message Broker (RabbitMQ / Kafka)
- Event serialization
- Dead letter queues
- Event versioning
- Consumer groups
- Retry mechanism
- Event replay
- Outbox pattern

```bash
# Install queue package with RabbitMQ support
```

### Base Domain Event
```php
<?php
namespace App\Events;

use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;
use Illuminate\Support\Str;

abstract class BaseDomainEvent
{
    use Dispatchable, SerializesModels;

    public string $eventId;
    public string $eventName;
    public string $occurredAt;
    public int $eventVersion;
    public string $sourceService;

    public function __construct()
    {
        $this->eventId = (string) Str::uuid();
        $this->occurredAt = now()->toISOString();
        $this->sourceService = config('app.name');
        $this->eventVersion = config("event-bus.events.{$this->eventName}.version", 1);
    }

    abstract public function toPayload(): array;

    public function toMessage(): array
    {
        return [
            'event_id' => $this->eventId,
            'event_name' => $this->eventName,
            'event_version' => $this->eventVersion,
            'source_service' => $this->sourceService,
            'occurred_at' => $this->occurredAt,
            'trace_id' => request()->header('X-Trace-Id', $this->eventId),
            'span_id' => request()->header('X-Span-Id'),
            'data' => $this->toPayload(),
        ];
    }

    public function getRoutingKey(): string
    {
        return config("event-bus.events.{$this->eventName}.routing_key", $this->eventName);
    }

    public function getExchange(): string
    {
        return config("event-bus.events.{$this->eventName}.exchange", 'microservices');
    }
}
```

### RabbitMQ Event Publisher
```php
<?php
namespace App\Services;

use App\Events\BaseDomainEvent;
use Illuminate\Support\Facades\Log;
use PhpAmqpLib\Connection\AMQPLazyConnection;
use PhpAmqpLib\Message\AMQPMessage;

class RabbitMQEventPublisher
{
    private $connection;
    private $channel;

    public function __construct()
    {
        $this->connect();
    }

    private function connect(): void
    {
        $config = config('event-bus.connections.rabbitmq');

        $this->connection = new AMQPLazyConnection(
            $config['host'],
            $config['port'],
            $config['login'],
            $config['password'],
            $config['vhost']
        );

        $this->channel = $this->connection->channel();
    }

    public function publish(BaseDomainEvent $event): void
    {
        $message = $event->toMessage();
        $routingKey = $event->getRoutingKey();
        $exchange = $event->getExchange();

        $this->channel->exchange_declare(
            $exchange,
            'topic',
            false,
            true,
            false
        );

        $msg = new AMQPMessage(
            json_encode($message),
            [
                'content_type' => 'application/json',
                'delivery_mode' => AMQPMessage::DELIVERY_MODE_PERSISTENT,
                'message_id' => $event->eventId,
                'timestamp' => time(),
                'headers' => [
                    'event_name' => $event->eventName,
                    'event_version' => $event->eventVersion,
                    'source_service' => $event->sourceService,
                ],
            ]
        );

        $this->channel->basic_publish($msg, $exchange, $routingKey);

        Log::info("Event published", [
            'event_name' => $event->eventName,
            'event_id' => $event->eventId,
            'exchange' => $exchange,
            'routing_key' => $routingKey,
        ]);
    }

    public function __destruct()
    {
        $this->channel?->close();
        $this->connection?->close();
    }
}
```

### Outbox Pattern Implementation
```php
<?php
namespace App\Services;

use App\Events\BaseDomainEvent;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class EventOutbox
{
    public function store(BaseDomainEvent $event): void
    {
        DB::connection(config('database.default'))
            ->table('event_outbox')
            ->insert([
                'event_id' => $event->eventId,
                'event_name' => $event->eventName,
                'event_version' => $event->eventVersion,
                'payload' => json_encode($event->toMessage()),
                'exchange' => $event->getExchange(),
                'routing_key' => $event->getRoutingKey(),
                'status' => 'pending',
                'created_at' => now(),
                'updated_at' => now(),
            ]);
    }

    public function publishPending(): int
    {
        $published = 0;

        $pendingEvents = DB::connection(config('database.default'))
            ->table('event_outbox')
            ->where('status', 'pending')
            ->where('attempts', '<', 5)
            ->orderBy('created_at')
            ->lockForUpdate()
            ->get();

        foreach ($pendingEvents as $row) {
            try {
                $publisher = app(RabbitMQEventPublisher::class);

                $message = json_decode($row->payload, true);

                $publisher->publishRaw(
                    $message,
                    $row->exchange,
                    $row->routing_key
                );

                DB::connection(config('database.default'))
                    ->table('event_outbox')
                    ->where('id', $row->id)
                    ->update([
                        'status' => 'published',
                        'published_at' => now(),
                        'updated_at' => now(),
                    ]);

                $published++;
            } catch (\Exception $e) {
                DB::connection(config('database.default'))
                    ->table('event_outbox')
                    ->where('id', $row->id)
                    ->increment('attempts');

                Log::error("Failed to publish event from outbox", [
                    'event_id' => $row->event_id,
                    'error' => $e->getMessage(),
                ]);
            }
        }

        return $published;
    }
}
```

### Event Versioning and Migration
```php
<?php
namespace App\Services;

use Illuminate\Support\Facades\Log;

class EventVersionResolver
{
    private static array $migrations = [
        1 => [],  // Version 1: initial
        2 => [
            'order.placed' => [
                'add_field' => ['promo_code', 'discount_amount'],
                'rename_field' => ['customer.email' => 'customer.customer_email'],
            ],
        ],
        3 => [
            'user.registered' => [
                'add_field' => ['referral_code'],
                'remove_field' => ['phone_number'],
            ],
        ],
    ];

    public static function upgrade(array $message, int $targetVersion): array
    {
        $currentVersion = $message['event_version'] ?? 1;
        $eventName = $message['event_name'];

        for ($v = $currentVersion; $v < $targetVersion; $v++) {
            $migrations = self::$migrations[$v][$eventName] ?? [];

            foreach ($migrations as $type => $fields) {
                $message['data'] = self::applyMigration($message['data'], $type, $fields);
            }

            $message['event_version'] = $v + 1;
        }

        return $message;
    }

    private static function applyMigration(array $data, string $type, array $fields): array
    {
        return match ($type) {
            'add_field' => array_merge($data, $fields),
            'remove_field' => array_diff_key($data, array_flip($fields)),
            'rename_field' => self::renameFields($data, $fields),
            default => $data,
        };
    }

    private static function renameFields(array $data, array $mapping): array
    {
        foreach ($mapping as $old => $new) {
            $keys = explode('.', $old);
            $value = $data;

            foreach ($keys as $key) {
                $value = $value[$key] ?? null;
            }

            if ($value !== null) {
                $newKeys = explode('.', $new);
                $ref = &$data;
                foreach ($newKeys as $i => $k) {
                    if ($i === count($newKeys) - 1) {
                        $ref[$k] = $value;
                    } else {
                        $ref = &$ref[$k];
                    }
                }

                $ref = &$data;
                foreach ($keys as $i => $k) {
                    if ($i === count($keys) - 1) {
                        unset($ref[$k]);
                    } else {
                        $ref = &$ref[$k];
                    }
                }
            }
        }

        return $data;
    }
}
```

### Consumer Worker Artisan Command
```php
<?php
namespace App\Console\Commands;

use App\Services\RabbitMQEventPublisher;
use App\Services\EventVersionResolver;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\Event;
use Illuminate\Support\Facades\Log;
use PhpAmqpLib\Connection\AMQPLazyConnection;

class ConsumeEvents extends Command
{
    protected $signature = 'events:consume {queue?}';
    protected $description = 'Consume events from RabbitMQ';

    public function handle(): int
    {
        $queueName = $this->argument('queue') ?? config('app.name') . '.events';

        $config = config('event-bus.connections.rabbitmq');
        $connection = new AMQPLazyConnection(
            $config['host'], $config['port'],
            $config['login'], $config['password'],
            $config['vhost']
        );
        $channel = $connection->channel();

        $channel->queue_declare($queueName, false, true, false, false);

        // Bind to all relevant exchanges
        foreach (config('event-bus.events') as $eventName => $eventConfig) {
            $channel->queue_bind(
                $queueName,
                $eventConfig['exchange'],
                $eventConfig['routing_key']
            );
        }

        $this->info("Listening for events on queue: {$queueName}");

        $channel->basic_consume(
            $queueName,
            '',
            false,
            false,
            false,
            false,
            function ($msg) {
                try {
                    $message = json_decode($msg->body, true);

                    $currentVersion = config("event-bus.events.{$message['event_name']}.version", 1);
                    $message = EventVersionResolver::upgrade($message, $currentVersion);

                    $eventClass = $this->resolveEventClass($message['event_name']);

                    if ($eventClass) {
                        Event::dispatch(new $eventClass(
                            ...$message['data']
                        ));

                        Log::info("Event consumed", [
                            'event_name' => $message['event_name'],
                            'event_id' => $message['event_id'],
                        ]);
                    }

                    $msg->ack();
                } catch (\Exception $e) {
                    Log::error("Failed to process event", [
                        'error' => $e->getMessage(),
                        'body' => substr($msg->body, 0, 500),
                    ]);

                    $msg->nack(false, true);
                }
            }
        );

        while ($channel->is_consuming()) {
            $channel->wait();
        }

        $channel->close();
        $connection->close();

        return self::SUCCESS;
    }

    private function resolveEventClass(string $eventName): ?string
    {
        $map = [
            'order.placed' => \App\Events\OrderPlaced::class,
            'payment.processed' => \App\Events\PaymentProcessed::class,
            'inventory.updated' => \App\Events\InventoryUpdated::class,
        ];

        return $map[$eventName] ?? null;
    }
}
```

### Event Replay Command
```php
<?php
namespace App\Console\Commands;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\DB;
use App\Services\RabbitMQEventPublisher;

class ReplayEvents extends Command
{
    protected $signature = 'events:replay {event_name?} {--from=} {--to=}';
    protected $description = 'Replay events from the event store';

    public function handle(): int
    {
        $query = DB::connection('event_store')
            ->table('domain_events')
            ->orderBy('created_at');

        if ($eventName = $this->argument('event_name')) {
            $query->where('event_type', $eventName);
        }

        if ($from = $this->option('from')) {
            $query->where('created_at', '>=', $from);
        }

        if ($to = $this->option('to')) {
            $query->where('created_at', '<=', $to);
        }

        $events = $query->get();
        $publisher = app(RabbitMQEventPublisher::class);

        $this->info("Replaying {$events->count()} events...");

        $bar = $this->output->createProgressBar($events->count());

        foreach ($events as $event) {
            $publisher->publishRaw(
                json_decode($event->event_data, true),
                config("event-bus.events.{$event->event_type}.exchange", 'microservices'),
                config("event-bus.events.{$event->event_type}.routing_key", $event->event_type)
            );
            $bar->advance();
        }

        $bar->finish();
        $this->newLine();
        $this->info('Events replayed successfully');

        return self::SUCCESS;
    }
}
```
