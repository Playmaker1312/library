# 418. Membangun API Gateway dengan Laravel untuk Single Entry Point

## Definisi & Target
Membangun API Gateway Service menggunakan Laravel sebagai reverse proxy yang menjadi single entry point untuk semua client request. Gateway menangani routing, autentikasi, rate limiting, aggregasi respons, dan transformasi protocol.

## Komponen yang Terlibat
- Laravel sebagai API Gateway
- HTTP Proxy / Forwarding
- Token validation (JWT / API Key)
- Rate limiting
- Request/Response transformation
- Service discovery integration
- Circuit breaker pattern
- Request logging & tracing

## Perintah Artisan CLI
```bash
# Buat project gateway
composer create-project laravel/laravel api-gateway

# Install packages untuk HTTP client & proxy
composer require guzzlehttp/guzzle
composer require spatie/laravel-http-logger

# Buat middleware gateway
php artisan make:middleware GatewayAuth
php artisan make:middleware GatewayRateLimiter
php artisan make:middleware RequestTransformer

# Buat controller
php artisan make:controller GatewayController
```

### Gateway Controller
```php
<?php
namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Cache;
use Illuminate\Http\Client\ConnectionException;

class GatewayController extends Controller
{
    private array $circuitBreakers = [];

    public function handle(Request $request, string $path): mixed
    {
        $route = $this->matchRoute($request->method(), $path);

        if (!$route) {
            return response()->json(['error' => 'Route not found'], 404);
        }

        $targetUrl = $this->buildTargetUrl($route, $path);

        if ($this->isCircuitOpen($route['prefix'])) {
            return response()->json(['error' => 'Service unavailable'], 503);
        }

        try {
            $response = Http::timeout($route['timeout'] ?? 30)
                ->withHeaders($this->buildHeaders($request, $route))
                ->withBody($request->getContent(), $request->header('Content-Type', 'application/json'))
                ->send($request->method(), $targetUrl);

            $this->recordSuccess($route['prefix']);

            $transformedResponse = $this->transformResponse(
                $response,
                $route
            );

            return response($transformedResponse->body(), $response->status())
                ->withHeaders($this->filterHeaders($transformedResponse->headers()));

        } catch (ConnectionException $e) {
            $this->recordFailure($route['prefix']);

            return response()->json([
                'error' => 'Service temporarily unavailable',
                'service' => $route['target'],
            ], 502);
        }
    }

    private function matchRoute(string $method, string $path): ?array
    {
        foreach (config('gateway.routes') as $route) {
            $pattern = str_replace('*', '.*', $route['prefix']);
            if (preg_match('#^' . $pattern . '$#', $method . ' ' . $path)) {
                return $route;
            }
        }
        return null;
    }

    private function buildTargetUrl(array $route, string $path): string
    {
        if ($route['strip_prefix']) {
            $prefix = rtrim(str_replace('*', '', $route['prefix']), '/');
            $path = substr($path, strlen($prefix));
        }

        return rtrim($route['target'], '/') . '/' . ltrim($path, '/');
    }

    private function buildHeaders(Request $request, array $route): array
    {
        $headers = [
            'X-Request-Id' => $request->header('X-Request-Id', (string) str()->uuid()),
            'X-Gateway-Host' => $request->getHost(),
            'X-Forwarded-For' => $request->ip(),
            'X-Forwarded-Proto' => $request->scheme(),
            'X-Trace-Id' => $request->header('X-Trace-Id', (string) str()->uuid()),
        ];

        if ($user = $request->user()) {
            $headers['X-Authenticated-User'] = $user->id;
            $headers['X-User-Roles'] = $user->roles->pluck('name')->implode(',');
        }

        return $headers;
    }

    private function filterHeaders(array $headers): array
    {
        unset($headers['Transfer-Encoding'], $headers['X-Powered-By']);
        return $headers;
    }

    private function transformResponse($response, array $route): mixed
    {
        if (isset($route['transform'])) {
            $transformer = app($route['transform']);
            return $transformer->transform($response);
        }
        return $response;
    }

    private function isCircuitOpen(string $service): bool
    {
        $key = 'circuit_breaker:' . md5($service);
        return Cache::get($key, false);
    }

    private function recordSuccess(string $service): void
    {
        $key = 'circuit_failures:' . md5($service);
        Cache::forget($key);
    }

    private function recordFailure(string $service): void
    {
        $key = 'circuit_failures:' . md5($service);
        $failures = Cache::increment($key, 1, 60);

        $threshold = config('gateway.aggregation.circuit_breaker.failure_threshold');

        if ($failures >= $threshold) {
            $circuitKey = 'circuit_breaker:' . md5($service);
            Cache::put($circuitKey, true, config('gateway.aggregation.circuit_breaker.recovery_time'));
        }
    }
}
```

### Aggregation Gateway (Multiple Services)
```php
<?php
namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;

class AggregationController extends Controller
{
    public function userProfile(Request $request, int $userId): mixed
    {
        $responses = Http::pool(fn ($pool) => [
            $pool->get(env('USER_SERVICE_URL') . "/api/users/{$userId}"),
            $pool->get(env('ORDER_SERVICE_URL') . "/api/users/{$userId}/orders"),
            $pool->get(env('PAYMENT_SERVICE_URL') . "/api/users/{$userId}/payments"),
        ]);

        $user = $responses[0]->json();
        $orders = $responses[1]->json();
        $payments = $responses[2]->json();

        return response()->json([
            'user' => $user,
            'recent_orders' => $orders['data'] ?? [],
            'payment_methods' => $payments['data'] ?? [],
            'order_count' => count($orders['data'] ?? []),
            'total_spent' => collect($payments['data'] ?? [])->sum('amount'),
        ]);
    }
}
```

### Gateway Middleware - Authentication
```php
<?php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;

class GatewayAuth
{
    public function handle(Request $request, Closure $next): mixed
    {
        $token = $request->bearerToken();

        if (!$token) {
            return response()->json(['error' => 'Unauthenticated'], 401);
        }

        $response = Http::withToken($token)
            ->timeout(5)
            ->get(env('USER_SERVICE_URL') . '/api/internal/verify-token');

        if (!$response->successful()) {
            return response()->json(['error' => 'Invalid token'], 401);
        }

        $request->setUserResolver(function () use ($response) {
            return (object) $response->json('user');
        });

        return $next($request);
    }
}
```
