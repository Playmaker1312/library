# 420. Service Discovery dan Registrasi Service Otomatis

## Definisi & Target
Mengimplementasikan service discovery agar microservices dapat menemukan dan berkomunikasi satu sama lain secara dinamis, tanpa hardcoded URL. Meliputi service registration, health checking, dan load balancing client-side.

## Komponen yang Terlibat
- Service Registry (Consul, etcd, atau database-based)
- Health check endpoints
- Client-side load balancing
- Service registration on boot
- Heartbeat mechanism
- Service deregistration on shutdown
- Caching service locations

## Perintah Artisan CLI
```bash
# Buat service discovery components
php artisan make:command RegisterService
php artisan make:command DeregisterService
php artisan make:command SendHeartbeat
php artisan make:middleware ServiceDiscovery
```

### Service Registry Model
```php
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Facades\Http;

class ServiceRegistry extends Model
{
    protected $table = 'service_registry';

    protected $fillable = [
        'service_name',
        'service_id',
        'host',
        'port',
        'health_check_url',
        'tags',
        'status',
        'last_heartbeat',
        'metadata',
    ];

    protected $casts = [
        'tags' => 'array',
        'metadata' => 'array',
        'last_heartbeat' => 'datetime',
    ];

    public function isHealthy(): bool
    {
        return $this->status === 'healthy'
            && $this->last_heartbeat
            && $this->last_heartbeat->diffInSeconds() < config('service-discovery.registry.ttl');
    }

    public function markUnhealthy(): void
    {
        $this->update(['status' => 'unhealthy']);
    }

    public function scopeHealthy($query)
    {
        $cutoff = now()->subSeconds(config('service-discovery.registry.ttl'));
        return $query->where('status', 'healthy')
            ->where('last_heartbeat', '>=', $cutoff);
    }

    public static function discover(string $serviceName): ?self
    {
        return static::healthy()->where('service_name', $serviceName)->first();
    }

    public static function discoverAll(string $serviceName): array
    {
        return static::healthy()
            ->where('service_name', $serviceName)
            ->get()
            ->toArray();
    }
}
```

### Service Registration Command
```php
<?php
namespace App\Console\Commands;

use App\Models\ServiceRegistry;
use Illuminate\Console\Command;

class RegisterService extends Command
{
    protected $signature = 'service:register';
    protected $description = 'Register this service in the registry';

    public function handle(): int
    {
        $config = config('service-discovery.this_service');

        $registry = ServiceRegistry::updateOrCreate(
            [
                'service_name' => $config['name'],
                'service_id' => gethostname(),
            ],
            [
                'host' => $config['host'],
                'port' => $config['port'],
                'health_check_url' => "http://{$config['host']}:{$config['port']}{$config['health_check']}",
                'tags' => $config['tags'],
                'status' => 'healthy',
                'last_heartbeat' => now(),
                'metadata' => [
                    'laravel_version' => app()->version(),
                    'php_version' => PHP_VERSION,
                    'environment' => app()->environment(),
                    'started_at' => now()->toISOString(),
                ],
            ]
        );

        $this->info("Service {$config['name']} registered successfully (ID: {$registry->id})");

        return self::SUCCESS;
    }
}
```

### Heartbeat Command
```php
<?php
namespace App\Console\Commands;

use App\Models\ServiceRegistry;
use Illuminate\Console\Command;

class SendHeartbeat extends Command
{
    protected $signature = 'service:heartbeat';
    protected $description = 'Send heartbeat to service registry';

    public function handle(): int
    {
        $config = config('service-discovery.this_service');

        $registry = ServiceRegistry::where('service_name', $config['name'])
            ->where('service_id', gethostname())
            ->first();

        if ($registry) {
            $registry->update([
                'last_heartbeat' => now(),
                'status' => 'healthy',
            ]);

            $this->info('Heartbeat sent successfully');
        } else {
            $this->warn('Service not registered. Running registration...');
            $this->call('service:register');
        }

        return self::SUCCESS;
    }
}
```

```php
use Illuminate\Console\Scheduling\Schedule;
use Illuminate\Foundation\Console\Kernel as ConsoleKernel;

class Kernel extends ConsoleKernel
{
    protected function schedule(Schedule $schedule): void
    {
        $schedule->command('service:heartbeat')
            ->everyTenSeconds()
            ->withoutOverlapping();

        $schedule->command('service:cleanup')
            ->everyMinute()
            ->withoutOverlapping();
    }
}
```

### Service Discovery Client (for ServiceMesh)
```php
<?php
namespace App\Services;

use App\Models\ServiceRegistry;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\Http;

class ServiceDiscoveryClient
{
    public function locate(string $serviceName): string
    {
        $cacheKey = "service_location:{$serviceName}";

        $location = Cache::remember($cacheKey, 10, function () use ($serviceName) {
            $instances = ServiceRegistry::discoverAll($serviceName);

            if (empty($instances)) {
                throw new ServiceNotFoundException("No healthy instances of {$serviceName}");
            }

            $instance = $this->loadBalance($instances);

            return "http://{$instance['host']}:{$instance['port']}";
        });

        return $location;
    }

    public function locateAll(string $serviceName): array
    {
        return ServiceRegistry::discoverAll($serviceName);
    }

    public function call(string $serviceName, string $method, string $path, array $data = []): mixed
    {
        $baseUrl = $this->locate($serviceName);

        $url = rtrim($baseUrl, '/') . '/' . ltrim($path, '/');

        $response = Http::withHeaders([
            'X-Service-Name' => config('app.name'),
            'X-Request-Id' => (string) str()->uuid(),
        ])->timeout(10)->$method($url, $data);

        if ($response->failed()) {
            Cache::forget("service_location:{$serviceName}");
            throw new ServiceCommunicationException(
                "Service {$serviceName} failed: {$response->status()}"
            );
        }

        return $response->json();
    }

    private function loadBalance(array $instances): array
    {
        return $instances[array_rand($instances)];
    }
}

class ServiceNotFoundException extends \RuntimeException {}
class ServiceCommunicationException extends \RuntimeException {}
```

### Health Check Controller
```php
<?php
namespace App\Http\Controllers;

use App\Models\ServiceRegistry;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Cache;

class HealthController extends Controller
{
    public function check(): mixed
    {
        $checks = [
            'database' => $this->checkDatabase(),
            'cache' => $this->checkCache(),
            'queue' => $this->checkQueue(),
            'storage' => $this->checkStorage(),
        ];

        $allHealthy = collect($checks)->every(fn ($check) => $check['status'] === 'healthy');

        $statusCode = $allHealthy ? 200 : 503;

        return response()->json([
            'service' => config('service-discovery.this_service.name'),
            'status' => $allHealthy ? 'healthy' : 'degraded',
            'checks' => $checks,
            'timestamp' => now()->toISOString(),
            'uptime' => $this->getUptime(),
        ], $statusCode);
    }

    public function info(): mixed
    {
        $thisService = config('service-discovery.this_service');

        $dependencies = [];
        foreach (config('service-discovery.services') as $name => $config) {
            try {
                $instances = ServiceRegistry::discoverAll($name);
                $dependencies[$name] = [
                    'healthy' => !empty($instances),
                    'instances' => count($instances),
                ];
            } catch (\Exception $e) {
                $dependencies[$name] = [
                    'healthy' => false,
                    'error' => $e->getMessage(),
                ];
            }
        }

        return response()->json([
            'service' => $thisService['name'],
            'host' => $thisService['host'],
            'port' => $thisService['port'],
            'environment' => app()->environment(),
            'dependencies' => $dependencies,
            'registered_services' => ServiceRegistry::select('service_name', 'status')
                ->distinct('service_name')
                ->get()
                ->groupBy('service_name')
                ->map(fn ($services) => [
                    'total' => $services->count(),
                    'healthy' => $services->where('status', 'healthy')->count(),
                ]),
        ]);
    }

    private function checkDatabase(): array
    {
        try {
            DB::connection()->getPdo();
            return ['status' => 'healthy'];
        } catch (\Exception $e) {
            return ['status' => 'unhealthy', 'error' => $e->getMessage()];
        }
    }

    private function checkCache(): array
    {
        try {
            Cache::store()->set('health_check', true, 1);
            return ['status' => 'healthy'];
        } catch (\Exception $e) {
            return ['status' => 'unhealthy', 'error' => $e->getMessage()];
        }
    }

    private function checkQueue(): array
    {
        try {
            app('queue')->connection()->size();
            return ['status' => 'healthy'];
        } catch (\Exception $e) {
            return ['status' => 'degraded', 'error' => $e->getMessage()];
        }
    }

    private function checkStorage(): array
    {
        $path = storage_path();
        return is_writable($path)
            ? ['status' => 'healthy']
            : ['status' => 'unhealthy', 'error' => 'Storage not writable'];
    }

    private function getUptime(): string
    {
        $started = cache()->get('service_started_at', now());
        return $started->diffForHumans();
    }
}
```

### Service Cleanup Command
```php
<?php
namespace App\Console\Commands;

use App\Models\ServiceRegistry;
use Illuminate\Console\Command;

class CleanupServices extends Command
{
    protected $signature = 'service:cleanup';
    protected $description = 'Remove stale service registrations';

    public function handle(): int
    {
        $cutoff = now()->subSeconds(
            config('service-discovery.registry.ttl') * 2
        );

        $stale = ServiceRegistry::where('last_heartbeat', '<', $cutoff)
            ->orWhere('status', 'unhealthy')
            ->get();

        $count = 0;
        foreach ($stale as $service) {
            $service->update(['status' => 'unhealthy']);
            $count++;
        }

        $this->info("Marked {$count} stale services as unhealthy");

        return self::SUCCESS;
    }
}
```
