# 422. Autentikasi dan Otorisasi Antar Service dengan JWT dan API Tokens

## Definisi & Target
Mengimplementasikan sistem autentikasi dan otorisasi yang aman untuk komunikasi antar microservices. Meliputi JWT (JSON Web Token) untuk user identity, API tokens untuk service-to-service, dan centralized authorization dengan policies.

## Komponen yang Terlibat
- JWT (JSON Web Token)
- API tokens (service-to-service)
- Public/Private key pairs
- Centralized auth service
- Token introspection
- Role-based access control (RBAC)
- Service identity verification
- Token caching & blacklisting

### User Service - JWT Auth Controller
```php
<?php
namespace App\Http\Controllers\Auth;

use App\Models\User;
use App\Services\JWTService;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Hash;
use Illuminate\Support\Facades\Validator;

class AuthController
{
    public function login(Request $request): mixed
    {
        $validator = Validator::make($request->all(), [
            'email' => 'required|email',
            'password' => 'required|string|min:6',
        ]);

        if ($validator->fails()) {
            return response()->json(['errors' => $validator->errors()], 422);
        }

        $user = User::where('email', $request->email)->first();

        if (!$user || !Hash::check($request->password, $user->password)) {
            return response()->json(['error' => 'Invalid credentials'], 401);
        }

        $token = JWTService::issueUserToken($user);

        return response()->json([
            'access_token' => $token,
            'token_type' => 'bearer',
            'expires_in' => config('jwt.ttl') * 60,
            'user' => [
                'id' => $user->id,
                'name' => $user->name,
                'email' => $user->email,
                'roles' => $user->roles->pluck('name'),
            ],
        ]);
    }

    public function me(Request $request): mixed
    {
        return response()->json(
            auth()->user()->load('roles.permissions')
        );
    }

    public function refresh(): mixed
    {
        $newToken = auth()->refresh();

        return response()->json([
            'access_token' => $newToken,
            'token_type' => 'bearer',
            'expires_in' => config('jwt.ttl') * 60,
        ]);
    }

    public function logout(): mixed
    {
        auth()->logout();

        return response()->json(['message' => 'Successfully logged out']);
    }

    public function introspect(Request $request): mixed
    {
        $token = $request->bearerToken();

        try {
            $payload = JWTService::introspectToken($token);

            return response()->json([
                'active' => true,
                'sub' => $payload['sub'],
                'user_id' => $payload['user_id'],
                'roles' => $payload['roles'],
                'service' => $payload['service'] ?? null,
                'exp' => $payload['exp'],
                'iat' => $payload['iat'],
            ]);
        } catch (\Exception $e) {
            return response()->json(['active' => false, 'error' => $e->getMessage()], 401);
        }
    }
}
```

### JWT Service
```php
<?php
namespace App\Services;

use App\Models\User;
use PHPOpenSourceSaver\JWTAuth\Facades\JWTAuth;
use Illuminate\Support\Str;

class JWTService
{
    public static function issueUserToken(User $user): string
    {
        $claims = [
            'user_id' => $user->id,
            'roles' => $user->roles->pluck('name')->toArray(),
            'service' => null,
        ];

        return JWTAuth::claims($claims)->fromUser($user);
    }

    public static function issueServiceToken(string $serviceName, array $permissions = []): string
    {
        $claims = [
            'user_id' => null,
            'roles' => $permissions,
            'service' => $serviceName,
            'sub' => 'service:' . $serviceName,
        ];

        return JWTAuth::claims($claims)->fromUser(
            new User(['id' => 0, 'email' => "{$serviceName}@services.local"])
        );
    }

    public static function introspectToken(string $token): array
    {
        $payload = JWTAuth::setToken($token)->getPayload();

        return [
            'sub' => $payload->get('sub'),
            'user_id' => $payload->get('user_id'),
            'roles' => $payload->get('roles', []),
            'service' => $payload->get('service'),
            'exp' => $payload->get('exp'),
            'iat' => $payload->get('iat'),
            'jti' => $payload->get('jti'),
        ];
    }

    public static function blacklistToken(string $token): void
    {
        JWTAuth::setToken($token)->invalidate(true);
    }

    public static function isServiceToken(array $payload): bool
    {
        return $payload['service'] !== null;
    }

    public static function hasRole(array $payload, string $role): bool
    {
        return in_array($role, $payload['roles'] ?? []);
    }

    public static function hasPermission(array $payload, string $permission): bool
    {
        return in_array($permission, $payload['roles'] ?? []);
    }
}
```

### Service-to-Service API Token Authentication
```php
<?php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Cache;

class ServiceAuth
{
    private array $allowedServices = [
        'user-service',
        'order-service',
        'product-service',
        'payment-service',
        'notification-service',
        'api-gateway',
    ];

    public function handle(Request $request, Closure $next): mixed
    {
        $serviceToken = $request->header('X-Service-Token');
        $serviceName = $request->header('X-Service-Name');

        if (!$serviceToken || !$serviceName) {
            return response()->json(['error' => 'Service credentials required'], 401);
        }

        if (!in_array($serviceName, $this->allowedServices)) {
            return response()->json(['error' => 'Unknown service'], 403);
        }

        $expectedToken = config("services.{$serviceName}.token");

        if (!$expectedToken || !hash_equals($expectedToken, $serviceToken)) {
            return response()->json(['error' => 'Invalid service token'], 403);
        }

        $request->merge([
            'authenticated_service' => $serviceName,
        ]);

        return $next($request);
    }
}

// Middleware for Token Introspection
class IntrospectToken
{
    public function handle(Request $request, Closure $next): mixed
    {
        $token = $request->bearerToken();

        if (!$token) {
            return response()->json(['error' => 'Token required'], 401);
        }

        $cacheKey = 'introspected_token:' . md5($token);

        $payload = Cache::remember($cacheKey, 300, function () use ($token) {
            $response = Http::withToken($token)
                ->timeout(5)
                ->get(env('USER_SERVICE_URL') . '/api/auth/introspect');

            if (!$response->successful()) {
                return null;
            }

            return $response->json();
        });

        if (!$payload || !($payload['active'] ?? false)) {
            return response()->json(['error' => 'Invalid or expired token'], 401);
        }

        $request->merge([
            'token_payload' => $payload,
            'authenticated_user_id' => $payload['user_id'],
            'user_roles' => $payload['roles'] ?? [],
        ]);

        return $next($request);
    }
}
```

### RBAC Implementation
```php
<?php
namespace App\Services;

use App\Models\Role;
use App\Models\Permission;
use Illuminate\Support\Facades\Cache;

class AuthorizationService
{
    public function can(string $action, string $resource, array $userContext): bool
    {
        $roles = $userContext['roles'] ?? [];

        foreach ($roles as $roleName) {
            $permissions = $this->getRolePermissions($roleName);

            if (in_array("{$action}:{$resource}", $permissions)) {
                return true;
            }
        }

        return false;
    }

    public function authorize(string $action, string $resource, array $userContext): void
    {
        if (!$this->can($action, $resource, $userContext)) {
            throw new AuthorizationException(
                "Unauthorized: {$action} on {$resource}"
            );
        }
    }

    public function authorizeService(string $serviceName, string $action): void
    {
        $allowedActions = config("service-permissions.{$serviceName}", []);

        if (!in_array($action, $allowedActions)) {
            throw new AuthorizationException(
                "Service {$serviceName} not allowed to perform {$action}"
            );
        }
    }

    private function getRolePermissions(string $roleName): array
    {
        return Cache::remember("role_permissions:{$roleName}", 3600, function () use ($roleName) {
            $role = Role::where('name', $roleName)
                ->with('permissions')
                ->first();

            if (!$role) {
                return [];
            }

            return $role->permissions
                ->map(fn ($perm) => "{$perm->action}:{$perm->resource}")
                ->toArray();
        });
    }
}

class AuthorizationException extends \RuntimeException {}
```

### Routes with Authentication
```php
<?php
// routes/api.php
use Illuminate\Support\Facades\Route;
use App\Http\Controllers\Auth\AuthController;
use App\Http\Controllers\OrderController;

// Public routes
Route::post('auth/login', [AuthController::class, 'login']);
Route::post('auth/refresh', [AuthController::class, 'refresh']);

// Protected routes - user token
Route::middleware(['auth:api', 'introspect'])->group(function () {
    Route::get('auth/me', [AuthController::class, 'me']);
    Route::post('auth/logout', [AuthController::class, 'logout']);

    Route::apiResource('orders', OrderController::class);
});

// Internal routes - service token only
Route::prefix('internal')->middleware(['service.auth'])->group(function () {
    Route::post('orders', [OrderController::class, 'storeInternal']);
    Route::get('orders/pending-payment', [OrderController::class, 'pendingPayment']);
    Route::post('verify-token', [AuthController::class, 'introspect']);
});
```

### Token Generation for Services
```php
<?php
namespace App\Console\Commands;

use App\Models\ServiceCredential;
use Illuminate\Console\Command;
use Illuminate\Support\Str;

class GenerateServiceToken extends Command
{
    protected $signature = 'service:generate-token {service_name}';
    protected $description = 'Generate API token for a service';

    public function handle(): int
    {
        $serviceName = $this->argument('service_name');
        $token = Str::random(64);

        ServiceCredential::updateOrCreate(
            ['service_name' => $serviceName],
            ['token' => hash('sha256', $token)]
        );

        $this->info("Service token for {$serviceName}:");
        $this->warn($token);
        $this->info("Store this token securely. It won't be shown again.");

        return self::SUCCESS;
    }
}
```
