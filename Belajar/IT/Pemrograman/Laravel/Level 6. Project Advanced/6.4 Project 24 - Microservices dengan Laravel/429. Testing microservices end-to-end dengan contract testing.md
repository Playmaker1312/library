# 429. Testing Microservices End-to-End dengan Contract Testing

## Definisi & Target
Mengimplementasikan strategi testing untuk microservices: unit test per service, integration test antar service, contract test (PACT) untuk API compatibility, dan end-to-end test untuk flow lengkap. Targetnya adalah memastikan setiap service tetap compatible selama development independen.

## Komponen yang Terlibat
- PHPUnit (unit & feature tests)
- PACT (contract testing)
- Docker Compose (integration test env)
- HttpFake / Mocking (Guzzle)
- Testcontainers (optional)
- API Blueprint / OpenAPI
- Smoke tests
- Chaos engineering

## Perintah Artisan CLI
```bash
# Install testing packages
composer require --dev phpunit/phpunit
composer require --dev pact-foundation/pact-php

# Make test files
php artisan make:test UserServiceTest
php artisan make:test OrderFlowTest
php artisan make:test ContractTest

# Create Pact files
php artisan pact:init
```

### Contract Test - Consumer Side (PACT)
```php
<?php
namespace Tests\Contract\Consumer;

use PHPUnit\Framework\TestCase;
use PhpPact\Consumer\Model\ConsumerRequest;
use PhpPact\Consumer\Model\ProviderResponse;
use PhpPact\Consumer\InteractionBuilder;
use PhpPact\Standalone\MockService\MockServerEnvConfig;
use Illuminate\Support\Facades\Http;

class UserServiceContractTest extends TestCase
{
    private MockServerEnvConfig $config;
    private InteractionBuilder $builder;

    protected function setUp(): void
    {
        parent::setUp();

        $this->config = new MockServerEnvConfig([
            'consumer' => 'OrderService',
            'provider' => 'UserService',
            'pact_dir' => storage_path('pacts'),
        ]);

        $this->builder = new InteractionBuilder($this->config);
    }

    public function testGetUserReturnsValidResponse(): void
    {
        $request = new ConsumerRequest();
        $request->setMethod('GET');
        $request->setPath('/api/users/1');
        $request->addHeader('Authorization', 'Bearer test-token');
        $request->addHeader('Accept', 'application/json');

        $response = new ProviderResponse();
        $response->setStatus(200);
        $response->addHeader('Content-Type', 'application/json');
        $response->setBody([
            'id' => 1,
            'name' => 'John Doe',
            'email' => 'john@example.com',
            'roles' => ['customer'],
        ]);

        $this->builder
            ->given('a user exists with ID 1')
            ->uponReceiving('a request for user details')
            ->with($request)
            ->willRespondWith($response);

        $result = Http::withToken('test-token')
            ->get("{$this->config->getBaseUri()}/api/users/1");

        $this->assertEquals(200, $result->status());
        $this->assertArrayHasKey('id', $result->json());
        $this->assertArrayHasKey('name', $result->json());
        $this->assertArrayHasKey('email', $result->json());

        $this->builder->verify();
    }

    public function testCreateUserReturnsCreatedResource(): void
    {
        $request = new ConsumerRequest();
        $request->setMethod('POST');
        $request->setPath('/api/users');
        $request->addHeader('Content-Type', 'application/json');
        $request->addHeader('Authorization', 'Bearer admin-token');
        $request->setBody([
            'name' => 'Jane Doe',
            'email' => 'jane@example.com',
            'password' => 'secret123',
            'roles' => ['customer'],
        ]);

        $response = new ProviderResponse();
        $response->setStatus(201);
        $response->addHeader('Content-Type', 'application/json');
        $response->setBody([
            'id' => 2,
            'name' => 'Jane Doe',
            'email' => 'jane@example.com',
            'roles' => ['customer'],
            'created_at' => '2024-01-15T10:00:00Z',
        ]);

        $this->builder
            ->given('the system is ready to create users')
            ->uponReceiving('a request to create a new user')
            ->with($request)
            ->willRespondWith($response);

        $result = Http::withToken('admin-token')
            ->post("{$this->config->getBaseUri()}/api/users", [
                'name' => 'Jane Doe',
                'email' => 'jane@example.com',
                'password' => 'secret123',
                'roles' => ['customer'],
            ]);

        $this->assertEquals(201, $result->status());
        $this->assertEquals('Jane Doe', $result->json('name'));

        $this->builder->verify();
    }
}
```

### Integration Test Environment Setup
```php
<?php
namespace Tests\Integration;

use Tests\TestCase;
use Illuminate\Support\Facades\Http;
use Illuminate\Foundation\Testing\RefreshDatabase;

class IntegrationTestCase extends TestCase
{
    protected array $serviceUrls = [];

    protected function setUp(): void
    {
        parent::setUp();

        $this->serviceUrls = [
            'user' => env('USER_SERVICE_URL', 'http://localhost:8001'),
            'product' => env('PRODUCT_SERVICE_URL', 'http://localhost:8002'),
            'order' => env('ORDER_SERVICE_URL', 'http://localhost:8003'),
            'payment' => env('PAYMENT_SERVICE_URL', 'http://localhost:8004'),
        ];

        Http::preventStrayRequests();
    }

    protected function callService(string $service, string $method, string $path, array $data = []): mixed
    {
        $url = rtrim($this->serviceUrls[$service], '/') . '/' . ltrim($path, '/');

        return Http::withHeaders([
            'X-Test-Mode' => 'true',
            'X-Service-Name' => config('app.name'),
        ])->$method($url, $data);
    }
}
```

### End-to-End Test for Order Flow
```php
<?php
namespace Tests\E2E;

use Tests\Integration\IntegrationTestCase;
use Illuminate\Support\Facades\Http;

class OrderFlowE2ETest extends IntegrationTestCase
{
    public function testCompleteOrderFlow(): void
    {
        // 1. Create user
        $user = $this->callService('user', 'post', '/api/test/users', [
            'name' => 'Test Customer',
            'email' => 'test-' . uniqid() . '@example.com',
            'password' => 'password',
        ])->json();

        $this->assertArrayHasKey('id', $user);
        $userId = $user['id'];

        // 2. Authenticate
        $auth = $this->callService('user', 'post', '/api/auth/login', [
            'email' => $user['email'],
            'password' => 'password',
        ])->json();

        $this->assertArrayHasKey('access_token', $auth);
        $token = $auth['access_token'];

        // 3. Get products
        $products = $this->callService('product', 'get', '/api/test/products', [])->json();
        $this->assertNotEmpty($products['data']);
        $product = $products['data'][0];

        // 4. Create order
        $order = Http::withToken($token)->post(
            $this->serviceUrls['order'] . '/api/orders',
            [
                'items' => [
                    ['product_id' => $product['id'], 'quantity' => 2],
                ],
                'shipping_address' => [
                    'street' => 'Jl. Merdeka No. 1',
                    'city' => 'Jakarta',
                    'postal_code' => '10110',
                ],
            ]
        )->json();

        $this->assertArrayHasKey('id', $order);
        $orderId = $order['id'];

        // 5. Verify order status
        $this->assertEquals('pending', $order['status']);

        // 6. Check stock reduced
        $updatedProduct = $this->callService('product', 'get', "/api/test/products/{$product['id']}")->json();
        $this->assertLessThan($product['stock'], $updatedProduct['stock']);

        // 7. Process payment
        $payment = Http::withToken($token)->post(
            $this->serviceUrls['payment'] . '/api/payments',
            [
                'order_id' => $orderId,
                'amount' => $order['total_amount'],
                'method' => 'bank_transfer',
            ]
        )->json();

        $this->assertArrayHasKey('id', $payment);
        $this->assertEquals('completed', $payment['status']);

        // 8. Verify notification was sent
        $notifications = $this->callService('user', 'get', "/api/test/users/{$userId}/notifications")->json();
        $this->assertNotEmpty($notifications);

        // 9. Verify order status updated to confirmed
        $updatedOrder = Http::withToken($token)->get(
            $this->serviceUrls['order'] . "/api/orders/{$orderId}"
        )->json();

        $this->assertEquals('confirmed', $updatedOrder['status']);
    }
}
```

### Service Mock for Testing
```php
<?php
namespace Tests\Mocks;

use Illuminate\Support\Facades\Http;
use Illuminate\Http\Client\Request;

class ServiceMockManager
{
    private array $mocks = [];

    public function mockUserService(): void
    {
        Http::fake([
            'user-service*' => function (Request $request) {
                return match (true) {
                    $request->method() === 'GET' && preg_match('#/api/users/(\d+)#', $request->url()) => $this->response([
                        'id' => (int) $request->segment(3),
                        'name' => 'Mock User',
                        'email' => 'mock@example.com',
                        'roles' => ['customer'],
                    ]),
                    $request->method() === 'POST' && str_contains($request->url(), '/api/auth/login') => $this->response([
                        'access_token' => 'mock-token-' . uniqid(),
                        'token_type' => 'bearer',
                        'expires_in' => 3600,
                    ]),
                    default => $this->response(['error' => 'Not mocked'], 404),
                };
            },
        ]);
    }

    public function mockProductService(): void
    {
        Http::fake([
            'product-service*' => function (Request $request) {
                return match (true) {
                    $request->method() === 'GET' && str_contains($request->url(), '/api/products') => $this->response([
                        'data' => [
                            ['id' => 1, 'name' => 'Product A', 'price' => 50000, 'stock' => 100],
                            ['id' => 2, 'name' => 'Product B', 'price' => 75000, 'stock' => 50],
                        ],
                    ]),
                    default => $this->response(['error' => 'Not mocked'], 404),
                };
            },
        ]);
    }

    public function mockAllServices(): void
    {
        $this->mockUserService();
        $this->mockProductService();

        Http::fake([
            'order-service*' => Http::response(['id' => 1, 'status' => 'pending', 'total_amount' => 100000], 201),
            'payment-service*' => Http::response(['id' => 1, 'status' => 'completed'], 200),
            'notification-service*' => Http::response(['status' => 'sent'], 200),
        ]);
    }

    private function response(array $data, int $status = 200): \Illuminate\Http\Client\Response
    {
        return Http::response($data, $status);
    }
}
```

### Chaos Engineering Test
```php
<?php
namespace Tests\Chaos;

use Tests\TestCase;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Cache;

class ChaosEngineeringTest extends TestCase
{
    public function testServiceDegradationWhenUserServiceDown(): void
    {
        Http::fake([
            'user-service*' => Http::response(null, 503),
            'product-service*' => Http::response(['data' => ['id' => 1, 'name' => 'Test Product']], 200),
        ]);

        $response = Http::get(env('API_GATEWAY_URL') . '/api/v1/products/1');

        $this->assertEquals(200, $response->status());
        // Should still return product data, even without user context
    }

    public function testOrderCreationWhenPaymentServiceSlow(): void
    {
        Http::fake([
            'user-service*' => Http::response(['id' => 1, 'name' => 'User']),
            'product-service*' => Http::response(['id' => 1, 'stock' => 10]),
            'payment-service*' => function () {
                sleep(35); // Simulate timeout
                return Http::response(null, 504);
            },
        ]);

        $start = microtime(true);

        $response = Http::post(env('ORDER_SERVICE_URL') . '/api/orders', [
            'user_id' => 1,
            'items' => [['product_id' => 1, 'quantity' => 1]],
        ]);

        $duration = microtime(true) - $start;

        // Should timeout quickly and not hang for 35 seconds
        $this->assertLessThan(15, $duration, "Request took too long: {$duration}s");
    }

    public function testCacheHitDuringServiceOutage(): void
    {
        Cache::put('product:1', [
            'id' => 1,
            'name' => 'Cached Product',
            'price' => 50000,
        ], 3600);

        Http::fake([
            'product-service*' => Http::response(null, 503),
        ]);

        $response = Http::get(env('PRODUCT_SERVICE_URL') . '/api/products/1');

        $this->assertEquals(200, $response->status());
        $data = $response->json();
        $this->assertArrayHasKey('_meta', $data);
        $this->assertEquals('stale_cache', $data['_meta']['source']);
    }
}
```

### Contract Validation Command
```php
<?php
namespace App\Console\Commands;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\Http;

class ValidateContracts extends Command
{
    protected $signature = 'contracts:validate {service?}';
    protected $description = 'Validate API contracts against live services';

    public function handle(): int
    {
        $services = [
            'user-service' => [
                'endpoints' => [
                    'GET /api/health' => ['status' => 200],
                    'GET /api/users/1' => ['status' => 401],
                ],
            ],
            'product-service' => [
                'endpoints' => [
                    'GET /api/health' => ['status' => 200],
                    'GET /api/products' => ['status' => 200],
                ],
            ],
            'order-service' => [
                'endpoints' => [
                    'GET /api/health' => ['status' => 200],
                ],
            ],
        ];

        $filter = $this->argument('service');

        foreach ($services as $name => $config) {
            if ($filter && $name !== $filter) continue;

            $this->info("Validating {$name}...");

            $url = env(strtoupper(str_replace('-', '_', $name)) . '_URL');

            foreach ($config['endpoints'] as $endpoint => $expectations) {
                [$method, $path] = explode(' ', $endpoint, 2);

                try {
                    $response = Http::timeout(5)->$method(rtrim($url, '/') . $path);
                    $statusOk = $response->status() === $expectations['status'];
                    $marker = $statusOk ? '✓' : '✗';
                    $this->info("  {$marker} {$method} {$path}: expected {$expectations['status']}, got {$response->status()}");
                } catch (\Exception $e) {
                    $this->error("  ✗ {$method} {$path}: {$e->getMessage()}");
                }
            }
        }

        return self::SUCCESS;
    }
}
```

### API Specification Test
```php
<?php
namespace Tests\Contract;

use Tests\TestCase;
use Illuminate\Support\Facades\Http;

class ApiSpecificationTest extends TestCase
{
    private array $spec;

    protected function setUp(): void
    {
        parent::setUp();
        $this->spec = json_decode(
            file_get_contents(base_path('docs/openapi.yaml')),
            true
        );
    }

    public function testAllSpecifiedEndpointsAreReachable(): void
    {
        $paths = array_keys($this->spec['paths'] ?? []);

        foreach ($paths as $path) {
            $url = env('API_GATEWAY_URL') . $path;

            $response = Http::timeout(3)->options($url);

            $this->assertContains(
                $response->status(),
                [200, 204, 401, 403],
                "OPTIONS {$path} failed with {$response->status()}"
            );
        }
    }

    public function testResponseMatchesSchema(): void
    {
        $testCases = [
            ['GET', '/api/v1/products', 200, '#/components/schemas/ProductList'],
            ['GET', '/api/v1/products/1', 401, null],
        ];

        foreach ($testCases as [$method, $path, $expectedStatus, $schema]) {
            $response = Http::timeout(5)->$method(env('API_GATEWAY_URL') . $path);

            $this->assertEquals($expectedStatus, $response->status(), "{$method} {$path}");

            if ($schema) {
                $this->validateSchema($response->json(), $schema);
            }
        }
    }

    private function validateSchema(array $data, string $schemaRef): void
    {
        $schemaPath = str_replace('#/', '', $schemaRef);
        $keys = explode('/', $schemaPath);
        $schema = $this->spec;

        foreach ($keys as $key) {
            $schema = $schema[$key] ?? [];
        }

        $required = $schema['required'] ?? [];

        foreach ($required as $field) {
            $this->assertArrayHasKey($field, $data, "Missing required field: {$field}");
        }
    }
}
```
