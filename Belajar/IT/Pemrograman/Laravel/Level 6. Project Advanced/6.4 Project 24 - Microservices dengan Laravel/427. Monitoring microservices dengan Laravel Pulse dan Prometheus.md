# 427. Monitoring Microservices dengan Laravel Pulse dan Prometheus

## Definisi & Target
Mengimplementasikan monitoring untuk seluruh microservices menggunakan Laravel Pulse untuk metrics internal dan Prometheus/Grafana untuk infrastructure monitoring. Meliputi metrics collection, alerting, dashboards, dan SLI/SLO tracking.

## Komponen yang Terlibat
- Laravel Pulse (internal metrics)
- Prometheus (metrics collection)
- Grafana (visualization)
- Node Exporter (system metrics)
- cAdvisor (container metrics)
- Alertmanager
- SLI/SLO definition
- Custom metrics exporters

## Perintah Artisan CLI
```bash
# Install Laravel Pulse
composer require laravel/pulse
php artisan vendor:publish --provider="Laravel\Pulse\PulseServiceProvider"
php artisan migrate

# Install Prometheus exporter
composer require promphp/prometheus_push_gateway

# Pulse commands
php artisan pulse:check

# Generate test metrics
php artisan make:command SimulateTraffic
```

### Custom Pulse Recorder for Microservices
```php
<?php
namespace App\Pulse\Recorders;

use Laravel\Pulse\Contracts\Recorder;
use Laravel\Pulse\Pulse;
use Illuminate\Support\Facades\Http;

class ServiceHealthRecorder implements Recorder
{
    private Pulse $pulse;

    public function __construct(Pulse $pulse)
    {
        $this->pulse = $pulse;
    }

    public function record(): void
    {
        $services = [
            'user-service' => env('USER_SERVICE_URL'),
            'product-service' => env('PRODUCT_SERVICE_URL'),
            'order-service' => env('ORDER_SERVICE_URL'),
            'payment-service' => env('PAYMENT_SERVICE_URL'),
            'notification-service' => env('NOTIFICATION_SERVICE_URL'),
        ];

        foreach ($services as $name => $url) {
            $healthy = false;
            $responseTime = 0;

            try {
                $start = microtime(true);
                $response = Http::timeout(5)->get(rtrim($url, '/') . '/api/health');
                $responseTime = (microtime(true) - $start) * 1000;
                $healthy = $response->successful();
            } catch (\Exception $e) {
                $healthy = false;
            }

            $this->pulse->record(
                'service_health',
                json_encode([
                    'service' => $name,
                    'healthy' => $healthy,
                    'response_time_ms' => round($responseTime, 2),
                    'checked_at' => now()->toISOString(),
                ])
            )->count();
        }
    }

    public function aggregate(): void
    {
        // Aggregate health metrics over time
    }
}
```

### Prometheus Metrics Export
```php
<?php
namespace App\Http\Controllers;

use Prometheus\CollectorRegistry;
use Prometheus\RenderTextFormat;
use Prometheus\Storage\Redis;

class MetricsController extends Controller
{
    private CollectorRegistry $registry;

    public function __construct()
    {
        $adapter = new Redis(['host' => env('REDIS_HOST', '127.0.0.1')]);
        $this->registry = new CollectorRegistry($adapter);
    }

    public function prometheus(): mixed
    {
        // HTTP request counter
        $requestCounter = $this->registry->registerCounter(
            'app_http_requests_total',
            'Total HTTP requests',
            ['method', 'endpoint', 'status', 'service']
        );

        // Request duration histogram
        $requestDuration = $this->registry->registerHistogram(
            'app_http_request_duration_seconds',
            'HTTP request duration in seconds',
            ['method', 'endpoint', 'service'],
            [0.01, 0.05, 0.1, 0.5, 1, 2, 5]
        );

        // Active users gauge
        $activeUsers = $this->registry->registerGauge(
            'app_active_users_total',
            'Total active users',
            ['service']
        );

        // Database query counter
        $dbQueries = $this->registry->registerCounter(
            'app_db_queries_total',
            'Total database queries',
            ['connection', 'service']
        );

        // Queue size gauge
        $queueSize = $this->registry->registerGauge(
            'app_queue_size',
            'Current queue size',
            ['queue', 'service']
        );

        // Service health gauge
        $serviceHealth = $this->registry->registerGauge(
            'app_service_health',
            'Service health status (1=healthy, 0=unhealthy)',
            ['service', 'target_service']
        );

        $renderer = new RenderTextFormat();
        $result = $renderer->render($this->registry->getMetricFamilySamples());

        return response($result, 200, ['Content-Type' => RenderTextFormat::MIME_TYPE]);
    }

    // Record metrics from middleware
    public static function recordHttpMetrics(string $method, string $endpoint, int $status, float $duration): void
    {
        try {
            $adapter = new Redis(['host' => env('REDIS_HOST', '127.0.0.1')]);
            $registry = new CollectorRegistry($adapter);

            $counter = $registry->registerCounter(
                'app_http_requests_total',
                'Total HTTP requests',
                ['method', 'endpoint', 'status', 'service']
            );
            $counter->incBy(1, [$method, $endpoint, $status, config('app.name')]);

            $histogram = $registry->registerHistogram(
                'app_http_request_duration_seconds',
                'HTTP request duration in seconds',
                ['method', 'endpoint', 'service'],
                [0.01, 0.05, 0.1, 0.5, 1, 2, 5]
            );
            $histogram->observe($duration, [$method, $endpoint, config('app.name')]);
        } catch (\Exception $e) {
            // Fail silently - metrics should not break the app
        }
    }
}
```

### Metrics Recording Middleware
```php
<?php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use App\Http\Controllers\MetricsController;

class RecordMetrics
{
    public function handle(Request $request, Closure $next): mixed
    {
        $start = microtime(true);

        $response = $next($request);

        $duration = microtime(true) - $start;

        MetricsController::recordHttpMetrics(
            $request->method(),
            $request->path(),
            $response->status(),
            $duration
        );

        return $response;
    }
}
```

### Alert Rules for Prometheus
```yaml
# prometheus/alerts.yml
groups:
  - name: microservices-alerts
    rules:
      - alert: ServiceDown
        expr: app_service_health == 0
        for: 1m
        labels:
          severity: critical
        annotations:
          summary: "Service {{ $labels.target_service }} is down"
          description: "{{ $labels.target_service }} has been unreachable for more than 1 minute"

      - alert: HighErrorRate
        expr: rate(app_http_requests_total{status=~"5.."}[5m]) / rate(app_http_requests_total[5m]) * 100 > 1
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "High error rate on {{ $labels.service }}"
          description: "Error rate is {{ $value }}% for the last 5 minutes"

      - alert: HighLatency
        expr: histogram_quantile(0.99, rate(app_http_request_duration_seconds_bucket[5m])) > 3
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "High latency on {{ $labels.service }}"
          description: "P99 latency is {{ $value }}s"

      - alert: QueueGrowing
        expr: app_queue_size > 1000
        for: 5m
        labels:
          severity: warning
        annotations:
          summary: "Queue {{ $labels.queue }} is growing"
          description: "Queue size is {{ $value }}"

      - alert: HighMemoryUsage
        expr: (node_memory_MemTotal_bytes - node_memory_MemAvailable_bytes) / node_memory_MemTotal_bytes * 100 > 90
        for: 5m
        labels:
          severity: critical
        annotations:
          summary: "High memory usage on {{ $labels.instance }}"
          description: "Memory usage is {{ $value }}%"

      - alert: ErrorBudgetBurned
        expr: (1 - (sum(rate(app_http_requests_total{status=~"5.."}[30d])) / sum(rate(app_http_requests_total[30d])))) < 0.999
        labels:
          severity: critical
        annotations:
          summary: "Error budget burned for {{ $labels.service }}"
          description: "SLO of 99.9% is at risk"
```

### Custom Pulse Dashboard Card
```php
<?php
namespace App\Pulse\Cards;

use Laravel\Pulse\Pulse;
use Laravel\Pulse\Livewire\Card;

class ServiceHealthCard extends Card
{
    public function render(): mixed
    {
        $services = [
            'User Service' => env('USER_SERVICE_URL'),
            'Product Service' => env('PRODUCT_SERVICE_URL'),
            'Order Service' => env('ORDER_SERVICE_URL'),
            'Payment Service' => env('PAYMENT_SERVICE_URL'),
            'Notification Service' => env('NOTIFICATION_SERVICE_URL'),
        ];

        $healthData = [];

        foreach ($services as $name => $url) {
            try {
                $start = microtime(true);
                $response = \Illuminate\Support\Facades\Http::timeout(3)->get(rtrim($url, '/') . '/api/health');
                $latency = (microtime(true) - $start) * 1000;

                $healthData[] = [
                    'name' => $name,
                    'healthy' => $response->successful(),
                    'latency' => round($latency, 0),
                    'status' => $response->json('status', 'unknown'),
                ];
            } catch (\Exception $e) {
                $healthData[] = [
                    'name' => $name,
                    'healthy' => false,
                    'latency' => null,
                    'status' => 'unreachable',
                ];
            }
        }

        return view('pulse-cards.service-health', [
            'services' => $healthData,
        ]);
    }
}
```

### Health Check Aggregation Endpoint
```php
<?php
namespace App\Http\Controllers;

use Illuminate\Http\Request;
use Illuminate\Support\Facades\Http;

class HealthAggregationController extends Controller
{
    public function all(): mixed
    {
        $services = [
            'user-service' => env('USER_SERVICE_URL'),
            'product-service' => env('PRODUCT_SERVICE_URL'),
            'order-service' => env('ORDER_SERVICE_URL'),
            'payment-service' => env('PAYMENT_SERVICE_URL'),
            'notification-service' => env('NOTIFICATION_SERVICE_URL'),
        ];

        $results = [];

        foreach ($services as $name => $url) {
            try {
                $response = Http::timeout(5)->get(rtrim($url, '/') . '/api/health');
                $results[$name] = [
                    'status' => $response->successful() ? 'healthy' : 'degraded',
                    'http_status' => $response->status(),
                ];
            } catch (\Exception $e) {
                $results[$name] = [
                    'status' => 'unhealthy',
                    'error' => $e->getMessage(),
                ];
            }
        }

        $allHealthy = collect($results)->every(fn ($r) => $r['status'] === 'healthy');

        return response()->json([
            'status' => $allHealthy ? 'healthy' : 'degraded',
            'services' => $results,
            'timestamp' => now()->toISOString(),
        ], $allHealthy ? 200 : 503);
    }
}
```
