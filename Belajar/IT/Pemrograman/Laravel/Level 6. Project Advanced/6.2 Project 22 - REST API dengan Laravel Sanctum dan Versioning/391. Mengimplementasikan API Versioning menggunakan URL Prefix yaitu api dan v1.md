# 391. Mengimplementasikan API Versioning Menggunakan URL Prefix yaitu api/v1 dan api/v2

## Definisi & Target
Mengimplementasikan versioning API menggunakan URL prefix (/api/v1/, /api/v2/) sehingga multiple versi API bisa berjalan bersamaan. Setiap versi memiliki controller, resource, dan logic sendiri yang independen. Targetnya adalah backward compatibility dan transisi mulus antar versi.

## Komponen yang Terlibat
- Route groups dengan prefix version
- Version-specific controllers
- Version-specific API Resources
- Version negotiation via header atau URL
- Default version fallback
- Version deprecation system

## Perintah Artisan CLI
```bash
# Membuat struktur folder
mkdir -p app/Http/Controllers/Api/{V1,V2}
mkdir -p app/Http/Resources/{V1,V2}
mkdir -p routes/api

# Membuat route files
touch routes/api/v1.php routes/api/v2.php

# Membuat controller per versi
php artisan make:controller Api/V1/ProductController --api
php artisan make:controller Api/V2/ProductController --api

# Membuat resource per versi
php artisan make:resource V1/ProductResource
php artisan make:resource V2/ProductResource
php artisan make:resource V1/ProductCollection
php artisan make:resource V2/ProductCollection
```

### Version Helper - AppServiceProvider
```php
<?php
namespace App\Providers;

use Illuminate\Support\ServiceProvider;
use Illuminate\Support\Facades\Request;

class AppServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        Request::macro('apiVersion', function () {
            return $this->segment(2) ?? 'v1';
        });

        Request::macro('isApiVersion', function (string $version) {
            return $this->apiVersion() === $version;
        });
    }
}
```

### Version-Specific Controller - V1/ProductController
```php
<?php
namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Http\Resources\V1\ProductResource;
use App\Models\Product;
use Illuminate\Http\Request;

class ProductController extends Controller
{
    public function index()
    {
        $products = Product::paginate(15);

        return ProductResource::collection($products);
    }

    public function show(Product $product)
    {
        return new ProductResource($product);
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'price' => 'required|numeric|min:0',
            'description' => 'nullable|string',
        ]);

        $product = Product::create($validated);

        return new ProductResource($product);
    }

    public function update(Request $request, Product $product)
    {
        $validated = $request->validate([
            'name' => 'string|max:255',
            'price' => 'numeric|min:0',
            'description' => 'nullable|string',
        ]);

        $product->update($validated);

        return new ProductResource($product);
    }

    public function destroy(Product $product)
    {
        $product->delete();

        return response()->json(['message' => 'Product deleted.']);
    }
}
```

### Version-Specific Controller - V2/ProductController (enhanced)
```php
<?php
namespace App\Http\Controllers\Api\V2;

use App\Http\Controllers\Controller;
use App\Http\Resources\V2\ProductResource;
use App\Models\Product;
use Illuminate\Http\Request;

class ProductController extends Controller
{
    public function index(Request $request)
    {
        $query = Product::with('category', 'variants');

        if ($request->category) {
            $query->where('category_id', $request->category);
        }

        if ($request->search) {
            $query->where(function ($q) use ($request) {
                $q->where('name', 'like', "%{$request->search}%")
                  ->orWhere('description', 'like', "%{$request->search}%");
            });
        }

        if ($request->sort) {
            $query->orderBy($request->sort, $request->order ?? 'asc');
        }

        $products = $query->cursorPaginate($request->per_page ?? 15);

        return ProductResource::collection($products);
    }

    public function show(Product $product)
    {
        $product->load('category', 'variants', 'reviews');

        return new ProductResource($product);
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'price' => 'required|numeric|min:0',
            'compare_price' => 'nullable|numeric|gt:price',
            'description' => 'nullable|string',
            'category_id' => 'nullable|exists:categories,id',
            'sku' => 'nullable|string|unique:products,sku',
            'tags' => 'nullable|array',
            'tags.*' => 'string',
            'variants' => 'nullable|array',
            'variants.*.name' => 'required|string',
            'variants.*.price' => 'required|numeric',
        ]);

        $product = Product::create($validated);

        if ($request->variants) {
            $product->variants()->createMany($request->variants);
        }

        return new ProductResource($product->load('variants'));
    }

    public function variants(Product $product)
    {
        return response()->json([
            'data' => $product->variants,
        ]);
    }
}
```

### Version-Specific Resource - V2/ProductResource (enhanced)

<?php
namespace App\Http\Resources\V2;

use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;

class ProductResource extends JsonResource
{
    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'name' => $this->name,
            'slug' => $this->slug,
            'price' => [
                'regular' => (float) $this->price,
                'compare' => (float) $this->compare_price,
                'currency' => 'IDR',
                'formatted' => 'Rp ' . number_format($this->price, 0, ',', '.'),
            ],
            'description' => $this->description,
            'category' => new CategoryResource($this->whenLoaded('category')),
            'variants' => VariantResource::collection($this->whenLoaded('variants')),
            'tags' => $this->tags,
            'sku' => $this->sku,
            'in_stock' => $this->stock > 0,
            'rating' => $this->when($this->relationLoaded('reviews'), function () {
                return $this->reviews->avg('rating');
            }),
            'created_at' => $this->created_at->toISO8601String(),
            'updated_at' => $this->updated_at->toISO8601String(),
            '_links' => [
                'self' => route('api.v2.products.show', $this),
                'collection' => route('api.v2.products.index'),
            ],
        ];
    }
}

### Middleware - SetDefaultVersion
```php
<?php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class SetDefaultVersion
{
    public function handle(Request $request, Closure $next): Response
    {
        $version = $request->segment(2);

        if (!$version || !in_array($version, ['v1', 'v2'])) {
            $version = 'v2';
            $uri = $request->getRequestUri();
            $segments = explode('/', $uri);
            array_splice($segments, 2, 0, [$version]);
            $newUri = implode('/', $segments);

            return redirect($newUri);
        }

        return $next($request);
    }
}
```
