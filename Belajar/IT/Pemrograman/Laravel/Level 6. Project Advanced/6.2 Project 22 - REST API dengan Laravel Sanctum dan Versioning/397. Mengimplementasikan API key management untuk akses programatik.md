# 397. Mengimplementasikan API Key Management untuk Akses Programatik

## Definisi & Target
Membuat sistem API key management untuk memberikan akses programatik ke aplikasi, service, atau developer eksternal. API Key berbeda dengan token user biasa karena bisa memiliki permissions sendiri, expiry, dan bisa direvoke secara independen. Targetnya adalah integrasi third-party yang aman dan terkelola.

## Komponen yang Terlibat
- ApiKey model dan migration
- ApiKey middleware
- Key generation dengan prefix
- Key permissions (abilities)
- Key usage logging
- Key expiration
- Dashboard management

## Perintah Artisan CLI
```bash
# Membuat model dan migration
php artisan make:model ApiKey -m

# Membuat middleware
php artisan make:middleware AuthenticateApiKey

# Membuat controller
php artisan make:controller Api/ApiKeyController

# Membuat command untuk generate key
php artisan make:command GenerateApiKey
```

### Model - ApiKey
```php
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\BelongsTo;
use Illuminate\Database\Eloquent\SoftDeletes;
use Illuminate\Support\Str;

class ApiKey extends Model
{
    use SoftDeletes;

    protected $fillable = [
        'user_id', 'tenant_id', 'name', 'key', 'prefix',
        'abilities', 'allowed_ips', 'allowed_domains',
        'expires_at', 'last_used_at', 'is_active',
    ];

    protected $casts = [
        'abilities' => 'array',
        'allowed_ips' => 'array',
        'allowed_domains' => 'array',
        'expires_at' => 'datetime',
        'last_used_at' => 'datetime',
        'is_active' => 'boolean',
    ];

    protected $hidden = ['key'];

    public function user(): BelongsTo
    {
        return $this->belongsTo(User::class);
    }

    public function tenant(): BelongsTo
    {
        return $this->belongsTo(Tenant::class);
    }

    public function logs()
    {
        return $this->hasMany(ApiKeyLog::class);
    }

    public static function generateKey(string $name, string $prefix = 'app'): array
    {
        $random = Str::random(32);
        $key = strtoupper($prefix) . '_' . $random;

        return [
            'key' => hash('sha256', $key),
            'raw_key' => $key,
            'prefix' => $prefix,
        ];
    }

    public static function createWithKey(array $data): self
    {
        $prefix = $data['prefix'] ?? 'app';
        $keyData = self::generateKey($data['name'], $prefix);

        return self::create(array_merge($data, [
            'key' => $keyData['key'],
            'prefix' => $keyData['prefix'],
        ]));
    }

    public function hasAbility(string $ability): bool
    {
        if (empty($this->abilities)) {
            return true;
        }

        if (in_array('*', $this->abilities)) {
            return true;
        }

        return in_array($ability, $this->abilities);
    }

    public function isExpired(): bool
    {
        return $this->expires_at && $this->expires_at->isPast();
    }

    public function isAllowedIp(string $ip): bool
    {
        if (empty($this->allowed_ips)) {
            return true;
        }

        return in_array($ip, $this->allowed_ips);
    }

    public function logRequest(string $endpoint, string $method, int $statusCode, string $ip = null, string $userAgent = null): void
    {
        $this->logs()->create([
            'endpoint' => $endpoint,
            'method' => $method,
            'status_code' => $statusCode,
            'ip_address' => $ip,
            'user_agent' => $userAgent,
        ]);

        $this->update(['last_used_at' => now()]);
    }
}
```

### Middleware - AuthenticateApiKey
```php
<?php
namespace App\Http\Middleware;

use App\Models\ApiKey;
use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class AuthenticateApiKey
{
    public function handle(Request $request, Closure $next, string ...$abilities): Response
    {
        $apiKey = $request->header('X-API-Key')
            ?? $request->header('Authorization')
            ?? $request->query('api_key');

        if (!$apiKey) {
            return response()->json([
                'message' => 'API key diperlukan.',
            ], 401);
        }

        if (str_starts_with($apiKey, 'Bearer ')) {
            $apiKey = substr($apiKey, 7);
        }

        $hashedKey = hash('sha256', $apiKey);

        $key = ApiKey::where('key', $hashedKey)
            ->where('is_active', true)
            ->first();

        if (!$key) {
            return response()->json([
                'message' => 'API key tidak valid.',
            ], 401);
        }

        if ($key->isExpired()) {
            return response()->json([
                'message' => 'API key sudah kadaluarsa.',
            ], 401);
        }

        if (!$key->isAllowedIp($request->ip())) {
            return response()->json([
                'message' => 'IP address tidak terdaftar.',
            ], 403);
        }

        foreach ($abilities as $ability) {
            if (!$key->hasAbility($ability)) {
                return response()->json([
                    'message' => 'API key tidak memiliki akses ke resource ini.',
                    'required_ability' => $ability,
                ], 403);
            }
        }

        $request->merge(['api_key_record' => $key]);
        $request->setUserResolver(function () use ($key) {
            return $key->user;
        });

        return $next($request);
    }

    public function terminate(Request $request, Response $response): void
    {
        if ($key = $request->get('api_key_record')) {
            $key->logRequest(
                $request->path(),
                $request->method(),
                $response->getStatusCode(),
                $request->ip(),
                $request->userAgent()
            );
        }
    }
}
```

### Controller - ApiKeyController
```php
<?php
namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Models\ApiKey;
use Illuminate\Http\Request;

class ApiKeyManagementController extends Controller
{
    public function index(Request $request)
    {
        $keys = $request->user()->apiKeys()
            ->withCount('logs')
            ->latest()
            ->get()
            ->map(function ($key) {
                return [
                    'id' => $key->id,
                    'name' => $key->name,
                    'prefix' => $key->prefix,
                    'abilities' => $key->abilities,
                    'allowed_ips' => $key->allowed_ips,
                    'expires_at' => $key->expires_at,
                    'last_used_at' => $key->last_used_at,
                    'is_active' => $key->is_active,
                    'usage_count' => $key->logs_count,
                    'created_at' => $key->created_at,
                ];
            });

        return response()->json(['data' => $keys]);
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'prefix' => 'nullable|string|max:10|alpha',
            'abilities' => 'nullable|array',
            'abilities.*' => 'string',
            'allowed_ips' => 'nullable|array',
            'allowed_ips.*' => 'ip',
            'expires_at' => 'nullable|date|after:today',
        ]);

        $validated['user_id'] = $request->user()->id;
        $validated['prefix'] ??= 'app';

        $apiKey = ApiKey::createWithKey($validated);

        $rawKey = $apiKey['raw_key'];
        unset($apiKey['raw_key']);

        return response()->json([
            'data' => [
                'id' => $apiKey->id,
                'name' => $apiKey->name,
                'key' => $rawKey,
                'prefix' => $apiKey->prefix,
                'abilities' => $apiKey->abilities,
                'expires_at' => $apiKey->expires_at,
            ],
            'message' => 'API Key berhasil dibuat. Simpan key ini karena tidak akan ditampilkan lagi.',
        ], 201);
    }

    public function show(Request $request, ApiKey $apiKey)
    {
        if ($apiKey->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        return response()->json([
            'data' => [
                'id' => $apiKey->id,
                'name' => $apiKey->name,
                'prefix' => $apiKey->prefix,
                'abilities' => $apiKey->abilities,
                'allowed_ips' => $apiKey->allowed_ips,
                'expires_at' => $apiKey->expires_at,
                'is_active' => $apiKey->is_active,
                'last_used_at' => $apiKey->last_used_at,
                'created_at' => $apiKey->created_at,
            ],
        ]);
    }

    public function update(Request $request, ApiKey $apiKey)
    {
        if ($apiKey->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        $validated = $request->validate([
            'name' => 'sometimes|string|max:255',
            'abilities' => 'sometimes|nullable|array',
            'abilities.*' => 'string',
            'allowed_ips' => 'sometimes|nullable|array',
            'allowed_ips.*' => 'ip',
            'expires_at' => 'sometimes|nullable|date',
            'is_active' => 'sometimes|boolean',
        ]);

        $apiKey->update($validated);

        return response()->json([
            'message' => 'API Key berhasil diperbarui.',
            'data' => $apiKey->fresh(),
        ]);
    }

    public function destroy(Request $request, ApiKey $apiKey)
    {
        if ($apiKey->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        $apiKey->delete();

        return response()->json(['message' => 'API Key berhasil dihapus.']);
    }

    public function revoke(Request $request, ApiKey $apiKey)
    {
        if ($apiKey->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        $apiKey->update(['is_active' => false]);

        return response()->json(['message' => 'API Key dinonaktifkan.']);
    }

    public function logs(Request $request, ApiKey $apiKey)
    {
        if ($apiKey->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        $logs = $apiKey->logs()->latest()->paginate(50);

        return response()->json([
            'data' => $logs->items(),
            'meta' => [
                'total' => $logs->total(),
                'per_page' => $logs->perPage(),
            ],
        ]);
    }
}
```

### Artisan Command - GenerateApiKey
```php
<?php
namespace App\Console\Commands;

use App\Models\ApiKey;
use App\Models\User;
use Illuminate\Console\Command;

class GenerateApiKey extends Command
{
    protected $signature = 'apikey:generate
        {user : User ID or email}
        {name : Key name}
        {--abilities=* : Key abilities}
        {--expires= : Expiration date}';

    protected $description = 'Generate API key for a user';

    public function handle(): int
    {
        $userInput = $this->argument('user');
        $user = User::where('id', $userInput)
            ->orWhere('email', $userInput)
            ->first();

        if (!$user) {
            $this->error('User not found.');
            return Command::FAILURE;
        }

        $data = [
            'user_id' => $user->id,
            'name' => $this->argument('name'),
            'abilities' => $this->option('abilities') ?: ['*'],
            'expires_at' => $this->option('expires'),
        ];

        $apiKey = ApiKey::createWithKey($data);

        $this->info('API Key generated successfully:');
        $this->line("Key: {$apiKey['raw_key']}");
        $this->line("Prefix: {$apiKey->prefix}");

        return Command::SUCCESS;
    }
}
```

## Cara Integrasi & Pengujian
1. Generate API key via command: php artisan apikey:generate 1 "My App"
2. Test request dengan header X-API-Key
3. Test tanpa API key (401)
4. Test dengan API key salah (401)
5. Test dengan API key expired (401)
6. Test dengan abilities check: akses read-only endpoint dengan write key (403)
7. Test IP restriction
8. Lihat logs penggunaan di endpoint /api-keys/{id}/logs
9. Revoke key dan pastikan tidak bisa digunakan lagi
10. Test regenerasi key
