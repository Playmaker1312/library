# 395. Mengimplementasikan Filtering, Sorting, dan Field Selection pada Semua Endpoint

## Definisi & Target
Membuat sistem filtering, sorting, dan field selection yang seragam untuk semua endpoint API. Filtering menggunakan query parameters, sorting dengan kolom yang diizinkan, dan field selection untuk memilih field spesifik. Targetnya adalah API yang fleksibel dan efisien untuk berbagai kebutuhan klien.

## Komponen yang Terlibat
- Query parameter parsers
- FilterService
- Allowed sorts configuration
- Spatie Query Builder (optional)
- Field selection implementation
- Request validation untuk parameter

## Perintah Artisan CLI

### Option A: Menggunakan Spatie Query Builder

#### Controller - ProductController with Query Builder
```php
<?php
namespace App\Http\Controllers\Api\V2;

use App\Http\Controllers\Controller;
use App\Http\Resources\V2\ProductResource;
use App\Models\Product;
use Illuminate\Http\Request;
use Spatie\QueryBuilder\AllowedFilter;
use Spatie\QueryBuilder\AllowedSort;
use Spatie\QueryBuilder\QueryBuilder;

class ProductController extends Controller
{
    public function index(Request $request)
    {
        $products = QueryBuilder::for(Product::class)
            ->with(['category', 'variants'])
            ->allowedFilters([
                AllowedFilter::exact('category_id'),
                AllowedFilter::exact('status'),
                AllowedFilter::partial('name'),
                AllowedFilter::scope('price_between'),
                AllowedFilter::scope('in_stock'),
                AllowedFilter::trashed(),
            ])
            ->allowedSorts([
                AllowedSort::field('name'),
                AllowedSort::field('price'),
                AllowedSort::field('created_at'),
                AllowedSort::field('stock'),
                AllowedSort::field('sold_count', 'sold'),
                AllowedSort::custom('popular', new PopularSort()),
            ])
            ->allowedFields([
                'id', 'name', 'slug', 'description', 'price',
                'category_id', 'status', 'stock', 'created_at',
            ])
            ->allowedIncludes([
                'category', 'variants', 'images', 'reviews',
            ])
            ->cursorPaginate($request->input('per_page', 15));

        return ProductResource::collection($products);
    }
}
```

#### Model Scopes for Filtering
```php
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Database\Eloquent\Model;

class Product extends Model
{
    public function scopePriceBetween(Builder $query, $value, string $property): Builder
    {
        if (is_array($value)) {
            $query->whereBetween('price', [$value[0], $value[1] ?? PHP_INT_MAX]);
        } elseif (is_string($value)) {
            [$min, $max] = explode(',', $value);
            $query->whereBetween('price', [(float) $min, (float) $max]);
        }

        return $query;
    }

    public function scopeInStock(Builder $query): Builder
    {
        return $query->where('stock', '>', 0);
    }

    public function scopeFeatured(Builder $query): Builder
    {
        return $query->where('is_featured', true);
    }

    public function scopeSearch(Builder $query, string $term): Builder
    {
        return $query->where(function ($q) use ($term) {
            $q->where('name', 'like', "%{$term}%")
              ->orWhere('description', 'like', "%{$term}%")
              ->orWhere('sku', 'like', "%{$term}%");
        });
    }
}
```

### Option B: Manual Filter Service

#### Service - FilterService
```php
<?php
namespace App\Services;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Http\Request;

class FilterService
{
    private array $allowedFilters = [];
    private array $exactFilters = [];
    private array $partialFilters = [];
    private array $scopeFilters = [];
    private array $rangeFilters = [];

    public function __construct(array $config = [])
    {
        $this->allowedFilters = $config['allowed'] ?? [];
        $this->exactFilters = $config['exact'] ?? [];
        $this->partialFilters = $config['partial'] ?? [];
        $this->scopeFilters = $config['scope'] ?? [];
        $this->rangeFilters = $config['range'] ?? [];
    }

    public function apply(Builder $query, Request $request): Builder
    {
        foreach ($this->exactFilters as $field => $column) {
            if ($request->has($field)) {
                $value = $request->input($field);
                if (is_array($value)) {
                    $query->whereIn(is_string($column) ? $column : $field, $value);
                } else {
                    $query->where(is_string($column) ? $column : $field, $value);
                }
            }
        }

        foreach ($this->partialFilters as $field => $column) {
            if ($request->has($field)) {
                $query->where(
                    is_string($column) ? $column : $field,
                    'like',
                    "%{$request->input($field)}%"
                );
            }
        }

        foreach ($this->scopeFilters as $scope) {
            if ($request->has($scope)) {
                $value = $request->input($scope);
                $query->{"scope" . ucfirst($scope)}($value);
            }
        }

        foreach ($this->rangeFilters as $field) {
            $min = $request->input("{$field}_min");
            $max = $request->input("{$field}_max");

            if ($min !== null) {
                $query->where($field, '>=', $min);
            }
            if ($max !== null) {
                $query->where($field, '<=', $max);
            }
        }

        return $query;
    }
}
```

#### Service - SortService
```php
<?php
namespace App\Services;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Http\Request;

class SortService
{
    private array $allowedSorts;

    public function __construct(array $allowedSorts = ['id', 'created_at'])
    {
        $this->allowedSorts = $allowedSorts;
    }

    public function apply(Builder $query, Request $request): Builder
    {
        $sort = $request->input('sort', 'created_at');
        $order = $request->input('order', 'desc');

        if (!in_array($sort, $this->allowedSorts)) {
            $sort = 'created_at';
        }

        $order = strtolower($order) === 'asc' ? 'asc' : 'desc';

        $query->orderBy($sort, $order);

        if ($sort !== 'id') {
            $query->orderBy('id', $order);
        }

        return $query;
    }
}
```

### Trait - HasApiQueryParams
```php
<?php
namespace App\Traits;

use App\Services\FilterService;
use App\Services\SortService;
use Illuminate\Database\Eloquent\Builder;
use Illuminate\Http\Request;

trait HasApiQueryParams
{
    public function scopeApplyFilters(Builder $query, Request $request, array $config = []): Builder
    {
        $filterConfig = array_merge([
            'exact' => ['status', 'category_id', 'user_id'],
            'partial' => ['name' => 'name', 'search' => 'name'],
            'scope' => ['inStock' => 'in_stock', 'featured'],
            'range' => ['price', 'created_at'],
        ], $config['filters'] ?? []);

        $filterService = new FilterService($filterConfig);
        $query = $filterService->apply($query, $request);

        $sortConfig = array_merge([
            'id', 'name', 'price', 'created_at', 'updated_at',
        ], $config['sorts'] ?? []);

        $sortService = new SortService($sortConfig);
        $query = $sortService->apply($query, $request);

        return $query;
    }

    public function scopeSelectFields(Builder $query, Request $request, array $defaultFields = ['*']): Builder
    {
        if (!$request->has('fields')) {
            return $query->select($defaultFields);
        }

        $allowedFields = $defaultFields === ['*']
            ? $query->getModel()->getFillable()
            : $defaultFields;

        $fields = array_intersect(
            explode(',', $request->fields),
            $allowedFields
        );

        if (!empty($fields)) {
            $query->select(array_unique(array_merge(['id'], $fields)));
        }

        return $query;
    }
}
```

### Controller Using Manual Filters
```php
<?php
namespace App\Http\Controllers\Api\V2;

use App\Http\Controllers\Controller;
use App\Http\Resources\V2\ProductResource;
use App\Models\Product;
use App\Traits\HasApiQueryParams;
use Illuminate\Http\Request;

class ProductController extends Controller
{
    use HasApiQueryParams;

    public function index(Request $request)
    {
        $products = Product::with(['category'])
            ->applyFilters($request, [
                'filters' => [
                    'exact' => ['status', 'category_id'],
                    'partial' => ['name', 'search'],
                    'scope' => ['inStock' => 'in_stock'],
                    'range' => ['price'],
                ],
                'sorts' => ['id', 'name', 'price', 'created_at', 'sold_count'],
            ])
            ->selectFields($request, ['id', 'name', 'price', 'description', 'status', 'stock'])
            ->cursorPaginate($request->input('per_page', 15));

        return ProductResource::collection($products);
    }
}
```

### Test - Filtering Tests
```php
<?php
namespace Tests\Feature;

use App\Models\Category;
use App\Models\Product;
use Tests\TestCase;

class ProductFilteringTest extends TestCase
{
    public function test_can_filter_by_category()
    {
        $category = Category::factory()->create();
        Product::factory()->count(3)->create(['category_id' => $category->id]);
        Product::factory()->count(2)->create();

        $response = $this->getJson("/api/v2/products?filter[category_id]={$category->id}");

        $response->assertOk();
        $response->assertJsonCount(3, 'data');
    }

    public function test_can_filter_by_price_range()
    {
        Product::factory()->create(['price' => 10000]);
        Product::factory()->create(['price' => 50000]);
        Product::factory()->create(['price' => 100000]);

        $response = $this->getJson('/api/v2/products?filter[price_between]=20000,75000');

        $response->assertOk();
        $response->assertJsonCount(1, 'data');
    }

    public function test_can_sort_by_price_ascending()
    {
        Product::factory()->create(['price' => 50000]);
        Product::factory()->create(['price' => 10000]);
        Product::factory()->create(['price' => 25000]);

        $response = $this->getJson('/api/v2/products?sort=price&order=asc');

        $response->assertOk();
        $prices = collect($response->json('data'))->pluck('price');
        $this->assertEquals($prices->sort()->values()->toArray(), $prices->toArray());
    }

    public function test_can_select_specific_fields()
    {
        Product::factory()->create([
            'name' => 'Test Product',
            'price' => 50000,
            'description' => 'Description here',
        ]);

        $response = $this->getJson('/api/v2/products?fields=name,price');

        $response->assertOk();
        $product = $response->json('data')[0];
        $this->assertArrayHasKey('name', $product);
        $this->assertArrayHasKey('price', $product);
        $this->assertArrayNotHasKey('description', $product);
    }

    public function test_invalid_sort_falls_back_to_default()
    {
        Product::factory()->count(3)->create();

        $response = $this->getJson('/api/v2/products?sort=invalid_field');

        $response->assertOk();
    }
}
```
