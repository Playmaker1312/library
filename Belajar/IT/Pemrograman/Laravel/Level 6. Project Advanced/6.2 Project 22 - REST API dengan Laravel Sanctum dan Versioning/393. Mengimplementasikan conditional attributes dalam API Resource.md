# 393. Mengimplementasikan Conditional Attributes dalam API Resource

## Definisi & Target
Mengimplementasikan conditional attributes di API Resource sehingga data yang dikembalikan bisa berbeda berdasarkan kondisi seperti: user role, relasi yang di-load, query parameter, atau konteks request. Targetnya adalah response yang efisien (tidak mengirim data tidak perlu) dan fleksibel.

## Komponen yang Terlibat
- when() method
- whenLoaded() method
- whenNotNull() method
- mergeWhen() method
- Conditional wrapping
- Query parameter includes
- User-based visibility

## Perintah Artisan CLI
```bash
# Tidak perlu artisan command khusus
# Cukup edit resource yang sudah ada
php artisan make:resource V1/UserResource
php artisan make:resource V1/AdminUserResource
```

## Kode Implementasi (Per File)

### Resource - Conditional User Visibility
```php
<?php
namespace App\Http\Resources\V1;

use Illuminate\Http\Request;

class UserResource extends BaseResource
{
    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'name' => $this->name,
            'email' => $this->email,
            'email_verified_at' => $this->when($this->email_verified_at, function () {
                return $this->email_verified_at->toISO8601String();
            }),
            'avatar' => $this->avatar_url,
            'bio' => $this->when($this->bio, $this->bio),
            'role' => $this->role,
            'is_admin' => $this->is_admin,

            // Conditional berdasarkan relasi yang di-load
            'profile' => new ProfileResource($this->whenLoaded('profile')),
            'orders' => OrderResource::collection($this->whenLoaded('orders')),
            'reviews' => ReviewResource::collection($this->whenLoaded('reviews')),

            // Conditional berdasarkan role user yang login
            'phone' => $this->when(
                $request->user()?->is_admin || $request->user()?->id === $this->id,
                $this->phone
            ),
            'address' => $this->when(
                $request->user()?->id === $this->id,
                $this->address
            ),

            // Conditional berdasarkan query parameter 'include'
            'stats' => $this->when($request->has('include') && in_array('stats', explode(',', $request->include)), function () {
                return [
                    'posts_count' => $this->posts()->count(),
                    'comments_count' => $this->comments()->count(),
                    'joined_at' => $this->created_at->diffForHumans(),
                ];
            }),

            // Conditional menggunakan whenNotNull
            'last_login' => $this->whenNotNull($this->last_login_at?->diffForHumans()),

            // Merge data kondisional
            ...$this->mergeWhen($request->user()?->is_admin, [
                'internal_notes' => $this->internal_notes,
                'last_login_ip' => $this->last_login_ip,
                'status' => $this->status,
                'suspicious_activities' => $this->login_attempts > 5,
            ]),

            ...$this->includeTimestamps(),
        ];
    }
}
```

### Resource - Product with Conditional Pricing
```php
<?php
namespace App\Http\Resources\V1;

use Illuminate\Http\Request;

class ProductResource extends BaseResource
{
    public function toArray(Request $request): array
    {
        $isWholesale = $request->user()?->role === 'wholesale';

        return [
            'id' => $this->id,
            'name' => $this->name,
            'slug' => $this->slug,
            'description' => $this->description,

            // Conditional price based on user role
            'price' => $this->mergeWhen(!$isWholesale, [
                'retail' => (float) $this->price,
                'formatted' => 'Rp ' . number_format($this->price, 0, ',', '.'),
            ]),

            // Wholesale price only for wholesale users
            'wholesale_price' => $this->when($isWholesale, function () {
                return [
                    'min_quantity' => $this->min_wholesale_qty,
                    'price' => (float) $this->wholesale_price,
                    'formatted' => 'Rp ' . number_format($this->wholesale_price, 0, ',', '.'),
                    'savings' => $this->price - $this->wholesale_price,
                ];
            }),

            // Conditional stock info
            'stock' => $this->when(
                $this->stock <= 10 || $request->user()?->is_admin,
                $this->stock
            ),

            'in_stock' => $this->stock > 0,
            'low_stock' => $this->stock > 0 && $this->stock <= 5,

            // Conditional based on loaded relations
            'category' => new CategoryResource($this->whenLoaded('category')),
            'variants' => VariantResource::collection($this->whenLoaded('variants')),

            // Conditional images with different resolutions
            'images' => $this->whenLoaded('images', function () {
                return $this->images->map(function ($image) {
                    return [
                        'thumbnail' => $image->thumbnail_url,
                        'medium' => $image->medium_url,
                        'large' => $image->large_url,
                        'original' => $image->original_url,
                    ];
                });
            }),

            // Conditional discount display
            'discount' => $this->when(
                $this->compare_price && $this->compare_price > $this->price,
                function () {
                    $percent = round((1 - $this->price / $this->compare_price) * 100);
                    return [
                        'percent' => $percent,
                        'label' => "Hemat {$percent}%",
                        'original_price' => (float) $this->compare_price,
                    ];
                }
            ),

            // Conditional based on query parameter
            'reviews_summary' => $this->when($request->has('with_reviews'), function () {
                return [
                    'average' => round($this->reviews()->avg('rating'), 1),
                    'total' => $this->reviews()->count(),
                ];
            }),

            // Always include timestamps
            ...$this->includeTimestamps(),
            ...$this->includeLinks('products'),
        ];
    }
}
```

### Resource - Conditionally Loaded Relationships with Merge
```php
<?php
namespace App\Http\Resources\V1;

use Illuminate\Http\Request;

class OrderResource extends BaseResource
{
    public function toArray(Request $request): array
    {
        $canViewDetails = $request->user()?->is_admin
            || $request->user()?->id === $this->user_id;

        return [
            'id' => $this->id,
            'order_number' => $this->order_number,
            'status' => $this->status,
            'status_history' => StatusHistoryResource::collection(
                $this->whenLoaded('statusHistory')
            ),

            // Items only when loaded
            'items' => OrderItemResource::collection($this->whenLoaded('items')),

            'total' => [
                'value' => (float) $this->total,
                'formatted' => 'Rp ' . number_format($this->total, 0, ',', '.'),
            ],

            // Financial details only for owner or admin
            ...$this->mergeWhen($canViewDetails, [
                'subtotal' => (float) $this->subtotal,
                'tax' => (float) $this->tax,
                'shipping_cost' => (float) $this->shipping_cost,
                'discount' => (float) $this->discount,
                'payment_method' => $this->payment_method,
                'payment_details' => new PaymentResource($this->whenLoaded('payment')),
            ]),

            // Shipping info when loaded
            'shipping_address' => $this->whenLoaded('shipping', function () {
                return [
                    'name' => $this->shipping->recipient_name,
                    'phone' => $this->shipping->phone,
                    'address' => $this->shipping->full_address,
                    'courier' => $this->shipping->courier,
                    'tracking_number' => $this->shipping->tracking_number,
                    'estimated_delivery' => $this->shipping->estimated_delivery?->format('d M Y'),
                ];
            }),

            // Tracking URL when shipped
            'tracking_url' => $this->when($this->tracking_number, function () {
                return "https://cekresi.com/{$this->tracking_number}";
            }),

            // User info for admin
            'user' => $this->when(
                $request->user()?->is_admin && $this->relationLoaded('user'),
                new UserResource($this->user)
            ),

            // Conditional notes
            'notes' => $this->when($this->notes, $this->notes),
            'internal_notes' => $this->when($request->user()?->is_admin, $this->internal_notes),

            ...$this->includeTimestamps(),
        ];
    }
}
```

### Conditional Attribute Helper - Trait
```php
<?php
namespace App\Http\Resources\V1\Concerns;

use Illuminate\Http\Request;

trait HasConditionalAttributes
{
    protected function whenAdmin(Request $request, array $attributes): array
    {
        return $this->mergeWhen(
            $request->user()?->is_admin,
            $attributes
        );
    }

    protected function whenOwner(Request $request, $ownerId, array $attributes): array
    {
        return $this->mergeWhen(
            $request->user()?->id === $ownerId,
            $attributes
        );
    }

    protected function whenAdminOrOwner(Request $request, $ownerId, array $attributes): array
    {
        return $this->mergeWhen(
            $request->user()?->is_admin || $request->user()?->id === $ownerId,
            $attributes
        );
    }

    protected function whenInclude(Request $request, string $key, callable $callback): mixed
    {
        if (!$request->has('include')) {
            return $this->when(false, null);
        }

        $includes = explode(',', $request->include);

        return $this->when(
            in_array($key, $includes),
            $callback
        );
    }

    protected function whenField(Request $request, string $field, mixed $value): mixed
    {
        if (!$request->has('fields')) {
            return $value;
        }

        $fields = explode(',', $request->fields);

        return $this->when(
            in_array($field, $fields),
            $value
        );
    }
}
```

### Resource Using Trait
```php
<?php
namespace App\Http\Resources\V1;

use App\Http\Resources\V1\Concerns\HasConditionalAttributes;
use Illuminate\Http\Request;

class PostResource extends BaseResource
{
    use HasConditionalAttributes;

    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'title' => $this->title,
            'content' => $this->when($this->status === 'published', $this->content),
            'excerpt' => substr($this->content, 0, 150),
            'author' => new UserResource($this->whenLoaded('author')),
            'comments' => CommentResource::collection($this->whenLoaded('comments')),
            'tags' => $this->tags,
            'status' => $this->status,
            'views_count' => $this->views_count,

            // Menggunakan trait methods
            ...$this->whenAdmin($request, [
                'draft_content' => $this->draft_content,
                'editor_notes' => $this->editor_notes,
                'seo_meta' => $this->seo_meta,
            ]),

            ...$this->whenOwner($request, $this->author_id, [
                'edit_history' => $this->edit_history,
            ]),

            ...$this->whenInclude($request, 'analytics', function () {
                return [
                    'views' => $this->views_count,
                    'unique_visitors' => $this->unique_visitors,
                    'avg_time' => $this->avg_reading_time,
                ];
            }),

            ...$this->includeTimestamps(),
        ];
    }
}
```

### Controller - Using Conditional Includes
```php
<?php
namespace App\Http\Controllers\Api\V1;

use App\Http\Resources\V1\ProductResource;
use App\Models\Product;
use Illuminate\Http\Request;

class ProductController extends Controller
{
    public function index(Request $request)
    {
        $query = Product::query();

        if ($request->has('include')) {
            $includes = explode(',', $request->include);

            if (in_array('category', $includes)) {
                $query->with('category');
            }

            if (in_array('variants', $includes)) {
                $query->with('variants');
            }

            if (in_array('reviews', $includes)) {
                $query->with('reviews');
            }
        }

        if ($request->has('fields')) {
            $fields = explode(',', $request->fields);
            $query->select(array_merge(['id'], $fields));
        }

        $products = $query->paginate(15);

        return ProductResource::collection($products);
    }

    public function show(Request $request, Product $product)
    {
        $includes = $request->has('include')
            ? explode(',', $request->include)
            : ['category', 'variants'];

        $product->load($includes);

        return new ProductResource($product);
    }
}
```
