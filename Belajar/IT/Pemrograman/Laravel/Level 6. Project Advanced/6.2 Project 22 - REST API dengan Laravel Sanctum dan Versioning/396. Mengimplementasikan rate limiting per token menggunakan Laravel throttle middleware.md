# 396. Mengimplementasikan Rate Limiting per Token Menggunakan Laravel Throttle Middleware

## Definisi & Target
Mengimplementasikan rate limiting untuk membatasi jumlah request API per token dalam periode waktu tertentu. Menggunakan Laravel built-in throttle middleware dengan konfigurasi berbeda per endpoint atau per user role. Targetnya adalah proteksi API dari abuse dan fair usage.

## Komponen yang Terlibat
- Laravel throttle middleware
- RateLimiter facade
- Custom rate limit definitions
- Per-token rate limiting
- Per-role rate limits
- X-RateLimit headers
- Cache driver (Redis recommended)

## Perintah Artisan CLI
```bash
# Rate limiter sudah built-in, tidak perlu install
# Membuat custom rate limiter
php artisan make:class Services/RateLimitService

# Membuat middleware custom (optional)
php artisan make:middleware ThrottleWithMeta
```

## Kode Implementasi (Per File)

### Rate Limiter Registration - AppServiceProvider
```php
<?php
namespace App\Providers;

use Illuminate\Cache\RateLimiting\Limit;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\RateLimiter;
use Illuminate\Support\ServiceProvider;

class AppServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        RateLimiter::for('api', function (Request $request) {
            $userId = $request->user()?->id ?: $request->ip();
            $tokenId = $request->user()?->currentAccessToken()?->id;

            $key = $tokenId ? "token_{$tokenId}" : "user_{$userId}";

            return Limit::perMinute(60)->by($key);
        });

        RateLimiter::for('auth', function (Request $request) {
            return Limit::perMinute(5)->by($request->ip());
        });

        RateLimiter::for('register', function (Request $request) {
            return Limit::perDay(3)->by($request->ip());
        });

        RateLimiter::for('uploads', function (Request $request) {
            $userId = $request->user()?->id;

            return Limit::perHour(20)->by("user_uploads_{$userId}");
        });

        RateLimiter::for('webhooks', function (Request $request) {
            $tenantId = tenant()?->id ?? $request->user()?->id;

            return Limit::perMinute(120)->by("webhooks_{$tenantId}");
        });

        RateLimiter::for('admin', function (Request $request) {
            return Limit::perMinute(200)->by("admin_{$request->user()?->id}");
        });

        RateLimiter::for('premium', function (Request $request) {
            $user = $request->user();

            if ($user && $user->isPremium()) {
                return Limit::perMinute(300)->by("premium_{$user->id}");
            }

            return Limit::perMinute(60)->by("free_{$user?->id}");
        });
    }
}
```

### Route Configuration - api.php with Throttle
```php
<?php
use Illuminate\Support\Facades\Route;

// Public routes with strict rate limiting
Route::post('/auth/login', [AuthController::class, 'login'])
    ->middleware('throttle:auth');

Route::post('/auth/register', [AuthController::class, 'register'])
    ->middleware('throttle:register');

// Protected routes with standard rate limiting
Route::middleware(['auth:sanctum', 'throttle:api'])->group(function () {
    Route::get('/user', [UserController::class, 'show']);
    Route::put('/user', [UserController::class, 'update']);
});

// Premium routes with higher limits
Route::middleware(['auth:sanctum', 'throttle:premium'])->group(function () {
    Route::get('/products', [ProductController::class, 'index']);
    Route::get('/products/{product}', [ProductController::class, 'show']);
});

// Upload routes with hourly limit
Route::middleware(['auth:sanctum', 'throttle:uploads'])->post('/uploads', [UploadController::class, 'store']);

// Admin routes
Route::middleware(['auth:sanctum', 'throttle:admin'])->prefix('admin')->group(function () {
    Route::get('/users', [AdminController::class, 'users']);
    Route::get('/stats', [AdminController::class, 'stats']);
});
```

### Custom Middleware - ThrottleWithMeta
```php
<?php
namespace App\Http\Middleware;

use Closure;
use Illuminate\Cache\RateLimiter;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class ThrottleWithMeta
{
    public function __construct(
        private RateLimiter $limiter
    ) {}

    public function handle(Request $request, Closure $next, string $limiterName, int $maxAttempts = 60, int $decayMinutes = 1): Response
    {
        $key = $this->resolveKey($request, $limiterName);

        if ($this->limiter->tooManyAttempts($key, $maxAttempts)) {
            return $this->buildResponse($key, $maxAttempts);
        }

        $this->limiter->hit($key, $decayMinutes * 60);

        $response = $next($request);

        return $this->addHeaders(
            $response,
            $maxAttempts,
            $this->limiter->remaining($key, $maxAttempts)
        );
    }

    private function resolveKey(Request $request, string $limiterName): string
    {
        $userId = $request->user()?->id;
        $tokenId = $request->user()?->currentAccessToken()?->id;
        $ip = $request->ip();

        return match ($limiterName) {
            'token' => "token_{$tokenId}_{$ip}",
            'user' => "user_{$userId}",
            'ip' => "ip_{$ip}",
            default => "{$limiterName}_{$userId}_{$ip}",
        };
    }

    private function buildResponse(string $key, int $maxAttempts): Response
    {
        $retryAfter = $this->limiter->availableIn($key);

        return response()->json([
            'message' => 'Terlalu banyak permintaan. Silakan coba lagi.',
            'retry_after_seconds' => $retryAfter,
            'retry_after' => now()->addSeconds($retryAfter)->diffForHumans(),
        ], 429)->withHeaders([
            'Retry-After' => $retryAfter,
            'X-RateLimit-Limit' => $maxAttempts,
            'X-RateLimit-Remaining' => 0,
        ]);
    }

    private function addHeaders(Response $response, int $maxAttempts, int $remaining): Response
    {
        if ($response instanceof \Illuminate\Http\JsonResponse) {
            $response->header('X-RateLimit-Limit', $maxAttempts);
            $response->header('X-RateLimit-Remaining', $remaining);
        }

        return $response;
    }
}
```

### Service - Dynamic Rate Limit Service
```php
<?php
namespace App\Services;

use App\Models\Plan;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\RateLimiter;

class RateLimitService
{
    public function getLimitForRequest(Request $request): array
    {
        $user = $request->user();

        if (!$user) {
            return ['max' => 30, 'decay' => 1, 'name' => 'guest'];
        }

        if ($user->is_admin) {
            return ['max' => 500, 'decay' => 1, 'name' => 'admin'];
        }

        $tenant = tenant();
        if ($tenant) {
            $plan = Plan::where('slug', $tenant->plan)->first();

            if ($plan) {
                $apiLimit = $plan->getFeatureValue('api_rate_limit');

                if ($apiLimit) {
                    return [
                        'max' => (int) $apiLimit,
                        'decay' => 1,
                        'name' => "plan_{$plan->slug}",
                    ];
                }
            }
        }

        return ['max' => 60, 'decay' => 1, 'name' => 'default'];
    }

    public function trackUsage(Request $request): void
    {
        $limit = $this->getLimitForRequest($request);
        $key = "dynamic_{$request->user()?->id}_{$request->ip()}";

        RateLimiter::hit($key, $limit['decay'] * 60);
    }

    public function remaining(Request $request): int
    {
        $limit = $this->getLimitForRequest($request);
        $key = "dynamic_{$request->user()?->id}_{$request->ip()}";

        return RateLimiter::remaining($key, $limit['max']);
    }
}
```

### Middleware - DynamicThrottle
```php
<?php
namespace App\Http\Middleware;

use App\Services\RateLimitService;
use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class DynamicThrottle
{
    public function __construct(
        private RateLimitService $rateLimitService
    ) {}

    public function handle(Request $request, Closure $next): Response
    {
        $limit = $this->rateLimitService->getLimitForRequest($request);
        $key = "dynamic_{$request->user()?->id}_{$request->ip()}";

        $limiter = app(\Illuminate\Cache\RateLimiter::class);

        if ($limiter->tooManyAttempts($key, $limit['max'])) {
            $retryAfter = $limiter->availableIn($key);

            return response()->json([
                'message' => 'Rate limit exceeded.',
                'limit' => $limit['max'],
                'retry_after' => $retryAfter,
            ], 429)->withHeaders([
                'X-RateLimit-Limit' => $limit['max'],
                'X-RateLimit-Remaining' => 0,
                'Retry-After' => $retryAfter,
            ]);
        }

        $limiter->hit($key, $limit['decay'] * 60);

        $response = $next($request);

        $response->header('X-RateLimit-Limit', $limit['max']);
        $response->header('X-RateLimit-Remaining', $limiter->remaining($key, $limit['max']));

        return $response;
    }
}
```

### Queue-based Rate Limit Reset
```php
<?php
namespace App\Console\Commands;

use Illuminate\Console\Command;
use Illuminate\Support\Facades\Cache;

class ResetRateLimits extends Command
{
    protected $signature = 'ratelimit:reset {userId?}';
    protected $description = 'Reset rate limits for a user or all users';

    public function handle(): int
    {
        $userId = $this->argument('userId');

        if ($userId) {
            Cache::tags(['rate-limits'])->forget("user_{$userId}");
            $this->info("Rate limit untuk user {$userId} telah di-reset.");
        } else {
            Cache::tags(['rate-limits'])->flush();
            $this->info('Semua rate limit telah di-reset.');
        }

        return Command::SUCCESS;
    }
}
```

### Test - Rate Limiting Tests
```php
<?php
namespace Tests\Feature;

use App\Models\User;
use Tests\TestCase;

class RateLimitingTest extends TestCase
{
    public function test_rate_limit_blocks_excessive_requests()
    {
        $user = User::factory()->create();
        $token = $user->createToken('test')->plainTextToken;

        $headers = ['Authorization' => "Bearer {$token}"];

        for ($i = 0; $i < 60; $i++) {
            $this->getJson('/api/v1/user', $headers);
        }

        $response = $this->getJson('/api/v1/user', $headers);

        $response->assertStatus(429);
        $response->assertJsonStructure(['message', 'retry_after_seconds']);
    }

    public function test_rate_limit_headers_are_present()
    {
        $user = User::factory()->create();
        $token = $user->createToken('test')->plainTextToken;

        $response = $this->withHeaders([
            'Authorization' => "Bearer {$token}",
        ])->getJson('/api/v1/user');

        $response->assertHeader('X-RateLimit-Limit');
        $response->assertHeader('X-RateLimit-Remaining');
    }

    public function test_auth_rate_limit_stricter()
    {
        for ($i = 0; $i < 6; $i++) {
            $this->postJson('/api/v1/auth/login', [
                'email' => 'test@test.com',
                'password' => 'wrong_password',
            ]);
        }

        $response = $this->postJson('/api/v1/auth/login', [
            'email' => 'test@test.com',
            'password' => 'wrong_password',
        ]);

        $response->assertStatus(429);
    }

    public function test_different_tokens_have_separate_limits()
    {
        $user = User::factory()->create();
        $token1 = $user->createToken('app1')->plainTextToken;
        $token2 = $user->createToken('app2')->plainTextToken;

        for ($i = 0; $i < 60; $i++) {
            $this->withHeaders(['Authorization' => "Bearer {$token1}"])
                ->getJson('/api/v1/user');
        }

        $response = $this->withHeaders(['Authorization' => "Bearer {$token2}"])
            ->getJson('/api/v1/user');

        $response->assertOk();
    }
}
```
