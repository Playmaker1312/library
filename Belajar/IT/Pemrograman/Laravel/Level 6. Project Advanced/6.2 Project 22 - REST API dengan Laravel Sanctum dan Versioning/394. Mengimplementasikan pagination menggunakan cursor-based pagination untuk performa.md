# 394. Mengimplementasikan Pagination Menggunakan Cursor-Based Pagination untuk Performa

## Definisi & Target
Mengimplementasikan cursor-based pagination sebagai alternatif offset-based pagination yang lebih performan untuk dataset besar. Cursor pagination menggunakan unique identifier (biasanya primary key atau timestamp) untuk menentukan posisi halaman berikutnya. Targetnya adalah query yang konsisten dan cepat meskipun ada data baru.

## Komponen yang Terlibat
- Laravel CursorPaginator
- Cursor parameter implementation
- Next/previous cursor links
- Sort column configuration
- Resource collection with cursor
- Infinite scroll integration

### Controller - Cursor Pagination Example
```php
<?php
namespace App\Http\Controllers\Api\V2;

use App\Http\Controllers\Controller;
use App\Http\Resources\V2\ProductResource;
use App\Models\Product;
use Illuminate\Http\Request;

class ProductController extends Controller
{
    public function index(Request $request)
    {
        $query = Product::with(['category', 'variants']);

        $validated = $request->validate([
            'cursor' => 'nullable|string',
            'per_page' => 'nullable|integer|min:1|max:100',
            'sort' => 'nullable|string|in:id,name,price,created_at',
            'order' => 'nullable|string|in:asc,desc',
        ]);

        $perPage = $validated['per_page'] ?? 15;
        $sortColumn = $validated['sort'] ?? 'id';
        $sortOrder = $validated['order'] ?? 'desc';

        $query->orderBy($sortColumn, $sortOrder);

        if ($sortColumn !== 'id') {
            $query->orderBy('id', 'desc');
        }

        $products = $query->cursorPaginate($perPage);

        return response()->json([
            'data' => ProductResource::collection($products),
            'meta' => [
                'per_page' => $products->perPage(),
                'sort' => $sortColumn,
                'order' => $sortOrder,
            ],
            'links' => [
                'next' => $products->nextCursor()?->encode(),
                'prev' => $products->previousCursor()?->encode(),
                'has_next' => $products->hasMorePages(),
                'has_prev' => $products->onFirstPage() === false,
                'path' => $request->url(),
            ],
        ]);
    }

    public function indexWithTimestamp(Request $request)
    {
        $query = Product::with(['category']);

        $perPage = $request->input('per_page', 15);

        $products = $query->orderBy('created_at', 'desc')
            ->orderBy('id', 'desc')
            ->cursorPaginate($perPage);

        $response = [
            'data' => ProductResource::collection($products),
            'links' => [
                'next' => null,
                'prev' => null,
            ],
        ];

        if ($products->hasMorePages()) {
            $nextCursor = $products->nextCursor();
            $response['links']['next'] = route('api.v2.products.index.cursor', [
                'cursor' => $nextCursor?->encode(),
                'per_page' => $perPage,
            ]);

            $response['links']['next_cursor_params'] = [
                'cursor' => $nextCursor?->encode(),
                'per_page' => $perPage,
            ];
        }

        $firstItem = $products->first();
        if (!$products->onFirstPage() && $firstItem) {
            $prevCursor = $products->previousCursor();
            $response['links']['prev'] = route('api.v2.products.index.cursor', [
                'cursor' => $prevCursor?->encode(),
                'per_page' => $perPage,
            ]);
        }

        return response()->json($response);
    }
}
```

### Custom Cursor Pagination Service
```php
<?php
namespace App\Services;

use Illuminate\Database\Eloquent\Builder;
use Illuminate\Pagination\CursorPaginator;
use Illuminate\Http\Request;

class CursorPaginationService
{
    public function paginate(Builder $query, Request $request, array $config = []): array
    {
        $defaults = [
            'per_page' => 15,
            'cursor_param' => 'cursor',
            'sort_column' => 'id',
            'sort_direction' => 'desc',
            'append_params' => [],
            'max_per_page' => 100,
        ];

        $config = array_merge($defaults, $config);

        $perPage = min(
            (int) $request->input('per_page', $config['per_page']),
            $config['max_per_page']
        );

        $sortColumn = $request->input('sort', $config['sort_column']);
        $sortDirection = $request->input('order', $config['sort_direction']);

        $allowedSorts = $config['allowed_sorts'] ?? [$config['sort_column']];

        if (!in_array($sortColumn, $allowedSorts)) {
            $sortColumn = $config['sort_column'];
        }

        $query->orderBy($sortColumn, $sortDirection);

        if ($sortColumn !== 'id') {
            $query->orderBy('id', 'desc');
        }

        $results = $query->cursorPaginate($perPage);

        $nextCursor = $results->nextCursor();
        $prevCursor = $results->previousCursor();

        return [
            'data' => $results->items(),
            'meta' => [
                'per_page' => $perPage,
                'sort' => $sortColumn,
                'order' => $sortDirection,
                'total_cached' => false,
            ],
            'links' => [
                'next' => $nextCursor?->encode(),
                'prev' => $prevCursor?->encode(),
                'has_next' => $results->hasMorePages(),
                'has_prev' => !$results->onFirstPage(),
            ],
            'paginator' => $results,
        ];
    }

    public function encodeCursor(mixed $cursor): ?string
    {
        return $cursor?->encode();
    }

    public function decodeCursor(string $encoded): mixed
    {
        return CursorPaginator::resolveCursor($encoded);
    }
}
```

### Controller - Using CursorPaginationService
```php
<?php
namespace App\Http\Controllers\Api\V2;

use App\Http\Controllers\Controller;
use App\Http\Resources\V2\PostResource;
use App\Models\Post;
use App\Services\CursorPaginationService;
use Illuminate\Http\Request;

class PostController extends Controller
{
    public function __construct(
        private CursorPaginationService $cursorService
    ) {}

    public function index(Request $request)
    {
        $query = Post::with(['author', 'tags'])
            ->where('status', 'published');

        $result = $this->cursorService->paginate($query, $request, [
            'per_page' => 20,
            'allowed_sorts' => ['id', 'created_at', 'title', 'views_count'],
            'sort_column' => 'created_at',
            'sort_direction' => 'desc',
        ]);

        return response()->json([
            'data' => PostResource::collection($result['data']),
            'meta' => $result['meta'],
            'links' => $result['links'],
        ]);
    }
}
```

### Frontend Integration - Infinite Scroll (Vue 3)
```javascript
// composables/useCursorPagination.js
import { ref, computed } from 'vue';
import api from '../api/axios';

export function useCursorPagination(endpoint, options = {}) {
    const items = ref([]);
    const loading = ref(false);
    const nextCursor = ref(null);
    const hasMore = ref(true);
    const error = ref(null);

    const fetchItems = async (cursor = null) => {
        if (loading.value) return;

        loading.value = true;
        error.value = null;

        try {
            const params = { per_page: options.perPage || 15 };
            if (cursor) params.cursor = cursor;

            const response = await api.get(endpoint, { params });

            const newItems = response.data.data;
            items.value = cursor ? [...items.value, ...newItems] : newItems;

            nextCursor.value = response.data.links.next;
            hasMore.value = response.data.links.has_next;

            return response.data;
        } catch (err) {
            error.value = err.response?.data?.message || 'Gagal memuat data';
        } finally {
            loading.value = false;
        }
    };

    const loadMore = () => {
        if (hasMore.value && !loading.value) {
            return fetchItems(nextCursor.value);
        }
    };

    const refresh = () => {
        items.value = [];
        nextCursor.value = null;
        hasMore.value = true;
        return fetchItems(null);
    };

    return {
        items,
        loading,
        hasMore,
        error,
        fetchItems,
        loadMore,
        refresh,
    };
}
```

### Vue Component - Infinite Scroll List
```vue
<template>
    <div>
        <div v-for="item in items" :key="item.id" class="card">
            <h3>{{ item.name }}</h3>
            <p>{{ item.description }}</p>
        </div>

        <div v-if="loading" class="text-center py-4">
            Loading...
        </div>

        <div v-if="error" class="text-red-500">{{ error }}</div>

        <button v-if="hasMore && !loading" @click="loadMore" class="load-more">
            Load More
        </button>

        <div v-if="!hasMore && items.length" class="text-center py-4 text-gray-500">
            Semua data telah dimuat.
        </div>
    </div>
</template>

<script setup>
import { onMounted } from 'vue';
import { useCursorPagination } from '../composables/useCursorPagination';

const { items, loading, hasMore, error, fetchItems, loadMore } = useCursorPagination(
    '/api/v2/products/cursor',
    { perPage: 20 }
);

onMounted(() => fetchItems());
</script>
```

### Performance Comparison Test
```php
<?php
namespace Tests\Feature;

use Tests\TestCase;

class PaginationPerformanceTest extends TestCase
{
    public function test_cursor_pagination_is_faster_than_offset()
    {
        $startOffset = microtime(true);
        $this->getJson('/api/v1/products?page=100');
        $offsetTime = microtime(true) - $startOffset;

        $startCursor = microtime(true);
        $response = $this->getJson('/api/v2/products?per_page=15');
        $cursorTime = microtime(true) - $startCursor;

        $this->assertLessThan(
            $offsetTime,
            $cursorTime,
            'Cursor pagination should be faster than offset at deep pages'
        );
    }

    public function test_cursor_consistency_with_new_data()
    {
        $response1 = $this->getJson('/api/v2/products?per_page=10');
        $nextCursor = $response1->json('links.next');

        Product::factory()->count(5)->create();

        $response2 = $this->getJson("/api/v2/products?per_page=10&cursor={$nextCursor}");

        // Cursor pagination should not be affected by new data
        $response2->assertOk();
        $this->assertNotEmpty($response2->json('data'));
    }
}
```
