# 392. Membuat API Resource yang Komprehensif untuk Setiap Model

## Definisi & Target
Membuat API Resource (Laravel Eloquent API Resource) yang lengkap dan konsisten untuk setiap model. Resource mengontrol format data yang dikembalikan ke klien, termasuk relasi, metadata, dan links. Targetnya adalah response JSON yang predictable, terstruktur, dan self-descriptive.

## Komponen yang Terlibat
- JsonResource class
- ResourceCollection class
- Resource relationships (whenLoaded)
- Conditional attributes
- Resource nesting
- Metadata dan links
- Pagination wrapper

## Perintah Artisan CLI
```bash
# Membuat resource per model
php artisan make:resource V1/UserResource
php artisan make:resource V1/UserCollection
php artisan make:resource V1/ProductResource
php artisan make:resource V1/ProductCollection
php artisan make:resource V1/OrderResource
php artisan make:resource V1/OrderCollection
php artisan make:resource V1/CategoryResource
php artisan make:resource V1/CategoryCollection
```

## Kode Implementasi (Per File)

### Base Resource - BaseResource
```php
<?php
namespace App\Http\Resources\V1;

use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\JsonResource;

abstract class BaseResource extends JsonResource
{
    protected function includeTimestamps(): array
    {
        return [
            'created_at' => $this->created_at?->toISO8601String(),
            'updated_at' => $this->updated_at?->toISO8601String(),
        ];
    }

    protected function includeLinks(string $resourceName): array
    {
        return [
            'self' => route("api.v1.{$resourceName}.show", $this),
            'collection' => route("api.v1.{$resourceName}.index"),
        ];
    }

    protected function includeAuthor(): ?array
    {
        if (!$this->relationLoaded('user')) {
            return null;
        }

        return [
            'id' => $this->user?->id,
            'name' => $this->user?->name,
        ];
    }

    public function with(Request $request): array
    {
        return [
            'meta' => [
                'api_version' => 'v1',
                'timestamp' => now()->toISO8601String(),
            ],
        ];
    }
}
```

### Resource - UserResource
```php
<?php
namespace App\Http\Resources\V1;

use Illuminate\Http\Request;

class UserResource extends BaseResource
{
    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'name' => $this->name,
            'email' => $this->email,
            'email_verified_at' => $this->email_verified_at?->toISO8601String(),
            'avatar' => $this->avatar_url,
            'role' => $this->role,
            'permissions' => $this->when($request->user()?->is_admin, function () {
                return $this->getAllPermissions()->pluck('name');
            }),
            'profile' => new ProfileResource($this->whenLoaded('profile')),
            'stats' => $this->when($this->relationLoaded('orders'), function () {
                return [
                    'total_orders' => $this->orders->count(),
                    'total_spent' => $this->orders->sum('total'),
                ];
            }),
            ...$this->includeTimestamps(),
            ...$this->includeLinks('users'),
        ];
    }
}
```

### Resource - ProductResource (Comprehensive)
```php
<?php
namespace App\Http\Resources\V1;

use Illuminate\Http\Request;

class ProductResource extends BaseResource
{
    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'name' => $this->name,
            'slug' => $this->slug,
            'description' => $this->description,
            'short_description' => substr($this->description, 0, 200),
            'price' => [
                'value' => (float) $this->price,
                'currency' => 'IDR',
                'formatted' => 'Rp ' . number_format($this->price, 0, ',', '.'),
            ],
            'compare_price' => $this->when($this->compare_price, function () {
                return [
                    'value' => (float) $this->compare_price,
                    'formatted' => 'Rp ' . number_format($this->compare_price, 0, ',', '.'),
                    'discount_percent' => $this->compare_price
                        ? round((1 - $this->price / $this->compare_price) * 100)
                        : 0,
                ];
            }),
            'images' => ProductImageResource::collection($this->whenLoaded('images')),
            'thumbnail' => $this->thumbnail_url,
            'category' => new CategoryResource($this->whenLoaded('category')),
            'variants' => VariantResource::collection($this->whenLoaded('variants')),
            'tags' => $this->tags ?? [],
            'sku' => $this->sku,
            'stock' => $this->stock,
            'in_stock' => $this->stock > 0,
            'weight' => $this->weight,
            'dimensions' => $this->when($this->length, function () {
                return [
                    'length' => $this->length,
                    'width' => $this->width,
                    'height' => $this->height,
                    'unit' => 'cm',
                ];
            }),
            'rating' => $this->when($this->relationLoaded('reviews'), function () {
                return [
                    'average' => round($this->reviews->avg('rating'), 1),
                    'count' => $this->reviews->count(),
                    'distribution' => $this->reviews->groupBy('rating')
                        ->map->count()
                        ->toArray(),
                ];
            }),
            'status' => $this->status,
            'is_featured' => $this->is_featured,
            ...$this->includeTimestamps(),
            ...$this->includeLinks('products'),
        ];
    }
}
```

### Resource - OrderResource (with nested relations)
```php
<?php
namespace App\Http\Resources\V1;

use Illuminate\Http\Request;

class OrderResource extends BaseResource
{
    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'order_number' => $this->order_number,
            'status' => $this->status,
            'status_label' => $this->status_label,
            'items' => OrderItemResource::collection($this->whenLoaded('items')),
            'items_count' => $this->when($this->relationLoaded('items'), function () {
                return $this->items->sum('quantity');
            }),
            'subtotal' => [
                'value' => (float) $this->subtotal,
                'formatted' => 'Rp ' . number_format($this->subtotal, 0, ',', '.'),
            ],
            'tax' => [
                'value' => (float) $this->tax,
                'formatted' => 'Rp ' . number_format($this->tax, 0, ',', '.'),
            ],
            'shipping_cost' => [
                'value' => (float) $this->shipping_cost,
                'formatted' => 'Rp ' . number_format($this->shipping_cost, 0, ',', '.'),
            ],
            'discount' => [
                'value' => (float) $this->discount,
                'formatted' => 'Rp ' . number_format($this->discount, 0, ',', '.'),
            ],
            'total' => [
                'value' => (float) $this->total,
                'formatted' => 'Rp ' . number_format($this->total, 0, ',', '.'),
            ],
            'payment' => new PaymentResource($this->whenLoaded('payment')),
            'shipping' => new ShippingResource($this->whenLoaded('shipping')),
            'user' => new UserResource($this->whenLoaded('user')),
            'notes' => $this->notes,
            'shipped_at' => $this->shipped_at?->toISO8601String(),
            'delivered_at' => $this->delivered_at?->toISO8601String(),
            ...$this->includeTimestamps(),
            ...$this->includeLinks('orders'),
        ];
    }
}
```

### Resource Collection - ProductCollection
```php
<?php
namespace App\Http\Resources\V1;

use Illuminate\Http\Request;
use Illuminate\Http\Resources\Json\ResourceCollection;

class ProductCollection extends ResourceCollection
{
    public $collects = ProductResource::class;

    public function toArray(Request $request): array
    {
        return [
            'data' => $this->collection,
            'meta' => [
                'current_page' => $this->currentPage(),
                'per_page' => $this->perPage(),
                'total' => $this->total(),
                'last_page' => $this->lastPage(),
                'from' => $this->firstItem(),
                'to' => $this->lastItem(),
            ],
            'links' => [
                'self' => $this->url($this->currentPage()),
                'first' => $this->url(1),
                'last' => $this->url($this->lastPage()),
                'prev' => $this->previousPageUrl(),
                'next' => $this->nextPageUrl(),
            ],
        ];
    }

    public function with(Request $request): array
    {
        return [
            'meta' => [
                'api_version' => 'v1',
                'timestamp' => now()->toISO8601String(),
            ],
        ];
    }
}
```

### Resource - Nested Resources Example
```php
<?php
namespace App\Http\Resources\V1;

use Illuminate\Http\Request;

class CategoryResource extends BaseResource
{
    public function toArray(Request $request): array
    {
        return [
            'id' => $this->id,
            'name' => $this->name,
            'slug' => $this->slug,
            'description' => $this->description,
            'parent' => new CategoryResource($this->whenLoaded('parent')),
            'children' => CategoryResource::collection($this->whenLoaded('children')),
            'products_count' => $this->when($this->products_count !== null, $this->products_count),
            'products' => ProductResource::collection($this->whenLoaded('products')),
            'icon' => $this->icon,
            'sort_order' => $this->sort_order,
            'is_active' => $this->is_active,
            ...$this->includeTimestamps(),
            ...$this->includeLinks('categories'),
        ];
    }
}
```

### Controller - Using Resources
```php
<?php
namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Http\Resources\V1\ProductResource;
use App\Http\Resources\V1\ProductCollection;
use App\Models\Product;
use Illuminate\Http\Request;

class ProductController extends Controller
{
    public function index(Request $request)
    {
        $products = Product::with(['category', 'images'])
            ->paginate($request->per_page ?? 15);

        return new ProductCollection($products);
    }

    public function show(Product $product)
    {
        $product->load(['category', 'variants', 'images', 'reviews']);

        return new ProductResource($product);
    }

    public function store(Request $request)
    {
        $product = Product::create($request->validated());

        return new ProductResource($product);
    }
}
```

## Cara Integrasi & Pengujian
1. Buat resource untuk setiap model dengan format konsisten
2. Gunakan whenLoaded untuk conditional relationships
3. Test response untuk single resource dan collection
4. Test nested resources (category -> products)
5. Test conditional attributes (admin-only fields)
6. Test pagination meta di collection
7. Test dengan include query parameter untuk eager loading
8. Pastikan format tanggal konsisten (ISO8601)
9. Test response ketika relasi tidak di-load
