# 389. Mengimplementasikan Token-Based Authentication Menggunakan Sanctum Personal Access Tokens

## Definisi & Target
Implementasi lengkap token-based authentication menggunakan Sanctum personal access tokens untuk klien eksternal seperti mobile apps, third-party integrations, dan API consumers. Mendukung multiple tokens per user dengan abilities/scopes berbeda.

## Komponen yang Terlibat
- PersonalAccessToken model
- Token abilities / scopes
- Token management controller
- Token expiration & refresh
- Middleware abilities check
- Token revocation

## Perintah Artisan CLI
```bash
# Membuat controller
php artisan make:controller Api/V1/TokenController

# Membuat middleware custom (optional)
php artisan make:middleware ValidateTokenAbility
```

## Kode Implementasi (Per File)

### Model - User Token Management
```php
<?php
namespace App\Models;

use Illuminate\Foundation\Auth\User as Authenticatable;
use Laravel\Sanctum\HasApiTokens;

class User extends Authenticatable
{
    use HasApiTokens;

    public function createApiToken(string $name, array $abilities = ['basic']): string
    {
        return $this->createToken($name, $abilities)->plainTextToken;
    }

    public function getValidTokens()
    {
        return $this->tokens()
            ->whereNull('expires_at')
            ->orWhere('expires_at', '>', now())
            ->get();
    }

    public function revokeTokenById(int $tokenId): bool
    {
        return $this->tokens()->where('id', $tokenId)->delete() > 0;
    }

    public function revokeAllTokensExceptCurrent(): int
    {
        return $this->tokens()
            ->where('id', '!=', $this->currentAccessToken()?->id)
            ->delete();
    }

    public function hasTokenAbility(string $ability): bool
    {
        return $this->tokenCan($ability);
    }
}
```

### Controller - TokenController
```php
<?php
namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use Illuminate\Http\Request;

class TokenController extends Controller
{
    public function index(Request $request)
    {
        $tokens = $request->user()->tokens()->get()->map(function ($token) {
            return [
                'id' => $token->id,
                'name' => $token->name,
                'abilities' => $token->abilities,
                'last_used_at' => $token->last_used_at,
                'created_at' => $token->created_at,
                'expires_at' => $token->expires_at,
            ];
        });

        return response()->json(['data' => $tokens]);
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'abilities' => 'sometimes|array',
            'abilities.*' => 'string|in:basic,admin,read,write,files,webhooks',
            'expires_at' => 'sometimes|date|after:today',
        ]);

        $abilities = $validated['abilities'] ?? ['basic'];
        $expiresAt = $validated['expires_at'] ?? null;

        $token = $request->user()->createToken(
            $validated['name'],
            $abilities,
            $expiresAt ? \Carbon\Carbon::parse($expiresAt) : null
        );

        return response()->json([
            'token' => $token->plainTextToken,
            'token_id' => $token->accessToken->id,
            'name' => $validated['name'],
            'abilities' => $abilities,
            'expires_at' => $expiresAt,
        ], 201);
    }

    public function show(Request $request, int $tokenId)
    {
        $token = $request->user()->tokens()->findOrFail($tokenId);

        return response()->json([
            'data' => [
                'id' => $token->id,
                'name' => $token->name,
                'abilities' => $token->abilities,
                'last_used_at' => $token->last_used_at,
                'created_at' => $token->created_at,
                'expires_at' => $token->expires_at,
            ]
        ]);
    }

    public function update(Request $request, int $tokenId)
    {
        $token = $request->user()->tokens()->findOrFail($tokenId);

        $validated = $request->validate([
            'name' => 'sometimes|string|max:255',
            'abilities' => 'sometimes|array',
            'abilities.*' => 'string|in:basic,admin,read,write,files,webhooks',
        ]);

        if (isset($validated['name'])) {
            $token->name = $validated['name'];
        }

        if (isset($validated['abilities'])) {
            $token->abilities = $validated['abilities'];
        }

        $token->save();

        return response()->json(['message' => 'Token diperbarui.']);
    }

    public function destroy(Request $request, int $tokenId)
    {
        $request->user()->tokens()->where('id', $tokenId)->delete();

        return response()->json(['message' => 'Token dihapus.']);
    }

    public function destroyAll(Request $request)
    {
        $count = $request->user()->tokens()->count();
        $request->user()->tokens()->delete();

        return response()->json([
            'message' => "{$count} token berhasil dihapus.",
        ]);
    }

    public function current(Request $request)
    {
        $token = $request->user()->currentAccessToken();

        return response()->json([
            'data' => [
                'id' => $token->id,
                'name' => $token->name,
                'abilities' => $token->abilities,
                'created_at' => $token->created_at,
            ]
        ]);
    }
}
```

### Middleware - CheckTokenAbilities
```php
<?php
namespace App\Http\Middleware;

use Illuminate\Auth\AuthenticationException;
use Illuminate\Http\Request;
use Closure;

class CheckTokenAbilities
{
    public function handle(Request $request, Closure $next, string ...$abilities)
    {
        if (!$request->user() || !$request->user()->currentAccessToken()) {
            throw new AuthenticationException('Token tidak valid.');
        }

        foreach ($abilities as $ability) {
            if ($request->user()->tokenCan($ability)) {
                return $next($request);
            }
        }

        return response()->json([
            'message' => 'Token tidak memiliki akses ke resource ini.',
            'required_abilities' => $abilities,
            'token_abilities' => $request->user()->currentAccessToken()->abilities,
        ], 403);
    }
}
```

### Route - api/v1/tokens
```php
<?php
use App\Http\Controllers\Api\V1\TokenController;

Route::middleware('auth:sanctum')->group(function () {
    Route::get('/tokens', [TokenController::class, 'index']);
    Route::post('/tokens', [TokenController::class, 'store']);
    Route::get('/tokens/current', [TokenController::class, 'current']);
    Route::get('/tokens/{token}', [TokenController::class, 'show']);
    Route::put('/tokens/{token}', [TokenController::class, 'update']);
    Route::delete('/tokens/{token}', [TokenController::class, 'destroy']);
    Route::delete('/tokens', [TokenController::class, 'destroyAll']);
});
```
