# 399. Mengimplementasikan Webhook System untuk Notifikasi Event ke Sistem Eksternal

## Definisi & Target
Membangun webhook system yang mengirim notifikasi HTTP ke URL eksternal ketika event tertentu terjadi di aplikasi. Webhook dapat dikonfigurasi per event, memiliki retry mechanism, signature verification, dan logging. Targetnya adalah integrasi real-time dengan sistem pihak ketiga.

## Komponen yang Terlibat
- Webhook model dan migration
- WebhookEvent model
- Webhook dispatch job
- Signature generation (HMAC)
- Retry mechanism with backoff
- Webhook dashboard management
- Event logging

### Model - Webhook
```php
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Relations\HasMany;
use Illuminate\Database\Eloquent\SoftDeletes;
use Illuminate\Support\Str;

class Webhook extends Model
{
    use SoftDeletes;

    protected $fillable = [
        'user_id', 'name', 'url', 'secret', 'events',
        'headers', 'signature_header', 'retry_count',
        'timeout', 'is_active',
    ];

    protected $casts = [
        'events' => 'array',
        'headers' => 'array',
        'is_active' => 'boolean',
        'retry_count' => 'integer',
        'timeout' => 'integer',
    ];

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public function events(): HasMany
    {
        return $this->hasMany(WebhookEvent::class);
    }

    public function shouldHandle(string $eventType): bool
    {
        if (empty($this->events)) {
            return true;
        }

        return in_array('*', $this->events) || in_array($eventType, $this->events);
    }

    public function generateSignature(string $payload): string
    {
        $secret = $this->secret ?? config('app.key');

        return hash_hmac('sha256', $payload, $secret);
    }

    public function getHeaders(): array
    {
        $defaultHeaders = [
            'Content-Type' => 'application/json',
            'User-Agent' => config('app.name') . '-Webhook/1.0',
            'X-Webhook-ID' => (string) $this->id,
        ];

        return array_merge($defaultHeaders, $this->headers ?? []);
    }

    public static function supportedEvents(): array
    {
        return [
            'user.registered' => 'User baru mendaftar',
            'user.deleted' => 'User dihapus',
            'order.created' => 'Pesanan baru dibuat',
            'order.updated' => 'Status pesanan berubah',
            'order.completed' => 'Pesanan selesai',
            'order.cancelled' => 'Pesanan dibatalkan',
            'payment.received' => 'Pembayaran diterima',
            'payment.failed' => 'Pembayaran gagal',
            'product.created' => 'Produk baru ditambahkan',
            'product.updated' => 'Produk diperbarui',
            'product.deleted' => 'Produk dihapus',
            'subscription.created' => 'Langganan baru',
            'subscription.cancelled' => 'Langganan dibatalkan',
            'invoice.paid' => 'Invoice dibayar',
            'invoice.overdue' => 'Invoice jatuh tempo',
        ];
    }
}
```

### Job - DispatchWebhookJob
```php
<?php
namespace App\Jobs;

use App\Models\Webhook;
use App\Models\WebhookEvent;
use App\Models\WebhookLog;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;
use Illuminate\Support\Facades\Http;

class DispatchWebhookJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, SerializesModels;

    public int $timeout = 30;
    public int $tries = 3;
    public array $backoff = [10, 60, 300];

    public function __construct(
        public Webhook $webhook,
        public WebhookEvent $event
    ) {}

    public function handle(): void
    {
        $payload = json_encode($this->event->payload);
        $signature = $this->webhook->generateSignature($payload);

        $headers = $this->webhook->getHeaders();
        $headers[$this->webhook->signature_header] = $signature;
        $headers['X-Webhook-Event'] = $this->event->event_type;
        $headers['X-Webhook-Timestamp'] = (string) now()->timestamp;

        $startTime = microtime(true);

        try {
            $response = Http::timeout($this->webhook->timeout)
                ->withHeaders($headers)
                ->withBody($payload, 'application/json')
                ->post($this->webhook->url);

            $duration = (microtime(true) - $startTime) * 1000;

            WebhookLog::create([
                'webhook_event_id' => $this->event->id,
                'attempt_number' => $this->event->attempts + 1,
                'status_code' => $response->status(),
                'response_body' => substr($response->body(), 0, 5000),
                'duration_ms' => $duration,
            ]);

            if ($response->successful()) {
                $this->event->update([
                    'status' => 'completed',
                    'attempts' => $this->event->attempts + 1,
                    'completed_at' => now(),
                ]);
            } else {
                $this->handleFailure("HTTP {$response->status()}: {$response->body()}");
            }
        } catch (\Exception $e) {
            $duration = (microtime(true) - $startTime) * 1000;

            WebhookLog::create([
                'webhook_event_id' => $this->event->id,
                'attempt_number' => $this->event->attempts + 1,
                'error_message' => $e->getMessage(),
                'duration_ms' => $duration,
            ]);

            $this->handleFailure($e->getMessage());
        }
    }

    private function handleFailure(string $error): void
    {
        $attempts = $this->event->attempts + 1;

        if ($attempts >= $this->webhook->retry_count) {
            $this->event->update([
                'status' => 'failed',
                'attempts' => $attempts,
                'last_attempt_at' => now(),
            ]);
        } else {
            $nextAttempt = now()->addSeconds($this->backoff[$attempts - 1] ?? 300);

            $this->event->update([
                'status' => 'retrying',
                'attempts' => $attempts,
                'last_attempt_at' => now(),
                'next_attempt_at' => $nextAttempt,
            ]);

            static::dispatch($this->webhook, $this->event)
                ->delay($nextAttempt);
        }
    }
}
```

### Webhook Service
```php
<?php
namespace App\Services;

use App\Jobs\DispatchWebhookJob;
use App\Models\Webhook;
use App\Models\WebhookEvent;
use App\Models\User;
use Illuminate\Support\Str;

class WebhookService
{
    public function dispatch(string $eventType, array $payload, ?User $user = null): void
    {
        $query = Webhook::where('is_active', true);

        if ($user) {
            $query->where('user_id', $user->id);
        }

        $webhooks = $query->get();

        foreach ($webhooks as $webhook) {
            if ($webhook->shouldHandle($eventType)) {
                $event = WebhookEvent::create([
                    'webhook_id' => $webhook->id,
                    'event_type' => $eventType,
                    'payload' => $payload,
                    'status' => 'pending',
                    'max_attempts' => $webhook->retry_count,
                ]);

                DispatchWebhookJob::dispatch($webhook, $event);
            }
        }
    }

    public function dispatchForAll(string $eventType, array $payload): void
    {
        Webhook::where('is_active', true)
            ->chunk(100, function ($webhooks) use ($eventType, $payload) {
                foreach ($webhooks as $webhook) {
                    if ($webhook->shouldHandle($eventType)) {
                        $event = WebhookEvent::create([
                            'webhook_id' => $webhook->id,
                            'event_type' => $eventType,
                            'payload' => $payload,
                            'status' => 'pending',
                            'max_attempts' => $webhook->retry_count,
                        ]);

                        DispatchWebhookJob::dispatch($webhook, $event);
                    }
                }
            });
    }
}
```

### Controller - WebhookManagementController
```php
<?php
namespace App\Http\Controllers\Api\V1;

use App\Http\Controllers\Controller;
use App\Models\Webhook;
use Illuminate\Http\Request;

class WebhookManagementController extends Controller
{
    public function index(Request $request)
    {
        $webhooks = $request->user()->webhooks()
            ->withCount(['events' => function ($q) {
                $q->where('status', 'failed');
            }])
            ->latest()
            ->get();

        return response()->json(['data' => $webhooks]);
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'url' => 'required|url|max:500',
            'events' => 'nullable|array',
            'events.*' => 'string|in:' . implode(',', array_keys(Webhook::supportedEvents())),
            'retry_count' => 'nullable|integer|min:1|max:10',
            'timeout' => 'nullable|integer|min:1|max:30',
            'secret' => 'nullable|string|min:16',
        ]);

        $validated['secret'] ??= Str::random(32);
        $validated['user_id'] = $request->user()->id;

        $webhook = Webhook::create($validated);

        return response()->json([
            'data' => $webhook,
            'secret' => $validated['secret'],
            'message' => 'Webhook berhasil dibuat. Simpan secret untuk verifikasi signature.',
        ], 201);
    }

    public function show(Request $request, Webhook $webhook)
    {
        if ($webhook->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        $webhook->load(['events' => function ($q) {
            $q->latest()->take(20);
        }]);

        return response()->json(['data' => $webhook]);
    }

    public function update(Request $request, Webhook $webhook)
    {
        if ($webhook->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        $validated = $request->validate([
            'name' => 'sometimes|string|max:255',
            'url' => 'sometimes|url|max:500',
            'events' => 'sometimes|nullable|array',
            'events.*' => 'string',
            'is_active' => 'sometimes|boolean',
            'retry_count' => 'sometimes|integer|min:1|max:10',
        ]);

        $webhook->update($validated);

        return response()->json([
            'data' => $webhook->fresh(),
            'message' => 'Webhook berhasil diperbarui.',
        ]);
    }

    public function destroy(Request $request, Webhook $webhook)
    {
        if ($webhook->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        $webhook->delete();

        return response()->json(['message' => 'Webhook berhasil dihapus.']);
    }

    public function events(Request $request, Webhook $webhook)
    {
        if ($webhook->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        $events = $webhook->events()->with('logs')->latest()->paginate(50);

        return response()->json([
            'data' => $events->items(),
            'meta' => [
                'total' => $events->total(),
                'per_page' => $events->perPage(),
                'failed_count' => $events->where('status', 'failed')->count(),
            ],
        ]);
    }

    public function redeliver(Request $request, WebhookEvent $event)
    {
        $webhook = $event->webhook;

        if ($webhook->user_id !== $request->user()->id) {
            return response()->json(['message' => 'Forbidden.'], 403);
        }

        $newEvent = WebhookEvent::create([
            'webhook_id' => $webhook->id,
            'event_type' => $event->event_type,
            'payload' => $event->payload,
            'status' => 'pending',
            'max_attempts' => $webhook->retry_count,
        ]);

        DispatchWebhookJob::dispatch($webhook, $newEvent);

        return response()->json([
            'message' => 'Webhook redelivery initiated.',
            'data' => $newEvent,
        ]);
    }

    public function supportedEvents()
    {
        return response()->json([
            'data' => Webhook::supportedEvents(),
        ]);
    }
}
```

### Usage Example - Dispatching Webhooks from Events
```php
<?php
namespace App\Listeners;

use App\Events\OrderCreated;
use App\Services\WebhookService;

class DispatchOrderWebhooks
{
    public function __construct(
        private WebhookService $webhookService
    ) {}

    public function handle(OrderCreated $event): void
    {
        $this->webhookService->dispatch('order.created', [
            'order_id' => $event->order->id,
            'order_number' => $event->order->order_number,
            'total' => $event->order->total,
            'status' => $event->order->status,
            'customer' => [
                'name' => $event->order->user->name,
                'email' => $event->order->user->email,
            ],
            'items' => $event->order->items->map(function ($item) {
                return [
                    'product_id' => $item->product_id,
                    'quantity' => $item->quantity,
                    'price' => $item->price,
                ];
            }),
            'timestamp' => now()->toISO8601String(),
        ], $event->order->user);
    }
}
```

## Cara Integrasi & Pengujian
1. Buat webhook via endpoint POST /api/v1/webhooks
2. Setup webhook receiver (requestbin.com atau webhook.site)
3. Trigger event (buat order baru)
4. Lihat webhook terkirim di receiver
5. Verifikasi signature HMAC di header
6. Cek webhook log di endpoint events
7. Test retry mechanism dengan membuat receiver offline
8. Test redeliver untuk event yang gagal
9. Test filtering event type
10. Test webhook deactivation dan activation
