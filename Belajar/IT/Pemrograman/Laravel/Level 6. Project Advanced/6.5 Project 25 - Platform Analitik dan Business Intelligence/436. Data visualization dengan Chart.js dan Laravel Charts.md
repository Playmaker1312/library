# 436. Data Visualization dengan Chart.js dan Laravel Charts

## Definisi & Target
Mengimplementasikan data visualization menggunakan Chart.js terintegrasi dengan Laravel. Meliputi berbagai tipe chart (bar, line, pie, scatter, heatmap), responsive dashboards, real-time updates, dan data export.

## Komponen yang Terlibat
- Chart.js library
- Laravel Blade components
- Chart data serialization
- Interactive dashboards
- Real-time chart updates via Reverb
- Chart options & theming
- Export chart as image
- Responsive design

## Perintah Artisan CLI
```bash
# Install Chart.js via NPM
npm install chart.js

# Buat Blade components untuk charts
php artisan make:component Chart/LineChart
php artisan make:component Chart/BarChart
php artisan make:component Chart/PieChart

# Buat controller untuk chart data
php artisan make:controller Api/ChartDataController
```

### Chart Data Serializer
```php
<?php
namespace App\Services\Analytics;

class ChartDataSerializer
{
    public function toLineChart(array $data, string $labelField, string $valueField, array $options = []): array
    {
        return [
            'type' => 'line',
            'data' => [
                'labels' => array_column($data, $labelField),
                'datasets' => [
                    [
                        'label' => $options['label'] ?? $valueField,
                        'data' => array_map(fn ($row) => (float) ($row[$valueField] ?? 0), $data),
                        'borderColor' => $options['color'] ?? '#3b82f6',
                        'backgroundColor' => $options['fillColor'] ?? 'rgba(59, 130, 246, 0.1)',
                        'tension' => $options['tension'] ?? 0.3,
                        'fill' => $options['fill'] ?? true,
                    ],
                ],
            ],
            'options' => $this->buildOptions($options),
        ];
    }

    public function toBarChart(array $data, string $labelField, string $valueField, array $options = []): array
    {
        return [
            'type' => 'bar',
            'data' => [
                'labels' => array_column($data, $labelField),
                'datasets' => [
                    [
                        'label' => $options['label'] ?? $valueField,
                        'data' => array_map(fn ($row) => (float) ($row[$valueField] ?? 0), $data),
                        'backgroundColor' => $options['colors'] ?? [
                            '#3b82f6', '#ef4444', '#10b981', '#f59e0b', '#8b5cf6',
                            '#ec4899', '#14b8a6', '#f97316', '#6366f1', '#84cc16',
                        ],
                        'borderRadius' => 4,
                    ],
                ],
            ],
            'options' => $this->buildOptions($options),
        ];
    }

    public function toPieChart(array $data, string $labelField, string $valueField, array $options = []): array
    {
        return [
            'type' => 'doughnut',
            'data' => [
                'labels' => array_column($data, $labelField),
                'datasets' => [
                    [
                        'data' => array_map(fn ($row) => (float) ($row[$valueField] ?? 0), $data),
                        'backgroundColor' => $options['colors'] ?? [
                            '#3b82f6', '#ef4444', '#10b981', '#f59e0b', '#8b5cf6',
                            '#ec4899', '#14b8a6', '#f97316',
                        ],
                        'borderWidth' => 2,
                    ],
                ],
            ],
            'options' => $this->buildOptions($options),
        ];
    }

    public function toMultiLineChart(array $data, string $labelField, array $series, array $options = []): array
    {
        $datasets = [];

        foreach ($series as $key => $config) {
            $datasets[] = [
                'label' => $config['label'] ?? $key,
                'data' => array_map(fn ($row) => (float) ($row[$key] ?? 0), $data),
                'borderColor' => $config['color'] ?? $this->getColor(count($datasets)),
                'backgroundColor' => $config['fillColor'] ?? $this->getColor(count($datasets), 0.1),
                'tension' => 0.3,
                'fill' => false,
            ];
        }

        return [
            'type' => 'line',
            'data' => [
                'labels' => array_column($data, $labelField),
                'datasets' => $datasets,
            ],
            'options' => $this->buildOptions($options),
        ];
    }

    public function toStackedBarChart(array $data, string $labelField, array $series, array $options = []): array
    {
        $datasets = [];

        foreach ($series as $key => $config) {
            $datasets[] = [
                'label' => $config['label'] ?? $key,
                'data' => array_map(fn ($row) => (float) ($row[$key] ?? 0), $data),
                'backgroundColor' => $config['color'] ?? $this->getColor(count($datasets)),
            ];
        }

        return [
            'type' => 'bar',
            'data' => [
                'labels' => array_column($data, $labelField),
                'datasets' => $datasets,
            ],
            'options' => array_merge($this->buildOptions($options), [
                'scales' => [
                    'x' => ['stacked' => true],
                    'y' => ['stacked' => true],
                ],
            ]),
        ];
    }

    public function toScatterChart(array $data, string $xField, string $yField, array $options = []): array
    {
        return [
            'type' => 'scatter',
            'data' => [
                'datasets' => [
                    [
                        'label' => $options['label'] ?? 'Data',
                        'data' => array_map(fn ($row) => [
                            'x' => (float) ($row[$xField] ?? 0),
                            'y' => (float) ($row[$yField] ?? 0),
                        ], $data),
                        'backgroundColor' => $options['color'] ?? '#3b82f6',
                    ],
                ],
            ],
            'options' => $this->buildOptions($options),
        ];
    }

    private function buildOptions(array $options): array
    {
        return [
            'responsive' => true,
            'maintainAspectRatio' => $options['maintainAspectRatio'] ?? false,
            'plugins' => [
                'legend' => [
                    'display' => $options['showLegend'] ?? true,
                    'position' => $options['legendPosition'] ?? 'top',
                ],
                'tooltip' => [
                    'enabled' => $options['showTooltips'] ?? true,
                    'callbacks' => [
                        'label' => $options['tooltipFormat'] ?? null,
                    ],
                ],
            ],
            'scales' => $options['scales'] ?? [
                'y' => [
                    'beginAtZero' => true,
                    'ticks' => [
                        'callback' => $options['yAxisFormat'] ?? null,
                    ],
                ],
            ],
            'animation' => [
                'duration' => $options['animationDuration'] ?? 500,
            ],
        ];
    }

    private function getColor(int $index, float $alpha = 1): string
    {
        $colors = [
            [59, 130, 246], [239, 68, 68], [16, 185, 129], [245, 158, 11], [139, 92, 246],
            [236, 72, 153], [20, 184, 166], [249, 115, 22], [99, 102, 241], [132, 204, 22],
        ];

        $color = $colors[$index % count($colors)];
        return $alpha < 1
            ? "rgba({$color[0]}, {$color[1]}, {$color[2]}, {$alpha})"
            : "rgb({$color[0]}, {$color[1]}, {$color[2]})";
    }
}
```

### Blade Component for Charts
```php
<?php
namespace App\View\Components\Chart;

use Illuminate\View\Component;
use Illuminate\Support\Str;

class LineChart extends Component
{
    public string $chartId;
    public array $chartData;
    public int $height;
    public int $width;

    public function __construct(
        array $chartData,
        string $chartId = null,
        int $height = 300,
        int $width = 600
    ) {
        $this->chartId = $chartId ?? 'chart-' . Str::random(8);
        $this->chartData = $chartData;
        $this->height = $height;
        $this->width = $width;
    }

    public function render()
    {
        return view('components.chart.line-chart');
    }
}

// Same for BarChart, PieChart, etc.
```

### Chart API Controller
```php
<?php
namespace App\Http\Controllers\Api;

use App\Services\Analytics\ChartDataSerializer;
use App\Services\Analytics\MetricQueryBuilder;
use Illuminate\Http\Request;

class ChartDataController extends Controller
{
    public function __construct(
        private ChartDataSerializer $serializer,
        private MetricQueryBuilder $queryBuilder
    ) {}

    public function revenueTrend(Request $request): mixed
    {
        $data = $this->queryBuilder
            ->from('agg_daily_sales')
            ->select(['date_key', 'total_revenue'])
            ->period($request->input('period', 'last_30_days'))
            ->groupBy(['date_key', 'total_revenue'])
            ->orderBy('date_key')
            ->get();

        return $this->serializer->toLineChart(
            $data,
            'date_key',
            'total_revenue',
            ['label' => 'Revenue', 'yAxisFormat' => 'formatCurrency']
        );
    }

    public function salesByCategory(Request $request): mixed
    {
        $data = $this->queryBuilder
            ->from('fact_sales')
            ->select(['dim_product.category'])
            ->sum('total_amount', 'revenue')
            ->groupBy(['dim_product.category'])
            ->orderBy('revenue', 'desc')
            ->get();

        return $this->serializer->toPieChart(
            $data,
            'category',
            'revenue'
        );
    }

    public function dailyComparison(Request $request): mixed
    {
        $currentPeriod = $request->input('period', 'last_7_days');
        $previousPeriod = $request->input('previous_period', 'previous_7_days');

        $current = $this->queryBuilder
            ->from('agg_daily_sales')
            ->select(['date_key', 'total_revenue'])
            ->period($currentPeriod)
            ->groupBy(['date_key', 'total_revenue'])
            ->orderBy('date_key')
            ->get();

        $previous = $this->queryBuilder
            ->from('agg_daily_sales')
            ->select(['date_key', 'total_revenue'])
            ->period($previousPeriod)
            ->groupBy(['date_key', 'total_revenue'])
            ->orderBy('date_key')
            ->get();

        // Align dates for comparison
        $combined = [];
        foreach ($current as $i => $row) {
            $prev = $previous[$i] ?? null;
            $combined[] = [
                'label' => "Day " . ($i + 1),
                'current' => $row['total_revenue'],
                'previous' => $prev['total_revenue'] ?? 0,
            ];
        }

        return $this->serializer->toMultiLineChart(
            $combined,
            'label',
            [
                'current' => ['label' => 'Current Period', 'color' => '#3b82f6'],
                'previous' => ['label' => 'Previous Period', 'color' => '#9ca3af'],
            ]
        );
    }

    public function conversionFunnel(Request $request): mixed
    {
        $events = $this->queryBuilder
            ->from('event_tracking')
            ->select(['event_type'])
            ->count('*', 'count')
            ->whereIn('event_type', [
                'page_view', 'add_to_cart', 'checkout_started',
                'payment_info', 'order_completed',
            ])
            ->period($request->input('period', 'last_30_days'))
            ->groupBy(['event_type'])
            ->orderBy('count', 'desc')
            ->get();

        // Sort by funnel order
        $funnelOrder = ['page_view', 'add_to_cart', 'checkout_started', 'payment_info', 'order_completed'];
        $sorted = collect($events)->sortBy(fn ($e) => array_search($e['event_type'], $funnelOrder))->values()->toArray();

        return [
            'type' => 'bar',
            'data' => [
                'labels' => array_map(fn ($e) => ucwords(str_replace('_', ' ', $e['event_type'])), $sorted),
                'datasets' => [
                    [
                        'label' => 'Users',
                        'data' => array_map(fn ($e) => (int) $e['count'], $sorted),
                        'backgroundColor' => ['#3b82f6', '#60a5fa', '#93c5fd', '#bfdbfe', '#dbeafe'],
                        'borderRadius' => 4,
                    ],
                ],
            ],
            'options' => [
                'responsive' => true,
                'maintainAspectRatio' => false,
                'plugins' => [
                    'legend' => ['display' => false],
                ],
                'scales' => [
                    'y' => ['beginAtZero' => true],
                ],
            ],
        ];
    }
}
```

### Dashboard Widget with Real-Time Updates
```php
<?php
namespace App\Http\Controllers\Analytics;

use App\Services\Analytics\ChartDataSerializer;
use Illuminate\Http\Request;

class DashboardController extends Controller
{
    public function __construct(
        private ChartDataSerializer $serializer
    ) {}

    public function index(): mixed
    {
        return view('analytics.dashboard', [
            'widgets' => $this->getDefaultWidgets(),
        ]);
    }

    public function widgetData(string $widgetKey, Request $request): mixed
    {
        return match ($widgetKey) {
            'revenue_today' => $this->getRevenueToday(),
            'active_users' => $this->getActiveUsers(),
            'sales_chart' => app(ChartDataController::class)->revenueTrend($request),
            'category_pie' => app(ChartDataController::class)->salesByCategory($request),
            'conversion_funnel' => app(ChartDataController::class)->conversionFunnel($request),
            default => response()->json(['error' => 'Unknown widget'], 404),
        };
    }

    public function getRevenueToday(): mixed
    {
        $revenue = DB::connection('warehouse')
            ->table('agg_daily_sales')
            ->where('date_key', (int) now()->format('Ymd'))
            ->value('total_revenue') ?? 0;

        $yesterday = DB::connection('warehouse')
            ->table('agg_daily_sales')
            ->where('date_key', (int) now()->subDay()->format('Ymd'))
            ->value('total_revenue') ?? 0;

        $growth = $yesterday > 0 ? round(($revenue - $yesterday) / $yesterday * 100, 1) : 0;

        return response()->json([
            'value' => $revenue,
            'prefix' => 'Rp',
            'growth' => $growth,
            'growthLabel' => 'vs yesterday',
        ]);
    }

    public function getActiveUsers(): mixed
    {
        $active = DB::connection('warehouse')
            ->table('agg_daily_users')
            ->where('date_key', (int) now()->format('Ymd'))
            ->value('active_users') ?? 0;

        return response()->json([
            'value' => $active,
            'icon' => 'users',
            'description' => 'Active users today',
        ]);
    }

    private function getDefaultWidgets(): array
    {
        return [
            ['key' => 'revenue_today', 'type' => 'stat', 'size' => '1/4', 'refresh' => 60],
            ['key' => 'active_users', 'type' => 'stat', 'size' => '1/4', 'refresh' => 30],
            ['key' => 'sales_chart', 'type' => 'chart', 'size' => '1/2', 'refresh' => 300],
            ['key' => 'category_pie', 'type' => 'chart', 'size' => '1/3', 'refresh' => 600],
            ['key' => 'conversion_funnel', 'type' => 'chart', 'size' => '2/3', 'refresh' => 600],
        ];
    }
}
```
