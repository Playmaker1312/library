# 435. OLAP Cube Implementation dengan Eloquent untuk Multi-Dimensional Analysis

## Definisi & Target
Mengimplementasikan OLAP (Online Analytical Processing) cube menggunakan Eloquent ORM untuk analisis multi-dimensional. Meliputi cube definition, dimension hierarchies, drill-down/roll-up, slice & dice, dan pivot operations.

## Komponen yang Terlibat
- OLAP Cube concept
- Dimensions & hierarchies
- Measures & facts
- Drill-down / Roll-up
- Slice & Dice
- Pivot (cross-tabulation)
- MDX-like queries
- Eloquent relationships for dimensions

### OLAP Cube Definition
```php
<?php
namespace App\Services\Analytics\Olap;

class OlapCube
{
    private string $name;
    private array $measures = [];
    private array $dimensions = [];
    private array $hierarchies = [];

    public function __construct(string $name)
    {
        $this->name = $name;
    }

    public function addMeasure(string $name, string $column, string $aggregation = 'SUM'): self
    {
        $this->measures[$name] = [
            'column' => $column,
            'aggregation' => $aggregation,
        ];
        return $this;
    }

    public function addDimension(string $name, string $table, string $key, array $hierarchy = []): self
    {
        $this->dimensions[$name] = [
            'table' => $table,
            'key' => $key,
            'hierarchy' => $hierarchy,
        ];
        return $this;
    }

    public function query(): CubeQueryBuilder
    {
        return new CubeQueryBuilder($this);
    }

    public function getName(): string
    {
        return $this->name;
    }

    public function getMeasures(): array
    {
        return $this->measures;
    }

    public function getDimensions(): array
    {
        return $this->dimensions;
    }

    public function getHierarchies(string $dimension): array
    {
        return $this->dimensions[$dimension]['hierarchy'] ?? [];
    }
}
```

### Cube Query Builder
```php
<?php
namespace App\Services\Analytics\Olap;

use Illuminate\Support\Facades\DB;

class CubeQueryBuilder
{
    private OlapCube $cube;
    private array $selectedMeasures = [];
    private array $rows = [];
    private array $columns = [];
    private array $filters = [];

    public function __construct(OlapCube $cube)
    {
        $this->cube = $cube;
    }

    public function select(array $measures): self
    {
        $this->selectedMeasures = $measures;
        return $this;
    }

    public function onRows(array $dimensions): self
    {
        $this->rows = $dimensions;
        return $this;
    }

    public function onColumns(array $dimensions): self
    {
        $this->columns = $dimensions;
        return $this;
    }

    public function where(string $dimension, string $operator, $value): self
    {
        $this->filters[] = compact('dimension', 'operator', 'value');
        return $this;
    }

    public function drillDown(string $dimension, string $hierarchyLevel): self
    {
        // Add lower hierarchy level to rows
        $hierarchy = $this->cube->getHierarchies($dimension);
        $levelIndex = array_search($hierarchyLevel, $hierarchy);

        if ($levelIndex !== false && isset($hierarchy[$levelIndex + 1])) {
            $this->rows[] = "{$dimension}.{$hierarchy[$levelIndex + 1]}";
        }

        return $this;
    }

    public function rollUp(string $dimension): self
    {
        // Remove lowest hierarchy level
        $dimRows = array_filter($this->rows, fn ($r) => str_starts_with($r, $dimension));
        if (!empty($dimRows)) {
            $lastKey = array_key_last($dimRows);
            unset($this->rows[$lastKey]);
        }

        return $this;
    }

    public function execute(): array
    {
        $factTable = $this->getFactTable();
        $query = DB::connection('warehouse')->table($factTable);

        // Join dimensions
        foreach ($this->getUsedDimensions() as $dimName => $dimConfig) {
            $query->join(
                $dimConfig['table'],
                "{$factTable}.{$dimConfig['key']}",
                '=',
                "{$dimConfig['table']}.{$dimConfig['key']}"
            );
        }

        // Select measures
        $measureSelects = [];
        foreach ($this->selectedMeasures as $measureName) {
            $measure = $this->cube->getMeasures()[$measureName];
            $alias = "measure_{$measureName}";
            $query->selectRaw("{$measure['aggregation']}({$measure['column']}) as {$alias}");
            $measureSelects[] = $alias;
        }

        // Select dimension columns for rows
        $groupBy = [];
        foreach ($this->rows as $row) {
            $parts = explode('.', $row);
            $table = $parts[0];
            $column = $this->getDimensionColumn($table, $parts[1] ?? null);
            $query->addSelect("{$column} as {$row}");
            $groupBy[] = $column;
        }

        // Select dimension columns for columns (pivot)
        foreach ($this->columns as $col) {
            $parts = explode('.', $col);
            $column = $this->getDimensionColumn($parts[0], $parts[1] ?? null);
            $query->addSelect("{$column} as {$col}");
            $groupBy[] = $column;
        }

        // Apply filters
        foreach ($this->filters as $filter) {
            $column = $this->getDimensionColumn(
                $filter['dimension'],
                $filter['operator']
            );
            // Simplified - should use proper operator handling
            $query->where($column, $filter['operator'], $filter['value']);
        }

        $query->groupBy($groupBy);

        $results = $query->get()->toArray();

        return $this->pivotResults($results);
    }

    private function getUsedDimensions(): array
    {
        $used = [];
        $allDimensions = array_merge($this->rows, $this->columns);

        foreach ($allDimensions as $dim) {
            $dimName = explode('.', $dim)[0];
            if (isset($this->cube->getDimensions()[$dimName])) {
                $used[$dimName] = $this->cube->getDimensions()[$dimName];
            }
        }

        return $used;
    }

    private function getFactTable(): string
    {
        // Assumes fact table naming convention
        return strtolower($this->cube->getName()) . '_fact';
    }

    private function getDimensionColumn(string $dimName, ?string $attribute): string
    {
        $dim = $this->cube->getDimensions()[$dimName] ?? [];
        if ($attribute) {
            return "{$dim['table']}.{$attribute}";
        }
        return "{$dim['table']}.{$dim['key']}";
    }

    private function pivotResults(array $results): array
    {
        if (empty($this->columns)) {
            return $results;
        }

        $pivoted = [];
        $rowKeys = $this->rows;
        $colKey = $this->columns[0];

        foreach ($results as $row) {
            $rowIdentifier = [];
            foreach ($rowKeys as $rk) {
                $rowIdentifier[$rk] = $row->$rk;
            }
            $rowHash = md5(json_encode($rowIdentifier));

            if (!isset($pivoted[$rowHash])) {
                $pivoted[$rowHash] = $rowIdentifier;
            }

            $colValue = $row->$colKey ?? 'unknown';
            foreach ($this->selectedMeasures as $measure) {
                $measureAlias = "measure_{$measure}";
                $pivoted[$rowHash]["{$colValue}_{$measure}"] = $row->$measureAlias ?? 0;
            }
        }

        return array_values($pivoted);
    }
}
```

### Sales Cube Definition
```php
<?php
namespace App\Services\Analytics\Olap\Cubes;

use App\Services\Analytics\Olap\OlapCube;

class SalesCube
{
    public static function define(): OlapCube
    {
        $cube = new OlapCube('sales');

        // Measures
        $cube->addMeasure('revenue', 'fact_sales.total_amount', 'SUM');
        $cube->addMeasure('units_sold', 'fact_sales.quantity', 'SUM');
        $cube->addMeasure('order_count', 'fact_sales.source_order_id', 'COUNT_DISTINCT');
        $cube->addMeasure('avg_order_value', 'fact_sales.total_amount', 'AVG');
        $cube->addMeasure('discount_total', 'fact_sales.discount_amount', 'SUM');

        // Dimensions
        $cube->addDimension('time', 'dim_date', 'date_key', [
            'year', 'quarter', 'month', 'day',
        ]);

        $cube->addDimension('product', 'dim_product', 'product_key', [
            'category', 'subcategory', 'brand', 'product',
        ]);

        $cube->addDimension('customer', 'dim_customer', 'customer_key', [
            'country', 'province', 'city',
        ]);

        return $cube;
    }
}
```

### OLAP Controller
```php
<?php
namespace App\Http\Controllers\Analytics;

use App\Services\Analytics\Olap\Cubes\SalesCube;
use App\Services\Analytics\Olap\OlapCube;
use Illuminate\Http\Request;

class OlapController extends Controller
{
    public function sales(Request $request): mixed
    {
        $cube = SalesCube::define();
        $builder = $cube->query();

        $measures = $request->input('measures', ['revenue', 'units_sold']);
        $rows = $request->input('rows', ['time.year']);
        $columns = $request->input('columns', []);

        $builder->select($measures);
        $builder->onRows($rows);
        $builder->onColumns($columns);

        if ($request->has('filters')) {
            foreach ($request->input('filters') as $filter) {
                $builder->where($filter['dimension'], $filter['operator'], $filter['value']);
            }
        }

        if ($request->has('drill')) {
            foreach ($request->input('drill') as $drill) {
                $builder->drillDown($drill['dimension'], $drill['level']);
            }
        }

        $results = $builder->execute();

        return response()->json([
            'cube' => $cube->getName(),
            'measures' => $measures,
            'dimensions' => [
                'rows' => $rows,
                'columns' => $columns,
            ],
            'total_cells' => count($results),
            'data' => $results,
        ]);
    }

    public function dimensions(): mixed
    {
        $cube = SalesCube::define();

        return response()->json([
            'measures' => $cube->getMeasures(),
            'dimensions' => collect($cube->getDimensions())->map(function ($dim, $name) {
                return [
                    'name' => $name,
                    'table' => $dim['table'],
                    'hierarchy' => $dim['hierarchy'],
                    'levels' => $dim['hierarchy'],
                ];
            }),
        ]);
    }

    public function drillOptions(): mixed
    {
        return response()->json([
            'time' => ['year', 'quarter', 'month', 'day'],
            'product' => ['category', 'subcategory', 'brand', 'product'],
            'customer' => ['country', 'province', 'city'],
        ]);
    }
}
```

### Cube Data Analysis Example
```php
<?php
namespace App\Services\Analytics\Olap;

class CubeAnalysisService
{
    public function analyzeSalesTrend(): array
    {
        $cube = Cubes\SalesCube::define();

        // Year-over-year comparison
        $thisYear = $cube->query()
            ->select(['revenue', 'units_sold', 'order_count'])
            ->onRows(['time.month'])
            ->where('time.year', '=', now()->year)
            ->execute();

        $lastYear = $cube->query()
            ->select(['revenue', 'units_sold', 'order_count'])
            ->onRows(['time.month'])
            ->where('time.year', '=', now()->subYear()->year)
            ->execute();

        return [
            'this_year' => $thisYear,
            'last_year' => $lastYear,
            'growth' => $this->calculateGrowth($thisYear, $lastYear),
        ];
    }

    public function analyzeProductPerformance(): array
    {
        $cube = Cubes\SalesCube::define();

        $byCategory = $cube->query()
            ->select(['revenue', 'units_sold'])
            ->onRows(['product.category'])
            ->execute();

        $byBrand = $cube->query()
            ->select(['revenue', 'units_sold'])
            ->onRows(['product.brand'])
            ->execute();

        return [
            'by_category' => $byCategory,
            'by_brand' => $byBrand,
        ];
    }

    public function crossTabProductByMonth(): array
    {
        $cube = Cubes\SalesCube::define();

        return $cube->query()
            ->select(['revenue'])
            ->onRows(['product.category'])
            ->onColumns(['time.month'])
            ->where('time.year', '=', now()->year)
            ->execute();
    }

    private function calculateGrowth(array $current, array $previous): array
    {
        $growth = [];

        foreach ($current as $i => $row) {
            $prev = $previous[$i] ?? null;
            if ($prev) {
                $currentRevenue = $row['measure_revenue'] ?? 0;
                $previousRevenue = $prev['measure_revenue'] ?? 0;
                $growth[] = [
                    'period' => $row['time.month'] ?? $i,
                    'current' => $currentRevenue,
                    'previous' => $previousRevenue,
                    'growth_pct' => $previousRevenue > 0
                        ? round(($currentRevenue - $previousRevenue) / $previousRevenue * 100, 2)
                        : 0,
                ];
            }
        }

        return $growth;
    }
}
```

### Eloquent Warehouse Models
```php
<?php
namespace App\Models\Warehouse;

use Illuminate\Database\Eloquent\Model;

class FactSales extends Model
{
    protected $connection = 'warehouse';
    protected $table = 'fact_sales';
    public $timestamps = false;

    public function product()
    {
        return $this->belongsTo(DimProduct::class, 'product_key');
    }

    public function date()
    {
        return $this->belongsTo(DimDate::class, 'date_key', 'date_key');
    }

    public function customer()
    {
        return $this->belongsTo(DimCustomer::class, 'customer_key');
    }
}

class DimProduct extends Model
{
    protected $connection = 'warehouse';
    protected $table = 'dim_product';
    protected $primaryKey = 'product_key';
    public $timestamps = false;

    public function sales()
    {
        return $this->hasMany(FactSales::class, 'product_key');
    }

    public function scopeCurrent($query)
    {
        return $query->where('is_current', true);
    }

    public function scopeCategory($query, string $category)
    {
        return $query->where('category', $category);
    }
}

class DimDate extends Model
{
    protected $connection = 'warehouse';
    protected $table = 'dim_date';
    protected $primaryKey = 'date_key';
    public $timestamps = false;

    public function sales()
    {
        return $this->hasMany(FactSales::class, 'date_key');
    }

    public function scopeYear($query, int $year)
    {
        return $query->where('year', $year);
    }

    public function scopeMonth($query, int $month)
    {
        return $query->where('month', $month);
    }
}
```
