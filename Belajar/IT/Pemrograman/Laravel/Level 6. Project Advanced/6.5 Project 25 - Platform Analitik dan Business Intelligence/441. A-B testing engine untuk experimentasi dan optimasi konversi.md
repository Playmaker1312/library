# 441. A/B Testing Engine untuk Experimentasi dan Optimasi Konversi

## Definisi & Target
Membangun A/B testing engine untuk menjalankan eksperimen terhadap UI, fitur, dan konten. Meliputi experiment creation, user assignment, variant rendering, metrics tracking, statistical analysis, dan result reporting.

## Komponen yang Terlibat
- Experiment model & management
- User assignment (consistent hashing)
- Variant rendering
- Conversion tracking
- Statistical significance testing
- Sample size calculation
- Experiment dashboard
- Feature flags integration

```php
namespace App\Models\Analytics;

    protected $fillable = [
        'name',
        'description',
        'hypothesis',
        'primary_metric',
        'secondary_metrics',
        'status',
        'type',
        'target_audience',
        'sample_size',
        'confidence_level',
        'minimum_effect',
        'started_at',
        'ended_at',
    ];

    protected $casts = [
        'secondary_metrics' => 'array',
        'target_audience' => 'array',
        'started_at' => 'datetime',
        'ended_at' => 'datetime',
    ];

    const STATUS_DRAFT = 'draft';
    const STATUS_RUNNING = 'running';
    const STATUS_PAUSED = 'paused';
    const STATUS_COMPLETED = 'completed';
    const STATUS_ANALYZING = 'analyzing';

    public function variants()
    {
        return $this->hasMany(ExperimentVariant::class);
    }

    public function assignments()
    {
        return $this->hasMany(ExperimentAssignment::class);
    }

    public function events()
    {
        return $this->hasMany(ExperimentEvent::class);
    }

    public function controlVariant()
    {
        return $this->variants()->where('is_control', true)->first();
    }

    public function isRunning(): bool
    {
        return $this->status === self::STATUS_RUNNING;
    }

    public function getTrafficAllocation(): array
    {
        $total = $this->variants->sum('traffic_percentage');
        return $this->variants->mapWithKeys(fn ($v) => [
            $v->id => $v->traffic_percentage / $total,
        ])->toArray();
    }
}
```

### Experiment Service
```php
<?php
namespace App\Services\Analytics;

use App\Models\Analytics\Experiment;
use App\Models\Analytics\ExperimentVariant;
use App\Models\Analytics\ExperimentAssignment;
use App\Models\Analytics\ExperimentEvent;
use Illuminate\Support\Facades\DB;

class ExperimentService
{
    public function assignUser(Experiment $experiment, int $userId, ?string $sessionId = null): ExperimentVariant
    {
        $existing = ExperimentAssignment::where('experiment_id', $experiment->id)
            ->where('user_id', $userId)
            ->first();

        if ($existing) {
            return $existing->variant;
        }

        $variant = $this->selectVariant($experiment, $userId);

        ExperimentAssignment::create([
            'experiment_id' => $experiment->id,
            'variant_id' => $variant->id,
            'user_id' => $userId,
            'session_id' => $sessionId,
            'assigned_at' => now(),
        ]);

        return $variant;
    }

    public function trackEvent(
        Experiment $experiment,
        int $userId,
        string $eventType,
        ?float $eventValue = null,
        array $metadata = []
    ): void {
        $assignment = ExperimentAssignment::where('experiment_id', $experiment->id)
            ->where('user_id', $userId)
            ->first();

        if (!$assignment) {
            return;
        }

        ExperimentEvent::create([
            'experiment_id' => $experiment->id,
            'variant_id' => $assignment->variant_id,
            'assignment_id' => $assignment->id,
            'user_id' => $userId,
            'session_id' => $assignment->session_id,
            'event_type' => $eventType,
            'event_value' => $eventValue,
            'metadata' => $metadata,
            'occurred_at' => now(),
        ]);
    }

    public function getResults(Experiment $experiment): array
    {
        $variants = $experiment->variants;
        $results = [];

        foreach ($variants as $variant) {
            $assignments = ExperimentAssignment::where('variant_id', $variant->id)->count();
            $conversions = ExperimentEvent::where('variant_id', $variant->id)
                ->where('event_type', $experiment->primary_metric)
                ->count();

            $results[$variant->id] = [
                'variant_name' => $variant->name,
                'is_control' => $variant->is_control,
                'assignments' => $assignments,
                'conversions' => $conversions,
                'conversion_rate' => $assignments > 0 ? $conversions / $assignments : 0,
                'traffic_percentage' => $variant->traffic_percentage,
            ];
        }

        $controlId = $experiment->controlVariant()->id;
        $controlRate = $results[$controlId]['conversion_rate'];

        foreach ($results as $id => &$result) {
            if ($id !== $controlId && $controlRate > 0) {
                $result['improvement'] = (($result['conversion_rate'] - $controlRate) / $controlRate) * 100;
                $result['significance'] = $this->calculateSignificance(
                    $results[$controlId]['assignments'],
                    $results[$controlId]['conversions'],
                    $result['assignments'],
                    $result['conversions']
                );
            } else {
                $result['improvement'] = 0;
                $result['significance'] = 0;
            }
        }

        return $results;
    }

    public function calculateSampleSize(float $baselineRate, float $minimumEffect, float $confidence = 0.95, float $power = 0.8): int
    {
        $alpha = 1 - $confidence;
        $beta = 1 - $power;

        $zAlpha = $this->zScore(1 - $alpha / 2);
        $zBeta = $this->zScore(1 - $beta);

        $p1 = $baselineRate;
        $p2 = $baselineRate * (1 + $minimumEffect);
        $pAvg = ($p1 + $p2) / 2;

        $numerator = pow($zAlpha * sqrt(2 * $pAvg * (1 - $pAvg)) + $zBeta * sqrt($p1 * (1 - $p1) + $p2 * (1 - $p2)), 2);
        $denominator = pow($p2 - $p1, 2);

        return (int) ceil($numerator / $denominator);
    }

    private function selectVariant(Experiment $experiment, int $userId): ExperimentVariant
    {
        $hash = crc32("{$experiment->id}_{$userId}");
        $normalized = abs($hash) / 2147483647;

        $cumulative = 0;
        $variants = $experiment->variants;

        foreach ($variants as $variant) {
            $cumulative += $variant->traffic_percentage / 100;
            if ($normalized <= $cumulative) {
                return $variant;
            }
        }

        return $variants->first();
    }

    private function calculateSignificance(int $n1, int $c1, int $n2, int $c2): float
    {
        if ($n1 === 0 || $n2 === 0) return 0;

        $p1 = $c1 / $n1;
        $p2 = $c2 / $n2;
        $pPooled = ($c1 + $c2) / ($n1 + $n2);

        $se = sqrt($pPooled * (1 - $pPooled) * (1/$n1 + 1/$n2));

        if ($se == 0) return 0;

        $z = ($p2 - $p1) / $se;
        $pValue = 2 * (1 - $this->normalCDF(abs($z)));

        return 1 - $pValue;
    }

    private function zScore(float $p): float
    {
        // Approximation of inverse normal CDF
        $t = sqrt(-2 * log(min($p, 1 - $p)));
        $c = [2.515517, 0.802853, 0.010328];
        $d = [1.432788, 0.189269, 0.001308];
        return $t - ($c[0] + $c[1] * $t + $c[2] * $t * $t) / (1 + $d[0] * $t + $d[1] * $t * $t + $d[2] * $t * $t * $t);
    }

    private function normalCDF(float $x): float
    {
        return 0.5 * (1 + erf($x / sqrt(2)));
    }
}
```

### Experiment Controller
```php
<?php
namespace App\Http\Controllers\Analytics;

use App\Models\Analytics\Experiment;
use App\Services\Analytics\ExperimentService;
use Illuminate\Http\Request;

class ExperimentController extends Controller
{
    public function __construct(private ExperimentService $experimentService) {}

    public function index(): mixed
    {
        return Experiment::with('variants')
            ->orderBy('created_at', 'desc')
            ->paginate(20);
    }

    public function store(Request $request): mixed
    {
        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'description' => 'nullable|string',
            'hypothesis' => 'required|string',
            'primary_metric' => 'required|string',
            'secondary_metrics' => 'nullable|array',
            'type' => 'required|in:ui,feature,content,algorithm',
            'target_audience' => 'nullable|array',
            'confidence_level' => 'nullable|numeric|min:0|max:1',
            'minimum_effect' => 'nullable|numeric|min:0|max:1',
            'variants' => 'required|array|min:2',
            'variants.*.name' => 'required|string',
            'variants.*.config' => 'nullable|array',
            'variants.*.traffic_percentage' => 'required|integer|min:0|max:100',
        ]);

        DB::beginTransaction();
        try {
            $experiment = Experiment::create([
                'name' => $validated['name'],
                'description' => $validated['description'],
                'hypothesis' => $validated['hypothesis'],
                'primary_metric' => $validated['primary_metric'],
                'secondary_metrics' => $validated['secondary_metrics'] ?? [],
                'type' => $validated['type'],
                'target_audience' => $validated['target_audience'] ?? [],
                'status' => Experiment::STATUS_DRAFT,
                'confidence_level' => $validated['confidence_level'] ?? 0.95,
                'minimum_effect' => $validated['minimum_effect'] ?? 0.05,
            ]);

            foreach ($validated['variants'] as $i => $variantData) {
                ExperimentVariant::create([
                    'experiment_id' => $experiment->id,
                    'name' => $variantData['name'],
                    'config' => $variantData['config'] ?? [],
                    'traffic_percentage' => $variantData['traffic_percentage'],
                    'is_control' => $i === 0,
                ]);
            }

            DB::commit();
            return response()->json($experiment->load('variants'), 201);
        } catch (\Exception $e) {
            DB::rollBack();
            throw $e;
        }
    }

    public function show(Experiment $experiment): mixed
    {
        return response()->json([
            'experiment' => $experiment->load('variants'),
            'results' => $this->experimentService->getResults($experiment),
        ]);
    }

    public function start(Experiment $experiment): mixed
    {
        $totalTraffic = $experiment->variants->sum('traffic_percentage');
        if ($totalTraffic !== 100) {
            return response()->json(['error' => 'Traffic allocation must total 100%'], 422);
        }

        $variantCount = $experiment->variants->count();
        if ($variantCount < 2) {
            return response()->json(['error' => 'At least 2 variants required'], 422);
        }

        $baseline = 0.1;
        $sampleSize = $this->experimentService->calculateSampleSize(
            $baseline,
            $experiment->minimum_effect,
            $experiment->confidence_level
        );

        $experiment->update([
            'status' => Experiment::STATUS_RUNNING,
            'sample_size' => $sampleSize,
            'started_at' => now(),
        ]);

        return response()->json(['message' => 'Experiment started', 'required_sample_size' => $sampleSize]);
    }

    public function stop(Experiment $experiment): mixed
    {
        $experiment->update([
            'status' => Experiment::STATUS_COMPLETED,
            'ended_at' => now(),
        ]);

        return response()->json(['message' => 'Experiment stopped']);
    }

    public function results(Experiment $experiment): mixed
    {
        return response()->json([
            'experiment' => $experiment->load('variants'),
            'results' => $this->experimentService->getResults($experiment),
            'is_significant' => $this->checkSignificance($experiment),
        ]);
    }

    private function checkSignificance(Experiment $experiment): bool
    {
        $results = $this->experimentService->getResults($experiment);
        $control = collect($results)->firstWhere('is_control', true);

        if (!$control || $control['assignments'] < 100) return false;

        foreach ($results as $result) {
            if (!$result['is_control'] && ($result['significance'] ?? 0) >= $experiment->confidence_level) {
                return true;
            }
        }

        return false;
    }
}
```

### Experiment Middleware (Variant Rendering)
```php
<?php
namespace App\Http\Middleware;

use App\Models\Analytics\Experiment;
use App\Services\Analytics\ExperimentService;
use Closure;
use Illuminate\Http\Request;

class ExperimentMiddleware
{
    public function __construct(private ExperimentService $experimentService) {}

    public function handle(Request $request, Closure $next): mixed
    {
        $user = $request->user();
        if (!$user) return $next($request);

        $activeExperiments = Experiment::where('status', Experiment::STATUS_RUNNING)->get();

        $variants = [];
        foreach ($activeExperiments as $experiment) {
            $variant = $this->experimentService->assignUser($experiment, $user->id, session()->getId());
            $variants[$experiment->id] = [
                'experiment' => $experiment->name,
                'variant' => $variant->name,
                'config' => $variant->config,
            ];
        }

        $request->attributes->set('experiments', $variants);
        view()->share('experiments', $variants);

        return $next($request);
    }
}
```

### Experiment Analysis Command
```php
<?php
namespace App\Console\Commands;

use App\Models\Analytics\Experiment;
use App\Services\Analytics\ExperimentService;
use Illuminate\Console\Command;

class AnalyzeExperiments extends Command
{
    protected $signature = 'experiments:analyze';
    protected $description = 'Analyze running experiments for significance';

    public function handle(ExperimentService $service): int
    {
        $experiments = Experiment::where('status', Experiment::STATUS_RUNNING)->get();

        foreach ($experiments as $experiment) {
            $this->line("Analyzing: {$experiment->name}");

            $results = $service->getResults($experiment);
            $control = collect($results)->firstWhere('is_control', true);

            if (!$control || $control['assignments'] < $experiment->sample_size) {
                $this->info("  Still collecting data ({$control['assignments']}/{$experiment->sample_size})");
                continue;
            }

            foreach ($results as $variantId => $result) {
                if ($result['is_control']) continue;

                if (($result['significance'] ?? 0) >= $experiment->confidence_level) {
                    $this->warn("  Winner found: {$result['variant_name']} " .
                        "({$result['improvement']}% improvement, " .
                        round($result['significance'] * 100, 1) . "% significance)");
                }
            }
        }

        return self::SUCCESS;
    }
}
```
