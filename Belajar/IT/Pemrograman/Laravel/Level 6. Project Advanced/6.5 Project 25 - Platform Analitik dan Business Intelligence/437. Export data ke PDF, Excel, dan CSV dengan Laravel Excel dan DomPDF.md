# 437. Export Data ke PDF, Excel, dan CSV dengan Laravel Excel dan DomPDF

## Definisi & Target
Mengimplementasikan export data untuk reports dalam berbagai format: Excel, CSV, PDF. Menggunakan Laravel Excel (Maatwebsite) untuk spreadsheet dan DomPDF/Barryvdh untuk PDF generation. Targetnya adalah report export yang reliable dengan formatting profesional.

## Komponen yang Terlibat
- Laravel Excel (Maatwebsite)
- DomPDF (Barryvdh)
- CSV generation
- Export jobs (queue)
- Template-based exports
- Large dataset streaming
- Export history
- Format configuration

## Perintah Artisan CLI
```bash
# Install packages
composer require maatwebsite/laravel-excel
composer require barryvdh/laravel-dompdf

# Publish config
php artisan vendor:publish --provider="Maatwebsite\LaravelExcel\ExcelServiceProvider"
php artisan vendor:publish --provider="Barryvdh\DomPDF\ServiceProvider"

# Buat export classes
php artisan make:export SalesReportExport
php artisan make:export UsersExport
php artisan make:export ProductsExport

# Buat job untuk background export
php artisan make:job GenerateReportJob
```

### Sales Report Export (Excel)
```php
<?php
namespace App\Exports;

use App\Services\Analytics\MetricQueryBuilder;
use Maatwebsite\Excel\Concerns\FromQuery;
use Maatwebsite\Excel\Concerns\WithHeadings;
use Maatwebsite\Excel\Concerns\WithMapping;
use Maatwebsite\Excel\Concerns\WithStyles;
use Maatwebsite\Excel\Concerns\ShouldAutoSize;
use Maatwebsite\Excel\Concerns\WithTitle;
use PhpOffice\PhpSpreadsheet\Worksheet\Worksheet;
use PhpOffice\PhpSpreadsheet\Style\NumberFormat;

class SalesReportExport implements FromQuery, WithHeadings, WithMapping, WithStyles, ShouldAutoSize, WithTitle
{
    public function __construct(
        private string $period = 'last_30_days',
        private ?string $category = null
    ) {}

    public function query()
    {
        $builder = app(MetricQueryBuilder::class);

        $builder->from('fact_sales')
            ->select([
                'dim_date.full_date',
                'dim_product.sku',
                'dim_product.name as product_name',
                'dim_product.category',
                'dim_customer.name as customer_name',
                'fact_sales.quantity',
                'fact_sales.unit_price',
                'fact_sales.discount_amount',
                'fact_sales.total_amount',
            ])
            ->period($this->period);

        if ($this->category) {
            $builder->where('dim_product.category', '=', $this->category);
        }

        $builder->orderBy('fact_sales.created_at', 'desc');

        return $builder;
    }

    public function headings(): array
    {
        return [
            'Date',
            'SKU',
            'Product',
            'Category',
            'Customer',
            'Quantity',
            'Unit Price',
            'Discount',
            'Total',
        ];
    }

    public function map($row): array
    {
        return [
            $row['full_date'],
            $row['sku'],
            $row['product_name'],
            $row['category'],
            $row['customer_name'],
            (int) $row['quantity'],
            (float) $row['unit_price'],
            (float) $row['discount_amount'],
            (float) $row['total_amount'],
        ];
    }

    public function styles(Worksheet $sheet): array
    {
        return [
            1 => [
                'font' => ['bold' => true, 'color' => ['rgb' => 'FFFFFF'], 'size' => 12],
                'fill' => ['fillType' => 'solid', 'startColor' => ['rgb' => '3B82F6']],
            ],
            'A1:I1' => [
                'alignment' => ['horizontal' => 'center'],
            ],
            'A' => ['numberFormat' => ['formatCode' => 'YYYY-MM-DD']],
            'G:I' => ['numberFormat' => ['formatCode' => NumberFormat::FORMAT_NUMBER_COMMA_SEPARATED1]],
        ];
    }

    public function title(): string
    {
        return 'Sales Report';
    }
}
```

### PDF Report Export
```php
<?php
namespace App\Exports\Pdf;

use Barryvdh\DomPDF\Facade\Pdf;
use App\Services\Analytics\ReportGenerator;

class PdfReportGenerator
{
    public function __construct(
        private ReportGenerator $reportGenerator
    ) {}

    public function generate(string $reportName, array $params = []): mixed
    {
        $data = $this->reportGenerator->generate($reportName, $params);

        $pdf = Pdf::loadView('exports.pdf.report', [
            'report' => $data,
            'generated_at' => now(),
            'params' => $params,
        ]);

        $pdf->setPaper('A4', 'landscape');
        $pdf->setOptions([
            'defaultFont' => 'sans-serif',
            'isHtml5ParserEnabled' => true,
            'isRemoteEnabled' => true,
        ]);

        return $pdf;
    }

    public function download(string $reportName, array $params = []): mixed
    {
        $pdf = $this->generate($reportName, $params);
        return $pdf->download("{$reportName}-" . now()->format('Y-m-d') . '.pdf');
    }

    public function stream(string $reportName, array $params = []): mixed
    {
        $pdf = $this->generate($reportName, $params);
        return $pdf->stream("{$reportName}-" . now()->format('Y-m-d') . '.pdf');
    }
}
```

### Export Controller
```php
<?php
namespace App\Http\Controllers\Analytics;

use App\Exports\SalesReportExport;
use App\Exports\ComprehensiveReportExport;
use App\Exports\Pdf\PdfReportGenerator;
use Illuminate\Http\Request;
use Maatwebsite\Excel\Facades\Excel;

class ExportController extends Controller
{
    public function __construct(
        private PdfReportGenerator $pdfGenerator
    ) {}

    public function exportExcel(string $reportName, Request $request): mixed
    {
        $format = $request->input('format', 'xlsx');

        $export = match ($reportName) {
            'sales' => new SalesReportExport(
                $request->input('period', 'last_30_days'),
                $request->input('category')
            ),
            'comprehensive' => new ComprehensiveReportExport(
                $request->input('period', 'last_30_days')
            ),
            default => throw new \InvalidArgumentException("Unknown report: {$reportName}"),
        };

        $filename = "{$reportName}-" . now()->format('Y-m-d') . ".{$format}";

        return Excel::download($export, $filename);
    }

    public function exportPdf(string $reportName, Request $request): mixed
    {
        return $this->pdfGenerator->download($reportName, $request->all());
    }

    public function exportCsv(string $reportName, Request $request): mixed
    {
        $export = new SalesReportExport(
            $request->input('period', 'last_30_days')
        );

        return Excel::download($export, "{$reportName}-" . now()->format('Y-m-d') . '.csv', \Maatwebsite\Excel\Excel::CSV);
    }
}
```

### Queue-based Export Job
```php
<?php
namespace App\Jobs;

use App\Exports\SalesReportExport;
use App\Models\ExportHistory;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;
use Illuminate\Support\Facades\Storage;
use Maatwebsite\Excel\Facades\Excel;

class GenerateReportJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $timeout = 600;
    public int $tries = 1;

    public function __construct(
        private int $exportHistoryId,
        private string $reportName,
        private array $params,
        private string $format
    ) {}

    public function handle(): void
    {
        $exportHistory = ExportHistory::findOrFail($this->exportHistoryId);
        $exportHistory->update(['status' => 'processing']);

        try {
            $export = new SalesReportExport(
                $this->params['period'] ?? 'last_30_days',
                $this->params['category'] ?? null
            );

            $filename = "exports/{$this->reportName}-" . now()->format('Y-m-d-His') . ".{$this->format}";

            Excel::store($export, $filename, 'public');

            $exportHistory->update([
                'status' => 'completed',
                'file_path' => $filename,
                'file_size' => Storage::disk('public')->size($filename),
                'completed_at' => now(),
            ]);
        } catch (\Exception $e) {
            $exportHistory->update([
                'status' => 'failed',
                'error_message' => $e->getMessage(),
            ]);

            throw $e;
        }
    }
}
```

### Export History Model
```php
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class ExportHistory extends Model
{
    protected $fillable = [
        'user_id',
        'report_name',
        'format',
        'params',
        'status',
        'file_path',
        'file_size',
        'error_message',
        'completed_at',
    ];

    protected $casts = [
        'params' => 'array',
        'completed_at' => 'datetime',
    ];

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public function scopePending($query)
    {
        return $query->where('status', 'pending');
    }

    public function scopeToday($query)
    {
        return $query->whereDate('created_at', today());
    }

    public function downloadUrl(): ?string
    {
        if ($this->file_path && $this->status === 'completed') {
            return Storage::disk('public')->url($this->file_path);
        }
        return null;
    }
}
```

### Export Request Validation
```php
<?php
namespace App\Http\Requests;

use Illuminate\Foundation\Http\FormRequest;

class ExportRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true;
    }

    public function rules(): array
    {
        return [
            'format' => 'sometimes|in:xlsx,csv,pdf',
            'period' => 'sometimes|in:today,yesterday,last_7_days,last_30_days,this_month,last_month,this_year',
            'filters' => 'sometimes|array',
            'filters.*.column' => 'required_with:filters|string',
            'filters.*.operator' => 'required_with:filters|in:=,!=,>,<,>=,<=,like',
            'filters.*.value' => 'required_with:filters',
        ];
    }

    public function messages(): array
    {
        return [
            'format.in' => 'Format must be one of: xlsx, csv, pdf',
            'period.in' => 'Invalid period selection',
        ];
    }
}
```
