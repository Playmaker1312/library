```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Cache;

class ForecastService
{
    public function forecastMovingAverage(string $metric, string $table, string $dateColumn, string $valueColumn, int $periods = 30): array
    {
        $historicalData = DB::connection('warehouse')
            ->table($table)
            ->select([$dateColumn, $valueColumn])
            ->orderBy($dateColumn, 'desc')
            ->limit(365)
            ->get()
            ->reverse()
            ->values();

        $values = $historicalData->pluck($valueColumn)->toArray();
        $dates = $historicalData->pluck($dateColumn)->toArray();

        // Calculate moving averages
        $ma7 = $this->movingAverage($values, 7);
        $ma30 = $this->movingAverage($values, 30);

        // Simple exponential smoothing forecast
        $alpha = 0.3;
        $forecast = $this->exponentialSmoothing($values, $alpha, $periods);

        // Generate future dates
        $lastDate = \Carbon\Carbon::parse(end($dates));
        $futureDates = [];
        for ($i = 1; $i <= $periods; $i++) {
            $futureDates[] = $lastDate->copy()->addDays($i)->format('Y-m-d');
        }

        // Calculate prediction intervals
        $residuals = $this->calculateResiduals($values, $forecast['fitted']);
        $stdError = sqrt(array_sum(array_map(fn ($r) => $r * $r, $residuals)) / max(count($residuals), 1));

        return [
            'metric' => $metric,
            'historical' => [
                'dates' => $dates,
                'values' => $values,
                'moving_average_7' => $ma7,
                'moving_average_30' => $ma30,
            ],
            'forecast' => [
                'dates' => $futureDates,
                'values' => $forecast['predicted'],
                'upper_bound' => array_map(fn ($v) => $v + 2 * $stdError, $forecast['predicted']),
                'lower_bound' => array_map(fn ($v) => max(0, $v - 2 * $stdError), $forecast['predicted']),
            ],
            'statistics' => [
                'mean' => array_sum($values) / count($values),
                'std_dev' => $stdError,
                'trend' => $this->calculateTrend($values),
                'seasonal_strength' => $this->calculateSeasonalStrength($values),
            ],
        ];
    }

    public function forecastLinearRegression(string $metric, string $table, string $dateColumn, string $valueColumn, int $periods = 30): array
    {
        $data = DB::connection('warehouse')
            ->table($table)
            ->select([$dateColumn, $valueColumn])
            ->orderBy($dateColumn)
            ->limit(365)
            ->get();

        $n = count($data);
        $x = range(1, $n);
        $y = array_map(fn ($d) => (float) $d->$valueColumn, iterator_to_array($data));

        // Linear regression
        $xMean = array_sum($x) / $n;
        $yMean = array_sum($y) / $n;

        $numerator = 0;
        $denominator = 0;
        for ($i = 0; $i < $n; $i++) {
            $numerator += ($x[$i] - $xMean) * ($y[$i] - $yMean);
            $denominator += ($x[$i] - $xMean) ** 2;
        }

        $slope = $denominator > 0 ? $numerator / $denominator : 0;
        $intercept = $yMean - $slope * $xMean;

        // Predict future
        $lastDate = \Carbon\Carbon::parse(end($data)->$dateColumn ?? now());
        $futureDates = [];
        $futureValues = [];

        for ($i = 1; $i <= $periods; $i++) {
            $futureDates[] = $lastDate->copy()->addDays($i)->format('Y-m-d');
            $futureValues[] = $slope * ($n + $i) + $intercept;
        }

        return [
            'metric' => $metric,
            'model' => [
                'type' => 'linear_regression',
                'slope' => $slope,
                'intercept' => $intercept,
                'r_squared' => $this->calculateRSquared($x, $y, $slope, $intercept),
            ],
            'historical' => [
                'dates' => array_map(fn ($d) => $d->$dateColumn, iterator_to_array($data)),
                'values' => $y,
            ],
            'forecast' => [
                'dates' => $futureDates,
                'values' => $futureValues,
            ],
        ];
    }

    public function detectTrendChange(string $metric, string $table, string $dateColumn, string $valueColumn): array
    {
        $data = DB::connection('warehouse')
            ->table($table)
            ->select([$dateColumn, $valueColumn])
            ->orderBy($dateColumn)
            ->limit(90)
            ->get();

        $values = array_map(fn ($d) => (float) $d->$valueColumn, iterator_to_array($data));
        $n = count($values);

        if ($n < 14) return ['trend' => 'insufficient_data'];

        // Compare first half vs second half
        $midpoint = (int) ($n / 2);
        $firstHalf = array_slice($values, 0, $midpoint);
        $secondHalf = array_slice($values, $midpoint);

        $firstMean = array_sum($firstHalf) / count($firstHalf);
        $secondMean = array_sum($secondHalf) / count($secondHalf);

        $changePct = $firstMean > 0 ? (($secondMean - $firstMean) / $firstMean) * 100 : 0;

        return [
            'metric' => $metric,
            'first_half_mean' => $firstMean,
            'second_half_mean' => $secondMean,
            'change_percentage' => round($changePct, 2),
            'trend' => match (true) {
                $changePct > 10 => 'strong_upward',
                $changePct > 3 => 'slight_upward',
                $changePct < -10 => 'strong_downward',
                $changePct < -3 => 'slight_downward',
                default => 'stable',
            },
        ];
    }

    private function movingAverage(array $values, int $window): array
    {
        $result = [];
        $sum = 0;

        for ($i = 0; $i < count($values); $i++) {
            $sum += $values[$i];
            if ($i >= $window) {
                $sum -= $values[$i - $window];
            }
            $result[] = $i >= $window - 1 ? $sum / min($window, $i + 1) : null;
        }

        return $result;
    }

    private function exponentialSmoothing(array $values, float $alpha, int $periods): array
    {
        $fitted = [];
        $smoothed = $values[0] ?? 0;

        foreach ($values as $v) {
            $smoothed = $alpha * $v + (1 - $alpha) * $smoothed;
            $fitted[] = $smoothed;
        }

        $predicted = [];
        for ($i = 0; $i < $periods; $i++) {
            $predicted[] = $smoothed;
        }

        return ['fitted' => $fitted, 'predicted' => $predicted];
    }

    private function calculateResiduals(array $actual, array $fitted): array
    {
        $residuals = [];
        for ($i = 0; $i < count($actual); $i++) {
            $residuals[] = $actual[$i] - ($fitted[$i] ?? $actual[$i]);
        }
        return $residuals;
    }

    private function calculateTrend(array $values): string
    {
        $n = count($values);
        if ($n < 2) return 'unknown';

        $recent = array_slice($values, -7);
        $older = array_slice($values, max(0, $n - 14), 7);

        if (empty($older)) return 'unknown';

        $recentAvg = array_sum($recent) / count($recent);
        $olderAvg = array_sum($older) / count($older);

        $diff = $olderAvg > 0 ? ($recentAvg - $olderAvg) / $olderAvg * 100 : 0;

        return match (true) {
            $diff > 5 => 'increasing',
            $diff < -5 => 'decreasing',
            default => 'stable',
        };
    }

    private function calculateSeasonalStrength(array $values): float
    {
        if (count($values) < 14) return 0;

        // Compare variance within weeks vs overall variance
        $weeks = array_chunk($values, 7);
        $weekMeans = array_map(fn ($w) => array_sum($w) / count($w), $weeks);
        $overallMean = array_sum($values) / count($values);

        $betweenWeekVar = 0;
        foreach ($weekMeans as $wm) {
            $betweenWeekVar += ($wm - $overallMean) ** 2;
        }
        $betweenWeekVar /= count($weekMeans);

        return $betweenWeekVar;
    }

    private function calculateRSquared(array $x, array $y, float $slope, float $intercept): float
    {
        $n = count($y);
        $yMean = array_sum($y) / $n;

        $ssRes = 0;
        $ssTot = 0;

        for ($i = 0; $i < $n; $i++) {
            $yPred = $slope * $x[$i] + $intercept;
            $ssRes += ($y[$i] - $yPred) ** 2;
            $ssTot += ($y[$i] - $yMean) ** 2;
        }

        return $ssTot > 0 ? 1 - ($ssRes / $ssTot) : 0;
    }
}
```

### Anomaly Detection Service
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\DB;

class AnomalyDetectionService
{
    public function detectZScoreAnomalies(string $metric, string $table, string $valueColumn, float $threshold = 3.0): array
    {
        $data = DB::connection('warehouse')
            ->table($table)
            ->select(['*'])
            ->orderBy('created_at', 'desc')
            ->limit(100)
            ->get()
            ->reverse();

        $values = array_map(fn ($d) => (float) $d->$valueColumn, iterator_to_array($data));
        $mean = array_sum($values) / count($values);
        $stdDev = $this->standardDeviation($values, $mean);

        $anomalies = [];
        foreach ($data as $i => $row) {
            $zScore = $stdDev > 0 ? abs(($row->$valueColumn - $mean) / $stdDev) : 0;
            if ($zScore > $threshold) {
                $anomalies[] = [
                    'date' => $row->created_at,
                    'value' => (float) $row->$valueColumn,
                    'expected' => $mean,
                    'z_score' => round($zScore, 2),
                    'deviation' => round((float) $row->$valueColumn - $mean, 2),
                    'severity' => $this->getSeverity($zScore),
                ];
            }
        }

        return [
            'metric' => $metric,
            'method' => 'z-score',
            'threshold' => $threshold,
            'mean' => $mean,
            'std_dev' => $stdDev,
            'anomalies' => $anomalies,
            'anomaly_count' => count($anomalies),
            'total_points' => count($values),
        ];
    }

    public function detectIQRAnomalies(string $metric, string $table, string $valueColumn, float $multiplier = 1.5): array
    {
        $data = DB::connection('warehouse')
            ->table($table)
            ->select(['*'])
            ->orderBy($valueColumn)
            ->get();

        $values = array_map(fn ($d) => (float) $d->$valueColumn, iterator_to_array($data));
        $n = count($values);

        if ($n < 4) return ['metric' => $metric, 'error' => 'Insufficient data'];

        // Calculate quartiles
        $q1 = $this->percentile($values, 0.25);
        $q3 = $this->percentile($values, 0.75);
        $iqr = $q3 - $q1;

        $lowerBound = $q1 - $multiplier * $iqr;
        $upperBound = $q3 + $multiplier * $iqr;

        $anomalies = [];
        foreach ($data as $row) {
            $value = (float) $row->$valueColumn;
            if ($value < $lowerBound || $value > $upperBound) {
                $anomalies[] = [
                    'date' => $row->created_at ?? $row->date,
                    'value' => $value,
                    'lower_bound' => $lowerBound,
                    'upper_bound' => $upperBound,
                    'type' => $value < $lowerBound ? 'low' : 'high',
                ];
            }
        }

        return [
            'metric' => $metric,
            'method' => 'iqr',
            'q1' => $q1,
            'q3' => $q3,
            'iqr' => $iqr,
            'lower_bound' => $lowerBound,
            'upper_bound' => $upperBound,
            'anomalies' => $anomalies,
            'anomaly_count' => count($anomalies),
        ];
    }

    public function detectMovingAverageAnomaly(string $metric, string $table, string $valueColumn, int $window = 7, float $threshold = 2.0): array
    {
        $data = DB::connection('warehouse')
            ->table($table)
            ->select(['*'])
            ->orderBy('created_at', 'desc')
            ->limit(365)
            ->get()
            ->reverse();

        $values = array_map(fn ($d) => (float) $d->$valueColumn, iterator_to_array($data));

        // Calculate rolling mean and std
        $anomalies = [];
        for ($i = $window; $i < count($values); $i++) {
            $windowValues = array_slice($values, $i - $window, $window);
            $mean = array_sum($windowValues) / $window;
            $std = $this->standardDeviation($windowValues, $mean);

            $zScore = $std > 0 ? abs(($values[$i] - $mean) / $std) : 0;

            if ($zScore > $threshold) {
                $anomalies[] = [
                    'date' => $data[$i]->created_at ?? $data[$i]->date,
                    'value' => $values[$i],
                    'expected' => $mean,
                    'z_score' => round($zScore, 2),
                    'severity' => $this->getSeverity($zScore),
                ];
            }
        }

        return [
            'metric' => $metric,
            'method' => 'moving_average',
            'window' => $window,
            'threshold' => $threshold,
            'anomalies' => $anomalies,
            'anomaly_count' => count($anomalies),
        ];
    }

    public function detectSeasonalAnomaly(string $metric, string $table, string $dateColumn, string $valueColumn): array
    {
        // Get daily data for the past year
        $dailyData = DB::connection('warehouse')
            ->table($table)
            ->select([$dateColumn, $valueColumn])
            ->orderBy($dateColumn)
            ->where($dateColumn, '>=', now()->subYear())
            ->get();

        if ($dailyData->isEmpty()) return ['metric' => $metric, 'error' => 'No data'];

        // Calculate day-of-week averages
        $dayOfWeekAverages = [];
        foreach ($dailyData as $row) {
            $dow = \Carbon\Carbon::parse($row->$dateColumn)->dayOfWeek;
            if (!isset($dayOfWeekAverages[$dow])) {
                $dayOfWeekAverages[$dow] = ['sum' => 0, 'count' => 0];
            }
            $dayOfWeekAverages[$dow]['sum'] += (float) $row->$valueColumn;
            $dayOfWeekAverages[$dow]['count']++;
        }

        foreach ($dayOfWeekAverages as &$avg) {
            $avg = $avg['count'] > 0 ? $avg['sum'] / $avg['count'] : 0;
        }

        // Detect anomalies by comparing to expected day-of-week value
        $anomalies = [];
        $today = \Carbon\Carbon::today();
        $dow = $today->dayOfWeek;
        $expectedValue = $dayOfWeekAverages[$dow] ?? 0;

        $todayValue = $dailyData->last()?->$valueColumn ?? 0;

        if ($expectedValue > 0) {
            $ratio = $todayValue / $expectedValue;
            if ($ratio > 1.5 || $ratio < 0.5) {
                $anomalies[] = [
                    'date' => $today->toDateString(),
                    'value' => $todayValue,
                    'expected' => $expectedValue,
                    'ratio' => round($ratio, 2),
                    'day_of_week' => $today->format('l'),
                ];
            }
        }

        return [
            'metric' => $metric,
            'method' => 'seasonal',
            'day_of_week_averages' => $dayOfWeekAverages,
            'anomalies' => $anomalies,
            'anomaly_count' => count($anomalies),
        ];
    }

    private function standardDeviation(array $values, ?float $mean = null): float
    {
        $mean = $mean ?? (array_sum($values) / count($values));
        $variance = array_sum(array_map(fn ($v) => ($v - $mean) ** 2, $values)) / count($values);
        return sqrt($variance);
    }

    private function percentile(array $sorted, float $percentile): float
    {
        $n = count($sorted);
        $index = $percentile * ($n - 1);
        $lower = floor($index);
        $upper = ceil($index);

        if ($lower == $upper) return $sorted[(int) $index];

        return $sorted[(int) $lower] + ($index - $lower) * ($sorted[(int) $upper] - $sorted[(int) $lower]);
    }

    private function getSeverity(float $zScore): string
    {
        return match (true) {
            $zScore > 5 => 'critical',
            $zScore > 4 => 'high',
            $zScore > 3 => 'medium',
            default => 'low',
        };
    }
}
```

### Forecast Command
```php
<?php
namespace App\Console\Commands;

use App\Services\Analytics\ForecastService;
use App\Services\Analytics\AnomalyDetectionService;
use Illuminate\Console\Command;

class ForecastMetrics extends Command
{
    protected $signature = 'analytics:forecast {--metric=}';
    protected $description = 'Generate forecasts for business metrics';

    public function handle(ForecastService $forecast, AnomalyDetectionService $anomaly): int
    {
        $metrics = $this->option('metric')
            ? [$this->option('metric')]
            : ['revenue', 'orders', 'users'];

        foreach ($metrics as $metric) {
            $this->info("Forecasting: {$metric}");

            [$table, $valueColumn] = $this->getMetricConfig($metric);

            $result = $forecast->forecastMovingAverage($metric, $table, 'date_key', $valueColumn, 30);

            $this->line("  Trend: {$result['statistics']['trend']}");
            $this->line("  Mean: " . number_format($result['statistics']['mean'], 0));
            $this->line("  Forecast: " . number_format(end($result['forecast']['values']), 0));

            // Check for anomalies
            $anomalies = $anomaly->detectZScoreAnomalies($metric, $table, $valueColumn);
            if ($anomalies['anomaly_count'] > 0) {
                $this->warn("  Found {$anomalies['anomaly_count']} anomalies");
            }
        }

        return self::SUCCESS;
    }

    private function getMetricConfig(string $metric): array
    {
        return match ($metric) {
            'revenue' => ['agg_daily_sales', 'total_revenue'],
            'orders' => ['agg_daily_sales', 'order_count'],
            'users' => ['agg_daily_users', 'active_users'],
            default => throw new \InvalidArgumentException("Unknown metric: {$metric}"),
        };
    }
}
```

### Anomaly Alert Model
```php
<?php
namespace App\Models\Analytics;

use Illuminate\Database\Eloquent\Model;

class AnomalyAlert extends Model
{
    protected $fillable = [
        'metric',
        'method',
        'value',
        'expected_value',
        'z_score',
        'severity',
        'message',
        'detected_at',
        'acknowledged_at',
        'acknowledged_by',
        'resolved_at',
    ];

    protected $casts = [
        'detected_at' => 'datetime',
        'acknowledged_at' => 'datetime',
        'resolved_at' => 'datetime',
    ];

    public function scopeUnacknowledged($query)
    {
        return $query->whereNull('acknowledged_at');
    }

    public function scopeCritical($query)
    {
        return $query->where('severity', 'critical');
    }

    public function acknowledge(int $userId): void
    {
        $this->update([
            'acknowledged_at' => now(),
            'acknowledged_by' => $userId,
        ]);
    }
}
```
