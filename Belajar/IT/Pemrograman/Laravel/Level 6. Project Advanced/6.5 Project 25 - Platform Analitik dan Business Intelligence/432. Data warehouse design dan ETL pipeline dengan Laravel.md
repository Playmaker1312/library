# 432. Data Warehouse Design dan ETL Pipeline dengan Laravel

## Definisi & Target
Membangun data warehouse menggunakan star schema dan implementasi ETL pipeline di Laravel. Meliputi data extraction dari berbagai sumber, transformation & cleaning, dan loading ke data warehouse dengan incremental updates.

## Komponen yang Terlibat
- Star schema (fact & dimension tables)
- Slowly Changing Dimensions (SCD Type 1 & 2)
- Incremental ETL
- Data cleaning & validation
- Batch processing
- Data lineage
- Error handling & retry
- Data quality checks

### ETL Extractor
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Http;
use Illuminate\Support\Facades\Log;

class Extractor
{
    private array $connections;

    public function __construct()
    {
        $this->connections = [
            'source_oltp' => DB::connection('mysql'),
            'warehouse' => DB::connection('warehouse'),
        ];
    }

    public function extractOrders(array $options = []): array
    {
        $since = $options['since'] ?? now()->subDay();
        $until = $options['until'] ?? now();

        $lastSync = $this->getLastSync('orders');

        $orders = $this->connections['source_oltp']
            ->table('orders')
            ->where('created_at', '>', $lastSync)
            ->where('created_at', '<=', $until)
            ->where('status', 'completed')
            ->get();

        $this->setLastSync('orders', $until);

        Log::info("Extracted {$orders->count()} orders");

        return $orders->toArray();
    }

    public function extractProducts(array $options = []): array
    {
        return $this->connections['source_oltp']
            ->table('products')
            ->where('updated_at', '>', $this->getLastSync('products'))
            ->get()
            ->toArray();
    }

    public function extractUsers(array $options = []): array
    {
        return $this->connections['source_oltp']
            ->table('users')
            ->where('updated_at', '>', $this->getLastSync('users'))
            ->get()
            ->toArray();
    }

    public function extractEvents(array $options = []): array
    {
        $since = $options['since'] ?? now()->subHour();

        return $this->connections['source_oltp']
            ->table('event_tracking')
            ->where('created_at', '>', $since)
            ->orderBy('created_at')
            ->limit(5000)
            ->get()
            ->toArray();
    }

    private function getLastSync(string $table): string
    {
        return $this->connections['warehouse']
            ->table('etl_control')
            ->where('table_name', $table)
            ->value('last_sync_at') ?? '2000-01-01 00:00:00';
    }

    private function setLastSync(string $table, string $timestamp): void
    {
        $this->connections['warehouse']
            ->table('etl_control')
            ->updateOrInsert(
                ['table_name' => $table],
                ['last_sync_at' => $timestamp, 'updated_at' => now()]
            );
    }
}
```

### ETL Transformer
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\Log;

class Transformer
{
    public function transformOrders(array $orders): array
    {
        $records = [];

        foreach ($orders as $order) {
            try {
                $records[] = [
                    'date_key' => (int) $order->created_at->format('Ymd'),
                    'product_key' => $this->getProductKey($order->product_id),
                    'customer_key' => $this->getCustomerKey($order->user_id),
                    'store_key' => $order->store_id,
                    'quantity' => (int) $order->quantity,
                    'unit_price' => (float) $order->unit_price,
                    'discount_amount' => (float) ($order->discount ?? 0),
                    'total_amount' => (float) $order->total_amount,
                    'cost_amount' => (float) ($order->cost ?? 0),
                    'source_order_id' => $order->id,
                    'created_at' => $order->created_at,
                ];
            } catch (\Exception $e) {
                Log::warning("Failed to transform order {$order->id}", [
                    'error' => $e->getMessage(),
                ]);
                continue;
            }
        }

        return $records;
    }

    public function transformProducts(array $products): array
    {
        $records = [];

        foreach ($products as $product) {
            $records[] = [
                'product_id' => $product->id,
                'sku' => $product->sku,
                'name' => $product->name,
                'category' => $product->category ?? 'Uncategorized',
                'subcategory' => $product->subcategory ?? null,
                'brand' => $product->brand ?? null,
                'supplier' => $product->supplier ?? null,
                'unit_price' => (float) $product->price,
                'valid_from' => now()->toDateString(),
                'is_current' => true,
            ];
        }

        return $records;
    }

    public function cleanData(array $data, array $rules = []): array
    {
        $cleaned = [];

        foreach ($data as $row) {
            $valid = true;

            foreach ($rules as $field => $rule) {
                $value = $row[$field] ?? null;

                switch ($rule) {
                    case 'required':
                        if (empty($value) && $value !== 0) $valid = false;
                        break;
                    case 'positive':
                        if ($value < 0) $row[$field] = abs($value);
                        break;
                    case 'not_null':
                        if (is_null($value)) $valid = false;
                        break;
                }
            }

            if ($valid) {
                $cleaned[] = $row;
            }
        }

        return $cleaned;
    }

    private function getProductKey(int $productId): int
    {
        return DB::connection('warehouse')
            ->table('dim_product')
            ->where('product_id', $productId)
            ->where('is_current', true)
            ->value('product_key') ?? $this->createUnknownProduct($productId);
    }

    private function getCustomerKey(int $customerId): int
    {
        return DB::connection('warehouse')
            ->table('dim_customer')
            ->where('customer_id', $customerId)
            ->value('customer_key') ?? -1;
    }

    private function createUnknownProduct(int $productId): int
    {
        return DB::connection('warehouse')
            ->table('dim_product')
            ->insertGetId([
                'product_id' => $productId,
                'sku' => 'UNKNOWN',
                'name' => 'Unknown Product',
                'category' => 'Unknown',
                'unit_price' => 0,
                'valid_from' => now()->toDateString(),
                'is_current' => true,
            ]);
    }
}
```

### ETL Loader
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class Loader
{
    private int $batchSize = 500;

    public function __construct()
    {
        DB::connection('warehouse')->statement('SET session transaction isolation level read committed');
    }

    public function loadFactSales(array $records): array
    {
        $loaded = 0;
        $failed = 0;

        foreach (array_chunk($records, $this->batchSize) as $batch) {
            try {
                DB::connection('warehouse')->beginTransaction();

                foreach ($batch as $record) {
                    DB::connection('warehouse')
                        ->table('fact_sales')
                        ->updateOrInsert(
                            ['source_order_id' => $record['source_order_id']],
                            $record
                        );
                    $loaded++;
                }

                DB::connection('warehouse')->commit();
            } catch (\Exception $e) {
                DB::connection('warehouse')->rollBack();
                $failed += count($batch);
                Log::error("Failed to load sales batch", ['error' => $e->getMessage()]);
            }
        }

        return ['loaded' => $loaded, 'failed' => $failed];
    }

    public function loadDimProduct(array $records): array
    {
        $loaded = 0;

        foreach ($records as $record) {
            try {
                $existing = DB::connection('warehouse')
                    ->table('dim_product')
                    ->where('product_id', $record['product_id'])
                    ->where('is_current', true)
                    ->first();

                if ($existing && $this->hasChanged($existing, $record)) {
                    // SCD Type 2: expire old record
                    DB::connection('warehouse')
                        ->table('dim_product')
                        ->where('product_key', $existing->product_key)
                        ->update([
                            'valid_to' => now()->subDay()->toDateString(),
                            'is_current' => false,
                        ]);

                    // Insert new version
                    DB::connection('warehouse')
                        ->table('dim_product')
                        ->insert($record);

                    $loaded++;
                } elseif (!$existing) {
                    DB::connection('warehouse')
                        ->table('dim_product')
                        ->insert($record);
                    $loaded++;
                }
            } catch (\Exception $e) {
                Log::error("Failed to load product", [
                    'product_id' => $record['product_id'],
                    'error' => $e->getMessage(),
                ]);
            }
        }

        return ['loaded' => $loaded, 'failed' => 0];
    }

    private function hasChanged(object $existing, array $new): bool
    {
        $compareFields = ['name', 'category', 'unit_price', 'brand', 'supplier'];
        foreach ($compareFields as $field) {
            if (($existing->$field ?? null) !== ($new[$field] ?? null)) {
                return true;
            }
        }
        return false;
    }
}
```

### ETL Orchestrator Command
```php
<?php
namespace App\Console\Commands;

use App\Services\Analytics\Extractor;
use App\Services\Analytics\Transformer;
use App\Services\Analytics\Loader;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\Log;

class EtlRun extends Command
{
    protected $signature = 'etl:run {--full : Run full load instead of incremental}';
    protected $description = 'Run ETL pipeline';

    public function handle(Extractor $extractor, Transformer $transformer, Loader $loader): int
    {
        $this->info('Starting ETL pipeline...');
        $startTime = microtime(true);

        // 1. Extract
        $this->info('Extracting data...');
        $orders = $extractor->extractOrders();
        $products = $extractor->extractProducts();

        // 2. Transform
        $this->info('Transforming data...');
        $transformedOrders = $transformer->transformOrders($orders);
        $transformedProducts = $transformer->transformProducts($products);

        // 3. Clean
        $cleanOrders = $transformer->cleanData($transformedOrders, [
            'total_amount' => 'required',
            'quantity' => 'positive',
        ]);

        // 4. Load
        $this->info('Loading data...');
        $orderResult = $loader->loadFactSales($cleanOrders);
        $productResult = $loader->loadDimProduct($transformedProducts);

        // 5. Update date dimension if needed
        $this->updateDateDimension();

        $duration = round((microtime(true) - $startTime) * 1000);

        $this->table(['Metric', 'Value'], [
            ['Orders loaded', $orderResult['loaded']],
            ['Orders failed', $orderResult['failed']],
            ['Products loaded', $productResult['loaded']],
            ['Duration (ms)', $duration],
        ]);

        Log::info('ETL completed', [
            'orders' => $orderResult,
            'products' => $productResult,
            'duration_ms' => $duration,
        ]);

        return self::SUCCESS;
    }

    private function updateDateDimension(): void
    {
        $existingDates = DB::connection('warehouse')
            ->table('dim_date')
            ->pluck('full_date')
            ->map(fn ($d) => $d->toDateString());

        $start = now()->subYear();
        $end = now()->addYear();

        $period = $start->toPeriod($end, 1, 'day');

        foreach ($period as $date) {
            $dateStr = $date->toDateString();

            if (!$existingDates->contains($dateStr)) {
                DB::connection('warehouse')
                    ->table('dim_date')
                    ->insert([
                        'date_key' => (int) $date->format('Ymd'),
                        'full_date' => $date,
                        'day' => (int) $date->format('d'),
                        'month' => (int) $date->format('m'),
                        'quarter' => (int) ceil($date->month / 3),
                        'year' => (int) $date->format('Y'),
                        'day_name' => $date->format('l'),
                        'month_name' => $date->format('F'),
                        'is_weekend' => $date->isWeekend(),
                        'is_holiday' => false,
                        'weekday_weekend' => $date->isWeekend() ? 'Weekend' : 'Weekday',
                    ]);
            }
        }
    }
}
```

### Data Quality Checker
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class DataQualityChecker
{
    public function runAllChecks(): array
    {
        return [
            'orphan_facts' => $this->checkOrphanFacts(),
            'duplicate_records' => $this->checkDuplicates(),
            'referential_integrity' => $this->checkReferentialIntegrity(),
            'data_freshness' => $this->checkFreshness(),
            'null_values' => $this->checkNullValues(),
        ];
    }

    public function checkOrphanFacts(): array
    {
        $orphans = DB::connection('warehouse')
            ->table('fact_sales')
            ->whereNotExists(function ($query) {
                $query->select(DB::raw(1))
                    ->from('dim_product')
                    ->whereColumn('dim_product.product_key', 'fact_sales.product_key');
            })
            ->count();

        return [
            'status' => $orphans === 0 ? 'pass' : 'fail',
            'orphan_count' => $orphans,
        ];
    }

    public function checkDuplicates(): array
    {
        $duplicates = DB::connection('warehouse')
            ->table('fact_sales')
            ->select('source_order_id', DB::raw('COUNT(*) as cnt'))
            ->groupBy('source_order_id')
            ->having('cnt', '>', 1)
            ->count();

        return [
            'status' => $duplicates === 0 ? 'pass' : 'fail',
            'duplicate_count' => $duplicates,
        ];
    }

    public function checkReferentialIntegrity(): array
    {
        $missingProducts = DB::connection('warehouse')
            ->table('fact_sales')
            ->whereNotExists(function ($q) {
                $q->select(DB::raw(1))->from('dim_product')
                    ->whereColumn('dim_product.product_key', 'fact_sales.product_key');
            })
            ->count();

        return [
            'status' => $missingProducts === 0 ? 'pass' : 'warn',
            'missing_products' => $missingProducts,
        ];
    }

    public function checkFreshness(): array
    {
        $lastLoad = DB::connection('warehouse')
            ->table('etl_control')
            ->where('table_name', 'orders')
            ->value('last_sync_at');

        $hoursSinceLastLoad = $lastLoad ? now()->diffInHours($lastLoad) : 999;

        return [
            'status' => $hoursSinceLastLoad < 24 ? 'pass' : 'fail',
            'hours_since_last_load' => $hoursSinceLastLoad,
        ];
    }

    public function checkNullValues(): array
    {
        $nullTotalAmount = DB::connection('warehouse')
            ->table('fact_sales')
            ->whereNull('total_amount')
            ->count();

        return [
            'status' => $nullTotalAmount === 0 ? 'pass' : 'warn',
            'null_total_amount' => $nullTotalAmount,
        ];
    }
}
```
