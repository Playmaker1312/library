- Metadata management
- Data quality monitoring
- Audit trail
- Data catalog
- Compliance reporting
- Data retention policies
- Data anonymization

### Data Quality Framework
```php
<?php
namespace App\Services\Governance;

use App\Models\Governance\DataQualityCheck;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Log;

class DataQualityService
{
    public function runAllChecks(): array
    {
        $results = [];

        $checks = DataQualityCheck::where('is_active', true)->get();

        foreach ($checks as $check) {
            try {
                $result = $this->executeCheck($check);
                $check->results()->create([
                    'status' => $result['status'],
                    'score' => $result['score'],
                    'details' => $result['details'],
                    'rows_checked' => $result['rows_checked'],
                    'rows_failed' => $result['rows_failed'],
                    'executed_at' => now(),
                ]);

                $results[] = $result;
            } catch (\Exception $e) {
                Log::error("Data quality check failed: {$check->name}", [
                    'error' => $e->getMessage(),
                ]);
                $results[] = [
                    'check' => $check->name,
                    'status' => 'error',
                    'error' => $e->getMessage(),
                ];
            }
        }

        return $results;
    }

    public function executeCheck(DataQualityCheck $check): array
    {
        $query = DB::connection($check->connection)->table($check->table);

        return match ($check->check_type) {
            'completeness' => $this->checkCompleteness($query, $check),
            'uniqueness' => $this->checkUniqueness($query, $check),
            'freshness' => $this->checkFreshness($query, $check),
            'consistency' => $this->checkConsistency($query, $check),
            'accuracy' => $this->checkAccuracy($query, $check),
            'custom_sql' => $this->checkCustomSQL($check),
            default => throw new \InvalidArgumentException("Unknown check type: {$check->check_type}"),
        };
    }

    public function checkCompleteness($query, DataQualityCheck $check): array
    {
        $total = (clone $query)->count();
        $nulls = (clone $query)->whereNull($check->column)->count();

        $score = $total > 0 ? (($total - $nulls) / $total) * 100 : 100;

        return [
            'check' => $check->name,
            'type' => 'completeness',
            'status' => $score >= ($check->threshold ?? 95) ? 'pass' : 'fail',
            'score' => round($score, 2),
            'rows_checked' => $total,
            'rows_failed' => $nulls,
            'details' => [
                'column' => $check->column,
                'total_rows' => $total,
                'null_count' => $nulls,
                'completeness_pct' => round($score, 2),
            ],
        ];
    }

    public function checkUniqueness($query, DataQualityCheck $check): array
    {
        $columns = explode(',', $check->column);
        $total = (clone $query)->count();
        $unique = (clone $query)->select($columns)->distinct()->count();
        $duplicates = $total - $unique;

        $score = $total > 0 ? ($unique / $total) * 100 : 100;

        return [
            'check' => $check->name,
            'type' => 'uniqueness',
            'status' => $score >= ($check->threshold ?? 100) ? 'pass' : 'fail',
            'score' => round($score, 2),
            'rows_checked' => $total,
            'rows_failed' => $duplicates,
            'details' => [
                'columns' => $columns,
                'total_rows' => $total,
                'unique_rows' => $unique,
                'duplicate_count' => $duplicates,
            ],
        ];
    }

    public function checkFreshness($query, DataQualityCheck $check): array
    {
        $latest = (clone $query)->max($check->column);
        $hoursSinceUpdate = $latest ? now()->diffInHours($latest) : 999;

        $threshold = $check->threshold ?? 24;
        $status = $hoursSinceUpdate <= $threshold ? 'pass' : 'fail';

        return [
            'check' => $check->name,
            'type' => 'freshness',
            'status' => $status,
            'score' => $hoursSinceUpdate <= $threshold ? 100 : max(0, 100 - ($hoursSinceUpdate - $threshold) * 10),
            'rows_checked' => 1,
            'rows_failed' => $status === 'pass' ? 0 : 1,
            'details' => [
                'column' => $check->column,
                'latest_value' => $latest,
                'hours_since_update' => $hoursSinceUpdate,
                'threshold_hours' => $threshold,
            ],
        ];
    }

    public function checkConsistency($query, DataQualityCheck $check): array
    {
        $config = $check->config;
        $sourceQuery = DB::connection($config['source.connection'])->table($config['source.table']);
        $targetQuery = DB::connection($config['target.connection'] ?? $check->connection)->table($config['target.table'] ?? '');

        $sourceCount = (clone $sourceQuery)->count();
        $targetCount = (clone $targetQuery)->count();

        $match = $sourceCount === $targetCount;

        return [
            'check' => $check->name,
            'type' => 'consistency',
            'status' => $match ? 'pass' : 'fail',
            'score' => $match ? 100 : 0,
            'rows_checked' => $sourceCount + $targetCount,
            'rows_failed' => abs($sourceCount - $targetCount),
            'details' => [
                'source_count' => $sourceCount,
                'target_count' => $targetCount,
                'difference' => $sourceCount - $targetCount,
            ],
        ];
    }

    public function checkAccuracy($query, DataQualityCheck $check): array
    {
        $config = $check->config;
        $rule = $config['rule'] ?? null;

        $total = (clone $query)->count();
        $failed = 0;

        if ($rule === 'range') {
            $min = $config['min'] ?? null;
            $max = $config['max'] ?? null;

            if ($min !== null) {
                $failed += (clone $query)->where($check->column, '<', $min)->count();
            }
            if ($max !== null) {
                $failed += (clone $query)->where($check->column, '>', $max)->count();
            }
        }

        $score = $total > 0 ? (($total - $failed) / $total) * 100 : 100;

        return [
            'check' => $check->name,
            'type' => 'accuracy',
            'status' => $score >= ($check->threshold ?? 95) ? 'pass' : 'fail',
            'score' => round($score, 2),
            'rows_checked' => $total,
            'rows_failed' => $failed,
            'details' => [
                'rule' => $rule,
                'total' => $total,
                'out_of_range' => $failed,
            ],
        ];
    }

    public function checkCustomSQL(DataQualityCheck $check): array
    {
        $result = DB::connection($check->connection)
            ->selectOne($check->config['sql']);

        $failed = (int) ($result->failed ?? $result->count ?? 0);
        $total = (int) ($result->total ?? $failed);

        $score = $total > 0 ? (($total - $failed) / $total) * 100 : 100;

        return [
            'check' => $check->name,
            'type' => 'custom_sql',
            'status' => $score >= ($check->threshold ?? 100) ? 'pass' : 'fail',
            'score' => round($score, 2),
            'rows_checked' => $total,
            'rows_failed' => $failed,
            'details' => (array) $result,
        ];
    }

    public function getOverallScore(): array
    {
        $recentChecks = DataQualityCheck::whereHas('results', function ($q) {
            $q->whereDate('executed_at', today());
        })->with('results')->get();

        $scores = [];
        foreach ($recentChecks as $check) {
            $latestResult = $check->results()->latest()->first();
            if ($latestResult) {
                $scores[] = $latestResult->score;
            }
        }

        return [
            'overall_score' => !empty($scores) ? round(array_sum($scores) / count($scores), 1) : 0,
            'checks_run' => count($scores),
            'passing_checks' => collect($scores)->filter(fn ($s) => $s >= 95)->count(),
            'failing_checks' => collect($scores)->filter(fn ($s) => $s < 95)->count(),
        ];
    }
}
```

### Audit Trail Service
```php
<?php
namespace App\Services\Governance;

use App\Models\Governance\AuditTrail;
use Illuminate\Support\Facades\DB;

class AuditTrailService
{
    public function log(string $action, string $resourceType, mixed $resourceId, array $data = []): AuditTrail
    {
        return AuditTrail::create([
            'user_id' => auth()->id(),
            'service' => config('app.name'),
            'action' => $action,
            'resource_type' => $resourceType,
            'resource_id' => (string) $resourceId,
            'data_before' => $data['before'] ?? null,
            'data_after' => $data['after'] ?? null,
            'ip_address' => request()->ip(),
            'user_agent' => request()->userAgent(),
            'session_id' => session()->getId(),
            'request_id' => request()->header('X-Request-Id'),
        ]);
    }

    public function logDataAccess(string $resourceType, mixed $resourceId, string $purpose): AuditTrail
    {
        return $this->log('data_access', $resourceType, $resourceId, [
            'purpose' => $purpose,
            'query_params' => request()->query(),
        ]);
    }

    public function logReportExport(string $reportName, string $format, array $filters): AuditTrail
    {
        return $this->log('report_export', 'report', $reportName, [
            'format' => $format,
            'filters' => $filters,
            'exported_at' => now()->toISOString(),
        ]);
    }

    public function queryAuditTrail(array $filters): array
    {
        $query = AuditTrail::query();

        if ($userId = $filters['user_id'] ?? null) {
            $query->where('user_id', $userId);
        }

        if ($action = $filters['action'] ?? null) {
            $query->where('action', $action);
        }

        if ($resourceType = $filters['resource_type'] ?? null) {
            $query->where('resource_type', $resourceType);
        }

        if ($dateFrom = $filters['date_from'] ?? null) {
            $query->where('created_at', '>=', $dateFrom);
        }

        if ($dateTo = $filters['date_to'] ?? null) {
            $query->where('created_at', '<=', $dateTo);
        }

        return [
            'total' => $query->count(),
            'data' => $query->orderBy('created_at', 'desc')
                ->paginate($filters['per_page'] ?? 50),
        ];
    }

    public function getDailySummary(): array
    {
        return AuditTrail::select(
            DB::raw("TO_CHAR(created_at, 'YYYY-MM-DD') as date"),
            DB::raw('COUNT(*) as total_events'),
            DB::raw("COUNT(DISTINCT user_id) as unique_users"),
            DB::raw("COUNT(DISTINCT action) as unique_actions"),
        )
        ->where('created_at', '>=', now()->subDays(30))
        ->groupBy(DB::raw("TO_CHAR(created_at, 'YYYY-MM-DD')"))
        ->orderBy('date', 'desc')
        ->get()
        ->toArray();
    }
}
```

### Data Governance Dashboard Controller
```php
<?php
namespace App\Http\Controllers\Governance;

use App\Models\Governance\DataCatalog;
use App\Models\Governance\AuditTrail;
use App\Services\Governance\DataQualityService;
use App\Services\Governance\AuditTrailService;
use Illuminate\Http\Request;

class GovernanceController extends Controller
{
    public function __construct(
        private DataQualityService $qualityService,
        private AuditTrailService $auditService
    ) {}

    public function dashboard(): mixed
    {
        return view('governance.dashboard', [
            'catalogCount' => DataCatalog::count(),
            'piiFields' => DataCatalog::pii()->count(),
            'qualityScore' => $this->qualityService->getOverallScore(),
            'recentAudits' => AuditTrail::latest()->limit(10)->get(),
        ]);
    }

    public function catalog(): mixed
    {
        $catalog = DataCatalog::with('qualityChecks')
            ->paginate(request('per_page', 20));

        return response()->json($catalog);
    }

    public function catalogShow(DataCatalog $catalog): mixed
    {
        return response()->json([
            'catalog' => $catalog->load('lineage', 'qualityChecks'),
            'lineage_graph' => $this->buildLineageGraph($catalog),
        ]);
    }

    public function runQualityChecks(Request $request): mixed
    {
        $catalogId = $request->input('catalog_id');

        $results = $catalogId
            ? $this->qualityService->runAllChecks()
            : $this->qualityService->runAllChecks();

        return response()->json([
            'message' => 'Quality checks completed',
            'results' => $results,
            'overall' => $this->qualityService->getOverallScore(),
        ]);
    }

    public function auditLog(Request $request): mixed
    {
        $filters = $request->validate([
            'user_id' => 'nullable|integer',
            'action' => 'nullable|string',
            'resource_type' => 'nullable|string',
            'date_from' => 'nullable|date',
            'date_to' => 'nullable|date',
            'per_page' => 'nullable|integer|min:1|max:100',
        ]);

        return response()->json(
            $this->auditService->queryAuditTrail($filters)
        );
    }

    public function complianceReport(Request $request): mixed
    {
        $type = $request->input('type', 'monthly');

        return response()->json([
            'report_type' => $type,
            'generated_at' => now()->toISOString(),
            'data_catalog' => [
                'total_datasets' => DataCatalog::count(),
                'pii_datasets' => DataCatalog::pii()->count(),
                'by_classification' => DataCatalog::selectRaw('classification, COUNT(*) as count')
                    ->groupBy('classification')->pluck('count', 'classification'),
            ],
            'data_quality' => $this->qualityService->getOverallScore(),
            'audit_summary' => $this->auditService->getDailySummary(),
            'access_events_30d' => AuditTrail::where('created_at', '>=', now()->subDays(30))->count(),
            'data_export_events' => AuditTrail::where('action', 'report_export')
                ->where('created_at', '>=', now()->subDays(30))->count(),
        ]);
    }

    private function buildLineageGraph(DataCatalog $catalog): array
    {
        $nodes = [];
        $edges = [];

        $nodes[$catalog->id] = [
            'id' => $catalog->id,
            'label' => $catalog->name,
            'type' => $catalog->type,
        ];

        foreach ($catalog->lineage as $lineage) {
            $source = $lineage->source;
            if ($source) {
                $nodes[$source->id] = [
                    'id' => $source->id,
                    'label' => $source->name ?? $source->table,
                    'type' => $source->type ?? 'table',
                ];
                $edges[] = [
                    'from' => $source->id,
                    'to' => $catalog->id,
                    'transformation' => $lineage->transformation,
                ];
            }
        }

        return ['nodes' => array_values($nodes), 'edges' => $edges];
    }
}
```

### Data Retention Command
```php
<?php
namespace App\Console\Commands;

use App\Models\Governance\DataCatalog;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\DB;

class EnforceDataRetention extends Command
{
    protected $signature = 'governance:enforce-retention {--dry-run : Simulate without deleting}';
    protected $description = 'Enforce data retention policies';

    public function handle(): int
    {
        $catalogs = DataCatalog::whereNotNull('retention_days')->get();

        foreach ($catalogs as $catalog) {
            $cutoffDate = now()->subDays($catalog->retention_days);

            $query = DB::connection($catalog->source)
                ->table($catalog->schema . '.' . $catalog->name)
                ->where('created_at', '<', $cutoffDate);

            $count = (clone $query)->count();

            if ($this->option('dry-run')) {
                $this->warn("[DRY RUN] Would delete {$count} rows from {$catalog->name}");
            } else {
                $query->delete();
                $this->info("Deleted {$count} rows from {$catalog->name} (retention: {$catalog->retention_days} days)");
            }
        }

        return self::SUCCESS;
    }
}
```
