### RFM Analysis Service
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\DB;

class RfmAnalysisService
{
    public function calculateRfm(): array
    {
        $customers = DB::connection('order_service')
            ->table('orders')
            ->select([
                'customer_id',
                DB::raw('MAX(created_at) as last_order_date'),
                DB::raw('COUNT(*) as frequency'),
                DB::raw('SUM(total_amount) as monetary'),
            ])
            ->where('status', 'completed')
            ->groupBy('customer_id')
            ->get();

        $scores = [];

        foreach ($customers as $customer) {
            $recency = $customer->last_order_date ? now()->diffInDays($customer->last_order_date) : 999;
            $frequency = (int) $customer->frequency;
            $monetary = (float) $customer->monetary;

            $scores[] = [
                'customer_id' => $customer->customer_id,
                'recency' => $recency,
                'frequency' => $frequency,
                'monetary' => $monetary,
                'r_score' => $this->scoreRecency($recency),
                'f_score' => $this->scoreFrequency($frequency),
                'm_score' => $this->scoreMonetary($monetary),
                'rfm_score' => null, // calculated after normalization
            ];
        }

        // Normalize scores (1-5 scale)
        $this->normalizeScores($scores);

        // Assign segments
        foreach ($scores as &$customer) {
            $customer['segment'] = $this->assignSegment(
                $customer['r_score'],
                $customer['f_score'],
                $customer['m_score']
            );
            $customer['rfm_score'] = $customer['r_score'] + $customer['f_score'] + $customer['m_score'];
        }

        return $scores;
    }

    public function getSegmentBreakdown(): array
    {
        $rfmData = $this->calculateRfm();
        $segments = [];

        foreach ($rfmData as $customer) {
            $segment = $customer['segment'];
            if (!isset($segments[$segment])) {
                $segments[$segment] = [
                    'name' => $segment,
                    'count' => 0,
                    'total_revenue' => 0,
                    'avg_frequency' => 0,
                ];
            }
            $segments[$segment]['count']++;
            $segments[$segment]['total_revenue'] += $customer['monetary'];
            $segments[$segment]['avg_frequency'] += $customer['frequency'];
        }

        foreach ($segments as &$seg) {
            $seg['avg_revenue'] = $seg['count'] > 0 ? $seg['total_revenue'] / $seg['count'] : 0;
            $seg['avg_frequency'] = $seg['count'] > 0 ? $seg['avg_frequency'] / $seg['count'] : 0;
        }

        return $segments;
    }

    public function getSegmentCustomers(string $segment): array
    {
        $all = $this->calculateRfm();
        return array_values(array_filter($all, fn ($c) => $c['segment'] === $segment));
    }

    private function scoreRecency(int $days): int
    {
        return match (true) {
            $days <= 7 => 5,
            $days <= 30 => 4,
            $days <= 90 => 3,
            $days <= 180 => 2,
            $days <= 365 => 1,
            default => 0,
        };
    }

    private function scoreFrequency(int $count): int
    {
        return match (true) {
            $count >= 20 => 5,
            $count >= 10 => 4,
            $count >= 5 => 3,
            $count >= 2 => 2,
            $count >= 1 => 1,
            default => 0,
        };
    }

    private function scoreMonetary(float $amount): int
    {
        return match (true) {
            $amount >= 10000000 => 5,
            $amount >= 5000000 => 4,
            $amount >= 1000000 => 3,
            $amount >= 500000 => 2,
            $amount >= 100000 => 1,
            default => 0,
        };
    }

    private function normalizeScores(array &$scores): void
    {
        $maxR = max(array_column($scores, 'r_score')) ?: 1;
        $maxF = max(array_column($scores, 'f_score')) ?: 1;
        $maxM = max(array_column($scores, 'm_score')) ?: 1;

        foreach ($scores as &$customer) {
            $customer['r_score'] = (int) round(($customer['r_score'] / $maxR) * 5);
            $customer['f_score'] = (int) round(($customer['f_score'] / $maxF) * 5);
            $customer['m_score'] = (int) round(($customer['m_score'] / $maxM) * 5);
        }
    }

    private function assignSegment(int $r, int $f, int $m): string
    {
        $total = $r + $f + $m;

        return match (true) {
            $r >= 4 && $f >= 4 && $m >= 4 => 'Champions',
            $r >= 4 && $f >= 3 && $m >= 3 => 'Loyal Customers',
            $r >= 4 && $f <= 2 && $m <= 2 => 'Recent Customers',
            $r >= 3 && $f >= 3 && $m >= 3 => 'Active Customers',
            $r >= 3 && $f <= 2 => 'At Risk',
            $r <= 2 && $f >= 3 => 'Need Attention',
            $r <= 2 && $f <= 2 && $m >= 3 => 'High Value Lost',
            $r <= 2 && $f <= 1 && $m <= 1 => 'Lost',
            $total >= 12 => 'Potential Loyalists',
            $total >= 9 => 'Promising',
            $total >= 6 => 'New Customers',
            default => 'Inactive',
        };
    }
}
```

### Cohort Analysis Service
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\DB;

class CohortAnalysisService
{
    public function buildCohorts(string $cohortType = 'registration_month', int $periods = 12): array
    {
        $cohorts = [];

        // Get all customers with their first order date
        $customers = DB::connection('order_service')
            ->table('orders')
            ->select(
                'customer_id',
                DB::raw('MIN(created_at) as first_order_date')
            )
            ->groupBy('customer_id')
            ->get();

        $monthlyOrders = DB::connection('order_service')
            ->table('orders')
            ->select(
                'customer_id',
                DB::raw("TO_CHAR(created_at, 'YYYY-MM') as period"),
                DB::raw('COUNT(*) as order_count'),
                DB::raw('SUM(total_amount) as revenue')
            )
            ->where('created_at', '>=', now()->subMonths($periods))
            ->groupBy('customer_id', 'period')
            ->get()
            ->groupBy('customer_id');

        // Group customers by cohort (registration month)
        $cohortGroups = [];
        foreach ($customers as $customer) {
            $cohortKey = \Carbon\Carbon::parse($customer->first_order_date)->format('Y-m');
            $cohortGroups[$cohortKey][] = $customer->customer_id;
        }

        // Calculate retention for each cohort
        foreach ($cohortGroups as $cohortKey => $customerIds) {
            $cohortData = [
                'cohort' => $cohortKey,
                'customers' => count($customerIds),
                'periods' => [],
            ];

            $cohortStart = \Carbon\Carbon::createFromFormat('Y-m', $cohortKey)->startOfMonth();

            for ($period = 0; $period <= $periods; $period++) {
                $periodDate = $cohortStart->copy()->addMonths($period);
                $periodKey = $periodDate->format('Y-m');

                if ($periodDate->isFuture()) break;

                $activeCustomers = 0;
                $totalRevenue = 0;

                foreach ($customerIds as $cid) {
                    if (isset($monthlyOrders[$cid])) {
                        $ordersInPeriod = $monthlyOrders[$cid]->firstWhere('period', $periodKey);
                        if ($ordersInPeriod) {
                            $activeCustomers++;
                            $totalRevenue += (float) $ordersInPeriod->revenue;
                        }
                    }
                }

                $cohortData['periods'][] = [
                    'period' => $period,
                    'period_label' => $periodKey,
                    'active_customers' => $activeCustomers,
                    'retention_rate' => $cohortData['customers'] > 0
                        ? round(($activeCustomers / $cohortData['customers']) * 100, 1)
                        : 0,
                    'revenue' => $totalRevenue,
                ];
            }

            $cohorts[] = $cohortData;
        }

        // Sort cohorts by date
        usort($cohorts, fn ($a, $b) => $a['cohort'] <=> $b['cohort']);

        return $cohorts;
    }

    public function getRetentionMatrix(): array
    {
        $cohorts = $this->buildCohorts();

        $matrix = [
            'headers' => ['Cohort', 'Size'],
            'rows' => [],
        ];

        $maxPeriods = 0;
        foreach ($cohorts as $cohort) {
            $maxPeriods = max($maxPeriods, count($cohort['periods']));
        }

        for ($i = 1; $i <= $maxPeriods; $i++) {
            $matrix['headers'][] = "M{$i}";
        }

        foreach ($cohorts as $cohort) {
            $row = [
                'cohort' => $cohort['cohort'],
                'size' => $cohort['customers'],
            ];

            foreach ($cohort['periods'] as $period) {
                $row[] = $period['retention_rate'] . '%';
            }

            // Pad with empty cells
            while (count($row) - 2 < $maxPeriods) {
                $row[] = '-';
            }

            $matrix['rows'][] = $row;
        }

        return $matrix;
    }

    public function getAverageRetention(): array
    {
        $cohorts = $this->buildCohorts();

        $averages = [];

        foreach ($cohorts as $cohort) {
            foreach ($cohort['periods'] as $period) {
                $month = $period['period'];
                if (!isset($averages[$month])) {
                    $averages[$month] = ['total' => 0, 'count' => 0];
                }
                $averages[$month]['total'] += $period['retention_rate'];
                $averages[$month]['count']++;
            }
        }

        return collect($averages)->map(fn ($avg) => round($avg['total'] / $avg['count'], 1))->toArray();
    }
}
```

### Segment Assignment Command
```php
<?php
namespace App\Console\Commands;

use App\Services\Analytics\RfmAnalysisService;
use App\Models\CustomerSegment;
use Illuminate\Console\Command;

class AssignSegments extends Command
{
    protected $signature = 'analytics:assign-segments';
    protected $description = 'Calculate RFM and assign customer segments';

    public function handle(RfmAnalysisService $rfm): int
    {
        $this->info('Calculating RFM scores...');
        $customers = $rfm->calculateRfm();

        $bar = $this->output->createProgressBar(count($customers));
        $bar->start();

        foreach ($customers as $customerData) {
            CustomerSegment::updateOrCreate(
                ['customer_id' => $customerData['customer_id']],
                [
                    'segment' => $customerData['segment'],
                    'rfm_score' => $customerData['rfm_score'],
                    'r_score' => $customerData['r_score'],
                    'f_score' => $customerData['f_score'],
                    'm_score' => $customerData['m_score'],
                    'recency_days' => $customerData['recency'],
                    'frequency' => $customerData['frequency'],
                    'monetary' => $customerData['monetary'],
                    'calculated_at' => now(),
                ]
            );
            $bar->advance();
        }

        $bar->finish();
        $this->newLine();

        $breakdown = $rfm->getSegmentBreakdown();
        $this->table(
            ['Segment', 'Customers', 'Total Revenue', 'Avg Revenue', 'Avg Frequency'],
            collect($breakdown)->map(fn ($s, $name) => [
                $name,
                $s['count'],
                number_format($s['total_revenue'], 0),
                number_format($s['avg_revenue'], 0),
                round($s['avg_frequency'], 1),
            ])->values()->toArray()
        );

        return self::SUCCESS;
    }
}
```

### Cohort Analysis Controller
```php
<?php
namespace App\Http\Controllers\Analytics;

use App\Services\Analytics\CohortAnalysisService;
use App\Services\Analytics\RfmAnalysisService;
use Illuminate\Http\Request;

class CustomerAnalyticsController extends Controller
{
    public function __construct(
        private CohortAnalysisService $cohortService,
        private RfmAnalysisService $rfmService
    ) {}

    public function rfm(): mixed
    {
        return response()->json([
            'segment_breakdown' => $this->rfmService->getSegmentBreakdown(),
            'segments' => [
                'Champions' => $this->rfmService->getSegmentCustomers('Champions'),
                'Loyal Customers' => $this->rfmService->getSegmentCustomers('Loyal Customers'),
                'At Risk' => $this->rfmService->getSegmentCustomers('At Risk'),
                'Lost' => $this->rfmService->getSegmentCustomers('Lost'),
            ],
        ]);
    }

    public function cohorts(Request $request): mixed
    {
        $periods = (int) $request->input('periods', 12);

        return response()->json([
            'matrix' => $this->cohortService->getRetentionMatrix(),
            'average_retention' => $this->cohortService->getAverageRetention(),
            'raw_data' => $this->cohortService->buildCohorts('registration_month', $periods),
        ]);
    }

    public function segment(string $segment): mixed
    {
        return response()->json([
            'segment' => $segment,
            'customers' => $this->rfmService->getSegmentCustomers($segment),
        ]);
    }

    public function segmentStats(): mixed
    {
        $breakdown = $this->rfmService->getSegmentBreakdown();

        return response()->json([
            'labels' => array_keys($breakdown),
            'values' => array_column($breakdown, 'count'),
            'revenue' => array_column($breakdown, 'total_revenue'),
        ]);
    }
}
```

### Customer Segment Model
```php
<?php
namespace App\Models\Analytics;

use Illuminate\Database\Eloquent\Model;

class CustomerSegment extends Model
{
    protected $fillable = [
        'customer_id',
        'segment',
        'rfm_score',
        'r_score',
        'f_score',
        'm_score',
        'recency_days',
        'frequency',
        'monetary',
        'calculated_at',
    ];

    protected $casts = [
        'calculated_at' => 'datetime',
    ];

    public function scopeBySegment($query, string $segment)
    {
        return $query->where('segment', $segment);
    }

    public function scopeTop($query, int $limit = 100)
    {
        return $query->orderBy('rfm_score', 'desc')->limit($limit);
    }

    public static function getSegments(): array
    {
        return [
            'Champions',
            'Loyal Customers',
            'Potential Loyalists',
            'Recent Customers',
            'Active Customers',
            'Promising',
            'New Customers',
            'At Risk',
            'Need Attention',
            'High Value Lost',
            'Lost',
            'Inactive',
        ];
    }
}
```
