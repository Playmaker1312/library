# 433. Event Tracking dan Data Collection untuk User Behavior Analytics

## Definisi & Target
Mengimplementasikan sistem event tracking untuk mengumpulkan data user behavior. Meliputi client-side events (page views, clicks, form submissions) dan server-side events (API calls, business events). Targetnya adalah data collection yang reliable dan scalable.

## Komponen yang Terlibat
- Event tracking system
- Client-side JavaScript tracker
- Server-side event collection
- Event queuing & batching
- Session management
- User identification (anonymous & authenticated)
- Event schema validation
- Data enrichment

### JavaScript Tracker
```javascript
// resources/js/analytics/tracker.js
class AnalyticsTracker {
    constructor(options = {}) {
        this.endpoint = options.endpoint || '/api/events/track';
        this.sessionId = this.getSessionId();
        this.anonymousId = this.getAnonymousId();
        this.batchSize = options.batchSize || 10;
        this.flushInterval = options.flushInterval || 5000;
        this.queue = [];
        this.userId = null;

        this.initSession();
        this.startBatchTimer();
        this.attachAutoTracking();
    }

    initSession() {
        if (!sessionStorage.getItem('analytics_session_id')) {
            sessionStorage.setItem('analytics_session_id', this.generateId());
        }
        this.sessionId = sessionStorage.getItem('analytics_session_id');
    }

    getAnonymousId() {
        let id = localStorage.getItem('analytics_anonymous_id');
        if (!id) {
            id = this.generateId();
            localStorage.setItem('analytics_anonymous_id', id);
        }
        return id;
    }

    generateId() {
        return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, (c) => {
            const r = Math.random() * 16 | 0;
            return (c === 'x' ? r : (r & 0x3 | 0x8)).toString(16);
        });
    }

    identify(userId, traits = {}) {
        this.userId = userId;
        this.track('identify', { user_id: userId, ...traits });
    }

    track(eventType, properties = {}) {
        const event = {
            session_id: this.sessionId,
            anonymous_id: this.anonymousId,
            user_id: this.userId,
            event_type: eventType,
            event_properties: properties,
            page_url: window.location.href,
            page_title: document.title,
            referrer_url: document.referrer,
            user_agent: navigator.userAgent,
            timestamp: new Date().toISOString(),
            screen_width: window.screen.width,
            screen_height: window.screen.height,
        };

        this.queue.push(event);

        if (this.queue.length >= this.batchSize) {
            this.flush();
        }
    }

    page(name, properties = {}) {
        this.track('page_view', { page_name: name, ...properties });
    }

    click(element, properties = {}) {
        this.track('click', {
            element_id: element.id,
            element_class: element.className,
            element_text: element.textContent?.substring(0, 100),
            tag_name: element.tagName,
            ...properties,
        });
    }

    formSubmit(form, properties = {}) {
        this.track('form_submit', {
            form_id: form.id,
            form_name: form.name,
            form_action: form.action,
            ...properties,
        });
    }

    error(error, properties = {}) {
        this.track('error', {
            error_message: error.message,
            error_stack: error.stack?.substring(0, 500),
            ...properties,
        });
    }

    startBatchTimer() {
        setInterval(() => this.flush(), this.flushInterval);
    }

    flush() {
        if (this.queue.length === 0) return;

        const batch = [...this.queue];
        this.queue = [];

        this.sendBatch(batch);
    }

    sendBatch(events) {
        if (navigator.sendBeacon) {
            navigator.sendBeacon(this.endpoint, JSON.stringify({ events }));
        } else {
            fetch(this.endpoint, {
                method: 'POST',
                headers: { 'Content-Type': 'application/json' },
                body: JSON.stringify({ events }),
                keepalive: true,
            });
        }
    }

    attachAutoTracking() {
        document.addEventListener('click', (e) => {
            const target = e.target.closest('[data-track]');
            if (target) {
                const eventName = target.dataset.track;
                const properties = {};
                for (const [key, value] of Object.entries(target.dataset)) {
                    if (key.startsWith('trackProp')) {
                        properties[key.replace('trackProp', '').toLowerCase()] = value;
                    }
                }
                this.click(target, { event_name: eventName, ...properties });
            }
        });

        document.addEventListener('submit', (e) => {
            if (e.target.matches('[data-track-form]')) {
                this.formSubmit(e.target);
            }
        });

        window.addEventListener('beforeunload', () => {
            this.flush();
        });
    }
}

// Initialize
window.analytics = new AnalyticsTracker({
    endpoint: '/api/events/track',
    batchSize: 10,
    flushInterval: 5000,
});

// Auto-track page view
document.addEventListener('DOMContentLoaded', () => {
    window.analytics.page(document.title);
});
```

### Server-Side Event Tracking Controller
```php
<?php
namespace App\Http\Controllers\Analytics;

use App\Models\Analytics\EventTracking;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Validator;
use Illuminate\Support\Facades\DB;

class EventTrackingController extends Controller
{
    public function track(Request $request): mixed
    {
        $validator = Validator::make($request->all(), [
            'events' => 'required|array|max:50',
            'events.*.event_type' => 'required|string|max:100',
            'events.*.session_id' => 'required|string|max:100',
        ]);

        if ($validator->fails()) {
            return response()->json(['error' => 'Invalid events'], 422);
        }

        $events = $request->input('events');

        DB::beginTransaction();
        try {
            $batch = [];

            foreach ($events as $event) {
                $enriched = $this->enrichEvent($event, $request);
                $batch[] = $enriched;

                $this->updateSession($enriched);
            }

            EventTracking::insert($batch);
            DB::commit();

            return response()->json(['accepted' => count($batch)]);
        } catch (\Exception $e) {
            DB::rollBack();
            logger()->error('Failed to track events', ['error' => $e->getMessage()]);
            return response()->json(['error' => 'Failed to process events'], 500);
        }
    }

    public function trackServerSide(Request $request): mixed
    {
        $validated = $request->validate([
            'event_type' => 'required|string|max:100',
            'event_category' => 'nullable|string|max:100',
            'event_action' => 'nullable|string|max:100',
            'event_label' => 'nullable|string|max:255',
            'event_properties' => 'nullable|array',
        ]);

        $event = $this->enrichEvent([
            'event_type' => $validated['event_type'],
            'event_category' => $validated['event_category'] ?? null,
            'event_action' => $validated['event_action'] ?? null,
            'event_label' => $validated['event_label'] ?? null,
            'event_properties' => $validated['event_properties'] ?? [],
            'session_id' => $request->header('X-Session-Id', (string) str()->uuid()),
            'anonymous_id' => $request->header('X-Anonymous-Id'),
            'user_id' => $request->user()?->id,
        ], $request);

        EventTracking::create($enriched);

        return response()->json(['status' => 'tracked', 'event_id' => $event->id]);
    }

    private function enrichEvent(array $event, Request $request): array
    {
        return [
            'session_id' => $event['session_id'],
            'user_id' => $event['user_id'] ?? $request->user()?->id,
            'anonymous_id' => $event['anonymous_id'] ?? $request->header('X-Anonymous-Id'),
            'event_type' => $event['event_type'],
            'event_category' => $event['event_category'] ?? null,
            'event_action' => $event['event_action'] ?? null,
            'event_label' => $event['event_label'] ?? null,
            'event_properties' => json_encode($event['event_properties'] ?? []),
            'page_url' => $event['page_url'] ?? $request->fullUrl(),
            'page_title' => $event['page_title'] ?? null,
            'referrer_url' => $event['referrer_url'] ?? $request->header('Referer'),
            'device_type' => $this->detectDeviceType($request->userAgent()),
            'browser' => $this->detectBrowser($request->userAgent()),
            'os' => $this->detectOS($request->userAgent()),
            'ip_address' => $request->ip(),
            'user_agent' => $request->userAgent(),
            'created_at' => now(),
        ];
    }

    private function updateSession(array $event): void
    {
        $session = DB::table('user_sessions')
            ->where('session_id', $event['session_id'])
            ->first();

        if (!$session) {
            DB::table('user_sessions')->insert([
                'session_id' => $event['session_id'],
                'user_id' => $event['user_id'],
                'anonymous_id' => $event['anonymous_id'],
                'started_at' => now(),
                'last_activity_at' => now(),
                'event_count' => 1,
                'landing_page' => $event['page_url'],
                'referrer' => $event['referrer_url'],
                'created_at' => now(),
                'updated_at' => now(),
            ]);
        } else {
            DB::table('user_sessions')
                ->where('session_id', $event['session_id'])
                ->update([
                    'last_activity_at' => now(),
                    'event_count' => $session->event_count + 1,
                    'exit_page' => $event['page_url'],
                    'user_id' => $event['user_id'] ?? $session->user_id,
                    'updated_at' => now(),
                ]);
        }
    }

    private function detectDeviceType(?string $ua): string
    {
        if (!$ua) return 'unknown';
        if (preg_match('/mobile|android|iphone|ipad|ipod/i', $ua)) return 'mobile';
        if (preg_match('/tablet|ipad/i', $ua)) return 'tablet';
        return 'desktop';
    }

    private function detectBrowser(?string $ua): string
    {
        if (!$ua) return 'unknown';
        if (str_contains($ua, 'Chrome')) return 'Chrome';
        if (str_contains($ua, 'Firefox')) return 'Firefox';
        if (str_contains($ua, 'Safari')) return 'Safari';
        if (str_contains($ua, 'Edge')) return 'Edge';
        return 'other';
    }

    private function detectOS(?string $ua): string
    {
        if (!$ua) return 'unknown';
        if (str_contains($ua, 'Windows')) return 'Windows';
        if (str_contains($ua, 'Mac')) return 'macOS';
        if (str_contains($ua, 'Linux')) return 'Linux';
        if (str_contains($ua, 'Android')) return 'Android';
        if (str_contains($ua, 'iOS')) return 'iOS';
        return 'other';
    }
}
```

### Business Event Tracking Trait
```php
<?php
namespace App\Traits;

use Illuminate\Support\Facades\Http;

trait Trackable
{
    public static function bootTrackable(): void
    {
        static::created(function ($model) {
            $model->trackBusinessEvent('created');
        });

        static::updated(function ($model) {
            $model->trackBusinessEvent('updated');
        });

        static::deleted(function ($model) {
            $model->trackBusinessEvent('deleted');
        });
    }

    public function trackBusinessEvent(string $action): void
    {
        $event = [
            'event_type' => 'business.' . class_basename($this),
            'event_category' => 'business',
            'event_action' => $action,
            'event_label' => get_class($this),
            'event_properties' => [
                'model_id' => $this->id,
                'changes' => $this->getChanges(),
                'original' => $this->getOriginal(),
            ],
        ];

        Http::post(url('/api/events/track-server-side'), $event);
    }
}
```

### Event Processing Command
```php
<?php
namespace App\Console\Commands;

use App\Models\Analytics\EventTracking;
use Illuminate\Console\Command;

class ProcessEvents extends Command
{
    protected $signature = 'analytics:process-events';
    protected $description = 'Process and aggregate raw events';

    public function handle(): int
    {
        $this->processSessions();
        $this->processPageViews();
        $this->processConversions();

        return self::SUCCESS;
    }

    private function processSessions(): void
    {
        $incompleteSessions = DB::table('user_sessions')
            ->whereNull('ended_at')
            ->where('last_activity_at', '<', now()->subMinutes(30))
            ->get();

        foreach ($incompleteSessions as $session) {
            DB::table('user_sessions')
                ->where('session_id', $session->session_id)
                ->update([
                    'ended_at' => $session->last_activity_at,
                    'duration_seconds' => $session->started_at->diffInSeconds($session->last_activity_at),
                ]);
        }

        $this->info('Processed ' . count($incompleteSessions) . ' sessions');
    }

    private function processPageViews(): void
    {
        $popularPages = EventTracking::query()
            ->select('page_url', DB::raw('COUNT(*) as views'), DB::raw('COUNT(DISTINCT session_id) as unique_visitors'))
            ->where('event_type', 'page_view')
            ->whereDate('created_at', today())
            ->groupBy('page_url')
            ->orderByDesc('views')
            ->limit(100)
            ->get();

        DB::table('daily_page_stats')->updateOrInsert(
            ['date' => today()],
            ['page_stats' => json_encode($popularPages), 'updated_at' => now()]
        );
    }

    private function processConversions(): void
    {
        $conversions = EventTracking::query()
            ->select(
                'session_id',
                DB::raw('MIN(CASE WHEN event_type = "page_view" THEN created_at END) as first_visit'),
                DB::raw('MIN(CASE WHEN event_type = "add_to_cart" THEN created_at END) as add_to_cart_time'),
                DB::raw('MIN(CASE WHEN event_type = "checkout_started" THEN created_at END) as checkout_time'),
                DB::raw('MIN(CASE WHEN event_type = "order_completed" THEN created_at END) as conversion_time')
            )
            ->whereDate('created_at', today())
            ->groupBy('session_id')
            ->havingNotNull('conversion_time')
            ->get();

        $this->info('Processed ' . count($conversions) . ' conversions today');
    }
}
```
