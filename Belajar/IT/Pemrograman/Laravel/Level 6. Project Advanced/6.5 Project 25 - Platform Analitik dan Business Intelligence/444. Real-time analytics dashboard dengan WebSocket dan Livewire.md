# 444. Real-Time Analytics Dashboard dengan WebSocket dan Livewire

## Definisi & Target
Mengimplementasikan real-time analytics dashboard yang update otomatis menggunakan Laravel Reverb WebSocket dan Livewire components. Meliputi live metrics, real-time charts, auto-refresh, event-driven updates, dan low-latency data streaming.

## Komponen yang Terlibat
- Laravel Reverb (WebSocket)
- Laravel Livewire
- Real-time chart updates
- Live metrics counters
- Event-driven data push
- Auto-refresh mechanism
- Connection management
- Throttling & debouncing

## Perintah Artisan CLI
```bash
# Install dependencies
composer require laravel/reverb

php artisan make:livewire RealTimeMetrics

# Buat events
```

### Real-Time Event
```php
<?php
namespace App\Events;

use Illuminate\Broadcasting\Channel;
use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class MetricsUpdated implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public string $metricKey,
        public mixed $value,
        public ?array $metadata = null
    ) {}

    public function broadcastOn(): array
    {
        return [
            new Channel('analytics.metrics'),
        ];
    }

    public function broadcastAs(): string
    {
        return 'metrics.updated';
    }

    public function broadcastWith(): array
    {
        return [
            'metric' => $this->metricKey,
            'value' => $this->value,
            'metadata' => $this->metadata,
            'timestamp' => now()->toISOString(),
        ];
    }
}

class NewOrderAlert implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public function __construct(
        public array $order,
        public float $revenue
    ) {}

    public function broadcastOn(): array
    {
        return [
            new Channel('analytics.alerts'),
        ];
    }

    public function broadcastAs(): string
    {
        return 'order.new';
    }

    public function broadcastWith(): array
    {
        return [
            'order_id' => $this->order['id'],
            'customer' => $this->order['customer_name'],
            'amount' => $this->revenue,
            'items' => count($this->order['items'] ?? []),
            'timestamp' => now()->toISOString(),
        ];
    }
}
```

### Livewire Real-Time Metrics Component
```php
<?php
namespace App\Http\Livewire;

use Livewire\Component;
use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Cache;

class RealTimeMetrics extends Component
{
    public float $revenueToday = 0;
    public int $ordersToday = 0;
    public int $activeVisitors = 0;
    public float $conversionRate = 0;
    public float $revenueGrowth = 0;

    protected $listeners = ['echo:analytics.metrics,MetricsUpdated' => 'refreshMetric'];

    public function mount(): void
    {
        $this->refreshAll();
    }

    public function refreshMetric(array $payload): void
    {
        $metric = $payload['metric'] ?? null;

        match ($metric) {
            'revenue_today' => $this->revenueToday = $payload['value'],
            'orders_today' => $this->ordersToday = $payload['value'],
            'active_visitors' => $this->activeVisitors = $payload['value'],
            default => null,
        };
    }

    public function refreshAll(): void
    {
        $today = now()->format('Ymd');

        $this->revenueToday = (float) Cache::remember('rt_revenue_today', 10, function () use ($today) {
            return DB::connection('warehouse')
                ->table('agg_daily_sales')
                ->where('date_key', (int) $today)
                ->value('total_revenue') ?? 0;
        });

        $this->ordersToday = (int) Cache::remember('rt_orders_today', 10, function () use ($today) {
            return DB::connection('warehouse')
                ->table('agg_daily_sales')
                ->where('date_key', (int) $today)
                ->value('order_count') ?? 0;
        });

        $this->activeVisitors = (int) Cache::get('active_visitors', 0);

        $yesterday = now()->subDay()->format('Ymd');
        $yesterdayRevenue = (float) DB::connection('warehouse')
            ->table('agg_daily_sales')
            ->where('date_key', (int) $yesterday)
            ->value('total_revenue') ?? 0;

        $this->revenueGrowth = $yesterdayRevenue > 0
            ? round(($this->revenueToday - $yesterdayRevenue) / $yesterdayRevenue * 100, 1)
            : 0;

        $visits = (int) Cache::get('daily_visits', 1);
        $this->conversionRate = $visits > 0
            ? round(($this->ordersToday / $visits) * 100, 2)
            : 0;
    }

    public function render()
    {
        return view('livewire.real-time-metrics');
    }
}
```

### Livewire Real-Time Chart Component
```php
<?php
namespace App\Http\Livewire;

use Livewire\Component;
use App\Services\Analytics\ChartDataSerializer;
use App\Services\Analytics\MetricQueryBuilder;

class RealTimeChart extends Component
{
    public string $chartType = 'line';
    public string $metric = 'revenue';
    public string $period = 'last_24_hours';
    public array $chartData = [];
    public int $refreshInterval = 30;

    protected $listeners = [
        'echo:analytics.metrics,MetricsUpdated' => 'handleRealtimeUpdate',
        'refreshChart' => 'loadData',
    ];

    public function mount(): void
    {
        $this->loadData();
    }

    public function loadData(): void
    {
        $serializer = app(ChartDataSerializer::class);
        $builder = app(MetricQueryBuilder::class);

        $data = $builder
            ->from($this->getTable())
            ->select($this->getSelectColumns())
            ->period($this->period)
            ->orderBy('date_key')
            ->get();

        $this->chartData = $this->period === 'last_24_hours'
            ? $this->formatHourlyData($data)
            : $serializer->toLineChart(
                $data,
                'date_key',
                $this->getValueColumn(),
                ['label' => $this->metric, 'fill' => true]
            );
    }

    public function handleRealtimeUpdate(array $payload): void
    {
        if ($payload['metric'] === $this->metric) {
            $this->chartData['data']['datasets'][0]['data'][] = $payload['value'];

            // Keep last 24 points
            if (count($this->chartData['data']['datasets'][0]['data']) > 24) {
                array_shift($this->chartData['data']['datasets'][0]['data']);
                array_shift($this->chartData['data']['labels']);
            }

            $this->chartData['data']['labels'][] = now()->format('H:i');
        }
    }

    public function updatePeriod(string $period): void
    {
        $this->period = $period;
        $this->loadData();
    }

    private function formatHourlyData(array $data): array
    {
        $hours = [];
        for ($i = 0; $i < 24; $i++) {
            $hours[now()->subHours(23 - $i)->format('H')] = 0;
        }

        foreach ($data as $row) {
            $hour = substr($row['date_key'] ?? '0000', -2);
            $hours[$hour] = $row[$this->getValueColumn()] ?? 0;
        }

        return [
            'type' => 'line',
            'data' => [
                'labels' => array_keys($hours),
                'datasets' => [
                    [
                        'label' => $this->metric,
                        'data' => array_values($hours),
                        'borderColor' => '#3b82f6',
                        'backgroundColor' => 'rgba(59, 130, 246, 0.1)',
                        'fill' => true,
                        'tension' => 0.3,
                    ],
                ],
            ],
            'options' => [
                'responsive' => true,
                'maintainAspectRatio' => false,
                'animation' => ['duration' => 300],
                'plugins' => ['legend' => ['display' => false]],
                'scales' => ['y' => ['beginAtZero' => true]],
            ],
        ];
    }

    private function getTable(): string
    {
        return match ($this->metric) {
            'revenue', 'orders' => 'agg_daily_sales',
            'users' => 'agg_daily_users',
            default => 'agg_daily_sales',
        };
    }

    private function getValueColumn(): string
    {
        return match ($this->metric) {
            'revenue' => 'total_revenue',
            'orders' => 'order_count',
            'users' => 'active_users',
            default => 'total_revenue',
        };
    }

    private function getSelectColumns(): array
    {
        return ['date_key', $this->getValueColumn()];
    }

    public function render()
    {
        return view('livewire.real-time-chart');
    }
}
```

### Livewire Active Visitors Component
```php
<?php
namespace App\Http\Livewire;

use Livewire\Component;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;

class LiveVisitorCounter extends Component
{
    public int $activeVisitors = 0;
    public int $todayVisits = 0;
    public int $pageViewsToday = 0;
    public array $topPages = [];
    public array $trafficSources = [];

    protected $listeners = [
        'echo:analytics.metrics,MetricsUpdated' => 'updateVisitorCount',
        'echo-presence:analytics.visitors,here' => 'setVisitors',
        'echo-presence:analytics.visitors,joining' => 'visitorJoined',
        'echo-presence:analytics.visitors,leaving' => 'visitorLeft',
    ];

    public function mount(): void
    {
        $this->refreshStats();
    }

    public function setVisitors(array $users): void
    {
        $this->activeVisitors = count($users);
        Cache::put('active_visitors', $this->activeVisitors, 30);
    }

    public function visitorJoined(array $user): void
    {
        $this->activeVisitors++;
        Cache::increment('active_visitors');
    }

    public function visitorLeft(array $user): void
    {
        $this->activeVisitors = max(0, $this->activeVisitors - 1);
        Cache::decrement('active_visitors');
    }

    public function updateVisitorCount(array $payload): void
    {
        if ($payload['metric'] === 'active_visitors') {
            $this->activeVisitors = $payload['value'];
        }
    }

    public function refreshStats(): void
    {
        $today = now()->toDateString();

        $this->todayVisits = DB::connection('warehouse')
            ->table('user_sessions')
            ->whereDate('started_at', $today)
            ->count();

        $this->pageViewsToday = DB::connection('warehouse')
            ->table('event_tracking')
            ->where('event_type', 'page_view')
            ->whereDate('created_at', $today)
            ->count();

        $this->topPages = DB::connection('warehouse')
            ->table('event_tracking')
            ->select('page_url', DB::raw('COUNT(*) as views'))
            ->where('event_type', 'page_view')
            ->whereDate('created_at', $today)
            ->groupBy('page_url')
            ->orderByDesc('views')
            ->limit(5)
            ->get()
            ->toArray();

        $this->activeVisitors = Cache::get('active_visitors', 0);
    }

    public function render()
    {
        return view('livewire.live-visitor-counter');
    }
}
```

### Alert Stream Livewire Component
```php
<?php
namespace App\Http\Livewire;

use Livewire\Component;
use Illuminate\Support\Facades\Cache;

class AlertStream extends Component
{
    public array $alerts = [];
    public int $maxAlerts = 50;

    protected $listeners = [
        'echo:analytics.alerts,NewOrderAlert' => 'addAlert',
    ];

    public function mount(): void
    {
        $this->alerts = Cache::get('recent_alerts', []);
    }

    public function addAlert(array $payload): void
    {
        $alert = [
            'type' => $payload['event'] ?? 'order',
            'message' => "New order #{$payload['order_id']} from {$payload['customer']}",
            'amount' => $payload['amount'] ?? 0,
            'timestamp' => now()->toISOString(),
            'id' => uniqid(),
        ];

        array_unshift($this->alerts, $alert);

        if (count($this->alerts) > $this->maxAlerts) {
            array_pop($this->alerts);
        }

        Cache::put('recent_alerts', array_slice($this->alerts, 0, 20), 60);
    }

    public function clearAlerts(): void
    {
        $this->alerts = [];
        Cache::forget('recent_alerts');
    }

    public function render()
    {
        return view('livewire.alert-stream');
    }
}
```

### Broadcast Metrics Command
```php
<?php
namespace App\Console\Commands;

use App\Events\MetricsUpdated;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\Cache;
use Illuminate\Support\Facades\DB;

class BroadcastMetrics extends Command
{
    protected $signature = 'analytics:broadcast-metrics';
    protected $description = 'Broadcast current metrics via WebSocket';

    public function handle(): int
    {
        $today = now()->format('Ymd');

        $revenue = (float) DB::connection('warehouse')
            ->table('agg_daily_sales')
            ->where('date_key', (int) $today)
            ->value('total_revenue') ?? 0;

        $orders = (int) DB::connection('warehouse')
            ->table('agg_daily_sales')
            ->where('date_key', (int) $today)
            ->value('order_count') ?? 0;

        $activeVisitors = Cache::get('active_visitors', 0);

        broadcast(new MetricsUpdated('revenue_today', $revenue));
        broadcast(new MetricsUpdated('orders_today', $orders));
        broadcast(new MetricsUpdated('active_visitors', $activeVisitors));

        $this->info("Broadcasted metrics: revenue={$revenue}, orders={$orders}, visitors={$activeVisitors}");

        return self::SUCCESS;
    }
}
```
