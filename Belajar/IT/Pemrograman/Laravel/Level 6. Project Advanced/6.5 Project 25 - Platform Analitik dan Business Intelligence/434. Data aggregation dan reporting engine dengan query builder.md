# 434. Data Aggregation dan Reporting Engine dengan Query Builder

## Definisi & Target
Membangun reporting engine yang powerful menggunakan Laravel Query Builder untuk aggregasi data multi-dimensional. Meliputi dynamic query building, metric calculations, time-series analysis, grouping, filtering, dan pagination untuk large datasets.

## Komponen yang Terlibat
- Query Builder (fluent)
- Dynamic aggregation
- Metric calculations
- Time-series queries
- Grouping & filtering
- Pivot tables
- Caching query results
- Query optimization (EXPLAIN)

### Metric Query Builder
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\DB;
use Illuminate\Support\Facades\Cache;

class MetricQueryBuilder
{
    private array $query;
    private string $connection = 'warehouse';

    public function __construct()
    {
        $this->reset();
    }

    public function reset(): void
    {
        $this->query = [
            'table' => null,
            'select' => [],
            'aggregations' => [],
            'filters' => [],
            'group_by' => [],
            'order_by' => [],
            'period' => null,
            'limit' => 100,
            'offset' => 0,
        ];
    }

    public function from(string $table): self
    {
        $this->query['table'] = $table;
        return $this;
    }

    public function select(array $columns): self
    {
        $this->query['select'] = $columns;
        return $this;
    }

    public function aggregate(string $function, string $column, string $alias = null): self
    {
        $this->query['aggregations'][] = [
            'function' => $function,
            'column' => $column,
            'alias' => $alias,
        ];
        return $this;
    }

    public function sum(string $column, string $alias = null): self
    {
        return $this->aggregate('SUM', $column, $alias);
    }

    public function count(string $column = '*', string $alias = null): self
    {
        return $this->aggregate('COUNT', $column, $alias);
    }

    public function avg(string $column, string $alias = null): self
    {
        return $this->aggregate('AVG', $column, $alias);
    }

    public function min(string $column, string $alias = null): self
    {
        return $this->aggregate('MIN', $column, $alias);
    }

    public function max(string $column, string $alias = null): self
    {
        return $this->aggregate('MAX', $column, $alias);
    }

    public function where(string $column, string $operator, mixed $value = null): self
    {
        $this->query['filters'][] = [
            'type' => 'basic',
            'column' => $column,
            'operator' => $operator,
            'value' => $value,
        ];
        return $this;
    }

    public function whereIn(string $column, array $values): self
    {
        $this->query['filters'][] = [
            'type' => 'in',
            'column' => $column,
            'values' => $values,
        ];
        return $this;
    }

    public function whereBetween(string $column, array $range): self
    {
        $this->query['filters'][] = [
            'type' => 'between',
            'column' => $column,
            'range' => $range,
        ];
        return $this;
    }

    public function whereDate(string $column, string $operator, string $date): self
    {
        return $this->where($column, $operator, $date);
    }

    public function groupBy(array|string $columns): self
    {
        $this->query['group_by'] = array_merge(
            $this->query['group_by'],
            is_array($columns) ? $columns : [$columns]
        );
        return $this;
    }

    public function orderBy(string $column, string $direction = 'desc'): self
    {
        $this->query['order_by'][] = compact('column', 'direction');
        return $this;
    }

    public function period(string $period): self
    {
        $this->query['period'] = $period;

        return match ($period) {
            'today' => $this->whereDate('created_at', '>=', today()),
            'yesterday' => $this->whereBetween('created_at', [today()->subDay(), today()]),
            'last_7_days' => $this->whereDate('created_at', '>=', today()->subDays(7)),
            'last_30_days' => $this->whereDate('created_at', '>=', today()->subDays(30)),
            'this_month' => $this->whereDate('created_at', '>=', now()->startOfMonth()),
            'last_month' => $this->whereBetween('created_at', [now()->subMonth()->startOfMonth(), now()->startOfMonth()]),
            'this_year' => $this->whereDate('created_at', '>=', now()->startOfYear()),
            'last_12_months' => $this->whereDate('created_at', '>=', now()->subMonths(12)),
            default => $this,
        };
    }

    public function limit(int $limit): self
    {
        $this->query['limit'] = $limit;
        return $this;
    }

    public function offset(int $offset): self
    {
        $this->query['offset'] = $offset;
        return $this;
    }

    public function get(): array
    {
        $cacheKey = 'metric_query:' . md5(json_encode($this->query));

        return Cache::remember($cacheKey, 300, function () {
            return $this->build()->get()->toArray();
        });
    }

    public function first(): ?array
    {
        return $this->limit(1)->get()[0] ?? null;
    }

    public function count(): int
    {
        return $this->build()->count();
    }

    public function paginate(int $perPage = 20): array
    {
        $page = request()->input('page', 1);
        $total = $this->count();

        $this->limit($perPage)->offset(($page - 1) * $perPage);

        return [
            'data' => $this->get(),
            'pagination' => [
                'total' => $total,
                'per_page' => $perPage,
                'current_page' => (int) $page,
                'last_page' => (int) ceil($total / $perPage),
            ],
        ];
    }

    public function toSql(): string
    {
        return $this->build()->toSql();
    }

    private function build(): \Illuminate\Database\Query\Builder
    {
        $query = DB::connection($this->connection)
            ->table($this->query['table']);

        // Aggregations
        foreach ($this->query['aggregations'] as $agg) {
            $expression = "{$agg['function']}({$agg['column']})";
            if ($agg['alias']) {
                $expression .= " as {$agg['alias']}";
            }
            $query->selectRaw($expression);
        }

        // Select columns
        if (!empty($this->query['select'])) {
            $query->addSelect($this->query['select']);
        }

        // Filters
        foreach ($this->query['filters'] as $filter) {
            match ($filter['type']) {
                'basic' => $query->where($filter['column'], $filter['operator'], $filter['value']),
                'in' => $query->whereIn($filter['column'], $filter['values']),
                'between' => $query->whereBetween($filter['column'], $filter['range']),
            };
        }

        // Group by
        if (!empty($this->query['group_by'])) {
            $query->groupBy($this->query['group_by']);
        }

        // Order by
        foreach ($this->query['order_by'] as $order) {
            $query->orderBy($order['column'], $order['direction']);
        }

        // Limit & offset
        $query->take($this->query['limit']);
        $query->skip($this->query['offset']);

        return $query;
    }
}
```

### Report Generator
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\Cache;

class ReportGenerator
{
    private MetricQueryBuilder $queryBuilder;
    private array $reportDefinitions;

    public function __construct(MetricQueryBuilder $queryBuilder)
    {
        $this->queryBuilder = $queryBuilder;
        $this->reportDefinitions = config('reports');
    }

    public function generate(string $reportName, array $params = []): array
    {
        $definition = $this->reportDefinitions[$reportName] ?? null;

        if (!$definition) {
            throw new \InvalidArgumentException("Unknown report: {$reportName}");
        }

        $cacheKey = "report:{$reportName}:" . md5(json_encode($params));

        return Cache::remember($cacheKey, $definition['cache_ttl'] ?? 300, function () use ($definition, $params) {
            return $this->buildReport($definition, $params);
        });
    }

    private function buildReport(array $definition, array $params): array
    {
        $builder = $this->queryBuilder;

        $builder->from($definition['table']);

        foreach ($definition['aggregations'] as $agg) {
            $builder->{$agg['function']}($agg['column'], $agg['alias'] ?? null);
        }

        if (isset($definition['group_by'])) {
            $builder->groupBy($definition['group_by']);
        }

        if (isset($params['period'])) {
            $builder->period($params['period']);
        }

        if (isset($params['filters'])) {
            foreach ($params['filters'] as $filter) {
                $builder->where($filter['column'], $filter['operator'], $filter['value']);
            }
        }

        if (isset($params['group_by'])) {
            $builder->groupBy($params['group_by']);
        }

        if (isset($definition['order_by'])) {
            $builder->orderBy($definition['order_by'][0], $definition['order_by'][1] ?? 'desc');
        }

        $data = $builder->get();

        $this->enrichWithMetadata($data, $definition, $params);

        return [
            'report_name' => $definition['name'],
            'generated_at' => now()->toISOString(),
            'period' => $params['period'] ?? 'all',
            'total_records' => count($data),
            'data' => $data,
            'summary' => $this->calculateSummary($data, $definition),
            'metadata' => $definition['metadata'] ?? [],
        ];
    }

    private function calculateSummary(array $data, array $definition): array
    {
        $summary = [];

        foreach ($definition['summary_fields'] ?? [] as $field => $functions) {
            $values = array_column($data, $field);

            foreach ($functions as $func) {
                $summary["{$field}_{$func}"] = match ($func) {
                    'sum' => array_sum($values),
                    'avg' => count($values) > 0 ? array_sum($values) / count($values) : 0,
                    'min' => min($values),
                    'max' => max($values),
                    'count' => count($values),
                    default => null,
                };
            }
        }

        return $summary;
    }

    private function enrichWithMetadata(array &$data, array $definition, array $params): void
    {
        // Add percentages, growth rates, etc.
    }
}
```

### Aggregation Service
```php
<?php
namespace App\Services\Analytics;

use Illuminate\Support\Facades\DB;

class AggregationService
{
    public function calculateDailyAggregates(): array
    {
        $results = [];

        $results['daily_revenue'] = $this->aggregateDailyRevenue();
        $results['daily_users'] = $this->aggregateDailyUsers();
        $results['daily_events'] = $this->aggregateDailyEvents();

        return $results;
    }

    public function aggregateDailyRevenue(): void
    {
        $dailyData = DB::connection('warehouse')
            ->table('fact_sales')
            ->select(
                'date_key',
                DB::raw('SUM(total_amount) as total_revenue'),
                DB::raw('SUM(quantity) as total_units'),
                DB::raw('COUNT(DISTINCT source_order_id) as order_count'),
                DB::raw('AVG(total_amount) as avg_order_value'),
            )
            ->whereDate('created_at', today())
            ->groupBy('date_key')
            ->get();

        foreach ($dailyData as $row) {
            DB::connection('warehouse')
                ->table('agg_daily_sales')
                ->updateOrInsert(
                    ['date_key' => $row->date_key],
                    [
                        'total_revenue' => $row->total_revenue,
                        'total_units' => $row->total_units,
                        'order_count' => $row->order_count,
                        'avg_order_value' => $row->avg_order_value,
                        'updated_at' => now(),
                    ]
                );
        }
    }

    public function aggregateDailyUsers(): void
    {
        $dailyData = DB::connection('warehouse')
            ->table('user_sessions')
            ->select(
                DB::raw("TO_CHAR(started_at, 'YYYYMMDD') as date_key"),
                DB::raw('COUNT(DISTINCT user_id) as active_users'),
                DB::raw('COUNT(DISTINCT session_id) as total_sessions'),
                DB::raw('AVG(duration_seconds) as avg_session_duration'),
                DB::raw('AVG(event_count) as avg_events_per_session'),
            )
            ->whereDate('started_at', today())
            ->groupBy(DB::raw("TO_CHAR(started_at, 'YYYYMMDD')"))
            ->get();

        foreach ($dailyData as $row) {
            DB::connection('warehouse')
                ->table('agg_daily_users')
                ->updateOrInsert(
                    ['date_key' => $row->date_key],
                    [
                        'active_users' => $row->active_users,
                        'total_sessions' => $row->total_sessions,
                        'avg_session_duration' => $row->avg_session_duration,
                        'avg_events_per_session' => $row->avg_events_per_session,
                        'updated_at' => now(),
                    ]
                );
        }
    }

    public function aggregateDailyEvents(): void
    {
        $dailyData = DB::connection('warehouse')
            ->table('event_tracking')
            ->select(
                DB::raw("TO_CHAR(created_at, 'YYYYMMDD') as date_key"),
                'event_type',
                DB::raw('COUNT(*) as event_count'),
                DB::raw('COUNT(DISTINCT session_id) as unique_sessions'),
            )
            ->whereDate('created_at', today())
            ->groupBy(DB::raw("TO_CHAR(created_at, 'YYYYMMDD')"), 'event_type')
            ->get();

        foreach ($dailyData as $row) {
            DB::connection('warehouse')
                ->table('agg_daily_events')
                ->updateOrInsert(
                    [
                        'date_key' => $row->date_key,
                        'event_type' => $row->event_type,
                    ],
                    [
                        'event_count' => $row->event_count,
                        'unique_sessions' => $row->unique_sessions,
                        'updated_at' => now(),
                    ]
                );
        }
    }
}
```

### Report API Controller
```php
<?php
namespace App\Http\Controllers\Analytics;

use App\Services\Analytics\ReportGenerator;
use App\Services\Analytics\MetricQueryBuilder;
use Illuminate\Http\Request;

class ReportController extends Controller
{
    public function __construct(
        private ReportGenerator $reportGenerator,
        private MetricQueryBuilder $queryBuilder
    ) {}

    public function show(string $reportName, Request $request): mixed
    {
        try {
            $data = $this->reportGenerator->generate($reportName, $request->all());
            return response()->json($data);
        } catch (\InvalidArgumentException $e) {
            return response()->json(['error' => $e->getMessage()], 404);
        }
    }

    public function query(Request $request): mixed
    {
        $validated = $request->validate([
            'table' => 'required|string',
            'aggregations' => 'required|array',
            'aggregations.*.function' => 'required|in:SUM,COUNT,AVG,MIN,MAX',
            'aggregations.*.column' => 'required|string',
            'filters' => 'nullable|array',
            'group_by' => 'nullable|array',
            'period' => 'nullable|string',
        ]);

        $builder = $this->queryBuilder
            ->from($validated['table']);

        foreach ($validated['aggregations'] as $agg) {
            $builder->aggregate($agg['function'], $agg['column'], $agg['alias'] ?? null);
        }

        if ($period = $validated['period'] ?? null) {
            $builder->period($period);
        }

        if ($groupBy = $validated['group_by'] ?? null) {
            $builder->groupBy($groupBy);
        }

        $result = $request->has('paginate')
            ? $builder->paginate($request->input('per_page', 20))
            : ['data' => $builder->get()];

        return response()->json($result);
    }
}
```
