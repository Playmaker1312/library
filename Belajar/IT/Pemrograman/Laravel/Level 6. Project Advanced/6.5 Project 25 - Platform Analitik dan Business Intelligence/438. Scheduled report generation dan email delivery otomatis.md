# 438. Scheduled Report Generation dan Email Delivery Otomatis

## Definisi & Target
Mengimplementasikan sistem scheduled reports yang menghasilkan laporan secara otomatis dan mengirimkannya via email. Meliputi job scheduling, report templates, dynamic recipient lists, multiple format support, dan delivery tracking.

- Mail notifications
- Queue jobs
- Report templates

- Delivery tracking

### Scheduled Report Model
```php
<?php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Builder;

class ScheduledReport extends Model
{
    protected $fillable = [
        'name',
        'description',
        'report_name',
        'params',
        'format',
        'schedule_frequency',
        'schedule_cron',
        'recipients',
        'subject_template',
        'body_template',
        'is_active',
        'last_run_at',
        'next_run_at',
    ];

    protected $casts = [
        'params' => 'array',
        'recipients' => 'array',
        'is_active' => 'boolean',
        'last_run_at' => 'datetime',
        'next_run_at' => 'datetime',
    ];

    public function scopeActive(Builder $query): Builder
    {
        return $query->where('is_active', true);
    }

    public function scopeDue(Builder $query): Builder
    {
        return $query->active()
            ->where('next_run_at', '<=', now())
            ->orWhereNull('next_run_at');
    }

    public function deliveries()
    {
        return $this->hasMany(ReportDelivery::class);
    }

    public function isDue(): bool
    {
        return $this->is_active && (
            is_null($this->next_run_at) ||
            $this->next_run_at <= now()
        );
    }

    public function calculateNextRun(): ?\DateTime
    {
        if (!$this->schedule_cron) {
            return null;
        }

        $cron = \Cron\CronExpression::factory($this->schedule_cron);
        return $cron->getNextRunDate();
    }
}
```

### Mail Notification for Report
```php
<?php
namespace App\Mail;

use Illuminate\Bus\Queueable;
use Illuminate\Mail\Mailable;
use Illuminate\Mail\Mailables\Content;
use Illuminate\Mail\Mailables\Envelope;
use Illuminate\Mail\Mailables\Attachment;
use Illuminate\Queue\SerializesModels;

class ReportMail extends Mailable
{
    use Queueable, SerializesModels;

    public function __construct(
        private string $reportName,
        private string $filePath,
        private string $format,
        private array $summary = []
    ) {}

    public function envelope(): Envelope
    {
        return new Envelope(
            subject: "Report: {$this->reportName} - " . now()->format('Y-m-d'),
            tags: ['report', $this->format],
        );
    }

    public function content(): Content
    {
        return new Content(
            view: 'emails.reports.delivery',
            with: [
                'reportName' => $this->reportName,
                'format' => strtoupper($this->format),
                'summary' => $this->summary,
                'generatedAt' => now()->format('Y-m-d H:i:s'),
            ],
        );
    }

    public function attachments(): array
    {
        return [
            Attachment::fromStorageDisk('public', $this->filePath)
                ->as(basename($this->filePath))
                ->withMime($this->getMimeType()),
        ];
    }

    private function getMimeType(): string
    {
        return match ($this->format) {
            'xlsx' => 'application/vnd.openxmlformats-officedocument.spreadsheetml.sheet',
            'csv' => 'text/csv',
            'pdf' => 'application/pdf',
            default => 'application/octet-stream',
        };
    }
}
```

### Report Delivery Job
```php
<?php
namespace App\Jobs;

use App\Exports\SalesReportExport;
use App\Mail\ReportMail;
use App\Models\ScheduledReport;
use App\Models\ReportDelivery;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;
use Illuminate\Support\Facades\Mail;
use Illuminate\Support\Facades\Storage;
use Maatwebsite\Excel\Facades\Excel;

class DeliverReportJob implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public int $timeout = 600;
    public int $tries = 3;

    public function __construct(
        private ScheduledReport $scheduledReport
    ) {}

    public function handle(): void
    {
        $report = $this->scheduledReport;

        if (!$report->is_active) {
            return;
        }

        try {
            $export = new SalesReportExport(
                $report->params['period'] ?? 'last_30_days',
                $report->params['category'] ?? null
            );

            $filename = "scheduled/{$report->id}/{$report->report_name}-" .
                now()->format('Y-m-d-His') . ".{$report->format}";

            Excel::store($export, $filename, 'public');

            $summary = $this->getSummary($report);
            $filePath = Storage::disk('public')->path($filename);

            foreach ($report->recipients as $recipient) {
                try {
                    Mail::to($recipient['email'], $recipient['name'] ?? null)
                        ->send(new ReportMail(
                            $report->name,
                            $filename,
                            $report->format,
                            $summary
                        ));

                    ReportDelivery::create([
                        'scheduled_report_id' => $report->id,
                        'recipient_email' => $recipient['email'],
                        'status' => 'delivered',
                        'delivered_at' => now(),
                    ]);
                } catch (\Exception $e) {
                    ReportDelivery::create([
                        'scheduled_report_id' => $report->id,
                        'recipient_email' => $recipient['email'],
                        'status' => 'failed',
                        'error_message' => $e->getMessage(),
                    ]);
                }
            }

            $report->update([
                'last_run_at' => now(),
                'next_run_at' => $report->calculateNextRun(),
            ]);

            Storage::disk('public')->delete($filename);

        } catch (\Exception $e) {
            $report->deliveries()->create([
                'recipient_email' => 'system',
                'status' => 'failed',
                'error_message' => $e->getMessage(),
            ]);

            if ($this->attempts() >= $this->tries) {
                $report->update(['is_active' => false]);
            }

            throw $e;
        }
    }

    private function getSummary(ScheduledReport $report): array
    {
        // Generate summary from report data
        return [];
    }
}
```

### Scheduler Command
```php
<?php
namespace App\Console\Commands;

use App\Jobs\DeliverReportJob;
use App\Models\ScheduledReport;
use Illuminate\Console\Command;

class GenerateScheduledReports extends Command
{
    protected $signature = 'reports:generate-scheduled';
    protected $description = 'Generate and deliver all due scheduled reports';

    public function handle(): int
    {
        $dueReports = ScheduledReport::due()->get();

        $this->info("Found {$dueReports->count()} due reports");

        foreach ($dueReports as $report) {
            $this->line("Processing: {$report->name}");

            DeliverReportJob::dispatch($report);

            $this->info("  Queued: {$report->name}");
        }

        $this->info('All scheduled reports have been queued for delivery');

        return self::SUCCESS;
    }
}
```

### Kernel Schedule Configuration
```php
<?php
// app/Console/Kernel.php
namespace App\Console;

use Illuminate\Console\Scheduling\Schedule;
use Illuminate\Foundation\Console\Kernel as ConsoleKernel;

class Kernel extends ConsoleKernel
{
    protected function schedule(Schedule $schedule): void
    {
        // Generate scheduled reports every 5 minutes
        $schedule->command('reports:generate-scheduled')
            ->everyFiveMinutes()
            ->withoutOverlapping()
            ->runInBackground();

        // Aggregation jobs
        $schedule->command('analytics:aggregate-daily')
            ->dailyAt('01:00')
            ->withoutOverlapping();

        // Cleanup old exports
        $schedule->command('exports:cleanup')
            ->dailyAt('03:00');

        // Warm report cache for common reports
        $schedule->command('reports:warm-cache')
            ->hourly();
    }
}
```

### Report Subscription Management
```php
<?php
namespace App\Http\Controllers\Analytics;

use App\Models\ScheduledReport;
use App\Models\ReportSubscription;
use Illuminate\Http\Request;

class ReportSubscriptionController extends Controller
{
    public function index(): mixed
    {
        $subscriptions = auth()->user()->reportSubscriptions()
            ->with('scheduledReport')
            ->active()
            ->get();

        return view('analytics.subscriptions', compact('subscriptions'));
    }

    public function subscribe(Request $request, ScheduledReport $report): mixed
    {
        $validated = $request->validate([
            'email' => 'required|email',
            'format' => 'sometimes|in:xlsx,csv,pdf',
        ]);

        $subscription = ReportSubscription::firstOrCreate(
            [
                'user_id' => auth()->id(),
                'scheduled_report_id' => $report->id,
            ],
            [
                'email' => $validated['email'],
                'format' => $validated['format'] ?? 'xlsx',
            ]
        );

        if (!$subscription->wasRecentlyCreated) {
            $subscription->update([
                'is_active' => true,
                'unsubscribed_at' => null,
            ]);
        }

        return back()->with('success', 'Subscribed to report successfully');
    }

    public function unsubscribe(string $token): mixed
    {
        $subscription = ReportSubscription::whereHas('user', function ($q) use ($token) {
            $q->where('unsubscribe_token', $token);
        })->firstOrFail();

        $subscription->update([
            'is_active' => false,
            'unsubscribed_at' => now(),
        ]);

        return redirect()->route('login')
            ->with('success', 'You have been unsubscribed from the report.');
    }
}
```

```php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;

{
```
