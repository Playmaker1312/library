# 439. Dashboard Builder Dinamis dengan Drag-and-Drop Widgets

## Definisi & Target
Membangun dashboard builder yang memungkinkan user menyusun dashboard kustom dengan drag-and-drop widgets. Meliputi grid layout, widget library, customizable filters, save/load layout, sharing, dan role-based access.

## Komponen yang Terlibat
- Grid layout system (Gridstack.js)
- Widget library
- Drag-and-drop interface
- Dashboard persistence
- Filter synchronization
- Widget configuration
- Dashboard sharing
- Role-based access

```php
<?php
namespace App\Models\Analytics;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\Builder;

class Dashboard extends Model
{
    protected $fillable = [
        'user_id',
        'name',
        'description',
        'layout',
        'widgets_config',
        'filters',
        'is_default',
        'is_public',
        'refresh_interval',
        'team_id',
        'shared_with',
    ];

    protected $casts = [
        'layout' => 'array',
        'widgets_config' => 'array',
        'filters' => 'array',
        'is_default' => 'boolean',
        'is_public' => 'boolean',
        'shared_with' => 'array',
    ];

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public function widgets()
    {
        return $this->hasMany(DashboardWidget::class)->orderBy('position');
    }

    public function scopeOwn($query)
    {
        return $query->where('user_id', auth()->id());
    }

    public function scopeAccessible($query)
    {
        return $query->where(function (Builder $q) {
            $q->where('user_id', auth()->id())
              ->orWhere('is_public', true)
              ->orWhereJsonContains('shared_with', auth()->id());
        });
    }

    public function duplicate(): self
    {
        $copy = $this->replicate()->fill([
            'name' => $this->name . ' (Copy)',
            'is_default' => false,
        ]);
        $copy->save();

        foreach ($this->widgets as $widget) {
            $widget->replicate()->fill(['dashboard_id' => $copy->id])->save();
        }

        return $copy->fresh();
    }
}

class DashboardWidget extends Model
{
    protected $fillable = [
        'dashboard_id',
        'widget_type',
        'widget_key',
        'title',
        'config',
        'position',
        'width',
        'height',
        'min_width',
        'min_height',
    ];

    protected $casts = [
        'config' => 'array',
        'position' => 'integer',
    ];

    public function dashboard()
    {
        return $this->belongsTo(Dashboard::class);
    }

    public function getData(): mixed
    {
        $provider = app("App\\Services\\Analytics\\Widgets\\{$this->widget_type}Widget");
        return $provider->getData($this->config);
    }
}
```

### Dashboard Controller
```php
<?php
namespace App\Http\Controllers\Analytics;

use App\Models\Analytics\Dashboard;
use App\Models\Analytics\DashboardWidget;
use Illuminate\Http\Request;

class DashboardController extends Controller
{
    public function index(): mixed
    {
        return Dashboard::accessible()
            ->with('user')
            ->orderBy('updated_at', 'desc')
            ->paginate(20);
    }

    public function store(Request $request): mixed
    {
        $validated = $request->validate([
            'name' => 'required|string|max:255',
            'description' => 'nullable|string',
            'layout' => 'nullable|array',
            'filters' => 'nullable|array',
            'refresh_interval' => 'nullable|integer|min:0',
        ]);

        $dashboard = auth()->user()->dashboards()->create($validated + [
            'layout' => $validated['layout'] ?? ['columns' => 12, 'rowHeight' => 100],
            'filters' => $validated['filters'] ?? [],
        ]);

        return response()->json($dashboard, 201);
    }

    public function show(Dashboard $dashboard): mixed
    {
        $this->authorize('view', $dashboard);

        return response()->json([
            'dashboard' => $dashboard->load('widgets'),
            'widget_data' => $this->getWidgetData($dashboard),
        ]);
    }

    public function update(Request $request, Dashboard $dashboard): mixed
    {
        $this->authorize('update', $dashboard);

        $validated = $request->validate([
            'name' => 'sometimes|string|max:255',
            'description' => 'nullable|string',
            'layout' => 'nullable|array',
            'widgets_config' => 'nullable|array',
            'filters' => 'nullable|array',
            'is_default' => 'boolean',
            'is_public' => 'boolean',
            'refresh_interval' => 'nullable|integer|min:0',
        ]);

        $dashboard->update($validated);

        return response()->json($dashboard->fresh());
    }

    public function destroy(Dashboard $dashboard): mixed
    {
        $this->authorize('delete', $dashboard);

        $dashboard->widgets()->delete();
        $dashboard->delete();

        return response()->json(['message' => 'Dashboard deleted']);
    }

    public function saveLayout(Request $request, Dashboard $dashboard): mixed
    {
        $this->authorize('update', $dashboard);

        $validated = $request->validate([
            'layout' => 'required|array',
            'widgets' => 'required|array',
            'widgets.*.id' => 'nullable|exists:dashboard_widgets,id',
            'widgets.*.widget_type' => 'required|string',
            'widgets.*.title' => 'required|string',
            'widgets.*.config' => 'nullable|array',
            'widgets.*.position' => 'required|integer',
            'widgets.*.width' => 'required|integer',
            'widgets.*.height' => 'required|integer',
        ]);

        // Sync widgets
        $existingIds = $dashboard->widgets()->pluck('id');
        $incomingIds = collect($validated['widgets'])->pluck('id')->filter();

        // Delete removed widgets
        $existingIds->diff($incomingIds)->each(fn ($id) => DashboardWidget::find($id)?->delete());

        // Update or create widgets
        foreach ($validated['widgets'] as $widgetData) {
            DashboardWidget::updateOrCreate(
                ['id' => $widgetData['id'] ?? null],
                [
                    'dashboard_id' => $dashboard->id,
                    'widget_type' => $widgetData['widget_type'],
                    'title' => $widgetData['title'],
                    'config' => $widgetData['config'] ?? [],
                    'position' => $widgetData['position'],
                    'width' => $widgetData['width'],
                    'height' => $widgetData['height'],
                ]
            );
        }

        $dashboard->update(['layout' => $validated['layout']]);

        return response()->json(['message' => 'Layout saved']);
    }

    public function setDefault(Dashboard $dashboard): mixed
    {
        $this->authorize('update', $dashboard);

        Dashboard::where('user_id', auth()->id())
            ->where('id', '!=', $dashboard->id)
            ->update(['is_default' => false]);

        $dashboard->update(['is_default' => true]);

        return response()->json(['message' => 'Default dashboard set']);
    }

    private function getWidgetData(Dashboard $dashboard): array
    {
        $data = [];
        foreach ($dashboard->widgets as $widget) {
            try {
                $data[$widget->id] = $widget->getData();
            } catch (\Exception $e) {
                $data[$widget->id] = ['error' => $e->getMessage()];
            }
        }
        return $data;
    }
}
```

### Widget Provider
```php
<?php
namespace App\Services\Analytics\Widgets;

abstract class BaseWidget
{
    abstract public function getType(): string;
    abstract public function getData(array $config): array;
    abstract public function getDefaultConfig(): array;

    public function validate(array $config): array
    {
        return $config;
    }

    public function getCacheKey(array $config): string
    {
        return "widget:{$this->getType()}:" . md5(json_encode($config));
    }
}

class RevenueChartWidget extends BaseWidget
{
    public function getType(): string
    {
        return 'revenue_chart';
    }

    public function getDefaultConfig(): array
    {
        return [
            'period' => 'last_30_days',
            'chart_type' => 'line',
            'show_previous_period' => true,
        ];
    }

    public function getData(array $config): array
    {
        $builder = app(MetricQueryBuilder::class);

        $data = $builder
            ->from('agg_daily_sales')
            ->select(['date_key', 'total_revenue', 'order_count'])
            ->period($config['period'] ?? 'last_30_days')
            ->orderBy('date_key')
            ->get();

        return app(ChartDataSerializer::class)->toLineChart(
            $data,
            'date_key',
            'total_revenue',
            ['label' => 'Revenue']
        );
    }
}

class StatCardWidget extends BaseWidget
{
    public function getType(): string
    {
        return 'stat_card';
    }

    public function getDefaultConfig(): array
    {
        return [
            'metric' => 'total_revenue',
            'label' => 'Total Revenue',
            'format' => 'currency',
            'period' => 'today',
            'comparison' => 'previous_period',
        ];
    }

    public function getData(array $config): array
    {
        $current = $this->getMetricValue($config['metric'], $config['period'] ?? 'today');
        $previous = $config['comparison']
            ? $this->getPreviousValue($config['metric'], $config['period'] ?? 'today')
            : null;

        $growth = $previous && $previous > 0
            ? round(($current - $previous) / $previous * 100, 1)
            : null;

        return [
            'value' => $current,
            'previous_value' => $previous,
            'growth' => $growth,
            'growth_label' => $growth !== null ? "vs " . $this->getPeriodLabel($config['period']) : null,
        ];
    }

    private function getMetricValue(string $metric, string $period): float
    {
        return DB::connection('warehouse')
            ->table('agg_daily_sales')
            ->when($period === 'today', fn ($q) => $q->where('date_key', (int) now()->format('Ymd')))
            ->when($period === 'yesterday', fn ($q) => $q->where('date_key', (int) now()->subDay()->format('Ymd')))
            ->when($period === 'this_month', fn ($q) => $q->where('month', now()->month)->where('year', now()->year))
            ->value(match ($metric) {
                'total_revenue' => 'SUM(total_revenue)',
                'order_count' => 'SUM(order_count)',
                'avg_order_value' => 'AVG(avg_order_value)',
                default => 'SUM(total_revenue)',
            }) ?? 0;
    }

    private function getPreviousValue(string $metric, string $period): float
    {
        return match ($period) {
            'today' => $this->getMetricValue($metric, 'yesterday'),
            'yesterday' => $this->getMetricValue($metric, 'day_before_yesterday'),
            'this_month' => $this->getMetricValue($metric, 'last_month'),
            default => 0,
        };
    }

    private function getPeriodLabel(string $period): string
    {
        return match ($period) {
            'today' => 'yesterday',
            'this_month' => 'last month',
            default => 'previous period',
        };
    }
}
```

### Frontend JavaScript (Gridstack Integration)
```javascript
// resources/js/analytics/dashboard-builder.js
import GridStack from 'gridstack';
import 'gridstack/dist/gridstack.min.css';

class DashboardBuilder {
    constructor(options) {
        this.dashboardId = options.dashboardId;
        this.widgetsEndpoint = options.widgetsEndpoint;
        this.saveEndpoint = options.saveEndpoint;
        this.grid = null;
        this.widgetTypes = options.widgetTypes || {};

        this.init();
    }

    init() {
        this.grid = GridStack.init({
            column: 12,
            cellHeight: 100,
            minRow: 3,
            margin: 8,
            float: false,
            acceptWidgets: true,
            removable: true,
            disableOneRowMode: true,
        });

        this.loadWidgets();
        this.setupEventListeners();
    }

    loadWidgets() {
        fetch(`${this.widgetsEndpoint}/${this.dashboardId}`)
            .then(response => response.json())
            .then(data => {
                const widgets = data.dashboard.widgets || [];
                const widgetData = data.widget_data || {};

                widgets.forEach(widget => {
                    this.addWidgetToGrid(widget, widgetData[widget.id]);
                });

                this.grid.commit();
            });
    }

    addWidgetToGrid(widget, data) {
        const content = document.createElement('div');
        content.className = 'grid-stack-item';
        content.innerHTML = this.renderWidget(widget, data);
        content.dataset.widgetId = widget.id;
        content.dataset.widgetType = widget.widget_type;

        this.grid.addWidget(content, {
            x: widget.position % 12,
            y: Math.floor(widget.position / 12),
            w: widget.width || 4,
            h: widget.height || 3,
            minW: 2,
            minH: 2,
        });
    }

    renderWidget(widget, data) {
        const widgetType = this.widgetTypes[widget.widget_type];
        if (!widgetType) return '<div>Unknown widget type</div>';

        return `
            <div class="widget-card">
                <div class="widget-header">
                    <h5 class="widget-title">${widget.title}</h5>
                    <div class="widget-actions">
                        <button class="btn btn-sm refresh-widget" title="Refresh">
                            <i class="icon-refresh"></i>
                        </button>
                        <button class="btn btn-sm configure-widget" title="Configure">
                            <i class="icon-settings"></i>
                        </button>
                        <button class="btn btn-sm remove-widget" title="Remove">
                            <i class="icon-close"></i>
                        </button>
                    </div>
                </div>
                <div class="widget-body" data-widget-key="${widget.widget_key}">
                    <canvas id="chart-${widget.id}"></canvas>
                </div>
                <div class="widget-loading" style="display:none">
                    <div class="spinner"></div>
                </div>
            </div>
        `;
    }

    setupEventListeners() {
        document.getElementById('add-widget-modal')?.addEventListener('show.bs.modal', () => {
            this.renderWidgetLibrary();
        });

        document.getElementById('save-dashboard')?.addEventListener('click', () => {
            this.saveLayout();
        });

        document.getElementById('filter-apply')?.addEventListener('click', () => {
            this.applyFilters();
        });

        // Auto-save on layout change
        let saveTimeout;
        this.grid.on('change', () => {
            clearTimeout(saveTimeout);
            saveTimeout = setTimeout(() => this.saveLayout(), 2000);
        });
    }

    saveLayout() {
        const items = this.grid.save();
        const widgets = items.map((item, index) => ({
            id: item.el.dataset.widgetId || null,
            widget_type: item.el.dataset.widgetType,
            title: item.el.querySelector('.widget-title')?.textContent || 'Widget',
            position: index,
            width: item.w,
            height: item.h,
        }));

        fetch(this.saveEndpoint, {
            method: 'PUT',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify({
                layout: { columns: 12, rowHeight: 100, items },
                widgets,
            }),
        });
    }

    applyFilters(filters) {
        const event = new CustomEvent('dashboard:filters:changed', {
            detail: { filters, dashboardId: this.dashboardId },
        });
        document.dispatchEvent(event);

        // Refresh all widgets with new filters
        this.refreshAllWidgets();
    }

    refreshAllWidgets() {
        const widgets = this.grid.getGridItems();
        widgets.forEach(item => {
            this.refreshWidget(item);
        });
    }

    refreshWidget(gridItem) {
        const widgetId = gridItem.el.dataset.widgetId;
        const loadingEl = gridItem.el.querySelector('.widget-loading');
        const bodyEl = gridItem.el.querySelector('.widget-body');

        loadingEl.style.display = 'block';

        fetch(`/api/widgets/${widgetId}/data`, {
            headers: {
                'X-Widget-Filters': JSON.stringify(this.getActiveFilters()),
            }
        })
        .then(response => response.json())
        .then(data => {
            this.renderWidgetData(gridItem, data);
        })
        .finally(() => {
            loadingEl.style.display = 'none';
        });
    }

    renderWidgetLibrary() {
        const container = document.getElementById('widget-library');
        container.innerHTML = '';

        Object.entries(this.widgetTypes).forEach(([type, config]) => {
            const card = document.createElement('div');
            card.className = 'widget-library-item';
            card.innerHTML = `
                <div class="widget-icon">${config.icon}</div>
                <div class="widget-info">
                    <h6>${config.name}</h6>
                    <small>${config.description}</small>
                </div>
                <button class="btn btn-sm btn-primary add-widget-btn"
                        data-type="${type}">Add</button>
            `;

            card.querySelector('.add-widget-btn').addEventListener('click', (e) => {
                this.addNewWidget(type);
            });

            container.appendChild(card);
        });
    }

    addNewWidget(type) {
        const widget = {
            title: this.widgetTypes[type].name,
            widget_type: type,
            config: {},
        };

        fetch(`/api/dashboards/${this.dashboardId}/widgets`, {
            method: 'POST',
            headers: { 'Content-Type': 'application/json' },
            body: JSON.stringify(widget),
        })
        .then(response => response.json())
        .then(widget => {
            const content = document.createElement('div');
            content.className = 'grid-stack-item';
            content.innerHTML = this.renderWidget(widget, null);
            content.dataset.widgetId = widget.id;
            content.dataset.widgetType = type;

            this.grid.addWidget(content, {
                x: 0, y: 0,
                w: widget.width || 4,
                h: widget.height || 3,
            });

            this.saveLayout();
        });
    }
}

export default DashboardBuilder;
```
