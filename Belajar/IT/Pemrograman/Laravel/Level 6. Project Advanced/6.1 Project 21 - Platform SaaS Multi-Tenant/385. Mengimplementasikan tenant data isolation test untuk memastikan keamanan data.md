# 385. Mengimplementasikan Tenant Data Isolation Test untuk Memastikan Keamanan Data

## Definisi & Target
Tenant data isolation test memastikan bahwa data satu tenant tidak bisa diakses oleh tenant lain. Pengujian meliputi: database isolation, cache isolation, queue isolation, session isolation, dan file storage isolation. Targetnya adalah jaminan keamanan data antar tenant.

## Komponen yang Terlibat
- PHPUnit test suite khusus isolation
- TenantAwareTestCase base class
- Cross-tenant access test
- Database isolation assertions
- Cache isolation assertions
- Storage isolation assertions
- Queue isolation assertions

## Perintah Artisan CLI
```bash
php artisan make:test Tenant/TenantDataIsolationTest
php artisan make:test Tenant/TenantDatabaseIsolationTest
```

### Base Test Case - TenantIsolationTestCase
```php
<?php
namespace Tests;

use App\Models\Tenant;
use App\Models\User;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Illuminate\Foundation\Testing\TestCase as BaseTestCase;

abstract class TenantIsolationTestCase extends BaseTestCase
{
    use CreatesApplication, RefreshDatabase;

    protected Tenant $tenantA;
    protected Tenant $tenantB;
    protected User $userA;
    protected User $userB;

    protected function setUp(): void
    {
        parent::setUp();

        $this->artisan('db:seed', ['--class' => 'PlanSeeder']);

        $this->tenantA = Tenant::factory()->create([
            'subdomain' => 'tenant-a',
            'plan' => 'professional',
            'status' => 'active',
        ]);

        $this->tenantB = Tenant::factory()->create([
            'subdomain' => 'tenant-b',
            'plan' => 'starter',
            'status' => 'active',
        ]);

        $this->createUsersForTenants();
    }

    protected function createUsersForTenants(): void
    {
        $this->tenantA->run(function () {
            $this->userA = User::factory()->create([
                'email' => 'admin@tenanta.com',
                'name' => 'Admin Tenant A',
            ]);
        });

        $this->tenantB->run(function () {
            $this->userB = User::factory()->create([
                'email' => 'admin@tenantb.com',
                'name' => 'Admin Tenant B',
            ]);
        });
    }

    protected function actAsTenantA(): void
    {
        $this->tenantA->makeCurrent();
        $this->actingAs($this->userA);
    }

    protected function actAsTenantB(): void
    {
        $this->tenantB->makeCurrent();
        $this->actingAs($this->userB);
    }

    protected function assertTenantIsolation(callable $tenantAAction, callable $tenantBAction): void
    {
        $this->actAsTenantA();
        $resultA = $tenantAAction();

        $this->actAsTenantB();
        $resultB = $tenantBAction();

        $this->assertNotEquals(
            $resultA,
            $resultB,
            'Tenant isolation failed: data should not be identical across tenants.'
        );
    }
}
```

### Test - TenantDatabaseIsolationTest
```php
<?php
namespace Tests\Tenant;

use App\Models\Product;
use App\Models\Tenant;
use Tests\TenantIsolationTestCase;

class TenantDatabaseIsolationTest extends TenantIsolationTestCase
{
    public function test_tenant_a_cannot_access_tenant_b_data_via_model()
    {
        $this->actAsTenantA();
        Product::create(['name' => 'Product Tenant A', 'price' => 100]);

        $this->actAsTenantB();
        Product::create(['name' => 'Product Tenant B', 'price' => 200]);

        $this->actAsTenantA();
        $products = Product::all();

        $this->assertCount(1, $products);
        $this->assertEquals('Product Tenant A', $products->first()->name);
    }

    public function test_tenant_b_cannot_access_tenant_a_data_via_raw_query()
    {
        $this->actAsTenantA();
        Product::create(['name' => 'Secret A', 'price' => 999]);

        $this->actAsTenantB();
        $products = Product::where('name', 'Secret A')->get();

        $this->assertTrue($products->isEmpty());
    }

    public function test_database_connection_isolation()
    {
        $tenantAConnection = $this->tenantA->database;
        $tenantBConnection = $this->tenantB->database;

        $this->assertNotNull($tenantAConnection);
        $this->assertNotNull($tenantBConnection);
        $this->assertNotEquals($tenantAConnection, $tenantBConnection);
    }

    public function test_tenant_data_isolation_using_tenant_scope()
    {
        $this->actAsTenantA();
        $productA = Product::create(['name' => 'Exclusive A', 'price' => 500]);

        $this->actAsTenantB();
        $productB = Product::create(['name' => 'Exclusive B', 'price' => 600]);

        $this->actAsTenantA();
        $found = Product::find($productB->id);

        $this->assertNull($found, 'Tenant A should not find Tenant B product by ID.');
    }

    public function test_cross_tenant_foreign_key_violation()
    {
        $this->actAsTenantA();
        $productA = Product::create(['name' => 'Test', 'price' => 100]);

        $this->actAsTenantB();
        $productB = Product::create(['name' => 'Test', 'price' => 200]);

        $this->assertNotEquals($productA->id, $productB->id);
    }

    public function test_tenant_data_isolation_with_relationships()
    {
        $this->actAsTenantA();
        $category = \App\Models\Category::create(['name' => 'Category A']);
        $category->products()->create(['name' => 'Product A1', 'price' => 50]);

        $this->actAsTenantB();
        $categories = \App\Models\Category::with('products')->get();
        $this->assertTrue($categories->isEmpty());
    }

    public function test_bulk_insert_isolation()
    {
        $this->actAsTenantA();
        Product::insert([
            ['name' => 'Bulk A1', 'price' => 10, 'created_at' => now(), 'updated_at' => now()],
            ['name' => 'Bulk A2', 'price' => 20, 'created_at' => now(), 'updated_at' => now()],
        ]);

        $this->actAsTenantB();
        $count = Product::count();

        $this->assertEquals(0, $count, 'Tenant B should have no products after Tenant A bulk insert.');
    }
}
```

### Test - TenantCacheIsolationTest
```php
<?php
namespace Tests\Tenant;

use Illuminate\Support\Facades\Cache;
use Tests\TenantIsolationTestCase;

class TenantCacheIsolationTest extends TenantIsolationTestCase
{
    public function test_cache_isolation_between_tenants()
    {
        $this->actAsTenantA();
        Cache::put('test_key', 'value_a', 60);

        $this->actAsTenantB();
        Cache::put('test_key', 'value_b', 60);

        $this->actAsTenantA();
        $value = Cache::get('test_key');

        $this->assertEquals('value_a', $value);
        $this->assertNotEquals('value_b', $value);
    }

    public function test_cache_tag_isolation()
    {
        $this->actAsTenantA();
        Cache::tags(['tenant_' . $this->tenantA->id])->put('data', 'secret_a', 60);

        $this->actAsTenantB();
        $value = Cache::tags(['tenant_' . $this->tenantA->id])->get('data');

        $this->assertNull($value, 'Tenant B should not access Tenant A cache tag.');
    }

    public function test_tenant_specific_cache_prefix()
    {
        $this->actAsTenantA();
        Cache::put('user_preferences', 'theme_dark', 60);

        $this->actAsTenantB();
        $prefs = Cache::get('user_preferences');

        $this->assertNull($prefs, 'Tenant B should not access Tenant A cached preferences.');
    }

    public function test_cache_invalidation_per_tenant()
    {
        $this->actAsTenantA();
        Cache::put('shared_key', 'tenant_a_value', 60);

        $this->actAsTenantA();
        Cache::forget('shared_key');

        $this->actAsTenantB();
        Cache::put('shared_key', 'tenant_b_value', 60);

        $this->actAsTenantB();
        $value = Cache::get('shared_key');
        $this->assertEquals('tenant_b_value', $value);
    }

    public function test_rate_limiter_isolation()
    {
        $this->actAsTenantA();
        $key = 'rate_limit:api_' . $this->userA->id;
        Cache::add($key, 0, 60);
        Cache::increment($key);

        $this->actAsTenantB();
        $hits = Cache::get('rate_limit:api_' . $this->userB->id);

        $this->assertNull($hits, 'Rate limiter should be isolated per tenant user.');
    }
}
```

### Test - TenantStorageIsolationTest
```php
<?php
namespace Tests\Tenant;

use Illuminate\Http\UploadedFile;
use Illuminate\Support\Facades\Storage;
use Tests\TenantIsolationTestCase;

class TenantStorageIsolationTest extends TenantIsolationTestCase
{
    public function test_storage_directory_isolation()
    {
        Storage::fake('local');

        $this->actAsTenantA();
        Storage::put("tenants/{$this->tenantA->id}/document.txt", 'Tenant A confidential');

        $this->actAsTenantB();
        $exists = Storage::exists("tenants/{$this->tenantA->id}/document.txt");

        $this->assertFalse($exists, 'Tenant B should not access Tenant A storage directory.');
    }

    public function test_file_upload_isolation()
    {
        Storage::fake('public');

        $this->actAsTenantA();
        $file = UploadedFile::fake()->image('photo.jpg');
        Storage::putFile("uploads/{$this->tenantA->id}", $file);

        $this->actAsTenantB();
        $files = Storage::files("uploads/{$this->tenantA->id}");

        $this->assertEmpty($files, 'Tenant B should not see Tenant A uploaded files.');
    }

    public function test_storage_delete_isolation()
    {
        Storage::fake('local');

        $this->actAsTenantA();
        Storage::put("data/{$this->tenantA->id}/file.txt", 'data');

        $this->actAsTenantB();
        Storage::deleteDirectory("data/{$this->tenantA->id}");

        $this->actAsTenantA();
        $exists = Storage::exists("data/{$this->tenantA->id}/file.txt");

        $this->assertTrue($exists, 'Tenant B should not be able to delete Tenant A files.');
    }
}
```

### Test - TenantApiIsolationTest
```php
<?php
namespace Tests\Tenant;

use App\Models\Product;
use Tests\TenantIsolationTestCase;

class TenantApiIsolationTest extends TenantIsolationTestCase
{
    public function test_api_returns_only_tenant_a_data()
    {
        $this->actAsTenantA();
        Product::create(['name' => 'Item A', 'price' => 100]);

        $this->actAsTenantB();
        Product::create(['name' => 'Item B', 'price' => 200]);

        $this->actAsTenantA();
        $response = $this->getJson('/api/products');

        $response->assertOk();
        $response->assertJsonCount(1, 'data');
        $response->assertJsonFragment(['name' => 'Item A']);
        $response->assertJsonMissing(['name' => 'Item B']);
    }

    public function test_api_post_isolation()
    {
        $this->actAsTenantA();
        $this->postJson('/api/products', ['name' => 'New A', 'price' => 50]);

        $this->actAsTenantB();
        $this->postJson('/api/products', ['name' => 'New B', 'price' => 75]);

        $this->actAsTenantA();
        $response = $this->getJson('/api/products');

        $response->assertJsonFragment(['name' => 'New A']);
        $response->assertJsonMissing(['name' => 'New B']);
    }

    public function test_cross_tenant_api_access_returns_404()
    {
        $this->actAsTenantA();
        $product = Product::create(['name' => 'Secret', 'price' => 999]);

        $this->actAsTenantB();
        $response = $this->getJson("/api/products/{$product->id}");

        $response->assertNotFound();
    }

    public function test_api_authentication_isolation()
    {
        $this->actAsTenantA();
        $token = $this->userA->createToken('test')->plainTextToken;

        $this->actAsTenantB();
        $response = $this->withHeaders([
            'Authorization' => 'Bearer ' . $token,
        ])->getJson('/api/user');

        $response->assertUnauthorized();
    }
}
```

### Test - TenantSecurityIsolationTest
```php
<?php
namespace Tests\Tenant;

use Tests\TenantIsolationTestCase;

class TenantSecurityIsolationTest extends TenantIsolationTestCase
{
    public function test_session_isolation()
    {
        $this->actAsTenantA();
        session(['tenant_a_data' => 'confidential']);

        $this->actAsTenantB();
        $value = session('tenant_a_data');

        $this->assertNull($value, 'Session data should be isolated per tenant.');
    }

    public function test_queue_job_isolation()
    {
        $this->actAsTenantA();
        \App\Jobs\ProcessTenantData::dispatch('job_a');

        $this->actAsTenantB();
        $jobs = \Illuminate\Support\Facades\Queue::pushedJobs();

        $this->assertArrayNotHasKey(\App\Jobs\ProcessTenantData::class, $jobs);
    }

    public function test_log_isolation()
    {
        $this->actAsTenantA();
        \Illuminate\Support\Facades\Log::channel('tenant')->info('Tenant A log');

        $this->actAsTenantB();
        $logFile = storage_path("logs/tenant-{$this->tenantA->id}.log");

        $this->assertFileDoesNotExist($logFile);
    }

    public function test_config_isolation()
    {
        $this->actAsTenantA();
        config(['app.name' => 'Tenant A App']);

        $this->actAsTenantB();
        $appName = config('app.name');

        $this->assertNotEquals('Tenant A App', $appName);
    }
}
```

## Cara Integrasi & Pengujian
1. Buat Tenant factory dan User factory dengan tenant-aware logic
2. Setup database isolation: setiap tenant punya database sendiri
3. Jalankan php artisan test tests/Tenant/TenantDatabaseIsolationTest
4. Verifikasi semua test passing (hijau)
5. Test manual: login sebagai tenant A, coba akses /api/products/{id} milik tenant B
6. Pastikan return 404 atau 403
7. Test cache: set cache di tenant A, coba baca dari tenant B
8. Test storage: upload file di tenant A, coba akses dari tenant B
9. Test queue: dispatch job di tenant A, pastikan tidak dieksekusi di tenant B
10. Jalankan full test suite untuk memastikan tidak ada yang rusak
