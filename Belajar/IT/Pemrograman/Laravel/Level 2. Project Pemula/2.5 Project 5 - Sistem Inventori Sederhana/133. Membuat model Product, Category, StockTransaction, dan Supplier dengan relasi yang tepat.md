# 133. Membuat model Product, Category, StockTransaction, dan Supplier dengan relasi yang tepat

## Definisi & Target
Membuat Eloquent Models untuk Product, Category, StockTransaction, dan Supplier dengan relasi yang tepat. Product milik satu Category dan Supplier, StockTransaction mencatat pergerakan stok Product.

## Komponen yang Terlibat
- Model: `app/Models/Category.php`
- Model: `app/Models/Supplier.php`
- Model: `app/Models/Product.php`
- Model: `app/Models/StockTransaction.php`

## Perintah Artisan CLI
```
php artisan make:model Product
php artisan make:model Supplier
php artisan make:model StockTransaction
php artisan make:model Category
```

## Kode Implementasi (Per File)

**app/Models/Category.php**
```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Str;

class Category extends Model
{
    use HasFactory;

    protected $fillable = ['name', 'slug', 'description'];

    protected static function boot()
    {
        parent::boot();
        static::creating(fn ($category) => $category->slug = Str::slug($category->name));
    }

    public function products()
    {
        return $this->hasMany(Product::class);
    }
}
```

**app/Models/Product.php**
```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class Product extends Model
{
    use HasFactory;

    protected $fillable = [
        'category_id', 'name', 'sku', 'description',
        'stock', 'minimum_stock', 'purchase_price',
        'selling_price', 'barcode',
    ];

    protected $casts = [
        'purchase_price' => 'decimal:2',
        'selling_price' => 'decimal:2',
    ];

    public function category()
    {
        return $this->belongsTo(Category::class);
    }

    public function stockTransactions()
    {
        return $this->hasMany(StockTransaction::class);
    }

    public function scopeLowStock($query)
    {
        return $query->whereColumn('stock', '<=', 'minimum_stock');
    }

    public function scopeOutOfStock($query)
    {
        return $query->where('stock', 0);
    }

    public function isLowStock(): bool
    {
        return $this->stock <= $this->minimum_stock;
    }

    public function isOutOfStock(): bool
    {
        return $this->stock <= 0;
    }
}
```

**app/Models/StockTransaction.php**
```php
<?php

namespace App\Models;

use Illuminate\Database\Eloquent\Factories\HasFactory;
use Illuminate\Database\Eloquent\Model;

class StockTransaction extends Model
{
    use HasFactory;

    protected $fillable = [
        'product_id', 'supplier_id', 'type',
        'quantity', 'stock_before', 'stock_after',
        'note', 'reference',
    ];

    public function product()
    {
        return $this->belongsTo(Product::class);
    }

    public function supplier()
    {
        return $this->belongsTo(Supplier::class);
    }

    public function scopeIn($query)
    {
        return $query->where('type', 'in');
    }

    public function scopeOut($query)
    {
        return $query->where('type', 'out');
    }
}
```
