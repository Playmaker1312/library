# 145. Memastikan semua fitur inventori berjalan akurat dan data selalu konsisten

## Definisi & Target
Memastikan seluruh fitur sistem inventori berjalan dengan akurat, data stok konsisten, transaksi tercatat dengan benar, dan tidak ada inkonsistensi data. Mengimplementasikan validasi dan pengecekan data.

## Komponen yang Terlibat
- ProductObserver (memastikan transaksi selalu sinkron)
- StockTransaction (validasi data)
- Unit test / pengecekan manual


## Kode Implementasi (Per File)

**app/Observers/ProductObserver.php (memastikan konsistensi)**
```php
<?php

namespace App\Observers;

use App\Models\Product;
use App\Models\StockTransaction;
use Illuminate\Support\Facades\DB;

class ProductObserver
{
    public function created(Product $product): void
    {
        if ($product->stock > 0) {
            DB::transaction(function () use ($product) {
                $product->stockTransactions()->create([
                    'type' => 'in',
                    'quantity' => $product->stock,
                    'stock_before' => 0,
                    'stock_after' => $product->stock,
                    'note' => 'Stok awal',
                    'reference' => 'INIT-' . $product->sku,
                ]);
            });
        }
    }

    public function updated(Product $product): void
    {
        if ($product->isDirty('stock')) {
            $oldStock = $product->getOriginal('stock');
            $newStock = $product->stock;
            $diff = $newStock - $oldStock;

            if ($diff !== 0) {
                DB::transaction(function () use ($product, $oldStock, $newStock, $diff) {
                    $product->stockTransactions()->create([
                        'type' => $diff > 0 ? 'in' : 'out',
                        'quantity' => abs($diff),
                        'stock_before' => $oldStock,
                        'stock_after' => $newStock,
                        'note' => 'Penyesuaian stok manual',
                        'reference' => 'ADJ-' . $product->sku . '-' . time(),
                    ]);
                });
            }
        }
    }

    public function deleted(Product $product): void
    {
        $product->stockTransactions()->delete();
    }
}
```

**database/seeders/ConsistencyChecker.php (pengecekan data)**
```php
<?php

namespace Database\Seeders;

use App\Models\Product;
use App\Models\StockTransaction;
use Illuminate\Database\Seeder;

class ConsistencyChecker extends Seeder
{
    public function run(): void
    {
        $products = Product::all();
        $issues = [];

        foreach ($products as $product) {
            $lastTransaction = $product->stockTransactions()
                ->latest()
                ->first();

            if ($lastTransaction && $lastTransaction->stock_after !== $product->stock) {
                $issues[] = "Produk {$product->name}: stok saat ini {$product->stock} tidak cocok dengan transaksi terakhir {$lastTransaction->stock_after}";
            }

            $calculatedStock = $product->stockTransactions()
                ->selectRaw('SUM(CASE WHEN type = "in" THEN quantity ELSE 0 END) - SUM(CASE WHEN type = "out" THEN quantity ELSE 0 END) as calculated')
                ->value('calculated');

            if ($calculatedStock !== null && (int) $calculatedStock !== $product->stock) {
                $issues[] = "Produk {$product->name}: stok {$product->stock} tidak cocok dengan kalkulasi transaksi {$calculatedStock}";
            }
        }

        if (count($issues) > 0) {
            echo "Ditemukan " . count($issues) . " masalah konsistensi:\n";
            foreach ($issues as $issue) {
                echo "- $issue\n";
            }
        } else {
            echo "Semua data konsisten. Tidak ada masalah ditemukan.\n";
        }

        echo "\nStatistik:\n";
        echo "Total produk: " . Product::count() . "\n";
        echo "Total transaksi: " . StockTransaction::count() . "\n";
        echo "Total stok: " . Product::sum('stock') . "\n";
    }
}
```

**Unit test sederhana (tests/Feature/InventoryConsistencyTest.php)**
```php
<?php

namespace Tests\Feature;

use App\Models\Category;
use App\Models\Product;
use App\Models\StockTransaction;
use Illuminate\Foundation\Testing\RefreshDatabase;
use Tests\TestCase;

class InventoryConsistencyTest extends TestCase
{
    use RefreshDatabase;

    public function test_product_stock_matches_transactions(): void
    {
        $category = Category::factory()->create();
        $product = Product::create([
            'category_id' => $category->id,
            'name' => 'Test Product',
            'sku' => 'TST-001',
            'stock' => 50,
            'minimum_stock' => 5,
        ]);

        $this->assertEquals(50, $product->stock);
        $this->assertEquals(1, $product->stockTransactions()->count());

        $transaction = $product->stockTransactions()->first();
        $this->assertEquals('in', $transaction->type);
        $this->assertEquals(50, $transaction->quantity);
        $this->assertEquals(0, $transaction->stock_before);
        $this->assertEquals(50, $transaction->stock_after);
    }

    public function test_stock_out_validation(): void
    {
        $category = Category::factory()->create();
        $product = Product::create([
            'category_id' => $category->id,
            'name' => 'Test Product',
            'sku' => 'TST-002',
            'stock' => 10,
            'minimum_stock' => 2,
        ]);

        $response = $this->post(route('stock-transactions.store-out'), [
            'product_id' => $product->id,
            'quantity' => 20,
        ]);

        $response->assertSessionHasErrors('quantity');
        $this->assertEquals(10, $product->fresh()->stock);
    }
}
```
