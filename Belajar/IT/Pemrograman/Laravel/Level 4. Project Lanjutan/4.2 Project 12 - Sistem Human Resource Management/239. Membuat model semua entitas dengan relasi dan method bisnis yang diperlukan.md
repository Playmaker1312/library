# 239. Membuat Model Semua Entitas dengan Relasi dan Method Bisnis yang Diperlukan

## Definisi & Target
Membuat model Eloquent untuk setiap entitas HRM dengan relasi yang tepat (belongsTo, hasMany, hasManyThrough) serta method bisnis seperti perhitungan gaji, sisa cuti, dan jam kerja.

## Komponen yang Terlibat
- Model Department, Position, Employee, Attendance, Leave, Payroll

## Perintah Artisan CLI
- `php artisan make:model Department`
- `php artisan make:model Position`
- `php artisan make:model Employee`
- `php artisan make:model Attendance`
- `php artisan make:model Leave`
- `php artisan make:model Payroll`

## Kode Implementasi (Per File)

**app/Models/Department.php:**
```php
class Department extends Model
{
    use HasFactory;

    protected $fillable = ['name', 'code', 'description', 'parent_id', 'is_active'];

    public function parent(): BelongsTo
    {
        return $this->belongsTo(Department::class, 'parent_id');
    }

    public function children(): HasMany
    {
        return $this->hasMany(Department::class, 'parent_id');
    }

    public function positions(): HasMany
    {
        return $this->hasMany(Position::class);
    }

    public function employees(): HasMany
    {
        return $this->hasMany(Employee::class);
    }
}
```

**app/Models/Employee.php:**
```php
class Employee extends Model
{
    use HasFactory;

    protected $fillable = ['user_id', 'position_id', 'department_id', 'employee_id', 'name', 'email', 'phone', 'address', 'birth_date', 'gender', 'photo', 'join_date', 'status', 'basic_salary'];

    protected $casts = ['birth_date' => 'date', 'join_date' => 'date'];

    public function user(): BelongsTo
    {
        return $this->belongsTo(User::class);
    }

    public function position(): BelongsTo
    {
        return $this->belongsTo(Position::class);
    }

    public function department(): BelongsTo
    {
        return $this->belongsTo(Department::class);
    }

    public function attendances(): HasMany
    {
        return $this->hasMany(Attendance::class);
    }

    public function leaves(): HasMany
    {
        return $this->hasMany(Leave::class);
    }

    public function payrolls(): HasMany
    {
        return $this->hasMany(Payroll::class);
    }

    public function getRemainingLeaveAttribute()
    {
        $totalDays = 12;
        $usedDays = Leave::where('employee_id', $this->id)
            ->where('type', 'annual')
            ->where('status', 'approved')
            ->whereYear('start_date', date('Y'))
            ->sum(DB::raw('DATEDIFF(end_date, start_date) + 1'));

        return max(0, $totalDays - $usedDays);
    }

    public function scopeActive($query)
    {
        return $query->where('status', 'active');
    }
}
```

```php
class Attendance extends Model

    protected $casts = ['clock_in' => 'datetime', 'clock_out' => 'datetime', 'date' => 'date'];

    public function employee(): BelongsTo
    {
        return $this->belongsTo(Employee::class);
    }
}
```

**app/Models/Leave.php:**
```php
class Leave extends Model
{
    use HasFactory;

    protected $fillable = ['employee_id', 'type', 'start_date', 'end_date', 'reason', 'status', 'approved_by', 'approved_at', 'rejection_reason'];

    protected $casts = ['start_date' => 'date', 'end_date' => 'date', 'approved_at' => 'datetime'];

    public function employee(): BelongsTo
    {
        return $this->belongsTo(Employee::class);
    }

    public function approver(): BelongsTo
    {
        return $this->belongsTo(Employee::class, 'approved_by');
    }

    public function getDurationAttribute()
    {
        return $this->start_date->diffInDays($this->end_date) + 1;
    }
}
```

**app/Models/Payroll.php:**
```php
class Payroll extends Model
{
    use HasFactory;

    protected $fillable = ['employee_id', 'period', 'basic_salary', 'allowances', 'deductions', 'overtime_pay', 'total', 'status', 'paid_at'];

    protected $casts = ['paid_at' => 'datetime'];

    public function employee(): BelongsTo
    {
        return $this->belongsTo(Employee::class);
    }
}
```
