# 250. Mengimplementasikan Export Data HRM ke Excel Menggunakan Laravel Excel

## Definisi & Target
Membangun fitur export data HRM (karyawan, absensi, cuti, payroll) ke format Excel (.xlsx) menggunakan library Laravel Excel.

## Komponen yang Terlibat
- Export Classes: `EmployeesExport`, `AttendancesExport`, `PayrollsExport`
- Controller: `ExportController`
- Library: maatwebsite/laravel-excel

## Perintah Artisan CLI
- `composer require maatwebsite/laravel-excel`
- `php artisan make:export EmployeesExport --model=Employee`
- `php artisan make:export AttendancesExport --model=Attendance`
- `php artisan make:export PayrollsExport --model=Payroll`

## Kode Implementasi (Per File)

**app/Exports/EmployeesExport.php:**
```php
class EmployeesExport implements FromQuery, WithHeadings, WithMapping, WithStyles
{
    public function query()
    {
        return Employee::query()->with('department', 'position');
    }

    public function headings(): array
    {
        return [
            'NIK', 'Nama', 'Email', 'Telepon', 'Departemen', 'Jabatan',
            'Tanggal Masuk', 'Status', 'Gaji Pokok', 'Jenis Kelamin'
        ];
    }

    public function map($employee): array
    {
        return [
            $employee->employee_id,
            $employee->name,
            $employee->email,
            $employee->phone,
            $employee->department->name ?? '-',
            $employee->position->name ?? '-',
            $employee->join_date->format('d/m/Y'),
            $employee->status,
            $employee->basic_salary,
            $employee->gender === 'male' ? 'Laki-laki' : 'Perempuan',
        ];
    }

    public function styles(Worksheet $sheet)
    {
        return [
            1 => ['font' => ['bold' => true, 'size' => 12], 'fill' => ['fillType' => 'solid', 'startColor' => ['rgb' => 'E2E8F0']]],
        ];
    }
}
```

**app/Exports/AttendancesExport.php:**
```php
class AttendancesExport implements FromQuery, WithHeadings, WithMapping
{
    public function __construct(
        public ?string $startDate = null,
        public ?string $endDate = null,
        public ?int $departmentId = null
    ) {}

    public function query()
    {
        return Attendance::query()
            ->with('employee.department')
            ->when($this->startDate, fn($q) => $q->whereDate('date', '>=', $this->startDate))
            ->when($this->endDate, fn($q) => $q->whereDate('date', '<=', $this->endDate))
            ->when($this->departmentId, fn($q) => $q->whereHas('employee', fn($qe) => $qe->where('department_id', $this->departmentId)))
            ->orderBy('date');
    }

    public function headings(): array
    {
        return ['NIK', 'Nama', 'Departemen', 'Tanggal', 'Clock In', 'Clock Out', 'Jam Kerja', 'Lembur', 'Status'];
    }

    public function map($attendance): array
    {
        return [
            $attendance->employee->employee_id,
            $attendance->employee->name,
            $attendance->employee->department->name ?? '-',
            $attendance->date,
            $attendance->clock_in ? \Carbon\Carbon::parse($attendance->clock_in)->format('H:i') : '-',
            $attendance->clock_out ? \Carbon\Carbon::parse($attendance->clock_out)->format('H:i') : '-',
            $attendance->work_hours,
            $attendance->overtime_hours,
            $attendance->status,
        ];
    }
}
```

**app/Exports/PayrollsExport.php:**
```php
class PayrollsExport implements FromQuery, WithHeadings, WithMapping
{
    public function __construct(public ?string $period = null) {}

    public function query()
    {
        return Payroll::query()
            ->with('employee.department')
            ->when($this->period, fn($q) => $q->where('period', $this->period))
            ->orderBy('period');
    }

    public function headings(): array
    {
        return ['NIK', 'Nama', 'Departemen', 'Periode', 'Gaji Pokok', 'Tunjangan', 'Lembur', 'Potongan', 'Total', 'Status'];
    }

    public function map($payroll): array
    {
        return [
            $payroll->employee->employee_id,
            $payroll->employee->name,
            $payroll->employee->department->name ?? '-',
            $payroll->period,
            $payroll->basic_salary,
            $payroll->allowances,
            $payroll->overtime_pay,
            $payroll->deductions,
            $payroll->total,
            $payroll->status,
        ];
    }
}
```

**app/Http/Controllers/ExportController.php:**
```php
class ExportController extends Controller
{
    public function employees()
    {
        return Excel::download(new EmployeesExport, 'data-karyawan-' . date('Y-m-d') . '.xlsx');
    }

    public function attendances(Request $request)
    {
        $export = new AttendancesExport(
            startDate: $request->get('start_date', now()->startOfMonth()->format('Y-m-d')),
            endDate: $request->get('end_date', now()->format('Y-m-d')),
            departmentId: $request->get('department_id'),
        );

        return Excel::download($export, 'data-absensi-' . date('Y-m-d') . '.xlsx');
    }

    public function payrolls(Request $request)
    {
        $period = $request->get('period', now()->format('Y-m'));

        return Excel::download(new PayrollsExport($period), 'data-payroll-' . $period . '.xlsx');
    }
}
```
