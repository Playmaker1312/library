# 247. Membangun Modul Penggajian dengan Perhitungan Gaji Dasar, Tunjangan, dan Potongan

## Definisi & Target
Membangun modul penggajian yang menghitung gaji karyawan berdasarkan gaji dasar, tunjangan (makan, transport, jabatan), potongan (BPJS, PPh 21, absensi), dan lembur.

## Komponen yang Terlibat
- Livewire Component: `PayrollGenerator`
- Model: Payroll, Employee, Attendance
- Service: `PayrollCalculationService`

## Perintah Artisan CLI
- `php artisan make:livewire PayrollGenerator`

## Kode Implementasi (Per File)

**app/Services/PayrollCalculationService.php:**
```php
class PayrollCalculationService
{
    private const BPJS_RATE = 0.02;
    private const PPH21_TAX_RATE = 0.05;

    public function calculate(Employee $employee, string $period): array
    {
        $basicSalary = $employee->basic_salary;
        $dailyRate = $basicSalary / 25;
        $hourlyRate = $dailyRate / 8;

        $allowances = $this->calculateAllowances($employee);
        $attendanceSummary = $this->getAttendanceSummary($employee->id, $period);
        $overtimePay = $this->calculateOvertimePay($attendanceSummary['total_overtime_hours'], $hourlyRate);
        $deductions = $this->calculateDeductions($basicSalary, $attendanceSummary['total_absent_days'], $dailyRate);

        $grossSalary = $basicSalary + array_sum($allowances) + $overtimePay;
        $netSalary = $grossSalary - array_sum($deductions);

        return [
            'basic_salary' => $basicSalary,
            'allowances' => $allowances,
            'allowances_total' => array_sum($allowances),
            'overtime_hours' => $attendanceSummary['total_overtime_hours'],
            'overtime_pay' => $overtimePay,
            'deductions' => $deductions,
            'deductions_total' => array_sum($deductions),
            'gross_salary' => $grossSalary,
            'net_salary' => $netSalary,
            'attendance' => $attendanceSummary,
        ];
    }

    private function calculateAllowances(Employee $employee): array
    {
        return [
            'makan' => 500000,
            'transport' => 300000,
            'jabatan' => $employee->position && $employee->position->level >= 3 ? 1000000 : 500000,
        ];
    }

    private function calculateOvertimePay(float $overtimeHours, float $hourlyRate): float
    {
        $overtimeRate = 1.5;
        $totalPay = 0;
        for ($i = 1; $i <= $overtimeHours; $i++) {
            $totalPay += $i === 1 ? $hourlyRate * $overtimeRate : $hourlyRate * 2;
        }
        return $totalPay;
    }

    private function calculateDeductions(float $basicSalary, int $absentDays, float $dailyRate): array
    {
        $bpjs = $basicSalary * self::BPJS_RATE;
        $pph21 = $basicSalary > 4500000 ? ($basicSalary - 4500000) * self::PPH21_TAX_RATE : 0;
        $absentDeduction = $absentDays * $dailyRate;

        return [
            'bpjs' => $bpjs,
            'pph21' => $pph21,
            'absensi' => $absentDeduction,
        ];
    }

    private function getAttendanceSummary(int $employeeId, string $period): array
    {
        $year = substr($period, 0, 4);
        $month = substr($period, 5, 2);

        $attendances = Attendance::where('employee_id', $employeeId)
            ->whereYear('date', $year)->whereMonth('date', $month)
            ->get();

        return [
            'total_present' => $attendances->where('status', 'present')->count(),
            'total_late' => $attendances->where('status', 'late')->count(),
            'total_absent_days' => $attendances->where('status', 'absent')->count(),
            'total_overtime_hours' => $attendances->sum('overtime_hours'),
        ];
    }
}
```

**app/Livewire/PayrollGenerator.php:**
```php
class PayrollGenerator extends Component
{
    public $period;
    public $employees = [];
    public $selectedEmployees = [];
    public $calculations = [];
    public $generating = false;

    public function mount()
    {
        $this->period = now()->format('Y-m');
        $this->loadEmployees();
    }

    public function loadEmployees()
    {
        $this->employees = Employee::active()
            ->with('department', 'position')
            ->get()
            ->toArray();
    }

    public function preview()
    {
        $service = app(PayrollCalculationService::class);

        foreach ($this->selectedEmployees as $employeeId) {
            $employee = Employee::find($employeeId);
            if ($employee) {
                $this->calculations[$employeeId] = $service->calculate($employee, $this->period);
            }
        }
    }

    public function generate()
    {
        $service = app(PayrollCalculationService::class);

        foreach ($this->selectedEmployees as $employeeId) {
            $employee = Employee::find($employeeId);
            if (!$employee) continue;

            $calc = $service->calculate($employee, $this->period);

            Payroll::updateOrCreate(
                ['employee_id' => $employeeId, 'period' => $this->period],
                [
                    'basic_salary' => $calc['basic_salary'],
                    'allowances' => $calc['allowances_total'],
                    'deductions' => $calc['deductions_total'],
                    'overtime_pay' => $calc['overtime_pay'],
                    'total' => $calc['net_salary'],
                    'status' => 'draft',
                ]
            );
        }

        $this->dispatch('payrollGenerated');
    }

    public function updatedPeriod()
    {
        $this->calculations = [];
        $this->selectedEmployees = [];
    }

    public function render()
    {
        return view('livewire.payroll-generator', [
            'existingPayrolls' => Payroll::where('period', $this->period)
                ->with('employee.department')->get(),
        ]);
    }
}
```
