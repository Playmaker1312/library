# 272. Mengimplementasikan Task dengan Subtask, Prioritas, Label, dan Assignee

## Definisi & Target
Membangun fitur task lengkap dengan subtask (parent-child), prioritas (urgent, high, medium, low), label (tag berwarna), dan assignee (penanggung jawab).

## Komponen yang Terlibat
- Livewire Component: `TaskForm`
- Model: Task, Label

## Perintah Artisan CLI
- `php artisan make:livewire TaskForm`

## Kode Implementasi (Per File)

**app/Livewire/TaskForm.php:**
```php
class TaskForm extends Component
{
    public $projectId;
    public $title;
    public $description;
    public $status = 'todo';
    public $priority = 'medium';
    public $assigneeId;
    public $dueDate;
    public $startDate;
    public $selectedLabels = [];
    public $subtasks = [];
    public $parentId;
    public $showForm = false;

    public function mount($projectId, $parentId = null)
    {
        $this->projectId = $projectId;
        $this->parentId = $parentId;
    }

    public function addSubtaskInput()
    {
        $this->subtasks[] = ['title' => '', 'done' => false];
    }

    public function removeSubtaskInput($index)
    {
        unset($this->subtasks[$index]);
        $this->subtasks = array_values($this->subtasks);
    }

    public function save()
    {
        $this->validate([
            'title' => 'required|string|max:255',
            'description' => 'nullable|string',
            'priority' => 'required|in:low,medium,high,urgent',
            'status' => 'required|in:todo,in_progress,review,done',
            'dueDate' => 'nullable|date|after_or_equal:today',
            'selectedLabels' => 'nullable|array',
        ]);

        DB::transaction(function () {
            $task = Task::create([
                'project_id' => $this->projectId,
                'parent_id' => $this->parentId,
                'workspace_id' => session('current_workspace_id'),
                'title' => $this->title,
                'description' => $this->description,
                'status' => $this->status,
                'priority' => $this->priority,
                'assignee_id' => $this->assigneeId ?: null,
                'start_date' => $this->startDate,
                'due_date' => $this->dueDate,
            ]);

            if (!empty($this->selectedLabels)) {
                $task->labels()->sync($this->selectedLabels);
            }

            foreach ($this->subtasks as $subtaskData) {
                if (!empty($subtaskData['title'])) {
                    Task::create([
                        'project_id' => $this->projectId,
                        'parent_id' => $task->id,
                        'workspace_id' => session('current_workspace_id'),
                        'title' => $subtaskData['title'],
                        'status' => 'todo',
                        'priority' => $this->priority,
                        'assignee_id' => $this->assigneeId ?: null,
                        'due_date' => $this->dueDate,
                    ]);
                }
            }
        });

        $this->dispatch('taskSaved');
    }

    public function render()
    {
        $members = WorkspaceMember::where('workspace_id', session('current_workspace_id'))
            ->with('user')->get();

        $labels = Label::where('workspace_id', session('current_workspace_id'))->get();

        return view('livewire.task-form', compact('members', 'labels'));
    }
}
```

- Tambah beberapa subtask
- Pilih label yang sudah dibuat
- Simpan task -> muncul di daftar task/Kanban
- Verifikasi task parent dan subtask (subtask punya parent_id)
- Label tersimpan di tabel pivot task_label
- Assignee terlihat di card task
