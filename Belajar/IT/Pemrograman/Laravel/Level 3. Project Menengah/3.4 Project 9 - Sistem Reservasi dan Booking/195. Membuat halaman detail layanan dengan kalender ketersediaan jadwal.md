# 195. Membuat halaman detail layanan dengan kalender ketersediaan jadwal

## Definisi & Target
Membuat halaman detail layanan yang menampilkan kalender ketersediaan slot waktu.

Target: User melihat jadwal yang tersedia dan memilih tanggal.

## Komponen yang Terlibat
- app/Http/Controllers/ServiceController (show)
- resources/views/services/show.blade.php

## Perintah Artisan CLI
Tidak ada perintah khusus.

## Kode Implementasi (Per File)

### Controller
File: app/Http/Controllers/ServiceController.php
```php
<?php

namespace App\Http\Controllers;

use App\Models\Service;
use Carbon\Carbon;
use Illuminate\Http\Request;
use Illuminate\View\View;

class ServiceController extends Controller
{
    public function index(): View
    {
        $services = Service::where('is_active', true)->get();
        return view('services.index', compact('services'));
    }

    public function show(Service $service): View
    {
        return view('services.show', compact('service'));
    }

    public function availability(Request $request, Service $service)
    {
        $date = $request->date ? Carbon::parse($request->date) : Carbon::today();

        $slots = $service->timeSlots()
            ->whereDate('date', $date)
            ->where('is_available', true)
            ->orderBy('start_time')
            ->get()
            ->filter(function ($slot) {
                return $slot->has_available_capacity;
            });

        return response()->json($slots->values());
    }
}
```

### Blade View
File: resources/views/services/show.blade.php
```blade
<x-app-layout>
    <div class="py-12">
        <div class="max-w-4xl mx-auto sm:px-6 lg:px-8">
            <div class="bg-white overflow-hidden shadow-sm sm:rounded-lg p-6">
                <div class="grid grid-cols-1 md:grid-cols-2 gap-8">
                    <div>
                        @if ($service->image)
                            <img src="{{ asset('storage/' . $service->image) }}" alt="{{ $service->name }}"
                                class="w-full rounded-lg">
                        @endif
                        <h1 class="text-3xl font-bold mt-4 mb-2">{{ $service->name }}</h1>
                        <p class="text-gray-600 mb-4">{{ $service->description }}</p>
                        <div class="flex items-center gap-4 text-sm text-gray-500">
                            <span class="text-2xl font-bold text-indigo-600">Rp {{ number_format($service->price, 0, ',', '.') }}</span>
                            <span>{{ $service->duration_minutes }} minutes</span>
                        </div>
                    </div>

                    <div>
                        <h2 class="text-xl font-semibold mb-4">Book Appointment</h2>

                        <div class="mb-4">
                            <label class="block text-sm font-medium text-gray-700 mb-1">Select Date</label>
                            <input type="date" id="booking-date" min="{{ Carbon::today()->format('Y-m-d') }}"
                                class="w-full rounded-md border-gray-300 shadow-sm focus:border-indigo-500 focus:ring-indigo-500">
                        </div>

                        <div id="time-slots" class="space-y-2">
                            <p class="text-gray-500 text-sm">Select a date to see available slots.</p>
                        </div>
                    </div>
                </div>
            </div>
        </div>
    </div>

    @push('scripts')
    <script>
        document.getElementById('booking-date').addEventListener('change', function() {
            const date = this.value;
            const container = document.getElementById('time-slots');

            if (!date) return;

            fetch(`/services/{{ $service->id }}/availability?date=${date}`)
                .then(res => res.json())
                .then(slots => {
                    if (slots.length === 0) {
                        container.innerHTML = '<p class="text-gray-500 text-sm">No available slots for this date.</p>';
                        return;
                    }

                    container.innerHTML = slots.map(slot => `
                        <form action="/bookings" method="POST" class="inline">
                            @csrf
                            <input type="hidden" name="service_id" value="{{ $service->id }}">
                            <input type="hidden" name="time_slot_id" value="${slot.id}">
                            <button type="submit"
                                class="w-full text-left p-3 border rounded-lg hover:bg-indigo-50 transition flex justify-between items-center">
                                <span>${slot.formatted_time}</span>
                                <span class="text-sm text-indigo-600 font-medium">Book</span>
                            </button>
                        </form>
                    `).join('');
                });
        });
    </script>
    @endpush
</x-app-layout>
```
