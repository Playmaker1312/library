# 179. Membuat CourseController untuk area publik dan admin

## Definisi & Target
Membuat controller untuk mengelola courses, dengan method untuk publik (index, show) dan admin (CRUD).

Target: Controller lengkap untuk operasi Course.

## Komponen yang Terlibat
- app/Http/Controllers/CourseController.php
- app/Http/Requests/CourseRequest.php

## Perintah Artisan CLI
```bash
php artisan make:controller CourseController --resource
php artisan make:request CourseRequest
```

## Kode Implementasi (Per File)

### Controller
File: app/Http/Controllers/CourseController.php
```php
<?php

namespace App\Http\Controllers;

use App\Http\Requests\CourseRequest;
use App\Models\Course;
use Illuminate\Http\RedirectResponse;
use Illuminate\Http\Request;
use Illuminate\View\View;

class CourseController extends Controller
{
    public function index(Request $request): View
    {
        $query = Course::with('author')->where('is_published', true);

        if ($request->filled('category')) {
            $query->where('category', $request->category);
        }

        if ($request->filled('level')) {
            $query->where('level', $request->level);
        }

        $courses = $query->paginate(12);

        return view('courses.index', compact('courses'));
    }

    public function show(Course $course): View
    {
        $course->load(['lessons', 'author']);

        $isEnrolled = auth()->check() && $course->enrollments()
            ->where('user_id', auth()->id())
            ->exists();

        return view('courses.show', compact('course', 'isEnrolled'));
    }

    public function create(): View
    {
        return view('courses.create');
    }

    public function store(CourseRequest $request): RedirectResponse
    {
        $course = auth()->user()->courses()->create($request->validated());

        return redirect()->route('courses.edit', $course)->with('success', 'Course created.');
    }

    public function edit(Course $course): View
    {
        return view('courses.edit', compact('course'));
    }

    public function update(CourseRequest $request, Course $course): RedirectResponse
    {
        $course->update($request->validated());

        return redirect()->route('courses.index')->with('success', 'Course updated.');
    }

    public function destroy(Course $course): RedirectResponse
    {
        $course->delete();

        return redirect()->route('courses.index')->with('success', 'Course deleted.');
    }
}
```

### Request
File: app/Http/Requests/CourseRequest.php
```php
<?php

namespace App\Http\Requests;

use Illuminate\Foundation\Http\FormRequest;

class CourseRequest extends FormRequest
{
    public function authorize(): bool
    {
        return true;
    }

    public function rules(): array
    {
        return [
            'title' => 'required|string|max:255',
            'description' => 'nullable|string',
            'category' => 'nullable|string|max:100',
            'level' => 'required|string|in:beginner,intermediate,advanced',
            'thumbnail' => 'nullable|image|max:2048',
            'duration_minutes' => 'nullable|integer|min:0',
            'is_published' => 'boolean',
        ];
    }
}
```

### Routes
File: routes/web.php
```php
Route::get('courses', [CourseController::class, 'index'])->name('courses.index');
Route::get('courses/{course}', [CourseController::class, 'show'])->name('courses.show');

Route::middleware(['auth', 'verified', 'role:super-admin|admin'])->group(function () {
    Route::get('courses/create', [CourseController::class, 'create'])->name('courses.create');
    Route::post('courses', [CourseController::class, 'store'])->name('courses.store');
    Route::get('courses/{course}/edit', [CourseController::class, 'edit'])->name('courses.edit');
    Route::put('courses/{course}', [CourseController::class, 'update'])->name('courses.update');
    Route::delete('courses/{course}', [CourseController::class, 'destroy'])->name('courses.destroy');
});
```

## Cara Integrasi & Pengujian
- Akses `/courses` untuk melihat daftar kursus publik.
- Klik kursus untuk melihat detail.
- Login sebagai admin, CRUD kursus.
- Filter berdasarkan kategori dan level.
