# 487. Laravel Dusk Browser Testing

## Definisi Singkat
Package browser testing untuk Laravel yang menggunakan ChromeDriver — memungkinkan testing interaksi pengguna di browser secara otomatis.

## Analogi
Seperti robot yang mengoperasikan komputer — robot membuka browser, klik tombol, isi form, screenshot, dan lapor hasil. Tanpa manusia duduk di depan layar.

## Poin-Poin Penting
- Testing langsung di Chrome browser sungguhan
- Interaksi: klik, ketik, scroll, upload file
- Assertions: lihat teks, URL, elemen
- Screenshot otomatis saat gagal
- Dapat digunakan dengan JavaScript SPA
- Headless mode untuk CI/CD

## Contoh Kode
```php
<?php

// Install
composer require laravel/dusk --dev
php artisan dusk:install

// Dasar test
namespace Tests\Browser;

use App\Models\User;
use Laravel\Dusk\Browser;
use Tests\DuskTestCase;

class LoginTest extends DuskTestCase
{
    public function test_user_can_login(): void
    {
        $user = User::factory()->create([
            'email' => 'test@example.com',
            'password' => bcrypt('password'),
        ]);

        $this->browse(function (Browser $browser) use ($user) {
            $browser->visit('/login')
                ->type('email', $user->email)
                ->type('password', 'password')
                ->press('Login')
                ->assertPathIs('/dashboard')
                ->assertSee('Dashboard')
                ->screenshot('login-success');
        });
    }
}

// Register test
public function test_user_can_register(): void
{
    $this->browse(function (Browser $browser) {
        $browser->visit('/register')
            ->type('name', 'New User')
            ->type('email', 'new@example.com')
            ->type('password', 'password')
            ->type('password_confirmation', 'password')
            ->press('Register')
            ->assertPathIs('/dashboard')
            ->assertSee('New User');
    });
}

// CRUD test
class PostTest extends DuskTestCase
{
    public function test_create_post(): void
    {
        $user = User::factory()->create();

        $this->browse(function (Browser $browser) use ($user) {
            $browser->loginAs($user)
                ->visit('/posts/create')
                ->type('title', 'My New Post')
                ->type('content', 'This is the content of my post')
                ->press('Save')
                ->assertSee('Post created!')
                ->assertSee('My New Post');
        });
    }

    public function test_search_post(): void
    {
        $user = User::factory()->create();

        $this->browse(function (Browser $browser) use ($user) {
            $browser->loginAs($user)
                ->visit('/posts')
                ->type('search', 'Laravel')
                ->waitForText('Laravel')
                ->assertSee('Laravel')
                ->assertDontSee('Vue.js');
        });
    }
}

// Testing JavaScript / Livewire
public function test_livewire_component(): void
{
    $this->browse(function (Browser $browser) {
        $browser->visit('/counter')
            ->assertSee('0')
            ->press('+')
            ->waitForText('1')
            ->assertSee('1')
            ->press('+')
            ->waitForText('2')
            ->assertSee('2');
    });
}

// Testing file upload
public function test_upload_avatar(): void
{
    $this->browse(function (Browser $browser) {
        $browser->visit('/profile')
            ->attach('photo', __DIR__ . '/test-avatar.jpg')
            ->press('Upload')
            ->waitForText('Upload successful')
            ->assertSee('Upload successful');
    });
}

// Multiple browser interaction (chat)
public function test_realtime_chat(): void
{
    $userA = User::factory()->create();
    $userB = User::factory()->create();

    $this->browse(function (Browser $first, Browser $second) use ($userA, $userB) {
        $first->loginAs($userA)
            ->visit('/chat')
            ->type('message', 'Hello from User A')
            ->press('Send');

        $second->loginAs($userB)
            ->visit('/chat')
            ->waitForText('Hello from User A')
            ->assertSee('Hello from User A');
    });
}

// Select2 / dropdown test
$browser->select('category', '1')
    ->assertSelected('category', '1');

// Modal test
$browser->click('#open-modal')
    ->waitFor('.modal')
    ->assertSee('Confirm Delete')
    ->press('Yes')
    ->waitUntilMissing('.modal');

// Custom assertion
$browser->assertScript('window.pageYOffset', 0);
$browser->script('window.scrollTo(0, 500);');

// env.dusk.local — environment khusus Dusk
// APP_ENV=dusk
// APP_URL=http://localhost:8000
```

```bash
# Run Dusk
php artisan dusk
php artisan dusk --group=login

# Start Dusk server
php artisan serve --env=dusk.local
```
