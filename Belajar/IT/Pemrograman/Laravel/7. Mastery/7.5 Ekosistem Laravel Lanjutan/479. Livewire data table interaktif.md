# 479. Livewire Data Table Interaktif

## Definisi Singkat
Membangun tabel data interaktif dengan Livewire — fitur search, sort, filter, pagination, bulk actions, dan inline editing tanpa JavaScript.

## Analogi
Seperti spreadsheet Excel online — klik header untuk sort, ketik di search untuk filter, centang baris untuk bulk action. Semua real-time.

## Poin-Poin Penting
- Search dengan `wire:model.live.debounce`
- Sort dengan click header + query string
- Pagination dengan `WithPagination` trait
- Filter dengan select/checkbox
- Bulk action dengan checkbox selection
- Inline editing dengan `wire:blur`

## Contoh Kode
```php
<?php

namespace App\Livewire;

use App\Models\Order;
use Livewire\Component;
use Livewire\WithPagination;

class OrdersTable extends Component
{
    use WithPagination;

    // Search & Filters
    public string $search = '';
    public string $status = '';
    public string $dateFrom = '';
    public string $dateTo = '';

    // Sort
    public string $sortField = 'created_at';
    public string $sortDirection = 'desc';

    // Bulk selection
    public array $selected = [];
    public bool $selectAll = false;

    // Inline editing
    public ?int $editingId = null;
    public string $editingField = '';
    public string $editingValue = '';

    // Query string
    protected $queryString = ['search', 'status', 'sortField', 'sortDirection'];

    // Listeners
    protected $listeners = ['refreshOrders' => '$refresh'];

    // Computed
    public function getOrdersProperty()
    {
        return Order::query()
            ->with('user')
            ->when($this->search, fn($q) =>
                $q->where('order_number', 'like', '%' . $this->search . '%')
                  ->orWhereHas('user', fn($q) =>
                      $q->where('name', 'like', '%' . $this->search . '%')
                  ))
            ->when($this->status, fn($q) => $q->where('status', $this->status))
            ->when($this->dateFrom, fn($q) => $q->whereDate('created_at', '>=', $this->dateFrom))
            ->when($this->dateTo, fn($q) => $q->whereDate('created_at', '<=', $this->dateTo))
            ->orderBy($this->sortField, $this->sortDirection)
            ->paginate(15);
    }

    public function sortBy(string $field): void
    {
        if ($this->sortField === $field) {
            $this->sortDirection = $this->sortDirection === 'asc' ? 'desc' : 'asc';
        } else {
            $this->sortField = $field;
            $this->sortDirection = 'asc';
        }
        $this->resetPage();
    }

    public function updatedSearch(): void
    {
        $this->resetPage();
    }

    // Select all
    public function updatedSelectAll(bool $value): void
    {
        $this->selected = $value ? $this->orders->pluck('id')->toArray() : [];
    }

    // Bulk actions
    public function bulkDelete(): void
    {
        Order::whereIn('id', $this->selected)->delete();
        $this->selected = [];
        session()->flash('message', count($this->selected) . ' orders deleted');
    }

    // Inline edit — start
    public function startEdit(int $id, string $field, mixed $value): void
    {
        $this->editingId = $id;
        $this->editingField = $field;
        $this->editingValue = $value;
    }

    // Inline edit — save
    public function saveEdit(): void
    {
        Order::find($this->editingId)->update([
            $this->editingField => $this->editingValue,
        ]);
        $this->editingId = null;
        $this->editingField = '';
        $this->editingValue = '';
    }

    // Export
    public function exportCsv(): void
    {
        $orders = Order::whereIn('id', $this->selected)->get();
        // return StreamedResponse...
    }

    public function render()
    {
        return view('livewire.orders-table', [
            'orders' => $this->orders,
        ]);
    }
}
```

```blade
<div>
    <input wire:model.live.debounce.300ms="search" placeholder="Search...">
    <select wire:model.live="status">
        <option value="">All Status</option>
        <option value="pending">Pending</option>
        <option value="completed">Completed</option>
    </select>

    <table>
        <thead>
            <tr>
                <th><input type="checkbox" wire:model.live="selectAll"></th>
                <th wire:click="sortBy('order_number')">Order #</th>
                <th wire:click="sortBy('total')">Total</th>
                <th>Status</th>
                <th>Actions</th>
            </tr>
        </thead>
        <tbody>
            @foreach ($orders as $order)
                <tr wire:key="order-{{ $order->id }}">
                    <td><input type="checkbox" value="{{ $order->id }}" wire:model.live="selected"></td>
                    <td>{{ $order->order_number }}</td>
                    <td>
                        @if ($editingId === $order->id && $editingField === 'total')
                            <input wire:model.live="editingValue"
                                   wire:blur="saveEdit"
                                   wire:keydown.enter="saveEdit">
                        @else
                            <span wire:click="startEdit({{ $order->id }}, 'total', {{ $order->total }})">
                                {{ number_format($order->total) }}
                            </span>
                        @endif
                    </td>
                    <td>{{ $order->status }}</td>
                    <td>
                        <button wire:click="$emit('deleteOrder', {{ $order->id }})">Delete</button>
                    </td>
                </tr>
            @endforeach
        </tbody>
    </table>
    {{ $orders->links() }}
</div>
```

## Kasus Penggunaan Nyata
Admin panel manajemen order dengan 50.000+ record — search, filter, sort real-time tanpa reload. Admin bisa edit total pesanan inline tanpa modal.
