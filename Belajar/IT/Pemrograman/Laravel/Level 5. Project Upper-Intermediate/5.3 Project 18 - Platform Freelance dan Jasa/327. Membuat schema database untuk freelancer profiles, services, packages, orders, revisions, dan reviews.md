```php
// app/Models/FreelancerProfile.php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class FreelancerProfile extends Model
{
    protected $fillable = [
        'user_id', 'headline', 'bio', 'hourly_rate',
        'skills', 'experience_level', 'response_time', 'avatar'
    ];

    protected $casts = ['skills' => 'array'];

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public function services()
    {
        return $this->hasMany(Service::class, 'user_id', 'user_id');
    }

    public function scopeBySkill($query, $skill)
    {
        return $query->whereJsonContains('skills', $skill);
    }
}
```

```php
// app/Models/Service.php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Support\Str;

class Service extends Model
{
    protected $fillable = [
        'user_id', 'category_id', 'title', 'slug',
        'description', 'gallery', 'faq', 'status'
    ];

    protected $casts = ['gallery' => 'array', 'faq' => 'array'];

    protected static function booted()
    {
        static::creating(fn($s) => $s->slug = Str::slug($s->title) . '-' . Str::random(5));
    }

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public function packages()
    {
        return $this->hasMany(ServicePackage::class)->orderBy('price');
    }

    public function category()
    {
        return $this->belongsTo(Category::class);
    }

    public function orders()
    {
        return $this->hasMany(Order::class);
    }

    public function basicPackage()
    {
        return $this->packages()->where('slug', 'basic')->first();
    }

    public function scopeActive($query)
    {
        return $query->where('status', 'active');
    }
}
```

```php
// app/Models/Order.php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Order extends Model
{
    protected $fillable = [
        'service_id', 'service_package_id', 'client_id', 'freelancer_id',
        'status', 'amount', 'requirements', 'revisions_left',
        'deadline', 'delivered_at', 'completed_at', 'cancel_reason'
    ];

    protected $casts = [
        'amount' => 'decimal:2',
        'deadline' => 'datetime',
        'delivered_at' => 'datetime',
        'completed_at' => 'datetime',
    ];

    public function service()
    {
        return $this->belongsTo(Service::class);
    }

    public function package()
    {
        return $this->belongsTo(ServicePackage::class, 'service_package_id');
    }

    public function client()
    {
        return $this->belongsTo(User::class, 'client_id');
    }

    public function freelancer()
    {
        return $this->belongsTo(User::class, 'freelancer_id');
    }

    public function files()
    {
        return $this->hasMany(OrderFile::class);
    }

    public function deliveryFiles()
    {
        return $this->files()->where('type', 'delivery');
    }

    public function revisionFiles()
    {
        return $this->files()->where('type', 'revision');
    }

    public function review()
    {
        return $this->hasOne(FreelanceReview::class);
    }

    public function canRequestRevision()
    {
        return $this->status === 'delivered' && $this->revisions_left > 0;
    }
}
```
