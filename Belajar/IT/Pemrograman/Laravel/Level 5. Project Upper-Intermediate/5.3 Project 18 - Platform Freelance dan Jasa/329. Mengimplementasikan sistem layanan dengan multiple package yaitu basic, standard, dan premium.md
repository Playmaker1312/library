# 329. Mengimplementasikan Sistem Layanan dengan Multiple Package yaitu Basic, Standard, dan Premium

## Definisi & Target
Membangun sistem layanan (service) dengan 3 tingkatan paket: Basic, Standard, Premium. Setiap paket memiliki harga, durasi, revisi, dan fitur berbeda.

## Komponen yang Terlibat
- Service model & ServicePackage model
- ServiceController
- Livewire ServiceForm
- Blade view detail layanan
- Perbandingan paket

## Perintah Artisan CLI
```bash
php artisan make:controller ServiceController --resource
php artisan make:livewire ServiceForm
php artisan make:livewire PackageManager
```

## Kode Implementasi (Per File)

### Controller
```php
// app/Http/Controllers/ServiceController.php
namespace App\Http\Controllers;

use App\Models\Service;
use App\Models\ServicePackage;
use App\Models\Category;
use Illuminate\Http\Request;

class ServiceController extends Controller
{
    public function index()
    {
        $services = Service::active()
            ->with(['user.freelancerProfile', 'packages', 'category'])
            ->latest()
            ->paginate(12);

        $categories = Category::has('services')->get();

        return view('services.index', compact('services', 'categories'));
    }

    public function create()
    {
        $categories = Category::all();
        return view('services.create', compact('categories'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'category_id' => 'required|exists:categories,id',
            'title' => 'required|max:255',
            'description' => 'required|max:10000',
            'packages' => 'required|array|min:1|max:3',
            'packages.*.name' => 'required|in:basic,standard,premium',
            'packages.*.price' => 'required|numeric|min:0',
            'packages.*.delivery_days' => 'required|integer|min:1',
            'packages.*.revisions' => 'required|integer|min:0',
            'packages.*.features' => 'nullable|array',
            'packages.*.description' => 'nullable|max:500',
        ]);

        $service = Service::create([
            'user_id' => auth()->id(),
            'category_id' => $request->category_id,
            'title' => $request->title,
            'description' => $request->description,
            'status' => 'active',
        ]);

        $sort = 0;
        foreach ($request->packages as $pkg) {
            $service->packages()->create([
                'name' => ucfirst($pkg['name']),
                'slug' => $pkg['name'],
                'price' => $pkg['price'],
                'delivery_days' => $pkg['delivery_days'],
                'revisions' => $pkg['revisions'],
                'features' => $pkg['features'] ?? [],
                'description' => $pkg['description'] ?? '',
                'sort' => $sort++,
            ]);
        }

        return redirect()->route('services.show', $service)
            ->with('success', 'Layanan berhasil dibuat.');
    }

    public function show(Service $service)
    {
        $service->load(['user.freelancerProfile', 'packages', 'category']);
        return view('services.show', compact('service'));
    }

    public function edit(Service $service)
    {
        $this->authorize('update', $service);
        $categories = Category::all();
        return view('services.edit', compact('service', 'categories'));
    }

    public function update(Request $request, Service $service)
    {
        $this->authorize('update', $service);

        $request->validate([
            'category_id' => 'required|exists:categories,id',
            'title' => 'required|max:255',
            'description' => 'required|max:10000',
            'packages' => 'required|array|min:1|max:3',
        ]);

        $service->update($request->only('category_id', 'title', 'description'));
        $service->packages()->delete();

        $sort = 0;
        foreach ($request->packages as $pkg) {
            $service->packages()->create([
                'name' => ucfirst($pkg['name']),
                'slug' => $pkg['name'],
                'price' => $pkg['price'],
                'delivery_days' => $pkg['delivery_days'],
                'revisions' => $pkg['revisions'],
                'features' => $pkg['features'] ?? [],
                'description' => $pkg['description'] ?? '',
                'sort' => $sort++,
            ]);
        }

        return redirect()->route('services.show', $service)
            ->with('success', 'Layanan diperbarui.');
    }

    public function destroy(Service $service)
    {
        $this->authorize('delete', $service);
        $service->delete();
        return redirect()->route('services.index')->with('success', 'Layanan dihapus.');
    }
}
```

### Routes
```php
// routes/web.php
use App\Http\Controllers\ServiceController;

Route::get('/services', [ServiceController::class, 'index'])->name('services.index');
Route::get('/services/{service}', [ServiceController::class, 'show'])->name('services.show');

Route::middleware('auth')->group(function () {
    Route::get('/services/create', [ServiceController::class, 'create'])->name('services.create');
    Route::post('/services', [ServiceController::class, 'store'])->name('services.store');
    Route::get('/services/{service}/edit', [ServiceController::class, 'edit'])->name('services.edit');
    Route::put('/services/{service}', [ServiceController::class, 'update'])->name('services.update');
    Route::delete('/services/{service}', [ServiceController::class, 'destroy'])->name('services.destroy');
});
```

### Livewire Component (Package Manager)
```php
// app/Livewire/PackageManager.php
namespace App\Livewire;

use Livewire\Component;

class PackageManager extends Component
{
    public $packages = [];

    public function mount($packages = [])
    {
        if (empty($packages)) {
            $this->packages = [
                ['name' => 'Basic', 'slug' => 'basic', 'price' => 50000, 'delivery_days' => 3, 'revisions' => 1, 'features' => ['Fitur 1'], 'description' => ''],
                ['name' => 'Standard', 'slug' => 'standard', 'price' => 100000, 'delivery_days' => 5, 'revisions' => 3, 'features' => ['Fitur 1', 'Fitur 2'], 'description' => ''],
                ['name' => 'Premium', 'slug' => 'premium', 'price' => 200000, 'delivery_days' => 7, 'revisions' => 5, 'features' => ['Fitur 1', 'Fitur 2', 'Fitur 3'], 'description' => ''],
            ];
        } else {
            $this->packages = $packages;
        }
    }

    public function addFeature($packageIndex)
    {
        $this->packages[$packageIndex]['features'][] = '';
    }

    public function removeFeature($packageIndex, $featureIndex)
    {
        unset($this->packages[$packageIndex]['features'][$featureIndex]);
        $this->packages[$packageIndex]['features'] = array_values($this->packages[$packageIndex]['features']);
    }

    public function render()
    {
        return view('livewire.package-manager');
    }
}
```

## Cara Integrasi & Pengujian
1. Buat kategori melalui seeder atau admin panel
2. Buat layanan dengan 3 paket: Basic (Rp50k, 3 hari, 1 revisi), Standard (Rp100k, 5 hari, 3 revisi), Premium (Rp200k, 7 hari, 5 revisi)
3. Verifikasi halaman detail menampilkan perbandingan paket
4. Edit layanan, ubah harga dan fitur
5. Hapus layanan (soft delete)
6. Pastikan hanya pemilik yang bisa edit/hapus
