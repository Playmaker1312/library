# 340. Membangun Dashboard Analytics untuk Freelancer dan Admin Platform

## Definisi & Target
Dashboard analytics untuk freelancer melihat performa layanan, pendapatan, order stats. Admin melihat overview seluruh platform.

## Komponen yang Terlibat
- FreelancerDashboardController
- AdminDashboardController
- Chart.js untuk grafik
- Export laporan

## Perintah Artisan CLI
```bash
php artisan make:controller FreelancerDashboardController
php artisan make:controller AdminDashboardController
php artisan make:service DashboardService
```

## Kode Implementasi (Per File)

### Service
```php
// app/Services/DashboardService.php
namespace App\Services;

use App\Models\Order;
use App\Models\Service;
use App\Models\FreelancerProfile;
use App\Models\FreelanceReview;
use App\Models\User;
use Illuminate\Support\Facades\DB;

class DashboardService
{
    public function freelancerStats($userId)
    {
        $orders = Order::where('freelancer_id', $userId);
        $services = Service::where('user_id', $userId);
        $reviews = FreelanceReview::where('to_user_id', $userId)
            ->where('type', 'client_to_freelancer');

        return [
            'total_orders' => (clone $orders)->count(),
            'completed_orders' => (clone $orders)->where('status', 'completed')->count(),
            'active_orders' => (clone $orders)->whereIn('status', ['active', 'in_progress', 'revision'])->count(),
            'total_revenue' => (clone $orders)->where('status', 'completed')->sum('amount'),
            'total_services' => (clone $services)->count(),
            'average_rating' => (clone $reviews)->avg('rating') ?? 0,
            'total_reviews' => (clone $reviews)->count(),
            'response_rate' => 95,
        ];
    }

    public function freelancerRevenueChart($userId, $months = 6)
    {
        return Order::select(
            DB::raw("DATE_FORMAT(completed_at, '%Y-%m') as month"),
            DB::raw('SUM(amount) as total')
        )
            ->where('freelancer_id', $userId)
            ->where('status', 'completed')
            ->where('completed_at', '>=', now()->subMonths($months))
            ->groupBy('month')
            ->orderBy('month')
            ->get();
    }

    public function freelancerOrderStatus($userId)
    {
        return Order::where('freelancer_id', $userId)
            ->select('status', DB::raw('COUNT(*) as total'))
            ->groupBy('status')
            ->pluck('total', 'status');
    }

    public function adminStats()
    {
        return [
            'total_users' => User::count(),
            'total_freelancers' => User::where('role', 'freelancer')->count(),
            'total_clients' => User::where('role', 'client')->count(),
            'total_orders' => Order::count(),
            'completed_orders' => Order::where('status', 'completed')->count(),
            'total_revenue' => Order::where('status', 'completed')->sum('amount'),
            'total_services' => Service::count(),
            'pending_orders' => Order::where('status', 'pending')->count(),
        ];
    }

    public function adminRevenueChart($months = 12)
    {
        return Order::select(
            DB::raw("DATE_FORMAT(completed_at, '%Y-%m') as month"),
            DB::raw('SUM(amount) as total')
        )
            ->where('status', 'completed')
            ->where('completed_at', '>=', now()->subMonths($months))
            ->groupBy('month')
            ->orderBy('month')
            ->get();
    }

    public function topFreelancers($limit = 5)
    {
        return User::where('role', 'freelancer')
            ->withCount(['freelanceReviewsReceived as avg_rating' => function ($q) {
                $q->select(DB::raw('coalesce(avg(rating), 0)'));
            }])
            ->withCount(['ordersReceived as completed_orders' => function ($q) {
                $q->where('status', 'completed');
            }])
            ->orderByDesc('completed_orders')
            ->limit($limit)
            ->get();
    }
}
```

### Controllers
```php
// app/Http/Controllers/FreelancerDashboardController.php
namespace App\Http\Controllers;

use App\Services\DashboardService;
use Illuminate\Http\Request;

class FreelancerDashboardController extends Controller
{
    protected DashboardService $dashboard;

    public function __construct(DashboardService $dashboard)
    {
        $this->dashboard = $dashboard;
    }

    public function index()
    {
        $stats = $this->dashboard->freelancerStats(auth()->id());
        $revenueChart = $this->dashboard->freelancerRevenueChart(auth()->id());
        $orderStatus = $this->dashboard->freelancerOrderStatus(auth()->id());

        $recentOrders = Order::where('freelancer_id', auth()->id())
            ->with(['service', 'client'])
            ->latest()
            ->take(5)
            ->get();

        return view('freelancer.dashboard', compact(
            'stats', 'revenueChart', 'orderStatus', 'recentOrders'
        ));
    }
}
```

```php
// app/Http/Controllers/AdminDashboardController.php
namespace App\Http\Controllers;

use App\Models\Order;
use App\Models\User;
use App\Services\DashboardService;
use Illuminate\Http\Request;

class AdminDashboardController extends Controller
{
    protected DashboardService $dashboard;

    public function __construct(DashboardService $dashboard)
    {
        $this->dashboard = $dashboard;
    }

    public function index()
    {
        $stats = $this->dashboard->adminStats();
        $revenueChart = $this->dashboard->adminRevenueChart();
        $topFreelancers = $this->dashboard->topFreelancers();

        $recentOrders = Order::with(['service', 'client', 'freelancer'])
            ->latest()
            ->take(10)
            ->get();

        $pendingWithdrawals = \App\Models\Withdrawal::where('status', 'pending')->count();

        return view('admin.dashboard', compact(
            'stats', 'revenueChart', 'topFreelancers',
            'recentOrders', 'pendingWithdrawals'
        ));
    }
}
```

## Cara Integrasi & Pengujian
1. Login sebagai freelancer, akses `/freelancer/analytics`
2. Pastikan statistik menampilkan data order yang benar
3. Verifikasi grafik pendapatan per bulan
4. Cek order terbaru muncul di tabel
5. Login sebagai admin, akses `/admin/dashboard`
6. Verifikasi overview seluruh platform
7. Cek daftar top freelancer dan revenue platform
8. Pastikan data kosong tidak menyebabkan error
