# 338. Mengimplementasikan Review dan Rating Setelah Order Selesai

## Definisi & Target
Klien dan freelancer saling memberi review dan rating setelah order completed. Target: transparansi reputasi, ditampilkan di profil.

## Komponen yang Terlibat
- Review model (freelance_reviews)
- ReviewController
- Livewire ReviewForm
- Rating stats di profil
- Notification

## Perintah Artisan CLI
```bash
php artisan make:controller FreelanceReviewController
php artisan make:notification NewReviewNotification
```

## Kode Implementasi (Per File)

### Model
```php
// app/Models/FreelanceReview.php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class FreelanceReview extends Model
{
    protected $table = 'freelance_reviews';

    protected $fillable = [
        'order_id', 'from_user_id', 'to_user_id',
        'rating', 'review', 'type'
    ];

    protected $casts = ['rating' => 'integer'];

    public function order()
    {
        return $this->belongsTo(Order::class);
    }

    public function fromUser()
    {
        return $this->belongsTo(User::class, 'from_user_id');
    }

    public function toUser()
    {
        return $this->belongsTo(User::class, 'to_user_id');
    }
}
```

```php
// app/Models/User.php (tambahan)
use App\Models\FreelanceReview;

public function freelanceReviewsReceived()
{
    return $this->hasMany(FreelanceReview::class, 'to_user_id');
}

public function freelanceReviewsGiven()
{
    return $this->hasMany(FreelanceReview::class, 'from_user_id');
}

public function averageFreelanceRating()
{
    return $this->freelanceReviewsReceived()
        ->where('type', 'client_to_freelancer')
        ->avg('rating') ?? 0;
}

public function totalFreelanceReviews()
{
    return $this->freelanceReviewsReceived()
        ->where('type', 'client_to_freelancer')
        ->count();
}
```

### Controller
```php
// app/Http/Controllers/FreelanceReviewController.php
namespace App\Http\Controllers;

use App\Models\FreelanceReview;
use App\Models\Order;
use App\Notifications\NewReviewNotification;
use Illuminate\Http\Request;

class FreelanceReviewController extends Controller
{
    public function createForFreelancer(Order $order)
    {
        if ($order->client_id !== auth()->id() || $order->status !== 'completed') {
            abort(403);
        }

        $existing = FreelanceReview::where('order_id', $order->id)
            ->where('from_user_id', auth()->id())
            ->where('type', 'client_to_freelancer')
            ->first();

        if ($existing) {
            return redirect()->route('orders.show', $order)
                ->with('info', 'Anda sudah memberikan review.');
        }

        return view('freelance-reviews.create', compact('order'));
    }

    public function createForClient(Order $order)
    {
        if ($order->freelancer_id !== auth()->id() || $order->status !== 'completed') {
            abort(403);
        }

        $existing = FreelanceReview::where('order_id', $order->id)
            ->where('from_user_id', auth()->id())
            ->where('type', 'freelancer_to_client')
            ->first();

        if ($existing) {
            return redirect()->route('orders.show', $order)
                ->with('info', 'Anda sudah memberikan review.');
        }

        return view('freelance-reviews.create-client', compact('order'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'order_id' => 'required|exists:orders,id',
            'to_user_id' => 'required|exists:users,id',
            'rating' => 'required|integer|min:1|max:5',
            'review' => 'nullable|string|max:5000',
            'type' => 'required|in:client_to_freelancer,freelancer_to_client',
        ]);

        $order = Order::findOrFail($request->order_id);
        $toUser = \App\Models\User::findOrFail($request->to_user_id);

        $existing = FreelanceReview::where('order_id', $order->id)
            ->where('from_user_id', auth()->id())
            ->where('type', $request->type)
            ->exists();

        if ($existing) {
            return back()->with('error', 'Anda sudah memberikan review.');
        }

        $review = FreelanceReview::create([
            'order_id' => $order->id,
            'from_user_id' => auth()->id(),
            'to_user_id' => $request->to_user_id,
            'rating' => $request->rating,
            'review' => $request->review,
            'type' => $request->type,
        ]);

        $toUser->notify(new NewReviewNotification($review));

        return redirect()->route('orders.show', $order)
            ->with('success', 'Review berhasil dikirim.');
    }

    public function showUserReviews($userId)
    {
        $user = \App\Models\User::findOrFail($userId);

        $reviews = FreelanceReview::where('to_user_id', $userId)
            ->where('type', 'client_to_freelancer')
            ->with(['fromUser', 'order.service'])
            ->latest()
            ->paginate(10);

        $stats = [
            'average' => $user->averageFreelanceRating(),
            'total' => $user->totalFreelanceReviews(),
            'distribution' => FreelanceReview::where('to_user_id', $userId)
                ->where('type', 'client_to_freelancer')
                ->selectRaw('rating, COUNT(*) as count')
                ->groupBy('rating')
                ->pluck('count', 'rating')
                ->toArray(),
        ];

        return view('freelance-reviews.user-reviews', compact('user', 'reviews', 'stats'));
    }
}
```

### Notification
```php
// app/Notifications/NewReviewNotification.php
namespace App\Notifications;

use App\Models\FreelanceReview;
use Illuminate\Bus\Queueable;
use Illuminate\Notifications\Notification;

class NewReviewNotification extends Notification
{
    use Queueable;

    public $review;

    public function __construct(FreelanceReview $review)
    {
        $this->review = $review;
    }

    public function via($notifiable)
    {
        return ['database'];
    }

    public function toArray($notifiable)
    {
        return [
            'type' => 'new_review',
            'order_id' => $this->review->order_id,
            'reviewer_name' => $this->review->fromUser->name,
            'rating' => $this->review->rating,
            'message' => "{$this->review->fromUser->name} memberi Anda rating {$this->review->rating} bintang.",
        ];
    }
}
```

## Cara Integrasi & Pengujian
1. Selesaikan order (completed)
2. Klien klik "Beri Review", diarahkan ke form review
3. Pilih rating (klik bintang) dan tulis ulasan
4. Submit, freelancer dapat notifikasi
5. Kunjungi profil freelancer, lihat rating dan review
6. Freelancer juga bisa review klien
7. Pastikan tidak bisa double review untuk order yang sama
8. Verifikasi stats: average, total, distribution
