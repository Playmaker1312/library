# 358. Mengimplementasikan Builder Survey dengan Drag and Drop Menggunakan SortableJS

## Definisi & Target
Survey builder dengan drag & drop untuk mengatur urutan pertanyaan menggunakan SortableJS. Target: UI intuitif untuk membuat survey.

## Komponen yang Terlibat
- SurveyController (create, edit)
- Livewire QuestionManager
- SortableJS library
- Question type selector
- Preview mode

## Perintah Artisan CLI
```bash
php artisan make:livewire QuestionManager
php artisan make:livewire QuestionEditor
```

## Kode Implementasi (Per File)

### Controller
```php
// app/Http/Controllers/SurveyController.php
namespace App\Http\Controllers;

use App\Models\Survey;
use Illuminate\Http\Request;

class SurveyController extends Controller
{
    public function index()
    {
        $surveys = Survey::where('user_id', auth()->id())
            ->orWhereHas('collaborators', fn($q) => $q->where('user_id', auth()->id()))
            ->withCount('responses')
            ->latest()
            ->paginate(10);

        return view('surveys.index', compact('surveys'));
    }

    public function create()
    {
        return view('surveys.create');
    }

    public function store(Request $request)
    {
        $validated = $request->validate([
            'title' => 'required|max:255',
            'description' => 'nullable|max:5000',
        ]);

        $survey = Survey::create([
            'user_id' => auth()->id(),
            'title' => $validated['title'],
            'description' => $validated['description'] ?? null,
            'status' => 'draft',
            'settings' => [
                'allow_anonymous' => true,
                'limit_one_per_ip' => false,
                'show_progress_bar' => true,
                'confirmation_message' => 'Terima kasih, respons Anda telah dicatat.',
            ],
        ]);

        return redirect()->route('surveys.edit', $survey)
            ->with('success', 'Survey berhasil dibuat. Sekarang tambahkan pertanyaan.');
    }

    public function show(Survey $survey)
    {
        $survey->load('questions');
        return view('surveys.show', compact('survey'));
    }

    public function edit(Survey $survey)
    {
        $survey->load('questions');
        return view('surveys.edit', compact('survey'));
    }

    public function update(Request $request, Survey $survey)
    {
        $validated = $request->validate([
            'title' => 'required|max:255',
            'description' => 'nullable|max:5000',
            'status' => 'in:draft,published,closed',
        ]);

        $survey->update($validated);

        return back()->with('success', 'Survey diperbarui.');
    }

    public function destroy(Survey $survey)
    {
        $survey->delete();
        return redirect()->route('surveys.index')->with('success', 'Survey dihapus.');
    }
}
```

### Livewire QuestionManager
```php
// app/Livewire/QuestionManager.php
namespace App\Livewire;

use App\Models\Question;
use App\Models\Survey;
use Livewire\Component;

class QuestionManager extends Component
{
    public Survey $survey;
    public $questions = [];
    public $editingIndex = null;

    protected $listeners = ['question-updated' => 'loadQuestions', 'reorder' => 'updateOrder'];

    public function mount(Survey $survey)
    {
        $this->survey = $survey;
        $this->loadQuestions();
    }

    public function loadQuestions()
    {
        $this->questions = $this->survey->questions()->orderBy('order')->get()->toArray();
    }

    public function addQuestion($type)
    {
        $maxOrder = $this->survey->questions()->max('order') ?? 0;

        $question = Question::create([
            'survey_id' => $this->survey->id,
            'title' => 'Pertanyaan baru',
            'type' => $type,
            'options' => in_array($type, ['radio', 'checkbox', 'dropdown'])
                ? ['Opsi 1', 'Opsi 2', 'Opsi 3']
                : ($type === 'rating' ? ['min' => 1, 'max' => 5] : null),
            'is_required' => false,
            'order' => $maxOrder + 1,
        ]);

        $this->loadQuestions();
        $this->dispatch('question-added');
    }

    public function updateOrder($orderedIds)
    {
        foreach ($orderedIds as $index => $id) {
            Question::where('id', $id)->update(['order' => $index + 1]);
        }
        $this->loadQuestions();
    }

    public function deleteQuestion($questionId)
    {
        Question::find($questionId)?->delete();
        $this->loadQuestions();
    }

    public function duplicateQuestion($questionId)
    {
        $original = Question::find($questionId);
        if (!$original) return;

        $maxOrder = $this->survey->questions()->max('order') ?? 0;
        $copy = $original->replicate();
        $copy->title = $original->title . ' (copy)';
        $copy->order = $maxOrder + 1;
        $copy->save();

        $this->loadQuestions();
    }

    public function render()
    {
        return view('livewire.question-manager');
    }
}
```

### Blade View
```blade
{{-- resources/views/livewire/question-manager.blade.php --}}
<div>
    <div class="flex gap-2 mb-4">
        <button wire:click="addQuestion('text')" class="btn btn-sm btn-outline">Teks</button>
        <button wire:click="addQuestion('textarea')" class="btn btn-sm btn-outline">Paragraf</button>
        <button wire:click="addQuestion('radio')" class="btn btn-sm btn-outline">Pilihan Ganda</button>
        <button wire:click="addQuestion('checkbox')" class="btn btn-sm btn-outline">Checkbox</button>
        <button wire:click="addQuestion('dropdown')" class="btn btn-sm btn-outline">Dropdown</button>
        <button wire:click="addQuestion('rating')" class="btn btn-sm btn-outline">Rating</button>
        <button wire:click="addQuestion('matrix')" class="btn btn-sm btn-outline">Matrix</button>
    </div>

    <div id="question-list" class="space-y-4" x-data="sortableQuestions()">
        @foreach ($questions as $index => $question)
            <div class="question-item bg-white p-4 rounded shadow border-l-4 border-blue-500"
                 data-id="{{ $question['id'] }}">
                <div class="flex items-start justify-between">
                    <div class="flex-1 cursor-move">
                        <span class="text-xs text-gray-400 uppercase font-semibold">{{ $question['type'] }}</span>
                        <p class="font-medium">{{ $question['title'] }}</p>
                    </div>
                    <div class="flex gap-2">
                        <button wire:click="duplicateQuestion({{ $question['id'] }})" class="text-gray-400 hover:text-blue-600">&plus;</button>
                        <button wire:click="$dispatch('edit-question', { id: {{ $question['id'] }} })" class="text-gray-400 hover:text-yellow-600">&#9998;</button>
                        <button wire:click="deleteQuestion({{ $question['id'] }})" class="text-gray-400 hover:text-red-600">&times;</button>
                    </div>
                </div>

                @if (in_array($question['type'], ['radio', 'checkbox', 'dropdown']) && $question['options'])
                    <div class="mt-2 space-y-1">
                        @foreach ($question['options'] as $option)
                            <div class="flex items-center gap-2 text-sm text-gray-600">
                                <span class="w-4 h-4 rounded-full border border-gray-300 inline-block"></span>
                                {{ is_string($option) ? $option : '' }}
                            </div>
                        @endforeach
                    </div>
                @endif
                @if ($question['type'] === 'rating')
                    <div class="mt-2 text-yellow-400 text-lg">
                        @for ($i = 1; $i <= ($question['options']['max'] ?? 5); $i++)
                            &#9733;
                        @endfor
                    </div>
                @endif
            </div>
        @endforeach
    </div>

    @push('scripts')
    <script src="https://cdn.jsdelivr.net/npm/sortablejs@1.15.0/Sortable.min.js"></script>
    <script>
        function sortableQuestions() {
            return {
                init() {
                    const el = document.getElementById('question-list');
                    if (!el) return;
                    Sortable.create(el, {
                        animation: 150,
                        handle: '.cursor-move',
                        onEnd: (evt) => {
                            const orderedIds = [];
                            el.querySelectorAll('.question-item').forEach(item => {
                                orderedIds.push(parseInt(item.dataset.id));
                            });
                            @this.call('updateOrder', orderedIds);
                        }
                    });
                }
            };
        }
    </script>
    @endpush
</div>
```

## Cara Integrasi & Pengujian
1. Buat survey, isi judul dan deskripsi
2. Tambahkan pertanyaan: text, radio, checkbox, dropdown, rating
3. Drag & drop untuk mengurutkan pertanyaan
4. Duplikat pertanyaan
5. Hapus pertanyaan
6. Publikasikan survey
7. Lihat survey di halaman publik
