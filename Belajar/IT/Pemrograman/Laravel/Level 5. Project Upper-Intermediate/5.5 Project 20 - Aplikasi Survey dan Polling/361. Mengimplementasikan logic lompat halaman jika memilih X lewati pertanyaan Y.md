# 361. Mengimplementasikan Logic Lompat Halaman (Jika Memilih X, Lewati Pertanyaan Y)

## Definisi & Target
Page jump logic: jika responden memilih jawaban tertentu, langsung lompat ke halaman/pertanyaan tertentu.

## Komponen yang Terlibat
- Question logic (`logic` column dengan `action: 'jump'`)
- SurveyForm (skip questions)

### 2. Evaluasi Jump di SurveyForm
```php
// app/Livewire/SurveyForm.php - tambah method
public function getVisibleQuestionsProperty()
{
    return $this->survey->questions->filter(function ($question) {
        if (!$question->logic || !($question->logic['enabled'] ?? false)) {
            return true;
        }

        $logic = $question->logic;
        $matches = collect($logic['conditions'])->map(function ($cond) {
            $answer = $this->answers[$cond['question_id']] ?? '';

            return match ($cond['operator']) {
                'equals' => $answer == $cond['value'],
                'not_equals' => $answer != $cond['value'],
                'contains' => str_contains((string)$answer, (string)$cond['value']),
                'greater_than' => (float)$answer > (float)$cond['value'],
                'less_than' => (float)$answer < (float)$cond['value'],
                default => false,
            };
        });

        $visible = $logic['operator'] === 'and'
            ? $matches->every(fn($m) => $m)
            : $matches->contains(fn($m) => $m);

        // Jika action adalah jump, dan kondisi terpenuhi, lewati question
        if (($logic['action'] ?? 'show') === 'jump' && $visible) {
            return false; // question disembunyikan
        }

        return !$visible; // show condition: tampilkan jika kondisi TERPENUHI
    })->values();
}
```

### 5. Fungsi JavaScript untuk Jump
```javascript
// Tambah di survey-form script
function applyJumpLogic() {
    document.querySelectorAll('[data-jump]').forEach(el => {
        const logic = JSON.parse(el.dataset.jump);
        if (!logic.enabled || logic.action !== 'jump') return;

        // Evaluasi kondisi
        let triggered = false;
        const conditions = logic.conditions || [];

        if (conditions.length === 0 && logic.jump_to_question) {
            // Direct jump: question ini selalu trigger lompatan
            triggered = true;
        } else {
            triggered = logic.operator === 'and'
                ? conditions.every(c => evaluateCondition(c))
                : conditions.some(c => evaluateCondition(c));
        }

        if (triggered && logic.jump_to_question) {
            // Sembunyikan semua question antara current dan target
            const target = document.querySelector(`[data-question-id="${logic.jump_to_question}"]`);
            if (target) {
                let next = el.nextElementSibling;
                while (next && next !== target) {
                    if (next.dataset.questionId) {
                        next.style.display = 'none';
                    }
                    next = next.nextElementSibling;
                }
                target.style.display = 'block';
            }
        }
    });
}

function evaluateCondition(cond) {
    const input = document.querySelector(`[name="answers.${cond.question_id}"]:checked`)
        || document.querySelector(`[wire\\:model="answers.${cond.question_id}"]`);
    const val = input ? input.value : '';
    switch (cond.operator) {
        case 'equals': return val == cond.value;
        case 'not_equals': return val != cond.value;
        default: return false;
    }
}

// Panggil setelah render dan saat ada perubahan
document.addEventListener('change', applyJumpLogic);
document.addEventListener('livewire:update', () => setTimeout(applyJumpLogic, 50));
```
