# 370. Permission dan Role pada Aplikasi Survey

## Definisi & Target
Sistem permission lengkap: owner, admin, editor, viewer. Setiap role memiliki akses terbatas sesuai wewenang.

## Komponen yang Terlibat
- Role enum/constant
- Middleware
- Gates/Policies
- UI conditional rendering
- Audit log

## Kode Implementasi (Per File)

### Role Enum
```php
// app/Enums/SurveyRole.php
namespace App\Enums;

enum SurveyRole: string
{
    case Owner = 'owner';
    case Admin = 'admin';
    case Editor = 'editor';
    case Viewer = 'viewer';

    public function label(): string
    {
        return match ($this) {
            self::Owner => 'Pemilik',
            self::Admin => 'Admin',
            self::Editor => 'Editor',
            self::Viewer => 'Viewer',
        };
    }

    public function permissions(): array
    {
        return match ($this) {
            self::Owner => ['view', 'create', 'edit', 'delete', 'manage_collaborators',
                            'publish', 'close', 'view_results', 'export', 'manage_settings'],
            self::Admin => ['view', 'edit', 'manage_collaborators', 'publish',
                            'close', 'view_results', 'export', 'manage_settings'],
            self::Editor => ['view', 'edit', 'view_results', 'export'],
            self::Viewer => ['view', 'view_results'],
        };
    }

    public function hasPermission(string $permission): bool
    {
        return in_array($permission, $this->permissions());
    }
}
```

### Permission Service
```php
// app/Services/SurveyPermissionService.php
namespace App\Services;

use App\Models\Survey;
use App\Models\User;
use App\Enums\SurveyRole;

class SurveyPermissionService
{
    public function getUserRole(Survey $survey, User $user): SurveyRole
    {
        if ($survey->user_id === $user->id) {
            return SurveyRole::Owner;
        }

        $collaborator = $survey->collaborators()->where('user_id', $user->id)->first();

        return match ($collaborator?->role) {
            'admin' => SurveyRole::Admin,
            'editor' => SurveyRole::Editor,
            'viewer' => SurveyRole::Viewer,
            default => throw new \RuntimeException('User tidak memiliki akses ke survey ini.')
        };
    }

    public function can(Survey $survey, User $user, string $permission): bool
    {
        try {
            $role = $this->getUserRole($survey, $user);
            return $role->hasPermission($permission);
        } catch (\RuntimeException) {
            return false;
        }
    }

    public function requirePermission(Survey $survey, User $user, string $permission): void
    {
        if (!$this->can($survey, $user, $permission)) {
            abort(403, "Anda tidak memiliki izin '{$permission}' untuk survey ini.");
        }
    }
}
```

### Middleware
```php
// app/Http/Middleware/CheckSurveyAccess.php
namespace App\Http\Middleware;

use App\Models\Survey;
use App\Services\SurveyPermissionService;
use Closure;
use Illuminate\Http\Request;
use Symfony\Component\HttpFoundation\Response;

class CheckSurveyAccess
{
    public function __construct(
        protected SurveyPermissionService $permissionService
    ) {}

    public function handle(Request $request, Closure $next, string $permission = 'view'): Response
    {
        $survey = $request->route('survey');

        if (!$survey instanceof Survey) {
            $survey = Survey::findOrFail($request->route('survey'));
            $request->route()->setParameter('survey', $survey);
        }

        if (!auth()->check()) {
            if ($permission === 'view' && $survey->status === 'published') {
                return $next($request);
            }
            abort(401);
        }

        $this->permissionService->requirePermission($survey, auth()->user(), $permission);

        return $next($request);
    }
}
```

```php

    // Owner/Admin
    Route::middleware('survey.access:manage_collaborators')->group(function () {
        Route::get('/surveys/{survey}/collaborators', [CollaboratorController::class, 'index'])->name('surveys.collaborators.index');
        Route::post('/surveys/{survey}/collaborators', [CollaboratorController::class, 'store'])->name('surveys.collaborators.store');

    // Owner
    Route::middleware('survey.access:delete')->group(function () {
        Route::delete('/surveys/{survey}', [SurveyController::class, 'destroy'])->name('surveys.destroy');
    });

    // Viewer/Editor/Admin/Owner
    Route::middleware('survey.access:view_results')->group(function () {
        Route::get('/surveys/{survey}/results', [SurveyResultController::class, 'overview'])->name('surveys.results.overview');
    });

// Public: anyone can view published surveys
```

### View Composer untuk Role
```php
// app/Providers/ViewServiceProvider.php
namespace App\Providers;

use App\Services\SurveyPermissionService;
use Illuminate\Support\Facades\View;
use Illuminate\Support\ServiceProvider;

class ViewServiceProvider extends ServiceProvider
{
    public function boot(): void
    {
        View::composer('surveys.*', function ($view) {
            $survey = $view->getData()['survey'] ?? null;
            if ($survey && auth()->check()) {
                try {
                    $role = app(SurveyPermissionService::class)->getUserRole($survey, auth()->user());
                    $view->with('userRole', $role);
                    $view->with('permissions', $role->permissions());
                } catch (\RuntimeException) {
                    $view->with('userRole', null);
                    $view->with('permissions', []);
                }
            }
        });
    }
}
```

### Audit Log
```php

use App\Traits\LogsSurveyActivity;

// app/Traits/LogsSurveyActivity.php
namespace App\Traits;

use App\Models\SurveyAuditLog;

trait LogsSurveyActivity
{
    public static function bootLogsSurveyActivity()
    {
        static::updated(function ($survey) {
            $changes = $survey->getChanges();
            unset($changes['updated_at']);
            if (!empty($changes)) {
                SurveyAuditLog::create([
                    'survey_id' => $survey->id,
                    'user_id' => auth()->id(),
                    'action' => 'update',
                    'changes' => $changes,
                ]);
            }
        });
    }

    public function auditLogs()
    {
        return $this->hasMany(SurveyAuditLog::class)->latest();
    }
}
```

### Owner Transfer
```php
// app/Http/Controllers/SurveyController.php - tambah
public function transferOwnership(Request $request, Survey $survey)
{
    abort_if($survey->user_id !== auth()->id(), 403);

    $validated = $request->validate([
        'user_id' => 'required|exists:users,id',
    ]);

    $newOwner = User::findOrFail($validated['user_id']);

    // Tambah owner lama sebagai admin
    SurveyCollaborator::updateOrCreate(
        ['survey_id' => $survey->id, 'user_id' => auth()->id()],
        ['role' => 'admin']
    );

    $survey->update(['user_id' => $newOwner->id]);

    return back()->with('success', "Kepemilikan dialihkan ke {$newOwner->name}.");
}
```

## Cara Integrasi & Pengujian
1. Buat survey sebagai User A
2. Login sebagai User B tanpa akses, verifikasi 403
3. Tambah User B sebagai viewer, verifikasi hanya bisa lihat hasil
4. Upgrade ke editor, verifikasi bisa edit pertanyaan
5. Upgrade ke admin, verifikasi bisa kelola kolaborator
6. Coba delete survey sebagai viewer → 403
7. Transfer kepemilikan ke user lain
8. Cek audit log untuk perubahan
