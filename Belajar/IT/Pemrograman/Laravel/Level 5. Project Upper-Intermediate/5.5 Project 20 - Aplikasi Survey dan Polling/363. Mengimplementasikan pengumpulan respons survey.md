```php
// app/Models/Survey.php - tambah method
public function canRespond($ip, $email = null)
{
    $settings = $this->settings ?? [];

    // Cek jika survey ditutup
    if ($this->status !== 'published') {
        return ['allowed' => false, 'reason' => 'Survey tidak tersedia.'];
    }

    // Cek batas waktu
    if ($this->starts_at && now()->lt($this->starts_at)) {
        return ['allowed' => false, 'reason' => 'Survey belum dimulai.'];
    }
    if ($this->ends_at && now()->gt($this->ends_at)) {
        return ['allowed' => false, 'reason' => 'Survey sudah berakhir.'];
    }

    // Cek batas maksimal respons
    if (($settings['max_responses'] ?? null) && $this->totalResponses() >= $settings['max_responses']) {
        return ['allowed' => false, 'reason' => 'Batas maksimal respons telah tercapai.'];
    }

    // Cek limit per IP
    if ($settings['limit_one_per_ip'] ?? false) {
        $existing = $this->responses()->where('respondent_ip', $ip)
            ->whereNotNull('completed_at')->exists();
        if ($existing) {
            return ['allowed' => false, 'reason' => 'Anda sudah mengisi survey ini.'];
        }
    }

    // Cek limit per email
    if ($email && ($settings['limit_one_per_email'] ?? false)) {
        $existing = $this->responses()->where('respondent_email', $email)
            ->whereNotNull('completed_at')->exists();
        if ($existing) {
            return ['allowed' => false, 'reason' => 'Email ini sudah digunakan.'];
        }
    }

    return ['allowed' => true];
}

public function totalResponses()
{
    return $this->responses()->whereNotNull('completed_at')->count();
}

public function completionRate()
{
    $total = $this->responses()->count();
    $completed = $this->totalResponses();
    return $total > 0 ? round(($completed / $total) * 100, 2) : 0;
}

public function averageTimeSpent()
{
    $seconds = $this->responses()
        ->whereNotNull('completed_at')
        ->whereNotNull('time_spent')
        ->avg('time_spent');

    if (!$seconds) return null;

    $minutes = floor($seconds / 60);
    $secs = $seconds % 60;
    return "{$minutes}m {$secs}d";
}
```

### Livewire SurveyForm - Response Management
```php
// app/Livewire/SurveyForm.php - tambah property dan method
public $responseId;
public $autoSaveTimer = 30; // detik

public function mount(Survey $survey)
{
    $this->survey = $survey->load('questions');

    // Cek apakah sudah pernah mengisi
    $check = $survey->canRespond(request()->ip(), request()->email);
    if (!$check['allowed']) {
        session()->flash('error', $check['reason']);
        return redirect()->route('surveys.closed');
    }

    // Cari draft atau buat baru
    $existing = SurveyResponse::where('survey_id', $survey->id)
        ->where('respondent_ip', request()->ip())
        ->whereNull('completed_at')
        ->latest()
        ->first();

    if ($existing) {
        $this->responseId = $existing->id;
        $this->loadDraftAnswers($existing);
    }

    $this->totalPages = max($survey->settings['pagination'] ?? 1, 1);
    $this->initializeAnswers();
}

public function initializeAnswers()
{
    $this->answers = collect($this->survey->questions)->mapWithKeys(fn($q) => [
        $q->id => $q->type === 'checkbox' ? [] : ($this->answers[$q->id] ?? null)
    ])->toArray();
}

public function loadDraftAnswers(SurveyResponse $response)
{
    $answers = $response->answers()->get()->keyBy('question_id');
    foreach ($answers as $answer) {
        $this->answers[$answer->question_id] = json_decode($answer->value, true) ?? $answer->value;
    }
    $this->startedAt = $response->started_at;
}

public function autoSave()
{
    if (!$this->responseId) {
        $this->responseId = SurveyResponse::create([
            'survey_id' => $this->survey->id,
            'respondent_ip' => request()->ip(),
            'respondent_email' => request()->email,
            'started_at' => $this->startedAt ?? now(),
        ])->id;
        $this->startedAt = now();
    }

    foreach ($this->answers as $questionId => $value) {
        ResponseAnswer::updateOrCreate(
            ['response_id' => $this->responseId, 'question_id' => $questionId],
            ['value' => is_array($value) ? json_encode($value) : $value]
        );
    }

    $this->dispatch('auto-saved');
}

public function submit()
{
    $this->validate();

    if (!$this->responseId) {
        $this->autoSave();
    }

    SurveyResponse::where('id', $this->responseId)->update([
        'completed_at' => now(),
        'time_spent' => $this->startedAt ? now()->diffInSeconds($this->startedAt) : null,
    ]);

    // Kirim notifikasi ke pembuat survey
    $this->survey->user->notify(new SurveyResponseReceived($this->survey, $this->responseId));

    return redirect()->route('surveys.thank-you', $this->survey);
}

public function saveDraft()
{
    $this->autoSave();
    session()->flash('info', 'Draft tersimpan.');
}
```

### Notifikasi ke Pembuat Survey
```php
// app/Notifications/SurveyResponseReceived.php
namespace App\Notifications;

use App\Models\Survey;
use Illuminate\Bus\Queueable;
use Illuminate\Notifications\Notification;
use Illuminate\Notifications\Messages\MailMessage;

class SurveyResponseReceived extends Notification
{
    use Queueable;

    public function __construct(
        public Survey $survey,
        public int $responseId
    ) {}

    public function via($notifiable)
    {
        return $notifiable->notification_preferences['survey_response'] ?? ['mail', 'database'];
    }

    public function toMail($notifiable): MailMessage
    {
        return (new MailMessage)
            ->subject("Respons baru untuk {$this->survey->title}")
            ->line("Survey Anda '{$this->survey->title}' menerima respons baru.")
            ->action('Lihat Detail', route('surveys.results', $this->survey));
    }

    public function toArray($notifiable): array
    {
        return [
            'survey_id' => $this->survey->id,
            'response_id' => $this->responseId,
            'message' => "Respons baru diterima untuk '{$this->survey->title}'",
        ];
    }
}
```

## Cara Integrasi & Pengujian
1. Publikasikan survey, atur limit 1 per IP
2. Isi survey dari browser biasa, submit
3. Coba isi lagi dari browser yang sama, harus ditolak
4. Buka incognito, isi sebagian (draft)
5. Tutup browser, buka lagi, harus melanjutkan draft
6. Cek notifikasi di email dan database notifications
7. Cek auto-save berjalan setiap 30 detik
