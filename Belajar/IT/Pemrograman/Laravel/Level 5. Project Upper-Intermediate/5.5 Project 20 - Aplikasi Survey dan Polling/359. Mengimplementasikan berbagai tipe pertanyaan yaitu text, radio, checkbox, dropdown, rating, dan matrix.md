# 359. Mengimplementasikan Berbagai Tipe Pertanyaan yaitu Text, Radio, Checkbox, Dropdown, Rating, dan Matrix

## Definisi & Target
Implementasi render tiap question type pada halaman publik survey. Target: setiap tipe pertanyaan tampil sesuai fungsinya dan datanya tersimpan dengan benar.

## Kode Implementasi (Per File)

### Livewire SurveyForm
```php
// app/Livewire/SurveyForm.php
namespace App\Livewire;

use App\Models\Survey;
use App\Models\SurveyResponse;
use App\Models\ResponseAnswer;
use Livewire\Component;
use Illuminate\Support\Str;

class SurveyForm extends Component
{
    public Survey $survey;
    public $answers = [];
    public $currentPage = 1;
    public $totalPages = 1;
    public $startedAt;

    protected function rules()
    {
        $rules = [];
        foreach ($this->survey->questions as $question) {
            $key = 'answers.' . $question->id;
            if ($question->is_required) {
                $rules[$key] = $question->type === 'checkbox' ? 'required|array|min:1' : 'required';
            } else {
                $rules[$key] = 'nullable';
            }

            if ($question->type === 'email') {
                $rules[$key] .= '|email';
            } elseif ($question->type === 'number') {
                $rules[$key] .= '|numeric';
            }
        }
        return $rules;
    }

    protected $messages = [
        'answers.*.required' => 'Pertanyaan ini wajib diisi.',
        'answers.*.array' => 'Pilih setidaknya satu opsi.',
        'answers.*.min' => 'Pilih setidaknya satu opsi.',
    ];

    public function mount(Survey $survey)
    {
        $this->survey = $survey->load('questions');
        $this->totalPages = $survey->settings['pagination'] ?? 1;
        $this->answers = collect($survey->questions)->mapWithKeys(fn($q) => [
            $q->id => $q->type === 'checkbox' ? [] : null
        ])->toArray();
    }

    public function startSurvey()
    {
        $token = Str::random(32);
        $responses = SurveyResponse::create([
            'survey_id' => $this->survey->id,
            'respondent_ip' => request()->ip(),
            'respondent_token' => $token,
            'started_at' => now(),
        ]);
        $this->startedAt = now();
        $this->response = $responses;
    }

    public function updatedAnswers($value, $key)
    {
        if ($this->survey->settings['show_live_preview'] ?? false) {
            $this->validateOnly('answers.' . explode('.', $key)[1]);
        }
    }

    public function nextPage()
    {
        $this->validate();
        $this->currentPage++;
    }

    public function previousPage()
    {
        $this->currentPage--;
    }

    public function submit()
    {
        $this->validate();

        if (!$this->response) {
            $token = Str::random(32);
            $this->response = SurveyResponse::create([
                'survey_id' => $this->survey->id,
                'respondent_ip' => request()->ip(),
                'respondent_token' => $token,
                'started_at' => $this->startedAt ?? now(),
            ]);
        }

        foreach ($this->answers as $questionId => $value) {
            ResponseAnswer::updateOrCreate(
                ['response_id' => $this->response->id, 'question_id' => $questionId],
                ['value' => is_array($value) ? json_encode($value) : $value]
            );
        }

        $this->response->update([
            'completed_at' => now(),
            'time_spent' => $this->startedAt ? now()->diffInSeconds($this->startedAt) : null,
        ]);

        return redirect()->route('surveys.thank-you', $this->survey);
    }

    public function render()
    {
        $questions = $this->survey->questions;
        if ($this->survey->settings['pagination'] ?? false) {
            $perPage = ceil($questions->count() / $this->totalPages);
            $questions = $questions->forPage($this->currentPage, $perPage);
        }
        return view('livewire.survey-form', compact('questions'));
    }
}
```

### Blade Survey Form
```blade
{{-- resources/views/livewire/survey-form.blade.php --}}
<div>
    @if (!$response)
        <div class="text-center py-12">
            <h1 class="text-2xl font-bold mb-2">{{ $survey->title }}</h1>
            <p class="text-gray-600 mb-6">{{ $survey->description }}</p>
            <button wire:click="startSurvey" class="btn btn-primary btn-lg">Mulai Survey</button>
        </div>
    @else
        @if ($survey->settings['show_progress_bar'] ?? true)
            <div class="w-full bg-gray-200 rounded-full h-2 mb-6">
                <div class="bg-blue-500 h-2 rounded-full transition-all"
                     style="width: {{ ($this->currentPage / max($this->totalPages, 1)) * 100 }}%">
                </div>
            </div>
        @endif

        <form wire:submit="{{ $this->currentPage < $this->totalPages ? 'nextPage' : 'submit' }}">
            @foreach ($questions as $question)
                <div class="mb-6 p-4 bg-white rounded shadow" wire:key="q-{{ $question->id }}">
                    <label class="block font-medium mb-2">
                        {{ $question->title }}
                        @if ($question->is_required) <span class="text-red-500">*</span> @endif
                    </label>
                    @if ($question->description)
                        <p class="text-sm text-gray-500 mb-2">{{ $question->description }}</p>
                    @endif

                    @switch($question->type)
                        @case('text')
                            <input type="text" wire:model.blur="answers.{{ $question->id }}"
                                   class="w-full border rounded p-2">
                            @break
                        @case('textarea')
                            <textarea wire:model.blur="answers.{{ $question->id }}"
                                      rows="4" class="w-full border rounded p-2"></textarea>
                            @break
                        @case('email')
                            <input type="email" wire:model.blur="answers.{{ $question->id }}"
                                   class="w-full border rounded p-2" placeholder="email@example.com">
                            @break
                        @case('number')
                            <input type="number" wire:model.blur="answers.{{ $question->id }}"
                                   class="w-full border rounded p-2">
                            @break
                        @case('radio')
                            @foreach ($question->options as $option)
                                <label class="flex items-center gap-2 mb-1 cursor-pointer">
                                    <input type="radio" wire:model="answers.{{ $question->id }}"
                                           value="{{ $option }}" class="accent-blue-500">
                                    {{ $option }}
                                </label>
                            @endforeach
                            @break
                        @case('checkbox')
                            @foreach ($question->options as $option)
                                <label class="flex items-center gap-2 mb-1 cursor-pointer">
                                    <input type="checkbox" wire:model="answers.{{ $question->id }}"
                                           value="{{ $option }}" class="accent-blue-500">
                                    {{ $option }}
                                </label>
                            @endforeach
                            @break
                        @case('dropdown')
                            <select wire:model.blur="answers.{{ $question->id }}" class="w-full border rounded p-2">
                                <option value="">-- Pilih --</option>
                                @foreach ($question->options as $option)
                                    <option value="{{ $option }}">{{ $option }}</option>
                                @endforeach
                            </select>
                            @break
                        @case('rating')
                            <div class="flex gap-1 text-3xl" x-data="{ value: @entangle('answers.' . $question->id) }">
                                @for ($i = 1; $i <= ($question->options['max'] ?? 5); $i++)
                                    <button type="button" @click="value = {{ $i }}"
                                            :class="value >= {{ $i }} ? 'text-yellow-400' : 'text-gray-300'"
                                            class="hover:text-yellow-400 transition">&starf;</button>
                                @endfor
                            </div>
                            @break
                        @case('matrix')
                            <table class="w-full border-collapse">
                                <thead>
                                    <tr>
                                        <th></th>
                                        @foreach (($question->options['columns'] ?? []) as $col)
                                            <th class="text-center text-sm p-2">{{ $col }}</th>
                                        @endforeach
                                    </tr>
                                </thead>
                                <tbody>
                                    @foreach (($question->options['rows'] ?? []) as $rowIndex => $row)
                                        <tr>
                                            <td class="p-2 font-medium">{{ $row }}</td>
                                            @foreach (($question->options['columns'] ?? []) as $colIndex => $col)
                                                <td class="text-center">
                                                    <input type="radio"
                                                           wire:model="answers.{{ $question->id }}.{{ $rowIndex }}"
                                                           value="{{ $colIndex }}">
                                                </td>
                                            @endforeach
                                        </tr>
                                    @endforeach
                                </tbody>
                            </table>
                            @break
                    @endswitch

                    @error("answers.{$question->id}")
                        <p class="text-red-500 text-sm mt-1">{{ $message }}</p>
                    @enderror
                </div>
            @endforeach

            <div class="flex justify-between">
                @if ($this->currentPage > 1)
                    <button type="button" wire:click="previousPage" class="btn btn-outline">Sebelumnya</button>
                @else
                    <div></div>
                @endif
                <button type="submit" class="btn btn-primary">
                    {{ $this->currentPage < $this->totalPages ? 'Selanjutnya' : 'Kirim' }}
                </button>
            </div>
        </form>
    @endif
</div>
```

## Cara Integrasi & Pengujian
1. Buat survey dengan tiap question type
2. Kunjungi halaman publik survey
3. Isi jawaban untuk tiap tipe, verifikasi input sesuai
4. Submit dan cek data di database (response_answers)
5. Uji validasi: required, email, number
6. Uji pagination jika diaktifkan
