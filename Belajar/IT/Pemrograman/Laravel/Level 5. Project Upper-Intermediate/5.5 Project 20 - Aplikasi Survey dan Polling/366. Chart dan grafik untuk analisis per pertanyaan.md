# 366. Chart dan Grafik untuk Analisis per Pertanyaan

## Definisi & Target
Visualisasi data respons menggunakan Chart.js: bar chart, pie chart, rating stars, dan word cloud untuk jawaban teks.

## Komponen yang Terlibat
- Chart.js
- Livewire ResultCharts
- Alpine.js untuk inisialisasi chart
- Data API endpoint

### Livewire ResultCharts
```php
// app/Livewire/ResultCharts.php
namespace App\Livewire;

use App\Models\Survey;
use Livewire\Component;

class ResultCharts extends Component
{
    public Survey $survey;
    public $questionData = [];

    public function mount(Survey $survey)
    {
        $this->survey = $survey->load('questions');
        $this->loadChartData();
    }

    public function loadChartData()
    {
        foreach ($this->survey->questions as $question) {
            $answers = $question->answers()
                ->whereHas('response', fn($q) => $q->whereNotNull('completed_at'))
                ->get();

            $data = match ($question->type) {
                'radio', 'dropdown' => $this->pieChartData($answers, $question),
                'checkbox' => $this->checkboxChartData($answers, $question),
                'rating' => $this->ratingChartData($answers, $question),
                'matrix' => $this->matrixChartData($answers, $question),
                default => null,
            };

            if ($data) {
                $this->questionData[$question->id] = $data;
            }
        }
    }

    private function pieChartData($answers, $question)
    {
        $counts = $answers->groupBy('value')->map(fn($g) => $g->count());

        $labels = $counts->keys()->toArray();
        $values = $counts->values()->toArray();

        $colors = ['#3b82f6', '#ef4444', '#10b981', '#f59e0b', '#8b5cf6', '#ec4899', '#14b8a6', '#f97316'];

        return [
            'type' => 'pie',
            'labels' => $labels,
            'datasets' => [[
                'data' => $values,
                'backgroundColor' => array_slice($colors, 0, count($labels)),
            ]],
            'total' => $answers->count(),
        ];
    }

    private function checkboxChartData($answers, $question)
    {
        $allOptions = $question->options ?? [];
        $counts = collect($allOptions)->mapWithKeys(fn($opt) => [$opt => 0]);

        foreach ($answers as $answer) {
            $selected = json_decode($answer->value, true) ?? [];
            foreach ($selected as $opt) {
                if (isset($counts[$opt])) $counts[$opt]++;
            }
        }

        return [
            'type' => 'bar',
            'labels' => $counts->keys()->toArray(),
            'datasets' => [[
                'label' => 'Jumlah Pilihan',
                'data' => $counts->values()->toArray(),
                'backgroundColor' => '#3b82f6',
            ]],
            'total' => $answers->count(),
        ];
    }

    private function ratingChartData($answers, $question)
    {
        $max = $question->options['max'] ?? 5;
        $ratings = collect(range(1, $max))->mapWithKeys(fn($i) => [$i => 0]);
        foreach ($answers as $answer) {
            $val = (int) $answer->value;
            if (isset($ratings[$val])) $ratings[$val]++;
        }

        $avg = $answers->avg('value');

        return [
            'type' => 'bar',
            'labels' => $ratings->keys()->toArray(),
            'datasets' => [[
                'label' => 'Jumlah',
                'data' => $ratings->values()->toArray(),
                'backgroundColor' => collect(range(1, $max))->map(fn($i) =>
                    $i <= round($avg) ? '#f59e0b' : '#e5e7eb'
                )->toArray(),
            ]],
            'average' => round($avg, 1),
            'total' => $answers->count(),
        ];
    }

    private function matrixChartData($answers, $question)
    {
        $rows = $question->options['rows'] ?? [];
        $cols = $question->options['columns'] ?? [];

        $matrix = collect($rows)->mapWithKeys(fn($r) => [
            $r => collect($cols)->mapWithKeys(fn($c) => [$c => 0])->toArray()
        ])->toArray();

        foreach ($answers as $answer) {
            $values = json_decode($answer->value, true) ?? [];
            foreach ($values as $rowIndex => $colIndex) {
                $rowName = $rows[$rowIndex] ?? $rowIndex;
                $colName = $cols[$colIndex] ?? $colIndex;
                if (isset($matrix[$rowName][$colName])) {
                    $matrix[$rowName][$colName]++;
                }
            }
        }

        return [
            'type' => 'matrix',
            'labels' => $cols,
            'rows' => $matrix,
            'total' => $answers->count(),
        ];
    }

    public function render()
    {
        return view('livewire.result-charts');
    }
}
```

### ResultCharts Blade
```blade
{{-- resources/views/livewire/result-charts.blade.php --}}
<div>
    @foreach ($survey->questions as $question)
        @if (isset($this->questionData[$question->id]))
            <div class="bg-white p-6 rounded shadow mb-6" wire:key="chart-{{ $question->id }}">
                <h3 class="font-semibold mb-1">{{ $question->title }}</h3>
                <p class="text-xs text-gray-400 mb-4">
                    {{ $question->type }} &middot;
                    {{ $this->questionData[$question->id]['total'] ?? 0 }} jawaban
                    @if (isset($this->questionData[$question->id]['average']))
                        &middot; Rata-rata: {{ $this->questionData[$question->id]['average'] }}
                    @endif
                </p>
                <canvas id="chart-{{ $question->id }}" height="200"></canvas>
            </div>

            @push('scripts')
            <script>
                document.addEventListener('livewire:init', () => {
                    const ctx = document.getElementById('chart-{{ $question->id }}');
                    if (!ctx) return;

                    const data = @json($this->questionData[$question->id]);

                    new Chart(ctx, {
                        type: data.type === 'pie' ? 'doughnut' : 'bar',
                        data: { labels: data.labels, datasets: data.datasets },
                        options: {
                            responsive: true,
                            plugins: {
                                legend: { display: data.type === 'pie' },
                                tooltip: {
                                    callbacks: {
                                        label: (ctx) => {
                                            const total = data.total || 1;
                                            const pct = ((ctx.raw / total) * 100).toFixed(1);
                                            return `${ctx.raw} (${pct}%)`;
                                        }
                                    }
                                }
                            },
                            scales: data.type !== 'pie' ? {
                                y: { beginAtZero: true, ticks: { precision: 0 } }
                            } : undefined
                        }
                    });
                });
            </script>
            @endpush
        @endif
    @endforeach

    @push('scripts')
    <script src="{{ asset('js/chart.js') }}"></script>
    @endpush
</div>
```

```php
public function exportChartImage($questionId)
{
    // HTML2Canvas approach di frontend
    $this->dispatch('export-chart', questionId: $questionId);
}
```

```javascript
// Tambah di scripts
document.addEventListener('export-chart', (e) => {
    const canvas = document.getElementById(`chart-${e.detail.questionId}`);
    if (canvas) {
        const link = document.createElement('a');
        link.download = `chart-${e.detail.questionId}.png`;
        link.href = canvas.toDataURL('image/png');
        link.click();
    }
});
```
