# 369. Collaboration (Multi-User Edit Survey)

## Definisi & Target
Fitur kolaborasi: beberapa pengguna bisa mengedit survey bersama-sama dengan role viewer, editor, admin.

## Komponen yang Terlibat
- SurveyCollaborator model
- CollaboratorController
- Livewire CollaboratorManager
- Invitation email
- Broadcasting untuk real-time edit

## Perintah Artisan CLI
```bash
php artisan make:controller CollaboratorController
php artisan make:livewire CollaboratorManager
php artisan make:notification SurveyInvitation
```

## Kode Implementasi (Per File)

### Controller
```php
// app/Http/Controllers/CollaboratorController.php
namespace App\Http\Controllers;

use App\Models\Survey;
use App\Models\SurveyCollaborator;
use App\Notifications\CollaboratorInvitation;
use Illuminate\Http\Request;

class CollaboratorController extends Controller
{
    public function index(Survey $survey)
    {
        $this->authorize('manage', $survey);
        $survey->load('collaborators.user');
        return view('surveys.collaborators', compact('survey'));
    }

    public function store(Request $request, Survey $survey)
    {
        $this->authorize('manage', $survey);

        $validated = $request->validate([
            'email' => 'required|email|exists:users,email',
            'role' => 'required|in:viewer,editor,admin',
        ]);

        $user = \App\Models\User::where('email', $validated['email'])->first();

        if ($user->id === $survey->user_id) {
            return back()->with('error', 'Anda adalah pemilik survey.');
        }

        $existing = SurveyCollaborator::where('survey_id', $survey->id)
            ->where('user_id', $user->id)->exists();

        if ($existing) {
            return back()->with('error', 'User sudah menjadi kolaborator.');
        }

        $collaborator = SurveyCollaborator::create([
            'survey_id' => $survey->id,
            'user_id' => $user->id,
            'role' => $validated['role'],
        ]);

        $user->notify(new CollaboratorInvitation($survey, $validated['role']));

        return back()->with('success', "{$user->name} ditambahkan sebagai {$validated['role']}.");
    }

    public function update(Request $request, Survey $survey, SurveyCollaborator $collaborator)
    {
        $this->authorize('manage', $survey);

        $validated = $request->validate([
            'role' => 'required|in:viewer,editor,admin',
        ]);

        $collaborator->update($validated);

        return back()->with('success', 'Role kolaborator diperbarui.');
    }

    public function destroy(Survey $survey, SurveyCollaborator $collaborator)
    {
        $this->authorize('manage', $survey);
        $collaborator->delete();
        return back()->with('success', 'Kolaborator dihapus.');
    }

    public function leave(Survey $survey)
    {
        $collaborator = SurveyCollaborator::where('survey_id', $survey->id)
            ->where('user_id', auth()->id())
            ->firstOrFail();

        $collaborator->delete();

        return redirect()->route('surveys.index')
            ->with('success', 'Anda keluar dari survey.');
    }
}
```

### Notification
```php
// app/Notifications/CollaboratorInvitation.php
namespace App\Notifications;

use App\Models\Survey;
use Illuminate\Bus\Queueable;
use Illuminate\Notifications\Notification;
use Illuminate\Notifications\Messages\MailMessage;

class CollaboratorInvitation extends Notification
{
    use Queueable;

    public function __construct(
        public Survey $survey,
        public string $role
    ) {}

    public function via($notifiable): array
    {
        return ['mail', 'database'];
    }

    public function toMail($notifiable): MailMessage
    {
        $roleLabels = ['viewer' => 'Viewer', 'editor' => 'Editor', 'admin' => 'Admin'];

        return (new MailMessage)
            ->subject("Undangan Kolaborasi: {$this->survey->title}")
            ->line("Anda diundang sebagai **{$roleLabels[$this->role]}** untuk survey '{$this->survey->title}'.")
            ->action('Buka Survey', route('surveys.edit', $this->survey))
            ->line("Terima kasih.");
    }

    public function toArray($notifiable): array
    {
        return [
            'survey_id' => $this->survey->id,
            'role' => $this->role,
            'message' => "Anda diundang sebagai {$this->role} untuk survey '{$this->survey->title}'.",
        ];
    }
}
```

### Livewire CollaboratorManager
```php
// app/Livewire/CollaboratorManager.php
namespace App\Livewire;

use App\Models\Survey;
use App\Models\User;
use Livewire\Component;

class CollaboratorManager extends Component
{
    public Survey $survey;
    public $email = '';
    public $role = 'editor';
    public $search = '';

    public function addCollaborator()
    {
        $this->validate([
            'email' => 'required|email|exists:users,email',
            'role' => 'required|in:viewer,editor,admin',
        ]);

        $user = User::where('email', $this->email)->first();

        if ($user->id === $this->survey->user_id) {
            $this->addError('email', 'Anda adalah pemilik survey.');
            return;
        }

        $existing = $this->survey->collaborators()
            ->where('user_id', $user->id)->exists();

        if ($existing) {
            $this->addError('email', 'User sudah menjadi kolaborator.');
            return;
        }

        $this->survey->collaborators()->create([
            'user_id' => $user->id,
            'role' => $this->role,
        ]);

        $user->notify(new \App\Notifications\CollaboratorInvitation($this->survey, $this->role));

        $this->email = '';
        $this->dispatch('collaborator-added', name: $user->name);
    }

    public function updateRole($collaboratorId, $role)
    {
        $this->survey->collaborators()
            ->where('id', $collaboratorId)
            ->update(['role' => $role]);

        $this->dispatch('role-updated');
    }

    public function removeCollaborator($collaboratorId)
    {
        $this->survey->collaborators()
            ->where('id', $collaboratorId)
            ->delete();

        $this->dispatch('collaborator-removed');
    }

    public function render()
    {
        $collaborators = $this->survey->collaborators()
            ->with('user')
            ->when($this->search, fn($q) => $q->whereHas('user', fn($u) =>
                $u->where('name', 'like', "%{$this->search}%")
                  ->orWhere('email', 'like', "%{$this->search}%")
            ))
            ->get();

        $users = User::where('id', '!=', auth()->id())
            ->whereDoesntHave('collaborations', fn($q) => $q->where('survey_id', $this->survey->id))
            ->take(10)
            ->get();

        return view('livewire.collaborator-manager', compact('collaborators', 'users'));
    }
}
```

### Permission Check (trait)
```php
// app/Traits/HasSurveyPermission.php
namespace App\Traits;

use App\Models\Survey;

trait HasSurveyPermission
{
    public function canViewSurvey(Survey $survey): bool
    {
        return $survey->user_id === $this->id
            || $survey->collaborators()->where('user_id', $this->id)->exists();
    }

    public function canEditSurvey(Survey $survey): bool
    {
        if ($survey->user_id === $this->id) return true;
        $collab = $survey->collaborators()->where('user_id', $this->id)->first();
        return $collab && in_array($collab->role, ['editor', 'admin']);
    }

    public function canManageSurvey(Survey $survey): bool
    {
        if ($survey->user_id === $this->id) return true;
        $collab = $survey->collaborators()->where('user_id', $this->id)->first();
        return $collab && $collab->role === 'admin';
    }
}
```

### Authorization Policy
```php
// app/Policies/SurveyPolicy.php
namespace App\Policies;

use App\Models\Survey;
use App\Models\User;

class SurveyPolicy
{
    public function view(User $user, Survey $survey): bool
    {
        return $user->canViewSurvey($survey);
    }

    public function update(User $user, Survey $survey): bool
    {
        return $user->canEditSurvey($survey);
    }

    public function delete(User $user, Survey $survey): bool
    {
        return $survey->user_id === $user->id;
    }

    public function manage(User $user, Survey $survey): bool
    {
        return $user->canManageSurvey($survey);
    }
}
```
