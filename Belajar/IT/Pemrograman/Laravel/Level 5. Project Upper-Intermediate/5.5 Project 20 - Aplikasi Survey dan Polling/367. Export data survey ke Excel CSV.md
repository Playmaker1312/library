# 367. Export Data Survey ke Excel/CSV

## Definisi & Target
Export semua respons survey ke format Excel (.xlsx) dan CSV. Target: data mudah dianalisis di spreadsheet.

## Komponen yang Terlibat
- SurveyExportController
- Laravel Excel (maatwebsite/laravel-excel)
- Export classes
- Filter respons

## Perintah Artisan CLI
```bash
composer require maatwebsite/excel
php artisan make:export SurveyResponsesExport
php artisan vendor:publish --provider="Maatwebsite\Excel\ExcelServiceProvider"
```

## Kode Implementasi (Per File)

### Export Class
```php
// app/Exports/SurveyResponsesExport.php
namespace App\Exports;

use App\Models\Survey;
use Maatwebsite\Excel\Concerns\FromCollection;
use Maatwebsite\Excel\Concerns\WithHeadings;
use Maatwebsite\Excel\Concerns\WithMapping;
use Maatwebsite\Excel\Concerns\WithTitle;
use Maatwebsite\Excel\Concerns\ShouldAutoSize;
use Maatwebsite\Excel\Concerns\WithStyles;
use PhpOffice\PhpSpreadsheet\Worksheet\Worksheet;

class SurveyResponsesExport implements FromCollection, WithHeadings, WithMapping, WithTitle, ShouldAutoSize, WithStyles
{
    protected $survey;

    public function __construct(public Survey $survey)
    {
        $this->survey = $survey->load('questions');
    }

    public function collection()
    {
        return $this->survey->responses()
            ->whereNotNull('completed_at')
            ->with('answers')
            ->get();
    }

    public function headings(): array
    {
        $headings = ['ID', 'Respondent Email', 'IP', 'Started At', 'Completed At', 'Time Spent (s)'];

        foreach ($this->survey->questions as $question) {
            $headings[] = $question->title;
        }

        return $headings;
    }

    public function map($response): array
    {
        $answersById = $response->answers->keyBy('question_id');
        $row = [
            $response->id,
            $response->respondent_email ?? '-',
            $response->respondent_ip ?? '-',
            $response->started_at ? $response->started_at->format('Y-m-d H:i:s') : '-',
            $response->completed_at ? $response->completed_at->format('Y-m-d H:i:s') : '-',
            $response->time_spent ?? '-',
        ];

        foreach ($this->survey->questions as $question) {
            $answer = $answersById->get($question->id);
            if (!$answer) {
                $row[] = '-';
                continue;
            }

            $value = $answer->value;
            if ($question->type === 'checkbox') {
                $decoded = json_decode($value, true);
                $row[] = is_array($decoded) ? implode('; ', $decoded) : $value;
            } elseif (in_array($question->type, ['radio', 'dropdown', 'text', 'textarea', 'email', 'number'])) {
                $row[] = $value;
            } elseif ($question->type === 'rating') {
                $row[] = $value;
            } elseif ($question->type === 'matrix') {
                $decoded = json_decode($value, true);
                $row[] = is_array($decoded) ? json_encode($decoded) : $value;
            } else {
                $row[] = $value;
            }
        }

        return $row;
    }

    public function title(): string
    {
        return substr($this->survey->title, 0, 31);
    }

    public function styles(Worksheet $sheet)
    {
        return [
            1 => ['font' => ['bold' => true]],
        ];
    }
}
```

### Multiple Sheet Export (per pertanyaan)
```php
// app/Exports/SurveyDetailedExport.php
namespace App\Exports;

use App\Models\Survey;
use Maatwebsite\Excel\Concerns\WithMultipleSheets;

class SurveyDetailedExport implements WithMultipleSheets
{
    public function __construct(public Survey $survey) {}

    public function sheets(): array
    {
        $sheets = [];

        // Sheet 1: All Responses
        $sheets[] = new SurveyResponsesExport($this->survey);

        // Per-question sheets
        $this->survey->load('questions');
        foreach ($this->survey->questions as $question) {
            $sheets[] = new QuestionDetailSheet($this->survey, $question);
        }

        return $sheets;
    }
}
```

```php
// app/Exports/QuestionDetailSheet.php
namespace App\Exports;

use App\Models\Question;
use App\Models\Survey;
use Maatwebsite\Excel\Concerns\FromArray;
use Maatwebsite\Excel\Concerns\WithTitle;
use Maatwebsite\Excel\Concerns\WithHeadings;

class QuestionDetailSheet implements FromArray, WithTitle, WithHeadings
{
    public function __construct(
        public Survey $survey,
        public Question $question
    ) {}

    public function array(): array
    {
        $answers = $this->question->answers()
            ->whereHas('response', fn($q) => $q->whereNotNull('completed_at'))
            ->get();

        if (in_array($this->question->type, ['radio', 'dropdown'])) {
            $counts = $answers->groupBy('value')->map(fn($g) => $g->count());
            return $counts->map(fn($count, $opt) => [
                'option' => $opt,
                'count' => $count,
                'percentage' => $counts->sum() > 0
                    ? round(($count / $counts->sum()) * 100, 1) . '%'
                    : '0%',
            ])->values()->toArray();
        }

        if ($this->question->type === 'rating') {
            $max = $this->question->options['max'] ?? 5;
            $ratings = collect(range(1, $max))->map(fn($i) => [
                'rating' => $i,
                'count' => $answers->where('value', $i)->count(),
            ]);
            $ratings->push([
                'rating' => 'Average',
                'count' => round($answers->avg('value'), 2),
            ]);
            return $ratings->toArray();
        }

        return $answers->map(fn($a) => [
            'response_id' => $a->response_id,
            'answer' => $a->value,
        ])->toArray();
    }

    public function headings(): array
    {
        $type = $this->question->type;
        if (in_array($type, ['radio', 'dropdown'])) return ['Option', 'Count', 'Percentage'];
        if ($type === 'rating') return ['Rating', 'Count'];
        return ['Response ID', 'Answer'];
    }

    public function title(): string
    {
        return substr($this->question->title, 0, 31);
    }
}
```

### Controller
```php
// app/Http/Controllers/SurveyExportController.php
namespace App\Http\Controllers;

use App\Exports\SurveyResponsesExport;
use App\Exports\SurveyDetailedExport;
use App\Models\Survey;
use Maatwebsite\Excel\Facades\Excel;

class SurveyExportController extends Controller
{
    public function exportExcel(Survey $survey)
    {
        $this->authorize('viewResults', $survey);
        return Excel::download(
            new SurveyResponsesExport($survey),
            "survey-{$survey->id}-responses.xlsx"
        );
    }

    public function exportDetailed(Survey $survey)
    {
        $this->authorize('viewResults', $survey);
        return Excel::download(
            new SurveyDetailedExport($survey),
            "survey-{$survey->id}-detailed.xlsx"
        );
    }

    public function exportCsv(Survey $survey)
    {
        $this->authorize('viewResults', $survey);
        return Excel::download(
            new SurveyResponsesExport($survey),
            "survey-{$survey->id}-responses.csv",
            \Maatwebsite\Excel\Excel::CSV
        );
    }
}
```

### Routes
```php
// routes/web.php
Route::middleware('auth')->prefix('surveys/{survey}/export')->name('surveys.export.')->group(function () {
    Route::get('/excel', [SurveyExportController::class, 'exportExcel'])->name('excel');
    Route::get('/detailed', [SurveyExportController::class, 'exportDetailed'])->name('detailed');
    Route::get('/csv', [SurveyExportController::class, 'exportCsv'])->name('csv');
});
```

## Cara Integrasi & Pengujian
1. Kumpulkan minimal 3-5 respons survey
2. Klik Export Excel, verifikasi file terdownload
3. Buka file Excel: heading benar, data per kolom sesuai pertanyaan
4. Uji Export CSV, buka di spreadsheet
5. Uji Detailed Export, cek multiple sheets
6. Cek data checkbox: multiple values terpisah ;
7. Cek rating: numeric value
