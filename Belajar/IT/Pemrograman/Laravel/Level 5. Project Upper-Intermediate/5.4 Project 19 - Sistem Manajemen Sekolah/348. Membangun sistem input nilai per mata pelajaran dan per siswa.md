# 348. Membangun Sistem Input Nilai per Mata Pelajaran dan per Siswa

## Definisi & Target
Guru mata pelajaran dapat menginput nilai tugas, UTS, UAS untuk setiap siswa di kelasnya. Target: input nilai mudah dengan tabel per kelas.

## Komponen yang Terlibat
- GradeController
- Grade model
- Livewire grade input
- Perhitungan nilai akhir
- Grade otomatis

## Perintah Artisan CLI
```bash
php artisan make:controller GradeController
php artisan make:livewire GradeInput
```

## Kode Implementasi (Per File)

### Controller
```php
// app/Http/Controllers/GradeController.php
namespace App\Http\Controllers;

use App\Helpers\GradeHelper;
use App\Models\Classroom;
use App\Models\Grade;
use App\Models\Student;
use App\Models\Subject;
use App\Models\AcademicYear;
use Illuminate\Http\Request;

class GradeController extends Controller
{
    public function index(Request $request)
    {
        $classes = Classroom::orderBy('level')->orderBy('name')->get();
        $subjects = Subject::all();

        $selectedClass = null;
        $selectedSubject = null;
        $students = collect();
        $grades = collect();

        if ($request->filled('class_id') && $request->filled('subject_id')) {
            $selectedClass = Classroom::find($request->class_id);
            $selectedSubject = Subject::find($request->subject_id);

            $students = Student::where('class_id', $request->class_id)
                ->where('status', 'active')
                ->orderBy('name')
                ->get();

            $grades = Grade::where('class_id', $request->class_id)
                ->where('subject_id', $request->subject_id)
                ->where('academic_year', AcademicYear::getActiveYear())
                ->where('semester', AcademicYear::getActiveSemester())
                ->get()
                ->keyBy('student_id');
        }

        return view('grades.index', compact(
            'classes', 'subjects', 'selectedClass',
            'selectedSubject', 'students', 'grades'
        ));
    }

    public function store(Request $request)
    {
        $request->validate([
            'class_id' => 'required|exists:classes,id',
            'subject_id' => 'required|exists:subjects,id',
            'grades' => 'required|array',
            'grades.*.student_id' => 'required|exists:students,id',
            'grades.*.assignment_score' => 'nullable|numeric|min:0|max:100',
            'grades.*.uts_score' => 'nullable|numeric|min:0|max:100',
            'grades.*.uas_score' => 'nullable|numeric|min:0|max:100',
        ]);

        $academicYear = AcademicYear::getActiveYear();
        $semester = AcademicYear::getActiveSemester();

        foreach ($request->grades as $data) {
            $assignment = $data['assignment_score'] ?? null;
            $uts = $data['uts_score'] ?? null;
            $uas = $data['uas_score'] ?? null;

            $finalScore = null;
            $grade = null;

            if ($assignment !== null && $uts !== null && $uas !== null) {
                $finalScore = GradeHelper::calculateFinal($assignment, $uts, $uas);
                $grade = GradeHelper::getGrade($finalScore);
            }

            Grade::updateOrCreate(
                [
                    'student_id' => $data['student_id'],
                    'subject_id' => $request->subject_id,
                    'class_id' => $request->class_id,
                    'academic_year' => $academicYear,
                    'semester' => $semester,
                ],
                [
                    'assignment_score' => $assignment,
                    'uts_score' => $uts,
                    'uas_score' => $uas,
                    'final_score' => $finalScore,
                    'grade' => $grade,
                    'teacher_id' => auth()->id(),
                ]
            );
        }

        return back()->with('success', 'Nilai berhasil disimpan.');
    }
}
```

### Livewire Component
```php
// app/Livewire/GradeInput.php
namespace App\Livewire;

use App\Helpers\GradeHelper;
use App\Models\AcademicYear;
use App\Models\Classroom;
use App\Models\Grade;
use App\Models\Student;
use App\Models\Subject;
use Livewire\Component;

class GradeInput extends Component
{
    public $classId;
    public $subjectId;
    public $grades = [];

    protected $rules = [
        'grades.*.assignment_score' => 'nullable|numeric|min:0|max:100',
        'grades.*.uts_score' => 'nullable|numeric|min:0|max:100',
        'grades.*.uas_score' => 'nullable|numeric|min:0|max:100',
    ];

    public function mount($classId, $subjectId)
    {
        $this->classId = $classId;
        $this->subjectId = $subjectId;
        $this->loadGrades();
    }

    public function loadGrades()
    {
        $students = Student::where('class_id', $this->classId)
            ->where('status', 'active')
            ->orderBy('name')
            ->get();

        $existingGrades = Grade::where('class_id', $this->classId)
            ->where('subject_id', $this->subjectId)
            ->where('academic_year', AcademicYear::getActiveYear())
            ->where('semester', AcademicYear::getActiveSemester())
            ->get()
            ->keyBy('student_id');

        $this->grades = $students->map(function ($student) use ($existingGrades) {
            $g = $existingGrades->get($student->id);
            return [
                'student_id' => $student->id,
                'student_name' => $student->name,
                'nis' => $student->nis,
                'assignment_score' => $g?->assignment_score,
                'uts_score' => $g?->uts_score,
                'uas_score' => $g?->uas_score,
                'final_score' => $g?->final_score,
                'grade' => $g?->grade,
            ];
        })->toArray();
    }

    public function calculate($index)
    {
        $g = $this->grades[$index];
        if ($g['assignment_score'] !== null && $g['uts_score'] !== null && $g['uas_score'] !== null) {
            $final = GradeHelper::calculateFinal(
                $g['assignment_score'], $g['uts_score'], $g['uas_score']
            );
            $this->grades[$index]['final_score'] = round($final, 2);
            $this->grades[$index]['grade'] = GradeHelper::getGrade($final);
        }
    }

    public function saveAll()
    {
        $this->validate();

        $academicYear = AcademicYear::getActiveYear();
        $semester = AcademicYear::getActiveSemester();

        foreach ($this->grades as $data) {
            Grade::updateOrCreate(
                [
                    'student_id' => $data['student_id'],
                    'subject_id' => $this->subjectId,
                    'class_id' => $this->classId,
                    'academic_year' => $academicYear,
                    'semester' => $semester,
                ],
                [
                    'assignment_score' => $data['assignment_score'],
                    'uts_score' => $data['uts_score'],
                    'uas_score' => $data['uas_score'],
                    'final_score' => $data['final_score'],
                    'grade' => $data['grade'],
                    'teacher_id' => auth()->id(),
                ]
            );
        }

        session()->flash('success', 'Semua nilai berhasil disimpan!');
    }

    public function render()
    {
        return view('livewire.grade-input');
    }
}
```

## Cara Integrasi & Pengujian
1. Pilih kelas dan mata pelajaran
2. Input nilai tugas, UTS, UAS untuk setiap siswa
3. Nilai akhir dan grade otomatis terhitung
4. Klik "Simpan Semua Nilai"
5. Refresh halaman, nilai tetap tersimpan
6. Edit nilai yang sudah ada, update
7. Pastikan GradeHelper::calculateFinal berfungsi dengan bobot yang benar
