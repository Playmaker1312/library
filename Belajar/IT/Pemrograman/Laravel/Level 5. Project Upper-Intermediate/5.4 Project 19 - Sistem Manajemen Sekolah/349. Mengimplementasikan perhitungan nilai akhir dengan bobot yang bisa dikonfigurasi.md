# 349. Mengimplementasikan Perhitungan Nilai Akhir dengan Bobot yang Bisa Dikonfigurasi

## Definisi & Target
Sistem perhitungan nilai akhir dengan bobot tugas, UTS, UAS yang bisa dikonfigurasi per mata pelajaran. Target: fleksibilitas penilaian.

## Komponen yang Terlibat
- GradeWeight model & migration
- GradeWeightController
- GradeHelper dengan bobot dinamis
- Konfigurasi per subject

```bash
php artisan make:model GradeWeight -m
php artisan make:controller GradeWeightController
```

### Model
```php
// app/Models/GradeWeight.php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class GradeWeight extends Model
{
    protected $fillable = [
        'subject_id', 'class_id',
        'assignment_weight', 'uts_weight', 'uas_weight',
        'passing_grade'
    ];

    protected $casts = [
        'assignment_weight' => 'decimal:2',
        'uts_weight' => 'decimal:2',
        'uas_weight' => 'decimal:2',
        'passing_grade' => 'decimal:2',
    ];

    public function subject()
    {
        return $this->belongsTo(Subject::class);
    }

    public function classroom()
    {
        return $this->belongsTo(Classroom::class, 'class_id');
    }

    public static function getWeight($subjectId = null, $classId = null)
    {
        $weight = null;

        if ($subjectId && $classId) {
            $weight = static::where('subject_id', $subjectId)
                ->where('class_id', $classId)
                ->first();
        }

        if (!$weight && $subjectId) {
            $weight = static::where('subject_id', $subjectId)
                ->whereNull('class_id')
                ->first();
        }

        if (!$weight) {
            $weight = (object) [
                'assignment_weight' => 25,
                'uts_weight' => 35,
                'uas_weight' => 40,
                'passing_grade' => 60,
            ];
        }

        return $weight;
    }
}
```

### Updated GradeHelper (dinamis)
```php
// app/Helpers/GradeHelper.php
namespace App\Helpers;

use App\Models\GradeWeight;

class GradeHelper
{
    public static function calculateFinal($assignment, $uts, $uas, $subjectId = null, $classId = null)
    {
        $weight = GradeWeight::getWeight($subjectId, $classId);

        return ($assignment * $weight->assignment_weight / 100) +
               ($uts * $weight->uts_weight / 100) +
               ($uas * $weight->uas_weight / 100);
    }

    public static function getGrade($score)
    {
        return match (true) {
            $score >= 92 => 'A',
            $score >= 83 => 'B+',
            $score >= 75 => 'B',
            $score >= 67 => 'C+',
            $score >= 60 => 'C',
            $score >= 50 => 'D',
            default => 'E',
        };
    }

    public static function getPredicate($grade)
    {
        return match ($grade) {
            'A' => 'Sangat Baik',
            'B+', 'B' => 'Baik',
            'C+', 'C' => 'Cukup',
            'D' => 'Kurang',
            'E' => 'Sangat Kurang',
            default => '-',
        };
    }

    public static function isPassed($score, $subjectId = null, $classId = null)
    {
        $weight = GradeWeight::getWeight($subjectId, $classId);
        return $score >= $weight->passing_grade;
    }
}
```

### Controller
```php
// app/Http/Controllers/GradeWeightController.php
namespace App\Http\Controllers;

use App\Models\GradeWeight;
use App\Models\Subject;
use App\Models\Classroom;
use Illuminate\Http\Request;

class GradeWeightController extends Controller
{
    public function index(Request $request)
    {
        $weights = GradeWeight::with(['subject', 'classroom'])->get();
        $subjects = Subject::all();
        $classes = Classroom::all();

        return view('grade-weights.index', compact('weights', 'subjects', 'classes'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'subject_id' => 'nullable|exists:subjects,id',
            'class_id' => 'nullable|exists:classes,id',
            'assignment_weight' => 'required|numeric|min:0|max:100',
            'uts_weight' => 'required|numeric|min:0|max:100',
            'uas_weight' => 'required|numeric|min:0|max:100',
            'passing_grade' => 'required|numeric|min:0|max:100',
        ]);

        $total = $request->assignment_weight + $request->uts_weight + $request->uas_weight;
        if ($total !== 100) {
            return back()->withErrors(['total' => 'Total bobot harus 100%. Saat ini: ' . $total . '%'])->withInput();
        }

        GradeWeight::updateOrCreate(
            [
                'subject_id' => $request->subject_id,
                'class_id' => $request->class_id,
            ],
            $request->only(['assignment_weight', 'uts_weight', 'uas_weight', 'passing_grade'])
        );

        return back()->with('success', 'Bobot nilai berhasil disimpan.');
    }

    public function update(Request $request, GradeWeight $gradeWeight)
    {
        $request->validate([
            'assignment_weight' => 'required|numeric|min:0|max:100',
            'uts_weight' => 'required|numeric|min:0|max:100',
            'uas_weight' => 'required|numeric|min:0|max:100',
            'passing_grade' => 'required|numeric|min:0|max:100',
        ]);

        $total = $request->assignment_weight + $request->uts_weight + $request->uas_weight;
        if ($total !== 100) {
            return back()->withErrors(['total' => 'Total bobot harus 100%.'])->withInput();
        }

        $gradeWeight->update($request->all());

        return back()->with('success', 'Bobot nilai diperbarui.');
    }

    public function destroy(GradeWeight $gradeWeight)
    {
        $gradeWeight->delete();
        return back()->with('success', 'Bobot nilai dihapus.');
    }

    public function recalculateAll(Request $request)
    {
        $subjectId = $request->subject_id;
        $classId = $request->class_id;

        $grades = \App\Models\Grade::query()
            ->when($subjectId, fn($q) => $q->where('subject_id', $subjectId))
            ->when($classId, fn($q) => $q->where('class_id', $classId))
            ->whereNotNull(['assignment_score', 'uts_score', 'uas_score'])
            ->get();

        $count = 0;
        foreach ($grades as $grade) {
            $final = self::calculateFinal(
                $grade->assignment_score,
                $grade->uts_score,
                $grade->uas_score,
                $grade->subject_id,
                $grade->class_id
            );
            $grade->update([
                'final_score' => $final,
                'grade' => self::getGrade($final),
            ]);
            $count++;
        }

        return back()->with('success', "{$count} nilai berhasil dihitung ulang.");
    }
}
```

## Cara Integrasi & Pengujian
1. Akses `/settings/grade-weights`
2. Set bobot: Tugas 20%, UTS 30%, UAS 50% untuk mapel Matematika
3. Input nilai di halaman grade, pastikan perhitungan sesuai bobot baru
4. Set bobot khusus per kelas
5. Uji validasi: total bobot harus 100%
6. Klik "Hitung Ulang" untuk memperbarui semua nilai yang ada
7. Verifikasi grade berubah sesuai bobot baru
