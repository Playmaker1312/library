# 346. Mengimplementasikan Pembagian Kelas dan Assignment Siswa ke Kelas

## Definisi & Target
Sistem pembagian kelas untuk mengelompokkan siswa berdasarkan tingkat/level. Admin dapat memindahkan siswa antar kelas dan mengatur wali kelas.

## Komponen yang Terlibat
- ClassroomController
- ClassAssignmentController
- Student model (class_id)
- Bulk assignment
- Drag & drop atau select

## Perintah Artisan CLI
```bash
php artisan make:controller ClassroomController --resource
php artisan make:controller ClassAssignmentController
```

## Kode Implementasi (Per File)

### Controller
```php
// app/Http/Controllers/ClassroomController.php
namespace App\Http\Controllers;

use App\Models\Classroom;
use App\Models\Teacher;
use App\Models\Student;
use Illuminate\Http\Request;

class ClassroomController extends Controller
{
    public function index()
    {
        $classes = Classroom::with(['teacher', 'students' => fn($q) => $q->where('status', 'active')])
            ->orderBy('level')
            ->orderBy('name')
            ->get()
            ->groupBy('level');

        return view('classes.index', compact('classes'));
    }

    public function create()
    {
        $teachers = Teacher::all();
        return view('classes.create', compact('teachers'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'name' => 'required|max:10',
            'level' => 'required|integer|min:1|max:12',
            'teacher_id' => 'nullable|exists:teachers,id',
            'academic_year' => 'required|max:9',
        ]);

        Classroom::create($request->all());

        return redirect()->route('classes.index')
            ->with('success', 'Kelas berhasil ditambahkan.');
    }

    public function show(Classroom $classroom)
    {
        $classroom->load(['teacher', 'students' => fn($q) => $q->where('status', 'active')->orderBy('name')]);

        $unassignedStudents = Student::where('status', 'active')
            ->whereNull('class_id')
            ->orWhere('class_id', '!=', $classroom->id)
            ->where('status', 'active')
            ->where('entry_year', '<=', date('Y'))
            ->get();

        return view('classes.show', compact('classroom', 'unassignedStudents'));
    }

    public function edit(Classroom $classroom)
    {
        $teachers = Teacher::all();
        return view('classes.edit', compact('classroom', 'teachers'));
    }

    public function update(Request $request, Classroom $classroom)
    {
        $request->validate([
            'name' => 'required|max:10',
            'level' => 'required|integer|min:1|max:12',
            'teacher_id' => 'nullable|exists:teachers,id',
            'academic_year' => 'required|max:9',
        ]);

        $classroom->update($request->all());

        return redirect()->route('classes.index')
            ->with('success', 'Kelas diperbarui.');
    }

    public function destroy(Classroom $classroom)
    {
        if ($classroom->students()->exists()) {
            return back()->with('error', 'Kelas masih memiliki siswa. Pindahkan siswa terlebih dahulu.');
        }

        $classroom->delete();
        return redirect()->route('classes.index')->with('success', 'Kelas dihapus.');
    }
}
```

```php
// app/Http/Controllers/ClassAssignmentController.php
namespace App\Http\Controllers;

use App\Models\Classroom;
use App\Models\Student;
use Illuminate\Http\Request;

class ClassAssignmentController extends Controller
{
    public function assign(Request $request, Classroom $classroom)
    {
        $request->validate([
            'student_ids' => 'required|array',
            'student_ids.*' => 'exists:students,id',
        ]);

        Student::whereIn('id', $request->student_ids)
            ->update(['class_id' => $classroom->id]);

        return redirect()->route('classes.show', $classroom)
            ->with('success', count($request->student_ids) . ' siswa ditambahkan ke kelas.');
    }

    public function remove(Classroom $classroom, Student $student)
    {
        if ($student->class_id !== $classroom->id) {
            return back()->with('error', 'Siswa tidak berada di kelas ini.');
        }

        $student->update(['class_id' => null]);

        return back()->with('success', 'Siswa dikeluarkan dari kelas.');
    }

    public function bulkMove(Request $request)
    {
        $request->validate([
            'from_class_id' => 'required|exists:classes,id',
            'to_class_id' => 'required|exists:classes,id|different:from_class_id',
            'student_ids' => 'required|array',
            'student_ids.*' => 'exists:students,id',
        ]);

        Student::whereIn('id', $request->student_ids)
            ->update(['class_id' => $request->to_class_id]);

        return back()->with('success', 'Siswa berhasil dipindahkan.');
    }

    public function promote(Request $request)
    {
        $request->validate([
            'from_level' => 'required|integer',
            'to_level' => 'required|integer',
        ]);

        $classes = Classroom::where('level', $request->from_level)->get();

        foreach ($classes as $class) {
            $targetClass = Classroom::where('level', $request->to_level)
                ->where('name', $class->name)
                ->first();

            if ($targetClass) {
                Student::where('class_id', $class->id)
                    ->where('status', 'active')
                    ->update(['class_id' => $targetClass->id]);
            }
        }

        return back()->with('success', 'Kenaikan kelas berhasil diproses.');
    }
}
```

### Routes
```php
// routes/web.php
use App\Http\Controllers\ClassroomController;
use App\Http\Controllers\ClassAssignmentController;

Route::middleware(['auth', 'role:admin'])->group(function () {
    Route::resource('classes', ClassroomController::class);
    Route::post('/classes/{classroom}/assign', [ClassAssignmentController::class, 'assign'])->name('classes.assign');
    Route::delete('/classes/{classroom}/students/{student}', [ClassAssignmentController::class, 'remove'])->name('classes.remove-student');
    Route::post('/classes/bulk-move', [ClassAssignmentController::class, 'bulkMove'])->name('classes.bulk-move');
    Route::post('/classes/promote', [ClassAssignmentController::class, 'promote'])->name('classes.promote');
});
```

## Cara Integrasi & Pengujian
1. Buat kelas baru dengan level, nama, wali kelas
2. Buka detail kelas, tambahkan siswa dari dropdown
3. Hapus siswa dari kelas
4. Edit kelas (ganti wali kelas)
5. Uji hapus kelas dengan siswa (harus ditolak)
6. Uji promosi/kenaikan kelas
7. Verifikasi student.class_id terupdate di database
