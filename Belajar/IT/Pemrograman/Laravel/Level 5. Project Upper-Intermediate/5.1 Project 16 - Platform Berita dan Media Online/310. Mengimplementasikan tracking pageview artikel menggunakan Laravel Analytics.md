# 310. Mengimplementasikan Tracking Pageview Artikel Menggunakan Laravel Analytics

## Definisi & Target
Melacak jumlah pageview artikel untuk analisis popularitas konten. Target: mencatat setiap kunjungan artikel, menyediakan data statistik harian/mingguan/bulanan.

## Komponen yang Terlibat
- PageView model & migration
- AnalyticsService
- Middleware pageview tracker
- Database atau cache untuk penyimpanan
- Dashboard statistik

## Perintah Artisan CLI
```bash
php artisan make:model PageView -m
php artisan make:model AnalyticsDaily -m
php artisan make:service AnalyticsService
php artisan make:middleware TrackPageview
php artisan make:controller AnalyticsController
```

### Middleware
```php
// app/Http/Middleware/TrackPageview.php
namespace App\Http\Middleware;

use App\Models\AnalyticsDaily;
use App\Models\PageView;
use Closure;
use Illuminate\Http\Request;
use Illuminate\Support\Facades\Cache;
use Jenssegers\Agent\Agent;

class TrackPageview
{
    public function handle(Request $request, Closure $next)
    {
        $response = $next($request);

        if ($response->isSuccessful() && $request->route() && !$request->is('livewire*')) {
            $viewable = $request->route('article') ?? $request->route('post');

            if ($viewable && method_exists($viewable, 'getKey')) {
                $this->recordView($request, $viewable);
            }
        }

        return $response;
    }

    protected function recordView(Request $request, $viewable)
    {
        $ip = $request->ip();
        $date = now()->toDateString();
        $key = "pageview_{$viewable->getMorphClass()}_{$viewable->getKey()}_{$ip}_{$date}";

        if (Cache::has($key)) return;

        Cache::put($key, true, now()->addDay());

        PageView::create([
            'viewable_type' => $viewable->getMorphClass(),
            'viewable_id' => $viewable->getKey(),
            'ip' => $ip,
            'user_agent' => $request->userAgent(),
            'referer' => $request->header('referer'),
            'user_id' => auth()->id(),
        ]);

        AnalyticsDaily::updateOrCreate(
            [
                'trackable_type' => $viewable->getMorphClass(),
                'trackable_id' => $viewable->getKey(),
                'date' => $date,
            ],
            [
                'views_count' => \DB::raw('views_count + 1'),
                'unique_visitors' => \DB::raw('(SELECT COUNT(DISTINCT ip) FROM page_views WHERE viewable_type = "' . $viewable->getMorphClass() . '" AND viewable_id = ' . $viewable->getKey() . ' AND DATE(created_at) = "' . $date . '")'),
            ]
        );
    }
}
```

### Service
```php
// app/Services/AnalyticsService.php
namespace App\Services;

use App\Models\AnalyticsDaily;
use App\Models\Article;
use Carbon\Carbon;
use Illuminate\Support\Facades\DB;

class AnalyticsService
{
    public function getArticleStats(Article $article, $days = 30)
    {
        return AnalyticsDaily::where('trackable_type', Article::class)
            ->where('trackable_id', $article->id)
            ->where('date', '>=', now()->subDays($days))
            ->orderBy('date')
            ->get();
    }

    public function getTotalViews(Article $article)
    {
        return AnalyticsDaily::where('trackable_type', Article::class)
            ->where('trackable_id', $article->id)
            ->sum('views_count');
    }

    public function getTopArticles($limit = 10, $days = 7)
    {
        return AnalyticsDaily::select(
            'trackable_id',
            DB::raw('SUM(views_count) as total_views'),
            DB::raw('SUM(unique_visitors) as total_visitors')
        )
            ->where('trackable_type', Article::class)
            ->where('date', '>=', now()->subDays($days))
            ->groupBy('trackable_id')
            ->orderByDesc('total_views')
            ->limit($limit)
            ->with('trackable')
            ->get();
    }

    public function getDailyTrend($days = 30)
    {
        return AnalyticsDaily::select(
            'date',
            DB::raw('SUM(views_count) as total_views'),
            DB::raw('SUM(unique_visitors) as total_visitors')
        )
            ->where('trackable_type', Article::class)
            ->where('date', '>=', now()->subDays($days))
            ->groupBy('date')
            ->orderBy('date')
            ->get();
    }

    public function getTotalViewsAll()
    {
        return AnalyticsDaily::where('trackable_type', Article::class)->sum('views_count');
    }

    public function getAverageDailyViews($days = 30)
    {
        $total = AnalyticsDaily::where('trackable_type', Article::class)
            ->where('date', '>=', now()->subDays($days))
            ->sum('views_count');

        return $days > 0 ? round($total / $days) : 0;
    }
}
```

### Controller
```php
// app/Http/Controllers/AnalyticsController.php
namespace App\Http\Controllers;

use App\Models\Article;
use App\Services\AnalyticsService;
use Illuminate\Http\Request;

class AnalyticsController extends Controller
{
    protected AnalyticsService $analytics;

    public function __construct(AnalyticsService $analytics)
    {
        $this->analytics = $analytics;
    }

    public function index()
    {
        $totalViews = $this->analytics->getTotalViewsAll();
        $dailyTrend = $this->analytics->getDailyTrend(30);
        $topArticles = $this->analytics->getTopArticles(10, 7);
        $avgDailyViews = $this->analytics->getAverageDailyViews(30);

        return view('analytics.index', compact(
            'totalViews', 'dailyTrend', 'topArticles', 'avgDailyViews'
        ));
    }

    public function articles(Request $request)
    {
        $query = Article::withCount(['pageViews as total_views'])
            ->where('status', 'published');

        if ($request->filled('period')) {
            $days = match ($request->period) {
                '7' => 7, '30' => 30, '90' => 90, default => 7
            };
            $query->whereHas('analyticsDailies', function ($q) use ($days) {
                $q->where('date', '>=', now()->subDays($days));
            });
        }

        $articles = $query->orderByDesc('total_views')->paginate(20);

        return view('analytics.articles', compact('articles'));
    }

    public function articleDetail(Article $article)
    {
        $stats = $this->analytics->getArticleStats($article, 30);
        $totalViews = $this->analytics->getTotalViews($article);

        return view('analytics.article-detail', compact('article', 'stats', 'totalViews'));
    }
}
```
