# 306. Mengintegrasikan Payment Gateway untuk Pembayaran Subscription

## Definisi & Target
Mengintegrasikan payment gateway (Midtrans atau Xendit) untuk memproses pembayaran subscription secara otomatis.

## Komponen yang Terlibat
- Midtrans/Xendit API
- Payment model
- Webhook handler
- Subscription activation setelah pembayaran

## Perintah Artisan CLI
```bash
composer require midtrans/midtrans-php
php artisan make:controller PaymentController
php artisan make:controller WebhookController
php artisan make:job ActivateSubscriptionAfterPayment
```

```php
// App/Http/Controllers/PaymentController.php
class PaymentController extends Controller
{
    public function checkout(SubscriptionPlan $plan)
    {
        $user = auth()->user();

        $orderId = 'SUB-' . strtoupper(Str::random(10));

        $params = [
            'transaction_details' => [
                'order_id' => $orderId,
                'gross_amount' => (int) $plan->price,
            ],
            'customer_details' => [
                'first_name' => $user->name,
                'email' => $user->email,
            ],
            'item_details' => [[
                'id' => $plan->id,
                'price' => (int) $plan->price,
                'quantity' => 1,
                'name' => $plan->name,
            ]],
        ];

        $snapToken = \Midtrans\Snap::getSnapToken($params);

        Payment::create([
            'user_id' => $user->id,
            'subscription_plan_id' => $plan->id,
            'order_id' => $orderId,
            'amount' => $plan->price,
            'status' => 'pending',
        ]);

        return view('payment.checkout', compact('snapToken', 'plan'));
    }
}
```

```php
// App/Http/Controllers/WebhookController.php
class WebhookController extends Controller
{
    public function midtrans()
    {
        $notif = new \Midtrans\Notification();

        $transaction = $notif->transaction_status;
        $orderId = $notif->order_id;

        $payment = Payment::where('order_id', $orderId)->firstOrFail();

        if (in_array($transaction, ['capture', 'settlement'])) {
            $payment->update(['status' => 'success']);
            ActivateSubscriptionAfterPayment::dispatch($payment);
        } elseif (in_array($transaction, ['deny', 'cancel', 'expire'])) {
            $payment->update(['status' => 'failed']);
        }

        return response('OK', 200);
    }
}
```

```php
// App/Jobs/ActivateSubscriptionAfterPayment.php
class ActivateSubscriptionAfterPayment implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public Payment $payment;

    public function __construct(Payment $payment) { $this->payment = $payment; }

    public function handle(): void
    {
        $this->payment->user->subscriptions()->create([
            'subscription_plan_id' => $this->payment->subscription_plan_id,
            'status' => 'active',
            'starts_at' => now(),
            'ends_at' => $this->payment->plan->interval === 'yearly' ? now()->addYear() : now()->addMonth(),
        ]);
    }
}
```
