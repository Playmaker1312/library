# 321. Mengimplementasikan Proses Pembayaran Pemenang Setelah Lelang Berakhir

## Definisi & Target
Membangun sistem pembayaran untuk pemenang lelang dengan integrasi payment gateway, penanganan batas waktu pembayaran, dan pengelolaan status.

## Komponen yang Terlibat
- PaymentController
- Payment model & migration
- Payment gateway integration (Midtrans/Xendit)
- Payment deadline checker
- Invoice generation

## Perintah Artisan CLI
```bash
php artisan make:model Payment -m
php artisan make:controller PaymentController
php artisan make:command CheckExpiredPayments
```

### Model
```php
// app/Models/Payment.php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Payment extends Model
{
    protected $fillable = [
        'payable_type', 'payable_id', 'user_id', 'amount',
        'status', 'payment_method', 'transaction_id',
        'gateway_response', 'paid_at', 'expired_at'
    ];

    protected $casts = [
        'amount' => 'decimal:2',
        'gateway_response' => 'array',
        'paid_at' => 'datetime',
        'expired_at' => 'datetime',
    ];

    public function payable()
    {
        return $this->morphTo();
    }

    public function user()
    {
        return $this->belongsTo(User::class);
    }

    public function scopePending($query)
    {
        return $query->where('status', 'pending');
    }

    public function scopeSuccess($query)
    {
        return $query->where('status', 'success');
    }

    public function isExpired()
    {
        return $this->expired_at && $this->expired_at->isPast();
    }
}
```

### Payment Service
```php
// app/Services/PaymentService.php
namespace App\Services;

use App\Models\AuctionResult;
use App\Models\Payment;
use Illuminate\Support\Facades\Http;

class PaymentService
{
    protected $serverKey;
    protected $baseUrl;

    public function __construct()
    {
        $this->serverKey = config('services.midtrans.server_key');
        $this->baseUrl = config('services.midtrans.is_production')
            ? 'https://app.midtrans.com/snap/v1'
            : 'https://app.sandbox.midtrans.com/snap/v1';
    }

    public function createTransaction(AuctionResult $result)
    {
        $orderId = 'AUCTION-' . $result->id . '-' . time();

        $payment = Payment::create([
            'payable_type' => AuctionResult::class,
            'payable_id' => $result->id,
            'user_id' => $result->winner_id,
            'amount' => $result->final_amount,
            'status' => 'pending',
            'expired_at' => $result->payment_deadline,
        ]);

        $payload = [
            'transaction_details' => [
                'order_id' => $orderId,
                'gross_amount' => (int) $result->final_amount,
            ],
            'customer_details' => [
                'first_name' => $result->winner->name,
                'email' => $result->winner->email,
            ],
            'expiry' => [
                'duration' => 3,
                'unit' => 'days',
            ],
        ];

        $response = Http::withBasicAuth($this->serverKey, '')
            ->post($this->baseUrl . '/transactions', $payload);

        if ($response->successful()) {
            $data = $response->json();
            $payment->update([
                'transaction_id' => $orderId,
                'gateway_response' => $data,
            ]);

            $result->update(['payment_id' => $payment->id]);

            return [
                'payment' => $payment,
                'redirect_url' => $data['redirect_url'],
                'token' => $data['token'],
            ];
        }

        throw new \Exception('Gagal membuat transaksi: ' . $response->body());
    }

    public function handleNotification($notification)
    {
        $orderId = $notification['order_id'];
        $transactionStatus = $notification['transaction_status'];
        $fraudStatus = $notification['fraud_status'];

        $payment = Payment::where('transaction_id', $orderId)->first();
        if (!$payment) return;

        if ($transactionStatus === 'capture' || $transactionStatus === 'settlement') {
            if ($fraudStatus === 'accept') {
                $payment->update([
                    'status' => 'success',
                    'paid_at' => now(),
                ]);

                $payment->payable->update(['payment_status' => 'paid', 'paid_at' => now()]);
            }
        } elseif ($transactionStatus === 'expire') {
            $payment->update(['status' => 'expired']);
            $payment->payable->update(['payment_status' => 'expired']);
        } elseif ($transactionStatus === 'deny' || $transactionStatus === 'cancel') {
            $payment->update(['status' => 'failed']);
        }
    }
}
```

### Controller
```php
// app/Http/Controllers/PaymentController.php
namespace App\Http\Controllers;

use App\Models\AuctionResult;
use App\Services\PaymentService;
use Illuminate\Http\Request;

class PaymentController extends Controller
{
    protected $paymentService;

    public function __construct(PaymentService $paymentService)
    {
        $this->paymentService = $paymentService;
    }

    public function auctionPayment(AuctionResult $result)
    {
        if ($result->payment_status === 'paid') {
            return redirect()->route('auction-results.show', $result)
                ->with('info', 'Pembayaran sudah dilakukan.');
        }

        if ($result->payment_status === 'expired') {
            return redirect()->route('auction-results.show', $result)
                ->with('error', 'Batas pembayaran telah lewat.');
        }

        try {
            $transaction = $this->paymentService->createTransaction($result);
            return redirect($transaction['redirect_url']);
        } catch (\Exception $e) {
            return back()->with('error', $e->getMessage());
        }
    }

    public function notification(Request $request)
    {
        $this->paymentService->handleNotification($request->all());
        return response()->json(['status' => 'ok']);
    }

    public function finish(Request $request)
    {
        return redirect()->route('auction-results.my-wins')
            ->with('success', 'Pembayaran berhasil diproses.');
    }

    public function unfinish(Request $request)
    {
        return redirect()->route('auction-results.my-wins')
            ->with('error', 'Pembayaran belum selesai.');
    }

    public function error(Request $request)
    {
        return redirect()->route('auction-results.my-wins')
            ->with('error', 'Pembayaran gagal. Silakan coba lagi.');
    }
}
```

### Command Cek Pembayaran Kedaluwarsa
```php
// app/Console/Commands/CheckExpiredPayments.php
namespace App\Console\Commands;

use App\Models\AuctionResult;
use Illuminate\Console\Command;

class CheckExpiredPayments extends Command
{
    protected $signature = 'payments:check-expired';
    protected $description = 'Menandai pembayaran yang melewati batas waktu';

    public function handle()
    {
        $expiredResults = AuctionResult::where('payment_status', 'pending')
            ->where('payment_deadline', '<=', now())
            ->get();

        foreach ($expiredResults as $result) {
            $result->update(['payment_status' => 'expired']);
            if ($result->payment) {
                $result->payment->update(['status' => 'expired']);
            }
        }

        $this->info("{$expiredResults->count()} pembayaran kadaluwarsa.");

        return Command::SUCCESS;
    }
}
```

## Cara Integrasi & Pengujian
1. Konfigurasi Midtrans/Xendit di .env
2. Menangkan lelang (status ended dengan winner)
3. Klik "Bayar Sekarang", diarahkan ke halaman payment gateway
4. Lakukan pembayaran (gunakan kartu uji Midtrans)
5. Verifikasi status berubah menjadi 'paid'
6. Uji kadaluwarsa dengan mengubah payment_deadline ke masa lalu
7. Jalankan `payments:check-expired` untuk verifikasi expired handling
