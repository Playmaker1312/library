# 320. Mengimplementasikan Penentuan Pemenang Otomatis saat Waktu Lelang Habis

## Definisi & Target
Sistem otomatis menentukan pemenang lelang ketika end_time tercapai, mencatat hasil, dan mengirim notifikasi ke pemenang.

## Komponen yang Terlibat
- Artisan command untuk penutupan lelang
- Scheduler
- AuctionResult model
- Notifikasi pemenang
- Queue job

## Perintah Artisan CLI
```bash
php artisan make:command CloseExpiredAuctions
php artisan make:notification AuctionWonNotification
php artisan make:notification AuctionEndedNoWinnerNotification
```

## Kode Implementasi (Per File)

### Artisan Command
```php
// app/Console/Commands/CloseExpiredAuctions.php
namespace App\Console\Commands;

use App\Models\AuctionItem;
use App\Models\AuctionResult;
use App\Models\Bid;
use App\Notifications\AuctionEndedNoWinnerNotification;
use App\Notifications\AuctionWonNotification;
use Illuminate\Console\Command;
use Illuminate\Support\Facades\DB;

class CloseExpiredAuctions extends Command
{
    protected $signature = 'auctions:close-expired';
    protected $description = 'Menutup lelang yang sudah berakhir dan menentukan pemenang';

    public function handle()
    {
        $expiredItems = AuctionItem::where('status', 'active')
            ->where('end_time', '<=', now())
            ->get();

        $count = 0;
        foreach ($expiredItems as $item) {
            DB::transaction(function () use ($item, &$count) {
                $item = AuctionItem::lockForUpdate()->find($item->id);

                if ($item->status !== 'active') return;

                $highestBid = Bid::where('auction_item_id', $item->id)
                    ->orderByDesc('amount')
                    ->first();

                if ($highestBid) {
                    $result = AuctionResult::create([
                        'auction_item_id' => $item->id,
                        'winner_id' => $highestBid->user_id,
                        'winning_bid_id' => $highestBid->id,
                        'final_amount' => $highestBid->amount,
                        'payment_status' => 'pending',
                        'payment_deadline' => now()->addDays(3),
                    ]);

                    $item->update(['status' => 'ended']);

                    $highestBid->user->notify(new AuctionWonNotification($result));
                } else {
                    $item->update(['status' => 'ended']);

                    $item->seller->notify(new AuctionEndedNoWinnerNotification($item));
                }

                $count++;
            });
        }

        $this->info("{$count} lelang telah ditutup.");

        return Command::SUCCESS;
    }
}
```

### Notifications
```php
// app/Notifications/AuctionWonNotification.php
namespace App\Notifications;

use App\Models\AuctionResult;
use Illuminate\Bus\Queueable;
use Illuminate\Notifications\Notification;

class AuctionWonNotification extends Notification
{
    use Queueable;

    public $result;

    public function __construct(AuctionResult $result)
    {
        $this->result = $result;
    }

    public function via($notifiable)
    {
        return ['database', 'mail'];
    }

    public function toArray($notifiable)
    {
        return [
            'type' => 'auction_won',
            'auction_item_id' => $this->result->auction_item_id,
            'auction_title' => $this->result->auctionItem->title,
            'final_amount' => (float) $this->result->final_amount,
            'payment_deadline' => $this->result->payment_deadline->toDateTimeString(),
            'message' => "Selamat! Anda memenangkan lelang {$this->result->auctionItem->title} " .
                "dengan Rp " . number_format($this->result->final_amount) .
                ". Lakukan pembayaran sebelum {$this->result->payment_deadline->format('d M Y H:i')}.",
            'action_url' => route('auction-results.show', $this->result),
        ];
    }

    public function toMail($notifiable)
    {
        return (new \Illuminate\Notifications\Messages\MailMessage)
            ->subject('Selamat! Anda Memenangkan Lelang')
            ->greeting("Halo {$notifiable->name}!")
            ->line("Anda memenangkan lelang {$this->result->auctionItem->title}")
            ->line("Harga akhir: Rp " . number_format($this->result->final_amount))
            ->line("Batas pembayaran: {$this->result->payment_deadline->format('d M Y H:i')}")
            ->action('Bayar Sekarang', route('auction-results.show', $this->result));
    }
}
```

```php
namespace App\Notifications;

use App\Models\AuctionItem;
use Illuminate\Bus\Queueable;

class AuctionEndedNoWinnerNotification extends Notification
{
    use Queueable;
```

### Scheduler
```php
// app/Console/Kernel.php (Laravel 10)
// atau routes/console.php (Laravel 11)

// Laravel 10:
// app/Console/Kernel.php
protected function schedule(Schedule $schedule)
{
    $schedule->command('auctions:close-expired')->everyMinute();
}

// Laravel 11:
// routes/console.php
use Illuminate\Support\Facades\Schedule;

Schedule::command('auctions:close-expired')->everyMinute();
```

### Controller untuk Hasil Lelang
```php
// app/Http/Controllers/AuctionResultController.php
namespace App\Http\Controllers;

use App\Models\AuctionResult;
use Illuminate\Http\Request;

class AuctionResultController extends Controller
{
    public function show(AuctionResult $auctionResult)
    {
        $auctionResult->load(['auctionItem.images', 'winner', 'winningBid']);

        return view('auction-results.show', compact('auctionResult'));
    }

    public function myWins()
    {
        $results = AuctionResult::where('winner_id', auth()->id())
            ->with('auctionItem')
            ->latest()
            ->paginate(10);

        return view('auction-results.my-wins', compact('results'));
    }

    public function mySales()
    {
        $results = AuctionResult::whereHas('auctionItem', function ($q) {
            $q->where('user_id', auth()->id());
        })->with('auctionItem', 'winner')
            ->latest()
            ->paginate(10);

        return view('auction-results.my-sales', compact('results'));
    }
}
```

## Cara Integrasi & Pengujian
1. Buat item lelang dengan end_time 2 menit dari sekarang
2. Jalankan `php artisan auctions:close-expired` secara manual
3. Pastikan status item berubah menjadi 'ended'
4. Verifikasi AuctionResult terbuat dengan winner yang benar
5. Cek database notifications untuk notifikasi pemenang
6. Tambahkan ke scheduler agar berjalan setiap menit
7. Uji lelang tanpa bid verifikasi tidak ada pemenang
