# 325. Mengimplementasikan Sistem Rating dan Review antara Buyer dan Seller

## Definisi & Target
Sistem rating dan review memungkinkan buyer dan seller saling memberi penilaian setelah transaksi selesai. Target: transparansi reputasi pengguna.

## Komponen yang Terlibat
- Review model & migration
- ReviewController
- Rating calculation
- Review display di profil

## Perintah Artisan CLI
```bash
php artisan make:model Review -m
php artisan make:controller ReviewController
php artisan make:livewire ReviewForm
```

### Model
```php
// app/Models/Review.php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;

class Review extends Model
{
    protected $fillable = [
        'from_user_id', 'to_user_id', 'auction_item_id',
        'auction_result_id', 'rating', 'review', 'type'
    ];

    protected $casts = ['rating' => 'integer'];

    public function fromUser()
    {
        return $this->belongsTo(User::class, 'from_user_id');
    }

    public function toUser()
    {
        return $this->belongsTo(User::class, 'to_user_id');
    }

    public function auctionItem()
    {
        return $this->belongsTo(AuctionItem::class);
    }

    public function auctionResult()
    {
        return $this->belongsTo(AuctionResult::class);
    }
}
```

```php
// app/Models/User.php (tambahan)
use App\Models\Review;

public function reviewsReceived()
{
    return $this->hasMany(Review::class, 'to_user_id');
}

public function reviewsGiven()
{
    return $this->hasMany(Review::class, 'from_user_id');
}

public function averageRating()
{
    return $this->reviewsReceived()->avg('rating') ?? 0;
}

public function ratingsCount()
{
    return $this->reviewsReceived()->count();
}

public function ratingDistribution()
{
    return $this->reviewsReceived()
        ->selectRaw('rating, COUNT(*) as total')
        ->groupBy('rating')
        ->pluck('total', 'rating')
        ->toArray();
}
```

### Controller
```php
// app/Http/Controllers/ReviewController.php
namespace App\Http\Controllers;

use App\Models\AuctionResult;
use App\Models\Review;
use App\Models\User;
use Illuminate\Http\Request;

class ReviewController extends Controller
{
    public function createForSeller(AuctionResult $result)
    {
        if ($result->winner_id !== auth()->id()) {
            abort(403);
        }

        $existing = Review::where('from_user_id', auth()->id())
            ->where('auction_result_id', $result->id)
            ->where('type', 'buyer_to_seller')
            ->first();

        if ($existing) {
            return redirect()->route('auction-results.show', $result)
                ->with('info', 'Anda sudah memberikan review.');
        }

        return view('reviews.create-seller', compact('result'));
    }

    public function createForBuyer(AuctionResult $result)
    {
        if ($result->auctionItem->user_id !== auth()->id()) {
            abort(403);
        }

        $existing = Review::where('from_user_id', auth()->id())
            ->where('auction_result_id', $result->id)
            ->where('type', 'seller_to_buyer')
            ->first();

        if ($existing) {
            return redirect()->route('auction-results.show', $result)
                ->with('info', 'Anda sudah memberikan review.');
        }

        return view('reviews.create-buyer', compact('result'));
    }

    public function store(Request $request)
    {
        $request->validate([
            'auction_result_id' => 'required|exists:auction_results,id',
            'to_user_id' => 'required|exists:users,id',
            'rating' => 'required|integer|min:1|max:5',
            'review' => 'nullable|string|max:1000',
            'type' => 'required|in:buyer_to_seller,seller_to_buyer',
        ]);

        $result = AuctionResult::findOrFail($request->auction_result_id);

        $existing = Review::where('from_user_id', auth()->id())
            ->where('auction_result_id', $result->id)
            ->where('type', $request->type)
            ->exists();

        if ($existing) {
            return back()->with('error', 'Anda sudah memberikan review.');
        }

        Review::create([
            'from_user_id' => auth()->id(),
            'to_user_id' => $request->to_user_id,
            'auction_item_id' => $result->auction_item_id,
            'auction_result_id' => $result->id,
            'rating' => $request->rating,
            'review' => $request->review,
            'type' => $request->type,
        ]);

        return redirect()->route('auction-results.show', $result)
            ->with('success', 'Review berhasil dikirim.');
    }

    public function showUserReviews(User $user)
    {
        $reviews = $user->reviewsReceived()
            ->with(['fromUser', 'auctionItem'])
            ->latest()
            ->paginate(10);

        $averageRating = $user->averageRating();
        $ratingDistribution = $user->ratingDistribution();
        $totalReviews = $user->ratingsCount();

        return view('reviews.user', compact(
            'user', 'reviews', 'averageRating',
            'ratingDistribution', 'totalReviews'
        ));
    }
}
```

### Routes
```php
// routes/web.php
use App\Http\Controllers\ReviewController;

Route::middleware('auth')->group(function () {
    Route::get('/reviews/seller/{auctionResult}', [ReviewController::class, 'createForSeller'])->name('reviews.create-seller');
    Route::get('/reviews/buyer/{auctionResult}', [ReviewController::class, 'createForBuyer'])->name('reviews.create-buyer');
    Route::post('/reviews', [ReviewController::class, 'store'])->name('reviews.store');
});

Route::get('/users/{user}/reviews', [ReviewController::class, 'showUserReviews'])->name('users.reviews');
```

## Cara Integrasi & Pengujian
1. Setup migration dan model
2. Buyer menangkan lelang, bayar, konfirmasi terima barang
3. Buyer beri rating dan review untuk seller
4. Seller beri rating untuk buyer
5. Cek halaman review user melihat rating dan distribusi
6. Pastikan tidak bisa double review untuk transaksi yang sama
7. Uji validasi rating 1-5 dan review max 1000 karakter
