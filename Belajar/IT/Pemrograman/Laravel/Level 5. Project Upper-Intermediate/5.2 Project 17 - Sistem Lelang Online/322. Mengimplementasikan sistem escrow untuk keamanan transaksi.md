# 322. Mengimplementasikan Sistem Escrow untuk Keamanan Transaksi

## Definisi & Target
Sistem escrow menahan dana pembayaran sampai buyer mengkonfirmasi penerimaan barang. Target: dana aman, release otomatis atau manual, dispute resolution.

## Komponen yang Terlibat
- Escrow model & migration (sudah di poin 312)
- EscrowController
- EscrowService
- Buyer confirmation flow
- Dispute management


## Kode Implementasi (Per File)

### Service
```php
// app/Services/EscrowService.php
namespace App\Services;

use App\Models\AuctionResult;
use App\Models\Escrow;
use App\Models\Dispute;
use Illuminate\Support\Facades\DB;

class EscrowService
{
    public function createEscrow(AuctionResult $result)
    {
        return Escrow::create([
            'auction_result_id' => $result->id,
            'buyer_id' => $result->winner_id,
            'seller_id' => $result->auctionItem->user_id,
            'amount' => $result->final_amount,
            'status' => 'held',
            'held_at' => now(),
        ]);
    }

    public function releaseToSeller(Escrow $escrow)
    {
        return DB::transaction(function () use ($escrow) {
            if ($escrow->status !== 'held') {
                throw new \Exception('Dana escrow sudah diproses.');
            }

            $escrow->update([
                'status' => 'released',
                'released_at' => now(),
            ]);

            $escrow->auctionResult->update([
                'payment_status' => 'completed',
            ]);

            $this->processPayoutToSeller($escrow);

            return $escrow;
        });
    }

    public function refundToBuyer(Escrow $escrow)
    {
        return DB::transaction(function () use ($escrow) {
            if ($escrow->status !== 'held') {
                throw new \Exception('Dana escrow sudah diproses.');
            }

            $escrow->update([
                'status' => 'refunded',
                'refunded_at' => now(),
            ]);

            $escrow->auctionResult->update([
                'payment_status' => 'refunded',
            ]);

            $this->processRefund($escrow);

            return $escrow;
        });
    }

    public function confirmReceived(Escrow $escrow)
    {
        if ($escrow->buyer_id !== auth()->id()) {
            throw new \Exception('Hanya buyer yang bisa konfirmasi.');
        }

        return $this->releaseToSeller($escrow);
    }

    public function raiseDispute(Escrow $escrow, $reason)
    {
        return Dispute::create([
            'escrow_id' => $escrow->id,
            'raised_by' => auth()->id(),
            'reason' => $reason,
            'status' => 'open',
        ]);
    }

    public function resolveDispute(Dispute $dispute, $resolution, $action)
    {
        return DB::transaction(function () use ($dispute, $resolution, $action) {
            $dispute->update([
                'resolution' => $resolution,
                'status' => 'resolved',
            ]);

            if ($action === 'release') {
                $this->releaseToSeller($dispute->escrow);
            } elseif ($action === 'refund') {
                $this->refundToBuyer($dispute->escrow);
            }

            return $dispute;
        });
    }

    protected function processPayoutToSeller(Escrow $escrow)
    {
        // Integrasi dengan payment gateway untuk payout ke seller
        // Contoh: kirim dana ke rekening seller via API bank/payment gateway
    }

    protected function processRefund(Escrow $escrow)
    {
        // Integrasi refund ke buyer
    }
}
```

### Controller
```php
// app/Http/Controllers/EscrowController.php
namespace App\Http\Controllers;

use App\Models\AuctionResult;
use App\Models\Escrow;
use App\Models\Dispute;
use App\Services\EscrowService;
use Illuminate\Http\Request;

class EscrowController extends Controller
{
    protected $escrowService;

    public function __construct(EscrowService $escrowService)
    {
        $this->escrowService = $escrowService;
    }

    public function show(Escrow $escrow)
    {
        $escrow->load(['auctionResult.auctionItem', 'buyer', 'seller', 'disputes']);
        return view('escrows.show', compact('escrow'));
    }

    public function confirmReceived(Escrow $escrow)
    {
        try {
            $this->escrowService->confirmReceived($escrow);
            return redirect()->route('escrows.show', $escrow)
                ->with('success', 'Barang diterima. Dana telah dilepas ke seller.');
        } catch (\Exception $e) {
            return back()->with('error', $e->getMessage());
        }
    }

    public function disputeForm(Escrow $escrow)
    {
        return view('escrows.dispute', compact('escrow'));
    }

    public function submitDispute(Request $request, Escrow $escrow)
    {
        $request->validate(['reason' => 'required|min:20']);

        try {
            $this->escrowService->raiseDispute($escrow, $request->reason);
            return redirect()->route('escrows.show', $escrow)
                ->with('success', 'Sengketa telah diajukan. Admin akan meninjau.');
        } catch (\Exception $e) {
            return back()->with('error', $e->getMessage());
        }
    }

    public function resolveDispute(Request $request, Dispute $dispute)
    {
        $request->validate([
            'resolution' => 'required',
            'action' => 'required|in:release,refund',
        ]);

        try {
            $this->escrowService->resolveDispute(
                $dispute,
                $request->resolution,
                $request->action
            );
            return redirect()->route('escrows.show', $dispute->escrow)
                ->with('success', 'Sengketa berhasil diselesaikan.');
        } catch (\Exception $e) {
            return back()->with('error', $e->getMessage());
        }
    }

    public function myEscrows()
    {
        $escrows = Escrow::where(function ($q) {
            $q->where('buyer_id', auth()->id())
              ->orWhere('seller_id', auth()->id());
        })->with(['auctionResult.auctionItem'])
            ->latest()
            ->paginate(10);

        return view('escrows.my-escrows', compact('escrows'));
    }
}
```

### Command
```php
// app/Console/Commands/ReleaseConfirmedEscrows.php
namespace App\Console\Commands;

use App\Models\Escrow;
use App\Services\EscrowService;
use Illuminate\Console\Command;

class ReleaseConfirmedEscrows extends Command
{
    protected $signature = 'escrows:release-confirmed';
    protected $description = 'Melepas dana escrow yang sudah dikonfirmasi buyer';

    protected $escrowService;

    public function __construct(EscrowService $escrowService)
    {
        parent::__construct();
        $this->escrowService = $escrowService;
    }

    public function handle()
    {
        // Auto-release setelah 7 hari jika buyer tidak ada tindakan
        $autoRelease = Escrow::where('status', 'held')
            ->where('created_at', '<=', now()->subDays(7))
            ->get();

        foreach ($autoRelease as $escrow) {
            try {
                $this->escrowService->releaseToSeller($escrow);
                $this->info("Escrow {$escrow->id} auto-released.");
            } catch (\Exception $e) {
                $this->error("Escrow {$escrow->id}: {$e->getMessage()}");
            }
        }

        return Command::SUCCESS;
    }
}
```

### Event Listener untuk Escrow Creation
```php
// app/Listeners/CreateEscrowAfterPayment.php
namespace App\Listeners;

use App\Models\Payment;
use App\Services\EscrowService;

class CreateEscrowAfterPayment
{
    protected $escrowService;

    public function __construct(EscrowService $escrowService)
    {
        $this->escrowService = $escrowService;
    }

    public function handle($event)
    {
        $payment = $event->payment;
        if ($payment->payable_type === 'App\Models\AuctionResult'
            && $payment->status === 'success') {
            $this->escrowService->createEscrow($payment->payable);
        }
    }
}
```

### Observer untuk Payment
```php
// app/Observers/PaymentObserver.php
namespace App\Observers;

use App\Models\Payment;
use App\Services\EscrowService;

class PaymentObserver
{
    protected $escrowService;

    public function __construct(EscrowService $escrowService)
    {
        $this->escrowService = $escrowService;
    }

    public function updated(Payment $payment)
    {
        if ($payment->wasChanged('status') && $payment->status === 'success') {
            if ($payment->payable_type === 'App\Models\AuctionResult') {
                $this->escrowService->createEscrow($payment->payable);
            }
        }
    }
}
```

## Cara Integrasi & Pengujian
1. Setup dan pastikan PaymentObserver terdaftar
2. Lakukan pembayaran lelang, escrow otomatis terbuat
3. Login sebagai buyer, konfirmasi penerimaan barang
4. Verifikasi status escrow berubah menjadi 'released'
5. Login sebagai seller, lihat dana sudah dilepas
6. Uji dispute: ajukan sengketa, resolve sebagai admin
7. Uji auto-release: buat escrow dengan tanggal 7 hari lalu, jalankan command
