# 315. Mengimplementasikan Sistem Penawaran atau Bidding secara Real-Time Menggunakan Pusher

## Definisi & Target
Mengimplementasikan sistem bidding real-time menggunakan Pusher/Laravel Broadcasting. Setiap bid baru akan langsung muncul tanpa refresh halaman.

## Komponen yang Terlibat
- Pusher (Laravel Broadcasting)
- BidController
- Bid model & event
- Livewire bid form
- Notifikasi real-time

### Event Broadcasting
```php
// app/Events/NewBidPlaced.php
namespace App\Events;

use App\Models\Bid;
use Illuminate\Broadcasting\Channel;
use Illuminate\Broadcasting\InteractsWithSockets;
use Illuminate\Broadcasting\PresenceChannel;
use Illuminate\Contracts\Broadcasting\ShouldBroadcast;
use Illuminate\Foundation\Events\Dispatchable;
use Illuminate\Queue\SerializesModels;

class NewBidPlaced implements ShouldBroadcast
{
    use Dispatchable, InteractsWithSockets, SerializesModels;

    public $bid;
    public $currentBid;
    public $bidCount;

    public function __construct(Bid $bid)
    {
        $this->bid = $bid;
        $this->currentBid = $bid->amount;
        $this->bidCount = $bid->auctionItem->bids()->count();
    }

    public function broadcastOn()
    {
        return new Channel('auction.' . $this->bid->auction_item_id);
    }

    public function broadcastAs()
    {
        return 'bid.placed';
    }

    public function broadcastWith()
    {
        return [
            'bid_id' => $this->bid->id,
            'user_name' => $this->bid->user->name,
            'amount' => $this->bid->amount,
            'current_bid' => $this->currentBid,
            'bid_count' => $this->bidCount,
            'time' => $this->bid->created_at->diffForHumans(),
        ];
    }
}
```

### Controller
```php
// app/Http/Controllers/BidController.php
namespace App\Http\Controllers;

use App\Events\NewBidPlaced;
use App\Models\AuctionItem;
use App\Models\Bid;
use Illuminate\Http\Request;

class BidController extends Controller
{
    public function store(Request $request, AuctionItem $auctionItem)
    {
        $request->validate([
            'amount' => 'required|numeric|min:' . $auctionItem->minNextBid(),
        ]);

        if (!$auctionItem->isActive()) {
            return back()->with('error', 'Lelang tidak aktif atau sudah berakhir.');
        }

        if ($auctionItem->user_id === auth()->id()) {
            return back()->with('error', 'Anda tidak bisa bid pada item sendiri.');
        }

        $bid = Bid::create([
            'auction_item_id' => $auctionItem->id,
            'user_id' => auth()->id(),
            'amount' => $request->amount,
        ]);

        $auctionItem->update(['current_bid' => $bid->amount]);

        broadcast(new NewBidPlaced($bid))->toOthers();

        return back()->with('success', 'Bid berhasil dikirim!');
    }

    public function history(AuctionItem $auctionItem)
    {
        $bids = $auctionItem->bids()
            ->with('user')
            ->latest()
            ->paginate(20);

        return view('bids.history', compact('auctionItem', 'bids'));
    }
}
```

### Livewire BidForm
```php
// app/Livewire/BidForm.php
namespace App\Livewire;

use App\Events\NewBidPlaced;
use App\Models\AuctionItem;
use App\Models\Bid;
use Livewire\Component;

class BidForm extends Component
{
    public AuctionItem $auctionItem;
    public $amount;
    public $minBid;

    protected function rules()
    {
        return [
            'amount' => ['required', 'numeric', 'min:' . $this->minBid],
        ];
    }

    public function mount(AuctionItem $auctionItem)
    {
        $this->auctionItem = $auctionItem;
        $this->minBid = $auctionItem->minNextBid();
        $this->amount = $this->minBid;
    }

    public function placeBid()
    {
        $this->validate();

        if (!$this->auctionItem->isActive()) {
            session()->flash('error', 'Lelang tidak aktif.');
            return;
        }

        $bid = Bid::create([
            'auction_item_id' => $this->auctionItem->id,
            'user_id' => auth()->id(),
            'amount' => $this->amount,
        ]);

        $this->auctionItem->update(['current_bid' => $this->amount]);
        $this->minBid = $this->auctionItem->minNextBid();
        $this->amount = $this->minBid;

        broadcast(new NewBidPlaced($bid))->toOthers();
        $this->dispatch('bid-placed', amount: $this->amount);
        session()->flash('success', 'Bid berhasil!');
    }

    public function render()
    {
        return view('livewire.bid-form');
    }
}
```

### Client-Side Echo Setup
```javascript
// resources/js/echo.js
import Echo from 'laravel-echo';
import Pusher from 'pusher-js';

window.Pusher = Pusher;
window.Echo = new Echo({
    broadcaster: 'pusher',
    key: import.meta.env.VITE_PUSHER_APP_KEY,
    cluster: import.meta.env.VITE_PUSHER_APP_CLUSTER,
    forceTLS: true,
});
```

```javascript
// resources/js/app.js (tambahan untuk halaman detail lelang)
import './bootstrap';

document.addEventListener('DOMContentLoaded', () => {
    const auctionId = document.body.dataset.auctionId;
    if (!auctionId) return;

    window.Echo.channel(`auction.${auctionId}`)
        .listen('.bid.placed', (e) => {
            const list = document.getElementById('bid-list');
            if (list) {
                const item = document.createElement('div');
                item.className = 'flex justify-between items-center p-2 bg-green-50 rounded border border-green-200';
                item.innerHTML = `
                    <div>
                        <span class="font-medium">${e.user_name}</span>
                        <span class="text-xs text-gray-500 ml-2">${e.time}</span>
                    </div>
                    <span class="font-bold text-green-600">Rp ${new Intl.NumberFormat('id-ID').format(e.amount)}</span>
                `;
                list.prepend(item);
            }

            const currentBidEl = document.getElementById('current-bid');
            if (currentBidEl) {
                currentBidEl.innerText = `Rp ${new Intl.NumberFormat('id-ID').format(e.current_bid)}`;
            }
        });
});
```

## Cara Integrasi & Pengujian
1. Install Pusher dan konfigurasi `.env`
2. Jalankan `npm install && npm run build`
3. Login dengan 2 akun berbeda untuk testing
4. Akun A buka halaman lelang, akun B melakukan bid
5. Pastikan bid muncul real-time di halaman akun A
6. Verifikasi current bid terupdate otomatis
7. Uji validasi: bid lebih rendah dari current bid
