# 317. Mengimplementasikan Auto-Bid atau Proxy Bidding hingga Batas Maksimum

## Definisi & Target
Fitur auto-bid memungkinkan user menentukan batas maksimum harga. Sistem akan otomatis menaikkan bid untuk user ketika ada bid baru yang mendekati batasnya.

## Komponen yang Terlibat
- Bid model (is_auto_bid, max_amount)
- AutoBidService
- Event listener untuk auto-bid trigger
- Queue job auto-bid

## Perintah Artisan CLI
```bash
php artisan make:service AutoBidService
php artisan make:job ProcessAutoBid
```

## Kode Implementasi (Per File)

### Service AutoBid
```php
// app/Services/AutoBidService.php
namespace App\Services;

use App\Events\NewBidPlaced;
use App\Models\AuctionItem;
use App\Models\Bid;
use Illuminate\Support\Facades\DB;

class AutoBidService
{
    public function processAutoBids(AuctionItem $item)
    {
        $autoBids = Bid::where('auction_item_id', $item->id)
            ->where('is_auto_bid', true)
            ->where('max_amount', '>', $item->current_bid ?? 0)
            ->where('user_id', '!=', $item->highestBid?->user_id)
            ->orderBy('max_amount', 'desc')
            ->orderBy('created_at')
            ->get();

        foreach ($autoBids as $autoBid) {
            if (!$item->isActive()) break;

            $currentHigh = $item->current_bid ?? $item->starting_bid;

            if ($autoBid->max_amount <= $currentHigh) continue;

            $nextBid = $currentHigh + $item->bid_increment;

            if ($nextBid > $autoBid->max_amount) continue;

            if ($nextBid <= $currentHigh) continue;

            $bid = Bid::create([
                'auction_item_id' => $item->id,
                'user_id' => $autoBid->user_id,
                'amount' => min($nextBid, $autoBid->max_amount),
                'is_auto_bid' => true,
                'max_amount' => $autoBid->max_amount,
            ]);

            $item->update(['current_bid' => $bid->amount]);
            broadcast(new NewBidPlaced($bid))->toOthers();
        }
    }

    public function placeAutoBid(AuctionItem $item, $userId, $maxAmount)
    {
        return DB::transaction(function () use ($item, $userId, $maxAmount) {
            $item = AuctionItem::lockForUpdate()->find($item->id);

            if (!$item->isActive()) {
                throw new \Exception('Lelang tidak aktif.');
            }

            $existingAuto = Bid::where('auction_item_id', $item->id)
                ->where('user_id', $userId)
                ->where('is_auto_bid', true)
                ->first();

            if ($existingAuto) {
                $existingAuto->update(['max_amount' => $maxAmount]);
                $autoBid = $existingAuto;
            } else {
                $autoBid = Bid::create([
                    'auction_item_id' => $item->id,
                    'user_id' => $userId,
                    'amount' => $item->minNextBid(),
                    'is_auto_bid' => true,
                    'max_amount' => $maxAmount,
                ]);

                $item->update(['current_bid' => $autoBid->amount]);
            }

            $this->processAutoBids($item);

            return $autoBid;
        });
    }
}
```

### Queue Job
```php
// app/Jobs/ProcessAutoBid.php
namespace App\Jobs;

use App\Models\AuctionItem;
use App\Services\AutoBidService;
use Illuminate\Bus\Queueable;
use Illuminate\Contracts\Queue\ShouldQueue;
use Illuminate\Foundation\Bus\Dispatchable;
use Illuminate\Queue\InteractsWithQueue;
use Illuminate\Queue\SerializesModels;

class ProcessAutoBid implements ShouldQueue
{
    use Dispatchable, InteractsWithQueue, Queueable, SerializesModels;

    public $auctionItemId;

    public function __construct($auctionItemId)
    {
        $this->auctionItemId = $auctionItemId;
    }

    public function handle(AutoBidService $service)
    {
        $item = AuctionItem::find($this->auctionItemId);
        if ($item && $item->isActive()) {
            $service->processAutoBids($item);
        }
    }
}
```

### Event Listener untuk Trigger Auto-Bid
```php
// app/Listeners/TriggerAutoBid.php
namespace App\Listeners;

use App\Events\NewBidPlaced;
use App\Jobs\ProcessAutoBid;

class TriggerAutoBid
{
    public function handle(NewBidPlaced $event)
    {
        ProcessAutoBid::dispatch($event->bid->auction_item_id)
            ->delay(now()->addSeconds(2));
    }
}
```

```php
// app/Providers/EventServiceProvider.php
use App\Events\NewBidPlaced;
use App\Listeners\TriggerAutoBid;

protected $listen = [
    NewBidPlaced::class => [
        TriggerAutoBid::class,
    ],
];
```

### Livewire Component AutoBid
```php
// app/Livewire/AutoBidForm.php
namespace App\Livewire;

use App\Models\AuctionItem;
use App\Services\AutoBidService;
use Livewire\Component;

class AutoBidForm extends Component
{
    public AuctionItem $auctionItem;
    public $maxAmount;
    public $isActive = false;
    public $currentMax = 0;

    public function mount(AuctionItem $auctionItem)
    {
        $this->auctionItem = $auctionItem;
        $existing = $auctionItem->bids()
            ->where('user_id', auth()->id())
            ->where('is_auto_bid', true)
            ->first();
        if ($existing) {
            $this->isActive = true;
            $this->currentMax = $existing->max_amount;
        }
    }

    public function setAutoBid(AutoBidService $service)
    {
        $this->validate(['maxAmount' => 'required|numeric|min:' . $this->auctionItem->minNextBid()]);

        $service->placeAutoBid($this->auctionItem, auth()->id(), $this->maxAmount);

        $this->isActive = true;
        $this->currentMax = $this->maxAmount;
        $this->dispatch('auto-bid-set');
    }

    public function cancelAutoBid()
    {
        $this->auctionItem->bids()
            ->where('user_id', auth()->id())
            ->where('is_auto_bid', true)
            ->delete();
        $this->isActive = false;
        $this->currentMax = 0;
    }

    public function render()
    {
        return view('livewire.auto-bid-form');
    }
}
```

## Cara Integrasi & Pengujian
1. User A memasang auto-bid dengan max 500000 pada item dengan current bid 100000
2. User B melakukan bid manual 150000
3. Sistem otomatis menaikkan bid User A ke 155000 (karena < max 500000)
4. User B bid lagi 200000, auto-bid naik ke 205000
5. User B bid 600000, auto-bid User A tidak bisa lanjut karena > max 500000
6. Verifikasi job queue berjalan dengan php artisan queue:work
