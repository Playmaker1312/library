// app/Models/AuctionItem.php
namespace App\Models;

use Illuminate\Database\Eloquent\Model;
use Illuminate\Database\Eloquent\SoftDeletes;
use Illuminate\Support\Str;

class AuctionItem extends Model
{
    use SoftDeletes;

    protected $fillable = [
        'user_id', 'title', 'slug', 'description',
        'starting_bid', 'bid_increment', 'buy_now_price',
        'current_bid', 'start_time', 'end_time', 'status'
    ];

    protected $casts = [
        'start_time' => 'datetime',
        'end_time' => 'datetime',
        'starting_bid' => 'decimal:2',
        'bid_increment' => 'decimal:2',
        'buy_now_price' => 'decimal:2',
        'current_bid' => 'decimal:2',
    ];

    protected static function booted()
    {
        static::creating(function ($item) {
            $item->slug = Str::slug($item->title) . '-' . Str::random(6);
        });
    }

    public function seller()
    {
        return $this->belongsTo(User::class, 'user_id');
    }

    public function bids()
    {
        return $this->hasMany(Bid::class);
    }

    public function images()
    {
        return $this->hasMany(AuctionItemImage::class)->orderBy('sort_order');
    }

    public function result()
    {
        return $this->hasOne(AuctionResult::class);
    }

    public function highestBid()
    {
        return $this->hasOne(Bid::class)->latestOfMany('amount');
    }

    public function scopeActive($query)
    {
        return $query->where('status', 'active')
            ->where('start_time', '<=', now())
            ->where('end_time', '>', now());
    }

    public function scopeEnded($query)
    {
        return $query->where('end_time', '<=', now());
    }

    public function isActive()
    {
        return $this->status === 'active'
            && $this->start_time <= now()
            && $this->end_time > now();
    }

    public function minNextBid()
    {
        return ($this->current_bid ?? $this->starting_bid) + $this->bid_increment;
    }
}
