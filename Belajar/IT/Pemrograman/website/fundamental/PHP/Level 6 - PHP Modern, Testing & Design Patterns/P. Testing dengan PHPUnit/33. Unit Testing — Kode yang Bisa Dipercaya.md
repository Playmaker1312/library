# Unit Testing — Kode yang Bisa Dipercaya

## Penjelasan

**Unit testing** adalah praktik menulis kode yang menguji kode lain — memastikan setiap unit (biasanya method atau fungsi) bekerja dengan benar dalam isolasi. Framework testing paling populer di ekosistem PHP adalah **PHPUnit** — ditulis oleh Sebastian Bergmann dan menjadi standar de facto.

Prinsip dasar unit testing:

1. **Setiap test menguji satu hal** — satu method, satu skenario, satu assertion
2. **Test harus independen** — tidak bergantung pada urutan eksekusi atau state global
3. **Test harus repeatable** — dijalankan kapan pun hasilnya sama
4. **Test harus cepat** — unit test berjalan dalam milidetik, bukan detik

PHPUnit mengikuti pola **xUnit** — class test me-extend `PHPUnit\Framework\TestCase`, setiap method public dengan prefix `test` atau annotation `#[Test]` (PHP 8+) adalah test case. Assertion seperti `assertEquals`, `assertTrue`, `assertNull` membandingkan hasil aktual dengan expected.

Struktur test yang baik mengikuti pola **AAA** (Arrange-Act-Assert):

```php
public function testHitungDendaMengembalikanAngka(): void {
    // Arrange — siapkan object dan data
    $kalkulator = new DendaCalculator();

    // Act — panggil method yang diuji
    $hasil = $kalkulator->hitung(5);

    // Assert — verifikasi hasil
    $this->assertEquals(10000, $hasil);
}
```

## Fungsi

Unit testing memberikan **kepercayaan** bahwa kode berjalan sesuai harapan. Saat Anda refactor atau menambah fitur, test akan langsung mendeteksi jika ada yang rusak (regression). Test juga berfungsi sebagai **dokumentasi yang hidup** — test menunjukkan bagaimana kode seharusnya digunakan dan apa output yang diharapkan. Tanpa testing, satu perubahan kecil bisa merusak fitur lain tanpa disadari sampai aplikasi di-production.

```php
<?php
// =====================
// File: src/DendaCalculator.php

declare(strict_types=1);

namespace Perpustakaan;

class DendaCalculator {
    private array $tarif = [
        0  => 0,
        3  => 1000,
        7  => 2000,
        14 => 5000,
    ];

    public function hitung(int $hariTelat): int {
        if ($hariTelat < 0) {
            throw new \InvalidArgumentException('Hari telat tidak boleh negatif');
        }

        foreach ($this->tarif as $batas => $denda) {
            if ($hariTelat <= $batas) {
                return $hariTelat * $denda;
            }
        }

        return $hariTelat * 10000;
    }

    public function formatRupiah(int $jumlah): string {
        return 'Rp ' . number_format($jumlah, 0, ',', '.');
    }
}

// =====================
// File: src/BukuRepository.php
// =====================

declare(strict_types=1);

namespace Perpustakaan;

interface BukuRepositoryInterface {
    public function cariById(int $id): ?array;
    public function simpan(array $data): int;
}

class BukuRepository implements BukuRepositoryInterface {
    private \PDO $pdo;

    public function __construct(\PDO $pdo) {
        $this->pdo = $pdo;
    }

    public function cariById(int $id): ?array {
        $stmt = $this->pdo->prepare("SELECT * FROM buku WHERE id = ?");
        $stmt->execute([$id]);
        $result = $stmt->fetch(\PDO::FETCH_ASSOC);
        return $result ?: null;
    }

    public function simpan(array $data): int {
        $stmt = $this->pdo->prepare(
            "INSERT INTO buku (judul, pengarang, tahun) VALUES (?, ?, ?)"
        );
        $stmt->execute([$data['judul'], $data['pengarang'], $data['tahun']]);
        return (int) $this->pdo->lastInsertId();
    }
}

declare(strict_types=1);

namespace Perpustakaan\Test;

use Perpustakaan\DendaCalculator;
use PHPUnit\Framework\TestCase;

class DendaCalculatorTest extends TestCase {
    private DendaCalculator $calculator;

    // setUp dijalankan SEBELUM setiap test method
    protected function setUp(): void {
        $this->calculator = new DendaCalculator();
    }

    // PHP 8 attribute — alternatif dari prefix "test"
    #[Test]
    public function hitungDendaTepatWaktu(): void {
        $this->assertEquals(0, $this->calculator->hitung(0));
    }

    #[Test]
    public function hitungDendaTelatRingan(): void {
        $this->assertEquals(2000, $this->calculator->hitung(2));  // 2 x 1000
    }

    #[Test]
    public function hitungDendaTelatSedang(): void {
        $this->assertEquals(10000, $this->calculator->hitung(5)); // 5 x 2000
    }

    #[Test]
    public function hitungDendaTelatBerat(): void {
        $this->assertEquals(70000, $this->calculator->hitung(14)); // 14 x 5000
    }

    #[Test]
    public function hitungDendaTelatEkstrem(): void {
        $this->assertEquals(150000, $this->calculator->hitung(15)); // 15 x 10000
    }

    #[Test]
    public function hitungDendaNegatifMelemparException(): void {
        $this->expectException(\InvalidArgumentException::class);
        $this->expectExceptionMessage('Hari telat tidak boleh negatif');
        $this->calculator->hitung(-1);
    }

    #[Test]
    public function formatRupiahBenar(): void {
        $this->assertEquals('Rp 10.000', $this->calculator->formatRupiah(10000));
        $this->assertEquals('Rp 1.000.000', $this->calculator->formatRupiah(1000000));
        $this->assertEquals('Rp 0', $this->calculator->formatRupiah(0));
    }

    // DATA PROVIDER — satu test method bisa dijalankan dengan banyak data
    #[Test]
    #[DataProvider('dataDenda')]
    public function hitungDendaDenganDataProvider(int $hariTelat, int $expected): void {
        $this->assertEquals($expected, $this->calculator->hitung($hariTelat));
    }

    // DataProvider bisa berupa method static
    public static function dataDenda(): array {
        return [
            'tepat waktu'         => [0, 0],
            'telat 1 hari'        => [1, 1000],
            'telat 3 hari'        => [3, 3000],
            'telat 4 hari'        => [4, 8000],
            'telat 7 hari'        => [7, 14000],
            'telat 8 hari'        => [8, 40000],
            'telat 14 hari'       => [14, 70000],
            'telat 15 hari'       => [15, 150000],
            'telat 30 hari'       => [30, 300000],
        ];
    }
}

declare(strict_types=1);

namespace Perpustakaan\Test;

use Perpustakaan\BukuRepository;
use PHPUnit\Framework\TestCase;

class BukuRepositoryTest extends TestCase {
    private BukuRepository $repo;
    private \PDO $pdo;

    protected function setUp(): void {
        // Gunakan SQLite in-memory agar test cepat dan terisolasi
        $this->pdo = new \PDO('sqlite::memory:');
        $this->pdo->setAttribute(\PDO::ATTR_ERRMODE, \PDO::ERRMODE_EXCEPTION);
        $this->pdo->exec("CREATE TABLE buku (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            judul TEXT NOT NULL,
            pengarang TEXT NOT NULL,
            tahun INTEGER
        )");

        $this->repo = new BukuRepository($this->pdo);
    }

    #[Test]
    public function simpanDanCariById(): void {
        $id = $this->repo->simpan([
            'judul'     => 'Belajar PHP Unit',
            'pengarang' => 'Andi',
            'tahun'     => 2024,
        ]);

        $this->assertIsInt($id);
        $this->assertGreaterThan(0, $id);

        $buku = $this->repo->cariById($id);
        $this->assertNotNull($buku);
        $this->assertEquals('Belajar PHP Unit', $buku['judul']);
        $this->assertEquals('Andi', $buku['pengarang']);
        $this->assertEquals(2024, $buku['tahun']);
    }

    #[Test]
    public function cariByIdMengembalikanNullJikaTidakDitemukan(): void {
        $this->assertNull($this->repo->cariById(999));
    }
}
```

## Analogi

Unit testing adalah **sistem quality control perpustakaan** sebelum buku dipajang ke rak.

- **Test case** adalah daftar periksa (checklist) kualitas: "Apakah sampul buku tidak sobek?", "Apakah nomor ISBN tercetak?", "Apakah barcode bisa discan?" — setiap item adalah satu test.
- **Assertion** adalah tindakan verifikasi: petugas memegang buku, melihat sampulnya, dan mencentang "sudah diperiksa". `assertEquals(200, $response->getStatusCode())` sama dengan "pastikan barcode menghasilkan angka 200".
- **SetUp** adalah persiapan sebelum pemeriksaan: petugas menyiapkan meja, alat scan, dan form checklist — sama setiap hari, tidak peduli buku apa yang akan diperiksa.
- **DataProvider** adalah daftar buku sampel dengan berbagai kondisi: buku baru (0 hari), buku telat 3 hari, buku telat 30 hari — satu prosedur pengecekan yang sama dijalankan dengan data berbeda-beda.
- **Mock object** adalah **boneka petugas perpustakaan** — berpura-pura menjadi petugas sungguhan, tapi tidak benar-benar memproses peminjaman di sistem. Anda cukup bilang "jika ada yang tanya buku ini, jawab saja tersedia". Testing jadi cepat karena tidak perlu koneksi database sungguhan.
- **Code coverage** adalah peta perpustakaan yang menunjukkan rak mana yang sudah diperiksa (line hijau) dan mana yang belum (line merah) — petugas tahu bagian mana yang masih perlu dicek.

## Dipakai untuk Apa

- **Memvalidasi logika bisnis** — perhitungan denda, status peminjaman, validasi anggota
- **Regression testing** — memastikan refactor atau penambahan fitur tidak merusak fungsi lama
- **Test-driven Development (TDD)** — menulis test dulu, baru implementasi kode
- **CI/CD pipeline** — test dijalankan otomatis setiap kali push ke repository
- **Dokumentasi** — test menunjukkan kontrak API dan expected behavior
- **Mocking dependency eksternal** — database, API pihak ketiga, file system — semuanya bisa di-mock agar test cepat dan terisolasi

## Kesalahan Umum yang Sering Terjadi

- **Test tidak independen**: Test A mengubah state yang mempengaruhi Test B — gunakan `setUp()` untuk reset state setiap kali.
- **Test terlalu umum**: `assertTrue($hasil)` — gunakan assertion yang spesifik (`assertEquals`, `assertContains`, `assertInstanceOf`).
- **Test menyentuh database sungguhan**: Test jadi lambat dan tidak repeatable — gunakan SQLite in-memory atau mock.
- **Lupa assert**: Test tanpa assertion (hanya memanggil method tanpa assert) akan selalu PASS — gunakan `$this->expectException()` untuk exception.
- **Testing kode yang tidak penting**: Getters, setters, atau method trivial — fokus test pada logika bisnis yang kompleks.
- **Data provider mengembalikan key numerik**: Gunakan key string deskriptif — saat test fail, Anda langsung tahu dataset mana yang gagal.
- **Test mencampur unit test dan integration test**: Unit test harus cepat (<100ms) — test yang perlu database, API, atau file system sebaiknya dipisah sebagai integration test.

## Hubungan dengan Materi Sebelumnya

Materi 15-17 (Fungsi) adalah unit terkecil yang bisa di-test — fungsi murni (pure function) paling mudah di-test karena tidak punya side effect. Materi 23-25 (OOP) membuat testing lebih terstruktur — class dengan dependency injection (materi 34) sangat mudah di-mock. Materi 26 (PDO) menunjukkan koneksi database — di unit test, database di-mock atau diganti SQLite in-memory. Materi 30 (Composer) diperlukan untuk menginstall PHPUnit. Materi 32 (PHP 8 Features) digunakan di test dengan attribute `#[Test]` dan `#[DataProvider]`. Setelah materi ini, kamu siap untuk Design Patterns (materi 34) yang sering ditest dengan PHPUnit.

## Soal Latihan & Jawaban

### Soal

Buat unit test untuk class `PeminjamanService` di bawah ini. Fokus pada:

1. Test untuk method `pinjamBuku()` — berhasil, buku tidak tersedia, anggota melebihi limit (maks 3 buku)
2. Test untuk method `kembalikanBuku()` — berhasil, buku sudah dikembalikan sebelumnya (exception)
3. Gunakan `setUp()` untuk inisialisasi, dan `createMock()` untuk mock dependency `BukuRepositoryInterface` dan `AnggotaRepositoryInterface`
4. Gunakan data provider untuk test kasus limit peminjaman
5. Jangan test database sungguhan — semua mock

```php
<?php
declare(strict_types=1);

namespace Perpustakaan;

class PeminjamanService {
    public function __construct(
        private BukuRepositoryInterface $bukuRepo,
        private AnggotaRepositoryInterface $anggotaRepo,
    ) {}

    public function pinjamBuku(int $anggotaId, int $bukuId): array {
        $anggota = $this->anggotaRepo->cariById($anggotaId);
        if (!$anggota) {
            throw new \InvalidArgumentException('Anggota tidak ditemukan');
        }

        $buku = $this->bukuRepo->cariById($bukuId);
        if (!$buku) {
            throw new \InvalidArgumentException('Buku tidak ditemukan');
        }

        if ($buku['status'] !== 'tersedia') {
            throw new \RuntimeException('Buku tidak tersedia');
        }

        $bukuDipinjam = $this->bukuRepo->countPinjamanAktif($anggotaId);
        if ($bukuDipinjam >= 3) {
            throw new \RuntimeException('Limit peminjaman tercapai (maks 3)');
        }

        $this->bukuRepo->updateStatus($bukuId, 'dipinjam');
        return ['message' => 'Buku berhasil dipinjam', 'buku_id' => $bukuId];
    }

    public function kembalikanBuku(int $bukuId): array {
        $buku = $this->bukuRepo->cariById($bukuId);
        if (!$buku) {
            throw new \InvalidArgumentException('Buku tidak ditemukan');
        }

        if ($buku['status'] === 'tersedia') {
            throw new \RuntimeException('Buku sudah tersedia, tidak ada peminjaman aktif');
        }

        $this->bukuRepo->updateStatus($bukuId, 'tersedia');
        return ['message' => 'Buku berhasil dikembalikan'];
    }
}

interface AnggotaRepositoryInterface {
    public function cariById(int $id): ?array;
}
```

### Jawaban

```php
<?php
declare(strict_types=1);

namespace Perpustakaan\Test;

use Perpustakaan\AnggotaRepositoryInterface;
use Perpustakaan\BukuRepositoryInterface;
use Perpustakaan\PeminjamanService;
use PHPUnit\Framework\Attributes\DataProvider;
use PHPUnit\Framework\Attributes\Test;
use PHPUnit\Framework\TestCase;

class PeminjamanServiceTest extends TestCase {
    private PeminjamanService $service;
    private BukuRepositoryInterface & \PHPUnit\Framework\MockObject\MockObject $bukuRepo;
    private AnggotaRepositoryInterface & \PHPUnit\Framework\MockObject\MockObject $anggotaRepo;

    protected function setUp(): void {
        $this->bukuRepo = $this->createMock(BukuRepositoryInterface::class);
        $this->anggotaRepo = $this->createMock(AnggotaRepositoryInterface::class);
        $this->service = new PeminjamanService($this->bukuRepo, $this->anggotaRepo);
    }

    #[Test]
    public function pinjamBukuBerhasil(): void {
        $this->anggotaRepo->method('cariById')->with(1)->willReturn(['id' => 1, 'nama' => 'Andi']);
        $this->bukuRepo->method('cariById')->with(1)->willReturn([
            'id' => 1, 'judul' => 'PHP Dasar', 'status' => 'tersedia',
        ]);
        $this->bukuRepo->method('countPinjamanAktif')->with(1)->willReturn(0);
        $this->bukuRepo->expects($this->once())->method('updateStatus')->with(1, 'dipinjam');

        $result = $this->service->pinjamBuku(1, 1);
        $this->assertEquals('Buku berhasil dipinjam', $result['message']);
        $this->assertEquals(1, $result['buku_id']);
    }

    #[Test]
    public function pinjamBukuAnggotaTidakDitemukan(): void {
        $this->anggotaRepo->method('cariById')->with(999)->willReturn(null);

        $this->expectException(\InvalidArgumentException::class);
        $this->expectExceptionMessage('Anggota tidak ditemukan');
        $this->service->pinjamBuku(999, 1);
    }

    #[Test]
    public function pinjamBukuBukuTidakDitemukan(): void {
        $this->anggotaRepo->method('cariById')->with(1)->willReturn(['id' => 1, 'nama' => 'Andi']);
        $this->bukuRepo->method('cariById')->with(999)->willReturn(null);

        $this->expectException(\InvalidArgumentException::class);
        $this->expectExceptionMessage('Buku tidak ditemukan');
        $this->service->pinjamBuku(1, 999);
    }

    #[Test]
    public function pinjamBukuBukuTidakTersedia(): void {
        $this->anggotaRepo->method('cariById')->with(1)->willReturn(['id' => 1, 'nama' => 'Andi']);
        $this->bukuRepo->method('cariById')->with(1)->willReturn([
            'id' => 1, 'judul' => 'PHP Dasar', 'status' => 'dipinjam',
        ]);

        $this->expectException(\RuntimeException::class);
        $this->expectExceptionMessage('Buku tidak tersedia');
        $this->service->pinjamBuku(1, 1);
    }

    #[Test]
    #[DataProvider('dataLimitPeminjaman')]
    public function pinjamBukuMelebihiLimit(int $jumlahDipinjam): void {
        $this->anggotaRepo->method('cariById')->with(1)->willReturn(['id' => 1, 'nama' => 'Andi']);
        $this->bukuRepo->method('cariById')->with(1)->willReturn([
            'id' => 1, 'judul' => 'PHP Dasar', 'status' => 'tersedia',
        ]);
        $this->bukuRepo->method('countPinjamanAktif')->with(1)->willReturn($jumlahDipinjam);

        $this->expectException(\RuntimeException::class);
        $this->expectExceptionMessage('Limit peminjaman tercapai (maks 3)');
        $this->service->pinjamBuku(1, 1);
    }

    public static function dataLimitPeminjaman(): array {
        return [
            'sudah 3 buku'  => [3],
            'sudah 5 buku'  => [5],
            'sudah 10 buku' => [10],
        ];
    }

    #[Test]
    public function kembalikanBukuBerhasil(): void {
        $this->bukuRepo->method('cariById')->with(1)->willReturn([
            'id' => 1, 'judul' => 'PHP Dasar', 'status' => 'dipinjam',
        ]);
        $this->bukuRepo->expects($this->once())->method('updateStatus')->with(1, 'tersedia');

        $result = $this->service->kembalikanBuku(1);
        $this->assertEquals('Buku berhasil dikembalikan', $result['message']);
    }

    #[Test]
    public function kembalikanBukuYangSudahTersedia(): void {
        $this->bukuRepo->method('cariById')->with(1)->willReturn([
            'id' => 1, 'judul' => 'PHP Dasar', 'status' => 'tersedia',
        ]);

        $this->expectException(\RuntimeException::class);
        $this->expectExceptionMessage('Buku sudah tersedia, tidak ada peminjaman aktif');
        $this->service->kembalikanBuku(1);
    }

    #[Test]
    public function kembalikanBukuTidakDitemukan(): void {
        $this->bukuRepo->method('cariById')->with(999)->willReturn(null);

        $this->expectException(\InvalidArgumentException::class);
        $this->expectExceptionMessage('Buku tidak ditemukan');
        $this->service->kembalikanBuku(999);
    }
}
```
