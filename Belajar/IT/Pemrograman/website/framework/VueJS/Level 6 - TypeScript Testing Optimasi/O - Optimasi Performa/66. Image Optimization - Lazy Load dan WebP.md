# 66. Image Optimization — Lazy Load dan WebP

## Penjelasan

Gambar adalah aset terbesar di hampir semua aplikasi web. Optimasi gambar mencakup tiga strategi utama:

1. **Lazy loading native** — atribut HTML `loading="lazy"` membuat browser hanya mengunduh gambar saat mendekati viewport.
2. **Placeholder** — gambar kecil atau skeleton yang muncul sebelum gambar asli selesai dimuat, mencegah layout shift (CLS).
3. **Format WebP** — format gambar modern dengan kompresi superior. Ukuran 25-35% lebih kecil dari JPEG/PNG dengan kualitas visual yang setara.

Browser modern mendukung WebP secara luas (>96%). Untuk browser lawas, siapkan fallback ke format asli.

## Fungsi

- **Tidak blocking render** — halaman bisa tampil tanpa menunggu semua gambar unduh
- **Bandwidth hemat** — terutama untuk pengguna mobile / data terbatas
- **CLS berkurang** — placeholder/reserved space mencegah layout tiba-tiba berubah
- **First Contentful Paint lebih cepat** — gambar di atas fold dimuat prioritas (eager), sisanya lazy
- **Pengalaman pengguna lebih baik** — konten utama muncul cepat, gambar mengisi secara bertahap

## Cara Implementasi

### 1. Native Lazy Loading + Decoding Async

```vue
<!-- BookCard.vue -->
<script setup lang="ts">
import { ref } from 'vue'

interface Props {
  book: {
    id: number
    title: string
    coverImage: string
    coverWebp?: string
  }
}

const props = defineProps<Props>()
const isLoaded = ref(false)
const hasError = ref(false)

function onImageLoad() {
  isLoaded.value = true
}

function onImageError() {
  hasError.value = true
  isLoaded.value = true
}
</script>

<template>
  <div class="book-card">
    <!-- Placeholder — muncul sebelum gambar asli -->
    <div
      v-if="!isLoaded"
      class="book-card__placeholder"
      :style="{ backgroundColor: '#e2e8f0' }"
    >
      <span class="book-card__skeleton-icon">📚</span>
    </div>

    <!-- Gambar dengan native lazy loading & WebP -->
    <picture>
      <source
        :srcset="book.coverWebp"
        type="image/webp"
      >
      <img
        :src="book.coverImage"
        :alt="book.title"
        class="book-card__image"
        :class="{ 'book-card__image--loaded': isLoaded }"
        loading="lazy"
        decoding="async"
        width="200"
        height="300"
        @load="onImageLoad"
        @error="onImageError"
      >
    </picture>
  </div>
</template>

<style scoped>
.book-card {
  position: relative;
  width: 200px;
  height: 300px;
}

.book-card__placeholder {
  position: absolute;
  inset: 0;
  display: flex;
  align-items: center;
  justify-content: center;
  border-radius: 8px;
  animation: pulse 1.5s infinite;
}

.book-card__image {
  width: 100%;
  height: 100%;
  object-fit: cover;
  border-radius: 8px;
  opacity: 0;
  transition: opacity 0.3s ease;
}

.book-card__image--loaded {
  opacity: 1;
}

@keyframes pulse {
  0%, 100% { opacity: 0.6; }
  50% { opacity: 1; }
}
</style>
```

### 2. Atur Prioritas: Eager untuk Above-the-Fold, Lazy untuk Sisanya

```vue
<!-- BookList.vue -->
<script setup lang="ts">
import BookCard from './BookCard.vue'

interface Book {
  id: number
  title: string
  coverImage: string
  coverWebp?: string
}

defineProps<{
  books: Book[]
}>()
</script>

<template>
  <div class="book-list">
    <!-- 3 buku pertama: eager (above the fold) -->
    <BookCard
      v-for="(book, index) in books"
      :key="book.id"
      :book="book"
      :loading="index < 3 ? 'eager' : 'lazy'"
    />
  </div>
</template>
```

Di komponen `BookCard`, terima prop `loading`:

```vue
<script setup lang="ts">
defineProps<{
  book: Book
  loading?: 'lazy' | 'eager'
}>()
</script>

<template>
  <img
    :src="book.coverImage"
    :loading="loading ?? 'lazy'"
    :decoding="loading === 'eager' ? 'sync' : 'async'"
  >
</template>
```

### 3. Intersection Observer untuk Custom Lazy Load (Fallback)

Jika butuh kontrol lebih (animasi masuk, preload, dll):

```typescript
// composables/useLazyImage.ts
import { ref, onMounted, onUnmounted } from 'vue'

export function useLazyImage(options?: IntersectionObserverInit) {
  const imageRef = ref<HTMLImageElement | null>(null)
  const isVisible = ref(false)
  const isLoaded = ref(false)

  let observer: IntersectionObserver | null = null

  onMounted(() => {
    if (!imageRef.value) return

    observer = new IntersectionObserver(([entry]) => {
      if (entry.isIntersecting) {
        isVisible.value = true
        observer?.unobserve(entry.target)
      }
    }, {
      rootMargin: '200px', // mulai unduh 200px sebelum terlihat
      ...options
    })

    observer.observe(imageRef.value)
  })

  onUnmounted(() => {
    observer?.disconnect()
  })

  return { imageRef, isVisible, isLoaded }
}
```

### 5. Komponen Gambar Umum dengan Semua Optimasi

```vue
<!-- components/common/OptimizedImage.vue -->
<script setup lang="ts">
import { ref, computed } from 'vue'

const props = defineProps<{
  src: string
  webpSrc?: string
  alt: string
  width?: number
  height?: number
  loading?: 'lazy' | 'eager'
  class?: string
}>()

const isLoaded = ref(false)
const hasError = ref(false)

const showPlaceholder = computed(() => !isLoaded.value && !hasError.value)
</script>

<template>
  <div
    class="optimized-image"
    :style="{
      width: width ? `${width}px` : '100%',
      height: height ? `${height}px` : 'auto'
    }"
  >
    <div v-if="showPlaceholder" class="optimized-image__skeleton" />

    <picture v-if="!hasError">
      <source v-if="webpSrc" :srcset="webpSrc" type="image/webp">
      <img
        :src="src"
        :alt="alt"
        :width="width"
        :height="height"
        :loading="loading ?? 'lazy'"
        :decoding="loading === 'eager' ? 'sync' : 'async'"
        class="optimized-image__img"
        :class="{ 'optimized-image__img--loaded': isLoaded }"
        @load="isLoaded = true"
        @error="hasError = true"
      >
    </picture>

    <div v-else class="optimized-image__fallback">
      <span>{{ alt.charAt(0).toUpperCase() }}</span>
    </div>
  </div>
</template>
```

## Analogi — Toko Buku

Dulu toko memajang **SEMUA gambar buku** di etalase sekaligus saat pintu dibuka — pengunjung harus menunggu SEMUA gambar selesai dicetak dan ditempel sebelum bisa lihat apa-apa (blocking render).

Sekarang toko menerapkan tiga strategi:

1. **Gambar dimuat SAAT pengunjung mendekati rak** (`loading="lazy"`). Pengunjung yang berdiri di rak "Fiksi" tidak perlu menunggu gambar rak "Sains" selesai dimuat.

2. **Saat gambar dimuat, pertama muncul siluet / bayangan buku** (placeholder), lalu gambar asli muncul (image loaded). Seperti etalase yang ditaruh kertas sampul dulu — pengunjung tahu itu buku meski belum jelas judulnya. Layout tidak berubah saat gambar asli datang.

3. **Semua foto buku dicetak ulang dengan kertas lebih efisien** (WebP) — kualitas visual sama, ukuran 30% lebih kecil. Seperti mengganti kertas foto glossy 200 gr dengan kertas premium 120 gr yang sama tajamnya tapi lebih ringan.

Hasilnya: toko terasa lebih cepat dibuka, pengunjung bisa langsung lihat etalase tanpa menunggu, dan toko menghemat biaya cetak (bandwidth).

## Dipakai untuk Apa

- **Gambar sampul buku** (BookStore) — thumbnail dalam grid
- **Galeri produk** (e-commerce) — banyak gambar per halaman
- **Avatar pengguna** — daftar ulasan dengan foto profil
- **Hero image / banner** — atur `loading="eager"` untuk gambar hero
- **Ilustrasi artikel** — blog atau konten dinamis
- **Foto upload pengguna** — konversi ke WebP di backend

## Kesalahan Umum

| Kesalahan | Dampak | Solusi |
|-----------|--------|--------|
| `loading="lazy"` pada gambar **above-the-fold** | Gambar hero muncul lambat — LCP memburuk | `loading="eager"` untuk 2-3 gambar pertama |
| Tidak memberikan `width` + `height` | Layout shift (CLS) saat gambar dimuat | Selalu set dimensi eksplisit atau reserved space CSS |
| Tidak provide fallback WebP | Browser lama (Safari <14) tidak menampilkan gambar | Gunakan tag `<picture>` dengan `<source webp>` + `<img fallback>` |
| Placeholder terlalu kompleks | Bundle size bertambah, animasi berat | Cukup skeleton sederhana dengan CSS |
| Lupa `decoding="async"` | Decoding gambar blocking main thread | Tambahkan `decoding="async"` |
| Semua gambar pakai lazy | Gambar penting (hero, logo) tertunda | Atur prioritas: eager untuk penting, lazy untuk sisanya |
| Tidak handle error image | Broken image icon jelek | Sediakan fallback (inisial, ikon, atau placeholder) |

## Hubungan dengan Materi Sebelumnya

**Level 1 (Dasar Vue & TypeScript):** Dulu kita pakai `<img :src="book.image">` langsung — tanpa optimasi apapun. Cukup untuk 5 buku, tidak untuk 500 buku.

**Level 2 (State Management & Composables):** Composable `useLazyImage` (Intersection Observer) memisahkan logika lazy loading dari UI — reusable di semua komponen.

**Level 3 (Vue Router):** Setiap halaman di-lazy load (poin 63). Tapi gambar di dalam halaman tetap besar — image optimization melengkapi lazy loading route.

**Level 4 (Testing):** Test komponen `OptimizedImage` untuk memastikan loading state, error state, dan fallback WebP bekerja. Gunakan Vitest + jsdom.

**Level 5 (Backend Integration):** Backend bisa menyediakan dua URL gambar: `.jpg` dan `.webp`. API response menyertakan `coverWebp` untuk gambar yang sudah dikonversi.

**Level 6 (Sekarang):** Setelah bundle size (poin 63), runtime (poin 64), dan analysis (poin 65), kita optimasi aset terbesar: gambar. Keempat poin ini adalah satu paket optimasi performa lengkap:

```
Poin 63 → Bundle lebih kecil (lazy loading)
Poin 64 → Render lebih efisien (v-memo, shallowRef)
Poin 65 → Analisis & perbaikan (bundle analysis, tree shaking)
Poin 66 → Aset lebih ringan (lazy image, WebP)
                              = Aplikasi super cepat
```

## Soal Latihan

### Soal 1

Di aplikasi BookStore, ada halaman HomePage dengan:

1. **Hero banner** (gambar besar 1920x600) — di atas fold
2. **Grid 50 buku** (masing-masing 200x300) — thumbnail sampul
3. **Logo toko** (200x50) — di navbar

Saat ini semua gambar menggunakan `<img :src="..." loading="lazy">`. Hasil Lighthouse menunjukkan LCP (Largest Contentful Paint) sangat buruk. Apa penyebabnya? Perbaiki strategi loading untuk ketiga jenis gambar.

<details>
<summary>Jawaban</summary>

**Penyebab:** Hero banner menggunakan `loading="lazy"` padahal ia adalah gambar **above-the-fold** (terlihat pertama kali saat halaman dimuat). Browser menganggapnya tidak prioritas karena atribut lazy — menunda unduhan hingga hero banner "mendekati viewport" (padahal sudah di viewport sejak awal).

**Perbaikan:**

```vue
<!-- 1. Hero Banner — EAGER (above the fold, prioritas tinggi) -->
<img
  :src="heroImage"
  alt="Promo Buku Terbaru"
  loading="eager"
  decoding="sync"
  width="1920"
  height="600"
  fetchpriority="high"
>

<!-- 2. Thumbnail Buku — LAZY (di bawah fold atau grid panjang) -->
<BookCard
  v-for="(book, index) in books"
  :key="book.id"
  :book="book"
  :loading="index < 6 ? 'eager' : 'lazy'"
/>

<!-- 3. Logo Navbar — EAGER (selalu terlihat) -->
<img
  :src="logoUrl"
  alt="BookStore"
  loading="eager"
  decoding="sync"
  width="200"
  height="50"
>
```

**Aturan praktis:**
- 1-3 gambar pertama (hero, logo, featured) → `eager`
- Sisanya (thumbnail, gambar artikel, galeri) → `lazy`
- Gunakan `fetchpriority="high"` untuk gambar paling penting

### Soal 2

Buat komponen `OptimizedImage` yang menerima props berikut:

- `src: string` — URL gambar utama (JPEG/PNG)
- `webpSrc?: string` — URL versi WebP (opsional)
- `alt: string` — teks alternatif
- `loading?: 'lazy' | 'eager'` — loading strategy (default lazy)

Komponen harus:

1. Menampilkan placeholder skeleton saat gambar belum dimuat
2. Menggunakan `<picture>` untuk WebP fallback
3. Memiliki transisi fade-in saat gambar selesai dimuat
4. Menampilkan inisial teks sebagai fallback jika gambar gagal dimuat
5. Menggunakan `decoding="async"` kecuali eager

<details>
<summary>Jawaban</summary>

```vue
<!-- components/common/OptimizedImage.vue -->
<script setup lang="ts">
import { ref, computed } from 'vue'

const props = withDefaults(defineProps<{
  src: string
  webpSrc?: string
  alt: string
  loading?: 'lazy' | 'eager'
  width?: number
  height?: number
}>(), {
  loading: 'lazy'
})

const isLoaded = ref(false)
const hasError = ref(false)

const showPlaceholder = computed(() => !isLoaded.value)
const decodingAttr = computed(() => props.loading === 'eager' ? 'sync' : 'async')
const fallbackInitial = computed(() => props.alt.charAt(0).toUpperCase())

function onLoad() {
  isLoaded.value = true
}

function onError() {
  hasError.value = true
  isLoaded.value = true
}
</script>

<template>
  <div
    class="optimized-image"
    :style="{
      width: width ? `${width}px` : '100%',
      height: height ? `${height}px` : 'auto',
      aspectRatio: width && height ? `${width} / ${height}` : undefined
    }"
  >
    <div v-if="showPlaceholder && !hasError" class="optimized-image__skeleton" />

    <picture v-if="!hasError">
      <source v-if="webpSrc" :srcset="webpSrc" type="image/webp">
      <img
        :src="src"
        :alt="alt"
        :width="width"
        :height="height"
        :loading="loading"
        :decoding="decodingAttr"
        class="optimized-image__img"
        :class="{ 'optimized-image__img--loaded': isLoaded }"
        @load="onLoad"
        @error="onError"
      >
    </picture>

    <div v-else class="optimized-image__fallback">
      <span>{{ fallbackInitial }}</span>
    </div>
  </div>
</template>

<style scoped>
.optimized-image {
  position: relative;
  overflow: hidden;
  border-radius: 8px;
  background-color: #e2e8f0;
}

.optimized-image__skeleton {
  position: absolute;
  inset: 0;
  animation: shimmer 1.5s infinite;
  background: linear-gradient(
    90deg,
    #e2e8f0 25%,
    #f1f5f9 50%,
    #e2e8f0 75%
  );
  background-size: 200% 100%;
}

.optimized-image__img {
  width: 100%;
  height: 100%;
  object-fit: cover;
  opacity: 0;
  transition: opacity 0.3s ease;
}

.optimized-image__img--loaded {
  opacity: 1;
}

  font-weight: 700;
}

@keyframes shimmer {
  0% { background-position: -200% 0; }

}
</style>
```
