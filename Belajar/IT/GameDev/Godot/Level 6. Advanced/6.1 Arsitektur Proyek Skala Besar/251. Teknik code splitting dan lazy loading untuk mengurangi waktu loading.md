# 251. Teknik code splitting dan lazy loading untuk mengurangi waktu loading

## Definisi Singkat
Code splitting adalah teknik membagi kode menjadi chunk-chunk kecil yang dapat dimuat secara terpisah, sedangkan lazy loading adalah strategi memuat resource hanya saat dibutuhkan. Dalam Godot, ini diimplementasikan menggunakan `ResourceLoader.load_threaded_request()` dan `load()` dinamis untuk mengurangi waktu loading awal game dan mengoptimalkan penggunaan memori.

## Analogi
Bayangkan kamu pindah ke rumah baru. Daripada membawa semua kotak sekaligus (eager loading), kamu hanya membawa kotak yang diperlukan hari ini — perlengkapan tidur dan perlengkapan mandi. Kotak furniture, buku, dan dekorasi lainnya baru diambil saat dibutuhkan (lazy loading). Ini membuat proses pindahan lebih cepat dan tidak melelahkan.

## Struktur Node (Scene Tree)
```
SceneManager (Node) — Mengelola loading dan transisi scene
├── ChunkLoader (Node) — Memuat scene dalam chunk
├── ResourceCache (Node) — Cache resource yang sudah dimuat
├── LoadingScreen (CanvasLayer) — UI loading screen
│   ├── ProgressBar
│   └── StatusLabel
└── PreloadedChunks (Node) — Container untuk chunk yang sudah di-preload
```

## Kode GDScript (Godot 4)

```gdscript
# lazy_loader.gd — Central lazy loading manager
class_name LazyLoader
extends Node

signal load_started(resource_path: String)
signal load_completed(resource_path: String)
signal load_failed(resource_path: String, error: String)

var loading_queue: Array[String] = []
var loaded_resources: Dictionary = {}
var active_loads: Dictionary = {}
var max_concurrent_loads: int = 3

func request_load(resource_path: String, callback: Callable = Callable()) -> void:
    if loaded_resources.has(resource_path):
        if callback.is_valid():
            callback.call(loaded_resources[resource_path])
        return

    if resource_path in loading_queue:
        return

    loading_queue.append(resource_path)

    if callback.is_valid():
        if not active_loads.has(resource_path):
            active_loads[resource_path] = []
        active_loads[resource_path].append(callback)

    _process_queue()

func _process_queue() -> void:
    while loading_queue.size() > 0 and _get_active_load_count() < max_concurrent_loads:
        var path = loading_queue.pop_front()
        _start_load(path)

func _get_active_load_count() -> int:
    var count = 0
    for path in active_loads:
        if active_loads[path].size() > 0:
            count += 1
    return count

func _start_load(resource_path: String) -> void:
    ResourceLoader.load_threaded_request(resource_path)
    load_started.emit(resource_path)

func _process(_delta: float) -> void:
    var completed_loads = []

    for path in active_loads.keys():
        var status = ResourceLoader.load_threaded_get_status(path)

        match status:
            ResourceLoader.THREAD_LOAD_LOADED:
                var resource = ResourceLoader.load_threaded_get(path)
                _on_load_complete(path, resource)
                completed_loads.append(path)
                load_completed.emit(path)
            ResourceLoader.THREAD_LOAD_FAILED:
                _on_load_failed(path, "Load failed")
                completed_loads.append(path)
                load_failed.emit(path, "Load failed")

    for path in completed_loads:
        active_loads.erase(path)

func _on_load_complete(path: String, resource: Resource) -> void:
    loaded_resources[path] = resource

    if active_loads.has(path):
        for callback in active_loads[path]:
            if callback.is_valid():
                callback.call(resource)

func _on_load_failed(path: String, error: String) -> void:
    push_warning("Failed to load: " + path + " - " + error)

func get_loaded_resource(path: String) -> Variant:
    return loaded_resources.get(path)

func unload_resource(path: String) -> void:
    if loaded_resources.has(path):
        loaded_resources.erase(path)
        ResourceLoader.get_cached_ref(path).unreference()
```

```gdscript
# scene_manager.gd — Manager untuk loading scene dengan lazy loading
class_name SceneManager
extends Node

var current_scene: Node
var lazy_loader: LazyLoader
var loading_screen: CanvasLayer

func _ready() -> void:
    lazy_loader = LazyLoader.new()
    add_child(lazy_loader)

    loading_screen = _create_loading_screen()
    add_child(loading_screen)
    loading_screen.visible = false

func load_scene(scene_path: String, show_loading: bool = true) -> void:
    if show_loading:
        loading_screen.visible = true

    var packed_scene = load(scene_path)
    if packed_scene:
        _transition_to_scene(packed_scene)
    else:
        lazy_loader.request_load(scene_path, _on_scene_loaded)

func _on_scene_loaded(resource: Resource) -> void:
    if resource is PackedScene:
        _transition_to_scene(resource)
    loading_screen.visible = false

func _transition_to_scene(packed_scene: PackedScene) -> void:
    if current_scene:
        current_scene.queue_free()

    var new_scene = packed_scene.instantiate()
    add_child(new_scene)
    current_scene = new_scene

func load_scene_async(scene_path: String) -> void:
    loading_screen.visible = true

    var thread = Thread.new()
    thread.start(_load_scene_thread.bind(scene_path, thread))

func _load_scene_thread(scene_path: String, thread: Thread) -> void:
    var loader = ResourceLoader.load_threaded_request(scene_path)

    while true:
        var status = ResourceLoader.load_threaded_get_status(scene_path)

        if status == ResourceLoader.THREAD_LOAD_LOADED:
            var resource = ResourceLoader.load_threaded_get(scene_path)
            call_deferred("_on_async_scene_loaded", resource)
            break
        elif status == ResourceLoader.THREAD_LOAD_FAILED:
            call_deferred("_on_async_scene_failed")
            break

        await get_tree().create_timer(0.1).timeout

func _on_async_scene_loaded(resource: Resource) -> void:
    if resource is PackedScene:
        _transition_to_scene(resource)
    loading_screen.visible = false

func _on_async_scene_failed() -> void:
    loading_screen.visible = false
    push_warning("Failed to load scene asynchronously")

func _create_loading_screen() -> CanvasLayer:
    var canvas = CanvasLayer.new()
    var bg = ColorRect.new()
    bg.color = Color(0, 0, 0, 0.8)
    bg.set_anchors_preset(Control.PRESET_FULL_RECT)
    canvas.add_child(bg)

    var progress = ProgressBar.new()
    progress.set_anchors_preset(Control.PRESET_CENTER)
    progress.size = Vector2(400, 30)
    canvas.add_child(progress)

    var label = Label.new()
    label.text = "Loading..."
    label.set_anchors_preset(Control.PRESET_CENTER)
    label.position = Vector2(0, -50)
    canvas.add_child(label)

    return canvas
```

```gdscript
# chunk_loader.gd — Loading game world dalam chunk-chunk kecil
class_name ChunkLoader
extends Node

signal chunk_loaded(chunk_id: Vector2i)
signal chunk_unloaded(chunk_id: Vector2i)

const CHUNK_SIZE: int = 64
const LOAD_RADIUS: int = 3

var loaded_chunks: Dictionary = {}
var chunk_cache: Dictionary = {}
var player_node: Node2D

func _ready() -> void:
    player_node = get_tree().get_first_node_in_group("player")

func _process(_delta: float) -> void:
    if player_node:
        var player_chunk = _get_chunk_position(player_node.global_position)
        _update_loaded_chunks(player_chunk)

func _get_chunk_position(world_pos: Vector2) -> Vector2i:
    return Vector2i(
        int(floor(world_pos.x / CHUNK_SIZE)),
        int(floor(world_pos.y / CHUNK_SIZE))
    )

func _update_loaded_chunks(center: Vector2i) -> void:
    var chunks_to_load = []

    for x in range(-LOAD_RADIUS, LOAD_RADIUS + 1):
        for y in range(-LOAD_RADIUS, LOAD_RADIUS + 1):
            var chunk_id = center + Vector2i(x, y)
            if not loaded_chunks.has(chunk_id):
                chunks_to_load.append(chunk_id)

    for chunk_id in chunks_to_load:
        _load_chunk_async(chunk_id)

    var chunks_to_unload = []
    for chunk_id in loaded_chunks:
        var distance = (chunk_id - center).length()
        if distance > LOAD_RADIUS + 1:
            chunks_to_unload.append(chunk_id)

    for chunk_id in chunks_to_unload:
        _unload_chunk(chunk_id)

func _load_chunk_async(chunk_id: Vector2i) -> void:
    var chunk_path = _get_chunk_path(chunk_id)

    if chunk_cache.has(chunk_id):
        _on_chunk_loaded(chunk_id, chunk_cache[chunk_id])
        return

    ResourceLoader.load_threaded_request(chunk_path)

    var check_interval = 0.05
    while ResourceLoader.load_threaded_get_status(chunk_path) == ResourceLoader.THREAD_LOAD_IN_PROGRESS:
        await get_tree().create_timer(check_interval).timeout

    var status = ResourceLoader.load_threaded_get_status(chunk_path)
    if status == ResourceLoader.THREAD_LOAD_LOADED:
        var resource = ResourceLoader.load_threaded_get(chunk_id)
        chunk_cache[chunk_id] = resource
        _on_chunk_loaded(chunk_id, resource)

func _on_chunk_loaded(chunk_id: Vector2i, resource: Resource) -> void:
    if resource is PackedScene:
        var chunk_node = resource.instantiate()
        chunk_node.position = Vector2(chunk_id.x * CHUNK_SIZE, chunk_id.y * CHUNK_SIZE)
        add_child(chunk_node)
        loaded_chunks[chunk_id] = chunk_node
        chunk_loaded.emit(chunk_id)

func _unload_chunk(chunk_id: Vector2i) -> void:
    if loaded_chunks.has(chunk_id):
        var chunk_node = loaded_chunks[chunk_id]
        chunk_node.queue_free()
        loaded_chunks.erase(chunk_id)
        chunk_unloaded.emit(chunk_id)

func _get_chunk_path(chunk_id: Vector2i) -> String:
    return "res://world/chunks/chunk_%d_%d.tscn" % [chunk_id.x, chunk_id.y]
```

```gdscript
# asset_preloader.gd — Preloading asset berdasarkan prioritas
class_name AssetPreloader
extends Node

var preload_queue: Array[Dictionary] = []
var current_preload: Dictionary = {}
var is_preloading: bool = false

func add_to_queue(asset_path: String, priority: int = 0, callback: Callable = Callable()) -> void:
    preload_queue.append({
        "path": asset_path,
        "priority": priority,
        "callback": callback
    })

    preload_queue.sort_custom(func(a, b): return a.priority > b.priority)

    if not is_preloading:
        _start_preloading()

func _start_preloading() -> void:
    is_preloading = true

    while preload_queue.size() > 0:
        var asset = preload_queue.pop_front()
        await _preload_asset(asset)

    is_preloading = false

func _preload_asset(asset_info: Dictionary) -> void:
    var path = asset_info.path
    ResourceLoader.load_threaded_request(path)

    while ResourceLoader.load_threaded_get_status(path) == ResourceLoader.THREAD_LOAD_IN_PROGRESS:
        await get_tree().create_timer(0.01).timeout

    var status = ResourceLoader.load_threaded_get_status(path)
    if status == ResourceLoader.THREAD_LOAD_LOADED:
        var resource = ResourceLoader.load_threaded_get(path)
        if asset_info.callback.is_valid():
            asset_info.callback.call(resource)

func preload_scene_batch(scene_paths: Array[String]) -> void:
    for i in range(scene_paths.size()):
        var priority = scene_paths.size() - i
        add_to_queue(scene_paths[i], priority)

func preload_level_assets(level_name: String) -> void:
    var level_data = _get_level_asset_list(level_name)
    for asset_path in level_data:
        add_to_queue(asset_path, 1)

func _get_level_asset_list(level_name: String) -> Array[String]:
    var asset_list: Array[String] = []

    var config_path = "res://levels/" + level_name + "/assets.cfg"
    var config = ConfigFile.new()

    if config.load(config_path) == OK:
        for key in config.get_section_keys("assets"):
            asset_list.append(config.get_value("assets", key))

    return asset_list
```

## Kasus Penggunaan Nyata
Game open-world seperti Elden Ring atau Cyberpunk 2077 memiliki dunia yang sangat besar. Jika semua asset dimuat sekaligus di awal, game akan membutuhkan waktu loading sangat lama dan memori ratusan GB. Dengan lazy loading, hanya area di sekitar pemain yang dimuat. Saat pemain bergerak ke area baru, area yang jauh di-unload dan area baru di-load. Teknik ini juga digunakan dalam game mobile seperti Genshin Impact untuk menangani keterbatasan memori perangkat mobile.
