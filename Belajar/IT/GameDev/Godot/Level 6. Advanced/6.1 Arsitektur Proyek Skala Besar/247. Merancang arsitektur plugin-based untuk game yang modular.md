# 247. Merancang arsitektur plugin-based untuk game yang modular

## Definisi Singkat
Plugin-based architecture adalah pendekatan desain di mana komponen game dikembangkan sebagai modul independen yang dapat ditambah, dihapus, atau diganti tanpa mempengaruhi inti aplikasi. Dalam Godot, ini diimplementasikan menggunakan fitur Plugin dan Custom Resources yang memungkinkan ekstensi engine tanpa modifikasi source code utama.

## Analogi
Bayangkan sebuah LEGO Technic. Setiap blok LEGO adalah modul independen dengan konektor standar. Kamu bisa mengganti roda, menambah motor, atau mengubah struktur tanpa merangkai ulang seluruh konstruksi. Plugin-based architecture bekerja persis sama — setiap fitur game adalah "blok LEGO" yang saling terhubung melalui interface yang telah didefinisikan.

## Struktur Node (Scene Tree)
```
GameRoot (Node)
├── Core (Node) — Inti game yang stabil
│   ├── PluginManager (Node) — Mengelola loading/unloading plugin
│   ├── EventBus (Node) — Pusat komunikasi
│   └── ConfigManager (Node) — Konfigurasi global
├── Plugins (Node) — Container untuk semua plugin
│   ├── CombatSystem/ — Plugin sistem pertarungan
│   │   ├── CombatPlugin.gd
│   │   └── CombatUI.tscn
│   ├── InventorySystem/ — Plugin sistem inventaris
│   │   ├── InventoryPlugin.gd
│   │   └── InventoryUI.tscn
│   └── DialogueSystem/ — Plugin sistem dialog
│       ├── DialoguePlugin.gd
│       └── DialogueUI.tscn
└── World (Node) — Dunia game
```

## Kode GDScript (Godot 4)

```gdscript
# plugin_interface.gd — Interface yang harus diimplementasikan setiap plugin
class_name PluginInterface
extends RefCounted

signal plugin_initialized
signal plugin_disabled

var plugin_name: String = ""
var plugin_version: String = "1.0.0"
var is_active: bool = false

func initialize() -> void:
    is_active = true
    plugin_initialized.emit()

func disable() -> void:
    is_active = false
    plugin_disabled.emit()

func get_plugin_name() -> String:
    return plugin_name

func get_plugin_version() -> String:
    return plugin_version

func _on_game_event(event_name: String, data: Dictionary) -> void:
    pass
```

```gdscript
# plugin_manager.gd — Central plugin management
extends Node

signal plugin_loaded(plugin_name: String)
signal plugin_unloaded(plugin_name: String)

var loaded_plugins: Dictionary = {}
var plugin_registry: Dictionary = {}

func register_plugin(plugin_id: String, plugin_path: String) -> void:
    plugin_registry[plugin_id] = plugin_path

func load_plugin(plugin_id: String) -> bool:
    if loaded_plugins.has(plugin_id):
        return false

    if not plugin_registry.has(plugin_id):
        return false

    var plugin_script = load(plugin_registry[plugin_id])
    var plugin_instance = plugin_script.new()

    if plugin_instance is PluginInterface:
        plugin_instance.initialize()
        loaded_plugins[plugin_id] = plugin_instance
        add_child(plugin_instance)
        plugin_loaded.emit(plugin_id)
        return true

    return false

func unload_plugin(plugin_id: String) -> bool:
    if not loaded_plugins.has(plugin_id):
        return false

    var plugin = loaded_plugins[plugin_id]
    plugin.disable()
    remove_child(plugin)
    plugin.queue_free()
    loaded_plugins.erase(plugin_id)
    plugin_unloaded.emit(plugin_id)
    return true

func get_plugin(plugin_id: String) -> PluginInterface:
    return loaded_plugins.get(plugin_id)

func get_all_active_plugins() -> Array:
    return loaded_plugins.keys()

func notify_all_plugins(event_name: String, data: Dictionary) -> void:
    for plugin_id in loaded_plugins:
        loaded_plugins[plugin_id]._on_game_event(event_name, data)
```

```gdscript
# combat_plugin.gd — Contoh plugin sistem pertarungan
class_name CombatPlugin
extends PluginInterface

var combat_entities: Array = []
var damage_calculator: Node

func _init() -> void:
    plugin_name = "CombatSystem"
    plugin_version = "2.1.0"

func initialize() -> void:
    super.initialize()
    damage_calculator = _create_damage_calculator()
    add_child(damage_calculator)

func register_combat_entity(entity: Node) -> void:
    combat_entities.append(entity)

func calculate_damage(attacker: Dictionary, defender: Dictionary) -> float:
    var base_damage = attacker.get("attack", 0)
    var defense = defender.get("defense", 0)
    var multiplier = attacker.get("damage_multiplier", 1.0)
    return max(base_damage * multiplier - defense, 0)

func _create_damage_calculator() -> Node:
    var calc = Node.new()
    calc.set_script(preload("res://addons/combat/damage_calculator.gd"))
    return calc

func _on_game_event(event_name: String, data: Dictionary) -> void:
    match event_name:
        "entity_registered":
            register_combat_entity(data.get("entity"))
        "combat_started":
            _start_combat_round(data)
        "combat_ended":
            _cleanup_combat()

func _start_combat_round(data: Dictionary) -> void:
    var attacker = data.get("attacker")
    var defender = data.get("defender")
    if attacker and defender:
        var damage = calculate_damage(attacker, defender)
        defender.take_damage(damage)

func _cleanup_combat() -> void:
    combat_entities.clear()
```

```gdscript
# plugin_loader.gd — Script untuk load semua plugin dari direktori
extends Node

const PLUGIN_DIR = "res://plugins/"

func _ready() -> void:
    _discover_plugins()

func _discover_plugins() -> void:
    var dir = DirAccess.open(PLUGIN_DIR)
    if dir:
        dir.list_dir_begin()
        var file_name = dir.get_next()
        while file_name != "":
            if dir.current_is_dir() and file_name != "." and file_name != "..":
                var plugin_path = PLUGIN_DIR + file_name + "/plugin.cfg"
                if FileAccess.file_exists(plugin_path):
                    _load_plugin_config(plugin_path, file_name)
            file_name = dir.get_next()

func _load_plugin_config(path: String, plugin_id: String) -> void:
    var config = ConfigFile.new()
    var err = config.load(path)
    if err == OK:
        var script_path = config.get_value("plugin", "script", "")
        if script_path != "":
            var manager = get_node("/root/GameRoot/PluginManager")
            manager.register_plugin(plugin_id, script_path)
            manager.load_plugin(plugin_id)
```

## Kasus Penggunaan Nyata
Dalam proyek RPG besar seperti game open-world, sistem pertarungan, inventaris, dan dialog dikembangkan oleh tim berbeda. Dengan arsitektur plugin-based, tim combat bisa merilis update v2.0 tanpa menunggu tim inventory selesai. Jika tim QA menemukan bug di dialog system, mereka bisa menonaktifkan plugin tersebut tanpa crash game. Contoh nyata adalah game seperti Gothic atau Mount & Blade yang menggunakan modding system serupa untuk memungkinkan komunitas menambah fitur baru.
