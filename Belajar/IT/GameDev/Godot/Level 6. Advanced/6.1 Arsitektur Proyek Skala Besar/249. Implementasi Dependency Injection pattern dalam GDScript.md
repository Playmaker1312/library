# 249. Implementasi Dependency Injection pattern dalam GDScript

## Definisi Singkat
Dependency Injection (DI) adalah design pattern di mana dependencies (dependensi) suatu objek diberikan dari luar, bukan dibuat di dalam objek itu sendiri. Dalam GDScript, ini dicapai dengan meng-inisialisasi dependencies melalui constructor injection, setter injection, atau service locator, sehingga kode lebih testable, loose coupled, dan mudah dimaintain.

## Analogi
Bayangkan kamu membuka restoran. Dulu, chef membuat semua bahan sendiri dari nol (tight coupling). Sekarang, ada supplier yang mengirim bahan baku berkualitas tinggi. Chef tidak perlu tahu bagaimana bahan diproduksi, dia hanya menerima dan menggunakannya. Dependency Injection bekerja sama — kelas menerima dependencies yang dibutuhkannya dari luar, bukan membuatnya sendiri.

## Kode GDScript (Godot 4)

```gdscript
# service_locator.gd — Central registry untuk semua services
class_name ServiceLocator
extends Node

static var instance: ServiceLocator

var services: Dictionary = {}

func _ready() -> void:
    instance = self

func register_service(service_name: String, service: Variant) -> void:
    services[service_name] = service

func get_service(service_name: String) -> Variant:
    if services.has(service_name):
        return services[service_name]
    push_warning("Service not found: " + service_name)
    return null

func has_service(service_name: String) -> bool:
    return services.has(service_name)

func unregister_service(service_name: String) -> void:
    services.erase(service_name)
```

```gdscript
# audio_manager.gd — Service untuk mengelola audio
class_name AudioManager
extends Node

var music_player: AudioStreamPlayer
var sfx_players: Array[AudioStreamPlayer] = []

func _ready() -> void:
    music_player = AudioStreamPlayer.new()
    add_child(music_player)

    for i in range(8):
        var sfx_player = AudioStreamPlayer.new()
        add_child(sfx_player)
        sfx_players.append(sfx_player)

func play_music(track: AudioStream, volume_db: float = 0.0) -> void:
    music_player.stream = track
    music_player.volume_db = volume_db
    music_player.play()

func play_sfx(sfx: AudioStream, volume_db: float = 0.0) -> void:
    for player in sfx_players:
        if not player.playing:
            player.stream = sfx
            player.volume_db = volume_db
            player.play()
            return

func stop_music() -> void:
    music_player.stop()
```

```gdscript
# player.gd — Contoh player dengan constructor injection
class_name Player
extends CharacterBody2D

var health_service: Variant
var audio_service: Variant
var save_service: Variant

var max_health: int = 100
var current_health: int = 100

func _init(health_svc: Variant, audio_svc: Variant, save_svc: Variant) -> void:
    health_service = health_svc
    audio_service = audio_svc
    save_service = save_svc

func take_damage(amount: int) -> void:
    current_health = health_service.calculate_damage(current_health, amount)
    audio_service.play_sfx(preload("res://assets/sfx/hit.wav"))

    if current_health <= 0:
        die()

func heal(amount: int) -> void:
    current_health = min(current_health + amount, max_health)
    audio_service.play_sfx(preload("res://assets/sfx/heal.wav"))

func die() -> void:
    audio_service.play_sfx(preload("res://assets/sfx/death.wav"))
    save_service.save_game_state()
    queue_free()

func save_data() -> Dictionary:
    return {
        "health": current_health,
        "position": {"x": global_position.x, "y": global_position.y}
    }
```

```gdscript
# health_service.gd — Service untuk perhitungan kesehatan
class_name HealthService
extends RefCounted

var max_health: int

func _init(max_hp: int = 100) -> void:
    max_health = max_hp

func calculate_damage(current_health: int, damage: int) -> int:
    var armor_reduction = int(damage * 0.1)
    var final_damage = max(damage - armor_reduction, 1)
    return max(current_health - final_damage, 0)

func is_alive(current_health: int) -> bool:
    return current_health > 0

func get_health_percentage(current_health: int) -> float:
    return float(current_health) / float(max_health)

func apply_heal(current_health: int, heal_amount: int) -> int:
    return min(current_health + heal_amount, max_health)
```

```gdscript
# player_factory.gd — Factory untuk membuat Player dengan dependencies
class_name PlayerFactory
extends RefCounted

var service_locator: ServiceLocator

func _init(locator: ServiceLocator) -> void:
    service_locator = locator

func create_player(spawn_position: Vector2) -> Player:
    var health_svc = service_locator.get_service("health")
    var audio_svc = service_locator.get_service("audio")
    var save_svc = service_locator.get_service("save")

    if health_svc == null:
        health_svc = HealthService.new(100)
        service_locator.register_service("health", health_svc)

    var player = Player.new(health_svc, audio_svc, save_svc)
    player.global_position = spawn_position
    return player
```

```gdscript
# enemy.gd — Contoh enemy dengan setter injection
class_name Enemy
extends CharacterBody2D

var target: Node2D
var damage_service: Variant
var navigation_agent: NavigationAgent2D

func set_damage_service(service: Variant) -> void:
    damage_service = service

func set_target(new_target: Node2D) -> void:
    target = new_target

func _ready() -> void:
    navigation_agent = NavigationAgent2D.new()
    add_child(navigation_agent)

func attack(target_node: Node2D) -> void:
    if damage_service and target_node.has_method("take_damage"):
        var damage = damage_service.calculate_enemy_damage()
        target_node.take_damage(damage)

func navigate_to_player() -> void:
    if target and navigation_agent:
        navigation_agent.target_position = target.global_position

        if navigation_agent.is_navigation_finished():
            return

        var next_path_position = navigation_agent.get_next_path_position()
        var direction = (next_path_position - global_position).normalized()
        velocity = direction * 200.0
        move_and_slide()
```

```gdscript
# game_initializer.gd — Script untuk setup semua dependencies
extends Node

func _ready() -> void:
    _setup_services()
    _setup_factories()
    _spawn_player()

func _setup_services() -> void:
    var locator = ServiceLocator.new()
    add_child(locator)

    var audio_manager = AudioManager.new()
    locator.register_service("audio", audio_manager)
    add_child(audio_manager)

    var health_service = HealthService.new(100)
    locator.register_service("health", health_service)

    var save_manager = Node.new()
    save_manager.set_script(load("res://scripts/save_manager.gd"))
    locator.register_service("save", save_manager)
    add_child(save_manager)

func _setup_factories() -> void:
    var locator = get_node("ServiceLocator")
    var player_factory = PlayerFactory.new(locator)
    set_meta("player_factory", player_factory)

func _spawn_player() -> void:
    var factory = get_meta("player_factory")
    var player = factory.create_player(Vector2(100, 100))
    add_child(player)
```

## Kasus Penggunaan Nyata
Dalam game RPG besar, sistem pertarungan membutuhkan akses ke damage calculator, audio manager, dan particle system. Tanpa DI, setiap enemy harus membuat instance baru dari service-service ini, menyebabkan duplikasi kode dan memory waste. Dengan DI, semua service didaftarkan di ServiceLocator saat game dimulai, dan setiap entity menerima reference ke service yang dibutuhkannya. Pendekatan ini digunakan oleh studio besar seperti Naughty Dog untuk game The Last of Us, di mana sistem AI, combat, dan rendering di-inject secara terpisah.
