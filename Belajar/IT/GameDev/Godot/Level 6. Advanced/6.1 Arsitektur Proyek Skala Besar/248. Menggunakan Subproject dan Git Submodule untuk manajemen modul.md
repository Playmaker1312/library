# 248. Menggunakan Subproject dan Git Submodule untuk manajemen modul

## Definisi Singkat
Subproject dan Git Submodule adalah teknik version control yang memungkinkan repository utama untuk menyertakan repository lain sebagai dependensi terpisah. Dalam konteks Godot, ini memungkinkan tim untuk mengelola modul game (seperti AI system, networking, atau UI framework) sebagai repository terpisah yang dapat di-update secara independen tanpa mempengaruhi codebase utama.

## Analogi
Bayangkan sebuah restoran franchise besar. Setiap cabang memiliki menu utama (repository utama), tetapi resep saus rahasia, roti khusus, dan bumbu marinasi masing-masing disimpan di buku resep terpisah (submodule). Ketika chef pusat mengubah resep saus, semua cabang bisa meng-update resep mereka tanpa harus mengubah seluruh menu. Saus tetap menjadi "bagian" dari menu, tetapi dikelola secara terpisah.

## Struktur Node (Scene Tree)
```
MainProject/ (Git Repository Utama)
├── .gitmodules
├── project.godot
├── game/
│   ├── main.tscn
│   └── scripts/
├── modules/ (Folder untuk semua submodule)
│   ├── ai_system/ (Git Submodule)
│   │   ├── project.godot
│   │   ├── ai_core.gd
│   │   └── behaviors/
│   ├── networking/ (Git Submodule)
│   │   ├── project.godot
│   │   ├── network_manager.gd
│   │   └── protocols/
│   └── ui_framework/ (Git Submodule)
│       ├── project.godot
│       ├── ui_base.gd
│       └── components/
└── addons/ (Godot Addons dari submodule)
    ├── ai_system/
    ├── networking/
    └── ui_framework/
```

## Kode GDScript (Godot 4)

```gdscript
# submodule_manager.gd — Script untuk mengelola loading modul dari submodule
extends Node

signal module_loaded(module_name: String)
signal module_failed(module_name: String, error: String)

var loaded_modules: Dictionary = {}
var module_paths: Dictionary = {
    "ai": "res://modules/ai_system/",
    "network": "res://modules/networking/",
    "ui": "res://modules/ui_framework/"
}

func load_module(module_name: String) -> bool:
    if loaded_modules.has(module_name):
        return true

    if not module_paths.has(module_name):
        module_failed.emit(module_name, "Module path not configured")
        return false

    var module_path = module_paths[module_name]
    var entry_script = module_path + "init.gd"

    if not FileAccess.file_exists(entry_script):
        module_failed.emit(module_name, "Entry script not found")
        return false

    var module_script = load(entry_script)
    var module_instance = module_script.new()

    if module_instance.has_method("initialize"):
        var init_result = module_instance.initialize()
        if init_result:
            loaded_modules[module_name] = module_instance
            add_child(module_instance)
            module_loaded.emit(module_name)
            return true

    module_failed.emit(module_name, "Initialization failed")
    return false

func get_module(module_name: String) -> Variant:
    return loaded_modules.get(module_name)

func unload_module(module_name: String) -> bool:
    if not loaded_modules.has(module_name):
        return false

    var module = loaded_modules[module_name]
    if module.has_method("cleanup"):
        module.cleanup()

    if module is Node:
        remove_child(module)
        module.queue_free()

    loaded_modules.erase(module_name)
    return true

func get_loaded_modules() -> Array:
    return loaded_modules.keys()
```

```gdscript
# ai_module_init.gd — Entry point untuk AI submodule
class_name AIModule
extends Node

var behavior_tree: Node
var pathfinding_system: Node

func initialize() -> bool:
    behavior_tree = _load_behavior_tree()
    pathfinding_system = _load_pathfinding()

    if behavior_tree == null or pathfinding_system == null:
        return false

    add_child(behavior_tree)
    add_child(pathfinding_system)
    return true

func cleanup() -> void:
    if behavior_tree:
        behavior_tree.queue_free()
    if pathfinding_system:
        pathfinding_system.queue_free()

func _load_behavior_tree() -> Node:
    var bt_path = "res://modules/ai_system/behaviors/behavior_tree.gd"
    if FileAccess.file_exists(bt_path):
        var bt_script = load(bt_path)
        return bt_script.new()
    return null

func _load_pathfinding() -> Node:
    var pf_path = "res://modules/ai_system/pathfinding/astar_grid.gd"
    if FileAccess.file_exists(pf_path):
        var pf_script = load(pf_path)
        return pf_script.new()
    return null
```

```gdscript
# project_initializer.gd — Script untuk inisialisasi project dengan semua submodule
extends SceneTree

func _init() -> void:
    _check_submodule_availability()
    _initialize_modules()

func _check_submodule_availability() -> void:
    var modules = ["ai", "network", "ui"]
    for module in modules:
        var path = "res://modules/" + module
        if DirAccess.dir_exists_absolute(path):
            print("Module found: ", module)
        else:
            push_warning("Module missing: ", module)

func _initialize_modules() -> void:
    var manager = Node.new()
    manager.set_script(load("res://scripts/submodule_manager.gd"))
    root.add_child(manager)

    var modules_to_load = ["ai", "network", "ui"]
    for module in modules_to_load:
        var success = manager.load_module(module)
        if success:
            print("Successfully loaded: ", module)
        else:
            push_warning("Failed to load: ", module)
```

```gdscript
# module_version_checker.gd — Memeriksa versi submodule yang terinstall
extends Node

func check_module_versions() -> Dictionary:
    var versions = {}
    var modules_dir = "res://modules/"

    var dir = DirAccess.open(modules_dir)
    if dir:
        dir.list_dir_begin()
        var file_name = dir.get_next()
        while file_name != "":
            if dir.current_is_dir() and file_name != "." and file_name != "..":
                var version_file = modules_dir + file_name + "/VERSION"
                if FileAccess.file_exists(version_file):
                    var file = FileAccess.open(version_file, FileAccess.READ)
                    if file:
                        versions[file_name] = file.get_as_text().strip_edges()
                        file.close()
            file_name = dir.get_next()

    return versions

func is_module_compatible(module_name: String, min_version: String) -> bool:
    var versions = check_module_versions()
    if not versions.has(module_name):
        return false

    var current_version = versions[module_name]
    return _compare_versions(current_version, min_version) >= 0

func _compare_versions(v1: String, v2: String) -> int:
    var parts1 = v1.split(".")
    var parts2 = v2.split(".")

    for i in range(max(parts1.size(), parts2.size())):
        var p1 = int(parts1[i]) if i < parts1.size() else 0
        var p2 = int(parts2[i]) if i < parts2.size() else 0

        if p1 > p2:
            return 1
        elif p1 < p2:
            return -1

    return 0
```

## Kasus Penggunaan Nyata
Studio game indie yang terdiri dari 5 orang sering menggunakan Git Submodule untuk mengelola engine tools, shader library, dan asset pipeline secara terpisah. Misalnya, repository untuk procedural generation bisa di-share ke 3 game berbeda tanpa copy-paste kode. Ketika ada bug fix di AI pathfinding, perubahan otomatis tersedia di semua game yang menggunakan submodule tersebut. Banyak studio profesional seperti CD Projekt Red dan Larian Studios menggunakan pendekatan serupa untuk mengelola toolchain mereka.
