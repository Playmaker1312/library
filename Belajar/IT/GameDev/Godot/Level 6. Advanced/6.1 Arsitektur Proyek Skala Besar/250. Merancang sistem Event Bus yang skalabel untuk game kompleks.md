# 250. Merancang sistem Event Bus yang skalabel untuk game kompleks

## Definisi Singkat
Event Bus adalah implementation dari publish-subscribe pattern yang berfungsi sebagai pusat komunikasi antar komponen game. Sistem ini memungkinkan node untuk mengirim dan menerima event tanpa langsung terhubung satu sama lain, menciptakan loose coupling yang memungkinkan skalabilitas tinggi pada game dengan banyak sistem yang saling berinteraksi.

## Analogi
Bayangkan sebuah stasiun radio. Ketika DJ memutar lagu (publish event), semua radio di kota yang menyiarkan frekuensi yang sama akan memutar lagu tersebut (subscribe). DJ tidak perlu tahu ada berapa radio yang mendengar, dan radio tidak perlu tahu DJ-nya siapa. Event Bus bekerja persis sama — pengirim event tidak peduli siapa yang menerima, dan penerima tidak peduli siapa yang mengirim.

## Struktur Node (Scene Tree)
```
EventBus (Node) — Singleton global event system
├── EventQueues (Node)
│   ├── ImmediateQueue — Event yang diproses langsung
│   ├── DeferredQueue — Event yang diproses di frame berikutnya
│   └── PersistentQueue — Event yang disimpan untuk diproses berulang
├── EventHistory (Node) — Log semua event untuk debugging
└── EventProcessors (Node)
    ├── PriorityProcessor — Memproses berdasarkan prioritas
    └── BatchProcessor — Memproses event dalam batch
```

## Kode GDScript (Godot 4)

```gdscript
# event_bus.gd — Central event bus singleton
extends Node

signal event_fired(event_name: String, data: Dictionary)

var subscribers: Dictionary = {}
var event_history: Array[Dictionary] = []
var max_history_size: int = 1000

var immediate_queue: Array[Dictionary] = []
var deferred_queue: Array[Dictionary] = []

func subscribe(event_name: String, callable: Callable, priority: int = 0) -> void:
    if not subscribers.has(event_name):
        subscribers[event_name] = []

    subscribers[event_name].append({
        "callable": callable,
        "priority": priority
    })

    subscribers[event_name].sort_custom(func(a, b): return a.priority > b.priority)

func unsubscribe(event_name: String, callable: Callable) -> void:
    if not subscribers.has(event_name):
        return

    var new_subscribers = []
    for sub in subscribers[event_name]:
        if not sub.callable == callable:
            new_subscribers.append(sub)

    subscribers[event_name] = new_subscribers

func emit(event_name: String, data: Dictionary = {}) -> void:
    _record_event(event_name, data)
    event_fired.emit(event_name, data)

    if not subscribers.has(event_name):
        return

    for sub in subscribers[event_name]:
        if sub.callable.is_valid():
            sub.callable.call(data)

func emit_deferred(event_name: String, data: Dictionary = {}) -> void:
    deferred_queue.append({
        "name": event_name,
        "data": data
    })

func _process(_delta: float) -> void:
    _process_deferred_queue()

func _process_deferred_queue() -> void:
    var current_queue = deferred_queue.duplicate()
    deferred_queue.clear()

    for event in current_queue:
        emit(event.name, event.data)

func _record_event(event_name: String, data: Dictionary) -> void:
    event_history.append({
        "name": event_name,
        "data": data,
        "timestamp": Time.get_unix_time_from_system()
    })

    if event_history.size() > max_history_size:
        event_history.pop_front()

func get_event_history(event_name: String = "") -> Array:
    if event_name == "":
        return event_history

    return event_history.filter(func(e): return e.name == event_name)

func clear_history() -> void:
    event_history.clear()

func get_subscriber_count(event_name: String) -> int:
    if subscribers.has(event_name):
        return subscribers[event_name].size()
    return 0
```

```gdscript
# combat_events.gd — Contoh penggunaan event untuk sistem pertarungan
class_name CombatEvents
extends Node

func _ready() -> void:
    EventBus.subscribe("damage_dealt", _on_damage_dealt)
    EventBus.subscribe("entity_died", _on_entity_died)
    EventBus.subscribe("heal_applied", _on_heal_applied)

func deal_damage(attacker: Node, target: Node, amount: int) -> void:
    EventBus.emit("damage_dealt", {
        "attacker": attacker,
        "target": target,
        "amount": amount,
        "timestamp": Time.get_ticks_msec()
    })

func _on_damage_dealt(data: Dictionary) -> void:
    var target = data.get("target")
    var amount = data.get("amount", 0)

    if target.has_method("take_damage"):
        target.take_damage(amount)

    EventBus.emit_deferred("damage_number_show", {
        "position": target.global_position,
        "amount": amount
    })

func _on_entity_died(data: Dictionary) -> void:
    var entity = data.get("entity")
    var killer = data.get("killer")

    EventBus.emit("experience_gained", {
        "entity": entity,
        "xp_amount": entity.get_meta("xp_reward", 10)
    })

    EventBus.emit("loot_dropped", {
        "position": entity.global_position,
        "loot_table": entity.get_meta("loot_table", [])
    })

func _on_heal_applied(data: Dictionary) -> void:
    var target = data.get("target")
    var amount = data.get("amount", 0)

    EventBus.emit("health_bar_update", {
        "entity": target,
        "new_health": target.current_health
    })
```

```gdscript
# ui_system.gd — Contoh UI yang mendengarkan event
extends CanvasLayer

var health_bars: Dictionary = {}
var damage_numbers: Array = []

func _ready() -> void:
    EventBus.subscribe("health_bar_update", _update_health_bar)
    EventBus.subscribe("damage_number_show", _show_damage_number)
    EventBus.subscribe("loot_dropped", _show_loot_notification)
    EventBus.subscribe("quest_completed", _show_quest_notification)

func _update_health_bar(data: Dictionary) -> void:
    var entity = data.get("entity")
    var new_health = data.get("new_health", 0)

    if not health_bars.has(entity):
        _create_health_bar(entity)

    var health_bar = health_bars[entity]
    var tween = create_tween()
    tween.tween_property(health_bar, "value", new_health, 0.3)

func _create_health_bar(entity: Node) -> void:
    var health_bar = ProgressBar.new()
    health_bar.max_value = entity.get_meta("max_health", 100)
    health_bar.value = entity.current_health
    health_bar.size = Vector2(100, 10)
    health_bar.position = entity.global_position + Vector2(-50, -30)
    add_child(health_bar)
    health_bars[entity] = health_bar

func _show_damage_number(data: Dictionary) -> void:
    var position = data.get("position", Vector2.ZERO)
    var amount = data.get("amount", 0)

    var label = Label.new()
    label.text = str(amount)
    label.position = position
    label.add_theme_color_override("font_color", Color.RED)
    add_child(label)

    var tween = create_tween()
    tween.tween_property(label, "position:y", position.y - 50, 0.5)
    tween.tween_callback(label.queue_free)

func _show_loot_notification(data: Dictionary) -> void:
    var loot = data.get("loot", "Unknown Item")
    print("Loot dropped: ", loot)

func _show_quest_notification(data: Dictionary) -> void:
    var quest_name = data.get("quest_name", "")
    print("Quest completed: ", quest_name)
```

```gdscript
# inventory_system.gd — Sistem inventaris dengan event-driven
extends Node

var items: Array[Dictionary] = []
var max_slots: int = 20

func add_item(item_data: Dictionary) -> bool:
    if items.size() >= max_slots:
        EventBus.emit("inventory_full", {"item": item_data})
        return false

    var existing_item = _find_item(item_data.id)
    if existing_item and existing_item.stackable:
        existing_item.quantity += item_data.get("quantity", 1)
        EventBus.emit("item_updated", {
            "item": existing_item,
            "action": "stack"
        })
        return true

    item_data.quantity = item_data.get("quantity", 1)
    items.append(item_data)

    EventBus.emit("item_added", {
        "item": item_data,
        "slot_index": items.size() - 1
    })
    return true

func remove_item(item_id: String, quantity: int = 1) -> bool:
    for i in range(items.size()):
        if items[i].id == item_id:
            items[i].quantity -= quantity
            if items[i].quantity <= 0:
                var removed_item = items[i]
                items.remove_at(i)
                EventBus.emit("item_removed", {
                    "item": removed_item,
                    "slot_index": i
                })
            else:
                EventBus.emit("item_updated", {
                    "item": items[i],
                    "action": "quantity_change"
                })
            return true
    return false

func _find_item(item_id: String) -> Variant:
    for item in items:
        if item.id == item_id:
            return item
    return null

func use_item(item_id: String) -> void:
    var item = _find_item(item_id)
    if item:
        EventBus.emit("item_used", {"item": item})

func get_item_count(item_id: String) -> int:
    var item = _find_item(item_id)
    return item.quantity if item else 0
```

## Kasus Penggunaan Nyata
Dalam game MMORPG seperti World of Warcraft atau Final Fantasy XIV, ada puluhan sistem yang saling berinteraksi: combat, inventory, quest, achievement, crafting, social, dan lainnya. Tanpa Event Bus, setiap sistem harus mengimpor dan memanggil metode sistem lain secara langsung, menciptakan spaghetti code yang sulit dimaintain. Dengan Event Bus, sistem combat mengirim event "monster_killed", dan sistem quest, achievement, crafting, serta loot secara independen merespon event tersebut. Studio seperti ArenaNet (Guild Wars 2) menggunakan pattern serupa untuk menangani kompleksitas sistem mereka.
