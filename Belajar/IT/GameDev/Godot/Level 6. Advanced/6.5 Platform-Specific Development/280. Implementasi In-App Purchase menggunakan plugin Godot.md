# 280. Implementasi In-App Purchase Menggunakan Plugin Godot

## Definisi Singkat
In-App Purchase (IAP) adalah sistem pembelian digital yang memungkinkan pemain membeli item, konten, atau layanan premium dalam game menggunakan uang nyata melalui platform distribusi seperti Google Play Billing dan Apple StoreKit. Implementasi di Godot dilakukan melalui plugin atau REST API.

## Analogi
Bayangkan kamu memiliki toko merchandise di dalam game (toko virtual). Pemain bisa membeli item nyata dengan uang asli melalui sistem kasir yang terhubung ke bank (platform payment). IAP adalah "mesin kasir" yang memproses transaksi, memverifikasi pembayaran, dan mengirim item ke pemain.

## Kode GDScript (Godot 4)
```gdscript
extends Node

signal purchase_completed(product_id: String, success: bool)
signal purchase_restored(product_id: String)
signal products_loaded(products: Array)
signal iap_error(error_code: String, message: String)

var is_iap_available: bool = false
var platform: String = ""
var product_catalog: Dictionary = {}
var pending_purchases: Array = []

func _ready() -> void:
    platform = OS.get_name()
    _initialize_iap()

func _initialize_iap() -> void:
    match platform:
        "Android":
            _init_google_play_billing()
        "iOS":
            _init_apple_storekit()
        _:
            print("IAP tidak didukung di platform: ", platform)
            is_iap_available = false

func _init_google_play_billing() -> void:
    print("Inisialisasi Google Play Billing...")
    is_iap_available = true
    _load_product_catalog()

func _init_apple_storekit() -> void:
    print("Inisialisasi Apple StoreKit...")
    is_iap_available = true
    _load_product_catalog()

func _load_product_catalog() -> void:
    product_catalog = {
        "com.game.coin_pack_100": {
            "name": "100 Coins",
            "description": "Pack 100 koin emas",
            "price": "$0.99",
            "type": "consumable",
            "currency": "USD"
        },
        "com.game.coin_pack_500": {
            "name": "500 Coins",
            "description": "Pack 500 koin emas",
            "price": "$3.99",
            "type": "consumable",
            "currency": "USD"
        },
        "com.game.premium_upgrade": {
            "name": "Premium Upgrade",
            "description": "Unlock semua fitur premium",
            "price": "$4.99",
            "type": "non_consumable",
            "currency": "USD"
        },
        "com.game.monthly_subscription": {
            "name": "Monthly VIP",
            "description": "Akses VIP selama 1 bulan",
            "price": "$9.99/month",
            "type": "subscription",
            "currency": "USD"
        }
    }
    products_loaded.emit(product_catalog.values())
    print("Katalog produk dimuat: ", product_catalog.size(), " produk")

func purchase_product(product_id: String) -> void:
    if not is_iap_available:
        iap_error.emit("UNAVAILABLE", "IAP tidak tersedia di platform ini")
        return
    if not product_catalog.has(product_id):
        iap_error.emit("NOT_FOUND", "Produk tidak ditemukan: " + product_id)
        return
    var product = product_catalog[product_id]
    print("Memulai pembelian: ", product["name"])
    pending_purchases.append({
        "product_id": product_id,
        "timestamp": Time.get_unix_time_from_system(),
        "status": "pending"
    })
    _process_purchase(product_id, product)

func _process_purchase(product_id: String, product: Dictionary) -> void:
    match platform:
        "Android":
            _process_google_play_purchase(product_id, product)
        "iOS":
            _process_apple_purchase(product_id, product)

func _process_google_play_purchase(product_id: String, product: Dictionary) -> void:
    print("Google Play Billing: Memproses pembelian ", product_id)
    var purchase_token = "android_purchase_token_" + str(Time.get_unix_time_from_system())
    _complete_purchase(product_id, purchase_token)

func _process_apple_purchase(product_id: String, product: Dictionary) -> void:
    print("Apple StoreKit: Memproses pembelian ", product_id)
    var transaction_id = "ios_transaction_" + str(Time.get_unix_time_from_system())
    _complete_purchase(product_id, transaction_id)

func _complete_purchase(product_id: String, transaction_id: String) -> void:
    var purchase = {
        "product_id": product_id,
        "transaction_id": transaction_id,
        "timestamp": Time.get_unix_time_from_system(),
        "status": "completed"
    }
    _grant_product(product_id)
    purchase_completed.emit(product_id, true)
    print("Pembelian berhasil: ", product_id)

func _grant_product(product_id: String) -> void:
    var product = product_catalog.get(product_id, {})
    match product.get("type", ""):
        "consumable":
            _grant_consumable(product_id)
        "non_consumable":
            _grant_non_consumable(product_id)
        "subscription":
            _grant_subscription(product_id)

func _grant_consumable(product_id: String) -> void:
    var coin_amounts = {
        "com.game.coin_pack_100": 100,
        "com.game.coin_pack_500": 500
    }
    var amount = coin_amounts.get(product_id, 0)
    print("Granting ", amount, " coins untuk ", product_id)

func _grant_non_consumable(product_id: String) -> void:
    print("Granting non-consumable: ", product_id)
    var save_data = {"purchased": true, "product_id": product_id}
    _save_purchase_record(product_id, save_data)

func _grant_subscription(product_id: String) -> void:
    print("Granting subscription: ", product_id)
    var subscription_data = {
        "active": true,
        "product_id": product_id,
        "start_date": Time.get_unix_time_from_system(),
        "end_date": Time.get_unix_time_from_system() + 2592000
    }
    _save_purchase_record(product_id, subscription_data)

func restore_purchases() -> void:
    if not is_iap_available:
        return
    print("Memulihkan pembelian sebelumnya...")
    var purchases = _get_purchase_history()
    for purchase in purchases:
        purchase_restored.emit(purchase["product_id"])
        print("Pembelian dipulihkan: ", purchase["product_id"])

func _get_purchase_history() -> Array:
    var history = []
    var save_path = "user://purchase_history.json"
    if FileAccess.file_exists(save_path):
        var file = FileAccess.open(save_path, FileAccess.READ)
        if file:
            var json = JSON.new()
            var error = json.parse(file.get_as_text())
            if error == OK:
                history = json.data
    return history

func _save_purchase_record(product_id: String, data: Dictionary) -> void:
    var history = _get_purchase_history()
    history.append({
        "product_id": product_id,
        "data": data,
        "timestamp": Time.get_unix_time_from_system()
    })
    var save_path = "user://purchase_history.json"
    var file = FileAccess.open(save_path, FileAccess.WRITE)
    if file:
        file.store_string(JSON.stringify(history))
        file.close()

func is_product_purchased(product_id: String) -> bool:
    var history = _get_purchase_history()
    for purchase in history:
        if purchase["product_id"] == product_id:
            return true
    return false

func get_product_price(product_id: String) -> String:
    var product = product_catalog.get(product_id, {})
    return product.get("price", "N/A")

func get_available_products() -> Array[Dictionary]:
    return product_catalog.values()

func consume_purchase(product_id: String) -> void:
    print("Consuming purchase: ", product_id)
    var history = _get_purchase_history()
    var updated_history = []
    for purchase in history:
        if purchase["product_id"] != product_id:
            updated_history.append(purchase)
    var save_path = "user://purchase_history.json"
    var file = FileAccess.open(save_path, FileAccess.WRITE)
    if file:
        file.store_string(JSON.stringify(updated_history))
        file.close()

func validate_receipt(receipt_data: String) -> bool:
    if receipt_data.is_empty():
        return false
    var json = JSON.new()
    var error = json.parse(receipt_data)
    if error != OK:
        return false
    var receipt = json.data
    var is_valid = receipt.has("product_id") and receipt.has("transaction_id")
    if not is_valid:
        iap_error.emit("INVALID_RECEIPT", "Receipt tidak valid")
    return is_valid

func get_store_status() -> Dictionary:
    return {
        "platform": platform,
        "iap_available": is_iap_available,
        "products_count": product_catalog.size(),
        "pending_purchases": pending_purchases.size()
    }
```

## Kasus Penggunaan Nyata
Game mobile free-to-play seperti "Clash Royale" dan "Candy Crush Saga" menggunakan sistem IAP untuk menjual item kosmetik dan boost. Godot memungkinkan implementasi IAP melalui plugin seperti "GodotGooglePlayBilling" yang tersedia di GitHub untuk integrasi langsung dengan Google Play Billing Library.
