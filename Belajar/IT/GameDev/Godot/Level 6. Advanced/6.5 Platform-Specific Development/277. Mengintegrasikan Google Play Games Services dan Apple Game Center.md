# 277. Mengintegrasikan Google Play Games Services dan Apple Game Center

## Definisi Singkat
Google Play Games Services dan Apple Game Center adalah platform layanan yang menyediakan fitur achievements, leaderboards, saved games, dan multiplayer untuk game mobile. Integrasi dilakukan menggunakan plugin Godot atau REST API untuk menghubungkan game dengan layanan cloud platform.

## Analogi
Bayangkan sebuah arcade game yang memiliki papan skor komunitas (leaderboard) dan lencana pencapaian (achievements). Google Play Games dan Apple Game Center adalah "papan skor dunia nyata" yang menghubungkan pemain di seluruh dunia, menyimpan kemajuan mereka di cloud, dan memungkinkan mereka bersaing secara online.

## Struktur Node (Scene Tree)
```
Main (Node)
├── GameManager (Node)
│   ├── AchievementManager (Node)
│   ├── LeaderboardManager (Node)
│   └── CloudSaveManager (Node)
├── PlatformBridge (Node)
│   ├── AndroidBridge (Node)
│   └── iOSBridge (Node)
└── UI (CanvasLayer)
    ├── AchievementsPanel (Control)
    ├── LeaderboardPanel (Control)
    └── StatusLabel (Label)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

signal achievement_unlocked(achievement_id: String)
signal leaderboard_updated(leaderboard_id: String)
signal cloud_save_completed(success: bool)

var platform: String = ""
var is_authenticated: bool = false
var player_id: String = ""
var player_display_name: String = ""

func _ready() -> void:
    platform = OS.get_name()
    _initialize_platform_services()

func _initialize_platform_services() -> void:
    match platform:
        "Android":
            _init_google_play_services()
        "iOS":
            _init_apple_game_center()
        _:
            print("Platform tidak didukung untuk game services: ", platform)

func _init_google_play_services() -> void:
    print("Inisialisasi Google Play Games Services...")
    is_authenticated = true
    player_id = "android_player_001"
    player_display_name = "Android Player"
    print("Google Play Games terhubung untuk: ", player_display_name)

func _init_apple_game_center() -> void:
    print("Inisialisasi Apple Game Center...")
    is_authenticated = true
    player_id = "ios_player_001"
    player_display_name = "iOS Player"
    print("Apple Game Center terhubung untuk: ", player_display_name)

func unlock_achievement(achievement_id: String, progress: float = 100.0) -> void:
    if not is_authenticated:
        push_warning("Belum terautentikasi")
        return
    var achievement_data = {
        "id": achievement_id,
        "progress": progress,
        "platform": platform,
        "timestamp": Time.get_unix_time_from_system()
    }
    print("Membuka achievement: ", achievement_data)
    achievement_unlocked.emit(achievement_id)
    _report_to_platform(achievement_data)

func submit_leaderboard_score(leaderboard_id: String, score: int) -> void:
    if not is_authenticated:
        push_warning("Belum terautentikasi")
        return
    var score_data = {
        "leaderboard_id": leaderboard_id,
        "score": score,
        "platform": platform,
        "timestamp": Time.get_unix_time_from_system()
    }
    print("Mengirim skor ke leaderboard: ", score_data)
    leaderboard_updated.emit(leaderboard_id)
    _report_to_platform(score_data)

func save_to_cloud(save_name: String, data: Dictionary) -> void:
    if not is_authenticated:
        push_warning("Belum terautentikasi")
        return
    var save_data = {
        "name": save_name,
        "data": data,
        "platform": platform,
        "timestamp": Time.get_unix_time_from_system(),
        "player_id": player_id
    }
    print("Menyimpan ke cloud: ", save_name)
    _report_to_platform(save_data)
    cloud_save_completed.emit(true)

func load_from_cloud(save_name: String) -> Dictionary:
    if not is_authenticated:
        return {}
    var load_request = {
        "name": save_name,
        "platform": platform,
        "player_id": player_id
    }
    print("Memuat dari cloud: ", load_request)
    return {"status": "success", "data": {}}

func get_achievements_list() -> Array[Dictionary]:
    return [
        {"id": "first_win", "name": "Kemenangan Pertama", "description": "Menangkan game pertama", "unlocked": false},
        {"id": "score_1000", "name": "Skor 1000", "description": "Raih skor 1000", "unlocked": false},
        {"id": "play_10_games", "name": "Pemain Aktif", "description": "Mainkan 10 game", "unlocked": false}
    ]

func get_leaderboard_entries(leaderboard_id: String, limit: int = 10) -> Array[Dictionary]:
    var entries = []
    for i in range(limit):
        entries.append({
            "rank": i + 1,
            "player_name": "Player_" + str(i + 1),
            "score": 10000 - (i * 500),
            "platform": platform
        })
    return entries

func _report_to_platform(data: Dictionary) -> void:
    match platform:
        "Android":
            _report_to_google_play(data)
        "iOS":
            _report_to_apple_game_center(data)

func _report_to_google_play(data: Dictionary) -> void:
    print("Google Play: ", data)

func _report_to_apple_game_center(data: Dictionary) -> void:
    print("Apple Game Center: ", data)

func get_player_stats() -> Dictionary:
    return {
        "player_id": player_id,
        "display_name": player_display_name,
        "platform": platform,
        "authenticated": is_authenticated,
        "total_games_played": 42,
        "high_score": 15000
    }
```

## Kasus Penggunaan Nyata
Game mobile seperti "Crossy Road" dan "Alto's Adventure" menggunakan Google Play Games untuk leaderboard global dan achievements. Godot memungkinkan integrasi melalui plugin seperti "Godot Google Play Services" yang tersedia di Asset Library untuk menambahkan fitur cloud save dan social features.
