# 276. Implementasi Fitur Mobile Native

## Definisi Singkat
Fitur mobile native adalah kemampuan akses hardware perangkat mobile seperti touch input, gesture, accelerometer, dan gyroscope melalui API Godot yang tersedia pada InputEvent dan JoyPad. Implementasinya memungkinkan game merespons input fisik perangkat secara langsung.

## Analogi
Bayangkan kamu bermain alat musik tradisional (game). Untuk memainkannya dengan baik, kamu perlu memahami cara jari-jarimu menyentuh senar (touch), menggoyangkan tubuhmu (accelerometer), dan membuat gerakan tertentu (gesture). Fitur mobile native adalah "mata dan tangan" yang membaca semua gerakan itu.

## Struktur Node (Scene Tree)
```
Main (Node2D)

```

## Kode GDScript (Godot 4)
```gdscript
extends Node2D

var touch_positions: Dictionary = {}
var swipe_threshold: float = 50.0
var swipe_start: Vector2 = Vector2.ZERO
var accelerometer_data: Vector3 = Vector3.ZERO
var is_swiping: bool = false

func _ready() -> void:
    Input.set_emulate_touch_from_mouse(true)
    Input.set_emulate_mouse_from_touch(false)

func _unhandled_input(event: InputEvent) -> void:
    if event is InputEventScreenTouch:
        _handle_touch(event)
    elif event is InputEventScreenDrag:
        _handle_drag(event)
    elif event is InputEventScreenPan:
        _handle_pan(event)

func _handle_touch(event: InputEventScreenTouch) -> void:
    if event.pressed:
        touch_positions[event.index] = event.position
        swipe_start = event.position
        is_swiping = true
        print("Touch dimulai pada: ", event.position)
    else:
        if is_swiping:
            var swipe_end = event.position
            var swipe_distance = swipe_start.distance_to(swipe_end)
            if swipe_distance > swipe_threshold:
                _detect_swipe_direction(swipe_start, swipe_end)
            is_swiping = false
        touch_positions.erase(event.index)

func _handle_drag(event: InputEventScreenDrag) -> void:
    touch_positions[event.index] = event.position
    if touch_positions.size() == 1:
        _process_single_touch(event.position)
    elif touch_positions.size() == 2:
        _process_pinch_zoom()

func _handle_pan(event: InputEventScreenPan) -> void:
    var pan_velocity = event.velocity
    if pan_velocity.length() > 100.0:
        _apply_camera_pan(pan_velocity)

func _process_single_touch(position: Vector2) -> void:
    var viewport_center = get_viewport().get_visible_rect().size / 2.0
    var direction = (position - viewport_center).normalized()
    _move_player(direction)

func _process_pinch_zoom() -> void:
    if touch_positions.size() != 2:
        return
    var keys = touch_positions.keys()
    var pos1 = touch_positions[keys[0]]
    var pos2 = touch_positions[keys[1]]
    var current_distance = pos1.distance_to(pos2)
    print("Pinch distance: ", current_distance)

func _detect_swipe_direction(start: Vector2, end: Vector2) -> void:
    var direction = (end - start).normalized()
    var angle = direction.angle()
    if angle > -PI/4 and angle < PI/4:
        print("Swipe kanan")
    elif angle > PI/4 and angle < 3*PI/4:
        print("Swipe bawah")
    elif angle < -PI/4 and angle > -3*PI/4:
        print("Swipe atas")
    else:
        print("Swipe kiri")

func _apply_camera_pan(velocity: Vector2) -> void:
    var camera = get_viewport().get_camera_2d()
    if camera:
        camera.position -= velocity * get_process_delta_time()

func _move_player(direction: Vector2) -> void:
    var player = get_node_or_null("Player")
    if player and player is CharacterBody2D:
        player.velocity = direction * 200.0
        player.move_and_slide()

func _process(delta: float) -> void:
    _read_accelerometer()
    _read_gyroscope()

func _read_accelerometer() -> void:
    if Input.get_accelerometer() != Vector3.ZERO:
        accelerometer_data = Input.get_accelerometer()
        var tilt_angle = atan2(accelerometer_data.x, accelerometer_data.y)
        print("Tilt angle: ", rad_to_deg(tilt_angle))

func _read_gyroscope() -> void:
    if Input.get_gyroscope() != Vector3.ZERO:
        var gyro = Input.get_gyroscope()
        print("Gyroscope: ", gyro)

func setup_virtual_joystick(center: Vector2, radius: float) -> Dictionary:
    return {
        "center": center,
        "radius": radius,
        "dead_zone": radius * 0.2,
        "input_vector": Vector2.ZERO
    }

func process_virtual_joystick(joystick: Dictionary, touch_pos: Vector2) -> Vector2:
    var raw_vector = touch_pos - joystick["center"]
    var distance = raw_vector.length()
    if distance < joystick["dead_zone"]:
        joystick["input_vector"] = Vector2.ZERO
    else:
        joystick["input_vector"] = raw_vector.normalized()
    return joystick["input_vector"]
```

## Kasus Penggunaan Nyata
Game mobile seperti "Alto's Odyssey" dan "Crossy Road" menggunakan touch gesture untuk kontrol swipe dan tilt. Godot memungkinkan implementasi virtual joystick seperti yang digunakan dalam game "Brawl Stars" untuk kontrol analog virtual yang responsif.
