# 279. Memahami Persyaratan Submission ke Platform Store

## Definisi Singkat
Persyaratan submission adalah serangkaian ketentuan teknis dan non-teknis yang harus dipenuhi agar game dapat diterima di platform distribusi digital seperti Steam, Google Play, dan App Store. Setiap platform memiliki standar kualitas, ukuran file, konten, dan dokumentasi yang berbeda.

## Analogi
Bayangkan kamu ingin menjual karya seni di galeri (platform store). Setiap galeri memiliki aturan tentang ukuran frame (ukuran file), jenis karya (konten), deskripsi (metadata), dan proses kurasi (review). Kamu harus memahami dan memenuhi semua aturan ini agar karyamu diterima dipamerkan.

## Kode GDScript (Godot 4)
```gdscript
extends Node

signal validation_completed(platform: String, passed: bool)
signal submission_ready(platform: String)

var platform_requirements: Dictionary = {}
var current_submission: Dictionary = {}

func _ready() -> void:
    _initialize_platform_requirements()

func _initialize_platform_requirements() -> void:
    platform_requirements = {
        "Steam": {
            "min_build_size_mb": 100,
            "max_build_size_gb": 10,
            "required_files": ["steam_appid.txt", "gameinfo.gi"],
            "supported_platforms": ["Windows", "macOS", "Linux"],
            "content_rating": "Universal",
            "review_time_days": 7
        },
        "Google Play": {
            "min_sdk": 21,
            "target_sdk": 33,
            "max_apk_size_mb": 150,
            "required_permissions": [],
            "content_rating_required": true,
            "privacy_policy_required": true,
            "review_time_hours": 24
        },
        "App Store": {
            "min_ios_version": "12.0",
            "max_build_size_gb": 4,
            "required_files": ["Info.plist", "Icon.icns"],
            "content_rating_required": true,
            "privacy_policy_required": true,
            "review_time_days": 7,
            "requires_app_review": true
        }
    }

func validate_submission(platform: String, build_path: String) -> Dictionary:
    var requirements = platform_requirements.get(platform, {})
    var validation_results = {
        "platform": platform,
        "passed": true,
        "errors": [],
        "warnings": [],
        "info": []
    }
    _check_build_size(platform, build_path, validation_results, requirements)
    _check_required_files(platform, build_path, validation_results, requirements)
    _check_platform_specific(platform, validation_results, requirements)
    _check_content_compliance(platform, validation_results, requirements)
    validation_completed.emit(platform, validation_results["passed"])
    return validation_results

func _check_build_size(platform: String, path: String, results: Dictionary, req: Dictionary) -> void:
    var dir = DirAccess.open(path)
    if dir:
        var total_size = 0
        dir.list_dir_begin()
        var file_name = dir.get_next()
        while file_name != "":
            if not dir.current_is_dir():
                total_size += dir.get_current_dir().path_join(file_name).get_file_size()
            file_name = dir.get_next()
        var size_mb = total_size / 1048576
        var max_size = req.get("max_build_size_mb", 1000)
        if size_mb > max_size:
            results["errors"].append("Ukuran build terlalu besar: %dMB > %dMB" % [size_mb, max_size])
            results["passed"] = false
        else:
            results["info"].append("Ukuran build: %dMB (OK)" % size_mb)

func _check_required_files(platform: String, path: String, results: Dictionary, req: Dictionary) -> void:
    var required_files = req.get("required_files", [])
    for file in required_files:
        var file_path = path.path_join(file)
        if not FileAccess.file_exists(file_path):
            results["errors"].append("File tidak ditemukan: %s" % file)
            results["passed"] = false
        else:
            results["info"].append("File ditemukan: %s" % file)

func _check_platform_specific(platform: String, results: Dictionary, req: Dictionary) -> void:
    match platform:
        "Steam":
            _check_steam_specific(results, req)
        "Google Play":
            _check_google_play_specific(results, req)
        "App Store":
            _check_app_store_specific(results, req)

func _check_steam_specific(results: Dictionary, req: Dictionary) -> void:
    results["info"].append("Steam: Pastikan steam_appid.txt berisi App ID yang benar")
    results["info"].append("Steam: Verifikasi Steamworks SDK terintegrasi")

func _check_google_play_specific(results: Dictionary, req: Dictionary) -> void:
    if req.get("privacy_policy_required", false):
        results["warnings"].append("Google Play: Siapkan URL Privacy Policy")
    if req.get("content_rating_required", false):
        results["warnings"].append("Google Play: Isi Content Rating Questionnaire")

func _check_app_store_specific(results: Dictionary, req: Dictionary) -> void:
    if req.get("requires_app_review", false):
        results["info"].append("App Store: Siapkan deskripsi fitur app review")
    if req.get("privacy_policy_required", false):
        results["warnings"].append("App Store: Siapkan URL Privacy Policy")

func _check_content_compliance(platform: String, results: Dictionary, req: Dictionary) -> void:
    results["info"].append("Pastikan tidak ada konten yang melanggar pedoman platform")

func prepare_store_metadata(platform: String) -> Dictionary:
    var metadata = {
        "title": ProjectSettings.get_setting("application/config/name"),
        "description": "",
        "short_description": "",
        "version": ProjectSettings.get_setting("application/config/version"),
        "developer": "",
        "icons": {},
        "screenshots": [],
        "tags": [],
        "category": ""
    }
    match platform:
        "Steam":
            metadata["tags"] = ["Singleplayer", "Indie", "Action"]
            metadata["category"] = "Game"
        "Google Play":
            metadata["category"] = "Games"
            metadata["content_rating"] = "Everyone"
        "App Store":
            metadata["category"] = "Games"
            metadata["age_rating"] = "4+"
    return metadata

func generate_submission_checklist(platform: String) -> Array[Dictionary]:
    var checklist = []
    match platform:
        "Steam":
            checklist = [
                {"item": "Build terkompilasi", "status": false},
                {"item": "steam_appid.txt", "status": false},
                {"item": "Screenshot 5-10", "status": false},
                {"item": "Trailer video", "status": false},
                {"item": "Deskripsi game", "status": false},
                {"item": "Store page setup", "status": false}
            ]
        "Google Play":
            checklist = [
                {"item": "APK/AAB terkompilasi", "status": false},
                {"item": "Ikon 512x512", "status": false},
                {"item": "Screenshot", "status": false},
                {"item": "Feature graphic", "status": false},
                {"item": "Privacy policy URL", "status": false},
                {"item": "Content rating", "status": false}
            ]
        "App Store":
            checklist = [
                {"item": "IPA terkompilasi", "status": false},
                {"item": "App icon 1024x1024", "status": false},
                {"item": "Screenshot", "status": false},
                {"item": "App description", "status": false},
                {"item": "Privacy policy URL", "status": false},
                {"item": "App Review information", "status": false}
            ]
    return checklist

func get_review_timeline(platform: String) -> Dictionary:
    var timelines = {
        "Steam": {"initial": "7-14 hari", "update": "1-3 hari"},
        "Google Play": {"initial": "24-48 jam", "update": "24 jam"},
        "App Store": {"initial": "7-14 hari", "update": "1-3 hari"}
    }
    return timelines.get(platform, {"initial": "Unknown", "update": "Unknown"})
```

## Kasus Penggunaan Nyata
Banyak game indie yang berhasil di Steam seperti "Hollow Knight" dan "Celeste" melewati proses submission yang ketat termasuk review konten dan kualitas teknis. Godot sendiri telah digunakan untuk game yang berhasil di Steam seperti "DungeonRim" yang memenuhi semua persyaratan Valve untuk distribusi digital.
