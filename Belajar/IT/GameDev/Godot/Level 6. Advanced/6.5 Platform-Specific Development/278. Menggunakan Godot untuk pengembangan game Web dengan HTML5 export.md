# 278. Menggunakan Godot untuk Pengembangan Game Web dengan HTML5 Export

## Definisi Singkat
HTML5 export adalah metode mengekspor game Godot menjadi file web yang dapat dijalankan di browser menggunakan WebGL/WebAssembly. Proses ini menghasilkan file HTML, JavaScript, dan PCK yang memungkinkan game dijalankan tanpa instalasi di platform web.

## Analogi
Bayangkan kamu memiliki sebuah mesin arcade (game Godot). Untuk memainkannya di kafe internet (browser), kamu perlu mengonversinya menjadi aplikasi web yang bisa diakses melalui browser. HTML5 export adalah proses "memindahkan" mesin arcade ke dalam bentuk yang bisa dijalankan di browser.

## Struktur Node (Scene Tree)
```
Main (Node2D)
├── WebGameManager (Node)
│   ├── BrowserDetector (Node)
│   ├── InputAdapter (Node)
│   └── AudioContext (Node)
├── GameWorld (Node2D)
│   ├── Player (CharacterBody2D)
│   └── Environment (Node2D)
└── WebUI (CanvasLayer)
    ├── LoadingScreen (Control)
    ├── PauseOverlay (Control)
    └── FullscreenButton (Button)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

var is_web: bool = false
var browser_info: Dictionary = {}
var is_fullscreen: bool = false
var canvas_element: JavaScriptObject

func _ready() -> void:
    is_web = OS.get_name() == "Web"
    if is_web:
        _initialize_web_features()
        _detect_browser()
        _setup_web_audio()

func _initialize_web_features() -> void:
    canvas_element = JavaScriptBridge.get_canvas()
    JavaScriptBridge.eval("document.body.style.margin = '0'")
    JavaScriptBridge.eval("document.body.style.overflow = 'hidden'")
    print("Web features diinisialisasi")

func _detect_browser() -> void:
    var user_agent = JavaScriptBridge.eval("navigator.userAgent")
    browser_info = {
        "user_agent": user_agent,
        "platform": JavaScriptBridge.eval("navigator.platform"),
        "language": JavaScriptBridge.eval("navigator.language"),
        "cookie_enabled": JavaScriptBridge.eval("navigator.cookieEnabled")
    }
    print("Browser terdeteksi: ", browser_info)

func _setup_web_audio() -> void:
    JavaScriptBridge.eval("""
        if (typeof AudioContext !== 'undefined') {
            window.audioContext = new AudioContext();
        }
    """)

func toggle_fullscreen() -> void:
    if not is_web:
        return
    is_fullscreen = !is_fullscreen
    if is_fullscreen:
        JavaScriptBridge.eval("""
            var canvas = document.querySelector('canvas');
            if (canvas.requestFullscreen) {
                canvas.requestFullscreen();
            } else if (canvas.webkitRequestFullscreen) {
                canvas.webkitRequestFullscreen();
            }
        """)
    else:
        JavaScriptBridge.eval("""
            if (document.exitFullscreen) {
                document.exitFullscreen();
            } else if (document.webkitExitFullscreen) {
                document.webkitExitFullscreen();
            }
        """)

func save_to_local_storage(key: String, data: Dictionary) -> void:
    if not is_web:
        return
    var json_string = JSON.stringify(data)
    JavaScriptBridge.eval("localStorage.setItem('%s', '%s')" % [key, json_string])
    print("Data disimpan ke localStorage: ", key)

func load_from_local_storage(key: String) -> Dictionary:
    if not is_web:
        return {}
    var result = JavaScriptBridge.eval("localStorage.getItem('%s')" % key)
    if result == "null" or result == "":
        return {}
    var json = JSON.new()
    var error = json.parse(result)
    if error == OK:
        return json.data
    return {}

func share_on_social_media(title: String, text: String, url: String) -> void:
    if not is_web:
        return
    var share_url = "https://twitter.com/intent/tweet?text=%s&url=%s" % [text.uri_encode(), url.uri_encode()]
    JavaScriptBridge.eval("window.open('%s', '_blank')" % share_url)

func copy_to_clipboard(text: String) -> void:
    if not is_web:
        return
    JavaScriptBridge.eval("navigator.clipboard.writeText('%s')" % text.uri_encode())

func get_web_performance() -> Dictionary:
    if not is_web:
        return {}
    var performance = JavaScriptBridge.eval("""
        JSON.stringify({
            memory: performance.memory ? {
                usedJSHeapSize: performance.memory.usedJSHeapSize,
                totalJSHeapSize: performance.memory.totalJSHeapSize
            } : null,
            timing: {
                loadEventEnd: performance.timing.loadEventEnd,
                navigationStart: performance.timing.navigationStart
            }
        })
    """)
    var json = JSON.new()
    var error = json.parse(performance)
    if error == OK:
        return json.data
    return {}

func _process(_delta: float) -> void:
    if is_web:
        _handle_web_input()

func _handle_web_input() -> void:
    if Input.is_action_just_pressed("ui_fullscreen"):
        toggle_fullscreen()

func create_web_save_system() -> Dictionary:
    return {
        "save": func(key: String, data: Dictionary):
            save_to_local_storage(key, data),
        "load": func(key: String) -> Dictionary:
            return load_from_local_storage(key),
        "delete": func(key: String):
            JavaScriptBridge.eval("localStorage.removeItem('%s')" % key),
        "clear": func():
            JavaScriptBridge.eval("localStorage.clear()")
    }

func get_web_audio_context() -> JavaScriptObject:
    if not is_web:
        return null
    return JavaScriptBridge.eval("window.audioContext")

func resume_web_audio() -> void:
    if not is_web:
        return
    JavaScriptBridge.eval("""
        if (window.audioContext && window.audioContext.state === 'suspended') {
            window.audioContext.resume();
        }
    """)

func get_clipboard_content() -> String:
    if not is_web:
        return ""
    var result = JavaScriptBridge.eval("navigator.clipboard.readText()")
    return str(result) if result else ""
```

## Kasus Penggunaan Nyata
Game web seperti "A Dark Room" dan "Universal Paperclips" menunjukkan potensi game browser yang sukses. Godot memungkinkan game seperti "DungeonRim" dan "City Building Simulator" diekspor ke web untuk distribusi langsung di itch.io tanpa perlu instalasi.
