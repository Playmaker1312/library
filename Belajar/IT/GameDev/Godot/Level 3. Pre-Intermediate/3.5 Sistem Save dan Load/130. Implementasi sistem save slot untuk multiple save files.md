# 130. Implementasi sistem save slot untuk multiple save files

## Definisi Singkat
Sistem save slot memungkinkan pemain menyimpan beberapa progres permainan secara terpisah dalam slot yang berbeda. Setiap slot berisi salinan independen dari state game yang bisa di-load atau di-overwrite secara terpisah.

## Analogi
Bayangkan seperti memiliki beberapa buku catatan terpisah. Buku pertama untuk petualangan karakter warrior, buku kedua untuk mage, dan buku ketiga untuk mencoba build berbeda. Kamu bisa berganti antar buku tanpa kehilangan catatan di buku lain.

## Struktur Node (Scene Tree)
```
MainMenu (Control)
├── VBoxContainer
│   ├── Slot1Button (Button)
│   ├── Slot2Button (Button)
│   ├── Slot3Button (Button)
│   └── BackButton (Button)
└── SaveManager (Node) ← Mengelola semua save slot

SaveSlotData (Resource) ← Resource kustom untuk data slot
├── slot_index: int
├── player_name: String
├── play_time: float
├── level: int
└── last_saved: String
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

const MAX_SLOTS = 3
const SAVE_DIR = "user://saves/"

var slot_data: Array = []

func _ready():
    DirAccess.make_dir_recursive_absolute(SAVE_DIR)
    for i in range(MAX_SLOTS):
        slot_data.append(create_empty_slot(i))

func create_empty_slot(index: int) -> Dictionary:
    return {
        "slot_index": index,
        "player_name": "",
        "level": 0,
        "play_time": 0.0,
        "last_saved": "",
        "is_empty": true
    }

func get_slot_path(slot_index: int) -> String:
    return SAVE_DIR + "slot_%d.json" % slot_index

func save_to_slot(slot_index: int, game_state: Dictionary):
    if slot_index < 0 or slot_index >= MAX_SLOTS:
        print("Invalid slot index!")
        return
    
    var save_info = {
        "slot_index": slot_index,
        "player_name": game_state.get("player_name", "Unknown"),
        "level": game_state.get("level", 1),
        "play_time": game_state.get("play_time", 0.0),
        "last_saved": Time.get_datetime_string_from_system(),
        "is_empty": false,
        "game_data": game_state
    }
    
    var json_string = JSON.stringify(save_info, "\t")
    var file = FileAccess.open(get_slot_path(slot_index), FileAccess.WRITE)
    if file:
        file.store_string(json_string)
        file.close()
        slot_data[slot_index] = save_info
        print("Saved to slot ", slot_index)

func load_from_slot(slot_index: int) -> Dictionary:
    if slot_index < 0 or slot_index >= MAX_SLOTS:
        print("Invalid slot index!")
        return {}
    
    var path = get_slot_path(slot_index)
    if not FileAccess.file_exists(path):
        print("Slot ", slot_index, " is empty!")
        return {}
    
    var file = FileAccess.open(path, FileAccess.READ)
    if file:
        var json_string = file.get_as_text()
        file.close()
        
        var json = JSON.new()
        if json.parse(json_string) == OK:
            slot_data[slot_index] = json.data
            print("Loaded from slot ", slot_index)
            return json.data
        else:
            print("Error parsing slot ", slot_index)
    
    return {}

func delete_slot(slot_index: int):
    if slot_index < 0 or slot_index >= MAX_SLOTS:
        return
    
    var path = get_slot_path(slot_index)
    if FileAccess.file_exists(path):
        DirAccess.remove_absolute(path)
        slot_data[slot_index] = create_empty_slot(slot_index)
        print("Deleted slot ", slot_index)

func get_all_slots() -> Array:
    var slots = []
    for i in range(MAX_SLOTS):
        var path = get_slot_path(i)
        if FileAccess.file_exists(path):
            var file = FileAccess.open(path, FileAccess.READ)
            if file:
                var json = JSON.new()
                if json.parse(file.get_as_text()) == OK:
                    slots.append(json.data)
                file.close()
        else:
            slots.append(create_empty_slot(i))
    return slots

func format_play_time(seconds: float) -> String:
    var hours = int(seconds) / 3600
    var minutes = (int(seconds) % 3600) / 60
    var secs = int(seconds) % 60
    return "%02d:%02d:%02d" % [hours, minutes, secs]

func save_game_example():
    var game_state = {
        "player_name": "Warrior",
        "level": 25,
        "play_time": 5432.5,
        "position": {"x": 100, "y": 200}
    }
    save_to_slot(0, game_state)

func _on_slot_button_pressed(slot_index: int):
    var data = load_from_slot(slot_index)
    if not data.is_empty:
        get_tree().change_scene_to_file("res://main_game.tscn")
```

## Kasus Penggunaan Nyata
Sistem save slot digunakan di hampir semua game modern termasuk The Legend of Zelda, Final Fantasy, dan Dark Souls. Pemain sering ingin mencoba berbagai strategi atau build karakter yang berbeda tanpa kehilangan progres sebelumnya. Fitur ini juga memungkinkan beberapa pemain menggunakan komputer yang sama untuk menyimpan progres masing-masing secara terpisah.
