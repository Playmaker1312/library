# 133. Mengenkripsi file save untuk mencegah cheating dasar

## Definisi Singkat
Enkripsi save file adalah teknik mengubah data save menjadi format yang tidak terbaca tanpa kunci dekripsi yang benar. Tujuannya adalah mencegah pemain memodifikasi data save secara langsung untuk keuntungan curang seperti mengedit jumlah uang atau item.

## Analogi
Enkripsi seperti menulis surat menggunakan bahasa rahasia yang hanya kamu dan penerima yang tahu artinya. Jika orang lain membuka surat tersebut, mereka hanya melihat tulisan acak yang tidak bisa dipahami tanpa kunci rahasia.

## Struktur Node (Scene Tree)
Tidak ada struktur node spesifik, tetapi biasanya diintegrasikan ke dalam SaveManager:

```
Main (Node)
└── SaveManager (Node)
    ├── EncryptionKey: String
    └── Script dengan fungsi encrypt/decrypt
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

const ENCRYPTION_KEY = "MySecretKey123!"
const SAVE_PATH = "user://save_encrypted.dat"

func xor_encrypt(data: String, key: String) -> PackedByteArray:
    var encrypted = PackedByteArray()
    var key_bytes = key.to_utf8_buffer()
    var data_bytes = data.to_utf8_buffer()
    
    for i in range(data_bytes.size()):
        encrypted.append(data_bytes[i] ^ key_bytes[i % key_bytes.size()])
    
    return encrypted

func xor_decrypt(data: PackedByteArray, key: String) -> String:
    var decrypted = PackedByteArray()
    var key_bytes = key.to_utf8_buffer()
    
    for i in range(data.size()):
        decrypted.append(data[i] ^ key_bytes[i % key_bytes.size()])
    
    return decrypted.get_string_from_utf8()

func simple_hash(data: String) -> String:
    var hash_value = 0
    for i in range(data.length()):
        hash_value = (hash_value * 31 + data.unicode_at(i)) % 1000000007
    return str(hash_value)

func save_encrypted(game_state: Dictionary):
    var json_string = JSON.stringify(game_state)
    var checksum = simple_hash(json_string)
    
    var save_data = {
        "data": json_string,
        "checksum": checksum,
        "version": "1.0"
    }
    
    var data_to_encrypt = JSON.stringify(save_data)
    var encrypted_data = xor_encrypt(data_to_encrypt, ENCRYPTION_KEY)
    
    var file = FileAccess.open(SAVE_PATH, FileAccess.WRITE)
    if file:
        file.store_buffer(encrypted_data)
        file.close()
        print("Encrypted save created!")

func load_encrypted() -> Dictionary:
    if not FileAccess.file_exists(SAVE_PATH):
        print("No encrypted save found!")
        return {}
    
    var file = FileAccess.open(SAVE_PATH, FileAccess.READ)
    if file:
        var encrypted_data = file.get_buffer(file.get_length())
        file.close()
        
        var decrypted_string = xor_decrypt(encrypted_data, ENCRYPTION_KEY)
        
        var json = JSON.new()
        if json.parse(decrypted_string) != OK:
            print("Failed to parse decrypted data!")
            return {}
        
        var save_wrapper = json.data
        var stored_checksum = save_wrapper["checksum"]
        var actual_checksum = simple_hash(save_wrapper["data"])
        
        if stored_checksum != actual_checksum:
            print("Save file has been tampered with!")
            return {}
        
        var game_json = JSON.new()
        if game_json.parse(save_wrapper["data"]) == OK:
            print("Encrypted save loaded successfully!")
            return game_json.data
    
    return {}

func save_with_validation(game_state: Dictionary):
    game_state["last_modified"] = Time.get_datetime_string_from_system()
    game_state["session_id"] = str(randi())
    
    var json_string = JSON.stringify(game_state)
    game_state["integrity_hash"] = simple_hash(json_string)
    
    save_encrypted(game_state)

func load_with_validation() -> Dictionary:
    var data = load_encrypted()
    if data.is_empty():
        return {}
    
    var stored_hash = data.get("integrity_hash", "")
    data.erase("integrity_hash")
    
    var current_hash = simple_hash(JSON.stringify(data))
    
    if stored_hash != current_hash:
        print("Data integrity check failed!")
        return {}
    
    return data

func backup_save():
    if FileAccess.file_exists(SAVE_PATH):
        var file = FileAccess.open(SAVE_PATH, FileAccess.READ)
        if file:
            var data = file.get_buffer(file.get_length())
            file.close()
            
            var backup_path = SAVE_PATH + ".backup"
            var backup_file = FileAccess.open(backup_path, FileAccess.WRITE)
            if backup_file:
                backup_file.store_buffer(data)
                backup_file.close()
                print("Backup created!")

func _ready():
    var test_state = {
        "player_name": "Hero",
        "gold": 5000,
        "level": 20
    }
    
    save_with_validation(test_state)
    var loaded = load_with_validation()
    print("Loaded: ", loaded)
    
    backup_save()
```

## Kasus Penggunaan Nyata
Enkripsi save file penting untuk game kompetitif atau game dengan microtransactions di mana cheating dapat merusak pengalaman pemain lain. Game seperti Diablo dan Path of Exile menggunakan enkripsi dan validasi server-side untuk mencegah manipulasi data. Meskipun tidak 100% aman terhadap hacker profesional, enkripsi dasar sudah cukup untuk mencegah sebagian besar pemain curang yang menggunakan text editor biasa.
