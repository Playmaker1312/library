# 110. Menggunakan Custom Data pada tile untuk informasi tambahan seperti damage atau speed modifier

## Definisi Singkat
Custom Data pada TileSet memungkinkan kamu menambahkan properti khusus (integer, string, boolean, float, dll.) ke setiap tile, seperti damage value untuk lava, speed modifier untuk lumpur, atau item drop untuk tile tertentu. Data ini bisa diakses saat runtime dari TileMap.

## Analogi
Bayangkan setiap kotak di papan Monopoli punya label tersembunyi di baliknya — kotak "Pajak" punya label "bayar 200", kotak "Penjara" punya label "skip 1 giliran." Label-label ini tidak terlihat secara visual, tetapi ketika pemain menginjak kotak, program membaca label untuk menentukan apa yang terjadi.

## Struktur Node (Scene Tree)
```
Node2D (Root)
├── TileMap
│   └── TileSet (dengan custom data layers)
├── CharacterBody2D (Player)
│   ├── CollisionShape2D

└── Area2D (tile detector)
    └── CollisionShape2D
```

## Kode GDScript (Godot 4)

### Setup custom data layer pada TileSet
```gdscript
extends Node2D

func _ready() -> void:
	var tileset = TileSet.new()
	tileset.tile_size = Vector2i(16, 16)

	tileset.add_custom_data_layer()
	tileset.set_custom_data_layer_name(0, "damage")
	tileset.set_custom_data_layer_type(0, TYPE_INT)

	tileset.add_custom_data_layer()
	tileset.set_custom_data_layer_name(1, "speed_modifier")
	tileset.set_custom_data_layer_type(1, TYPE_FLOAT)

	tileset.add_custom_data_layer()
	tileset.set_custom_data_layer_name(2, "is_dangerous")
	tileset.set_custom_data_layer_type(2, TYPE_BOOL)

	var source = TileSetAtlasSource.new()
	source.texture = preload("res://assets/tileset.png")
	source.texture_region_size = Vector2i(16, 16)

	source.create_tile(Vector2i(0, 0))
	var normal_data = source.get_tile_data(Vector2i(0, 0))
	normal_data.set_custom_data(0, 0)
	normal_data.set_custom_data(1, 1.0)
	normal_data.set_custom_data(2, false)

	source.create_tile(Vector2i(1, 0))
	var lava_data = source.get_tile_data(Vector2i(1, 0))
	lava_data.set_custom_data(0, 25)
	lava_data.set_custom_data(1, 0.3)
	lava_data.set_custom_data(2, true)

	source.create_tile(Vector2i(2, 0))
	var mud_data = source.get_tile_data(Vector2i(2, 0))
	mud_data.set_custom_data(0, 0)
	mud_data.set_custom_data(1, 0.5)
	mud_data.set_custom_data(2, false)

	tileset.add_source(source, 0)
	$TileMap.tile_set = tileset
```

### Membaca custom data saat runtime
```gdscript
extends Node2D

func get_tile_custom_data(layer_name: String, cell_pos: Vector2i) -> Variant:
	var tilemap = $TileMap
	var source_id = tilemap.get_cell_source_id(0, cell_pos)
	if source_id == -1:
		return null

	var tile_data = tilemap.get_cell_tile_data(0, cell_pos)
	if tile_data == null:
		return null

	var tileset = tilemap.tile_set
	for i in range(tileset.get_custom_data_layers_count()):
		if tileset.get_custom_data_layer_name(i) == layer_name:
			return tile_data.get_custom_data(i)

	return null

func _ready() -> void:
	var damage = get_tile_custom_data("damage", Vector2i(1, 0))
	var speed = get_tile_custom_data("speed_modifier", Vector2i(2, 0))
	var danger = get_tile_custom_data("is_dangerous", Vector2i(1, 0))
	print("Damage: ", damage, " Speed: ", speed, " Dangerous: ", danger)
```

### Player membaca tile di bawahnya
```gdscript
extends CharacterBody2D

@onready var tilemap: TileMap = null

func _physics_process(delta: float) -> void:
	if tilemap == null:
		return

	var cell_pos = tilemap.local_to_map(global_position)
	var damage = get_data_from_tile("damage", cell_pos)
	var speed_mod = get_data_from_tile("speed_modifier", cell_pos)

	if damage and damage > 0:
		take_damage(damage * delta)

	if speed_mod and speed_mod != 1.0:
		velocity *= speed_mod

func get_data_from_tile(layer_name: String, cell: Vector2i) -> Variant:
	var tile_data = tilemap.get_cell_tile_data(0, cell)
	if tile_data == null:
		return null

	var tileset = tilemap.tile_set
	for i in range(tileset.get_custom_data_layers_count()):
		if tileset.get_custom_data_layer_name(i) == layer_name:
			return tile_data.get_custom_data(i)
	return null

func take_damage(amount: float) -> void:
	print("Taking damage: ", amount)
```

## Kasus Penggunaan Nyata
Dalam game seperti Dead Cells, setiap tile di level memiliki custom data seperti damage untuk trap, knockback force untuk explosion tile, dan loot drop untuk destructible tile. Saat pemain menabrak tile, game membaca custom data ini untuk menentukan efek gameplay tanpa hardcode di script pemain.
