# 113. Membuat level editor sederhana menggunakan tool script

## Definisi Singkat
Tool script adalah script GDScript yang berjalan di editor Godot secara langsung (bukan saat runtime), memungkinkan kamu membangun custom editor tools seperti level painter, tile randomizer, atau validator level tanpa harus menjalankan game terlebih dahulu.

## Analogi
Bayangkan kamu sedang merancang peta harta karun. Tool script seperti pena ajaib yang bisa kamu gunakan langsung di atas kertas — setiap kali kamu menggambar tanda X, tool secara otomatis menuliskan koordinat, menghitung jarak ke X berikutnya, dan bahkan memperingatkan jika harta karun terlalu dekat satu sama lain.

## Struktur Node (Scene Tree)
```
@tool
Node2D (Root — LevelEditor)
├── TileMap
├── GridOverlay (Node2D — visualisasi grid)
├── Cursor (Sprite2D — kursor editor)
└── UI (CanvasLayer)
    ├── Toolbar (VBoxContainer)
    └── StatusLabel (Label)
```

## Kode GDScript (Godot 4)

### Tool script untuk level editor
```gdscript
@tool
extends Node2D

@export var grid_size: Vector2i = Vector2i(16, 16)
@export var show_grid: bool = true
@export var level_width: int = 50
@export var level_height: int = 20

var current_brush: int = 0
var is_painting: bool = false
var cursor_pos: Vector2i = Vector2i.ZERO

func _ready() -> void:
	if Engine.is_editor_hint():
		print("Level editor active")

func _draw() -> void:
	if not Engine.is_editor_hint():
		return

	if show_grid:
		draw_grid()

	draw_cursor()

func _process(delta: float) -> void:
	if Engine.is_editor_hint():
		queue_redraw()

func draw_grid() -> void:
	var color = Color(1, 1, 1, 0.1)
	for x in range(level_width + 1):
		var from = Vector2(x * grid_size.x, 0)
		var to = Vector2(x * grid_size.x, level_height * grid_size.y)
		draw_line(from, to, color, 1.0)

	for y in range(level_height + 1):
		var from = Vector2(0, y * grid_size.y)
		var to = Vector2(level_width * grid_size.x, y * grid_size.y)
		draw_line(from, to, color, 1.0)

func draw_cursor() -> void:
	var rect = Rect2(
		Vector2(cursor_pos) * Vector2(grid_size),
		Vector2(grid_size)
	)
	draw_rect(rect, Color.YELLOW, false, 2.0)

func _input(event: InputEvent) -> void:
	if not Engine.is_editor_hint():
		return

	if event is InputEventMouseMotion:
		cursor_pos = get_global_mouse_position() / Vector2(grid_size)
		cursor_pos = cursor_pos.floor()
		queue_redraw()

	if event is InputEventMouseButton:
		if event.button_index == MOUSE_BUTTON_LEFT:
			if event.pressed:
				is_painting = true
				paint_tile(cursor_pos, current_brush)
			else:
				is_painting = false

		if event.button_index == MOUSE_BUTTON_RIGHT:
			paint_tile(cursor_pos, -1)

		if event.button_index == MOUSE_BUTTON_WHEEL_UP:
			current_brush = clampi(current_brush + 1, 0, 5)
			print("Brush: ", current_brush)

		if event.button_index == MOUSE_BUTTON_WHEEL_DOWN:
			current_brush = clampi(current_brush - 1, 0, 5)
			print("Brush: ", current_brush)

	if is_painting and event is InputEventMouseMotion:
		paint_tile(cursor_pos, current_brush)

func paint_tile(pos: Vector2i, tile_id: int) -> void:
	var tilemap = $TileMap
	if tile_id == -1:
		tilemap.set_cell(0, pos, -1)
	else:
		var atlas_pos = Vector2i(tile_id % 8, tile_id / 8)
		tilemap.set_cell(0, pos, 0, atlas_pos)
```

### Validator script untuk memeriksa level
```gdscript
@tool
extends Node

func validate_level(tilemap: TileMap) -> Array[String]:
	var errors: Array[String] = []
	var used_cells = tilemap.get_used_cells(0)

	if used_cells.is_empty():
		errors.append("Level is empty")

	var min_x = INF
	var max_x = -INF
	for cell in used_cells:
		min_x = minf(min_x, cell.x)
		max_x = maxf(max_x, cell.x)

	if max_x - min_x < 10:
		errors.append("Level too short")

	var has_ground = false
	for cell in used_cells:
		if tilemap.get_cell_atlas_coords(0, cell) == Vector2i(0, 0):
			has_ground = true
			break

	if not has_ground:
		errors.append("No ground tiles found")

	var floating_tiles = 0
	for cell in used_cells:
		var below = cell + Vector2i(0, 1)
		if tilemap.get_cell_source_id(0, below) == -1:
			floating_tiles += 1

	if floating_tiles > used_cells.size() * 0.3:
		errors.append("Too many floating tiles")

	return errors

func auto_fix_level(tilemap: TileMap) -> void:
	var errors = validate_level(tilemap)
	if errors.is_empty():
		return

	var used_cells = tilemap.get_used_cells(0)
	for cell in used_cells:
		var below = cell + Vector2i(0, 1)
		if tilemap.get_cell_source_id(0, below) == -1:
			tilemap.set_cell(0, below, 0, Vector2i(0, 0))
```

### Export level ke file
```gdscript
@tool
extends Node

func export_level(tilemap: TileMap, filepath: String) -> void:
	var data = {
		"version": "1.0",
		"tile_size": [tilemap.tile_set.tile_size.x, tilemap.tile_set.tile_size.y],
		"layers": [],
	}

	for layer_idx in range(tilemap.get_layers_count()):
		var layer_data = {
			"name": tilemap.get_layer_name(layer_idx),
			"cells": []
		}

		var used_cells = tilemap.get_used_cells(layer_idx)
		for cell in used_cells:
			var atlas = tilemap.get_cell_atlas_coords(layer_idx, cell)
			layer_data["cells"].append({
				"x": cell.x,
				"y": cell.y,
				"atlas_x": atlas.x,
				"atlas_y": atlas.y
			})

		data["layers"].append(layer_data)

	var file = FileAccess.open(filepath, FileAccess.WRITE)
	if file:
		file.store_string(JSON.stringify(data, "\t"))
		file.close()
		print("Level exported to ", filepath)

func import_level(tilemap: TileMap, filepath: String) -> void:
	var file = FileAccess.open(filepath, FileAccess.READ)
	if not file:
		return

	var json = JSON.new()
	var result = json.parse(file.get_as_text())
	file.close()

	if result != OK:
		return

	var data = json.data
	for layer_idx in range(min(data["layers"].size(), tilemap.get_layers_count())):
		var layer_data = data["layers"][layer_idx]
		for cell in layer_data["cells"]:
			tilemap.set_cell(
				layer_idx,
				Vector2i(cell["x"], cell["y"]),
				0,
				Vector2i(cell["atlas_x"], cell["atlas_y"])
			)
```

## Kasus Penggunaan Nyata
Banyak game indie menggunakan tool script untuk membangun level secara procedural atau memberikan level designer tool kustom. Game seperti Dead Cells memiliki level editor internal yang dibangun dengan pendekatan serupa — memungkinkan desainer membangun, menguji, dan memvalidasi level langsung di editor tanpa harus menjalankan game setiap kali ada perubahan kecil.
