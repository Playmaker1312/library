# 180. Implementasi Wave Function Collapse sederhana untuk map generation

## Definisi Singkat
Wave Function Collapse (WFC) adalah algoritma procedural generation yang menghasilkan pola berulang dengan mempertimbangkan aturan koneksi antar tile, diinspirasi dari konsep mekanika kuantum dalam fisika.

## Analogi
Bayangkan kamu sedang menyelesaikan puzzle jigsaw. Setiap kepingan puzzle hanya bisa cocok dengan kepingan tertentu berdasarkan pola di tepinya. WFC bekerja dengan cara yang sama: memilih tile secara acak, lalu menentukan tile tetangga berdasarkan aturan kompatibilitas.

## Struktur Node (Scene Tree)
```
WFCGenerator (Node2D)
├── TileMap (TileMapLayer)
└── UI (CanvasLayer)
    ├── Button (Generate)
    └── Label (Status)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node2D

class WFCTile:
	var id: int
	var name: String
	var connections: Dictionary

	func _init(p_id: int, p_name: String, p_connections: Dictionary) -> void:
		id = p_id
		name = p_name
		connections = p_connections

class WFCCell:
	var position: Vector2i
	var possible: Array[int]
	var collapsed: bool = false

	func _init(p_position: Vector2i, all_ids: Array[int]) -> void:
		position = p_position
		possible = all_ids.duplicate()

@export var grid_size: Vector2i = Vector2i(20, 15)
@export var tile_size: int = 16

var tiles: Array[WFCTile]
var grid: Array[WFCCell]
var tile_map: TileMapLayer

func _ready() -> void:
	tile_map = $TileMapLayer
	setup_tiles()
	generate()

func setup_tiles() -> void:
	tiles.append(WFCTile.new(0, "grass", {"right": [0, 1], "down": [0, 1]}))
	tiles.append(WFCTile.new(1, "dirt", {"right": [1, 2], "down": [1, 2]}))
	tiles.append(WFCTile.new(2, "water", {"right": [2], "down": [2]}))

func generate() -> void:
	tile_map.clear()
	grid.clear()
	var all_ids: Array[int] = []
	for t in tiles:
		all_ids.append(t.id)
	for y in range(grid_size.y):
		for x in range(grid_size.x):
			grid.append(WFCCell.new(Vector2i(x, y), all_ids))
	while true:
		var cell: WFCCell = get_lowest_entropy_cell()
		if not cell:
			break
		collapse_cell(cell)
		propagate(cell)
	draw_grid()

func get_lowest_entropy_cell() -> WFCCell:
	var min_entropy: int = 999
	var candidates: Array[WFCCell] = []
	for cell in grid:
		if not cell.collapsed and cell.possible.size() < min_entropy:
			min_entropy = cell.possible.size()
			candidates.clear()
			candidates.append(cell)
		elif not cell.collapsed and cell.possible.size() == min_entropy:
			candidates.append(cell)
	if candidates.is_empty():
		return null
	return candidates[randi() % candidates.size()]

func collapse_cell(cell: WFCCell) -> void:
	cell.possible = [cell.possible[randi() % cell.possible.size()]]
	cell.collapsed = true

func propagate(source: WFCCell) -> void:
	var stack: Array[WFCCell] = [source]
	while not stack.is_empty():
		var current: WFCCell = stack.pop_back()
		var neighbors: Array[Vector2i] = [
			current.position + Vector2i(1, 0),
			current.position + Vector2i(-1, 0),
			current.position + Vector2i(0, 1),
			current.position + Vector2i(0, -1)
		]
		var directions: Array[String] = ["right", "left", "down", "up"]
		for i in range(neighbors.size()):
			var neighbor_pos: Vector2i = neighbors[i]
			var neighbor: WFCCell = get_cell_at(neighbor_pos)
			if not neighbor or neighbor.collapsed:
				continue
			var allowed: Array[int] = get_allowed_tiles(current.possible[0], directions[i])
			var new_possible: Array[int] = []
			for p in neighbor.possible:
				if p in allowed:
					new_possible.append(p)
			if new_possible.size() < neighbor.possible.size():
				neighbor.possible = new_possible
				stack.append(neighbor)

func get_allowed_tiles(tile_id: int, direction: String) -> Array[int]:
	var tile: WFCTile = tiles[tile_id]
	if direction in tile.connections:
		return tile.connections[direction]
	return []

func get_cell_at(pos: Vector2i) -> WFCCell:
	for cell in grid:
		if cell.position == pos:
			return cell
	return null

func draw_grid() -> void:
	for cell in grid:
		if cell.collapsed:
			var atlas_coords: Vector2i = Vector2i(cell.possible[0], 0)
			tile_map.set_cell(cell.position, 0, atlas_coords)

func _on_button_pressed() -> void:
	generate()
```

## Kasus Penggunaan Nyata
WFC digunakan dalam game seperti Townscaper untuk menghasilkan kota-kota kecil yang koheren secara visual. Algoritma ini juga digunakan dalam pembuatan level game puzzle dan generasi terrain yang memiliki pola natural seperti hutan, padang rumput, dan danau.
