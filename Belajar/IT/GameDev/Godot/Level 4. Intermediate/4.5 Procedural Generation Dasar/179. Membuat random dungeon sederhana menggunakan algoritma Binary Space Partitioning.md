# 179. Membuat random dungeon sederhana menggunakan algoritma Binary Space Partitioning

## Definisi Singkat
Binary Space Partitioning (BSP) adalah algoritma yang membagi ruang menjadi bagian-bagian lebih kecil secara rekursif menggunakan pemotongan vertikal atau horizontal, kemudian menghubungkan bagian-bagian tersebut dengan koridor.

## Analogi
Seperti memotong kertas menjadi dua bagian secara acak, lalu memotong setiap bagian lagi menjadi dua, dan seterusnya. Setelah selesai, kamu membuat pintu di dinding antar ruangan agar bisa berpindah dari satu ruangan ke ruangan lain.

## Struktur Node (Scene Tree)
```
DungeonGenerator (Node2D)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node2D

class BSPNode:
	var rect: Rect2i
	var left: BSPNode
	var right: BSPNode
	var room: Rect2i

	func _init(p_rect: Rect2i) -> void:
		rect = p_rect

	func split(min_size: int) -> bool:
		if left or right:
			return false
		var split_h: bool = randf() > 0.5
		if rect.size.x > rect.size.y * 1.25:
			split_h = false
		elif rect.size.y > rect.size.x * 1.25:
			split_h = true
		var max_size: int
		if split_h:
			max_size = rect.size.y - min_size
		else:
			max_size = rect.size.x - min_size
		if max_size < min_size:
			return false
		var split_pos: int = randi_range(min_size, max_size)
		if split_h:
			left = BSPNode.new(Rect2i(rect.position, Vector2i(rect.size.x, split_pos)))
			right = BSPNode.new(Rect2i(rect.position + Vector2i(0, split_pos), Vector2i(rect.size.x, rect.size.y - split_pos)))
		else:
			left = BSPNode.new(Rect2i(rect.position, Vector2i(split_pos, rect.size.y)))
			right = BSPNode.new(Rect2i(rect.position + Vector2i(split_pos, 0), Vector2i(rect.size.x - split_pos, rect.size.y)))
		return true

	func generate_room(min_size: int) -> Rect2i:
		if left and right:
			left.generate_room(min_size)
			right.generate_room(min_size)
		elif not left and not right:
			var room_w: int = randi_range(min_size, rect.size.x - 2)
			var room_h: int = randi_range(min_size, rect.size.y - 2)
			var room_x: int = randi_range(rect.position.x + 1, rect.end.x - room_w - 1)
			var room_y: int = randi_range(rect.position.y + 1, rect.end.y - room_h - 1)
			room = Rect2i(Vector2i(room_x, room_y), Vector2i(room_w, room_h))
		return room

	func get_leaves() -> Array[BSPNode]:
		var leaves: Array[BSPNode] = []
		if not left and not right:
			leaves.append(self)
		else:
			if left:
				leaves.append_array(left.get_leaves())
			if right:
				leaves.append_array(right.get_leaves())
		return leaves

@export var map_size: Vector2i = Vector2i(80, 60)
@export var min_room_size: int = 6
@export var split_iterations: int = 5
@export var wall_tile: Vector2i = Vector2i(0, 0)
@export var floor_tile: Vector2i = Vector2i(1, 0)

var root: BSPNode
var tile_map: TileMapLayer

func _ready() -> void:
	tile_map = $TileMapLayer
	generate_dungeon()

func generate_dungeon() -> void:
	tile_map.clear()
	root = BSPNode.new(Rect2i(Vector2i.ZERO, map_size))
	var stack: Array[BSPNode] = [root]
	for i in range(split_iterations):
		var new_stack: Array[BSPNode] = []
		for node in stack:
			if node.split(min_room_size):
				new_stack.append(node.left)
				new_stack.append(node.right)
		stack = new_stack
	root.generate_room(min_room_size)
	var leaves: Array[BSPNode] = root.get_leaves()
	for leaf in leaves:
		if leaf.room.size != Vector2i.ZERO:
			for x in range(leaf.room.position.x, leaf.room.end.x):
				for y in range(leaf.room.position.y, leaf.room.end.y):
					tile_map.set_cell(Vector2i(x, y), 0, floor_tile)
	for leaf in leaves:
		if leaf.room.size != Vector2i.ZERO:
			for x in range(leaf.room.position.x - 1, leaf.room.end.x + 1):
				for y in range(leaf.room.position.y - 1, leaf.room.end.y + 1):
					if tile_map.get_cell_source_id(Vector2i(x, y)) == -1:
						tile_map.set_cell(Vector2i(x, y), 0, wall_tile)
	connect_rooms(root)

func connect_rooms(node: BSPNode) -> void:
	if node.left and node.right:
		connect_rooms(node.left)
		connect_rooms(node.right)
		if node.left.room.size != Vector2i.ZERO and node.right.room.size != Vector2i.ZERO:
			var center_a: Vector2 = Vector2(node.left.room.get_center())
			var center_b: Vector2 = Vector2(node.right.room.get_center())
			carve_corridor(center_a, center_b)

func carve_corridor(from: Vector2, to: Vector2) -> void:
	var current: Vector2i = Vector2i(from)
	var target: Vector2i = Vector2i(to)
	while current != target:
		tile_map.set_cell(current, 0, floor_tile)
		if randf() > 0.5:
			if current.x != target.x:
				current.x += 1 if target.x > current.x else -1
			else:
				current.y += 1 if target.y > current.y else -1
		else:
			if current.y != target.y:
				current.y += 1 if target.y > current.y else -1
			else:
				current.x += 1 if target.x > current.x else -1
```

## Kasus Penggunaan Nyata
BSP banyak digunakan dalam game roguelike seperti Doom dan Rogue untuk membuat dungeon dengan ruangan-ruangan yang terhubung oleh koridor. Algoritma ini memastikan setiap ruangan dapat diakses dan menciptakan struktur peta yang logis untuk eksplorasi pemain.
