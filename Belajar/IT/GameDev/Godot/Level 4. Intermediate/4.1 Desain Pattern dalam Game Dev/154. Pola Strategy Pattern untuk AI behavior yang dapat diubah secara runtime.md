# 154. Pola Strategy Pattern untuk AI behavior yang dapat diubah secara runtime

## Definisi Singkat
Strategy Pattern memungkinkan algoritma atau perilaku tertentu dipilih secara runtime. Pattern ini memisahkan algoritma dari objek yang menggunakannya sehingga algoritma dapat diubah tanpa mengubah kode klien yang menggunakannya.

## Analogi
Bayangkan seorang pegawai kantor. Pegawai ini memiliki beberapa cara untuk menyelesaikan tugas: cara biasa (cara normal), cara cepat (cara efisien), atau cara hati-hati (cara teliti). Atasan dapat meminta pegawai mengubah cara kerja sesuai kebutuhan tanpa harus mengganti pegawai tersebut. Yang berubah hanyalah strategi cara kerja, bukan pegawainya.

## Struktur Node (Scene Tree)
```
AIEnemy (CharacterBody2D)
├── Sprite2D
├── CollisionShape2D
├── NavigationAgent2D
└── AIStrategyManager (Node)
    ├── AggressiveStrategy (Node)
    ├── DefensiveStrategy (Node)
    └── PassiveStrategy (Node)
```

```gdscript
# AggressiveStrategy.gd
extends AIStrategy

@export var chase_speed: float = 200.0
@export var attack_range: float = 60.0
@export var attack_damage: int = 15

var player: CharacterBody2D = null
var is_attacking: bool = false

func enter():
    is_attacking = false

func execute(enemy: CharacterBody2D, delta: float):
    player = _find_player(enemy)
    
    if player == null:
        return
    
    var distance = enemy.global_position.distance_to(player.global_position)
    
    if distance <= attack_range:
        _attack(enemy)
    else:
        _chase(enemy, delta)

func _chase(enemy: CharacterBody2D, delta: float):
    var direction = (player.global_position - enemy.global_position).normalized()
    enemy.velocity.x = direction.x * chase_speed
    
    if direction.x < 0:
        enemy.get_node("Sprite2D").flip_h = true
    else:
        enemy.get_node("Sprite2D").flip_h = false
    
    enemy.get_node("AnimationPlayer").play("run")

func _attack(enemy: CharacterBody2D):
    if is_attacking:
        return
    
    is_attacking = true
    enemy.get_node("AnimationPlayer").play("attack")
    
    if player.has_method("take_damage"):
        player.take_damage(attack_damage)
    
    await enemy.get_node("AnimationPlayer").animation_finished
    is_attacking = false

func _find_player(enemy: CharacterBody2D) -> Node:
    var players = enemy.get_tree().get_nodes_in_group("player")
    if players.size() > 0:
        return players[0]
    return null
```

```gdscript
# DefensiveStrategy.gd
extends AIStrategy

@export var patrol_speed: float = 100.0
@export var retreat_speed: float = 150.0
@export var health_threshold: float = 0.3

var patrol_points: Array[Vector2] = []
var current_point_index: int = 0
var is_retreating: bool = false

func enter():
    is_retreating = false
    _generate_patrol_points()

func execute(enemy: CharacterBody2D, delta: float):
    var health_percent = enemy.get("health") / enemy.get("max_health")
    
    if health_percent <= health_threshold:
        _retreat(enemy, delta)
    else:
        _patrol(enemy, delta)

func _patrol(enemy: CharacterBody2D, delta: float):
    is_retreating = false
    
    if patrol_points.is_empty():
        return
    
    var target = patrol_points[current_point_index]
    var direction = (target - enemy.global_position).normalized()
    
    enemy.velocity.x = direction.x * patrol_speed
    
    if enemy.global_position.distance_to(target) < 10.0:
        current_point_index = (current_point_index + 1) % patrol_points.size()
    
    enemy.get_node("AnimationPlayer").play("walk")

func _retreat(enemy: CharacterBody2D, delta: float):
    is_retreating = true
    var player = _find_player(enemy)
    
    if player == null:
        _patrol(enemy, delta)
        return
    
    var direction = (enemy.global_position - player.global_position).normalized()
    enemy.velocity.x = direction.x * retreat_speed
    
    enemy.get_node("AnimationPlayer").play("run")

func _generate_patrol_points():
    patrol_points.clear()
    patrol_points.append(Vector2(100, 0))
    patrol_points.append(Vector2(-100, 0))

func _find_player(enemy: CharacterBody2D) -> Node:
    var players = enemy.get_tree().get_nodes_in_group("player")
    if players.size() > 0:
        return players[0]
    return null
```

```gdscript
# PassiveStrategy.gd
extends AIStrategy

func execute(enemy: CharacterBody2D, delta: float):
    enemy.velocity.x = 0
    enemy.get_node("AnimationPlayer").play("idle")
```

```gdscript
# AIStrategyManager.gd
extends Node

@export var enemy: CharacterBody2D

var current_strategy: AIStrategy
var strategies: Dictionary = {}

func _ready():
    for child in get_children():
        if child is AIStrategy:
            strategies[child.name.to_lower()] = child
    
    if strategies.has("passive"):
        set_strategy("passive")

func _process(delta):
    if current_strategy:
        current_strategy.execute(enemy, delta)

func set_strategy(strategy_name: String):
    if not strategies.has(strategy_name):
        return
    
    if current_strategy:
        current_strategy.exit()
    
    current_strategy = strategies[strategy_name]
    current_strategy.enter()

func get_current_strategy_name() -> String:
    if current_strategy:
        return current_strategy.name.to_lower()
    return ""
```

```gdscript
# AIEnemy.gd
extends CharacterBody2D

@export var max_health: int = 100
var health: int

@onready var strategy_manager = $AIStrategyManager

func _ready():
    health = max_health

func take_damage(amount: int):
    health -= amount
    
    if health <= 0:
        queue_free()
    
    _check_strategy_change()

func _check_strategy_change():
    var health_percent = float(health) / float(max_health)
    
    if health_percent <= 0.3:
        strategy_manager.set_strategy("defensive")
    elif health_percent <= 0.6:
        strategy_manager.set_strategy("aggressive")
    else:
        strategy_manager.set_strategy("passive")
```

## Kasus Penggunaan Nyata
Strategy Pattern sangat berguna untuk AI yang memiliki perilaku kompleks dan dinamis. Contoh nyata termasuk musuh yang mengubah taktika berdasarkan health, boss yang memiliki phase berbeda, atau NPC yang berubah perilaku berdasarkan waktu atau event. Pattern ini memungkinkan designer menambahkan strategi baru tanpa mengubah kode AI utama.
