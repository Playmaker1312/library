# 155. Event Bus yaitu sistem komunikasi global menggunakan Autoload dan Signal

## Definisi Singkat
Event Bus adalah pola komunikasi global yang memungkinkan node berkomunikasi tanpa dependensi langsung. Dalam Godot, implementasi Event Bus menggunakan Autoload singleton yang menampung signal-signal global dan dapat diakses dari mana saja dalam scene tree.

## Analogi
Bayangkan sebuah papan pengumuman di kantor. Setiap karyawan dapat menempel pengumuman di papan tersebut. Karyawan lain yang melihat pengumuman dapat merespon sesuai kebutuhan. Papan pengumuman tidak peduli siapa yang menempel atau siapa yang membaca. Yang terpenting adalah informasi tersampaikan ke pihak yang membutuhkan.

## Struktur Node (Scene Tree)
```
EventBus (Autoload - Node)
├── Signals
│   ├── player_damaged
│   ├── enemy_killed
│   ├── item_collected
│   ├── level_completed
│   └── game_saved

Main (Node2D)
├── Player (CharacterBody2D)
├── Enemy (CharacterBody2D)
├── UI (CanvasLayer)
└── Items (Node2D)
```

```gdscript
# Player.gd
extends CharacterBody2D

@export var max_health: int = 100
var current_health: int

func _ready():
    current_health = max_health
    EventBus.player_damaged.connect(_on_player_damaged)
    EventBus.player_healed.connect(_on_player_healed)

func take_damage(amount: int):
    current_health = clampi(current_health - amount, 0, max_health)
    EventBus.player_damaged.emit(amount, current_health)
    
    if current_health <= 0:
        EventBus.player_died.emit()

func heal(amount: int):
    current_health = clampi(current_health + amount, 0, max_health)
    EventBus.player_healed.emit(amount, current_health)

func _on_player_damaged(amount: int, new_health: int):
    _play_damage_animation()

func _on_player_healed(amount: int, new_health: int):
    _play_heal_animation()

func _play_damage_animation():
    modulate = Color.RED
    var tween = create_tween()
    tween.tween_property(self, "modulate", Color.WHITE, 0.3)

func _play_heal_animation():
    modulate = Color.GREEN
    var tween = create_tween()
    tween.tween_property(self, "modulate", Color.WHITE, 0.3)
```

# Enemy.gd
extends CharacterBody2D

@export var enemy_name: String = "Enemy"
@export var health: int = 30
@export var score_value: int = 100

func _ready():
    EventBus.enemy_spawned.emit(enemy_name)

func take_damage(amount: int):
    health -= amount
    
    if health <= 0:
        EventBus.enemy_killed.emit(enemy_name, global_position, score_value)
        queue_free()
```

# CollectibleItem.gd
extends Area2D

@export var item_name: String = "Coin"
@export var value: int = 10

func _ready():
    body_entered.connect(_on_body_entered)

func _on_body_entered(body):
    if body.is_in_group("player"):
        EventBus.item_collected.emit(item_name, value)
        _play_collect_animation()

func _play_collect_animation():
    var tween = create_tween()
    tween.tween_property(self, "position:y", position.y - 50, 0.3)
    tween.parallel().tween_property(self, "modulate:a", 0.0, 0.3)
    tween.tween_callback(queue_free)
```

```gdscript
# HUD.gd
extends CanvasLayer

@onready var health_label = $HealthLabel
@onready var score_label = $ScoreLabel
@onready var item_label = $ItemLabel

var score: int = 0
var items_collected: int = 0

func _ready():
    EventBus.player_damaged.connect(_on_player_damaged)
    EventBus.player_healed.connect(_on_player_healed)
    EventBus.enemy_killed.connect(_on_enemy_killed)
    EventBus.item_collected.connect(_on_item_collected)
    EventBus.level_completed.connect(_on_level_completed)
    EventBus.achievement_unlocked.connect(_on_achievement_unlocked)

func _on_player_damaged(amount: int, current_health: int):
    health_label.text = "Health: " + str(current_health)
    _flash_label(health_label, Color.RED)

func _on_player_healed(amount: int, current_health: int):
    health_label.text = "Health: " + str(current_health)
    _flash_label(health_label, Color.GREEN)

func _on_enemy_killed(enemy_name: String, position: Vector2, score_value: int):
    score += score_value
    score_label.text = "Score: " + str(score)
    _show_floating_text(position, "+" + str(score_value))

func _on_item_collected(item_name: String, value: int):
    items_collected += 1
    item_label.text = "Items: " + str(items_collected)
    _flash_label(item_label, Color.YELLOW)

func _on_level_completed(level_name: String):
    _show_notification("Level Completed!")

func _on_achievement_unlocked(achievement_name: String):
    _show_notification("Achievement: " + achievement_name)

func _flash_label(label: Label, color: Color):
    label.modulate = color
    var tween = create_tween()
    tween.tween_property(label, "modulate", Color.WHITE, 0.5)

func _show_floating_text(position: Vector2, text: String):
    var floating_text = Label.new()
    floating_text.text = text
    floating_text.global_position = position
    get_tree().current_scene.add_child(floating_text)
    
    var tween = create_tween()
    tween.tween_property(floating_text, "position:y", position.y - 50, 1.0)
    tween.parallel().tween_property(floating_text, "modulate:a", 0.0, 1.0)
    tween.tween_callback(floating_text.queue_free)

func _show_notification(text: String):
    var notification = Label.new()
    notification.text = text
    notification.horizontal_alignment = HORIZONTAL_ALIGNMENT_CENTER
    notification.vertical_alignment = VERTICAL_ALIGNMENT_CENTER
    notification.set_anchors_preset(Control.PRESET_CENTER)
    add_child(notification)
    
    var tween = create_tween()
    tween.tween_interval(2.0)
    tween.tween_callback(notification.queue_free)
```

```gdscript
# GameManager.gd
extends Node

func _ready():
    EventBus.game_paused.connect(_on_game_paused)
    EventBus.level_started.connect(_on_level_started)
    EventBus.level_completed.connect(_on_level_completed)

func save_game():
    var save_data = {
        "score": 0,
        "health": 100,
        "level": "Level1"
    }
    
    var file = FileAccess.open("user://savegame.json", FileAccess.WRITE)
    file.store_string(JSON.stringify(save_data))
    file.close()
    
    EventBus.game_saved.emit()

func load_game():
    if not FileAccess.file_exists("user://savegame.json"):
        return
    
    var file = FileAccess.open("user://savegame.json", FileAccess.READ)
    var data = JSON.parse_string(file.get_as_text())
    file.close()
    
    EventBus.game_loaded.emit()

func _on_game_paused(is_paused: bool):
    get_tree().paused = is_paused

func _on_level_started(level_name: String):
    print("Level started: " + level_name)

func _on_level_completed(level_name: String):
    print("Level completed: " + level_name)
    save_game()
```

## Kasus Penggunaan Nyata
Event Bus sangat berguna untuk komunikasi global antar sistem yang terpisah. Contoh nyata termasuk notifikasi achievement saat pencapaian tercapai, update UI saat score berubah, save/load game system, dan koordinasi antar scene. Pattern ini mengurangi dependensi antar node dan membuat kode lebih modular serta mudah di-maintain.
