# 150. Implementasi State Machine untuk AI enemy dengan state patrol chase attack

## Definisi Singkat
State Machine untuk AI enemy mengatur perilaku musuh berdasarkan kondisi tertentu seperti jarak ke player, visibility, atau health. Setiap state menentukan aksi yang dilakukan enemy pada kondisi tertentu.

## Analogi
Bayangkan seorang satpam gedung. Satpam ini memiliki state: Patrol (berkeliling), Chase (mengejar penyusup), dan Attack (bertarung). Saat patrol, satpam berjalan mengelilingi area. Jika melihat penyusup, satpam berubah ke state Chase. Jika penyusup sudah dekat, satpam berubah ke state Attack. Setelah penyusup kalah, satpam kembali ke state Patrol.

## Struktur Node (Scene Tree)
```
Enemy (CharacterBody2D)
├── StateMachine (Node)
│   ├── Patrol (Node "Patrol")
│   ├── Chase (Node "Chase")
│   └── Attack (Node "Attack")
├── Sprite2D
├── CollisionShape2D
├── DetectionArea (Area2D)
│   └── CollisionShape2D
├── AnimationPlayer
└── NavigationAgent2D
```

## Kode GDScript (Godot 4)

```gdscript
# Enemy.gd
extends CharacterBody2D

@export var speed: float = 150.0
@export var attack_damage: int = 10
@export var detection_range: float = 200.0
@export var attack_range: float = 50.0

var gravity = ProjectSettings.get_setting("physics/2d/default_gravity")
var player: CharacterBody2D = null

@onready var state_machine = $StateMachine
@onready var sprite = $Sprite2D
@onready var animation_player = $AnimationPlayer
@onready var nav_agent = $NavigationAgent2D
@onready var detection_area = $DetectionArea

func _ready():
    detection_area.body_entered.connect(_on_detection_area_body_entered)
    detection_area.body_exited.connect(_on_detection_area_body_exited)

func _physics_process(delta):
    if not is_on_floor():
        velocity.y += gravity * delta
    
    velocity = move_and_slide(velocity)

func _on_detection_area_body_entered(body):
    if body.is_in_group("player"):
        player = body

func _on_detection_area_body_exited(body):
    if body == player:
        player = null

func is_player_in_range() -> bool:
    if player == null:
        return false
    return global_position.distance_to(player.global_position) <= detection_range

func is_player_in_attack_range() -> bool:
    if player == null:
        return false
    return global_position.distance_to(player.global_position) <= attack_range
```

```gdscript
# Patrol.gd
extends EnemyState

var patrol_points: Array[Vector2] = []
var current_point_index: int = 0

func enter():
    animation_player.play("patrol")
    _generate_patrol_points()

func _generate_patrol_points():
    patrol_points.clear()
    var offset = 100.0
    patrol_points.append(enemy.global_position + Vector2(offset, 0))
    patrol_points.append(enemy.global_position + Vector2(-offset, 0))

func physics_process(delta: float):
    if enemy.is_player_in_range():
        transitioned.emit("Chase")
        return
    
    if patrol_points.is_empty():
        return
    
    var target = patrol_points[current_point_index]
    var direction = (target - enemy.global_position).normalized()
    
    enemy.velocity.x = direction.x * enemy.speed
    
    if enemy.global_position.distance_to(target) < 10.0:
        current_point_index = (current_point_index + 1) % patrol_points.size()
    
    if direction.x < 0:
        enemy.sprite.flip_h = true
    elif direction.x > 0:
        enemy.sprite.flip_h = false
```

```gdscript
# Chase.gd
extends EnemyState

func enter():
    animation_player.play("chase")

func physics_process(delta: float):
    if enemy.player == null:
        transitioned.emit("Patrol")
        return
    
    if enemy.is_player_in_attack_range():
        transitioned.emit("Attack")
        return
    
    if not enemy.is_player_in_range():
        transitioned.emit("Patrol")
        return
    
    var direction = (enemy.player.global_position - enemy.global_position).normalized()
    enemy.velocity.x = direction.x * enemy.speed * 1.5
    
    if direction.x < 0:
        enemy.sprite.flip_h = true
    elif direction.x > 0:
        enemy.sprite.flip_h = false
```

```gdscript
# Attack.gd
extends EnemyState

var attack_cooldown: float = 1.0
var current_cooldown: float = 0.0
var is_attacking: bool = false

func enter():
    animation_player.play("attack")
    is_attacking = false
    current_cooldown = 0.0

func exit():
    is_attacking = false

func physics_process(delta: float):
    if current_cooldown > 0:
        current_cooldown -= delta
    
    if enemy.player == null:
        transitioned.emit("Patrol")
        return
    
    if not enemy.is_player_in_attack_range():
        transitioned.emit("Chase")
        return
    
    if current_cooldown <= 0 and not is_attacking:
        _perform_attack()

func _perform_attack():
    is_attacking = true
    animation_player.play("attack")
    
    if enemy.player.has_method("take_damage"):
        enemy.player.take_damage(enemy.attack_damage)
    
    current_cooldown = attack_cooldown
    is_attacking = false
```

```gdscript
# EnemyStateMachine.gd
extends Node

@export var initial_state: NodePath

var current_state: EnemyState
var states: Dictionary = {}

func _ready():
    for child in get_children():
        if child is EnemyState:
            states[child.name] = child
            child.transitioned.connect(_on_child_transition)
    
    if initial_state:
        current_state = get_node(initial_state)
        current_state.enter()

func _process(delta):
    if current_state:
        current_state.process(delta)

func _physics_process(delta):
    if current_state:
        current_state.physics_process(delta)

func _on_child_transition(new_state_name: String):
    if not states.has(new_state_name):
        return
    
    current_state.exit()
    current_state = states[new_state_name]
    current_state.enter()
```

## Kasus Penggunaan Nyata
AI enemy dengan state patrol, chase, dan attack adalah implementasi dasar untuk musuh dalam game aksi atau RPG. Pattern ini memungkinkan developer membuat perilaku musuh yang lebih kompleks dengan menambahkan state baru seperti Flee (melarikan diri), Search (mencari player), atau Support (mendukung musuh lain). Setiap state dapat diatur secara independen.
