# 197. Implementasi Boss Fight dengan Multiple Phase menggunakan State Machine

## Definisi Singkat
Boss fight adalah pertarungan melawan musuh kuat yang memiliki multiple phase atau tahapan. Setiap phase memiliki perilaku dan serangan berbeda, dan transisi antar phase dikendalikan oleh State Machine.

## Analogi
Bayangkan seperti pertarungan melawan final boss di game "Dark Souls". Boss memiliki fase-fase berbeda di mana ia berubah menjadi lebih kuat atau menggunakan serangan baru. Setiap fase seperti level berbeda yang harus diselesaikan sebelum bisa lanjut ke fase berikutnya.

## Struktur Node (Scene Tree)
```
Boss (CharacterBody2D)
├── Sprite2D
├── CollisionShape2D
├── AnimationPlayer
├── HitBox (Area2D)
│   └── CollisionShape2D
├── HurtBox (Area2D)
│   └── CollisionShape2D
├── BossStateMachine (Node)
│   ├── IdleState (Node)
│   ├── Phase1State (Node)
│   ├── Phase2State (Node)
│   ├── Phase3State (Node)
│   ├── AttackState (Node)
│   └── DefeatedState (Node)
├── AttackTimer (Timer)
└── PhaseTransitionTimer (Timer)
```

## Kode GDScript (Godot 4) - Boss State Machine

```gdscript
extends Node

class_name BossStateMachine

var current_state: BossState
var states: Dictionary = {}
var boss: CharacterBody2D

signal state_changed(old_state: BossState, new_state: BossState)

func initialize(boss_node: CharacterBody2D) -> void:
	boss = boss_node
	for child in get_children():
		if child is BossState:
			states[child.name] = child
			child.state_machine = self
			child.boss = boss
	if states.size() > 0:
		current_state = states.values()[0]
		current_state.enter()

func transition_to(state_name: String) -> void:
	if states.has(state_name) and current_state != states[state_name]:
		var old_state = current_state
		current_state.exit()
		current_state = states[state_name]
		current_state.enter()
		state_changed.emit(old_state, current_state)

func update(delta: float) -> void:
	if current_state:
		current_state.update(delta)

func physics_update(delta: float) -> void:
	if current_state:
		current_state.physics_update(delta)
```

## Kode GDScript (Godot 4) - Phase 1 State

```gdscript
extends BossState

class_name Phase1State

var attack_cooldown: float = 2.0
var current_cooldown: float = 0.0
var health_threshold: float = 0.7

func enter() -> void:
	current_cooldown = attack_cooldown
	boss.get_node("Sprite2D").modulate = Color.WHITE

func update(delta: float) -> void:
	current_cooldown -= delta
	if current_cooldown <= 0:
		_perform_attack()
		current_cooldown = attack_cooldown
	if boss.stats.current_health <= boss.stats.max_health * health_threshold:
		state_machine.transition_to("Phase2State")

func _perform_attack() -> void:
	var player = get_tree().get_first_node_in_group("player")
	if player:
		var direction: Vector2 = (player.global_position - boss.global_position).normalized()
		boss.velocity = direction * 100
		await get_tree().create_timer(0.5).timeout
		boss.velocity = Vector2.ZERO
```

## Kode GDScript (Godot 4) - Phase 2 State

```gdscript
extends BossState

class_name Phase2State

var attack_cooldown: float = 1.5
var current_cooldown: float = 0.0
var health_threshold: float = 0.3

func enter() -> void:
	current_cooldown = attack_cooldown
	boss.get_node("Sprite2D").modulate = Color.YELLOW
	boss.stats.increase_stat("attack", 5)

func update(delta: float) -> void:
	current_cooldown -= delta
	if current_cooldown <= 0:
		_perform_special_attack()
		current_cooldown = attack_cooldown
	if boss.stats.current_health <= boss.stats.max_health * health_threshold:
		state_machine.transition_to("Phase3State")

func _perform_special_attack() -> void:
	var player = get_tree().get_first_node_in_group("player")
	if player:
		for i in range(4):
			var angle: float = i * PI / 2
			var direction: Vector2 = Vector2(cos(angle), sin(angle))
			_spawn_projectile(direction)
			await get_tree().create_timer(0.1).timeout

func _spawn_projectile(direction: Vector2) -> void:
	var projectile = preload("res://scenes/projectiles/enemy_projectile.tscn").instantiate()
	projectile.global_position = boss.global_position
	projectile.direction = direction
	get_tree().current_scene.add_child(projectile)
```

## Kode GDScript (Godot 4) - Phase 3 State

```gdscript
extends BossState

class_name Phase3State

var attack_cooldown: float = 1.0
var current_cooldown: float = 0.0
var rage_mode: bool = false

func enter() -> void:
	current_cooldown = attack_cooldown
	boss.get_node("Sprite2D").modulate = Color.RED
	boss.stats.increase_stat("speed", 50)
	rage_mode = true

func update(delta: float) -> void:
	current_cooldown -= delta
	if current_cooldown <= 0:
		if rage_mode:
			_perform_rage_attack()
		else:
			_perform_normal_attack()
		current_cooldown = attack_cooldown
	if boss.stats.current_health <= 0:
		state_machine.transition_to("DefeatedState")

func _perform_rage_attack() -> void:
	var player = get_tree().get_first_node_in_group("player")
	if player:
		var direction: Vector2 = (player.global_position - boss.global_position).normalized()
		boss.velocity = direction * 200
		await get_tree().create_timer(0.3).timeout
		boss.velocity = Vector2.ZERO
		for i in range(8):
			var angle: float = i * PI / 4
			var projectile_dir: Vector2 = Vector2(cos(angle), sin(angle))
			_spawn_projectile(projectile_dir)

func _perform_normal_attack() -> void:
	_perform_rage_attack()

func _spawn_projectile(direction: Vector2) -> void:
	var projectile = preload("res://scenes/projectiles/enemy_projectile.tscn").instantiate()
	projectile.global_position = boss.global_position
	projectile.direction = direction
	get_tree().current_scene.add_child(projectile)
```

## Kasus Penggunaan Nyata
Sistem boss fight seperti ini digunakan di game "Hollow Knight" di mana setiap boss memiliki multiple phase dengan serangan berbeda. Di "Dark Souls", boss seperti Ornstein and Smough memiliki phase transformasi saat salah satu dikalahkan. Multi-phase boss fight memberikan tantangan yang meningkat dan membuat pertarungan lebih menarik dan berkesan.
