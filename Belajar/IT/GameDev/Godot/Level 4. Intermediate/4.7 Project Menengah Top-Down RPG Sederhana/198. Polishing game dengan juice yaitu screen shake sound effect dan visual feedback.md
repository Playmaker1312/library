# 198. Polishing Game dengan Juice: Screen Shake, Sound Effect, dan Visual Feedback

## Definisi Singkat
"Juice" adalah istilah dalam game development untuk elemen-elemen yang membuat game terasa lebih responsif dan menyenangkan. Ini termasuk screen shake (getaran layar), sound effect (efek suara), dan visual feedback (umpan balik visual) yang memberikan konfirmasi tindakan pemain.

## Analogi
Bayangkan seperti menambahkan bumbu pada masakan. Tanpa bumbu, makanan terasa hambar. Dengan bumbu yang tepat, makanan menjadi lezat dan memuaskan. Juice membuat game terasa "lezat" dan memberikan kepuasan instan kepada pemain.

## Struktur Node (Scene Tree)

## Kode GDScript (Godot 4) - Screen Shake

```gdscript
extends Node

class_name ScreenShakeManager

var camera: Camera2D
var shake_intensity: float = 0.0
var shake_duration: float = 0.0
var shake_timer: float = 0.0
var original_offset: Vector2 = Vector2.ZERO

func initialize(cam: Camera2D) -> void:
	camera = cam
	original_offset = camera.offset

func shake(intensity: float, duration: float) -> void:
	shake_intensity = intensity
	shake_duration = duration
	shake_timer = duration

func _process(delta: float) -> void:
	if shake_timer > 0:
		shake_timer -= delta
		var progress: float = shake_timer / shake_duration
		var current_intensity: float = shake_intensity * progress
		var random_offset: Vector2 = Vector2(
			randf_range(-1, 1) * current_intensity,
			randf_range(-1, 1) * current_intensity
		)
		camera.offset = original_offset + random_offset
	else:
		camera.offset = original_offset
```

## Kode GDScript (Godot 4) - Sound Manager

```gdscript
extends Node

class_name SoundManager

var music_player: AudioStreamPlayer
var sfx_players: Array[AudioStreamPlayer] = []
var sfx_pool_size: int = 8

var music_volume: float = 1.0
var sfx_volume: float = 1.0

func _ready() -> void:
	music_player = AudioStreamPlayer.new()
	music_player.bus = "Music"
	add_child(music_player)
	for i in range(sfx_pool_size):
		var sfx_player = AudioStreamPlayer.new()
		sfx_player.bus = "SFX"
		add_child(sfx_player)
		sfx_players.append(sfx_player)

func play_music(music: AudioStream, fade_time: float = 1.0) -> void:
	if music_player.playing:
		var tween: Tween = create_tween()
		tween.tween_property(music_player, "volume_db", -80, fade_time / 2)
		await tween.finished
	music_player.stream = music
	music_player.play()
	var tween: Tween = create_tween()
	tween.tween_property(music_player, "volume_db", linear_to_db(music_volume), fade_time / 2)

func stop_music(fade_time: float = 1.0) -> void:
	var tween: Tween = create_tween()
	tween.tween_property(music_player, "volume_db", -80, fade_time)
	await tween.finished
	music_player.stop()

func play_sfx(sfx: AudioStream, volume_scale: float = 1.0) -> void:
	var player: AudioStreamPlayer = _get_free_sfx_player()
	if player:
		player.stream = sfx
		player.volume_db = linear_to_db(sfx_volume * volume_scale)
		player.play()

func _get_free_sfx_player() -> AudioStreamPlayer:
	for player in sfx_players:
		if not player.playing:
			return player
	return sfx_players[0]

func set_music_volume(volume: float) -> void:
	music_volume = clampf(volume, 0.0, 1.0)
	music_player.volume_db = linear_to_db(music_volume)

func set_sfx_volume(volume: float) -> void:
	sfx_volume = clampf(volume, 0.0, 1.0)
```

## Kode GDScript (Godot 4) - Visual Feedback

```gdscript
extends Node

class_name VisualFeedbackManager

var damage_number_scene: PackedScene = preload("res://scenes/ui/damage_number.tscn")
var hit_flash_scene: PackedScene = preload("res://scenes/effects/hit_flash.tscn")

func show_damage_number(position: Vector2, damage: int, is_critical: bool = false) -> void:
	var damage_number = damage_number_scene.instantiate()
	damage_number.global_position = position
	damage_number.initialize(damage, is_critical)
	get_tree().current_scene.add_child(damage_number)

func flash_sprite(sprite: Sprite2D, color: Color, duration: float = 0.1) -> void:
	var original_modulate: Color = sprite.modulate
	sprite.modulate = color
	await get_tree().create_timer(duration).timeout
	sprite.modulate = original_modulate

func create_hit_effect(position: Vector2, direction: Vector2) -> void:
	var hit_effect = hit_flash_scene.instantiate()
	hit_effect.global_position = position
	hit_effect.rotation = direction.angle()
	get_tree().current_scene.add_child(hit_effect)
	await get_tree().create_timer(0.3).timeout
	hit_effect.queue_free()

func pulse_scale(node: Node2D, target_scale: Vector2, duration: float = 0.2) -> void:
	var original_scale: Vector2 = node.scale
	var tween: Tween = create_tween()
	tween.tween_property(node, "scale", target_scale, duration / 2)
	await tween.finished
	var tween2: Tween = create_tween()
	tween2.tween_property(node, "scale", original_scale, duration / 2)
	await tween2.finished
```

## Kode GDScript (Godot 4) - Damage Number

```gdscript
extends Node2D

class_name DamageNumber

@onready var label: Label = $Label
@onready var animation_player: AnimationPlayer = $AnimationPlayer

var damage: int = 0
var is_critical: bool = false

func initialize(dmg: int, critical: bool = false) -> void:
	damage = dmg
	is_critical = critical
	label.text = str(damage)
	if is_critical:
		label.modulate = Color.RED
		label.scale = Vector2(1.5, 1.5)
	else:
		label.modulate = Color.WHITE

func _ready() -> void:
	animation_player.play("float_up")
	await animation_player.animation_finished
	queue_free()
```

## Kasus Penggunaan Nyata
Juice sangat penting dalam game seperti "Celeste" di mana setiap lompatan dan dash memiliki feedback visual dan audio yang memuaskan. Di "Hollow Knight", setiap serangan menghasilkan screen shake kecil dan efek partikel yang membuat pertarungan terasa berbobot. Tanpa juice, game akan terasa hambar dan kurang responsif, meskipun mekanikdasarnya sudah baik.
