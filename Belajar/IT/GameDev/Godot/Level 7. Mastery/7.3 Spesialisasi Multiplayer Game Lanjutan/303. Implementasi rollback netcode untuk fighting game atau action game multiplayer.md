# 303. Implementasi Rollback Netcode untuk Fighting Game atau Action Game Multiplayer

## Definisi Singkat
Rollback netcode adalah teknik prediksi di mana klien menjalankan permainan secara optimistis berdasarkan input sendiri, lalu melakukan rollback (pengembalian) ke state sebelumnya dan memperbaikinya saat input lawan diterima. Sistem ini memberikan responsivitas tinggi meskipun latency jaringan tinggi.

## Analogi
Bayangkan dua orang bermain catur melalui telepon. Setiap pemain langsung memindahkan bidaknya di papan sendiri (optimistic execution). Saat pesanan lawan datang, jika ternyata prediksi salah, pemain mengembalikan bidak ke posisi semula dan memperbaiki berdasarkan gerakan lawan yang sebenarnya (rollback and re-simulasi). Pemain harus memiliki catatan langkah sebelumnya (state history) untuk bisa mundur ke posisi yang benar.

## Struktur Node (Scene Tree)
```
Game (Main.tscn)
├── NetworkManager (Node)
│   ├── InputBuffer (Node)
│   └── StateHistory (Node)
├── GameState (Node)
│   ├── Player1 (CharacterBody2D)
│   ├── Player2 (CharacterBody2D)
│   └── FrameCounter (Label)
├── InputHandler (Node)
└── UI (CanvasLayer)
    ├── PingLabel (Label)
    └── RollbackLabel (Label)
```

## Kode GDScript (Godot 4)

### rollback_manager.gd
```gdscript
extends Node

const MAX_ROLLBACK_FRAMES: int = 8
const INPUT_DELAY: int = 2
const MAX_PLAYERS: int = 2

var current_frame: int = 0
var input_buffer: Array = []
var state_history: Array = []
var confirmed_inputs: Dictionary = {}
var local_player_id: int = 0
var remote_player_id: int = 0
var ping: int = 0
var rollback_count: int = 0

func _ready():
    for i in range(MAX_ROLLBACK_FRAMES + INPUT_DELAY + 2):
        input_buffer.append([0, 0])
        state_history.append(null)

func initialize(p1_id: int, p2_id: int):
    local_player_id = p1_id
    remote_player_id = p2_id

func advance_frame():
    var local_input = get_local_input()
    var frame_to_send = current_frame + INPUT_DELAY
    store_input(frame_to_send, local_player_id, local_input)
    send_input_to_remote.rpc_id(remote_player_id, frame_to_send, local_input)
    save_state(current_frame)
    var inputs = get_confirmed_inputs(current_frame)
    simulate_frame(inputs)
    current_frame += 1
    cleanup_old_states()

func get_local_input() -> int:
    var input = 0
    if Input.is_action_pressed("punch"):
        input |= 1
    if Input.is_action_pressed("kick"):
        input |= 2
    if Input.is_action_pressed("block"):
        input |= 4
    if Input.is_action_pressed("move_left"):
        input |= 8
    if Input.is_action_pressed("move_right"):
        input |= 16
    if Input.is_action_pressed("jump"):
        input |= 32
    return input

func store_input(frame: int, player_id: int, input: int):
    if frame < 0 or frame >= input_buffer.size():
        return
    var idx = frame % input_buffer.size()
    if player_id == local_player_id:
        input_buffer[idx][0] = input
    else:
        input_buffer[idx][1] = input

func get_confirmed_inputs(frame: int) -> Array:
    var idx = frame % input_buffer.size()
    return input_buffer[idx]

func save_state(frame: int):
    var idx = frame % state_history.size()
    state_history[idx] = capture_game_state()

func capture_game_state() -> Dictionary:
    var state = {
        "frame": current_frame,
        "player1_pos": $"/root/Main/Game/Player1".position,
        "player1_health": $"/root/Main/Game/Player1".health,
        "player1_state": $"/root/Main/Game/Player1".current_state,
        "player2_pos": $"/root/Main/Game/Player2".position,
        "player2_health": $"/root/Main/Game/Player2".health,
        "player2_state": $"/root/Main/Game/Player2".current_state
    }
    return state

func restore_state(state: Dictionary):
    $"/root/Main/Game/Player1".position = state.player1_pos
    $"/root/Main/Game/Player1".health = state.player1_health
    $"/root/Main/Game/Player1".current_state = state.player1_state
    $"/root/Main/Game/Player2".position = state.player2_pos
    $"/root/Main/Game/Player2".health = state.player2_health
    $"/root/Main/Game/Player2".current_state = state.player2_state

func simulate_frame(inputs: Array):
    $"/root/Main/Game/Player1".process_input(inputs[0])
    $"/root/Main/Game/Player2".process_input(inputs[1])
    $"/root/Main/Game/Player1".update_physics()
    $"/root/Main/Game/Player2".update_physics()
    check_hitboxes()

func check_hitboxes():
    var p1 = $"/root/Main/Game/Player1"
    var p2 = $"/root/Main/Game/Player2"
    if p1.is_attacking and p1.attack_frame == 3:
        if p1.position.distance_to(p2.position) < 80:
            if not p2.is_blocking:
                p2.take_damage(10)
    if p2.is_attacking and p2.attack_frame == 3:
        if p2.position.distance_to(p1.position) < 80:
            if not p1.is_blocking:
                p1.take_damage(10)

@rpc("any_peer", "call_remote", "unreliable")
func send_input_to_remote(frame: int, input: int):
    var sender_id = multiplayer.get_remote_sender_id()
    var player_id = sender_id
    store_input(frame, player_id, input)
    send_input_confirmation.rpc_id(sender_id, frame)

@rpc("any_peer", "call_remote", "unreliable")
func send_input_confirmation(frame: int):
    var sender_id = multiplayer.get_remote_sender_id()
    confirmed_inputs[frame] = true

func on_remote_input_received(frame: int, input: int):
    var last_confirmed = get_last_confirmed_frame()
    if frame <= last_confirmed:
        return
    var current_state = capture_game_state()
    rollback_and_resimulate(frame, input)
    rollback_count += 1

func rollback_and_resimulate(target_frame: int, new_input: int):
    var frames_to_replay = current_frame - target_frame
    var target_idx = target_frame % state_history.size()
    if state_history[target_idx] == null:
        return
    restore_state(state_history[target_idx])
    var saved_frame = current_frame
    current_frame = target_frame
    store_input(target_frame, remote_player_id, new_input)
    for i in range(frames_to_replay + 1):
        var replay_frame = target_frame + i
        var inputs = get_confirmed_inputs(replay_frame)
        simulate_frame(inputs)
        save_state(replay_frame)
        current_frame += 1
    current_frame = saved_frame

func get_last_confirmed_frame() -> int:
    var last = current_frame - MAX_ROLLBACK_FRAMES
    return max(last, 0)

func cleanup_old_states():
    var cutoff = current_frame - MAX_ROLLBACK_FRAMES
    if cutoff < 0:
        return
    var idx = cutoff % state_history.size()
    state_history[idx] = null

func get_display_ping() -> int:
    return ping
```

### player_fighter.gd
```gdscript
extends CharacterBody2D

@export var speed: float = 200.0
@export var jump_force: float = -400.0
@export var max_health: int = 100

var health: int = 100
var current_state: String = "idle"
var is_attacking: bool = false
var is_blocking: bool = false
var attack_frame: int = 0
var attack_duration: int = 8
var gravity: float = 980.0

func process_input(input: int):
    is_blocking = (input & 4) != 0
    if is_blocking:
        current_state = "blocking"
        velocity.x = 0
        return
    var move_dir = 0
    if input & 8:
        move_dir -= 1
    if input & 16:
        move_dir += 1
    velocity.x = move_dir * speed
    if (input & 32) and is_on_floor():
        velocity.y = jump_force
    if (input & 1) and not is_attacking:
        start_attack("punch")
    elif (input & 2) and not is_attacking:
        start_attack("kick")

func start_attack(type: String):
    is_attacking = true
    attack_frame = 0
    current_state = "attacking_%s" % type

func update_physics():
    if not is_on_floor():
        velocity.y += gravity
    move_and_slide()
    if is_attacking:
        attack_frame += 1
        if attack_frame >= attack_duration:
            is_attacking = false
            attack_frame = 0
            current_state = "idle"
    if velocity.x != 0 and not is_attacking:
        current_state = "walking"
    elif is_on_floor() and not is_attacking:
        current_state = "idle"

func take_damage(amount: int):
    health -= amount
    if health < 0:
        health = 0
    current_state = "hitstun"

func get_state_snapshot() -> Dictionary:
    return {
        "health": health,
        "state": current_state,
        "position": position,
        "velocity": velocity,
        "is_attacking": is_attacking,
        "attack_frame": attack_frame
    }

func load_state_snapshot(snapshot: Dictionary):
    health = snapshot.health
    current_state = snapshot.state
    position = snapshot.position
    velocity = snapshot.velocity
    is_attacking = snapshot.is_attacking
    attack_frame = snapshot.attack_frame
```

### network_sync.gd
```gdscript
extends Node

var rollback_manager: Node

func _ready():
    rollback_manager = $"/root/Main/NetworkManager/RollbackManager"

func _physics_process(_delta):
    if multiplayer.is_server() or multiplayer.get_peers().size() > 0:
        rollback_manager.advance_frame()

func get_state_summary() -> String:
    var ping = rollback_manager.get_display_ping()
    var frame = rollback_manager.current_frame
    var rollbacks = rollback_manager.rollback_count
    return "Frame: %d | Ping: %dms | Rollbacks: %d" % [frame, ping, rollbacks]
```

## Kasus Penggunaan Nyata
Rollback netcode digunakan oleh game fighting seperti Street Fighter 6, Guilty Gear Strive, dan Tekken 8 yang memerlukan presisi input frame-by-frame. Teknik ini memungkinkan pemain di Asia bisa bermain melawan pemain di Eropa dengan latency 150ms+ tanpa lag yang terasa, karena setiap klien langsung menampilkan respons input sendiri dan hanya melakukan koreksi kecil saat prediksi meleset.
