# 304. Mengintegrasikan Layanan Backend Seperti Nakama atau PlayFab dengan Godot

## Definisi Singkat
Layanan backend seperti Nakama dan PlayFab menyediakan fitur server-side untuk game multipemain termasuk akun pemain, matchmaking, leaderboard, dan penyimpanan data cloud. Integrasi dilakukan melalui SDK atau REST API yang berkomunikasi dengan Godot client.

## Analogi
Bayangkan sebuah arcade game center. PlayFab/Nakama adalah mesin tiket (server backend) yang mencatat skor, mengelola tiket hadiah, dan mengatur turnamen. Godot adalah mesin game itu sendiri yang mengirim data permainan ke mesin tiket dan menerima informasi seperti leaderboard atau pencapaian pemain.

## Struktur Node (Scene Tree)
```
Game (Main.tscn)
├── BackendManager (Autoload)
│   ├── HttpClient (Node)
│   ├── WebSocket (Node)
│   └── SessionManager (Node)
├── UIManager (CanvasLayer)
│   ├── LoginScreen (Control)
│   ├── LeaderboardUI (Control)
│   └── MatchmakingUI (Control)
└── GameWorld (Node2D)
```

## Kode GDScript (Godot 4)

### backend_manager.gd (Nakama integration)
```gdscript
extends Node

const NAKAMA_HOST: String = "127.0.0.1"
const NAKAMA_PORT: int = 7350
const NAKAMA_SERVER_KEY: String = "defaultkey"
const NAKAMA_HTTP_PORT: int = 7349

var session: NakamaSession
var client: NakamaClient
var socket: NakamaSocket
var authenticated: bool = false
var player_data: Dictionary = {}

signal login_completed(success: bool, message: String)
signal match_found(match_data: Dictionary)
signal leaderboard_updated(entries: Array)

func _ready():
    client = Nakama.create_client(NAKAMA_SERVER_KEY, NAKAMA_HOST, NAKAMA_PORT, "http")
    socket = Nakama.create_socket(client)

func authenticate_device(device_id: String):
    session = await client.authenticate_device_async(device_id)
    if session and session.is_exception():
        login_completed.emit(false, "Autentikasi gagal: %s" % session.exception.message)
        return
    authenticated = true
    await connect_socket()
    await load_player_data()
    login_completed.emit(true, "Login berhasil")

func authenticate_email(email: String, password: String):
    session = await client.authenticate_email_async(email, password)
    if session and session.is_exception():
        login_completed.emit(false, "Login gagal: %s" % session.exception.message)
        return
    authenticated = true
    await connect_socket()
    await load_player_data()
    login_completed.emit(true, "Login berhasil")

func connect_socket():
    var connect_result = await socket.connect_async(session)
    if connect_result.is_exception():
        push_error("Gagal koneksi socket: %s" % connect_result.exception.message)
        return
    socket.received_notification.connect(_on_notification)
    socket.received_match_presence.connect(_on_match_presence)
    socket.received_match_state.connect(_on_match_state)
    print("Socket Nakama terhubung")

func load_player_data():
    var storage_objects = await client.read_storage_objects_async(session, [
        NakamaStorageObjectId.new("player_profile", "data", session.user_id, 1)
    ])
    if storage_objects.objects.size() > 0:
        var json = JSON.new()
        var error = json.parse(storage_objects.objects[0].value)
        if error == OK:
            player_data = json.data

func save_player_data(data: Dictionary):
    var object = NakamaWriteStorageObject.new(
        "player_profile", "data", "private", "player_data", JSON.stringify(data)
    )
    await client.write_storage_objects_async(session, [object])
    player_data = data

func update_leaderboard(leaderboard_id: String, score: int):
    var record = await client.write_leaderboard_record_async(
        session, leaderboard_id, score, 0, ""
    )
    if record.is_exception():
        push_error("Gagal update leaderboard: %s" % record.exception.message)

func get_leaderboard(leaderboard_id: String, limit: int = 10):
    var records = await client.get_leaderboard_records_async(
        session, leaderboard_id, limit, limit, 0
    )
    var entries = []
    for record in records.records:
        entries.append({
            "owner": record.owner,
            "username": record.username,
            "score": record.score,
            "rank": record.rank
        })
    leaderboard_updated.emit(entries)
    return entries

func join_matchmaker(min_count: int, max_count: int, query: String = ""):
    var matchmaker_ticket = await socket.add_matchmaker_async(
        query, min_count, max_count
    )
    if matchmaker_ticket.is_exception():
        push_error("Matchmaker error: %s" % matchmaker_ticket.exception.message)
        return
    print("Masuk matchmaking. Ticket: %s" % matchmaker_ticket.ticket)

func create_match(match_name: String, max_players: int):
    var match_result = await socket.create_match_async(match_name, max_players)
    if match_result.is_exception():
        push_error("Gagal membuat match: %s" % match_result.exception.message)
        return
    return match_result

func join_match(match_id: String):
    var match_result = await socket.join_match_async(match_id)
    if match_result.is_exception():
        push_error("Gagal join match: %s" % match_result.exception.message)
        return
    return match_result

func send_match_state(match_id: int, op_code: int, state: Dictionary):
    await socket.send_match_state_async(match_id, op_code, JSON.stringify(state))

func _on_notification(notification: NakamaNotification):
    print("Notifikasi diterima: %s" % notification.content)

func _on_match_presence(presence: NakamaMatchPresenceEvent):
    print("Match presence: %s" % presence.joins.size() + presence.leaves.size())

func _on_match_state(data: NakamaMatchData):
    var json = JSON.new()
    var error = json.parse(data.state)
    if error == OK:
        var state = json.data
        match_found.emit(state)

func get_player_inventory():
    var inventory = await client.get_inventory_async(session)
    return inventory

func add_virtual_currency(amount: int, currency_type: String):
    var result = await client.add_currency_virtual_async(session, currency_type, amount)
    return result
```

### playfab_integration.gd (PlayFab alternative)
```gdscript
extends Node

const PLAYFAB_TITLE_ID: String = "YOUR_TITLE_ID"
const PLAYFAB_API_URL: String = "https://YOUR_TITLE_ID.playfabapi.com"

var session_ticket: String = ""
var playfab_id: String = ""
var authenticated: bool = false

signal login_completed(success: bool, message: String)
signal leaderboard_updated(entries: Array)

var http_request: HTTPRequest

func _ready():
    http_request = HTTPRequest.new()
    add_child(http_request)
    http_request.request_completed.connect(_on_request_completed)

func login_with_custom_id(custom_id: String):
    var url = PLAYFAB_API_URL + "/Client/LoginWithCustomID"
    var payload = JSON.stringify({
        "TitleId": PLAYFAB_TITLE_ID,
        "CustomId": custom_id,
        "CreateAccount": true
    })
    var headers = ["Content-Type: application/json"]
    var error = http_request.request(url, headers, HTTPClient.METHOD_POST, payload)
    if error != OK:
        login_completed.emit(false, "Request gagal")

func _on_request_completed(result: int, response_code: int, _headers: PackedStringArray, body: PackedByteArray):
    if result != HTTPRequest.RESULT_SUCCESS:
        login_completed.emit(false, "Koneksi gagal")
        return
    var json = JSON.new()
    var error = json.parse(body.get_string_from_utf8())
    if error != OK:
        login_completed.emit(false, "Response tidak valid")
        return
    var response = json.data
    if response_code == 200:
        session_ticket = response.get("SessionTicket", "")
        playfab_id = response.get("PlayFabId", "")
        authenticated = true
        login_completed.emit(true, "Login berhasil")
    else:
        var error_msg = response.get("errorMessage", "Unknown error")
        login_completed.emit(false, error_msg)

func update_player_stat(stat_name: String, value: int):
    var url = PLAYFAB_API_URL + "/Client/UpdatePlayerStatistics"
    var payload = JSON.stringify({
        "SessionTicket": session_ticket,
        "Statistics": [{
            "StatisticName": stat_name,
            "Value": value
        }]
    })
    var headers = ["Content-Type: application/json"]
    http_request.request(url, headers, HTTPClient.METHOD_POST, payload)

func get_leaderboard(stat_name: String, start_position: int = 0, max_results: int = 10):
    var url = PLAYFAB_API_URL + "/Client/GetLeaderboard"
    var payload = JSON.stringify({
        "SessionTicket": session_ticket,
        "StatisticName": stat_name,
        "StartPosition": start_position,
        "MaxResultsCount": max_results
    })
    var headers = ["Content-Type: application/json"]
    var error = http_request.request(url, headers, HTTPClient.METHOD_POST, payload)
    if error != OK:
        return []

func get_player_data(key: String):
    var url = PLAYFAB_API_URL + "/Client/GetUserData"
    var payload = JSON.stringify({
        "SessionTicket": session_ticket,
        "Keys": [key]
    })
    var headers = ["Content-Type: application/json"]
    var error = http_request.request(url, headers, HTTPClient.METHOD_POST, payload)
    if error != OK:
        return null

func set_player_data(key: String, value: String):
    var url = PLAYFAB_API_URL + "/Client/UpdateUserData"
    var payload = JSON.stringify({
        "SessionTicket": session_ticket,
        "Data": {key: value}
    })
    var headers = ["Content-Type: application/json"]
    http_request.request(url, headers, HTTPClient.METHOD_POST, payload)
```

### matchmaking_ui.gd
```gdscript
extends Control

@onready var backend = $"/root/BackendManager"
@onready var status_label = $VBoxContainer/StatusLabel
@onready var find_match_button = $VBoxContainer/FindMatchButton
@onready var leaderboard_list = $VBoxContainer/LeaderboardList

func _ready():
    backend.login_completed.connect(_on_login_completed)
    backend.leaderboard_updated.connect(_on_leaderboard_updated)
    find_match_button.pressed.connect(_on_find_match_pressed)

func _on_find_match_pressed():
    status_label.text = "Mencari match..."
    find_match_button.disabled = true
    backend.join_matchmaker(2, 2)

func _on_login_completed(success: bool, message: String):
    status_label.text = message
    if success:
        await backend.get_leaderboard("wins")

func _on_leaderboard_updated(entries: Array):
    leaderboard_list.clear()
    for entry in entries:
        var text = "#%d %s - %d wins" % [entry.rank, entry.username, entry.score]
        leaderboard_list.add_item(text)
```

## Kasus Penggunaan Nyata
Nakama digunakan oleh banyak game indie dan mobile untuk fitur seperti sistem akun, leaderboard global, dan matchmaking cross-platform. PlayFab dari Microsoft digunakan oleh game AAA seperti Halo Infinite untuk menangani jutaan pemain secara simultan dengan analitik, virtual economy, dan live ops events yang terintegrasi langsung ke dalam game client.
