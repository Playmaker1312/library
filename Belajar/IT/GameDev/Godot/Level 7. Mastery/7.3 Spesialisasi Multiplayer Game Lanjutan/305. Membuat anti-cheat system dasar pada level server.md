# 305. Membuat Anti-Cheat System Dasar pada Level Server

## Definisi Singkat
Anti-cheat system pada level server adalah mekanisme yang memvalidasi semua input dan perilaku pemain di sisi server untuk mendeteksi dan mencegah kecurangan seperti speed hack, wall hack, atau packet manipulation. Server menjadi otoritas tunggal yang memverifikasi legalitas setiap aksi pemain.

## Analogi
Bayangkan sebuah pertandingan olahraga dengan wasit VAR (Video Assistant Referee). Wasit utama (server) tidak hanya mempercayai klaim pemain, tetapi juga merekam dan memverifikasi setiap gerakan menggunakan kamera (logging dan validasi). Jika ada gerakan yang tidak mungkin secara fisik (cheat), wasit dapat memperingatkan atau mengeluarkan pemain tersebut.

## Struktur Node (Scene Tree)
```
Server (Dedicated Server)
├── AntiCheatManager (Node)
│   ├── InputValidator (Node)
│   ├── PositionValidator (Node)
│   ├── RateLimiter (Node)
│   └── CheatLogger (Node)
├── PlayerManager (Node)
│   └── PlayerSessions (Dictionary)
└── GameWorld (Node2D)

Client (Game)
├── PlayerController (CharacterBody2D)
├── InputHandler (Node)
└── NetworkManager (Node)
```

## Kode GDScript (Godot 4)

### anti_cheat_manager.gd
```gdscript
extends Node

const MAX_SPEED: float = 400.0
const MAX_HEALTH: int = 100
const MAX_DAMAGE_PER_SECOND: int = 50
const MAX_POSITION_DELTA: float = 500.0
const RATE_LIMIT_WINDOW: float = 1.0
const MAX_ACTIONS_PER_SECOND: int = 20
const SUSPICIOUS_THRESHOLD: int = 5

var player_violations: Dictionary = {}
var player_action_counts: Dictionary = {}
var player_last_positions: Dictionary = {}
var player_last_action_times: Dictionary = {}
var cheat_logs: Array = []

signal cheat_detected(player_id: int, cheat_type: String, evidence: Dictionary)
signal player_banned(player_id: int, reason: String)

func _ready():
    cheat_detected.connect(_on_cheat_detected)

func validate_movement(player_id: int, current_pos: Vector2, delta: float) -> bool:
    var last_pos = player_last_positions.get(player_id, current_pos)
    var distance = last_pos.distance_to(current_pos)
    var max_allowed = MAX_SPEED * delta * 1.5
    if distance > max_allowed and distance > MAX_POSITION_DELTA:
        report_cheat(player_id, "speed_hack", {
            "distance": distance,
            "max_allowed": max_allowed,
            "last_position": last_pos,
            "current_position": current_pos
        })
        return false
    var time_delta = player_last_action_times.get(player_id + "_movement", 0.0)
    var speed = distance / max(delta, 0.001)
    if speed > MAX_SPEED * 2:
        report_cheat(player_id, "impossible_speed", {
            "speed": speed,
            "max_speed": MAX_SPEED
        })
        return false
    player_last_positions[player_id] = current_pos
    return true

func validate_input_rate(player_id: int) -> bool:
    var current_time = Time.get_ticks_msec() / 1000.0
    if not player_action_counts.has(player_id):
        player_action_counts[player_id] = {"count": 0, "window_start": current_time}
    var data = player_action_counts[player_id]
    if current_time - data.window_start > RATE_LIMIT_WINDOW:
        if data.count > MAX_ACTIONS_PER_SECOND:
            report_cheat(player_id, "input_flooding", {
                "actions_per_second": data.count,
                "limit": MAX_ACTIONS_PER_SECOND
            })
            player_action_counts[player_id] = {"count": 0, "window_start": current_time}
            return false
        data.count = 0
        data.window_start = current_time
    data.count += 1
    return true

func validate_damage(player_id: int, damage: int, target_id: int, attacker_pos: Vector2, target_pos: Vector2) -> bool:
    if damage <= 0 or damage > 100:
        report_cheat(player_id, "invalid_damage", {
            "damage": damage,
            "target": target_id
        })
        return false
    var distance = attacker_pos.distance_to(target_pos)
    if distance > 200.0:
        report_cheat(player_id, "distance_attack", {
            "distance": distance,
            "max_range": 200.0
        })
        return false
    var dps_key = "%d_dps" % player_id
    var current_time = Time.get_ticks_msec() / 1000.0
    if not player_action_counts.has(dps_key):
        player_action_counts[dps_key] = {"total_damage": 0, "window_start": current_time}
    var dps_data = player_action_counts[dps_key]
    if current_time - dps_data.window_start > 1.0:
        if dps_data.total_damage > MAX_DAMAGE_PER_SECOND:
            report_cheat(player_id, "damage_hack", {
                "damage_per_second": dps_data.total_damage,
                "limit": MAX_DAMAGE_PER_SECOND
            })
            dps_data.total_damage = 0
            dps_data.window_start = current_time
            return false
        dps_data.total_damage = 0
        dps_data.window_start = current_time
    dps_data.total_damage += damage
    return true

func validate_item_usage(player_id: int, item_id: String) -> bool:
    var player_data = get_player_inventory(player_id)
    if not player_data.has(item_id) or player_data[item_id] <= 0:
        report_cheat(player_id, "item_duplication", {
            "item_id": item_id,
            "inventory": player_data
        })
        return false
    return true

func validate_position_sanity(player_id: int, position: Vector2, map_bounds: Rect2) -> bool:
    if not map_bounds.has_point(position):
        report_cheat(player_id, "out_of_bounds", {
            "position": position,
            "map_bounds": map_bounds
        })
        return false
    return true

func report_cheat(player_id: int, cheat_type: String, evidence: Dictionary):
    if not player_violations.has(player_id):
        player_violations[player_id] = []
    var violation = {
        "type": cheat_type,
        "evidence": evidence,
        "timestamp": Time.get_datetime_string_from_system(),
        "severity": get_severity(cheat_type)
    }
    player_violations[player_id].append(violation)
    log_cheat(player_id, violation)
    var violation_count = player_violations[player_id].size()
    if violation_count >= SUSPICIOUS_THRESHOLD:
        ban_player(player_id, "Terlalu banyak pelanggaran: %d" % violation_count)
    cheat_detected.emit(player_id, cheat_type, evidence)

func get_severity(cheat_type: String) -> String:
    match cheat_type:
        "speed_hack": return "high"
        "damage_hack": return "critical"
        "item_duplication": return "critical"
        "input_flooding": return "medium"
        "out_of_bounds": return "low"
        "distance_attack": return "high"
        "impossible_speed": return "critical"
        _: return "medium"

func ban_player(player_id: int, reason: String):
    player_banned.emit(player_id, reason)
    print("PEMAIN DIBANNED: %d - Alasan: %s" % [player_id, reason])
    multiplayer.disconnect_peer(player_id)

func log_cheat(player_id: int, violation: Dictionary):
    var log_entry = {
        "player_id": player_id,
        "violation": violation,
        "server_time": Time.get_datetime_string_from_system()
    }
    cheat_logs.append(log_entry)
    print("[CHEAT LOG] Player %d: %s (Severity: %s)" % [
        player_id, violation.type, violation.severity
    ])

func get_player_violations(player_id: int) -> Array:
    return player_violations.get(player_id, [])

func _on_cheat_detected(player_id: int, cheat_type: String, evidence: Dictionary):
    print("Cheat terdeteksi dari pemain %d: %s" % [player_id, cheat_type])

func get_player_inventory(player_id: int) -> Dictionary:
    return {}

func reset_player_data(player_id: int):
    player_violations.erase(player_id)
    player_action_counts.erase(player_id)
    player_last_positions.erase(player_id)
```

### server_validated_player.gd
```gdscript
extends CharacterBody2D

@export var speed: float = 300.0
@export var max_health: int = 100

var health: int = 100
var player_id: int = 0
var anti_cheat: Node

func _ready():
    anti_cheat = $"/root/Main/Server/AntiCheatManager"
    player_id = get_multiplayer_authority()

func _physics_process(delta):
    if multiplayer.is_server():
        server_validate_and_apply(delta)

func server_validate_and_apply(delta: float):
    var input = get_validated_input()
    var movement = input * speed * delta
    var new_position = position + movement
    var map_bounds = Rect2(0, 0, 1920, 1080)
    if anti_cheat.validate_position_sanity(player_id, new_position, map_bounds):
        position = new_position
    move_and_slide()
    sync_position_to_client.rpc(position)

@rpc("any_peer", "call_remote", "unreliable")
func client_send_input(input_vector: Vector2):
    var sender_id = multiplayer.get_remote_sender_id()
    if sender_id != player_id:
        return
    if not anti_cheat.validate_input_rate(player_id):
        return
    var clamped_input = input_vector.clamp(Vector2(-1, -1), Vector2(1, 1))
    apply_server_movement(clamped_input)

func apply_server_movement(input: Vector2):
    var movement = input * speed
    velocity = movement
    move_and_slide()

func get_validated_input() -> Vector2:
    return Vector2.ZERO

func take_damage_validated(damage: int, attacker_id: int, attacker_pos: Vector2):
    if not multiplayer.is_server():
        return
    if anti_cheat.validate_damage(attacker_id, damage, player_id, attacker_pos, position):
        health -= damage
        if health < 0:
            health = 0
        sync_health_to_client.rpc(health)
        if health <= 0:
            die.rpc()

func use_item_validated(item_id: String):
    if not multiplayer.is_server():
        return
    if anti_cheat.validate_item_usage(player_id, item_id):
        apply_item_effect(item_id)

func apply_item_effect(item_id: String):
    print("Item %s digunakan oleh pemain %d" % [item_id, player_id])

@rpc("authority", "call_remote", "unreliable")
func sync_position_to_client(server_position: Vector2):
    if not is_multiplayer_authority():
        var predicted_pos = position
        var error = server_position.distance_to(predicted_pos)
        if error > 10.0:
            position = lerp(position, server_position, 0.3)

@rpc("authority", "call_remote", "reliable")
func sync_health_to_client(new_health: int):
    health = new_health

@rpc("authority", "call_remote", "reliable")
func die():
    queue_free()
```

### rate_limiter.gd
```gdscript
extends Node

const WINDOW_SIZE: float = 1.0
const MAX_REQUESTS: int = 30

var request_counts: Dictionary = {}
var window_start: Dictionary = {}

func check_rate_limit(player_id: int, action_type: String = "default") -> bool:
    var key = "%d_%s" % [player_id, action_type]
    var current_time = Time.get_ticks_msec() / 1000.0
    if not request_counts.has(key):
        request_counts[key] = 0
        window_start[key] = current_time
    if current_time - window_start[key] > WINDOW_SIZE:
        request_counts[key] = 0
        window_start[key] = current_time
    request_counts[key] += 1
    if request_counts[key] > MAX_REQUESTS:
        print("[RATE LIMIT] Player %d melewati batas aksi: %s" % [player_id, action_type])
        return false
    return true

func reset_player_limits(player_id: int):
    var keys_to_remove = []
    for key in request_counts:
        if key.begins_with(str(player_id)):
            keys_to_remove.append(key)
    for key in keys_to_remove:
        request_counts.erase(key)
        window_start.erase(key)

func get_current_rate(player_id: int, action_type: String = "default") -> int:
    var key = "%d_%s" % [player_id, action_type]
    return request_counts.get(key, 0)
```

### packet_validator.gd
```gdscript
extends Node

const PACKET_SIGNATURE: String = "game_v1"
const MAX_PACKET_SIZE: int = 1024

func validate_packet(sender_id: int, data: PackedByteArray) -> bool:
    if data.size() > MAX_PACKET_SIZE:
        print("[PACKET] Ukuran paket terlalu besar dari player %d: %d bytes" % [sender_id, data.size()])
        return false
    var header = data.slice(0, 8).get_string_from_utf8()
    if not header.begins_with(PACKET_SIGNATURE):
        print("[PACKET] Signature tidak valid dari player %d" % sender_id)
        return false
    if not validate_checksum(data):
        print("[PACKET] Checksum tidak valid dari player %d" % sender_id)
        return false
    return true

func sign_packet(data: PackedByteArray) -> PackedByteArray:
    var checksum = calculate_checksum(data)
    var signed = PACKET_SIGNATURE.to_utf8_buffer() + checksum.to_utf8_buffer() + data
    return signed

func calculate_checksum(data: PackedByteArray) -> int:
    var checksum = 0
    for byte in data:
        checksum = (checksum + byte) % 256
    return checksum

func validate_checksum(data: PackedByteArray) -> bool:
    if data.size() < 12:
        return false
    var received_checksum = data.slice(8, 12).get_string_from_utf8().to_int()
    var payload = data.slice(12)
    var calculated = calculate_checksum(payload)
    return received_checksum == calculated

func decrypt_packet(data: PackedByteArray, key: String) -> PackedByteArray:
    var decrypted = PackedByteArray()
    var key_bytes = key.to_utf8_buffer()
    for i in range(data.size()):
        decrypted.append(data[i] ^ key_bytes[i % key_bytes.size()])
    return decrypted

func encrypt_packet(data: PackedByteArray, key: String) -> PackedByteArray:
    return decrypt_packet(data, key)
```

## Kasus Penggunaan Nyata
Sistem anti-cheat server-side seperti ini digunakan oleh game kompetitif seperti Valorant dan CS2 yang menggunakan Vanguard dan VAC untuk mendetekti cheat. Pada game multiplayer Godot, pendekatan ini memastikan bahwa meskipun client dimodifikasi, server tidak akan menerima atau memproses input yang tidak valid, menjaga integritas gameplay untuk semua pemain jujur.
