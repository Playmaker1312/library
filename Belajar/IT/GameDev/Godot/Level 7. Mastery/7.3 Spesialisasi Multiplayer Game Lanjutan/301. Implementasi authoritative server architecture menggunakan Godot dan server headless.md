# 301. Implementasi Authoritative Server Architecture Menggunakan Godot dan Server Headless

## Definisi Singkat
Authoritative server architecture adalah pola desain di mana server memiliki otoritas penuh atas state permainan, menghitung logika gameplay, dan mengirimkan hasil validasi kepada semua klien. Server berjalan tanpa grafis (headless mode) untuk efisiensi resource.

## Analogi
Bayangkan sebuah pertandingan sepak profesional. Wasit (server) adalah satu-satunya yang menentukan gol valid, pelanggaran, dan skor. Pemain (klien) hanya mengirim input gerakan, tetapi tidak bisa mengubah skor sendiri. Wasit tidak perlu tribun penonton (grafis) untuk menjalankan tugasnya.

## Kode GDScript (Godot 4)

### network_manager.gd (autoload)
```gdscript
extends Node

const PORT: int = 7000
const MAX_PLAYERS: int = 8

var players: Dictionary = {}
var player_id: int = 0

func _ready():
    multiplayer.peer_connected.connect(_on_peer_connected)
    multiplayer.peer_disconnected.connect(_on_peer_disconnected)
    multiplayer.connected_to_server.connect(_on_connected_to_server)

func start_server():
    var peer = ENetMultiplayerPeer.new()
    var error = peer.create_server(PORT, MAX_PLAYERS)
    if error != OK:
        push_error("Gagal membuat server: %s" % error)
        return
    multiplayer.multiplayer_peer = peer
    print("Server dimulai di port %d" % PORT)

func join_server(address: String):
    var peer = ENetMultiplayerPeer.new()
    var error = peer.create_client(address, PORT)
    if error != OK:
        push_error("Gagal bergabung ke server: %s" % error)
        return
    multiplayer.multiplayer_peer = peer
    player_id = multiplayer.get_unique_id()
    print("Bergabung ke server. ID: %d" % player_id)

func _on_peer_connected(id: int):
    print("Pemain terhubung: %d" % id)
    if multiplayer.is_server():
        register_player(id)

func _on_peer_disconnected(id: int):
    print("Pemain terputus: %d" % id)
    if multiplayer.is_server():
        unregister_player(id)

func _on_connected_to_server():
    player_id = multiplayer.get_unique_id()
    print("Tersambung ke server. ID: %d" % player_id)

@rpc("authority", "call_remote", "reliable")
func register_player(id: int):
    players[id] = {"id": id, "score": 0}
    spawn_player_on_all_clients.rpc(id)

@rpc("authority", "call_remote", "reliable")
func unregister_player(id: int):
    players.erase(id)
    despawn_player_on_all_clients.rpc(id)

@rpc("authority", "call_remote", "reliable")
func spawn_player_on_all_clients(id: int):
    var player = preload("res://scenes/player.tscn").instantiate()
    player.name = "Player_%d" % id
    player.set_multiplayer_authority(id)
    $"/root/Main/World".add_child(player, true)

@rpc("authority", "call_remote", "reliable")
func despawn_player_on_all_clients(id: int):
    var player_node = $"/root/Main/World/Player_%d" % id
    if player_node:
        player_node.queue_free()
```

### player.gd (server-side movement validation)
```gdscript
extends CharacterBody2D

@export var speed: float = 300.0
@export var max_health: int = 100

var health: int = 100
var input_velocity: Vector2 = Vector2.ZERO

func _physics_process(_delta):
    if is_multiplayer_authority():
        input_velocity = Vector2(
            Input.get_axis("move_left", "move_right"),
            Input.get_axis("move_up", "move_down")
        ).normalized()
        send_input_to_server.rpc(input_velocity)

@rpc("any_peer", "call_remote", "unreliable")
func send_input_to_server(direction: Vector2):
    if multiplayer.is_server():
        var clamped_dir = direction.clamp(Vector2(-1, -1), Vector2(1, 1))
        velocity = clamped_dir * speed
        move_and_slide()
        update_position_on_clients.rpc(position)

@rpc("authority", "call_remote", "unreliable")
func update_position_on_clients(new_position: Vector2):
    if not is_multiplayer_authority():
        position = new_position

@rpc("authority", "call_remote", "reliable")
func take_damage(damage: int, attacker_id: int):
    if not multiplayer.is_server():
        return
    health -= damage
    if health <= 0:
        health = 0
        die.rpc()

@rpc("authority", "call_remote", "reliable")
func die():
    queue_free()
    if is_multiplayer_authority():
        NetworkManager.register_player_score.rpc(get_multiplayer_authority(), 10)
```

### game_manager.gd
```gdscript
extends Node

var game_state: String = "waiting"
var min_players: int = 2

func _process(_delta):
    if multiplayer.is_server():
        check_game_start()

func check_game_start():
    if game_state == "waiting" and NetworkManager.players.size() >= min_players:
        game_state = "playing"
        start_game.rpc()

@rpc("authority", "call_remote", "reliable")
func start_game():
    game_state = "playing"
    print("Permainan dimulai!")

func end_game(winner_id: int):
    if multiplayer.is_server():
        game_state = "waiting"
        end_game_rpc.rpc(winner_id)

@rpc("authority", "call_remote", "reliable")
func end_game_rpc(winner_id: int):
    game_state = "waiting"
    print("Pemain %d menang!" % winner_id)
```

## Kasus Penggunaan Nyata
Server headless digunakan oleh game seperti Counter-Strike 2 dan Dota 2 yang menjalankan dedicated server tanpa GPU. Arsitektur ini mencegahcheat karena klien tidak memiliki otoritas atas game state, dan memungkinkan kompetisi ranked yang adil di mana semua pemain menerima informasi gameplay dari sumber yang sama.
