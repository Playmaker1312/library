# 302. Membuat Matchmaking System Sederhana Menggunakan Relay Server

## Definisi Singkat
Matchmaking system adalah layanan yang secara otomatis mengelompokkan pemain berdasarkan kriteria tertentu seperti skill rating, region, atau preferensi game mode. Relay server berfungsi sebagai perantara yang meneruskan paket data antar klien tanpa memproses logika gameplay.

## Analogi
Bayangkan sebuah restoran buffet. Pelanggan (pemain) datang ke meja penerima tamu (matchmaking server) dan menyatakan preferensi makanan (kriteria matching). Penerima tamu mengarahkan pelanggan ke meja yang sesuai (lobby/game room). Relay server seperti pelayan yang membawa piring dari dapur ke meja tanpa ikut makan.

## Kode GDScript (Godot 4)

### matchmaking_server.gd (server-side)
```gdscript
extends Node

const MATCHMAKING_PORT: int = 8000
const RELAY_PORT: int = 7000
const MAX_PLAYERS_PER_LOBBY: int = 4

var http_server: TCPServer
var player_sessions: Dictionary = {}
var lobbies: Array = []
var waiting_queue: Array = []

func _ready():
    start_matchmaking_server()

func start_matchmaking_server():
    http_server = TCPServer.new()
    var error = http_server.listen(MATCHMAKING_PORT)
    if error != OK:
        push_error("Gagal memulai matchmaking server: %s" % error)
        return
    print("Matchmaking server berjalan di port %d" % MATCHMAKING_PORT)

func _process(_delta):
    if http_server and http_server.is_connection_available():
        var connection = http_server.take_connection()
        handle_http_request(connection)

func handle_http_request(connection: StreamPeerTCP):
    var request_data = ""
    while connection.get_available_bytes() > 0:
        request_data += connection.get_utf8_string(connection.get_available_bytes())
    connection.disconnect_from_host()
    var response = parse_matchmaking_request(request_data)
    send_http_response(connection, response)

func parse_matchmaking_request(request: String) -> Dictionary:
    var lines = request.split("\r\n")
    if lines.size() < 1:
        return {"status": 400, "body": "Bad Request"}
    var parts = lines[0].split(" ")
    if parts.size() < 3:
        return {"status": 400, "body": "Bad Request"}
    var method = parts[0]
    var endpoint = parts[1]
    if method == "POST" and endpoint == "/match":
        return process_match_request(request)
    elif method == "GET" and endpoint.begins_with("/lobby/"):
        return process_lobby_status(endpoint)
    return {"status": 404, "body": "Not Found"}

func process_match_request(request: String) -> Dictionary:
    var body = extract_json_body(request)
    if body.is_empty():
        return {"status": 400, "body": "Invalid JSON"}
    var player_id = body.get("player_id", "")
    var skill_rating = body.get("skill_rating", 1000)
    var region = body.get("region", "asia")
    if player_id.is_empty():
        return {"status": 400, "body": "Missing player_id"}
    var match_result = find_or_create_lobby(player_id, skill_rating, region)
    return {"status": 200, "body": JSON.stringify(match_result)}

func find_or_create_lobby(player_id: String, skill: int, region: String) -> Dictionary:
    for lobby in lobbies:
        if lobby.region == region and lobby.players.size() < MAX_PLAYERS_PER_LOBBY:
            var skill_diff = abs(lobby.avg_skill - skill)
            if skill_diff <= 200:
                lobby.players.append(player_id)
                lobby.avg_skill = calculate_avg_skill(lobby.players)
                if lobby.players.size() >= MAX_PLAYERS_PER_LOBBY:
                    lobby.status = "ready"
                    start_relay_for_lobby(lobby)
                return {"lobby_id": lobby.id, "status": lobby.status}
    var new_lobby = {
        "id": generate_lobby_id(),
        "players": [player_id],
        "avg_skill": skill,
        "region": region,
        "status": "waiting",
        "relay_address": ""
    }
    lobbies.append(new_lobby)
    return {"lobby_id": new_lobby.id, "status": "waiting"}

func calculate_avg_skill(players: Array) -> int:
    var total = 0
    for p in players:
        total += player_sessions.get(p, {}).get("skill", 1000)
    return total / max(players.size(), 1)

func generate_lobby_id() -> String:
    return "lobby_%d_%d" % [Time.get_unix_time_from_system(), randi() % 10000]

func start_relay_for_lobby(lobby: Dictionary):
    lobby.relay_address = "relay://%s:%d" % ["127.0.0.1", RELAY_PORT]
    lobby.status = "starting"
    print("Memulai relay untuk lobby %s" % lobby.id)

func extract_json_body(request: String) -> Dictionary:
    var parts = request.split("\r\n\r\n")
    if parts.size() < 2:
        return {}
    var json = JSON.new()
    var error = json.parse(parts[1])
    if error != OK:
        return {}
    return json.data

func send_http_response(connection: StreamPeerTCP, response: Dictionary):
    var body = response.get("body", "")
    var status = response.get("status", 200)
    var status_text = "OK" if status == 200 else "Error"
    var header = "HTTP/1.1 %d %s\r\nContent-Type: application/json\r\nContent-Length: %d\r\n\r\n" % [status, status_text, body.length()]
    connection.put_data((header + body).to_utf8_buffer())

func process_lobby_status(endpoint: String) -> Dictionary:
    var lobby_id = endpoint.split("/")[2]
    for lobby in lobbies:
        if lobby.id == lobby_id:
            return {"status": 200, "body": JSON.stringify(lobby)}
    return {"status": 404, "body": "Lobby not found"}
```

### matchmaking_client.gd (client-side)
```gdscript
extends Node

signal match_found(lobby_id: String, relay_address: String)
signal match_failed(reason: String)
signal match_status_changed(status: String)

const SERVER_URL: String = "http://127.0.0.1:8000"

var http_request: HTTPRequest
var current_lobby_id: String = ""

func _ready():
    http_request = HTTPRequest.new()
    add_child(http_request)
    http_request.request_completed.connect(_on_request_completed)

func find_match(skill_rating: int, region: String):
    var player_id = str(multiplayer.get_unique_id())
    var payload = JSON.stringify({
        "player_id": player_id,
        "skill_rating": skill_rating,
        "region": region
    })
    var headers = ["Content-Type: application/json"]
    http_request.request(SERVER_URL + "/match", headers, HTTPClient.METHOD_POST, payload)
    match_status_changed.emit("Mencari match...")

func check_lobby_status(lobby_id: String):
    var error = http_request.request(SERVER_URL + "/lobby/" + lobby_id)
    if error != OK:
        match_failed.emit("Gagal memeriksa status lobby")

func _on_request_completed(result: int, response_code: int, _headers: PackedStringArray, body: PackedByteArray):
    if result != HTTPRequest.RESULT_SUCCESS:
        match_failed.emit("Koneksi ke matchmaking server gagal")
        return
    var json = JSON.new()
    var error = json.parse(body.get_string_from_utf8())
    if error != OK:
        match_failed.emit("Response tidak valid dari server")
        return
    var response = json.data
    if response_code == 200:
        var lobby_id = response.get("lobby_id", "")
        var status = response.get("status", "")
        current_lobby_id = lobby_id
        match_status_changed.emit(status)
        if status == "ready":
            var relay_address = response.get("relay_address", "")
            match_found.emit(lobby_id, relay_address)
        elif status == "waiting":
            await get_tree().create_timer(2.0).timeout
            check_lobby_status(lobby_id)
    else:
        match_failed.emit("Matchmaking gagal: %d" % response_code)

func cancel_matchmaking():
    current_lobby_id = ""
    match_status_changed.emit("Dibatalkan")
```

### relay_server.gd (simple relay)
```gdscript
extends Node

const RELAY_PORT: int = 7000

var server_peer: ENetMultiplayerPeer
var relay_rooms: Dictionary = {}

func start_relay():
    server_peer = ENetMultiplayerPeer.new()
    var error = server_peer.create_server(RELAY_PORT, 32)
    if error != OK:
        push_error("Gagal memulai relay server: %s" % error)
        return
    multiplayer.multiplayer_peer = server_peer
    server_peer.peer_connected.connect(_on_peer_connected)
    server_peer.peer_disconnected.connect(_on_peer_disconnected)
    print("Relay server berjalan di port %d" % RELAY_PORT)

func _on_peer_connected(id: int):
    print("Relay: Peer terhubung %d" % id)

func _on_peer_disconnected(id: int):
    print("Relay: Peer terputus %d" % id)
    for room_id in relay_rooms:
        relay_rooms[room_id].erase(id)

@rpc("any_peer", "call_remote", "reliable")
func join_relay_room(room_id: String, player_name: String):
    var sender_id = multiplayer.get_remote_sender_id()
    if not relay_rooms.has(room_id):
        relay_rooms[room_id] = {}
    relay_rooms[room_id][sender_id] = player_name
    broadcast_room_update.rpc(room_id, relay_rooms[room_id])

@rpc("authority", "call_remote", "reliable")
func broadcast_room_update(room_id: String, players: Dictionary):
    for player_id in players:
        var player_name = players[player_id]
        print("Room %s: Pemain %s (ID: %d)" % [room_id, player_name, player_id])

@rpc("any_peer", "call_remote", "unreliable")
func relay_packet(target_id: int, data: PackedByteArray):
    var sender_id = multiplayer.get_remote_sender_id()
    if sender_id in relay_rooms.values():
        send_relay_packet.rpc_id(target_id, sender_id, data)

@rpc("authority", "call_remote", "unreliable")
func send_relay_packet(from_id: int, data: PackedByteArray):
    pass
```

## Kasus Penggunaan Nyata
Sistem matchmaking seperti ini digunakan oleh game mobile seperti Mobile Legends dan PUBG Mobile yang menggabungkan pemain berdasarkan rank dan region. Relay server memungkinkan pemain di belakang NAT atau firewall tetap bisa bermain bersama tanpa perlu port forwarding, karena semua komunikasi diteruskan melalui server perantara.
