# 297. Implementasi Dungeon Generation Lanjutan Menggunakan Algoritma Cellular Automata

## Definisi Singkat
Cellular Automata adalah algoritma generasi dungeon berbasis grid yang mensimulasikan pertumbuhan organik dengan menerapkan aturan tetangga secara iteratif, menghasilkan gua dan ruangan dengan bentuk alami dan tidak terstruktur.

## Analogi
Seperti menonton pertumbuhan kristal di bawah mikroskop. Awalnya titik-titik acak ditempatkan pada grid, lalu setiap iterasi menentukan apakah suatu sel menjadi batu atau ruang berdasarkan berapa banyak tetangganya yang batu. Hasilnya adalah formasi gua yang terlihat natural seperti yang terbentuk oleh erosi alami.

## Kode GDScript (Godot 4)
```gdscript
extends Node2D

class_name CellularAutomataGenerator

const TILE_WALL := Vector2i(0, 0)
const TILE_FLOOR := Vector2i(1, 0)

@export var width: int = 64
@export var height: int = 64
@export var fill_probability: float = 0.45
@export var iterations: int = 5
@export var smoothing_iterations: int = 2
@export var min_room_size: int = 50

var grid: Array[Array] = []
var tile_map: TileMapLayer

func _ready() -> void:
    tile_map = $TileMapLayer

func generate(seed_value: int = -1) -> Dictionary:
    if seed_value >= 0:
        seed(seed_value)
    _initialize_grid()
    _run_cellular_automata()
    _smooth_caves()
    var rooms = _flood_fill_rooms()
    _ensure_connectivity(rooms)
    _render_tilemap()
    return {"rooms": rooms, "grid": grid}

func _initialize_grid() -> void:
    grid.clear()
    for y in range(height):
        var row: Array = []
        for x in range(width):
            if x == 0 or x == width - 1 or y == 0 or y == height - 1:
                row.append(1)
            else:
                row.append(1 if randf() < fill_probability else 0)
        grid.append(row)

func _run_cellular_automata() -> void:
    for _i in range(iterations):
        var new_grid: Array[Array] = []
        for y in range(height):
            var row: Array = []
            for x in range(width):
                var neighbors = _count_wall_neighbors(x, y)
                row.append(1 if neighbors > 4 else 0)
            new_grid.append(row)
        grid = new_grid

func _count_wall_neighbors(x: int, y: int) -> int:
    var count := 0
    for dy in range(-1, 2):
        for dx in range(-1, 2):
            if dx == 0 and dy == 0:
                continue
            var nx = x + dx
            var ny = y + dy
            if nx < 0 or nx >= width or ny < 0 or ny >= height:
                count += 1
            elif grid[ny][nx] == 1:
                count += 1
    return count

func _smooth_caves() -> void:
    for _i in range(smoothing_iterations):
        var new_grid: Array[Array] = []
        for y in range(height):
            var row: Array = []
            for x in range(width):
                var neighbors = _count_wall_neighbors(x, y)
                row.append(1 if neighbors >= 5 else 0)
            new_grid.append(row)
        grid = new_grid

func _flood_fill_rooms() -> Array[Dictionary]:
    var visited: Array[Array] = []
    for y in range(height):
        visited.append(PackedInt32Array().duplicate())
        for _x in range(width):
            visited[y].append(0)
    var rooms: Array[Dictionary] = []
    for y in range(1, height - 1):
        for x in range(1, width - 1):
            if grid[y][x] == 0 and visited[y][x] == 0:
                var room = _flood_fill(x, y, visited)
                if room["tiles"].size() >= min_room_size:
                    rooms.append(room)
    return rooms

func _flood_fill(start_x: int, start_y: int, visited: Array[Array]) -> Dictionary:
    var tiles: Array[Vector2i] = []
    var stack: Array[Vector2i] = [Vector2i(start_x, start_y)]
    var bounds = {"min_x": start_x, "max_x": start_x, "min_y": start_y, "max_y": start_y}
    while stack.size() > 0:
        var pos = stack.pop_back()
        if pos.x < 0 or pos.x >= width or pos.y < 0 or pos.y >= height:
            continue
        if visited[pos.y][pos.x] == 1 or grid[pos.y][pos.x] == 1:
            continue
        visited[pos.y][pos.x] = 1
        tiles.append(pos)
        bounds["min_x"] = min(bounds["min_x"], pos.x)
        bounds["max_x"] = max(bounds["max_x"], pos.x)
        bounds["min_y"] = min(bounds["min_y"], pos.y)
        bounds["max_y"] = max(bounds["max_y"], pos.y)
        stack.append(Vector2i(pos.x + 1, pos.y))
        stack.append(Vector2i(pos.x - 1, pos.y))
        stack.append(Vector2i(pos.x, pos.y + 1))
        stack.append(Vector2i(pos.x, pos.y - 1))
    return {"tiles": tiles, "bounds": bounds, "center": Vector2i(
        (bounds["min_x"] + bounds["max_x"]) / 2,
        (bounds["min_y"] + bounds["max_y"]) / 2
    )}

func _ensure_connectivity(rooms: Array[Dictionary]) -> void:
    if rooms.size() <= 1:
        return
    var connected: Array[Dictionary] = [rooms[0]]
    var unconnected: Array[Dictionary] = rooms.slice(1)
    while unconnected.size() > 0:
        var best_distance := INF
        var best_connected: Dictionary = {}
        var best_unconnected: Dictionary = {}
        var best_index := 0
        for c in connected:
            for i in range(unconnected.size()):
                var u = unconnected[i]
                var dist = c["center"].distance_to(u["center"])
                if dist < best_distance:
                    best_distance = dist
                    best_connected = c
                    best_unconnected = u
                    best_index = i
        _carve_corridor(best_connected["center"], best_unconnected["center"])
        connected.append(best_unconnected)
        unconnected.remove_at(best_index)

func _carve_corridor(from: Vector2i, to: Vector2i) -> void:
    var current = from
    while current != to:
        if current.x != to.x:
            current.x += 1 if to.x > current.x else -1
        elif current.y != to.y:
            current.y += 1 if to.y > current.y else -1
        grid[current.y][current.x] = 0
        if randf() < 0.3:
            var extra = current + Vector2i(randi_range(-1, 1), randi_range(-1, 1))
            if extra.x > 0 and extra.x < width - 1 and extra.y > 0 and extra.y < height - 1:
                grid[extra.y][extra.x] = 0

func _render_tilemap() -> void:
    tile_map.clear()
    for y in range(height):
        for x in range(width):
            var tile_pos = Vector2i(x, y)
            if grid[y][x] == 1:
                tile_map.set_cell(tile_pos, 0, TILE_WALL)
            else:
                tile_map.set_cell(tile_pos, 0, TILE_FLOOR)
```
