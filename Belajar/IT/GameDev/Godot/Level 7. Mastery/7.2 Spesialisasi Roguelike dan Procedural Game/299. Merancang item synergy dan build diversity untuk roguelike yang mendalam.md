# 299. Merancang Item Synergy dan Build Diversity untuk Roguelike yang Mendalam

## Definisi Singkat
Item synergy adalah interaksi antara dua atau lebih item yang menghasilkan efek lebih kuat dari item individu, sedangkan build diversity adalah variasi strategi yang tersedia bagi pemain melalui kombinasi item yang berbeda.

## Analogi
Seperti memasak. Tepung, telur, dan gula masing-masing memiliki rasa sendiri, tetapi ketika digabungkan menjadi cake, hasilnya jauh lebih lezat. Synergy item bekerja sama dengan prinsip yang sama: kombinasi yang tepat menciptakan sesuatu yang lebih dari jumlah bagiannya.

## Struktur Node (Scene Tree)
```
ItemSynergySystem (Node)
├── SynergyDatabase (Node)
├── ItemDatabase (Node)
├── PlayerBuildManager (Node)

│   └── StatModifier (Node)
└── SynergyUI (CanvasLayer)

```

```gdscript
var synergy_database: Array[Dictionary] = [

var active_items: Array[String] = []
var active_synergies: Array[Dictionary] = []

func add_item(item_id: String) -> void:
    if item_id not in active_items:
        active_items.append(item_id)
        _recalculate_synergies()

func remove_item(item_id: String) -> void:
    active_items.erase(item_id)
    _recalculate_synergies()

func _recalculate_synergies() -> void:
    active_synergies.clear()
    for synergy in synergy_database:
        if _check_synergy_requirement(synergy):
            active_synergies.append(synergy)
            emit_signal("synergy_activated", synergy)
    emit_signal("build_changed", get_build_data())

func _check_synergy_requirement(synergy: Dictionary) -> bool:
    var owned_items: Array[Dictionary] = []
    for item_id in active_items:
        if item_id in item_database:
            owned_items.append(item_database[item_id])
    var has_required_items := true
    for required in synergy["required_items"]:
        if required not in active_items:
            has_required_items = false
            break
    if not has_required_items:
        return false
    var all_tags: Array[String] = []
    var elements: Array[String] = []
    var magic_count := 0
    var melee_count := 0
    var poison_count := 0
    var debuff_count := 0
    for item in owned_items:
        for tag in item["tags"]:
            all_tags.append(tag)
        if item["element"] != "none":
            elements.append(item["element"])
        if "magic" in item["tags"]:
            magic_count += 1
        if "melee" in item["tags"]:
            melee_count += 1
        if "poison" in item["tags"]:
            poison_count += 1
        if "debuff" in item["tags"]:
            debuff_count += 1
    var unique_elements = []
    for e in elements:
        if e not in unique_elements:
            unique_elements.append(e)
    if synergy.has("min_elements") and unique_elements.size() < synergy["min_elements"]:
        return false
    if synergy.has("required_tags") and all_tags.size() < synergy["required_tags"]:
        return false
    if synergy.has("min_magic_tags") and magic_count < synergy["min_magic_tags"]:
        return false
    if synergy.has("min_melee_tags") and melee_count < synergy["min_melee_tags"]:
        return false
    if synergy.has("min_poison_tags") and poison_count < synergy["min_poison_tags"]:
        return false
    if synergy.has("min_debuff_tags") and debuff_count < synergy["min_debuff_tags"]:
        return false
    return true

func get_build_data() -> Dictionary:
    var build = {
        "items": active_items.duplicate(),
        "synergies": active_synergies.duplicate(),
        "total_stats": _calculate_total_stats(),
        "tags": _get_all_tags(),
        "elements": _get_unique_elements()
    }
    return build

func _calculate_total_stats() -> Dictionary:
    var total_stats := {
        "attack": 0,
        "defense": 0,
        "magic_power": 0,
        "crit_chance": 0.0,
        "lifesteal": 0.0,
        "cast_speed": 0.0
    }
    for item_id in active_items:
        if item_id in item_database:
            var item = item_database[item_id]
            for stat in item["stats"]:
                total_stats[stat] = total_stats.get(stat, 0) + item["stats"][stat]
    return total_stats

func _get_all_tags() -> Array[String]:
    var tags: Array[String] = []
    for item_id in active_items:
        if item_id in item_database:
            for tag in item_database[item_id]["tags"]:
                if tag not in tags:
                    tags.append(tag)
    return tags

func _get_unique_elements() -> Array[String]:
    var elements: Array[String] = []
    for item_id in active_items:
        if item_id in item_database:
            var elem = item_database[item_id]["element"]
            if elem != "none" and elem not in elements:
                elements.append(elem)
    return elements

func get_synergy_damage_multiplier() -> float:
    var multiplier := 1.0
    for synergy in active_synergies:
        if synergy["effect"]["type"] == "damage_boost":
            multiplier *= synergy["effect"]["value"]
    return multiplier

func get_cooldown_multiplier() -> float:
    var multiplier := 1.0
    for synergy in active_synergies:
        if synergy["effect"]["type"] == "cooldown_reduction":
            multiplier *= (1.0 - synergy["effect"]["value"])
    return multiplier
```

## Kasus Penggunaan Nyata
The Binding of Isaac memiliki lebih dari 700 items dengan ratusan synergy yang bisa dikombinasikan. Setiap item memiliki tag dan efek yang berinteraksi, memungkinkan pemain menemukan build unik seperti "Tech X + Mom's Knife" yang menghasilkan senjata baru. Sistem ini menciptakan replayability yang sangat tinggi karena pemain selalu penasaran dengan kombinasi baru.
