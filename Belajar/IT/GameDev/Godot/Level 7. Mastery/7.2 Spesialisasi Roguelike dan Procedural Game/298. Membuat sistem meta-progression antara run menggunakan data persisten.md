# 298. Membuat Sistem Meta-Progression Antara Run Menggunakan Data Persisten

## Definisi Singkat
Meta-progression adalah sistem yang menyimpan kemajuan pemain secara permanen di luar satu run, seperti unlockable items, karakter baru, atau upgrade yang tetap ada meskipun run berakhir dengan kematian.

## Analogi
Seperti program gym membership. Setiap kunjungan (run) memiliki tantangan sendiri, tetapi tingkat kebugaran (meta-progression) yang meningkat tetap tersimpan. Semakin sering ke gym, semakin banyak latihan baru yang bisa dilakukan, meskipun setiap sesi dimulai dari nol.

## Struktur Node (Scene Tree)
```
MetaProgressionManager (Node)
├── SaveSystem (Node)
│   ├── FileAccessManager (Node)
│   └── EncryptionManager (Node)
├── UnlockTracker (Node)
├── CurrencyManager (Node)
└── AchievementSystem (Node)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

class_name MetaProgressionManager

signal currency_changed(amount: int)
signal item_unlocked(item_id: String)
signal character_unlocked(character_id: String)

const SAVE_PATH := "user://meta_save.dat"

var meta_data: Dictionary = {
    "currency": 0,
    "total_runs": 0,
    "total_kills": 0,
    "total_gold_earned": 0,
    "best_floor": 0,
    "best_time": 0.0,
    "unlocked_items": [],
    "unlocked_characters": ["warrior"],
    "active_character": "warrior",
    "permanent_upgrades": {},
    "achievements": [],
    "run_history": []
}

var unlock_database: Dictionary = {
    "items": {
        "health_potion": {"cost": 0, "condition": "default"},
        "fire_sword": {"cost": 500, "condition": "kill_100_enemies"},
        "shield_ring": {"cost": 300, "condition": "reach_floor_5"},
        "speed_boots": {"cost": 400, "condition": "complete_run_5_times"},
        "vampire_fang": {"cost": 800, "condition": "kill_boss_no_damage"}
    },
    "characters": {
        "warrior": {"cost": 0, "condition": "default"},
        "mage": {"cost": 1000, "condition": "reach_floor_10"},
        "rogue": {"cost": 1200, "condition": "kill_500_enemies"},
        "necromancer": {"cost": 2000, "condition": "complete_run_10_times"}
    },
    "upgrades": {
        "max_health": {"max_level": 10, "cost_per_level": 100, "effect": 10},
        "attack_power": {"max_level": 8, "cost_per_level": 150, "effect": 5},
        "movement_speed": {"max_level": 5, "cost_per_level": 200, "effect": 0.05},
        "crit_chance": {"max_level": 10, "cost_per_level": 120, "effect": 0.02}
    }
}

func _ready() -> void:
    load_meta_data()

func add_currency(amount: int) -> void:
    meta_data["currency"] += amount
    emit_signal("currency_changed", meta_data["currency"])

func spend_currency(amount: int) -> bool:
    if meta_data["currency"] >= amount:
        meta_data["currency"] -= amount
        emit_signal("currency_changed", meta_data["currency"])
        return true
    return false

func unlock_item(item_id: String) -> bool:
    if item_id in meta_data["unlocked_items"]:
        return false
    var unlock_info = unlock_database["items"].get(item_id, {})
    if unlock_info.is_empty():
        return false
    if spend_currency(unlock_info["cost"]):
        meta_data["unlocked_items"].append(item_id)
        emit_signal("item_unlocked", item_id)
        save_meta_data()
        return true
    return false

func unlock_character(character_id: String) -> bool:
    if character_id in meta_data["unlocked_characters"]:
        return false
    var unlock_info = unlock_database["characters"].get(character_id, {})
    if unlock_info.is_empty():
        return false
    if spend_currency(unlock_info["cost"]):
        meta_data["unlocked_characters"].append(character_id)
        emit_signal("character_unlocked", character_id)
        save_meta_data()
        return true
    return false

func purchase_upgrade(upgrade_id: String) -> bool:
    var current_level = meta_data["permanent_upgrades"].get(upgrade_id, 0)
    var upgrade_info = unlock_database["upgrades"].get(upgrade_id, {})
    if upgrade_info.is_empty():
        return false
    if current_level >= upgrade_info["max_level"]:
        return false
    var cost = upgrade_info["cost_per_level"] * (current_level + 1)
    if spend_currency(cost):
        meta_data["permanent_upgrades"][upgrade_id] = current_level + 1
        save_meta_data()
        return true
    return false

func get_upgrade_value(upgrade_id: String) -> float:
    var level = meta_data["permanent_upgrades"].get(upgrade_id, 0)
    var upgrade_info = unlock_database["upgrades"].get(upgrade_id, {})
    if upgrade_info.is_empty():
        return 0.0
    return upgrade_info["effect"] * level

func update_run_stats(run_data: Dictionary) -> void:
    meta_data["total_runs"] += 1
    meta_data["total_kills"] += run_data.get("enemies_killed", 0)
    meta_data["total_gold_earned"] += run_data.get("gold_earned", 0)
    if run_data.get("floor", 0) > meta_data["best_floor"]:
        meta_data["best_floor"] = run_data["floor"]
    if run_data.get("victory", false):
        var time = run_data.get("time_elapsed", 0.0)
        if meta_data["best_time"] == 0.0 or time < meta_data["best_time"]:
            meta_data["best_time"] = time
    meta_data["run_history"].append(run_data)
    if meta_data["run_history"].size() > 50:
        meta_data["run_history"] = meta_data["run_history"].slice(-50)
    check_conditional_unlocks()
    save_meta_data()

func check_conditional_unlocks() -> void:
    for item_id in unlock_database["items"]:
        var condition = unlock_database["items"][item_id]["condition"]
        if _check_condition(condition) and item_id not in meta_data["unlocked_items"]:
            meta_data["unlocked_items"].append(item_id)
            emit_signal("item_unlocked", item_id)

func _check_condition(condition: String) -> bool:
    match condition:
        "kill_100_enemies":
            return meta_data["total_kills"] >= 100
        "kill_500_enemies":
            return meta_data["total_kills"] >= 500
        "reach_floor_5":
            return meta_data["best_floor"] >= 5
        "reach_floor_10":
            return meta_data["best_floor"] >= 10
        "complete_run_5_times":
            return meta_data["total_runs"] >= 5
        "complete_run_10_times":
            return meta_data["total_runs"] >= 10
        _:
            return false

func save_meta_data() -> void:
    var file = FileAccess.open(SAVE_PATH, FileAccess.WRITE)
    if file:
        file.store_var(meta_data)
        file.close()

func load_meta_data() -> void:
    if not FileAccess.file_exists(SAVE_PATH):
        return
    var file = FileAccess.open(SAVE_PATH, FileAccess.READ)
    if file:
        var loaded = file.get_var()
        if loaded is Dictionary:
            for key in meta_data:
                if key in loaded:
                    meta_data[key] = loaded[key]
        file.close()
```

## Kasus Penggunaan Nyata
Dead Cells menggunakan meta-progression untuk unlock weapons dan skills baru. Setiap run mengumpulkan Cells yang bisa dibelanjakan untuk permanent unlocks di Collector, menciptakan motivasi untuk terus bermain meskipun harus memulai dari awal setiap kali mati.
