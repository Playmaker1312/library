# 300. Implementasi Seed-Based Generation untuk Reproducible World

## Definisi Singkat
Seed-based generation menggunakan angka seed sebagai input untuk random number generator, sehingga世界 yang dihasilkan selalu sama untuk seed yang sama. Ini memungkinkan pemain membagikan world yang spesifik dan kompetisi speedrun.

## Analogi
Seperti resep kue. Jika kamu memberikan resep yang persis sama (seed) kepada 10 orang berbeda, mereka akan menghasilkan kue yang hampir identik. Seed dalam game bekerja dengan prinsip yang sama: angka seed menentukan seluruh world yang akan dihasilkan.

## Struktur Node (Scene Tree)
```
SeedWorldGenerator (Node)
├── RNGManager (Node)
│   ├── SeededRNG (RandomNumberGenerator)
│   └── NoiseGenerator (FastNoiseLite)
├── WorldBuilder (Node2D)
├── LootTableManager (Node)
└── SpawnManager (Node)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

class_name SeedWorldGenerator

signal world_generated(seed_value: int)
signal chunk_generated(chunk_pos: Vector2i)

const WORLD_SEED_KEY := "world_seed"

var seeded_rng: RandomNumberGenerator = RandomNumberGenerator.new()
var noise: FastNoiseLite = FastNoiseLite.new()
var world_seed: int = 0

var world_config: Dictionary = {
    "chunk_size": 32,
    "world_size": 100,
    "biome_threshold": 0.3,
    "resource_density": 0.1,
    "enemy_density": 0.05
}

var biome_noise: FastNoiseLite = FastNoiseLite.new()
var resource_noise: FastNoiseLite = FastNoiseLite.new()
var enemy_noise: FastNoiseLite = FastNoiseLite.new()

func _ready() -> void:
    _setup_noise_generators()

func _setup_noise_generators() -> void:
    biome_noise.noise_type = FastNoiseLite.TYPE_SIMPLEX_SMOOTH
    biome_noise.seed = world_seed
    biome_noise.frequency = 0.02
    biome_noise.fractal_octaves = 4

    resource_noise.noise_type = FastNoiseLite.TYPE_VALUE
    resource_noise.seed = world_seed + 1000
    resource_noise.frequency = 0.05

    enemy_noise.noise_type = FastNoiseLite.TYPE_SIMPLEX
    enemy_noise.seed = world_seed + 2000
    enemy_noise.frequency = 0.03

func set_seed(seed_value: int) -> void:
    world_seed = seed_value
    seeded_rng.seed = seed_value
    noise.seed = seed_value
    _update_noise_seeds()

func _update_noise_seeds() -> void:
    biome_noise.seed = world_seed
    resource_noise.seed = world_seed + 1000
    enemy_noise.seed = world_seed + 2000

func generate_world(seed_value: int = -1) -> Dictionary:
    if seed_value >= 0:
        set_seed(seed_value)
    else:
        world_seed = randi()
        set_seed(world_seed)

    var world_data = {
        "seed": world_seed,
        "chunks": {},
        "spawn_points": [],
        "resources": [],
        "enemies": [],
        "pois": []
    }

    var world_size = world_config["world_size"]
    for y in range(-world_size, world_size):
        for x in range(-world_size, world_size):
            var chunk_pos = Vector2i(x, y)
            var chunk_data = generate_chunk(chunk_pos)
            world_data["chunks"][chunk_pos] = chunk_data
            world_data["spawn_points"].append_array(chunk_data["spawn_points"])
            world_data["resources"].append_array(chunk_data["resources"])
            world_data["enemies"].append_array(chunk_data["enemies"])
            if chunk_data.has("poi"):
                world_data["pois"].append(chunk_data["poi"])

    emit_signal("world_generated", world_seed)
    return world_data

func generate_chunk(chunk_pos: Vector2i) -> Dictionary:
    var chunk_data = {
        "position": chunk_pos,
        "biome": _get_biome(chunk_pos),
        "tiles": [],
        "spawn_points": [],
        "resources": [],
        "enemies": []
    }

    var chunk_size = world_config["chunk_size"]
    for y in range(chunk_size):
        var row: Array = []
        for x in range(chunk_size):
            var world_pos = Vector2(
                chunk_pos.x * chunk_size + x,
                chunk_pos.y * chunk_size + y
            )
            var tile_type = _get_tile_type(world_pos, chunk_data["biome"])
            row.append(tile_type)
            _process_tile_features(world_pos, tile_type, chunk_data)
        chunk_data["tiles"].append(row)

    emit_signal("chunk_generated", chunk_pos)
    return chunk_data

func _get_biome(chunk_pos: Vector2i) -> String:
    var biome_value = biome_noise.get_noise_2d(chunk_pos.x, chunk_pos.y)
    if biome_value < -0.5:
        return "desert"
    elif biome_value < 0.0:
        return "plains"
    elif biome_value < 0.5:
        return "forest"
    else:
        return "mountain"

func _get_tile_type(world_pos: Vector2, biome: String) -> int:
    var elevation = noise.get_noise_2d(world_pos.x, world_pos.y)
    var moisture = biome_noise.get_noise_2d(world_pos.x + 500, world_pos.y + 500)

    match biome:
        "desert":
            if elevation > 0.7:
                return 3
            elif elevation > 0.3:
                return 1
            else:
                return 2
        "plains":
            if elevation > 0.6:
                return 3
            elif elevation > 0.2:
                return 0
            else:
                return 4
        "forest":
            if elevation > 0.5:
                return 3
            elif moisture > 0.3:
                return 5
            else:
                return 0
        "mountain":
            if elevation > 0.8:
                return 6
            elif elevation > 0.4:
                return 3
            else:
                return 0
    return 0

func _process_tile_features(world_pos: Vector2, tile_type: int, chunk_data: Dictionary) -> void:
    var resource_roll = resource_noise.get_noise_2d(world_pos.x, world_pos.y)
    if resource_roll > 1.0 - world_config["resource_density"]:
        var resource = _generate_resource(world_pos, chunk_data["biome"])
        chunk_data["resources"].append(resource)

    var enemy_roll = enemy_noise.get_noise_2d(world_pos.x, world_pos.y)
    if enemy_roll > 1.0 - world_config["enemy_density"]:
        var enemy = _generate_enemy(world_pos, chunk_data["biome"])
        chunk_data["enemies"].append(enemy)

func _generate_resource(pos: Vector2, biome: String) -> Dictionary:
    var resource_types = _get_resource_types_for_biome(biome)
    var type_index = seeded_rng.randi_range(0, resource_types.size() - 1)
    return {
        "type": resource_types[type_index],
        "position": pos,
        "amount": seeded_rng.randi_range(1, 5),
        "respawn_time": seeded_rng.randi_range(300, 600)
    }

func _generate_enemy(pos: Vector2, biome: String) -> Dictionary:
    var enemy_types = _get_enemy_types_for_biome(biome)
    var type_index = seeded_rng.randi_range(0, enemy_types.size() - 1)
    var level = seeded_rng.randi_range(1, 10)
    return {
        "type": enemy_types[type_index],
        "position": pos,
        "level": level,
        "loot_table": _get_loot_table(enemy_types[type_index], level)
    }

func _get_resource_types_for_biome(biome: String) -> Array[String]:
    match biome:
        "desert":
            return ["cactus", "sandstone", "mirage_orb"]
        "plains":
            return ["wheat", "flower", "mushroom"]
        "forest":
            return ["wood", "berry", "herb"]
        "mountain":
            return ["iron_ore", "crystal", "dragon_egg"]
    return ["generic_resource"]

func _get_enemy_types_for_biome(biome: String) -> Array[String]:
    match biome:
        "desert":
            return ["scorpion", "sand_worm", "mummy"]
        "plains":
            return ["slime", "goblin", "bandit"]
        "forest":
            return ["spider", "wolf", "treant"]
        "mountain":
            return ["golem", "wyvern", "dragon"]
    return ["generic_enemy"]

func _get_loot_table(enemy_type: String, level: int) -> Dictionary:
    var base_drops = {
        "scorpion": {"gold": [5, 15], "items": ["venom_sac", "stinger"]},
        "goblin": {"gold": [10, 25], "items": ["goblin_dagger", "gold_ring"]},
        "slime": {"gold": [3, 10], "items": ["slime_jelly", "slime_core"]},
        "wolf": {"gold": [8, 20], "items": ["wolf_pelt", "wolf_fang"]},
        "spider": {"gold": [6, 18], "items": ["spider_silk", "venom_sac"]},
        "golem": {"gold": [20, 50], "items": ["golem_heart", "iron_chunk"]},
        "mummy": {"gold": [15, 35], "items": ["ancient_scroll", "embalming_fluid"]},
        "wyvern": {"gold": [30, 60], "items": ["wyvern_scale", "wyvern_wing"]}
    }
    var base = base_drops.get(enemy_type, {"gold": [5, 15], "items": []})
    var gold_min = base["gold"][0] * level
    var gold_max = base["gold"][1] * level
    return {
        "gold": [gold_min, gold_max],
        "items": base["items"],
        "drop_chance": 0.3 + (level * 0.02)
    }

func get_world_info(seed_value: int) -> Dictionary:
    set_seed(seed_value)
    var info = {
        "seed": seed_value,
        "spawn_biome": _get_biome(Vector2i(0, 0)),
        "nearby_biomes": [],
        "difficulty_zones": []
    }
    for angle in range(0, 360, 45):
        var rad = deg_to_rad(angle)
        var check_pos = Vector2i(cos(rad) * 10, sin(rad) * 10)
        info["nearby_biomes"].append(_get_biome(check_pos))
    return info

func share_seed(seed_value: int) -> String:
    return "GODOT-ROGUE-%d" % seed_value

func validate_seed(seed_string: String) -> int:
    if seed_string.begins_with("GODOT-ROGUE-"):
        var seed_num = seed_string.substr(12).to_int()
        if seed_num > 0:
            return seed_num
    return -1
```

## Kasus Penggunaan Nyata
Minecraft menggunakan seed-based generation untuk membuat world yang bisa dibagikan antar pemain. Seed seperti "404" menjadi legendaris karena menghasilkan world dengan gua raksasa di spawn. Dalam roguelike, seed digunakan untuk speedrun competition di mana semua pemain bermain di world yang sama.
