# 294. Memahami psikologi pemain yaitu flow theory intrinsic motivation dan reward schedule

## Definisi Singkat
Psikologi pemain dalam game design mencakup tiga konsep kunci: **Flow Theory** (keseimbangan antara tantangan dan skill), **Intrinsic Motivation** (motivasi dari dalam diri untuk bermain), dan **Reward Schedule** (pola pemberian hadiah yang mempengaruhi perilaku pemain). Ketiganya bekerja sama untuk menciptakan pengalaman bermain yang adictive dan memuaskan.

## Analogi
Bayangkan memancing di danau yang tenang. **Flow Theory** adalah saat Anda menemukan spot yang tepat di mana ikan tidak terlalu banyak (terlalu mudah) dan tidak terlalu sedikit (terlalu sulit) - ada tantangan tapi Anda tetap bisa menangkap ikan. **Intrinsic Motivation** adalah kenikmatan Anda menikmati alam, ketenangan, dan proses menunggu, bukan hanya hasil tangkapan. **Reward Schedule** adalah pola Anda mendapat ikan - kadang cepat, kadang lambat, tapi cukup sering untuk membuat Anda tetap duduk dan menunggu dengan sabar.


## Kode GDScript (Godot 4)
```gdscript
extends Node

signal flow_state_changed(new_state: String)
signal reward_granted(reward_type: String, amount: int)
signal motivation_boost(source: String, intensity: float)

enum FlowState { BORED, ANXIOUS, FLOW, CONTROL, AROUSAL }

var challenge_level: float = 0.5
var skill_level: float = 0.5
var current_flow_state: FlowState = FlowState.FLOW
var reward_history: Array[Dictionary] = []
var intrinsic_motivation_score: float = 0.7

@export var reward_pools: Dictionary = {
    "common": {"chance": 0.7, "rewards": ["coin_small", "health_potion"]},
    "uncommon": {"chance": 0.2, "rewards": ["coin_medium", "mana_potion"]},
    "rare": {"chance": 0.08, "rewards": ["coin_large", "rare_item"]},
    "legendary": {"chance": 0.02, "rewards": ["legendary_weapon", "special_ability"]}
}

func update_flow_state(new_challenge: float, new_skill: float) -> void:
    challenge_level = clamp(new_challenge, 0.0, 1.0)
    skill_level = clamp(new_skill, 0.0, 1.0)
    var balance: float = challenge_level - skill_level
    var old_state: FlowState = current_flow_state
    if balance < -0.2:
        current_flow_state = FlowState.BORED
    elif balance > 0.2:
        current_flow_state = FlowState.ANXIOUS
    elif abs(balance) < 0.1:
        current_flow_state = FlowState.FLOW
    elif balance < 0:
        current_flow_state = FlowState.CONTROL
    else:
        current_flow_state = FlowState.AROUSAL
    if old_state != current_flow_state:
        flow_state_changed.emit(_flow_state_to_string(current_flow_state))

func grant_reward(event_type: String) -> Dictionary:
    var reward: Dictionary = _calculate_reward()
    reward_history.append({
        "type": reward["type"],
        "item": reward["item"],
        "event": event_type,
        "timestamp": Time.get_unix_time_from_system()
    })
    reward_granted.emit(reward["item"], 1)
    _update_motivation(reward["type"])
    return reward

func _calculate_reward() -> Dictionary:
    var roll: float = randf()
    var cumulative: float = 0.0
    for rarity in reward_pools:
        cumulative += reward_pools[rarity]["chance"]
        if roll <= cumulative:
            var pool = reward_pools[rarity]
            var item = pool["rewards"][randi() % pool["rewards"].size()]
            return {"type": rarity, "item": item}
    return {"type": "common", "item": "coin_small"}

func _update_motivation(reward_rarity: String) -> void:
    var motivation_boost: float = 0.0
    match reward_rarity:
        "common":
            motivation_boost = 0.05
        "uncommon":
            motivation_boost = 0.15
        "rare":
            motivation_boost = 0.3
        "legendary":
            motivation_boost = 0.5
    intrinsic_motivation_score = min(intrinsic_motivation_score + motivation_boost, 1.0)
    motivation_boost.emit("reward", motivation_boost)

func get_player_engagement() -> Dictionary:
    return {
        "flow_state": _flow_state_to_string(current_flow_state),
        "motivation_level": intrinsic_motivation_score,
        "challenge_balance": challenge_level - skill_level,
        "total_rewards": reward_history.size()
    }

func _flow_state_to_string(state: FlowState) -> String:
    match state:
        FlowState.BORED:
            return "bored"
        FlowState.ANXIOUS:
            return "anxious"
        FlowState.FLOW:
            return "flow"
        FlowState.CONTROL:
            return "control"
        FlowState.AROUSAL:
            return "arousal"
    return "unknown"
```

## Kasus Penggunaan Nyata
**World of Warcraft** mengimplementasikan ketiga konsep psikologi ini dengan sempurna. **Flow Theory** dijaga melalui leveling sistem yang menyesuaikan konten dengan level pemain. **Intrinsic Motivation** didorong melalui cerita yang mendalam dan kemampuan untuk mengeksplorasi dunia terbuka. **Reward Schedule** menggunakan variable ratio (random loot drops) yang membuat pemain terus bermain karena tidak tahu kapan akan mendapatkan item langka berikutnya.
