# 295. Merancang accessibility features yaitu color blind mode subtitle dan control remapping

## Definisi Singkat
Accessibility features dalam game design adalah fitur yang dirancang untuk memastikan semua pemain, termasuk mereka dengan keterbatasan fisik atau kondisi tertentu, dapat menikmati game secara penuh. Contoh utama meliputi color blind mode (mode buta warna), subtitle (teks dialog), dan control remapping (pengaturan ulang kontrol).

## Analogi
Bayangkan sebuah gedung perkantoran. **Color blind mode** seperti menambahkan label teks pada lift selain warna tombol, sehingga orang yang tidak bisa membedakan warna tetap tahu mana tombol naik dan turun. **Subtitle** seperti penerjemah bahasa isyarat yang berdiri di samping pembicara, membantu orang yang tidak bisa mendengar untuk memahami presentasi. **Control remapping** seperti menyediakan berbagai jenis pintu - ada yang pakai gagang putar, ada yang pakai panel sentuh, ada yang pakai sensor - sehingga semua orang bisa masuk meski dengan kemampuan fisik yang berbeda.

## Struktur Node (Scene Tree)
```
AccessibilityManager (Node)
├── ColorBlindMode (Node)
│   ├── ColorFilter (ColorRect)
│   ├── FilterType (enum)
│   └── Intensity (float)
├── SubtitleSystem (CanvasLayer)
│   ├── SubtitlePanel (Control)
│   ├── SubtitleLabel (RichTextLabel)
│   └── SpeakerName (Label)
├── ControlRemapping (Node)
│   ├── InputMapping (Dictionary)
│   ├── GamepadSupport (bool)
│   └── KeyboardSupport (bool)
└── SettingsPersistence (Node)
    ├── ConfigFile (ConfigFile)
    └── SettingsLoaded (bool)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

signal accessibility_setting_changed(setting: String, value: Variant)
signal subtitle_displayed(speaker: String, text: String, duration: float)

enum ColorBlindType { NONE, PROTANOPIA, DEUTERANOPIA, TRITANOPIA }

var color_blind_type: ColorBlindType = ColorBlindType.NONE
var color_blind_intensity: float = 1.0
var subtitle_enabled: bool = true
var subtitle_font_size: int = 24
var subtitle_background_opacity: float = 0.7
var control_mappings: Dictionary = {
    "move_up": KEY_W,
    "move_down": KEY_S,
    "move_left": KEY_A,
    "move_right": KEY_D,
    "jump": KEY_SPACE,
    "attack": KEY_J,
    "interact": KEY_E
}
var original_mappings: Dictionary = {}

func _ready() -> void:
    original_mappings = control_mappings.duplicate()
    _load_settings()
    _apply_color_blind_filter()
    _setup_subtitle_display()

func set_color_blind_mode(type: ColorBlindType, intensity: float = 1.0) -> void:
    color_blind_type = type
    color_blind_intensity = clamp(intensity, 0.0, 2.0)
    _apply_color_blind_filter()
    accessibility_setting_changed.emit("color_blind_mode", type)

func _apply_color_blind_filter() -> void:
    var filter = $ColorBlindMode/ColorFilter
    match color_blind_type:
        ColorBlindType.NONE:
            filter.visible = false
        ColorBlindType.PROTANOPIA:
            filter.visible = true
            filter.color = Color(0.5, 1.0, 0.5, 0.3 * color_blind_intensity)
        ColorBlindType.DEUTERANOPIA:
            filter.visible = true
            filter.color = Color(1.0, 0.5, 0.5, 0.3 * color_blind_intensity)
        ColorBlindType.TRITANOPIA:
            filter.visible = true
            filter.color = Color(0.5, 0.5, 1.0, 0.3 * color_blind_intensity)

func set_subtitle(enabled: bool, font_size: int = 24, bg_opacity: float = 0.7) -> void:
    subtitle_enabled = enabled
    subtitle_font_size = font_size
    subtitle_background_opacity = clamp(bg_opacity, 0.0, 1.0)
    _setup_subtitle_display()
    accessibility_setting_changed.emit("subtitle", enabled)

func _setup_subtitle_display() -> void:
    var panel = $SubtitleSystem/SubtitlePanel
    var label = $SubtitleSystem/SubtitleLabel
    panel.visible = subtitle_enabled
    label.add_theme_font_size_override("normal_font_size", subtitle_font_size)
    panel.modulate.a = subtitle_background_opacity

func display_subtitle(speaker: String, text: String, duration: float = 3.0) -> void:
    if not subtitle_enabled:
        return
    var label = $SubtitleSystem/SubtitleLabel
    var speaker_label = $SubtitleSystem/SpeakerName
    speaker_label.text = speaker
    label.text = text
    $SubtitleSystem/SubtitlePanel.visible = true
    subtitle_displayed.emit(speaker, text, duration)
    await get_tree().create_timer(duration).timeout
    $SubtitleSystem/SubtitlePanel.visible = false

func remap_control(action: String, new_key: int) -> bool:
    if not control_mappings.has(action):
        return false
    for existing_action in control_mappings:
        if existing_action != action and control_mappings[existing_action] == new_key:
            return false
    control_mappings[action] = new_key
    _update_input_map(action, new_key)
    accessibility_setting_changed.emit("control_remapping", action)
    return true

func _update_input_map(action: String, key: int) -> void:
    InputMap.action_erase_events(action)
    var event = InputEventKey.new()
    event.keycode = key
    InputMap.add_action_event(action, event)

func reset_controls_to_default() -> void:
    control_mappings = original_mappings.duplicate()
    for action in control_mappings:
        _update_input_map(action, control_mappings[action])
    accessibility_setting_changed.emit("control_remapping", "reset")

func get_current_settings() -> Dictionary:
    return {
        "color_blind_mode": color_blind_type,
        "color_blind_intensity": color_blind_intensity,
        "subtitle_enabled": subtitle_enabled,
        "subtitle_font_size": subtitle_font_size,
        "control_mappings": control_mappings.duplicate()
    }

func save_settings() -> void:
    var config = ConfigFile.new()
    config.set_value("accessibility", "color_blind_type", color_blind_type)
    config.set_value("accessibility", "color_blind_intensity", color_blind_intensity)
    config.set_value("accessibility", "subtitle_enabled", subtitle_enabled)
    config.set_value("accessibility", "subtitle_font_size", subtitle_font_size)
    config.set_value("accessibility", "control_mappings", control_mappings)
    config.save("user://accessibility_settings.cfg")

func _load_settings() -> void:
    var config = ConfigFile.new()
    if config.load("user://accessibility_settings.cfg") == OK:
        color_blind_type = config.get_value("accessibility", "color_blind_type", ColorBlindType.NONE)
        color_blind_intensity = config.get_value("accessibility", "color_blind_intensity", 1.0)
        subtitle_enabled = config.get_value("accessibility", "subtitle_enabled", true)
        subtitle_font_size = config.get_value("accessibility", "subtitle_font_size", 24)
        var saved_mappings = config.get_value("accessibility", "control_mappings", {})
        for action in saved_mappings:
            if control_mappings.has(action):
                control_mappings[action] = saved_mappings[action]
                _update_input_map(action, saved_mappings[action])
```

## Kasus Penggunaan Nyata
**The Last of Us Part II** oleh Naughty Dog dianggap sebagai game paling accessible dalam sejarah gaming. Game ini menyediakan lebih dari 60 opsi aksesibilitas termasuk audio descriptions untuk tunanetra, haptic feedback untuk tunarungu, dan custom control layouts untuk pemain dengan keterbatasan motorik. Pendekatan ini membuktikan bahwa accessibility features tidak mengurangi kualitas game, malah justru memperluas jumlah pemain yang bisa menikmati pengalaman gaming.
