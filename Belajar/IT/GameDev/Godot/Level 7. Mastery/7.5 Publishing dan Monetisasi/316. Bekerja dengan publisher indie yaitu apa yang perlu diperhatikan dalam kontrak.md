# 316. Bekerja dengan publisher indie yaitu apa yang perlu diperhatikan dalam kontrak

## Definisi Singkat
Publisher indie adalah perusahaan yang membantu developer mempublikasikan, memasarkan, dan mendistribusikan game di berbagai platform. Kontrak dengan publisher mencakup hak-hak seperti hak distribusi, pembagian revenue (revenue share), hak kekayaan intelektual (IP rights), dan kewajiban masing-masing pihak.

## Analogi
Bekerja dengan publisher seperti menyewa agen properti untuk menjual rumahmu. Agen punya jaringan lebih luas dan tahu cara memasarkan rumah, tapi mereka mengambil komisi dari hasil penjualan. Kamu perlu pastikan kontraknya jelas: berapa komisi, siapa yang bayar biaya iklan, dan yang paling penting — tetap siapa yang punya rumah (hak IP).

## Kode GDScript (Godot 4)
```gdscript
extends Node

var contract_data: Dictionary = {
	"publisher_name": "",
	"developer_name": "",
	"game_title": "",
	"signing_date": "",
	"revenue_share_publisher": 30,
	"revenue_share_developer": 70,
	"platforms": [],
	"territory": "worldwide",
	"duration_months": 24,
	"ip_owner": "developer",
	"minimum_guarantee": 0,
	"marketing_budget": 0,
	"milestones": []
}

var revenue_data: Dictionary = {
	"total_sales": 0,
	"total_revenue": 0.0,
	"publisher_share": 0.0,
	"developer_share": 0.0,
	"platform_fees": 0.0
}

func setup_contract(data: Dictionary):
	for key in data:
		if key in contract_data:
			contract_data[key] = data[key]
	print("Contract setup for: " + contract_data["game_title"])

func calculate_revenue_split(gross_revenue: float) -> Dictionary:
	var platform_fee_rate: float = 0.30
	var platform_fee: float = gross_revenue * platform_fee_rate
	var net_revenue: float = gross_revenue - platform_fee
	var publisher_amount: float = net_revenue * (contract_data["revenue_share_publisher"] / 100.0)
	var developer_amount: float = net_revenue * (contract_data["revenue_share_developer"] / 100.0)
	return {
		"gross": gross_revenue,
		"platform_fee": platform_fee,
		"net_revenue": net_revenue,
		"publisher": publisher_amount,
		"developer": developer_amount
	}

func record_sale(platform: String, amount: float, units: int):
	revenue_data["total_sales"] += units
	revenue_data["total_revenue"] += amount
	var split = calculate_revenue_split(amount)
	revenue_data["publisher_share"] += split["publisher"]
	revenue_data["developer_share"] += split["developer"]
	revenue_data["platform_fees"] += split["platform_fee"]

func is_minimum_guarantee_met() -> bool:
	return revenue_data["developer_share"] >= contract_data["minimum_guarantee"]

func get_remaining_contract_duration() -> int:
	var signing = Time.get_datetime_dict_from_system_from_string(contract_data["signing_date"])
	var contract_end = Time.get_unix_time_from_dict(signing) + (contract_data["duration_months"] * 30 * 86400)
	var remaining = contract_end - Time.get_unix_time_from_system()
	return maxi(0, int(remaining / 86400))

func get_contract_summary() -> Dictionary:
	return {
		"game": contract_data["game_title"],
		"publisher": contract_data["publisher_name"],
		"revenue_split": str(contract_data["revenue_share_developer"]) + "/" + str(contract_data["revenue_share_publisher"]),
		"ip_owner": contract_data["ip_owner"],
		"duration_months": contract_data["duration_months"],
		"territory": contract_data["territory"]
	}

func generate_terminology_checklist() -> Array:
	return [
		"Revenue share percentage and calculation method",
		"Platform fees handling",
		"Marketing budget responsibility",
		"IP ownership and licensing terms",
		"Exclusivity clause and duration",
		"Termination conditions and notice period",
		"Minimum guarantee amount",
		"Reporting and payment schedule",
		" sequel and expansion rights",
		"Source code escrow provisions"
	]

func get_red_flags() -> Array:
	return [
		"Publisher owns IP permanently",
		"No termination clause",
		"Unlimited exclusivity period",
		"Developer pays all marketing costs",
		"No revenue reporting transparency",
		"Publisher can cancel without notice",
		"Revenue share below 60/40 for developer",
		"No milestone-based payment schedule"
	]
