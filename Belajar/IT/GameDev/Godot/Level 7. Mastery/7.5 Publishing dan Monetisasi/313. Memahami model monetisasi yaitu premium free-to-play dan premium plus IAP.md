# 313. Memahami model monetisasi yaitu premium free-to-play dan premium plus IAP

## Definisi Singkat
Premium adalah model di mana pemain membayar di muka untuk mengakses seluruh game. Free-to-play (F2P) memungkinkan pemain bermain gratis tetapi menghasilkan uang melalui iklan atau pembelian dalam aplikasi (IAP). Premium plus IAP menggabungkan harga awal dengan opsi pembelian tambahan seperti kosmetik atau expansion.

## Analogi
Premium seperti membeli buku novel penuh di toko buku, kamu bayar sekali dan dapat semua cerita. Free-to-play seperti membaca novel gratis di internet tapi ada iklan di setiap bab dan kamu bisa bayar untuk hapus iklan atau dapat bonus cerita sampingan. Premium plus IAP seperti beli buku novel lengkap tapi ada edisi spesial yang bisa kamu beli terpisah untuk dapat ilustrasi warna atau chapter bonus.

## Kode GDScript (Godot 4)
```gdscript
extends Node

enum MonetizationModel { PREMIUM, FREE_TO_PLAY, PREMIUM_PLUS_IAP }

var current_model: MonetizationModel = MonetizationModel.FREE_TO_PLAY
var player_currency: int = 0
var owned_products: Array[String] = []

var product_catalog: Dictionary = {
	"skin_warrior": {"name": "Warrior Skin", "price": 500, "type": "cosmetic"},
	"skin_mage": {"name": "Mage Skin", "price": 500, "type": "cosmetic"},
	"coin_pack_small": {"name": "1000 Coins", "price": 99, "type": "currency", "amount": 1000},
	"coin_pack_large": {"name": "5000 Coins", "price": 399, "type": "currency", "amount": 5000},
	"expansion_1": {"name": "DLC Chapter 1", "price": 1499, "type": "expansion"}
}

func is_game_free() -> bool:
	return current_model == MonetizationModel.FREE_TO_PLAY

func can_player_access_content(content_id: String) -> bool:
	if current_model == MonetizationModel.PREMIUM:
		return true
	if content_id in owned_products:
		return true
	var product = product_catalog.get(content_id)
	if product and product["type"] == "cosmetic":
		return false
	if current_model == MonetizationModel.PREMIUM_PLUS_IAP:
		return product == null
	return false

func purchase_product(product_id: String) -> bool:
	if product_id not in product_catalog:
		return false
	var product = product_catalog[product_id]
	if player_currency < product["price"]:
		return false
	player_currency -= product["price"]
	owned_products.append(product_id)
	if product["type"] == "currency":
		player_currency += product["amount"]
	print("Purchased: " + product["name"])
	return true

func get_available_products() -> Array:
	var available: Array = []
	for id in product_catalog:
		if id not in owned_products:
			available.append({"id": id, "data": product_catalog[id]})
	return available

func get_total_revenue_estimate(purchases: Array) -> int:
	var total: int = 0
	for purchase in purchases:
		if purchase in product_catalog:
			total += product_catalog[purchase]["price"]
	return total
