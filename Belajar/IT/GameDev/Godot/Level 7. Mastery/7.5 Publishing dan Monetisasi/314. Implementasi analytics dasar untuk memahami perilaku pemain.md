# 314. Implementasi analytics dasar untuk memahami perilaku pemain

## Definisi Singkat
Game analytics adalah proses pengumpulan dan analisis data perilaku pemain seperti waktu bermain, level yang diselesaikan, titik di mana pemain berhenti bermain, dan pola interaksi lainnya. Data ini membantu developer memahami apa yang berhasil dan apa yang perlu diperbaiki dalam game.

## Analogi
Analytics dalam game seperti kamera CCTV di toko retail. Kamu bisa melihat ke mana pelanggan pergi, di rak mana mereka berhenti paling lama, dan di mana mereka akhirnya meninggalkan toko tanpa membeli apa-apa. Dengan informasi ini, kamu bisa menata ulang toko agar lebih menarik dan meningkatkan penjualan.

## Struktur Node (Scene Tree)
```
Node (AnalyticsManager)
├── SessionTracker
├── EventQueue
└── DataExporter
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

var session_start_time: float = 0.0
var events: Array[Dictionary] = []
var session_id: String = ""
var player_data: Dictionary = {
	"total_playtime": 0.0,
	"levels_completed": [],
	"deaths": 0,
	"items_collected": [],
	"highest_level_reached": 0
}

func _ready():
	session_id = _generate_session_id()
	session_start_time = Time.get_unix_time_from_system()
	track_event("session_start", {})

func _generate_session_id() -> String:
	var time = Time.get_datetime_dict_from_system()
	return "%04d%02d%02d_%02d%02d%02d" % [
		time.year, time.month, time.day,
		time.hour, time.minute, time.second
	]

func track_event(event_name: String, data: Dictionary):
	var event = {
		"event": event_name,
		"timestamp": Time.get_unix_time_from_system(),
		"session_id": session_id,
		"data": data
	}
	events.append(event)
	print("Analytics Event: " + event_name)

func track_level_start(level_number: int):
	track_event("level_start", {"level": level_number})

func track_level_complete(level_number: int, time_taken: float):
	track_event("level_complete", {"level": level_number, "time": time_taken})
	if level_number not in player_data["levels_completed"]:
		player_data["levels_completed"].append(level_number)
	if level_number > player_data["highest_level_reached"]:
		player_data["highest_level_reached"] = level_number

func track_player_death(level_number: int, cause: String):
	track_event("player_death", {"level": level_number, "cause": cause})
	player_data["deaths"] += 1

func track_item_collected(item_id: String, level_number: int):
	track_event("item_collected", {"item": item_id, "level": level_number})
	if item_id not in player_data["items_collected"]:
		player_data["items_collected"].append(item_id)

func track_menu_interaction(menu_name: String, action: String):
	track_event("menu_interaction", {"menu": menu_name, "action": action})

func get_session_duration() -> float:
	return Time.get_unix_time_from_system() - session_start_time

func get_retention_data() -> Dictionary:
	return {
		"session_id": session_id,
		"duration": get_session_duration(),
		"events_count": events.size(),
		"levels_completed": player_data["levels_completed"].size(),
		"deaths": player_data["deaths"]
	}

func export_analytics_data() -> Dictionary:
	return {
		"sessions": events,
		"player_summary": player_data,
		"export_time": Time.get_datetime_string_from_system()
	}

func save_analytics_to_file():
	var data = export_analytics_data()
	var file = FileAccess.open("user://analytics_" + session_id + ".json", FileAccess.WRITE)
	if file:
		file.store_string(JSON.stringify(data, "\t"))
		file.close()
