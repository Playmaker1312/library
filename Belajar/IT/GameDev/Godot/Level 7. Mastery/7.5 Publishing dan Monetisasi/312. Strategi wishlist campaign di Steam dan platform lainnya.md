# 312. Strategi wishlist campaign di Steam dan platform lainnya

## Definisi Singkat
Wishlist di Steam adalah fitur yang memungkinkan pemain menyimpan game yang menarik di daftar keinginan mereka. Wishlist campaign adalah strategi terencana untuk meningkatkan jumlah wishlist sebelum dan sesudah game dirilis, karena jumlah wishlist berkorelasi langsung dengan penjualan di hari pertama.

## Analogi
Wishlist itu seperti antrian pre-order tiket konser. Semakin banyak orang yang mengantri (wishlist), semakin besar peluang venue penuh terjual habis di hari pertama. Strategi wishlist campaign itu seperti kampanye promosi yang kamu lakukan supaya orang-orang tahu konsermu dan mau mengantri lebih awal.

## Kode GDScript (Godot 4)
```gdscript
extends Node

const STEAM_APP_ID: int = 12345

var wishlist_count: int = 0
var wishlist_goal: int = 10000

func _ready():
	fetch_wishlist_count()

func fetch_wishlist_count():
	var http = HTTPRequest.new()
	add_child(http)
	http.request_completed.connect(_on_wishlist_fetched)
	var url = "https://store.steampowered.com/api/appdetails?appids=" + str(STEAM_APP_ID)
	http.request(url)

func _on_wishlist_fetched(result: int, response_code: int, headers: PackedStringArray, body: PackedByteArray):
	if response_code == 200:
		var json = JSON.new()
		var parse_result = json.parse(body.get_string_from_utf8())
		if parse_result == OK:
			print("App data fetched successfully")

func get_wishlist_percentage() -> float:
	if wishlist_goal == 0:
		return 0.0
	return float(wishlist_count) / float(wishlist_goal) * 100.0

func is_wishlist_goal_reached() -> bool:
	return wishlist_count >= wishlist_goal

func generate_wishlist_link(steam_app_id: int) -> String:
	return "https://store.steampowered.com/app/" + str(steam_app_id)

func generate_wishlist_cta() -> String:
	var percentage: float = get_wishlist_percentage()
	if percentage >= 100.0:
		return "Wishlist goal reached! Thank you!"
	elif percentage >= 75.0:
		return "Almost there! " + str(int(percentage)) + "% of wishlist goal!"
	elif percentage >= 50.0:
		return "Halfway to our goal! " + str(int(percentage)) + "%"
	else:
		return "Help us reach our wishlist goal! " + str(int(percentage)) + "% done"

func create_steam_news_post(title: String, body: String) -> Dictionary:
	return {
		"appid": STEAM_APP_ID,
		"title": title,
		"body": body,
		"language": "english",
		"visibility": 0
	}

func log_wishlist_event(source: String):
	print("Wishlist added from: " + source)
	wishlist_count += 1
