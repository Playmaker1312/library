# 309. Implementasi Choice Consequence System yang Mempengaruhi Akhir Cerita

## Definisi Singkat
Choice consequence system adalah mekanisme yang melacak pilihan pemain dan mengubah status game berdasarkan keputusan tersebut, yang pada akhirnya menentukan akhir cerita yang berbeda.

## Analogi
Bayangkan sebuah papan permainan ular tangga di mana setiap pilihan mengubah posisi Anda. Beberapa jalur mengarah ke kemenangan, yang lain ke kekalahan. Consequence system adalah papan yang mencatat semua langkah Anda.

## Struktur Node (Scene Tree)
```
GameManager (Node)
├── StoryState (Node)
│   ├── Variables (Dictionary)
│   └── History (Array)
├── QuestTracker (Node)
├── RelationshipSystem (Node)
└── EndingEvaluator (Node)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

signal variable_changed(key: String, value: Variant)
signal quest_completed(quest_id: String)
signal relationship_changed(character: String, value: int)
signal ending_reached(ending_id: String)

var variables: Dictionary = {}
var history: Array[Dictionary] = []
var quests: Dictionary = {}
var relationships: Dictionary = {}

func set_variable(key: String, value: Variant) -> void:
    variables[key] = value
    variable_changed.emit(key, value)
    _record_history("set_variable", {"key": key, "value": value})

func get_variable(key: String, default: Variant = null) -> Variant:
    return variables.get(key, default)

func add_to_variable(key: String, amount: int) -> void:
    if variables.has(key):
        variables[key] += amount
    else:
        variables[key] = amount
    variable_changed.emit(key, variables[key])
    _record_history("add_variable", {"key": key, "amount": amount})

func modify_relationship(character: String, amount: int) -> void:
    if not relationships.has(character):
        relationships[character] = 0
    relationships[character] += amount
    relationships[character] = clampi(relationships[character], -100, 100)
    relationship_changed.emit(character, relationships[character])
    _record_history("modify_relationship", {"character": character, "amount": amount})

func get_relationship(character: String) -> int:
    return relationships.get(character, 0)

func start_quest(quest_id: String, data: Dictionary = {}) -> void:
    quests[quest_id] = {
        "status": "active",
        "objectives": data.get("objectives", []),
        "completed_objectives": []
    }

func complete_quest(quest_id: String) -> void:
    if quests.has(quest_id):
        quests[quest_id]["status"] = "completed"
        quest_completed.emit(quest_id)
        _record_history("complete_quest", {"quest_id": quest_id})

func complete_objective(quest_id: String, objective_index: int) -> void:
    if quests.has(quest_id):
        var quest = quests[quest_id]
        if objective_index < quest["objectives"].size():
            quest["completed_objectives"].append(objective_index)
            if quest["completed_objectives"].size() >= quest["objectives"].size():
                complete_quest(quest_id)

func check_ending_conditions(endings: Array[Dictionary]) -> String:
    for ending in endings:
        if _check_conditions(ending.get("conditions", {})):
            ending_reached.emit(ending["id"])
            return ending["id"]
    return "default"

func _check_conditions(conditions: Dictionary) -> bool:
    for key in conditions:
        match key:
            "variable":
                for var_name in conditions["variable"]:
                    if not variables.has(var_name):
                        return false
                    if variables[var_name] != conditions["variable"][var_name]:
                        return false
            "relationship":
                for char_name in conditions["relationship"]:
                    var required = conditions["relationship"][char_name]
                    var current = get_relationship(char_name)
                    if current < required:
                        return false
            "quest":
                for quest_id in conditions["quest"]:
                    if not quests.has(quest_id):
                        return false
                    if quests[quest_id]["status"] != "completed":
                        return false
            "history":
                for event in conditions["history"]:
                    if not _check_history_event(event):
                        return false
    return true

func _check_history_event(event: Dictionary) -> bool:
    for record in history:
        if record["action"] == event.get("action", ""):
            if event.has("data"):
                var match_data = true
                for key in event["data"]:
                    if not record["data"].has(key) or record["data"][key] != event["data"][key]:
                        match_data = false
                        break
                if match_data:
                    return true
    return false

func _record_history(action: String, data: Dictionary) -> void:
    history.append({
        "action": action,
        "data": data,
        "timestamp": Time.get_unix_time_from_system()
    })

func get_ending_data() -> Dictionary:
    return {
        "variables": variables.duplicate(),
        "relationships": relationships.duplicate(),
        "completed_quests": _get_completed_quests(),
        "history_size": history.size()
    }

func _get_completed_quests() -> Array[String]:
    var completed: Array[String] = []
    for quest_id in quests:
        if quests[quest_id]["status"] == "completed":
            completed.append(quest_id)
    return completed

func reset() -> void:
    variables.clear()
    history.clear()
    quests.clear()
    relationships.clear()
```

## Kasus Penggunaan Nyata
Choice consequence system digunakan dalam game seperti "The Witcher 3" di mana pilihan dialog mempengaruhi nasib karakter dan "Mass Effect" di mana keputusan pemain menentukan akhir trilogi. Sistem ini memungkinkan game memiliki multiple endings yang terasa bermakna karena didasarkan pada akumulasi pilihan pemain sepanjang permainan.
