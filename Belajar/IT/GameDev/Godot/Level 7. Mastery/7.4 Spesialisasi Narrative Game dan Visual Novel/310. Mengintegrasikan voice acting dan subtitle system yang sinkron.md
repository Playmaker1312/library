# 310. Mengintegrasikan Voice Acting dan Subtitle System yang Sinkron

## Definisi Singkat
Voice acting dan subtitle system adalah mekanisme untuk menampilkan dialog dengan suara aktor dan teks subtitle yang sinkron. Sistem ini meningkatkan imersi dan aksesibilitas game narrative.

## Analogi
Bayangkan menonton film asing dengan subtitle. Voice acting adalah suara aktor, subtitle adalah terjemahan di layar, dan sinkronisasi memastikan teks muncul tepat saat kata-kata diucapkan.

## Struktur Node (Scene Tree)
```
VoiceSubtitleSystem (Node)
├── AudioStreamPlayer (AudioStreamPlayer)
├── SubtitleContainer (VBoxContainer)
│   ├── SpeakerLabel (Label)
│   ├── SubtitleLabel (RichTextLabel)
│   └── ProgressBar (ProgressBar)
├── Timer (Timer)
└── CaptionParser (Node)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node

signal voice_started()
signal voice_finished()
signal subtitle_updated(text: String, timestamp: float)
signal caption_hit(caption_data: Dictionary)

@onready var audio_player: AudioStreamPlayer = %AudioStreamPlayer
@onready var speaker_label: Label = %SpeakerLabel
@onready var subtitle_label: RichTextLabel = %SubtitleLabel
@onready var progress_bar: ProgressBar = %ProgressBar
@onready var timer: Timer = %Timer

var captions: Array[Dictionary] = []
var current_caption_index: int = 0
var is_playing: bool = false
var caption_data: Dictionary = {}

func _ready() -> void:
    audio_player.finished.connect(_on_audio_finished)
    timer.timeout.connect(_on_timer_timeout)
    timer.wait_time = 0.05

func play_voice_line(voice_path: String, caption_file: String, speaker: String = "") -> void:
    if not ResourceLoader.exists(voice_path):
        push_error("Voice file not found: " + voice_path)
        return
    
    audio_player.stream = load(voice_path)
    audio_player.play()
    
    speaker_label.text = speaker
    captions = _load_captions(caption_file)
    current_caption_index = 0
    is_playing = true
    
    timer.start()
    voice_started.emit()

func _load_captions(file_path: String) -> Array[Dictionary]:
    if not ResourceLoader.exists(file_path):
        return []
    
    var file = FileAccess.open(file_path, FileAccess.READ)
    if file == null:
        return []
    
    var json = JSON.new()
    var result = json.parse(file.get_as_text())
    file.close()
    
    if result != OK:
        return []
    
    var data = json.data
    if data is Array:
        return data
    return []

func _on_timer_timeout() -> void:
    if not is_playing:
        return
    
    var current_time = audio_player.get_playback_position()
    progress_bar.value = current_time / audio_player.stream.get_length() * 100
    
    while current_caption_index < captions.size():
        var caption = captions[current_caption_index]
        if current_time >= caption.get("start", 0):
            if current_time <= caption.get("end", 0):
                _display_caption(caption)
                caption_hit.emit(caption)
                break
            else:
                current_caption_index += 1
        else:
            break

func _display_caption(caption: Dictionary) -> void:
    var text = caption.get("text", "")
    subtitle_label.text = text
    subtitle_updated.emit(text, caption.get("start", 0))

func stop_voice() -> void:
    audio_player.stop()
    timer.stop()
    is_playing = false
    subtitle_label.text = ""
    progress_bar.value = 0

func _on_audio_finished() -> void:
    timer.stop()
    is_playing = false
    subtitle_label.text = ""
    progress_bar.value = 100
    voice_finished.emit()

func set_caption_data(data: Dictionary) -> void:
    caption_data = data

func get_caption_at_time(time: float) -> Dictionary:
    for caption in captions:
        if time >= caption.get("start", 0) and time <= caption.get("end", 0):
            return caption
    return {}

func format_time(seconds: float) -> String:
    var mins = int(seconds) / 60
    var secs = int(seconds) % 60
    return "%02d:%02d" % [mins, secs]

func create_caption_file(audio_path: String, output_path: String) -> void:
    var captions_template: Array[Dictionary] = [
        {"start": 0.0, "end": 2.0, "text": "Contoh subtitle pertama"},
        {"start": 2.5, "end": 4.5, "text": "Contoh subtitle kedua"}
    ]
    
    var file = FileAccess.open(output_path, FileAccess.WRITE)
    if file != null:
        file.store_string(JSON.stringify(captions_template, "\t"))
        file.close()

func get_audio_length() -> float:
    if audio_player.stream != null:
        return audio_player.stream.get_length()
    return 0.0

func seek_to(time: float) -> void:
    if audio_player.stream != null:
        audio_player.seek(time)
        current_caption_index = 0
        for i in range(captions.size()):
            if captions[i].get("start", 0) >= time:
                current_caption_index = i
                break
```

## Kasus Penggunaan Nyata
Voice acting dan subtitle system digunakan dalam game seperti "Final Fantasy VII Remake" dan "The Last of Us Part II" yang memiliki ribuan baris dialog dengan kualitas sinematik. Sistem ini penting untuk aksesibilitas karena membantu pemain dengan gangguan pendengaran dan memungkinkan bermain dalam bahasa yang berbeda melalui subtitle.
