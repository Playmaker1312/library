# 308. Membuat Cutscene System Menggunakan AnimationPlayer dan Sinematografi Dasar

## Definisi Singkat
Cutscene system adalah kerangka kerja untuk membuat adegan sinematik dalam game yang menggabungkan animasi, audio, dan interaksi pemain. AnimationPlayer di Godot adalah alat utama untuk membuat transisi visual yang halus dan efek sinematik.

## Analogi
Cutscene seperti film mini di dalam game. AnimationPlayer adalah kamera dan alat editing yang mengatur pergerakan, zoom, dan efek visual untuk menciptakan pengalaman sinematik.

## Struktur Node (Scene Tree)
```
CutsceneSystem (Node2D)
├── Camera2D (Camera2D)
├── AnimationPlayer (AnimationPlayer)
├── DialogueSystem (Node)
│   ├── DialogueBox (PanelContainer)
│   └── NameLabel (Label)
├── EffectsLayer (CanvasLayer)
│   ├── FadeRect (ColorRect)
│   └── VignetteRect (ColorRect)
└── AudioPlayers (Node)
    ├── MusicPlayer (AudioStreamPlayer)
    └── SFXPlayer (AudioStreamPlayer)
```

## Kode GDScript (Godot 4)
```gdscript
extends Node2D

signal cutscene_started()
signal cutscene_ended()
signal cutscene_step_completed(step: int)

@onready var camera: Camera2D = %Camera2D
@onready var animation_player: AnimationPlayer = %AnimationPlayer
@onready var dialogue_box: PanelContainer = %DialogueBox
@onready var name_label: Label = %NameLabel
@onready var dialogue_label: RichTextLabel = %DialogueLabel
@onready var fade_rect: ColorRect = %FadeRect
@onready var music_player: AudioStreamPlayer = %MusicPlayer
@onready var sfx_player: AudioStreamPlayer = %SFXPlayer

var cutscene_data: Array[Dictionary] = []
var current_step: int = 0
var is_playing: bool = false

func play_cutscene(data: Array[Dictionary]) -> void:
    cutscene_data = data
    current_step = 0
    is_playing = true
    cutscene_started.emit()
    _process_step()

func _process_step() -> void:
    if current_step >= cutscene_data.size():
        _end_cutscene()
        return
    
    var step = cutscene_data[current_step]
    var step_type = step.get("type", "")
    
    match step_type:
        "camera_move":
            _animate_camera(step)
        "fade":
            _animate_fade(step)
        "dialogue":
            _show_dialogue(step)
        "animation":
            _play_animation(step)
        "wait":
            await get_tree().create_timer(step.get("duration", 1.0)).timeout
            _advance_step()
        "sfx":
            _play_sfx(step)
        "music":
            _play_music(step)

func _animate_camera(step: Dictionary) -> void:
    var target_pos = step.get("position", Vector2.ZERO)
    var duration = step.get("duration", 1.0)
    var zoom = step.get("zoom", Vector2.ONE)
    
    var tween = create_tween()
    tween.set_parallel(true)
    tween.tween_property(camera, "position", target_pos, duration)
    tween.tween_property(camera, "zoom", zoom, duration)
    tween.chain().tween_callback(_advance_step)

func _animate_fade(step: Dictionary) -> void:
    var target_alpha = step.get("alpha", 1.0)
    var duration = step.get("duration", 1.0)
    var color = fade_rect.color
    
    var tween = create_tween()
    color.a = 1.0 - target_alpha
    tween.tween_property(fade_rect, "color", color, duration)
    tween.tween_callback(_advance_step)

func _show_dialogue(step: Dictionary) -> void:
    name_label.text = step.get("speaker", "")
    dialogue_label.text = step.get("text", "")
    dialogue_box.visible = true
    
    dialogue_label.visible_characters = 0
    var tween = create_tween()
    var text_length = dialogue_label.get_total_character_count()
    tween.tween_property(dialogue_label, "visible_characters", text_length, text_length * 0.03)
    tween.tween_callback(_on_dialogue_finished)

func _on_dialogue_finished() -> void:
    if not dialogue_box.visible:
        return
    
    dialogue_box.visible = false
    _advance_step()

func _play_animation(step: Dictionary) -> void:
    var anim_name = step.get("animation", "")
    if animation_player.has_animation(anim_name):
        animation_player.play(anim_name)
        await animation_player.animation_finished
    _advance_step()

func _play_sfx(step: Dictionary) -> void:
    var sfx_path = step.get("path", "")
    if sfx_path != "" and ResourceLoader.exists(sfx_path):
        sfx_player.stream = load(sfx_path)
        sfx_player.play()
    _advance_step()

func _play_music(step: Dictionary) -> void:
    var music_path = step.get("path", "")
    var fade_time = step.get("fade_time", 1.0)
    
    if music_path != "" and ResourceLoader.exists(music_path):
        var tween = create_tween()
        tween.tween_property(music_player, "volume_db", -40, fade_time)
        await tween.finished
        
        music_player.stream = load(music_path)
        music_player.play()
        
        tween = create_tween()
        tween.tween_property(music_player, "volume_db", 0, fade_time)
    _advance_step()

func _advance_step() -> void:
    cutscene_step_completed.emit(current_step)
    current_step += 1
    _process_step()

func _end_cutscene() -> void:
    is_playing = false
    cutscene_ended.emit()

func skip_cutscene() -> void:
    animation_player.stop()
    _end_cutscene()
```

## Kasus Penggunaan Nyata
Cutscene system digunakan dalam game seperti "Celeste" untuk adegan naratif dan "Hollow Knight" untuk memperkenalkan boss. AnimationPlayer memungkinkan pembuatan transisi layar, zoom kamera dramatis, dan efek visual yang memberikan pengalaman sinematik tanpa memerlukan external video files.
