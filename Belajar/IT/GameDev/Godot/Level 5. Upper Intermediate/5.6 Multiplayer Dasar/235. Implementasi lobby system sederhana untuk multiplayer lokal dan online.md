# 235. Implementasi Lobby System Sederhana untuk Multiplayer Lokal dan Online

## Definisi Singkat
Lobby system adalah mekanisme yang memungkinkan pemain berkumpul dalam satu ruangan virtual sebelum memulai permainan. Sistem ini menangani pembuatan room, join, matchmaking, dan manajemen pemain sebelum game dimulai.

## Analogi
Bayangkan sebuah lobi hotel sebelum masuk ke ruang pertemuan. Di lobi, resepsionis (server) mencatat siapa saja yang datang, mengatur ruangan mana yang mereka masuki, dan memberikan informasi tentang acara yang akan dimulai. Pemain bisa membuat ruangan baru (create room), masuk ke ruangan yang sudah ada (join room), atau menunggu di lobi umum (matchmaking).

## Struktur Node (Scene Tree)

```
Lobby (Control)

```

## Kode GDScript (Godot 4)

```gdscript
extends Control

@onready var player_list: ItemList = %PlayerList
@onready var room_list: ItemList = %RoomList
@onready var chat_box: TextEdit = %ChatBox
@onready var message_input: LineEdit = %MessageInput
@onready var create_room_btn: Button = %CreateRoomBtn
@onready var join_room_btn: Button = %JoinRoomBtn
@onready var start_game_btn: Button = %StartGameBtn
@onready var disconnect_btn: Button = %DisconnectBtn

var peer: ENetMultiplayerPeer
var rooms: Dictionary = {}
var players_in_lobby: Dictionary = {}
var current_room: String = ""

func _ready() -> void:
	create_room_btn.pressed.connect(_on_create_room_pressed)
	join_room_btn.pressed.connect(_on_join_room_pressed)
	start_game_btn.pressed.connect(_on_start_game_pressed)
	disconnect_btn.pressed.connect(_on_disconnect_pressed)
	message_input.text_submitted.connect(_on_message_submitted)
	start_game_btn.visible = false

func host_game(port: int = 7777) -> void:
	peer = ENetMultiplayerPeer.new()
	peer.create_server(port, 10)
	multiplayer.peer = peer
	multiplayer.peer_connected.connect(_on_player_connected)
	multiplayer.peer_disconnected.connect(_on_player_disconnected)
	_add_player(multiplayer.get_unique_id(), "Host")
	refresh_ui()

func join_game(address: String, port: int = 7777) -> void:
	peer = ENetMultiplayerPeer.new()
	peer.create_client(address, port)
	multiplayer.peer = peer
	multiplayer.connected_to_server.connect(_on_connected_to_server)

func _on_player_connected(id: int) -> void:
	_add_player(id, "Player_" + str(id))
	rpc("sync_lobby_state", players_in_lobby, rooms)

func _on_player_disconnected(id: int) -> void:
	players_in_lobby.erase(id)
	refresh_ui()

func _on_connected_to_server() -> void:
	_add_player(multiplayer.get_unique_id(), "Player_" + str(multiplayer.get_unique_id()))
	refresh_ui()

func _add_player(id: int, player_name: String) -> void:
	players_in_lobby[id] = player_name

@rpc("any_peer", "call_local", "reliable")
func create_room(room_name: String) -> void:
	rooms[room_name] = {"host": multiplayer.get_remote_sender_id(), "players": []}
	current_room = room_name
	refresh_ui()

@rpc("any_peer", "call_local", "reliable")
func join_room(room_name: String) -> void:
	if rooms.has(room_name):
		rooms[room_name]["players"].append(multiplayer.get_remote_sender_id())
		current_room = room_name
		refresh_ui()

@rpc("authority", "call_local", "reliable")
func sync_lobby_state(players: Dictionary, available_rooms: Dictionary) -> void:
	players_in_lobby = players
	rooms = available_rooms
	refresh_ui()

func refresh_ui() -> void:
	player_list.clear()
	for id in players_in_lobby:
		player_list.add_item(players_in_lobby[id] + " (ID: " + str(id) + ")")
	room_list.clear()
	for room_name in rooms:
		var player_count = rooms[room_name]["players"].size()
		room_list.add_item(room_name + " (" + str(player_count) + " players)")
	start_game_btn.visible = multiplayer.is_server()

func _on_create_room_pressed() -> void:
	var room_name = "Room_" + str(randi())
	rpc("create_room", room_name)

func _on_join_room_pressed() -> void:
	var selected = room_list.get_selected_items()
	if selected.size() > 0:
		var room_name = room_list.get_item_text(selected[0]).split(" (")[0]
		rpc("join_room", room_name)

func _on_start_game_pressed() -> void:
	rpc("start_game")

@rpc("authority", "call_local", "reliable")
func start_game() -> void:
	get_tree().change_scene_to_file("res://scenes/game.tscn")

func _on_disconnect_pressed() -> void:
	multiplayer.peer = null
	peer = null
	get_tree().change_scene_to_file("res://scenes/main_menu.tscn")

func _on_message_submitted(message: String) -> void:
	if message.strip_edges() != "":
		rpc("send_chat_message", message)
		message_input.clear()

@rpc("any_peer", "call_local", "reliable")
func send_chat_message(message: String) -> void:
	var sender = players_in_lobby.get(multiplayer.get_remote_sender_id(), "Unknown")
	chat_box.append_text(sender + ": " + message + "\n")
```

## Kasus Penggunaan Nyata
Lobby system digunakan di hampir semua game multiplayer seperti Valorant (lobby untuk ranked match), Minecraft (server list), dan Roblox (game lobby). Pada game mobile seperti Mobile Legends, lobby berfungsi sebagai tempat pemain berkumpul sebelum matchmaking dan menunggu pemain lain bergabung.
