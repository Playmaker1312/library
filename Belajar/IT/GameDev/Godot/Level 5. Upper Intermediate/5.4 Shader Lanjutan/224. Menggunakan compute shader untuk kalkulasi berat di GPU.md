# 224. Menggunakan compute shader untuk kalkulasi berat di GPU

## Definisi Singkat
Compute shader adalah program yang berjalan di GPU untuk melakukan kalkulasi paralel seperti fisika, partikel, atau kalkulasi berat kompleks tanpa mempengaruhi CPU.

## Analogi
Bayangkan sebuah pabrik dengan ribuan pekerja (GPU cores) yang masing-masing menghitung angka secara bersamaan, jauh lebih cepat daripada satu orang (CPU) yang menghitung satu per satu.

## Struktur Node (Scene Tree)
```
Node3D (Compute Shader Controller)
├── MeshInstance3D (untuk visualisasi hasil)
├── ComputeShader (Resource)
└── RenderingServer (akses ke rendering pipeline)
```

## Kode GDScript (Godot 4)

### Compute Shader Code (save as .glsl):
```glsl
#version 450

// INPUT DATA
layout(local_size_x = 256) in;

// BUFFER
layout(std430, binding = 0) readonly buffer InputBuffer {
    float input_data[];
};

layout(std430, binding = 1) writeonly buffer OutputBuffer {
    float output_data[];
};

layout(std430, binding = 2) readonly buffer WeightBuffer {
    float weights[];
};

// UNIFORMS
uniform int data_size;
uniform float time;

void main() {
    uint idx = gl_GlobalInvocationID.x;
    
    if (idx >= data_size) return;
    
    // KALKULASI BERAT DENGAN FORMULA
    float input_val = input_data[idx];
    float weight = weights[idx];
    
    // KOMBINASI INPUT DENGAN WEIGHT
    float result = input_val * weight;
    
    // TAMBAH EFFECT WAVE
    result += sin(time + float(idx) * 0.1) * 0.1;
    
    output_data[idx] = result;
}
```

### GDScript Controller:
```gdscript
extends Node3D

@export var data_size: int = 1024
@export var compute_shader_path: String = "res://compute_shader.glsl"

var rd: RenderingDevice
var shader_rid: RID
var input_buffer: RID
var output_buffer: RID
var weight_buffer: RID
var uniform_set: RID
var pipeline: RID

var input_data: PackedFloat32Array
var weight_data: PackedFloat32Array
var output_data: PackedFloat32Array

func _ready():
    input_data.resize(data_size)
    weight_data.resize(data_size)
    output_data.resize(data_size)
    
    for i in data_size:
        input_data[i] = randf_range(0.0, 1.0)
        weight_data[i] = randf_range(0.5, 2.0)
    
    init_compute_shader()
    execute_compute()

func init_compute_shader():
    rd = RenderingServer.get_rendering_device()
    
    var shader_file = load(compute_shader_path)
    var shader_spirv = shader_file.get_spirv()
    shader_rid = rd.shader_create_from_spirv(shader_spirv)
    
    input_buffer = rd.storage_buffer_create(
        data_size * 4,
        input_data.to_byte_array()
    )
    
    weight_buffer = rd.storage_buffer_create(
        data_size * 4,
        weight_data.to_byte_array()
    )
    
    output_buffer = rd.storage_buffer_create(data_size * 4)
    
    var uniforms = []
    uniforms.append(RID())
    uniforms[0] = rd.uniform_buffer_create(0, 4)
    
    var input_uni = RDUniform.new()
    input_uni.uniform_type = RenderingDevice.UNIFORM_TYPE_STORAGE_BUFFER
    input_uni.binding = 0
    input_uni.add_id(input_buffer)
    
    var weight_uni = RDUniform.new()
    weight_uni.uniform_type = RenderingDevice.UNIFORM_TYPE_STORAGE_BUFFER
    weight_uni.binding = 2
    weight_uni.add_id(weight_buffer)
    
    var output_uni = RDUniform.new()
    output_uni.uniform_type = RenderingDevice.UNIFORM_TYPE_STORAGE_BUFFER
    output_uni.binding = 1
    output_uni.add_id(output_buffer)
    
    var uniforms_array = [input_uni, output_uni, weight_uni]
    uniform_set = rd.uniform_set_create(uniforms_array, shader_rid, 0)
    
    pipeline = rd.compute_pipeline_create(shader_rid)

func execute_compute():
    var compute_list = rd.compute_list_begin()
    rd.compute_list_bind_compute_pipeline(compute_list, pipeline)
    rd.compute_list_bind_uniform_set(compute_list, uniform_set, 0)
    rd.compute_list_dispatch(compute_list, ceili(data_size / 256.0), 1, 1)
    rd.compute_list_end()
    
    rd.submit()
    rd.sync()
    
    var output_bytes = rd.buffer_get_data(output_buffer)
    output_data = output_bytes.to_float32_array()

func get_output() -> PackedFloat32Array:
    return output_data

func update_input(new_input: PackedFloat32Array):
    input_data = new_input
    rd.buffer_update(input_buffer, 0, input_data.to_byte_array())
    execute_compute()

func _process(delta):
    var time_data = PackedFloat32Array([Time.get_ticks_msec() / 1000.0])
    rd.buffer_update(weight_buffer, 0, time_data.to_byte_array())
```

## Kasus Penggunaan Nyata
Compute shader digunakan untuk kalkulasi fisika massal, simulasi partikel ribuan, atau machine learning inference. Dalam konteks game, bisa digunakan untuk kalkulasi weight neural network, simulasi air, atau crowd simulation dengan performa tinggi.
