# 225. Membuat post-processing effect menggunakan SubViewport dan shader

## Definisi Singkat
Post-processing effect adalah efek visual yang diterapkan pada seluruh layar setelah rendering selesai, menggunakan SubViewport untuk menangkap adegan dan shader untuk memanipulasi hasilnya.

## Analogi
Bayangkan seperti menggunakan filter di aplikasi foto: setelah foto diambil (rendering), filter (shader) diterapkan untuk mengubah tampilan keseluruhan seperti efek blur, warna, atau distorsi.

## Struktur Node (Scene Tree)
```
CanvasLayer (untuk post-processing)
└── SubViewportContainer
    └── SubViewport
        ├── Camera3D
        └── Adegan 3D
    └── ColorRect (dengan post-process shader)
```

## Kode GDScript (Godot 4)

### Post-Process Shader Code (save as .gdshader):
```glsl
shader_type canvas_item;

uniform sampler2D screen_texture : hint_screen_texture, filter_nearest;
uniform sampler2D depth_texture : hint_depth_texture, filter_nearest;

uniform bool enable_bloom = true;
uniform float bloom_threshold: hint_range(0.0, 1.0) = 0.8;
uniform float bloom_intensity: hint_range(0.0, 2.0) = 0.5;

uniform bool enable_vignette = true;
uniform float vignette_intensity: hint_range(0.0, 1.0) = 0.4;
uniform float vignette_radius: hint_range(0.0, 2.0) = 0.8;

uniform bool enable_chromatic_aberration = true;
uniform float chromatic_strength: hint_range(0.0, 10.0) = 2.0;

uniform float time;

vec3 bloom_effect(vec2 uv) {
    vec3 color = texture(screen_texture, uv).rgb;
    float brightness = dot(color, vec3(0.2126, 0.7152, 0.0722));
    
    if (brightness > bloom_threshold) {
        vec3 bloom = color * bloom_intensity;
        return bloom;
    }
    return vec3(0.0);
}

float vignette_effect(vec2 uv) {
    vec2 center = uv - 0.5;
    float dist = length(center);
    return smoothstep(vignette_radius, vignette_radius - vignette_intensity, dist);
}

vec3 chromatic_aberration(vec2 uv) {
    vec2 offset = (uv - 0.5) * chromatic_strength;
    float r = texture(screen_texture, uv + offset * 0.01).r;
    float g = texture(screen_texture, uv).g;
    float b = texture(screen_texture, uv - offset * 0.01).b;
    return vec3(r, g, b);
}

void fragment() {
    vec2 uv = SCREEN_UV;
    vec3 color = texture(screen_texture, uv).rgb;
    
    if (enable_chromatic_aberration) {
        color = chromatic_aberration(uv);
    }
    
    if (enable_bloom) {
        color += bloom_effect(uv);
    }
    
    if (enable_vignette) {
        color *= vignette_effect(uv);
    }
    
    // SCANLINE EFFECT
    float scanline = sin(uv.y * 800.0 + time * 10.0) * 0.02;
    color -= scanline;
    
    ALBEDO = color;
    ALPHA = 1.0;
}
```

### GDScript Controller:
```gdscript
extends ColorRect

@export var sub_viewport: SubViewport
@export var enable_bloom: bool = true
@export var enable_vignette: bool = true
@export var enable_chromatic: bool = true

var post_process_mat: ShaderMaterial

func _ready():
    post_process_mat = ShaderMaterial.new()
    post_process_mat.shader = load("res://post_process.gdshader")
    post_process_mat.set_shader_parameter("enable_bloom", enable_bloom)
    post_process_mat.set_shader_parameter("enable_vignette", enable_vignette)
    post_process_mat.set_shader_parameter("enable_chromatic_aberration", enable_chromatic)
    
    material = post_process_mat
    
    if sub_viewport:
        sub_viewport.render_target_update_mode = SubViewport.UPDATE_ALWAYS

func _process(delta):
    post_process_mat.set_shader_parameter("time", Time.get_ticks_msec() / 1000.0)

func toggle_bloom():
    enable_bloom = !enable_bloom
    post_process_mat.set_shader_parameter("enable_bloom", enable_bloom)

func toggle_vignette():
    enable_vignette = !enable_vignette
    post_process_mat.set_shader_parameter("enable_vignette", enable_vignette)

func toggle_chromatic():
    enable_chromatic = !enable_chromatic
    post_process_mat.set_shader_parameter("enable_chromatic_aberration", enable_chromatic)

func set_bloom_threshold(value: float):
    post_process_mat.set_shader_parameter("bloom_threshold", value)

func set_vignette_intensity(value: float):
    post_process_mat.set_shader_parameter("vignette_intensity", value)

func set_chromatic_strength(value: float):
    post_process_mat.set_shader_parameter("chromatic_strength", value)
```

## Kasus Penggunaan Nyata
Post-processing digunakan untuk efek visual game seperti bloom untuk cahaya terang, vignette untuk fokus perhatian, chromatic aberration untuk efek kamera, atau scanline untuk estetika retro. SubViewport memungkinkan rendering adegan ke texture yang kemudian dimanipulasi oleh shader.
