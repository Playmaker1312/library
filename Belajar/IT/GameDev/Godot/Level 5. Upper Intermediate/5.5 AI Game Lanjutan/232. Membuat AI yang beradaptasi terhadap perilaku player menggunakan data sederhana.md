# 232. Membuat AI yang beradaptasi terhadap perilaku player menggunakan data sederhana

## Definisi Singkat
Adaptive AI adalah sistem AI yang mengumpulkan data dari perilaku player dalam waktu nyata dan mengubah strategi berdasarkan pola yang terdeteksi. Pendekatan sederhana menggunakan statistik dasar seperti frekuensi serangan, pola gerakan, dan waktu reaksi untuk menyesuaikan kesulitan dan taktik AI.

## Analogi
Bayangkan kamu bermain catur dengan teman yang suka mengorbankan bidak di awal. Setelah beberapa ronde, kamu belajar pola ini dan mulai berhati-hati saat dia mengorbankan bidak. Adaptive AI bekerja seperti otak manusia yang belajar dari pengalaman - mengamati, mencatat pola, dan menyesuaikan strategi.

## Kode GDScript (Godot 4)
```gdscript
extends CharacterBody3D

@export var adaptation_rate: float = 0.1
@export var tracking_window: float = 30.0

var player_data: Dictionary = {
    "attack_frequency": 0.0,
    "dodge_rate": 0.0,
    "preferred_distance": 5.0,
    "reaction_time": 0.5,
    "total_encounters": 0
}

var current_strategy: String = "balanced"
var encounter_timer: float = 0.0
var last_player_attack_time: float = 0.0
var player_attack_count: int = 0
var player_dodge_count: int = 0
var player_total_actions: int = 0

func _ready() -> void:
    load_player_profile()

func _process(delta: float) -> void:
    encounter_timer += delta
    if encounter_timer >= tracking_window:
        analyze_and_adapt()
        encounter_timer = 0.0

func record_player_attack() -> void:
    var time_since_last: float = Time.get_ticks_msec() / 1000.0 - last_player_attack_time
    player_data["reaction_time"] = lerp(
        player_data["reaction_time"],
        time_since_last,
        adaptation_rate
    )
    last_player_attack_time = Time.get_ticks_msec() / 1000.0
    player_attack_count += 1
    player_total_actions += 1

func record_player_dodge() -> void:
    player_dodge_count += 1
    player_total_actions += 1

func record_player_distance(distance: float) -> void:
    player_data["preferred_distance"] = lerp(
        player_data["preferred_distance"],
        distance,
        adaptation_rate
    )

func analyze_and_adapt() -> void:
    if player_total_actions == 0:
        return
    
    player_data["attack_frequency"] = float(player_attack_count) / tracking_window
    player_data["dodge_rate"] = float(player_dodge_count) / player_total_actions
    player_data["total_encounters"] += 1
    
    determine_strategy()
    save_player_profile()
    reset_counters()

func determine_strategy() -> void:
    if player_data["attack_frequency"] > 0.5:
        current_strategy = "defensive"
    elif player_data["dodge_rate"] > 0.4:
        current_strategy = "predictive"
    elif player_data["preferred_distance"] < 3.0:
        current_strategy = "ranged"
    else:
        current_strategy = "balanced"

func get_current_strategy() -> String:
    return current_strategy

func get_adapted_behavior() -> Dictionary:
    match current_strategy:
        "defensive":
            return {
                "aggression": 0.3,
                "block_chance": 0.7,
                "counter_attack": true,
                "distance_preference": 2.0
            }
        "predictive":
            return {
                "aggression": 0.5,
                "anticipate_dodge": true,
                "feint_attack": true,
                "distance_preference": player_data["preferred_distance"]
            }
        "ranged":
            return {
                "aggression": 0.4,
                "keep_distance": 8.0,
                "use_projectile": true,
                "dodge频繁": true
            }
        _:
            return {
                "aggression": 0.5,
                "block_chance": 0.3,
                "distance_preference": 4.0
            }

func reset_counters() -> void:
    player_attack_count = 0
    player_dodge_count = 0
    player_total_actions = 0

func save_player_profile() -> void:
    var config = ConfigFile.new()
    for key in player_data:
        config.set_value("player_profile", key, player_data[key])
    config.save("user://player_profile.cfg")

func load_player_profile() -> void:
    var config = ConfigFile.new()
    if config.load("user://player_profile.cfg") == OK:
        for key in player_data:
            if config.has_section_key("player_profile", key):
                player_data[key] = config.get_value("player_profile", key)
```

## Kasus Penggunaan Nyata
Adaptive AI digunakan dalam game fighting seperti Tekken dan Street Fighter untuk menyesuaikan kesulitan AI berdasarkan skill player. Di game seperti Resident Evil 4, sistem Dynamic Difficulty Adjustment (DDA) menyesuaikan jumlah musuh dan drop item berdasarkan performa player, menciptakan pengalaman yang challenging tapi tidak frustasi.
